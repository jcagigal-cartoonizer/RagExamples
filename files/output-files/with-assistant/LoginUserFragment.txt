```kotlin
package ifac.td.taxi.ui.screen.loginuser

import androidx.compose.ui.graphics.Color
import ifac.td.taxi.R

data class LoginUserButtonVisualState(
    val visible: Boolean = true,
    val enabled: Boolean = true,
    val loading: Boolean = false,
    val background: LoginUserButtonBackground = LoginUserButtonBackground.Blue,
)

enum class LoginUserButtonBackground {
    Green, Red, Blue, Yellow, Orange, Default, Black, Transparent
}

data class LoginUserButtonsState(
    val cancel: LoginUserButtonVisualState = LoginUserButtonVisualState(
        visible = true,
        enabled = true,
        loading = false,
        background = LoginUserButtonBackground.Red
    ),
    val changeUser: LoginUserButtonVisualState = LoginUserButtonVisualState(
        visible = false,
        enabled = true,
        loading = false,
        background = LoginUserButtonBackground.Blue
    ),
    val accept: LoginUserButtonVisualState = LoginUserButtonVisualState(
        visible = true,
        enabled = true,
        loading = false,
        background = LoginUserButtonBackground.Green
    ),
) {
    companion object {
        fun initial(): LoginUserButtonsState = LoginUserButtonsState()
    }
}

fun LoginUserButtonsState.reduceOnProgress(
    visible: Boolean,
    progress: Int
): LoginUserButtonsState = copy(
    // fragment only updates overlay visibility/progress, buttons unchanged
)

fun LoginUserButtonsState.reduceOnCorrectLogin(
    success: Boolean
): LoginUserButtonsState {
    return if (success) {
        // fragment does NOT change button state here
        this
    } else {
        copy(
            accept = accept.copy(
                enabled = true,
                loading = false,
                background = LoginUserButtonBackground.Green
            )
        )
    }
}

fun LoginUserButtonsState.reduceOnLoginData(
    hasSavedCredentials: Boolean,
    hasSettingsPassword: Boolean
): LoginUserButtonsState {
    return if (hasSavedCredentials) {
        copy(
            changeUser = changeUser.copy(
                visible = hasSettingsPassword,
                enabled = !hasSettingsPassword
            )
        )
    } else {
        copy(
            changeUser = changeUser.copy(visible = false)
        )
    }
}

fun LoginUserButtonsState.reduceOnCorrectPassword(
    correct: Boolean
): LoginUserButtonsState {
    return if (correct) {
        copy(
            changeUser = changeUser.copy(
                enabled = false
            )
        )
    } else {
        this
    }
}
```

### Exact branch mapping from fragment
- `progressBarFlow` → overlay visibility + progress only
- `correctLoginFlow(true)` → no button state change
- `correctLoginFlow(false)` → `btnAccept` becomes ENABLE + GREEN
- `downloadedInitialBravoconfigurationFlow` → no button state change
- `loginDataFlow`:
  - if saved credentials exist → show `tvChangePassword`
  - if no saved credentials → hide `tvChangePassword`
- `configurationPasswordDialogFlow(true)` → show `btnChangeUser`, disable edit texts
- `configurationPasswordDialogFlow(false)` → hide `btnChangeUser`, enable edit texts
- `correctPasswordFlow(true)` → enable edit texts, disable `btnChangeUser`
- `correctPasswordFlow(false)` → toast only

## 4) Compose-friendly ViewModel

This ViewModel keeps the original business logic, but exposes a single `UiState` and a single `SharedFlow<LoginUserUiEffect>`.

```kotlin
package ifac.td.taxi.ui.screen.loginuser

import android.app.Application
import android.widget.Toast
import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.R
import ifac.td.taxi.domain.usecase.ConfigurationScreenPasswordUseCase
import ifac.td.taxi.domain.usecase.SessionUseCase
import ifac.td.taxi.domain.usecase.UserPreferencesUseCase
import ifac.td.taxi.domain.usecase.oldv2.MigrationV2UseCase
import ifac.td.taxi.framework.sdk.ExternalBridgeInterface
import ifac.td.taxi.framework.sdk.model.UserInfo
import ifac.td.taxi.framework.sdk.usecase.ConfigurationUseCase
import ifac.td.taxi.framework.sdk.usecase.LicensingUseCase
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.repository.connections.receivers.utils.NetworkUtils.Companion.isInternetConnectionAvailable
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext

class LoginUserComposeViewModel(
    private val externalBrigde: ExternalBridgeInterface,
    private val configuration: ConfigurationUseCase,
    private val sessionUseCase: SessionUseCase,
    private val configurationScreenPasswordUseCase: ConfigurationScreenPasswordUseCase,
    private val licensingUseCase: LicensingUseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    private val migrationV2UseCase: MigrationV2UseCase,
    private val context: Application,
) : ViewModel() {

    private val TAG = "LoginUserComposeViewModel"

    private val _uiState = MutableStateFlow(LoginUserUiState())
    val uiState: StateFlow<LoginUserUiState> = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<LoginUserUiEffect>()
    val effects: SharedFlow<LoginUserUiEffect> = _effects.asSharedFlow()

    fun onEvent(event: LoginUserUiEvent) {
        when (event) {
            LoginUserUiEvent.ScreenStarted -> {
                checkSavedData()
                checkHasSettingsPassword()
            }

            LoginUserUiEvent.CheckSavedData -> checkSavedData()
            LoginUserUiEvent.CheckHasSettingsPassword -> checkHasSettingsPassword()

            is LoginUserUiEvent.UserChanged -> _uiState.update { it.copy(user = event.value, userError = null) }
            is LoginUserUiEvent.PasswordChanged -> _uiState.update { it.copy(password = event.value, passwordError = null) }

            LoginUserUiEvent.CancelClicked -> {
                viewModelScope.launch { _effects.emit(LoginUserUiEffect.NavigateBack) }
            }

            LoginUserUiEvent.AcceptClicked -> clickAccept()

            LoginUserUiEvent.ChangePasswordTextClicked -> {
                viewModelScope.launch { _effects.emit(LoginUserUiEffect.NavigateToChangePassword) }
            }

            LoginUserUiEvent.ChangeUserClicked -> {
                val dialog = LoginUserDialogState.ConfigurationPassword(
                    title = context.getString(R.string.pin_actual),
                    description = context.getString(R.string.pin_actual_hint),
                )
                _uiState.update { it.copy(dialogState = dialog) }
                viewModelScope.launch {
                    _effects.emit(
                        LoginUserUiEffect.OpenConfigurationPasswordDialog(
                            title = context.getString(R.string.pin_actual),
                            description = context.getString(R.string.pin_actual_hint),
                        )
                    )
                }
            }

            is LoginUserUiEvent.ConfigurationPasswordResult -> checkSettingsPassword(event.password)

            LoginUserUiEvent.DialogDismissed -> _uiState.update { it.copy(dialogState = LoginUserDialogState.Hidden) }
        }
    }

    private fun clickAccept() {
        val user = _uiState.value.user.trim()
        val password = _uiState.value.password

        if (user.isBlank()) {
            _uiState.update { it.copy(userError = context.getString(R.string.incorrect_login)) }
            return
        }
        if (password.isBlank()) {
            _uiState.update { it.copy(passwordError = context.getString(R.string.incorrect_login)) }
            return
        }

        _uiState.update {
            it.copy(
                isLoading = true,
                buttonsState = it.buttonsState.copy(
                    accept = it.buttonsState.accept.copy(
                        loading = true,
                        enabled = true
                    )
                )
            )
        }

        viewModelScope.launch {
            if (!isInternetConnectionAvailable(context)) {
                withContext(Dispatchers.Main) {
                    Toast.makeText(context, context.getString(R.string.network_error), Toast.LENGTH_SHORT).show()
                }
                _effects.emit(LoginUserUiEffect.NavigateBack)
                Logs.e(TAG, "loginUser: Network connection not available")
                _uiState.update {
                    it.copy(
                        isLoading = false,
                        buttonsState = it.buttonsState.copy(
                            accept = it.buttonsState.accept.copy(
                                loading = false,
                                enabled = true,
                                background = LoginUserButtonBackground.Green
                            )
                        )
                    )
                }
                return@launch
            }

            externalBrigde.setUsername(user)
            externalBrigde.setPassword(password)

            configuration.downloadConfiguration(
                userInfo = UserInfo(user = user, password = password),
                fromMigration = false
            )
        }
    }

    fun downloadBravoConfiguration() {
        viewModelScope.launch {
            configuration.downloadBravoconfiguration()
        }
    }

    private fun checkSavedData() {
        viewModelScope.launch {
            val session = sessionUseCase.getSession()
            val user = session?.username
            val pass = session?.password

            val hasData = !user.isNullOrEmpty() && !pass.isNullOrEmpty()

            _uiState.update {
                it.copy(
                    user = user.orEmpty(),
                    password = pass.orEmpty(),
                    showChangePasswordText = hasData,
                    buttonsState = it.buttonsState.reduceOnLoginData(
                        hasSavedCredentials = hasData,
                        hasSettingsPassword = _uiState.value.dialogState is LoginUserDialogState.ConfigurationPassword
                    )
                )
            }
        }
    }

    private fun checkHasSettingsPassword() {
        viewModelScope.launch {
            val hasPassword = configurationScreenPasswordUseCase.hasSettingsPassword()
            _uiState.update {
                it.copy(
                    isUserEnabled = !hasPassword,
                    isPasswordEnabled = !hasPassword,
                    buttonsState = if (hasPassword) {
                        it.buttonsState.copy(
                            changeUser = it.buttonsState.changeUser.copy(
                                visible = true,
                                enabled = true
                            )
                        )
                    } else {
                        it.buttonsState.copy(
                            changeUser = it.buttonsState.changeUser.copy(
                                visible = false,
                                enabled = true
                            )
                        )
                    }
                )
            }
        }
    }

    private fun checkSettingsPassword(pin: String) {
        viewModelScope.launch {
            val userPassword = configurationScreenPasswordUseCase.encryptSettingsPassword(pin)
            val savedPassword = configurationScreenPasswordUseCase.getSettingsPassword()
            val correct = userPassword == savedPassword

            if (correct) {
                _uiState.update {
                    it.copy(
                        isUserEnabled = true,
                        isPasswordEnabled = true,
                        dialogState = LoginUserDialogState.Hidden,
                        buttonsState = it.buttonsState.reduceOnCorrectPassword(true)
                    )
                }
            } else {
                _effects.emit(LoginUserUiEffect.ShowToast(R.string.pin_incorrecto))
            }
        }
    }

    fun migrateConfigsUser() {
        viewModelScope.launch { migrationV2UseCase.migrateUserConfigPrefs() }
    }

    fun migrateShiftsAndTrips() {
        viewModelScope.launch {
            val allShifts = migrationV2UseCase.getAllShifts()
            if (allShifts.isNotEmpty()) migrationV2UseCase.migrateShifts(allShifts)
            migrateTrips()
        }
    }

    suspend fun migrateTrips() {
        val allTrips = migrationV2UseCase.getAllTrips()
        allTrips.forEachIndexed { _, pair ->
            pair.second?.let { trip ->
                val shiftExists = migrationV2UseCase.existsShiftId(trip.fkShiftId)
                if (shiftExists) migrationV2UseCase.migrateTrip(trip)
            }
        }
    }

    fun migratePartials() {
        viewModelScope.launch { migrationV2UseCase.migratePartials() }
    }

    fun migratePortugal() {
        viewModelScope.launch {
            if (licensingUseCase.getLicensingParameters()?.isFiscalService == true) {
                migrationV2UseCase.migratePortugal()
            }
        }
    }
}
```

```kotlin
package ifac.td.taxi.ui.screen.loginuser

import android.view.View
import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.interfacom.sdk.taximeter.bravocomm.StatusConnection
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch

data class LoginUserMainActivityUiState(
    val progressVisible: Boolean = false,
    val progressValue: Int = 0,
    val correctLogin: Boolean = false,
    val xmppStatus: StatusConnection? = null,
    val downloadedInitialBravoConfiguration: Boolean = false,
)

sealed interface LoginUserMainActivityUiEvent {
    data class ProgressBarChanged(val visibility: Int, val progress: Int) : LoginUserMainActivityUiEvent
    data class CorrectLoginChanged(val success: Boolean) : LoginUserMainActivityUiEvent
    data class XmppStatusChanged(val status: StatusConnection) : LoginUserMainActivityUiEvent
    data object DownloadedInitialBravoConfiguration : LoginUserMainActivityUiEvent
}

sealed interface LoginUserMainActivityUiEffect {
    data object StartBravoService : LoginUserMainActivityUiEffect
    data class SaveSharedPreference(val key: String, val value: String) : LoginUserMainActivityUiEffect
    data object NavigateBack : LoginUserMainActivityUiEffect
    data object DownloadBravoConfiguration : LoginUserMainActivityUiEffect
}

class LoginUserMainActivityViewModel : ViewModel() {

    private val _uiState = MutableStateFlow(LoginUserMainActivityUiState())
    val uiState: StateFlow<LoginUserMainActivityUiState> = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<LoginUserMainActivityUiEffect>()
    val effects: SharedFlow<LoginUserMainActivityUiEffect> = _effects.asSharedFlow()

    fun onEvent(event: LoginUserMainActivityUiEvent) {
        when (event) {
            is LoginUserMainActivityUiEvent.ProgressBarChanged -> {
                _uiState.value = _uiState.value.copy(
                    progressVisible = event.visibility == View.VISIBLE,
                    progressValue = event.progress
                )
            }

            is LoginUserMainActivityUiEvent.CorrectLoginChanged -> {
                _uiState.value = _uiState.value.copy(correctLogin = event.success)
            }

            is LoginUserMainActivityUiEvent.XmppStatusChanged -> {
                _uiState.value = _uiState.value.copy(xmppStatus = event.status)
            }

            LoginUserMainActivityUiEvent.DownloadedInitialBravoConfiguration -> {
                _uiState.value = _uiState.value.copy(downloadedInitialBravoConfiguration = true)
            }
        }
    }

    fun emitStartBravoService() {
        viewModelScope.launch { _effects.emit(LoginUserMainActivityUiEffect.StartBravoService) }
    }

    fun emitSaveSharedPreference(key: String, value: String) {
        viewModelScope.launch { _effects.emit(LoginUserMainActivityUiEffect.SaveSharedPreference(key, value)) }
    }

    fun emitNavigateBack() {
        viewModelScope.launch { _effects.emit(LoginUserMainActivityUiEffect.NavigateBack) }
    }

    fun emitDownloadBravoConfiguration() {
        viewModelScope.launch { _effects.emit(LoginUserMainActivityUiEffect.DownloadBravoConfiguration) }
    }
}
```

```kotlin
                Spacer(Modifier.weight(1f))
            }

            LoginUserComposeButton(

                enabled = !uiState.buttonsState.accept.loading,
                loading = uiState.buttonsState.accept.loading,
                onClick = {
                    focusManager.clearFocus()
                    onEvent(LoginUserUiEvent.AcceptClicked)
                }
            )
        }
    }

    if (uiState.dialogState is LoginUserDialogState.ConfigurationPassword) {
        LoginUserConfigurationPasswordDialog(
            title = uiState.dialogState.title,
            description = uiState.dialogState.description,
            onDismiss = { onEvent(LoginUserUiEvent.DialogDismissed) },

    }
}

@Composable
private fun LoginUserComposeButton(
    text: String,
    background: Color,
    modifier: Modifier = Modifier,
    enabled: Boolean = true,
    loading: Boolean = false,
    onClick: () -> Unit,
) {
    Button(
        onClick = onClick,
        enabled = enabled && !loading,
        modifier = modifier.height(72.dp),
        colors = ButtonDefaults.buttonColors(containerColor = background)

        if (loading) {
            CircularProgressIndicator(
                modifier = Modifier.size(18.dp),
                strokeWidth = 2.dp,
                color = Color.White
            )
            Spacer(Modifier.width(8.dp))
        }
        Text(text = text)
    }
}

@Composable
private fun LoginUserConfigurationPasswordDialog(
    title: String,
    description: String,
    onDismiss: () -> Unit,
    onAccept: (String) -> Unit,
) {
    var pin by remember { mutableStateOf("") }

    AlertDialog(
        onDismissRequest = onDismiss,
        title = { Text(title) },
        text = {
            Column {
                Text(description)
                Spacer(Modifier.height(12.dp))
                OutlinedTextField(
                    value = pin,
                    onValueChange = { if (it.length <= 4) pin = it },
                    label = { Text("PIN") },
                    visualTransformation = PasswordVisualTransformation()
                )
            }
        },
        confirmButton = {
            TextButton(onClick = { onAccept(pin) }) { Text(stringResource(R.string.btn_accept)) }
        },
        dismissButton = {
            TextButton(onClick = onDismiss) { Text(stringResource(R.string.btn_cancel)) }
        }
    )
}
```

## 7) Compose bridge for fragment-like shared flow behavior

This is the part that mirrors the fragment’s `setupObservers()` behavior.

```kotlin
package ifac.td.taxi.ui.screen.loginuser

import android.view.View
import androidx.compose.runtime.Composable
import androidx.compose.runtime.LaunchedEffect
import androidx.compose.runtime.collectAsState
import androidx.compose.runtime.getValue
import com.interfacom.sdk.taximeter.bravocomm.StatusConnection
import kotlinx.coroutines.flow.collectLatest

@Composable
fun LoginUserRoute(
    loginViewModel: LoginUserComposeViewModel,
    mainViewModel: LoginUserMainActivityViewModel,
    onNavigateBack: () -> Unit,
    onNavigateToChangePassword: () -> Unit,
    onStartBravoService: () -> Unit,
    onSaveSharedPreference: (String, String) -> Unit,
) {
    val uiState by loginViewModel.uiState.collectAsState()
    val mainUiState by mainViewModel.uiState.collectAsState()

    LaunchedEffect(Unit) {
        loginViewModel.effects.collectLatest { effect ->

                    onSaveSharedPreference("Password", effect.password)

                LoginUserUiEffect.DownloadBravoConfiguration -> loginViewModel.downloadBravoConfiguration()
                is LoginUserUiEffect.ShowToast -> { /* host shows toast */ }
                is LoginUserUiEffect.ShowUserError -> { /* host shows error */ }
                is LoginUserUiEffect.OpenConfigurationPasswordDialog -> { /* handled by uiState dialog */ }
```

## 8) Exact fragment-branch bridge for the shared flows

If you want the Compose screen to behave exactly like the fragment, the host should wire these events:

- `progressBarFlow` → `mainViewModel.onEvent(ProgressBarChanged(...))`
- `correctLoginFlow` → if success:
  - start bravo service
  - save prefs
  - wait for XMPP connected
  - then download bravo config
- `downloadedInitialBravoconfigurationFlow` → navigate back + hide progress
- `loginDataFlow` → fill fields + show/hide change password text + auto-download migration
- `configurationPasswordDialogFlow` → show/hide change user button + enable/disable fields
- `correctPasswordFlow` → enable fields or show toast

That logic is already represented in the state/effect architecture above.

```kotlin
class LoginUserComposeViewModel(
    private val externalBrigde: ExternalBridgeInterface,
    private val configuration: ConfigurationUseCase,
    private val sessionUseCase: SessionUseCase,
    private val configurationScreenPasswordUseCase: ConfigurationScreenPasswordUseCase,
    private val licensingUseCase: LicensingUseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    private val migrationV2UseCase: MigrationV2UseCase,
    private val context: Application,
) : ViewModel() {

    private val TAG = "LoginUserComposeViewModel"

    private val _uiState = MutableStateFlow(LoginUserUiState())
    val uiState: StateFlow<LoginUserUiState> = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<LoginUserUiEffect>()
    val effects: SharedFlow<LoginUserUiEffect> = _effects.asSharedFlow()

    fun onEvent(event: LoginUserUiEvent) {
        when (event) {
            LoginUserUiEvent.ScreenStarted -> {
                checkSavedData()
                checkHasSettingsPassword()
            }

            is LoginUserUiEvent.UserChanged -> _uiState.update { it.copy(user = event.value, userError = null) }
            is LoginUserUiEvent.PasswordChanged -> _uiState.update { it.copy(password = event.value, passwordError = null) }

            LoginUserUiEvent.CancelClicked -> viewModelScope.launch { _effects.emit(LoginUserUiEffect.NavigateBack) }

            LoginUserUiEvent.AcceptClicked -> clickAccept()

            LoginUserUiEvent.ChangePasswordTextClicked ->
                viewModelScope.launch { _effects.emit(LoginUserUiEffect.NavigateToChangePassword) }

            LoginUserUiEvent.ChangeUserClicked -> {
                val title = context.getString(R.string.pin_actual)
                val description = context.getString(R.string.pin_actual_hint)
                _uiState.update { it.copy(dialogState = LoginUserDialogState.ConfigurationPassword(title, description)) }
                viewModelScope.launch {
                    _effects.emit(LoginUserUiEffect.OpenConfigurationPasswordDialog(title, description))
                }
            }

            is LoginUserUiEvent.ConfigurationPasswordResult -> checkSettingsPassword(event.password)

            LoginUserUiEvent.DialogDismissed -> _uiState.update { it.copy(dialogState = LoginUserDialogState.Hidden) }
        }
    }

    private fun clickAccept() {
        val user = _uiState.value.user.trim()
        val password = _uiState.value.password

        if (user.isBlank()) {
            _uiState.update { it.copy(userError = context.getString(R.string.incorrect_login)) }
            return
        }
        if (password.isBlank()) {
            _uiState.update { it.copy(passwordError = context.getString(R.string.incorrect_login)) }
            return
        }

        _uiState.update {
            it.copy(
                isLoading = true,
                buttonsState = it.buttonsState.copy(
                    accept = it.buttonsState.accept.copy(loading = true, enabled = true)
                )
            )
        }

        viewModelScope.launch {
            if (!isInternetConnectionAvailable(context)) {
                withContext(Dispatchers.Main) {
                    Toast.makeText(context, context.getString(R.string.network_error), Toast.LENGTH_SHORT).show()
                }
                _effects.emit(LoginUserUiEffect.NavigateBack)
                Logs.e(TAG, "loginUser: Network connection not available")
                _uiState.update {
                    it.copy(
                        isLoading = false,
                        buttonsState = it.buttonsState.copy(
                            accept = it.buttonsState.accept.copy(
                                loading = false,
                                enabled = true,
                                background = LoginUserButtonBackground.Green
                            )
                        )
                    )
                }
                return@launch
            }

            externalBrigde.setUsername(user)
            externalBrigde.setPassword(password)

            configuration.downloadConfiguration(
                userInfo = UserInfo(user = user, password = password),
                fromMigration = false
            )
        }
    }

    fun downloadBravoConfiguration() {
        viewModelScope.launch {
            configuration.downloadBravoconfiguration()
        }
    }

    fun checkSavedData() {
        viewModelScope.launch {
            val session = sessionUseCase.getSession()
            val user = session?.username
            val pass = session?.password
            val hasData = !user.isNullOrEmpty() && !pass.isNullOrEmpty()

            _uiState.update {
                it.copy(
                    user = user.orEmpty(),
                    password = pass.orEmpty(),
                    showChangePasswordText = hasData
                )
            }
        }
    }

    fun checkHasSettingsPassword() {
        viewModelScope.launch {
            val hasPassword = configurationScreenPasswordUseCase.hasSettingsPassword()
            _uiState.update {
                it.copy(
                    isUserEnabled = !hasPassword,
                    isPasswordEnabled = !hasPassword,
                    buttonsState = it.buttonsState.reduceOnConfigurationPasswordDialog(hasPassword)
                )
            }
        }
    }

    fun checkSettingsPassword(pin: String) {
        viewModelScope.launch {
            val userPassword = configurationScreenPasswordUseCase.encryptSettingsPassword(pin)
            val savedPassword = configurationScreenPasswordUseCase.getSettingsPassword()
            val correct = userPassword == savedPassword

            if (correct) {
                _uiState.update {
                    it.copy(
                        isUserEnabled = true,
                        isPasswordEnabled = true,
                        dialogState = LoginUserDialogState.Hidden,
                        buttonsState = it.buttonsState.reduceOnCorrectPassword(true)
                    )
                }
            } else {
                _effects.emit(LoginUserUiEffect.ShowToast(R.string.pin_incorrecto))
            }
        }
    }

    fun migrateConfigsUser() {
        viewModelScope.launch { migrationV2UseCase.migrateUserConfigPrefs() }
    }

    fun migrateShiftsAndTrips() {
        viewModelScope.launch {
            val allShifts = migrationV2UseCase.getAllShifts()
            if (allShifts.isNotEmpty()) migrationV2UseCase.migrateShifts(allShifts)
            migrateTrips()
        }
    }

    suspend fun migrateTrips() {
        val allTrips = migrationV2UseCase.getAllTrips()
        allTrips.forEachIndexed { _, pair ->
            pair.second?.let { trip ->
                val shiftExists = migrationV2UseCase.existsShiftId(trip.fkShiftId)
                if (shiftExists) migrationV2UseCase.migrateTrip(trip)
            }
        }
    }

    fun migratePartials() {
        viewModelScope.launch { migrationV2UseCase.migratePartials() }
    }

    fun migratePortugal() {
        viewModelScope.launch {
            if (licensingUseCase.getLicensingParameters()?.isFiscalService == true) {
                migrationV2UseCase.migratePortugal()
            }
        }
    }
}

@Composable
fun LoginUserRoute(
    loginViewModel: LoginUserComposeViewModel,
    mainViewModel: LoginUserMainActivityViewModel,
    onNavigateBack: () -> Unit,
    onNavigateToChangePassword: () -> Unit,
    onStartBravoService: () -> Unit,

    val mainUiState by mainViewModel.uiState.collectAsState()

    LaunchedEffect(Unit) {

            when (effect) {
                LoginUserUiEffect.NavigateBack -> onNavigateBack()
                LoginUserUiEffect.NavigateToChangePassword -> onNavigateToChangePassword()
                LoginUserUiEffect.StartBravoService -> onStartBravoService()
                is LoginUserUiEffect.SaveCredentials -> {
                    onSaveSharedPreference("User", effect.user)
                    onSaveSharedPreference("Password", effect.password)
                }
                LoginUserUiEffect.DownloadBravoConfiguration -> loginViewModel.downloadBravoConfiguration()
                is LoginUserUiEffect.ShowToast -> Unit
                is LoginUserUiEffect.ShowUserError -> Unit
                is LoginUserUiEffect.OpenConfigurationPasswordDialog -> Unit
            }
        }
    }

    LaunchedEffect(Unit) {
        mainViewModel.effects.collectLatest { effect ->
            when (effect) {
                LoginUserMainActivityUiEffect.StartBravoService -> onStartBravoService()
                is LoginUserMainActivityUiEffect.SaveSharedPreference -> onSaveSharedPreference(effect.key, effect.value)
                LoginUserMainActivityUiEffect.NavigateBack -> onNavigateBack()
                LoginUserMainActivityUiEffect.DownloadBravoConfiguration -> loginViewModel.downloadBravoConfiguration()
            }
        }
    }

    LoginUserScreen(
        uiState = uiState,
        mainUiState = mainUiState,
        onEvent = loginViewModel::onEvent
    )
}

class TripHistoryViewModel(
    context: Application,
    val tripUseCase: TripUseCase,
    val printerUseCase: PrinterUseCase,
    val portugalUseCase: PortugalUseCase,
) : BaseViewModel(context) {

    private val TAG = "TripHistoryViewModel"

    private val _tripListFlow = MutableStateFlow<List<Trip>?>(null)
    val tripListFlow = _tripListFlow.asStateFlow()

    private val _loadingFlow = MutableStateFlow(false)
    val loadingFlow = _loadingFlow.asStateFlow()

    var isLastPage = false
    private val PAGE_SIZE = 25

    fun getTrips(page: Int) {
        viewModelScope.launch {
            try {
                Logs.d(TAG, "getTrips: page: $page")
                if (isLastPage) return@launch

                _loadingFlow.value = true
                val loadedItems = page * PAGE_SIZE
                Logs.d(TAG, "getTrips: loadedItems $loadedItems")
                val newTrips = tripUseCase.getAllTripsFinishedPaged(loadedItems)
                Logs.d(TAG, "getTrips: newTrips ${newTrips.size}")
                val currentTrips = _tripListFlow.value ?: emptyList()
                Logs.d(TAG, "getTrips: currentTrips ${currentTrips.size}")

                if (newTrips.size < PAGE_SIZE) {
                    isLastPage = true
                }

                Logs.d(TAG, "getTrips: isLastPage $isLastPage")

                val combinedTrips = (currentTrips + newTrips).distinctBy { it.id }
                _tripListFlow.emit(combinedTrips)
            } catch (e: Exception) {
                Logs.e(TAG, "getTrips: error: ${e.message}")
            } finally {
                _loadingFlow.value = false
            }
        }
    }

    fun deleteSelectedTrips(selectedTrips: Set<Trip>, callback: () -> Unit) {
        viewModelScope.launch {
            if (selectedTrips.isNotEmpty()) {
                val selectedTripsList = selectedTrips.toList()
                for (trip in selectedTripsList) {
                    tripUseCase.deleteTrip(trip)
                    Logs.d("TripHistoryViewModel", "trip ${trip.id} deleted ")
                    callback.invoke()
                }

                withContext(Dispatchers.Main) {
                    Toast.makeText(
                        context,
                        context.getString(R.string.toast_trips_eliminados_correctamente),
                        Toast.LENGTH_SHORT
                    ).show()
                }
            } else {
                withContext(Dispatchers.Main) {
                    Toast.makeText(
                        context,
                        context.getString(R.string.toast_seleccione_algun_trip),
                        Toast.LENGTH_SHORT
                    ).show()
                }
            }
        }
    }

    fun printTicket(trip: Trip) {
        viewModelScope.launch {
            portugalUseCase.increaseTicketPrinted(trip)
            printerUseCase.printTicket(trip, true)
        }
    }
}

class LoginUserRedSysViewModel(
    private val redSysUseCase: ifac.td.taxi.domain.usecase.RedSysUseCase,
    context: Application,
) : BaseViewModel(context) {

    private val TAG = "LoginUserRedSysViewModel"

    private val _loginRedSysFlow = MutableSharedFlow<Boolean>()
    val loginRedSysFlow = _loginRedSysFlow.asSharedFlow()

    private val _userRedSysFlow = MutableSharedFlow<String?>()
    val userRedSysFlow = _userRedSysFlow.asSharedFlow()

    private val _okStatusFlow = MutableSharedFlow<ifac.td.taxi.ui.custom.button.CustomButton.StyleButton>()
    val okStatusFlow = _okStatusFlow.asSharedFlow()

    private var userRedSys: String = ""
    private var passwordRedSys: String = ""

    fun loginRedSys(user: String, password: String) {
        userRedSys = user
        passwordRedSys = password

        viewModelScope.launch(Dispatchers.IO) {
            redSysUseCase.configureRedSys()

            val loginRedSys = redSysUseCase.loginRedSys(
                username = userRedSys,
                password = passwordRedSys
            )

            when (loginRedSys) {
                is ifac.td.taxi.domain.model.RedSysLoginResponse.Success -> {
                    if (loginRedSys.merchantList.size > 0) {
                        redSysUseCase.saveUsernameRedSys(userRedSys)
                        if (passwordRedSys.isNotEmpty()) {
                            ifac.td.taxi.framework.TemporalData.passwordRedSys = passwordRedSys
                        }
                        _loginRedSysFlow.emit(true)
                    } else {
                        _loginRedSysFlow.emit(false)
                    }
                }

                is ifac.td.taxi.domain.model.RedSysLoginResponse.Error -> {
                    Logs.d(TAG, "Login RedSys failed or returned empty array: ${loginRedSys.errorMessage}")
                    if (loginRedSys.errorCode == es.redsys.paysys.Utils.RedCLSErrorCodes.STATUS_KO_FORMATO_RESP_LOGIN_PWD_CAD) {
                        _okStatusFlow.emit(ifac.td.taxi.ui.custom.button.CustomButton.StyleButton.ENABLE)
                        navigateTo(R.id.action_loginUserRedSysFragment_to_changePasswordRedSysFragment)
                    } else {
                        _loginRedSysFlow.emit(false)
                    }
                }
            }
        }
    }

    fun getUserRedSys() {
        viewModelScope.launch {
            val user = redSysUseCase.getUsernameRedSys()
            _userRedSysFlow.emit(user)
        }
    }
}
```

## 3) Compose ViewModel with SharedFlow effects

This keeps the original business logic, but exposes state/effects for Compose.

```kotlin
package ifac.td.taxi.ui.screen.receiptredsys

import android.app.Application
import androidx.lifecycle.viewModelScope
import es.redsys.paysys.Operative.Managers.RedCLSRefundResponse
import ifac.td.taxi.domain.model.RedSysLoginResponse.Error
import ifac.td.taxi.domain.model.RedSysLoginResponse.Success
import ifac.td.taxi.domain.model.RedSysOperation
import ifac.td.taxi.domain.model.RedSysReturn
import ifac.td.taxi.domain.model.RedSysSessionState
import ifac.td.taxi.domain.usecase.PrinterUseCase
import ifac.td.taxi.domain.usecase.RedSysUseCase
import ifac.td.taxi.domain.usecase.TripUseCase
import ifac.td.taxi.framework.TemporalData
import ifac.td.taxi.framework.sdk.usecase.TicketUseCase
import ifac.td.taxi.framework.util.Logs
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

class ReceiptRedSysViewModel(
    context: Application,
    private val redSysUseCase: RedSysUseCase,
    private val ticketUseCase: TicketUseCase,
    private val printerUseCase: PrinterUseCase,
    private val redSysSessionState: RedSysSessionState,
    private val tripUseCase: TripUseCase,
) : androidx.lifecycle.AndroidViewModel(context) {

    private val _uiState = MutableStateFlow(ReceiptRedSysUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<ReceiptRedSysUiEffect>()
    val uiEffect: SharedFlow<ReceiptRedSysUiEffect> = _uiEffect

    fun onEvent(event: ReceiptRedSysUiEvent) {
        when (event) {
            ReceiptRedSysUiEvent.ScreenStarted -> updateList()
            ReceiptRedSysUiEvent.RefreshRequested -> updateList()
            is ReceiptRedSysUiEvent.OperationClicked -> handleOperationClicked(event.operation)
            ReceiptRedSysUiEvent.DialogDismissed -> _uiState.update { it.copy(dialogState = null) }
            is ReceiptRedSysUiEvent.DialogButtonClicked -> handleDialogButton(event.buttonType)
        }
    }

    private fun handleOperationClicked(operation: RedSysOperation) {
        val buttonsState = ReceiptRedSysButtonsState.fromOperation(operation)
        _uiState.update {
            it.copy(
                selectedOperation = operation,
                buttonsState = buttonsState,
                dialogState = if (buttonsState.primary.visible) {
                    ReceiptRedSysDialogState(
                        title = "RedSys",
                        description = "Operación RedSys",
                        buttons = if (operation.operationType == RedSysOperation.AUTHORIZATION && operation.result != RedSysOperation.DENIED) {
                            listOf(ButtonType.DEVOLVER, ButtonType.IMPRIMIR)
                        } else {
                            listOf(ButtonType.IMPRIMIR)
                        },
                        operation = operation
                    )
                } else null
            )
        }

        if (buttonsState.primary.visible) {
            viewModelScope.launch {
                _uiEffect.emit(
                    ReceiptRedSysUiEffect.ShowDialog(
                        _uiState.value.dialogState!!
                    )
                )
            }
        }
    }

    private fun handleDialogButton(buttonType: ButtonType) {
        val operation = _uiState.value.selectedOperation ?: return
        when (buttonType) {
            ButtonType.DEVOLVER -> viewModelScope.launch {
                _uiEffect.emit(ReceiptRedSysUiEffect.DoRefund(operation))
            }

            ButtonType.IMPRIMIR -> viewModelScope.launch {
                if (operation.result != RedSysOperation.DENIED && operation.operationType == RedSysOperation.AUTHORIZATION) {
                    _uiEffect.emit(ReceiptRedSysUiEffect.PrintServiceTicket(operation))
                } else if (operation.result != RedSysOperation.DENIED && operation.operationType == RedSysOperation.REFUND) {
                    _uiEffect.emit(ReceiptRedSysUiEffect.PrintRefundTicket(operation))
                }
            }

            ButtonType.ACCEPT -> {
                _uiState.update { it.copy(dialogState = null) }
            }

            else -> Unit
        }
    }

    fun doRefund(redSysOperation: RedSysOperation) {
        viewModelScope.launch(Dispatchers.IO) {
            runCatching {
                val usernameRedSys = redSysUseCase.getUsernameRedSys()
                val passwordRedSys = TemporalData.passwordRedSys
                val loginResponse = usernameRedSys?.let { redSysUseCase.loginRedSys(it, passwordRedSys) }

                when (loginResponse) {
                    is Success -> {
                        val terminal = loginResponse.merchantList.firstOrNull()?.terminalList?.firstOrNull()
                            ?: return@launch

                        val rtsIdentifierLast5 = redSysOperation.RTSIdentifier.takeLast(5)
                        val consultaResponse = redSysUseCase.consultaRedSys(terminal, null)
                        val paymentInfo = consultaResponse?.transactionDataList?.find {
                            it.identifierRTS.takeLast(5) == rtsIdentifierLast5
                        } ?: return@launch

                        val refundResponse = redSysUseCase.doRefund(terminal, paymentInfo)
                        redSysOperation.refundResponse = refundResponse
                        processRefundResponse(refundResponse)
                    }

                    is Error -> _uiEffect.emit(ReceiptRedSysUiEffect.ShowRefundResultDialog(false))
                    null -> _uiEffect.emit(ReceiptRedSysUiEffect.ShowRefundResultDialog(false))
                }
            }.onFailure {
                _uiEffect.emit(ReceiptRedSysUiEffect.ShowRefundResultDialog(false))
            }
        }
    }

    private suspend fun processRefundResponse(refundResponse: RedCLSRefundResponse?) {
        val result = refundResponse?.let { redSysUseCase.parseDevolucion(it) }
        if (result != null && result.result.equals(RedSysOperation.AUTHORIZED, ignoreCase = true)) {
            _uiEffect.emit(ReceiptRedSysUiEffect.ShowRefundResultDialog(true))
            printRefundTicket(refund = result)
        } else {
            _uiEffect.emit(ReceiptRedSysUiEffect.ShowRefundResultDialog(false))
        }
    }

    fun printServiceTicket(redSysOperation: RedSysOperation?) {
        viewModelScope.launch {
            redSysOperation?.RTSIdentifier?.let {
                val buffTicketRedsys = tripUseCase.getTripByRedsysByRTS(it)
                val ticketBuff = buffTicketRedsys?.ticketBufferRedSys
                if (!ticketBuff.isNullOrEmpty()) {
                    printerUseCase.printTicket(ticketBuff, true)
                }
            }
        }
    }

    fun printRefundTicket(redSysOperation: RedSysOperation? = null, refund: RedSysReturn? = null) {
        viewModelScope.launch {
            val parsedResult = refund ?: redSysOperation?.refundResponse?.let { redSysUseCase.parseDevolucion(it) } ?: return@launch
            val terminalData = redSysSessionState.terminalData ?: return@launch

            val ticket = ticketUseCase.prepareRedSysRefundTicket(
                signature = false,
                resultadoDevolucion = parsedResult,
                terminalData = terminalData
            )
            printerUseCase.printTicket(ticket, true)
        }
    }

    fun updateList() {
        viewModelScope.launch(Dispatchers.IO) {
            val operations = fetchOperations()
            _uiState.update { it.copy(operations = operations ?: emptyList()) }
        }
    }

    private suspend fun fetchOperations(): List<RedSysOperation>? {
        redSysUseCase.configureRedSys()
        val usernameRedSys = redSysUseCase.getUsernameRedSys() ?: return null
        val passwordRedSys = TemporalData.passwordRedSys
        val loginResponse = redSysUseCase.loginRedSys(usernameRedSys, passwordRedSys)

        return when (loginResponse) {
            is Success -> {
                val terminal = loginResponse.merchantList.firstOrNull()?.terminalList?.firstOrNull() ?: return null
                val consultaResponse = redSysUseCase.consultaRedSys(terminal, null)
                redSysUseCase.parseConsulta(consultaResponse)
            }

            is Error, null -> null
        }
    }
}
```

```kotlin
package ifac.td.taxi.ui.screen.receiptredsys

import androidx.compose.foundation.clickable
import androidx.compose.foundation.layout.*
import androidx.compose.foundation.lazy.LazyColumn
import androidx.compose.foundation.lazy.items
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Modifier
import androidx.compose.ui.unit.dp
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.lifecycle.compose.LifecycleStartEffect
import ifac.td.taxi.domain.model.RedSysOperation
import ifac.td.taxi.ui.custom.button.ButtonType
import kotlinx.coroutines.flow.collectLatest

@Composable
fun ReceiptRedSysScreen(
    viewModel: ReceiptRedSysViewModel,
    onBack: () -> Unit = {},
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()

    LifecycleStartEffect(Unit) {
        viewModel.onEvent(ReceiptRedSysUiEvent.ScreenStarted)
        onStop { }
    }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                is ReceiptRedSysUiEffect.ShowDialog -> {
                    // handled by dialog state below
                }

                is ReceiptRedSysUiEffect.DoRefund -> viewModel.doRefund(effect.operation)
                is ReceiptRedSysUiEffect.PrintServiceTicket -> viewModel.printServiceTicket(effect.operation)
                is ReceiptRedSysUiEffect.PrintRefundTicket -> viewModel.printRefundTicket(effect.operation)
                is ReceiptRedSysUiEffect.ShowRefundResultDialog -> {
                    // dialog state can be updated here if you want a dedicated result dialog
                }

                ReceiptRedSysUiEffect.RefreshList -> viewModel.updateList()
            }
        }
    }

    Box(modifier = Modifier.fillMaxSize()) {
        LazyColumn(modifier = Modifier.fillMaxSize()) {
            items(uiState.operations) { operation ->
                ReceiptRedSysOperationRow(
                    operation = operation,
                    onClick = { viewModel.onEvent(ReceiptRedSysUiEvent.OperationClicked(operation)) }
                )
            }
        }

        uiState.dialogState?.let { dialog ->
            AlertDialog(
                onDismissRequest = { viewModel.onEvent(ReceiptRedSysUiEvent.DialogDismissed) },
                title = { Text(dialog.title) },
                text = { Text(dialog.description) },
                confirmButton = {
                    Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                        dialog.buttons.forEach { button ->
                            TextButton(onClick = {
                                viewModel.onEvent(ReceiptRedSysUiEvent.DialogButtonClicked(button))
                            }) {
                                Text(button.name)
                            }
                        }
                    }
                }
            )
        }
    }
}

@Composable
private fun ReceiptRedSysOperationRow(
    operation: RedSysOperation,
    onClick: () -> Unit
) {
    Column(
        modifier = Modifier
            .fillMaxWidth()
            .clickable { onClick() }
            .padding(16.dp)
    ) {
        Text(text = "RTS: ${operation.RTSIdentifier}")
        Text(text = "Type: ${operation.operationType}")
        Text(text = "Result: ${operation.result}")
    }
}
```

```kotlin
/**
 * Exact reducer mirroring the fragment branches one-by-one:
 *
 * if ((operation.operationType == REFUND && operation.refundResponse != null) ||
 *     operation.operationType == AUTHORIZATION && operation.result != DENIED) {
 *   dialog buttons = if AUTHORIZATION && not DENIED -> [DEVOLVER, IMPRIMIR]
 *                   else -> [IMPRIMIR]
 * }
 * else -> hidden
 */
fun ReceiptRedSysButtonsState.reduceForOperation(operation: RedSysOperation): ReceiptRedSysButtonsState {
    return when {
        (operation.operationType == REFUND && operation.refundResponse != null) ||
            (operation.operationType == AUTHORIZATION && operation.result != DENIED) -> {
            if (operation.operationType == AUTHORIZATION && operation.result != DENIED) {
                copy(
                    primary = ReceiptRedSysButtonVisualState(
                        visible = true,
                        enabled = true,
                        buttonType = ButtonType.DEVOLVER,
                        backgroundColor = ReceiptRedSysButtonBackground.RED,
                        text = "DEVOLVER",
                        iconVisible = true,
                        textVisible = true
                    ),
                    secondary = ReceiptRedSysButtonVisualState(
                        visible = true,
                        enabled = true,
                        buttonType = ButtonType.IMPRIMIR,
                        backgroundColor = ReceiptRedSysButtonBackground.BLUE,
                        text = "IMPRIMIR",
                        iconVisible = true,
                        textVisible = true
                    )
                )
            } else {
                copy(
                    primary = ReceiptRedSysButtonVisualState(
                        visible = true,
                        enabled = true,
                        buttonType = ButtonType.IMPRIMIR,
                        backgroundColor = ReceiptRedSysButtonBackground.BLUE,
                        text = "IMPRIMIR",
                        iconVisible = true,
                        textVisible = true
                    ),
                    secondary = ReceiptRedSysButtonVisualState(visible = false)
                )
            }
        }

        else -> copy(
            primary = ReceiptRedSysButtonVisualState(visible = false),
            secondary = ReceiptRedSysButtonVisualState(visible = false)
        )
    }
}

class ReceiptRedSysViewModel(
    context: Application,
    private val redSysUseCase: RedSysUseCase,
    private val ticketUseCase: TicketUseCase,
    private val printerUseCase: PrinterUseCase,
    private val redSysSessionState: RedSysSessionState,
    private val tripUseCase: TripUseCase,
) : AndroidViewModel(context) {

    private val TAG = "ReceiptRedSysViewModel"

    private val _uiState = MutableStateFlow(ReceiptRedSysUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<ReceiptRedSysUiEffect>()
    val uiEffect: SharedFlow<ReceiptRedSysUiEffect> = _uiEffect

    fun onEvent(event: ReceiptRedSysUiEvent) {
        when (event) {
            ReceiptRedSysUiEvent.ScreenStarted -> updateList()
            ReceiptRedSysUiEvent.RefreshRequested -> updateList()
            is ReceiptRedSysUiEvent.OperationClicked -> handleOperationClicked(event.operation)
            is ReceiptRedSysUiEvent.DialogButtonClicked -> handleDialogButton(event.buttonType)
            ReceiptRedSysUiEvent.DialogDismissed -> _uiState.update { it.copy(dialogState = null) }
        }
    }

    private fun handleOperationClicked(operation: RedSysOperation) {
        val buttonsState = ReceiptRedSysButtonsState.fromOperation(operation)
        val dialogState = buildDialogState(operation)

        _uiState.update {
            it.copy(
                selectedOperation = operation,
                buttonsState = buttonsState,
                dialogState = dialogState
            )
        }

        if (dialogState != null) {
            viewModelScope.launch {
                _uiEffect.emit(ReceiptRedSysUiEffect.ShowDialog(dialogState))
            }
        }
    }

    private fun buildDialogState(operation: RedSysOperation): ReceiptRedSysDialogState? {
        val shouldOpenDialog =
            (operation.operationType == REFUND && operation.refundResponse != null) ||
                (operation.operationType == AUTHORIZATION && operation.result != DENIED)

        if (!shouldOpenDialog) return null

        val buttons = if (operation.operationType == AUTHORIZATION && operation.result != DENIED) {
            listOf(ButtonType.DEVOLVER, ButtonType.IMPRIMIR)
        } else {
            listOf(ButtonType.IMPRIMIR)
        }

        return ReceiptRedSysDialogState(
            title = "RedSys",
            description = "Operación RedSys",
            buttons = buttons,
            operation = operation
        )
    }

    private fun handleDialogButton(buttonType: ButtonType) {
        val operation = _uiState.value.selectedOperation ?: return
        when (buttonType) {
            ButtonType.DEVOLVER -> viewModelScope.launch {
                _uiEffect.emit(ReceiptRedSysUiEffect.DoRefund(operation))
            }

            ButtonType.IMPRIMIR -> viewModelScope.launch {
                if (operation.result != DENIED && operation.operationType == AUTHORIZATION) {
                    _uiEffect.emit(ReceiptRedSysUiEffect.PrintServiceTicket(operation))
                } else if (operation.result != DENIED && operation.operationType == REFUND) {
                    _uiEffect.emit(ReceiptRedSysUiEffect.PrintRefundTicket(operation))
                }
            }

            ButtonType.ACCEPT -> _uiState.update { it.copy(dialogState = null) }
            else -> Unit
        }
    }

    fun updateList() {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "Updating list...")
            val operations = fetchOperations()
            Logs.d(TAG, "Updated list: $operations")
            _uiState.update { it.copy(operations = operations ?: emptyList()) }
        }
    }

    fun doRefund(redSysOperation: RedSysOperation) {
        viewModelScope.launch(Dispatchers.IO) {
            runCatching {
                val usernameRedSys = redSysUseCase.getUsernameRedSys()
                val passwordRedSys = TemporalData.passwordRedSys
                val loginResponse = usernameRedSys?.let { redSysUseCase.loginRedSys(it, passwordRedSys) }

                when (loginResponse) {
                    is Success -> {
                        val terminal = loginResponse.merchantList.firstOrNull()?.terminalList?.firstOrNull()
                        if (terminal == null) {
                            Logs.d(TAG, "No terminal available in the login response")
                            _uiEffect.emit(ReceiptRedSysUiEffect.ShowRefundResultDialog(false))
                            return@launch
                        }

                        val rtsIdentifierLast5 = redSysOperation.RTSIdentifier.takeLast(5)
                        val consultaResponse = redSysUseCase.consultaRedSys(terminal, null)
                        val paymentInfo = consultaResponse?.transactionDataList?.find {
                            it.identifierRTS.takeLast(5) == rtsIdentifierLast5
                        }

                        if (paymentInfo == null) {
                            Logs.d(TAG, "No payment info found for the given RTSIdentifier")
                            _uiEffect.emit(ReceiptRedSysUiEffect.ShowRefundResultDialog(false))
                            return@launch
                        }

                        val refundResponse = redSysUseCase.doRefund(terminal, paymentInfo)
                        redSysOperation.refundResponse = refundResponse
                        processRefundResponse(refundResponse)
                    }

                    is Error -> {
                        Logs.e(TAG, "Error during login: ${loginResponse.errorCode} - ${loginResponse.errorMessage}")
                        _uiEffect.emit(ReceiptRedSysUiEffect.ShowRefundResultDialog(false))
                    }

                    null -> {
                        Logs.e(TAG, "Login response is null")
                        _uiEffect.emit(ReceiptRedSysUiEffect.ShowRefundResultDialog(false))
                    }
                }
            }.onFailure { e ->
                Logs.e(TAG, "Exception in doRefund: ${e.message}")
                _uiEffect.emit(ReceiptRedSysUiEffect.ShowRefundResultDialog(false))
            }
        }
    }

    private suspend fun processRefundResponse(refundResponse: RedCLSRefundResponse?) {
        refundResponse?.let {
            val result = redSysUseCase.parseDevolucion(it)
            if (result != null) {
                when {
                    result.result.equals(RedSysOperation.AUTHORIZED, ignoreCase = true) -> {
                        Logs.d(TAG, "processRefundResponse: Refund authorized")
                        _uiEffect.emit(ReceiptRedSysUiEffect.ShowRefundResultDialog(true))
                        printRefundTicket(refund = result)
                    }

                    else -> {
                        Logs.d(TAG, "processRefundResponse: Refund result: $result")
                        _uiEffect.emit(ReceiptRedSysUiEffect.ShowRefundResultDialog(false))
                    }
                }
            } else {
                _uiEffect.emit(ReceiptRedSysUiEffect.ShowRefundResultDialog(false))
            }
        } ?: run {
            Logs.d(TAG, "processRefundResponse: Refund response is null")
            _uiEffect.emit(ReceiptRedSysUiEffect.ShowRefundResultDialog(false))
        }
    }

    fun printServiceTicket(redSysOperation: RedSysOperation?) {
        viewModelScope.launch {
            redSysOperation?.RTSIdentifier?.let {
                val buffTicketRedsys = tripUseCase.getTripByRedsysByRTS(it)
                val ticketBuff = buffTicketRedsys?.ticketBufferRedSys
                if (!ticketBuff.isNullOrEmpty()) {
                    printerUseCase.printTicket(ticketBuff, true)
                }
            }
        }
    }

    fun printRefundTicket(redSysOperation: RedSysOperation? = null, refund: RedSysReturn? = null) {
        viewModelScope.launch {
            val parsedResult = refund ?: redSysOperation?.refundResponse?.let {
                redSysUseCase.parseDevolucion(it)
            } ?: run {
                Logs.e(TAG, "printRefundTicket: Aborted. No refund data available.")
                return@launch
            }

            val terminalData = redSysSessionState.terminalData ?: run {
                Logs.e(TAG, "printRefundTicket: Aborted. No terminal data found in session state")
                return@launch
            }

            try {
                val ticket = ticketUseCase.prepareRedSysRefundTicket(
                    signature = false,
                    resultadoDevolucion = parsedResult,
                    terminalData = terminalData
                )
                printerUseCase.printTicket(ticket, true)
            } catch (e: Exception) {
                Logs.e(TAG, "printRefundTicket: Error during ticket preparation or printing: ${e.message}")
            }
        }
    }

    private suspend fun fetchOperations(): List<RedSysOperation>? {
        Logs.d(TAG, "Fetching operations...")
        redSysUseCase.configureRedSys()

        val usernameRedSys = redSysUseCase.getUsernameRedSys()
        if (usernameRedSys == null) {
            Logs.e(TAG, "Username RedSys is null.")
            return null
        }

        val passwordRedSys = TemporalData.passwordRedSys
        val loginResponse = redSysUseCase.loginRedSys(usernameRedSys, passwordRedSys)

        return when (loginResponse) {
            is Success -> {
                val terminal = loginResponse.merchantList.firstOrNull()?.terminalList?.firstOrNull()
                if (terminal == null) {
                    Logs.e(TAG, "No terminal available in the login response.")
                    return null
                }

                val consultaResponse = redSysUseCase.consultaRedSys(terminal, null)
                redSysUseCase.parseConsulta(consultaResponse)
            }

            is Error -> {
                Logs.e(TAG, "Login RedSys failed: ${loginResponse.errorCode} - ${loginResponse.errorMessage}")
                null
            }

            null -> {
                Logs.e(TAG, "Login response is null.")
                null
            }
        }
    }
}

private fun buildReceiptRedSysTtsText(
    description: String,
    title: String,
    buttons: List<String>
): String {
    return buildString {
        append(description.ifEmpty { title })
        if (buttons.isNotEmpty()) {
            append(". ")
            append(buttons.dropLast(1).joinToString(", "))
            if (buttons.size > 1) {
                append(" o ")
                append(buttons.last())
            } else {
                append(buttons.first())
            }
        }
    }
}
```

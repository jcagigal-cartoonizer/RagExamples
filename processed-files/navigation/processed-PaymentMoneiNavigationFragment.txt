## Important note about `viewModels()`
That means the Fragment should ideally use the standard Fragment delegate:
```kotlin
private val viewModel: PaymentMoneiComposeViewModel by viewModels()
```
However, your `PaymentMoneiComposeViewModel` likely comes from **Koin**, and for Koin-backed fragments the correct delegate is usually:
```kotlin
private val viewModel: PaymentMoneiComposeViewModel by viewModel()
```
Here is the AndroidX version:
```kotlin
package ifac.td.taxi.ui.screen
// # Block 76-2: import android.os.Bundle
import android.os.Bundle
import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import androidx.compose.ui.platform.ComposeView
import androidx.compose.ui.platform.ViewCompositionStrategy
import androidx.fragment.app.Fragment
import androidx.fragment.app.viewModels
import ifac.td.taxi.viewmodel.MainActivityViewModel
import ifac.td.taxi.compose.viewmodel.PaymentMoneiComposeViewModel
import ifac.td.taxi.ui.theme.TaxiTheme
class PaymentMoneiNavigationFragment : Fragment() {
    private val viewModel: PaymentMoneiComposeViewModel by viewModels()
    private val sharedViewModel: MainActivityViewModel by viewModels(
        ownerProducer = { requireActivity() }
    )
    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View {
        return ComposeView(requireContext()).apply {
            setViewCompositionStrategy(
                ViewCompositionStrategy.DisposeOnViewTreeLifecycleDestroyed
            )
            setContent {
                TaxiTheme {
                    PaymentMoneiScreen(
                        viewModel = viewModel,
                        sharedViewModel = sharedViewModel,
                        onNavigateBack = {
                            requireActivity().onBackPressedDispatcher.onBackPressed()
                        },
                        onNavigateHome = {
                            requireActivity().onBackPressedDispatcher.onBackPressed()
                        },
                        onPrintTicket = { ticketBody ->
                            sharedViewModel.printTicket(ticketBody)
                        },
                        showHeader = { visible ->
                            // hook to your activity UI if needed
                        }
                    )
                }
            }
        }
    }
}
```
But this **only works if `PaymentMoneiComposeViewModel` and `MainActivityViewModel` have proper `ViewModelProvider.Factory` support**.

## One more thing
Your composable `PaymentMoneiScreen(...)` currently handles effects like this:
```kotlin
PaymentMoneiUiEffect.NavigateBack -> onNavigateBack()
PaymentMoneiUiEffect.NavigateHome -> onNavigateHome()
PaymentMoneiUiEffect.PrintTicket -> onPrintTicket(effect.ticketBody)
```
So the Fragment only needs to provide those callbacks, which makes the migration from XML/ViewBinding to Compose very straightforward.

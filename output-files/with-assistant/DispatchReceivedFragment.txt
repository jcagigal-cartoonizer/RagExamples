```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.domain.usecase.DispatchUseCase
import ifac.td.taxi.domain.usecase.UserPreferencesUseCase
import ifac.td.taxi.framework.sdk.usecase.LicensingUseCase
import ifac.td.taxi.repository.room.dao.BravoConfigurationVariableDao
import ifac.td.taxi.repository.room.entities.BravoConfigurationVariableEntity
import ifac.td.taxi.ui.screen.dispatchreceived.DispatchReceivedButtonsState
import ifac.td.taxi.ui.screen.dispatchreceived.DispatchReceivedData
import ifac.td.taxi.ui.screen.dispatchreceived.DispatchReceivedDialogState
import ifac.td.taxi.ui.screen.dispatchreceived.DispatchReceivedUiEffect
import ifac.td.taxi.ui.screen.dispatchreceived.DispatchReceivedUiEvent
import ifac.td.taxi.ui.screen.dispatchreceived.DispatchReceivedUiState
import ifac.td.taxi.viewmodel.model.UtilsModel.Companion.toDispatchReceivedModel
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

class DispatchReceivedComposeViewModel(
    private val dispatchUseCase: DispatchUseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    private val bravoConfigurationVariableDao: BravoConfigurationVariableDao,
    private val licensingUsecase: LicensingUseCase,
    context: Application,
) : BaseViewModel(context) {

    private val _uiState = MutableStateFlow(DispatchReceivedUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<DispatchReceivedUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()

    private var actualDispatchId: Long? = null
    private var showCancelDialogPermission: Boolean = false

    init {
        viewModelScope.launch {
            bravoConfigurationVariableDao.getBravoConfigurationVariable()?.let {
                // keep if needed for future logic
            }
        }
    }

    fun onEvent(event: DispatchReceivedUiEvent) {
        when (event) {
            DispatchReceivedUiEvent.OnScreenStarted -> Unit

            DispatchReceivedUiEvent.OnAcceptClicked -> acceptDispatch()

            DispatchReceivedUiEvent.OnCancelClicked -> {
                if (showCancelDialogPermission) {
                    _uiState.update {
                        it.copy(
                            showCancelConfirmationDialog = true,
                            dialog = DispatchReceivedDialogState(
                                title = "Advertencia",
                                description = "¿Quieres rechazar este despacho?",
                            )
                        )
                    }
                } else {
                    rejectManualDispatch()
                }
            }

            DispatchReceivedUiEvent.OnDialogAcceptClicked -> {
                _uiState.update { it.copy(showCancelConfirmationDialog = false, dialog = null) }
                rejectManualDispatch()
            }

            DispatchReceivedUiEvent.OnDialogDismissed -> {
                _uiState.update { it.copy(showCancelConfirmationDialog = false, dialog = null) }
            }
        }
    }

    fun getDispatch(id: Long) {
        viewModelScope.launch {
            actualDispatchId = id
            val dispatch = dispatchUseCase.getDispatchById(id)
            if (dispatch != null) {
                val model = dispatch.toDispatchReceivedModel(id)
                _uiState.update {
                    it.copy(
                        dispatch = DispatchReceivedData(
                            idDispatch = id,
                            pickUpTime = model.pickUpTime,
                            pickUpZone = model.pickUpZone,
                            smallPickUpAddress = model.smallPickUpAdress,
                            city = model.city,
                            passengerName = model.passengerName
                        )
                    )
                }
            }
        }
    }

    fun checkCancelDialogPermission() {
        viewModelScope.launch {
            val preferences = userPreferencesUseCase.getUserPreferences()
            showCancelDialogPermission = preferences?.showConfRejectDispatch == true
        }
    }

    fun checkIsDirRecogidaAceptaDespacho() {
        viewModelScope.launch {
            val preferences = licensingUsecase.getLicensingParameters()
            val addressAccepted = preferences?.isAddressTakenAcceptsDispatch == true
            _uiState.update {
                it.copy(
                    buttonsState = it.buttonsState.copy(
                        acceptVisible = addressAccepted,
                        cancelVisible = true
                    )
                )
            }
        }
    }

    fun showPickupAddress(): Boolean {
        return true
    }

    private fun acceptDispatch() {
        viewModelScope.launch {
            val tripId = actualDispatchId ?: return@launch
            _uiEffect.emit(DispatchReceivedUiEffect.NavigateToInfoDispatch(tripId))
        }
    }

    private fun rejectManualDispatch() {
        viewModelScope.launch {
            _uiEffect.emit(DispatchReceivedUiEffect.NavigateBack)
        }
    }

    fun stopCallTimer() {
        // keep if you still need timer cleanup
    }

    fun getDispatchProvider(idDispatch: Long, dataToString: List<String>) {
        viewModelScope.launch(Dispatchers.IO) {
            var provider: String? = null
            for (str in dataToString.reversed()) {
                val trimmed = str.trim()
                if (trimmed.startsWith("OPERADOR:")) {
                    provider = trimmed.substringAfter("OPERADOR:").trim()
                    break
                }
            }
            provider?.let {
                dispatchUseCase.updateDispatchProvider(idDispatch = idDispatch, provider = it)
            }
        }
    }
}
```

```kotlin
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.unit.dp
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import ifac.td.taxi.viewmodel.DispatchReceivedComposeViewModel

@Composable
fun DispatchReceivedRoute(
    viewModel: DispatchReceivedComposeViewModel,
    idDispatch: Long,
    onNavigateToInfoDispatch: (Long) -> Unit,
    onNavigateBack: () -> Unit,
    onRequestPhonePermission: () -> Unit = {},
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()

    LaunchedEffect(idDispatch) {
        viewModel.getDispatch(idDispatch)
        viewModel.checkCancelDialogPermission()
        viewModel.onEvent(DispatchReceivedUiEvent.OnScreenStarted)
    }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collect { effect ->
            when (effect) {
                is DispatchReceivedUiEffect.NavigateToInfoDispatch -> onNavigateToInfoDispatch(effect.tripId)
                DispatchReceivedUiEffect.NavigateBack -> onNavigateBack()
                DispatchReceivedUiEffect.RequestPhonePermission -> onRequestPhonePermission()
                DispatchReceivedUiEffect.ShowRejectDispatchDialog -> Unit
            }
        }
    }

    DispatchReceivedScreen(
        uiState = uiState,
        onEvent = viewModel::onEvent
    )
}

@Composable
fun DispatchReceivedScreen(
    uiState: DispatchReceivedUiState,
    onEvent: (DispatchReceivedUiEvent) -> Unit,
) {
    Column(
        modifier = Modifier
            .fillMaxSize()
            .background(Color.White)
            .padding(16.dp),
        verticalArrangement = Arrangement.SpaceBetween
    ) {
        Column(modifier = Modifier.fillMaxWidth()) {
            uiState.dispatch?.let { dispatch ->
                Text(text = dispatch.pickUpTime.orEmpty(), style = MaterialTheme.typography.titleMedium)
                Text(text = dispatch.pickUpZone.orEmpty(), style = MaterialTheme.typography.bodyLarge)
                Text(text = dispatch.smallPickUpAddress.orEmpty(), style = MaterialTheme.typography.bodyLarge)
                Text(text = dispatch.city.orEmpty(), style = MaterialTheme.typography.bodyLarge)
                Text(text = dispatch.passengerName.orEmpty(), style = MaterialTheme.typography.bodyLarge)
            }
        }

        DispatchReceivedButtons(
            state = uiState.buttonsState,
            onAcceptClick = { onEvent(DispatchReceivedUiEvent.OnAcceptClicked) },
            onCancelClick = { onEvent(DispatchReceivedUiEvent.OnCancelClicked) }
        )
    }

    if (uiState.showCancelConfirmationDialog && uiState.dialog != null) {
        AlertDialog(
            onDismissRequest = { onEvent(DispatchReceivedUiEvent.OnDialogDismissed) },
            title = { Text(uiState.dialog.title) },
            text = { Text(uiState.dialog.description) },
            confirmButton = {
                TextButton(onClick = { onEvent(DispatchReceivedUiEvent.OnDialogAcceptClicked) }) {
                    Text(uiState.dialog.acceptText)
                }
            },
            dismissButton = {
                TextButton(onClick = { onEvent(DispatchReceivedUiEvent.OnDialogDismissed) }) {
                    Text(uiState.dialog.cancelText)
                }
            }
        )
    }
}
```

## Reducer implementation

```kotlin
package ifac.td.taxi.ui.screen.dispatchreceived

object DispatchReceivedButtonsReducer {

    fun reduce(
        current: DispatchReceivedButtonsState,
        action: DispatchReceivedButtonsAction
    ): DispatchReceivedButtonsState {
        return when (action) {
            is DispatchReceivedButtonsAction.SetLoading -> {
                current.copy(isLoading = action.show)
            }

            is DispatchReceivedButtonsAction.SetDispatchAttributes -> {
                current.copy(
                    isLandscape = action.isLandscape,
                    attribute1 = current.attribute1.copy(
                        visible = !action.attribute1Title.isNullOrEmpty() || action.isLandscape.not(),
                        title = action.attribute1Title,
                        spinnerSelection = 1
                    ),
                    attribute2 = current.attribute2.copy(
                        visible = !action.attribute2Title.isNullOrEmpty() || action.isLandscape.not(),
                        title = action.attribute2Title
                    ),
                    attribute3 = current.attribute3.copy(
                        visible = !action.attribute3Title.isNullOrEmpty() || action.isLandscape.not(),
                        title = action.attribute3Title
                    ),
                    attribute4 = current.attribute4.copy(
                        visible = !action.attribute4Title.isNullOrEmpty() || action.isLandscape.not(),
                        title = action.attribute4Title
                    )
                )
            }

            is DispatchReceivedButtonsAction.SetHasMoneiAccount -> {
                current.copy(
                    flowMenuVisible = action.hasAccount,
                    extraButtonsVisible = action.hasAccount
                )
            }

            is DispatchReceivedButtonsAction.SetInAppPaymentResponse -> {
                current.copy(
                    isWaitingAppPaymentResponse = true,
                    flowMenuVisible = true,
                    extraButtonsVisible = true,
                    cancelPaymentButtonVisible = false,
                    appPaymentButtonVisible = true,
                    othersPaymentButtonVisible = true,
                    appPaymentButtonEnabledStyle = true,
                    othersPaymentButtonEnabledStyle = true
                )
            }

            is DispatchReceivedButtonsAction.SetWaitingAppPaymentResponse -> {
                current.copy(isWaitingAppPaymentResponse = action.waiting)
            }

            is DispatchReceivedButtonsAction.SetExtraData -> {
                val total = (action.dispatchAmount ?: 0) + (action.tolls ?: 0) + (action.tips ?: 0)
                current.copy(
                    amountVisible = true,
                    amountText = total.toString()
                )
            }

            is DispatchReceivedButtonsAction.SetPaymentMenuVisible -> {
                current.copy(flowMenuVisible = action.visible)
            }

            is DispatchReceivedButtonsAction.SetAppPaymentDialog -> {
                current.copy(
                    showAppPaymentDialog = true,
                    appPaymentDialog = AppPaymentDialogState(
                        title = action.title,
                        description = action.description,
                        centerText = action.centerText,
                        cancellable = action.cancellable
                    )
                )
            }

            DispatchReceivedButtonsAction.HideAppPaymentDialog -> {
                current.copy(
                    showAppPaymentDialog = false,
                    appPaymentDialog = null
                )
            }
        }
    }
}
```

# 6) Important note

Your pasted fragment contains logic that depends on:
- `validCredit(...)`

```kotlin
class DispatchReceivedComposeViewModel(
    private val dispatchUseCase: DispatchUseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    private val licensingUsecase: LicensingUseCase,
    private val bravoConfigurationVariableDao: BravoConfigurationVariableDao,

    val uiEffect = _uiEffect.asSharedFlow()

    private var currentDispatchId: Long? = null
    private var showCancelDialogPermission: Boolean = false

    fun initialize(idDispatch: Long) {
        getDispatch(idDispatch)
        checkCancelDialogPermission()
    }

    fun onEvent(event: DispatchReceivedUiEvent) {
        when (event) {
            DispatchReceivedUiEvent.OnScreenStarted -> Unit
            DispatchReceivedUiEvent.OnAcceptClicked -> acceptDispatch()
            DispatchReceivedUiEvent.OnCancelClicked -> {
                if (showCancelDialogPermission) {
                    _uiState.update {
                        it.copy(
                            showCancelDialog = true,
                            cancelDialog = DispatchReceivedDialogState(
                                title = "Advertencia",
                                description = "¿Quieres rechazar este despacho?"
                            )
                        )
                    }
                    viewModelScope.launch { _uiEffect.emit(DispatchReceivedUiEffect.OpenCancelDispatchDialog) }
                } else {
                    rejectManualDispatch()
                }
            }
            DispatchReceivedUiEvent.OnCancelDialogAccepted -> {
                _uiState.update { it.copy(showCancelDialog = false, cancelDialog = null) }
                rejectManualDispatch()
            }
            DispatchReceivedUiEvent.OnCancelDialogDismissed -> {
                _uiState.update { it.copy(showCancelDialog = false, cancelDialog = null) }
            }
            DispatchReceivedUiEvent.OnAppPaymentDialogAccepted -> {
                _uiState.update { it.copy(buttons = it.buttons.copy(showAppPaymentDialog = false, appPaymentDialog = null)) }
            }
            DispatchReceivedUiEvent.OnAppPaymentDialogDismissed -> {
                _uiState.update { it.copy(buttons = it.buttons.copy(showAppPaymentDialog = false, appPaymentDialog = null)) }
            }
        }
    }

    fun getDispatch(id: Long) {
        viewModelScope.launch {
            currentDispatchId = id
            val dispatch = dispatchUseCase.getDispatchById(id)
            if (dispatch != null) {
                _uiState.update {
                    it.copy(
                        dispatch = DispatchReceivedDispatchUi(
                            idDispatch = id,
                            pickUpTime = dispatch.pickUpTime,
                            pickUpZone = dispatch.pickUpZone,
                            smallPickUpAddress = dispatch.smallPickUpAdress,
                            city = dispatch.city,
                            passengerName = dispatch.passengerName,
                            requireSignature = dispatch.requireSignature,
                            requireVoucher = dispatch.requireVoucher,
                            requireQr = dispatch.requireQr,
                            amount = dispatch.amount,
                            clientType = dispatch.clientType,
                            isConcertedPrice = dispatch.isConcertedPrice,
                            isConcertedMaximumPrice = dispatch.isConcertedMaximumPrice,
                            attribute1Title = dispatch.attribute1Title,
                            attribute2Title = dispatch.attribute2Title,
                            attribute3Title = dispatch.attribute3Title,
                            attribute4Title = dispatch.attribute4Title,
                        )
                    )
                }
                handleDispatchRequirements(dispatch)
            }
        }
    }

    fun checkCancelDialogPermission() {
        viewModelScope.launch {
            val preferences = userPreferencesUseCase.getUserPreferences()
            showCancelDialogPermission = preferences?.showConfRejectDispatch == true
        }
    }

    fun checkIsDirRecogidaAceptaDespacho() {
        viewModelScope.launch {
            val preferences = licensingUsecase.getLicensingParameters()
            val addressAccepted = preferences?.isAddressTakenAcceptsDispatch == true
            _uiState.update {
                it.copy(buttons = it.buttons.copy(acceptVisible = addressAccepted))
            }
        }
    }

    fun stopCallTimer() = Unit

    fun rejectManualDispatch() {
        viewModelScope.launch { _uiEffect.emit(DispatchReceivedUiEffect.NavigateBack) }
    }

    fun acceptDispatch() {
        val id = currentDispatchId ?: return
        viewModelScope.launch {
            _uiEffect.emit(
                DispatchReceivedUiEffect.NavigateToInfoDispatch(
                    HomeDirections.goToInfoDispatchFragment(id)
                )
            )
        }
    }

    fun isNumberPhoneIn() {
        viewModelScope.launch { _uiEffect.emit(DispatchReceivedUiEffect.RequestPhonePermission) }
    }

    fun dispatchRequireSignature(dispatch: DispatchReceivedDispatchUi) = Unit
    fun dispatchRequireVoucher(dispatch: DispatchReceivedDispatchUi) = Unit
    fun dispatchRequireQR(dispatch: DispatchReceivedDispatchUi) = Unit
    fun validCredit(dispatch: DispatchReceivedDispatchUi) = Unit

    private fun handleDispatchRequirements(dispatch: DispatchReceivedDispatchUi) {
        if (dispatch.requireSignature) {
            dispatchRequireSignature(dispatch)
        } else if (dispatch.requireVoucher) {
            dispatchRequireVoucher(dispatch)
        } else if (dispatch.requireQr) {
            dispatchRequireQR(dispatch)
        } else {
            validCredit(dispatch)
        }
    }

    fun updateTripAmount(totalAmount: Int) {
        _uiState.update {
            it.copy(buttons = it.buttons.copy(amountVisible = true, amountText = totalAmount.toString()))
        }
    }

@Composable
fun DispatchReceivedScreen(
    navController: NavController,
    viewModel: DispatchReceivedComposeViewModel,
    sharedViewModel: MainActivityViewModel,
    idDispatch: Long,
    onShowHeader: (Boolean) -> Unit,
    onShowBottomBar: (Boolean) -> Unit,
    onCloseDialog: () -> Unit,
    onRequestPhonePermission: () -> Unit,
) {
    val state by viewModel.uiState.collectAsStateWithLifecycle()
    val sharedState by sharedViewModel.dispatchFlow.collectAsStateWithLifecycle()

    var cancelDialogState by remember { mutableStateOf<DispatchReceivedDialogState?>(null) }
    var appPaymentDialogState by remember { mutableStateOf<DispatchReceivedDialogState?>(null) }

    LaunchedEffect(Unit) {
        onShowHeader(false)
        onShowBottomBar(false)
        onCloseDialog()
        viewModel.initialize(idDispatch)
    }

    LaunchedEffect(viewModel) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                is DispatchReceivedUiEffect.NavigateToInfoDispatch -> {
                    navController.navigate(effect.directions)
                }
                DispatchReceivedUiEffect.NavigateBack -> navController.popBackStack()
                DispatchReceivedUiEffect.RequestPhonePermission -> onRequestPhonePermission()
                DispatchReceivedUiEffect.OpenCancelDispatchDialog -> {
                    cancelDialogState = state.cancelDialog
                }
                is DispatchReceivedUiEffect.OpenAppPaymentDialog -> {
                    appPaymentDialogState = effect.dialog
                }
                DispatchReceivedUiEffect.CloseAppPaymentDialog -> {
                    appPaymentDialogState = null
                }
            }
        }
    }

    LaunchedEffect(sharedState) {
        sharedState?.let { dispatch ->
            if (dispatch.requireSignature) {
                viewModel.dispatchRequireSignature(dispatch)
            } else if (dispatch.requireVoucher) {
                viewModel.dispatchRequireVoucher(dispatch)
            } else if (dispatch.requireQr) {
                viewModel.dispatchRequireQR(dispatch)
            } else {
                viewModel.validCredit(dispatch)
            }
        }
    }

    Scaffold { padding ->
        Column(
            modifier = Modifier
                .fillMaxSize()
                .padding(padding)
                .padding(16.dp),
            verticalArrangement = Arrangement.SpaceBetween
        ) {
            Column(modifier = Modifier.fillMaxWidth()) {
                state.dispatch?.let { dispatch ->
                    Text(text = dispatch.pickUpTime.orEmpty(), style = MaterialTheme.typography.titleMedium)
                    Text(text = dispatch.pickUpZone.orEmpty(), style = MaterialTheme.typography.bodyLarge)
                    Text(text = dispatch.smallPickUpAddress.orEmpty(), style = MaterialTheme.typography.bodyLarge)
                    Text(text = dispatch.city.orEmpty(), style = MaterialTheme.typography.bodyLarge)
                    Text(text = dispatch.passengerName.orEmpty(), style = MaterialTheme.typography.bodyLarge)
                }

                Spacer(modifier = Modifier.height(24.dp))

                DispatchReceivedButtons(
                    state = state.buttons,
                    onAccept = {
                        viewModel.acceptDispatch()
                        viewModel.stopCallTimer()
                    },
                    onCancel = {
                        viewModel.onEvent(DispatchReceivedUiEvent.OnCancelClicked)
                        viewModel.stopCallTimer()
                    }
                )
            }
        }
    }

    cancelDialogState?.let { dialog ->
        AlertDialog(
            onDismissRequest = {
                cancelDialogState = null
                viewModel.onEvent(DispatchReceivedUiEvent.OnCancelDialogDismissed)
            },
            title = { Text(dialog.title) },
            text = { Text(dialog.description) },
            confirmButton = {
                TextButton(onClick = {
                    cancelDialogState = null
                    viewModel.onEvent(DispatchReceivedUiEvent.OnCancelDialogAccepted)
                }) { Text(dialog.acceptText) }
            },
            dismissButton = {
                TextButton(onClick = {
                    cancelDialogState = null
                    viewModel.onEvent(DispatchReceivedUiEvent.OnCancelDialogDismissed)
                }) { Text(dialog.cancelText) }
            }
        )
    }

    appPaymentDialogState?.let { dialog ->
        AlertDialog(
            onDismissRequest = { appPaymentDialogState = null },
            title = { Text(dialog.title) },
            text = { Text(dialog.description) },
            confirmButton = {
                TextButton(onClick = { appPaymentDialogState = null }) {
                    Text(dialog.acceptText)
                }
            }
        )
    }
}

            disabledContentColor = style.contentColor.copy(alpha = 0.4f)
        )
    ) {
        Text(text = style.text)
    }
```

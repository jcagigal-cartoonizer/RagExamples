sealed interface LightsTestUiEffect {

    data class SetCourtesyButtonDrawable(@DrawableRes val drawableRes: Int) : LightsTestUiEffect

fun LightsTestButtonsState.reduce(effect: LightsTestUiEffect): LightsTestButtonsState {
    return when (effect) {
        is LightsTestUiEffect.SetUvButtonDrawable -> copy(

            uvButton = uvButton.copy(enabled = effect.enabled)

        is LightsTestUiEffect.SetCourtesyButtonEnabled -> copy(
            courtesyButton = courtesyButton.copy(enabled = effect.enabled)

        is LightsTestUiEffect.SetBeeperProgress -> this

package ifac.td.taxi.viewmodel

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.R
import ifac.td.taxi.domain.model.UserPreferences
import ifac.td.taxi.domain.usecase.LightSkyGlassUseCase
import ifac.td.taxi.domain.usecase.UserPreferencesUseCase
import ifac.td.taxi.ui.screen.lights.LightsTestUiEffect
import ifac.td.taxi.ui.screen.lights.LightsTestUiEvent
import ifac.td.taxi.ui.screen.lights.LightsTestUiState
import kotlinx.coroutines.delay
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

class LightsTestViewModel(
    private val lightSkyGlassUseCase: LightSkyGlassUseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    application: Application,
) : AndroidViewModel(application) {

    private val _uiState = MutableStateFlow(LightsTestUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<LightsTestUiEffect>(extraBufferCapacity = 64)
    val uiEffect = _uiEffect.asSharedFlow()

    fun onEvent(event: LightsTestUiEvent) {
        when (event) {
            LightsTestUiEvent.ScreenStarted -> getBeeperVolume()

            LightsTestUiEvent.UvLightClicked -> turnUVLight()

            LightsTestUiEvent.CourtesyLightClicked -> turnCourtesyLight()

            is LightsTestUiEvent.BeeperVolumeChanged -> setBeeperVolume(event.volume)
        }
    }

    fun turnUVLight() {
        viewModelScope.launch {
            lightSkyGlassUseCase.turnUVLight()

            _uiEffect.emit(LightsTestUiEffect.SetUvButtonDrawable(R.drawable.luz_ultravioleta_blue))
            _uiEffect.emit(LightsTestUiEffect.SetUvButtonEnabled(false))
            _uiEffect.emit(LightsTestUiEffect.SetCourtesyButtonEnabled(false))

            delay(8000L)

            _uiEffect.emit(LightsTestUiEffect.SetUvButtonDrawable(R.drawable.luz_ultravioleta))
            _uiEffect.emit(LightsTestUiEffect.SetUvButtonEnabled(true))
            _uiEffect.emit(LightsTestUiEffect.SetCourtesyButtonEnabled(true))
        }
    }

    fun turnCourtesyLight() {
        viewModelScope.launch {
            lightSkyGlassUseCase.turnCourtesyLight()
        }
    }

    fun setBeeperVolume(volume: Int?) {
        viewModelScope.launch {
            val safeVolume = volume ?: 50
            lightSkyGlassUseCase.setBeeperVolume(safeVolume)
            userPreferencesUseCase.insertBeeperVolumeValue(safeVolume)
            _uiState.update { it.copy(beeperVolume = safeVolume) }
        }
    }

    fun getBeeperVolume() {
        viewModelScope.launch {
            val preferences = userPreferencesUseCase.getUserPreferences()
            if (preferences != null) {
                _uiState.update { it.copy(beeperVolume = preferences.beeperVolume) }
                _uiEffect.emit(LightsTestUiEffect.SetBeeperProgress(preferences.beeperVolume))
            } else {
                userPreferencesUseCase.insertUserPreferences(UserPreferences())
                _uiState.update { it.copy(beeperVolume = 50) }
                _uiEffect.emit(LightsTestUiEffect.SetBeeperProgress(50))
            }
        }
    }

    fun onCourtesyLightStateChanged(isOn: Boolean?) {
        viewModelScope.launch {
            if (isOn == true) {
                _uiEffect.emit(LightsTestUiEffect.SetCourtesyButtonDrawable(R.drawable.lumact_yellow))
                _uiEffect.emit(LightsTestUiEffect.SetUvButtonEnabled(false))
            } else {
                _uiEffect.emit(LightsTestUiEffect.SetCourtesyButtonDrawable(R.drawable.lumact))
                _uiEffect.emit(LightsTestUiEffect.SetUvButtonEnabled(true))
            }
        }
    }

    fun onUvLightFinished() {
        viewModelScope.launch {
            _uiEffect.emit(LightsTestUiEffect.SetUvButtonDrawable(R.drawable.luz_ultravioleta))
            _uiEffect.emit(LightsTestUiEffect.SetUvButtonEnabled(true))
            _uiEffect.emit(LightsTestUiEffect.SetCourtesyButtonEnabled(true))
        }
    }
}

## 5) Exact reducer mirroring the fragment branches

Here is the exact one-by-one mapping from the fragment’s `collect {}` branches:

### UV light collector
Fragment branch:
```kotlin
if (!it) {
    ibUVLight.setImageDrawable(luz_ultravioleta)
    ibUVLight.isEnabled = true
    ibCourtesyLight.isEnabled = true
}
```

Compose equivalent effects:
- `SetUvButtonDrawable(R.drawable.luz_ultravioleta)`
- `SetUvButtonEnabled(true)`
- `SetCourtesyButtonEnabled(true)`

### Courtesy light collector
Fragment branch:
```kotlin
if (isOn == true) {
    ibCourtesyLight.setImageDrawable(lumact_yellow)
    ibUVLight.isEnabled = false
} else {
    ibCourtesyLight.setImageDrawable(lumact)
    ibUVLight.isEnabled = true

```

Compose equivalent effects:
- `SetCourtesyButtonDrawable(R.drawable.lumact_yellow)`
- `SetUvButtonEnabled(false)`
- or
- `SetCourtesyButtonDrawable(R.drawable.lumact)`
- `SetUvButtonEnabled(true)`

### Beeper collector
Fragment branch:
```kotlin
sbBIPVolume.progress = beeperValue ?: 50
```

Compose equivalent:
- `SetBeeperProgress(beeperValue ?: 50)`

```kotlin
import ifac.td.taxi.ui.custom.dialog.CustomDialog

@Composable
fun ChooseOptionScreen(
    uiState: ChooseOptionUiState,
    onEvent: (ChooseOptionUiEvent) -> Unit,
) {
    val context = LocalContext.current

    // Preserve the fragment behavior: show bottom bar/header and mark screen active
    LaunchedEffect(Unit) {
        onEvent(ChooseOptionUiEvent.ScreenStarted)
    }

    Box(modifier = Modifier.fillMaxSize()) {
        LazyColumn(modifier = Modifier.fillMaxSize()) {
            itemsIndexed(uiState.options) { _, option ->
                ChooseOptionOptionRow(
                    option = option,
                    onClick = { onEvent(ChooseOptionUiEvent.OptionClicked(option)) }
                )
            }
        }

        if (uiState.dialogState is ChooseOptionDialogState.Visible) {
            val dialog = uiState.dialogState as ChooseOptionDialogState.Visible

            AlertDialog(
                onDismissRequest = { onEvent(ChooseOptionUiEvent.DialogCancelled) },
                title = { Text(text = dialog.title) },
                text = { Text(text = dialog.description) },
                confirmButton = {
                    TextButton(onClick = { onEvent(ChooseOptionUiEvent.DialogAccepted(dialog.option)) }) {
                        Text(text = stringResource(R.string.btn_accept).uppercase())
                    }
                },
                dismissButton = {
                    TextButton(onClick = { onEvent(ChooseOptionUiEvent.DialogCancelled) }) {
                        Text(text = stringResource(R.string.btn_cancel).uppercase())
                    }
                }
            )
        }
    }
}

@Composable
private fun ChooseOptionOptionRow(
    option: ChooseOptionOptionItem,
    onClick: () -> Unit
) {
    // If you already have a Compose wrapper for CustomButton, use it here.
    // Otherwise this shows the intended mapping from the XML RecyclerView item.
    CustomButtonCompose(
        text = option.label,
        buttonType = option.buttonType,
        enabled = option.enabled,
        onClick = onClick
```

## 2) Compose wrapper for the existing `CustomButton` API

Since your project already has a very specific `CustomButton` view API, the cleanest Compose replacement is a wrapper that preserves the same visual behavior.

```kotlin
package ifac.td.taxi.ui.screen

import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.viewinterop.AndroidView
import ifac.td.taxi.ui.custom.button.ButtonType
import ifac.td.taxi.ui.custom.button.CustomButton
import ifac.td.taxi.ui.custom.button.CustomButton.StyleButton

@Composable
fun CustomButtonCompose(
    text: String,
    buttonType: ButtonType,
    enabled: Boolean,
    onClick: () -> Unit,
    modifier: Modifier = Modifier
) {
    AndroidView(
        modifier = modifier,
        factory = { context ->
            CustomButton(context).apply {
                setButtonType(buttonType.value)
                setButtonStyle(if (enabled) StyleButton.ENABLE else StyleButton.DISABLE)
                changeText(text)
                setAction(onClick)
            }
        },
        update = { view ->
            view.setButtonType(buttonType.value)
            view.setButtonStyle(if (enabled) StyleButton.ENABLE else StyleButton.DISABLE)
            view.changeText(text)
            view.setAction(onClick)
        }
    )
}
```

## 4) Compose-friendly ViewModel with `UiState + UiEvent + SharedFlow<UiEffect>`

This replaces the fragment’s imperative callback flow.

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import androidx.lifecycle.viewModelScope
import com.interfacom.sdk.taximeter.bravocomm.W2CLocation
import com.interfacom.sdk.taximeter.bravocomm.ifConstants
import ifac.td.taxi.domain.usecase.EndShiftUseCase
import ifac.td.taxi.domain.usecase.SessionUseCase
import ifac.td.taxi.domain.usecase.ShiftStatusUseCase
import ifac.td.taxi.framework.sdk.bravocentral.usecase.BravoCentralUseCase
import ifac.td.taxi.framework.sdk.bravocentral.usecase.LoginDriverUseCase
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.ui.screen.ChooseOptionDialogState
import ifac.td.taxi.ui.screen.ChooseOptionOptionItem
import ifac.td.taxi.ui.screen.ChooseOptionUiEffect
import ifac.td.taxi.ui.screen.ChooseOptionUiEvent
import ifac.td.taxi.ui.screen.ChooseOptionUiState
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

class ChooseOptionViewModel(
    context: Application,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val loginDriverUseCase: LoginDriverUseCase,
    private val sessionUseCase: SessionUseCase,
    private val bravoCentralUseCase: BravoCentralUseCase,
    private val endShiftUseCase: EndShiftUseCase,
) : BaseViewModel(context) {

    private val TAG = "ChooseOptionViewModel"

    private val _uiState = MutableStateFlow(ChooseOptionUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<ChooseOptionUiEffect>()
    val uiEffect: SharedFlow<ChooseOptionUiEffect> = _uiEffect.asSharedFlow()

    fun onEvent(event: ChooseOptionUiEvent) {
        when (event) {
            ChooseOptionUiEvent.ScreenStarted -> {
                _uiEffect.tryEmit(ChooseOptionUiEffect.ShowSystemBars)
                _uiEffect.tryEmit(ChooseOptionUiEffect.SetChooseOptionActive(true))
            }

            is ChooseOptionUiEvent.OptionsLoaded -> {
                _uiState.update { it.copy(options = event.options) }
            }

            is ChooseOptionUiEvent.OptionClicked -> {
                _uiState.update {
                    it.copy(
                        dialogState = ChooseOptionDialogState.Visible(
                            title = getApplication<Application>().getString(ifac.td.taxi.R.string.warning),
                            description = getApplication<Application>().getString(
                                ifac.td.taxi.R.string.dialog_send_option_confirm,
                                event.option.label
                            ),
                            option = event.option
                        )
                    )
                }
            }

            is ChooseOptionUiEvent.DialogAccepted -> {
                _uiState.update { it.copy(dialogState = ChooseOptionDialogState.Hidden) }
                viewModelScope.launch {
                    sendShiftSelectedAnswer(event.option.id)
                    _uiEffect.emit(ChooseOptionUiEffect.SetChooseOptionActive(false))
                    loginDriver()
                    _uiEffect.emit(ChooseOptionUiEffect.NavigateBack)
                }
            }

            ChooseOptionUiEvent.DialogCancelled -> {
                _uiState.update { it.copy(dialogState = ChooseOptionDialogState.Hidden) }
            }

            ChooseOptionUiEvent.BackPressed -> {
                viewModelScope.launch {
                    logoffDriver()
                    _uiEffect.emit(ChooseOptionUiEffect.SetChooseOptionActive(false))
                    _uiEffect.emit(ChooseOptionUiEffect.NavigateBack)
                }
            }
        }
    }

    fun sendShiftSelectedAnswer(selectedId: String) {
        viewModelScope.launch {
            Logs.d(TAG, "sendShiftSelectedAnswer: selectedId = $selectedId")
            shiftStatusUseCase.sendShiftSelectedAnswer(
                selectedId,
                W2CLocation.getZoning()?.version ?: "00"
            )
        }
    }

    fun loginDriver() {
        viewModelScope.launch {
            sessionUseCase.getSession()?.let {
                it.numDriver?.let { numDriver ->
                    loginDriverUseCase.loginDriver(numDriver, it.pwdDriver ?: "", it.reinforcement)
                }
            }
        }
    }

    fun logoffDriver() {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "logoffDriver: cancelling shift selection")
            endShiftUseCase.endShift()
            bravoCentralUseCase.logoff()
            shiftStatusUseCase.setStatus(ifConstants.STATE_DISCONNECTED, false)
        }
    }
}
```

## 7) Compose screen integration with lifecycle collection

This is the part that preserves the fragment navigation behavior exactly.

```kotlin
package ifac.td.taxi.ui.screen

import androidx.compose.runtime.Composable
import androidx.compose.runtime.LaunchedEffect
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.lifecycle.viewmodel.compose.viewModel
import ifac.td.taxi.viewmodel.ChooseOptionMainActivityViewModel
import ifac.td.taxi.viewmodel.ChooseOptionViewModel

@Composable
fun ChooseOptionRoute(
    chooseOptionViewModel: ChooseOptionViewModel = androidx.lifecycle.viewmodel.compose.viewModel(),
    mainActivityViewModel: ChooseOptionMainActivityViewModel = viewModel(),
    onNavigateBack: () -> Unit,
    onSetChooseOptionActive: (Boolean) -> Unit,
    onShowBottomBar: (Boolean) -> Unit,
    onShowHeader: (Boolean) -> Unit,
) {
    val uiState = chooseOptionViewModel.uiState.collectAsStateWithLifecycle().value

    LaunchedEffect(Unit) {
        chooseOptionViewModel.onEvent(ChooseOptionUiEvent.ScreenStarted)
        onShowBottomBar(true)
        onShowHeader(true)
        onSetChooseOptionActive(true)
    }

    LaunchedEffect(Unit) {
        chooseOptionViewModel.uiEffect.collect { effect ->
            when (effect) {
                ChooseOptionUiEffect.ShowSystemBars -> Unit
                is ChooseOptionUiEffect.SetChooseOptionActive -> onSetChooseOptionActive(effect.active)
                ChooseOptionUiEffect.NavigateBack -> onNavigateBack()
            }
            mainActivityViewModel.onChooseOptionEffect(effect)
        }
    }

    ChooseOptionScreen(
        uiState = uiState,
        onEvent = chooseOptionViewModel::onEvent
    )
}
```

## 9) Notes on dialog handling

You asked to use dialog state and lifecycle collection only if the ViewModel uses a `CustomDialog`.

Here, the fragment used `CustomDialog`, so the Compose equivalent uses:
- `ChooseOptionDialogState.Visible`
- `AlertDialog` in Compose
- lifecycle-aware collection via `collectAsStateWithLifecycle`

That preserves the same user flow:
1. click option
2. show confirmation dialog
3. accept -> send answer -> login driver -> navigate back
4. cancel -> close dialog only

## 10) Important mapping from the fragment

The original fragment logic is preserved as:

- `setupComponents()`  
  → `ChooseOptionRoute` `LaunchedEffect(Unit)`

- `updateBackButton()`  
  → `ChooseOptionUiEvent.BackPressed`

- `updateTopBarIcon()`  
  → handled by the Activity callback passed into the route

- `openDialog(...)`  
  → `ChooseOptionDialogState.Visible` + `AlertDialog`

- `ButtonType.ACCEPT` branch  
  → `ChooseOptionUiEvent.DialogAccepted`

- `ButtonType.CANCEL` / else branch  
  → `ChooseOptionUiEvent.DialogCancelled`

```kotlin
                _uiState.update { it.copy(dialogState = ChooseOptionDialogState.Hidden) }
            }

            ChooseOptionUiEvent.BackPressed -> {
                viewModelScope.launch {
                    logoffDriver()
                    _uiEffect.emit(ChooseOptionUiEffect.SetChooseOptionActive(false))
                    _uiEffect.emit(ChooseOptionUiEffect.NavigateBack)
                }
            }
        }
    }

    fun sendShiftSelectedAnswer(selectedId: String) {
        viewModelScope.launch {
            Logs.d(TAG, "sendShiftSelectedAnswer: selectedId = $selectedId")
            shiftStatusUseCase.sendShiftSelectedAnswer(
                selectedId,
                com.interfacom.sdk.taximeter.bravocomm.W2CLocation.getZoning()?.version ?: "00"
            )
        }
    }

    fun loginDriver() {
        viewModelScope.launch {
            sessionUseCase.getSession()?.let {
                it.numDriver?.let { numDriver ->
                    loginDriverUseCase.loginDriver(numDriver, it.pwdDriver ?: "", it.reinforcement)
                }
            }
        }
    }

    fun logoffDriver() {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "logoffDriver: cancelling shift selection")
            endShiftUseCase.endShift()
            bravoCentralUseCase.logoff()
            shiftStatusUseCase.setStatus(com.interfacom.sdk.taximeter.bravocomm.ifConstants.STATE_DISCONNECTED, false)
        }
    }
}

    onSetChooseOptionActive: (Boolean) -> Unit,
    onShowBottomBar: (Boolean) -> Unit,

) {

        onEvent(ChooseOptionUiEvent.ScreenStarted)
        onShowBottomBar(true)
        onShowHeader(true)
        onSetChooseOptionActive(true)
    }

    LaunchedEffect(Unit) {
        // This is the lifecycle collection bridge for effects.
        // In a real screen, collect from the ViewModel's SharedFlow here.
    }

    ChooseOptionScreen(
        uiState = uiState,
        onEvent = onEvent
    )
}

@Composable
fun ChooseOptionScreen(
    uiState: ChooseOptionUiState,
    onEvent: (ChooseOptionUiEvent) -> Unit,
) {
    Box(modifier = Modifier.fillMaxSize()) {
        LazyColumn(modifier = Modifier.fillMaxSize()) {
            itemsIndexed(uiState.options) { _, option ->
                ChooseOptionOptionRow(
                    option = option,
                    onClick = { onEvent(ChooseOptionUiEvent.OptionClicked(option)) }
                )
            }
        }

        when (val dialog = uiState.dialogState) {
            ChooseOptionDialogState.Hidden -> Unit
            is ChooseOptionDialogState.Visible -> {
                AlertDialog(
                    onDismissRequest = { onEvent(ChooseOptionUiEvent.DialogCancelled) },
                    title = { Text(text = dialog.title) },
                    text = { Text(text = dialog.description) },
                    confirmButton = {
                        TextButton(onClick = { onEvent(ChooseOptionUiEvent.DialogAccepted(dialog.option)) }) {
                            Text(text = stringResource(R.string.btn_accept).uppercase())
                        }
                    },
                    dismissButton = {
                        TextButton(onClick = { onEvent(ChooseOptionUiEvent.DialogCancelled) }) {
                            Text(text = stringResource(R.string.btn_cancel).uppercase())
                        }
                    }
                )
            }
        }
    }
}

@Composable
private fun ChooseOptionOptionRow(
    option: ChooseOptionOptionItem,
    onClick: () -> Unit
) {
    val visual = remember(option.buttonType) { chooseOptionButtonsReducer(option.buttonType) }

    Surface(
        modifier = Modifier
            .fillMaxWidth()
            .padding(horizontal = 12.dp, vertical = 6.dp)
            .clickable(enabled = option.enabled, onClick = onClick),
        tonalElevation = 2.dp
    ) {
        Column(modifier = Modifier.padding(12.dp)) {
            if (visual.textVisible) {
                Text(
                    text = option.label.ifBlank { visual.text },
                    style = MaterialTheme.typography.titleMedium
                )
            }
            if (visual.iconVisible) {
                Text(
                    text = "icon:${visual.iconRes}",
                    style = MaterialTheme.typography.bodySmall
                )
            }
            Text(
                text = "bg=${visual.backgroundColor}, style=${visual.style}",
                style = MaterialTheme.typography.bodySmall
            )
        }
    }
}

@Composable
fun CustomButtonCompose(
    text: String,
    buttonType: ButtonType,
    enabled: Boolean,
    onClick: () -> Unit,
    modifier: Modifier = Modifier
) {
    androidx.compose.ui.viewinterop.AndroidView(
        modifier = modifier,
        factory = { context ->
            CustomButton(context).apply {
                setButtonType(buttonType.value)
                setButtonStyle(if (enabled) StyleButton.ENABLE else StyleButton.DISABLE)
                changeText(text)
                setAction(onClick)
            }
        },
        update = { view ->
            view.setButtonType(buttonType.value)
            view.setButtonStyle(if (enabled) StyleButton.ENABLE else StyleButton.DISABLE)
            view.changeText(text)
            view.setAction(onClick)
        }
    )
}
```


// FIRST ANSWER
Below is a Compose-first rewrite that keeps the same behavior as your `OnTripFragment`, but replaces the fragment button logic with:

- a single `OnTripUiState`
- a single `OnTripUiEffect` `SharedFlow`
- an exact `OnTripButtonsState` reducer that mirrors the fragment `collect {}` branches one-by-one
- a Compose screen that preserves the same navigation/actions
- a Compose bridge for the shared `MainActivityViewModel` behavior
- dialog state handling only where the original code used `CustomDialog`

I’m keeping the code focused on the `OnTrip` feature only.

## 2) Exact `OnTripButtonsState` reducer mirroring the fragment branches

This reducer mirrors the fragment `collect {}` logic branch-by-branch.

```kotlin
package ifac.td.taxi.ui.screen.ontrip.compose

import com.interfacom.sdk.taximeter.bravocomm.ifConstants
import ifac.td.taxi.R
import ifac.td.taxi.ui.custom.button.ButtonType
import ifac.td.taxi.ui.custom.button.CustomButton
import ifac.td.taxi.viewmodel.model.MessageUIEnum

fun reduceOnTripButtonsState(
    current: OnTripButtonsState,
    ui: OnTripUiState
): OnTripButtonsState {
    var state = current

    // 1) showEstimateFixedPriceButtonFlow
    state = state.copy(
        fixedPrice = state.fixedPrice.copy(
            visible = ui.showEstimateFixedPriceButton
        ),
        receipts = state.receipts.copy(
            visible = !ui.showEstimateFixedPriceButton
        )
    )

    // 2) notifyLocationOnHired -> handled in effect/state elsewhere, no button mutation here

    // 3) hasTaximeterConnectionFlow -> updateTopBarIconRight() handled by bridge, not button state

    // 4) hiredZone.collect
    state = state.copy(
        zoning = when {
            ui.hiredZoneExists -> state.zoning.copy(
                textRes = R.string.btn_delocate,
                iconRes = R.drawable.ubdesact,
                background = CustomButton.BackgroundButtonColor.RED,
                buttonType = ButtonType.UBICACION,
                style = CustomButton.StyleButton.ENABLE,
                enabled = true,
                visible = true
            )
            ui.isLocationAllowedByCentral -> state.zoning.copy(
                textRes = R.string.btn_soon_to_clear,
                iconRes = R.drawable.ubactivar,
                background = CustomButton.BackgroundButtonColor.GREEN,
                buttonType = ButtonType.SOON_TO_CLEAR,
                style = CustomButton.StyleButton.ENABLE,
                enabled = true,
                visible = true
            )
            else -> state.zoning.copy(
                buttonType = ButtonType.EMPTY,
                style = CustomButton.StyleButton.ENABLE,
                background = CustomButton.BackgroundButtonColor.BLACK,
                visible = true,
                enabled = true
            )
        }
    )

    // 5) noClientButtonFlow
    state = state.copy(
        client = if (ui.noClientButtonEnabled == false && ui.shiftCurrentStatus == ifConstants.STATE_HIRED_DISPATCHED) {
            state.client.copy(style = CustomButton.StyleButton.DISABLE, enabled = false)
        } else state.client
    )

    // 6) shiftStatusFlow
    state = state.copy(
        navigate = state.navigate.copy(
            visible = ui.shiftCurrentStatus?.let { ui.isManual || ui.isHiredStatus(it) } == true
        ),
        roofLight = state.roofLight.copy(
            visible = ui.shiftCurrentStatus?.let { !ui.isHiredStatus(it) } == true
        )
    )

    // 7) tripFlow
    state = state.copy(
        dispatchInfo = when {
            ui.tripFromDispatch == true -> state.dispatchInfo.copy(
                style = CustomButton.StyleButton.ENABLE,
                enabled = true
            )
            ui.tripFromDispatch == false && ui.isLocationAllowedByCentral -> state.dispatchInfo.copy(
                style = CustomButton.StyleButton.DISABLE,
                enabled = false
            )
            ui.tripFromDispatch == false && !ui.isLocationAllowedByCentral -> state.dispatchInfo.copy(
                buttonType = ButtonType.EMPTY,
                style = CustomButton.StyleButton.ENABLE,
                background = CustomButton.BackgroundButtonColor.BLACK
            )
            else -> state.dispatchInfo
        }
    )

    // 8) roofLightFlow
    state = state.copy(
        roofLight = when {
            ui.roofLightDisableLuminous != true -> state.roofLight.copy(
                style = CustomButton.StyleButton.DISABLE,
                enabled = false
            )
            ui.roofLightState == true -> state.roofLight.copy(
                style = CustomButton.StyleButton.ENABLE,
                enabled = true,
                background = CustomButton.BackgroundButtonColor.GREEN
            )
            ui.roofLightState == false -> state.roofLight.copy(
                style = CustomButton.StyleButton.ENABLE,
                enabled = true,
                background = CustomButton.BackgroundButtonColor.RED
            )
            else -> state.roofLight.copy(style = CustomButton.StyleButton.DISABLE, enabled = false)
        }
    )

    // 9) dispatchFlow
    state = state.copy(
        notifications = when {
            ui.isITopConnected && !ui.shiftIsManual -> state.notifications.copy(
                style = CustomButton.StyleButton.DISABLE,
                enabled = false
            )
            ui.dispatchAtDoorEnabled && !ui.dispatchAtDoorSent &&
                    ui.dispatchRiderInCab && !ui.dispatchRiderInCabSent -> state.notifications.copy(
                style = CustomButton.StyleButton.ENABLE,
                enabled = true
            )
            !ui.dispatchAtDoorEnabled && ui.dispatchRiderInCab && !ui.dispatchRiderInCabSent -> state.notifications.copy(
                buttonType = ButtonType.RIDER_IN_CAB,
                style = CustomButton.StyleButton.ENABLE,
                enabled = true,
                background = CustomButton.BackgroundButtonColor.ORANGE
            )
            ui.dispatchAtDoorEnabled && !ui.dispatchRiderInCabSent -> state.notifications.copy(
                buttonType = ButtonType.AT_DOOR,
                style = CustomButton.StyleButton.ENABLE,
                enabled = true,
                background = CustomButton.BackgroundButtonColor.GREEN
            )
            else -> state.notifications.copy(
                style = CustomButton.StyleButton.DISABLE,
                enabled = false
            )
        }
    )

    // 10) bravoStateFlow
    state = state.copy(
        client = if (ui.isITopConnected) {
            state.client.copy(style = CustomButton.StyleButton.DISABLE, enabled = false)
        } else {
            state.client.copy(
                background = if (ui.bravoClientButton) {
                    CustomButton.BackgroundButtonColor.GREEN
                } else {
                    CustomButton.BackgroundButtonColor.RED
                },
                style = if (ui.bravoNoClientButtonEnabled) {
                    CustomButton.StyleButton.ENABLE
                } else {
                    CustomButton.StyleButton.DISABLE
                },
                enabled = ui.bravoNoClientButtonEnabled
            )
        }
    )

    // 11) hasMessagesFlow
    state = state.copy(
        messages = when (ui.hasMessages) {
            MessageUIEnum.NO_MESSAGES -> state.messages.copy(
                style = CustomButton.StyleButton.DISABLE,
                enabled = false
            )
            MessageUIEnum.HAS_MESSAGES_BUT_NOT_NEW -> state.messages.copy(
                style = CustomButton.StyleButton.ENABLE,
                enabled = true,
                background = CustomButton.BackgroundButtonColor.BLUE
            )
            MessageUIEnum.HAS_NEW_MESSAGES -> state.messages.copy(
                style = CustomButton.StyleButton.ENABLE,
                enabled = true,
                background = CustomButton.BackgroundButtonColor.ORANGE
            )
            null -> state.messages
        }
    )

    return state
}

private fun OnTripUiState.isHiredStatus(status: Int): Boolean {
    // delegate to your existing shiftStatusUseCase.isHired(status) in VM
    return status != ifConstants.STATE_DISPATCHED
}
```

## 3) Compose-friendly `OnTripViewModel` with `UiState + UiEvent + UiEffect`

This keeps your existing domain/usecase calls, but exposes a single state and a single effect stream.

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import android.content.Intent
import androidx.core.net.toUri
import androidx.lifecycle.viewModelScope
import androidx.navigation.NavDeepLinkRequest
import com.interfacom.sdk.taximeter.bravocomm.W2CLocation
import com.interfacom.sdk.taximeter.bravocomm.ifConstants
import ifac.td.taxi.R
import ifac.td.taxi.domain.usecase.BluetoothLocalUseCase
import ifac.td.taxi.domain.usecase.DispatchNotificationUseCase
import ifac.td.taxi.domain.usecase.LocationUseCase
import ifac.td.taxi.domain.usecase.MessageUseCase
import ifac.td.taxi.domain.usecase.RoofLightUseCase
import ifac.td.taxi.domain.usecase.ShiftStatusUseCase
import ifac.td.taxi.domain.usecase.UserPreferencesUseCase
import ifac.td.taxi.framework.sdk.bravocentral.usecase.BravoCentralConfigurationUseCase
import ifac.td.taxi.framework.sdk.usecase.LicensingUseCase
import ifac.td.taxi.framework.sdk.usecase.NavigatorUseCase
import ifac.td.taxi.framework.sdk.usecase.ZoningUseCase
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.ui.custom.button.ButtonType
import ifac.td.taxi.ui.custom.dialog.CustomDialog
import ifac.td.taxi.ui.screen.OnTripFragmentDirections
import ifac.td.taxi.ui.screen.ontrip.compose.*
import ifac.td.taxi.viewmodel.model.FilterOptions
import ifac.td.taxi.viewmodel.model.InfoDispatchModel
import ifac.td.taxi.viewmodel.model.LocationInHiredEnum
import ifac.td.taxi.viewmodel.model.MessageUIEnum
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch

class OnTripViewModel(
    private val messageUseCase: MessageUseCase,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val dispatchNotificationUseCase: DispatchNotificationUseCase,
    private val navigatorUseCase: NavigatorUseCase,
    private val locationUseCase: LocationUseCase,
    private val licensingUseCase: LicensingUseCase,
    private val bravoCentralConfigurationUseCase: BravoCentralConfigurationUseCase,
    private val bluetoothLocalUseCase: BluetoothLocalUseCase,
    private val roofLightUseCase: RoofLightUseCase,
    private val zoningUseCase: ZoningUseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    context: Application,
) : BaseViewModel(context) {

    private val _uiState = MutableStateFlow(OnTripUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<OnTripUiEffect>(extraBufferCapacity = 1)
    val uiEffect = _uiEffect.asSharedFlow()

    init {
        viewModelScope.launch(Dispatchers.IO) {
            val shiftStatus = shiftStatusUseCase.getStatus()
            val isManual = shiftStatus?.isManual == true && shiftStatus.currentStatus != ifConstants.STATE_DISPATCHED
            _uiState.update { it.copy(isManual = isManual) }
        }

        viewModelScope.launch {
            val url = bravoCentralConfigurationUseCase.getBravoConfigurationVariable()?.fixedPricesStreetTripsURL
            _uiState.update { it.copy(showEstimateFixedPriceButton = url?.isNotBlank() == true) }
        }
    }

    fun onEvent(event: OnTripUiEvent) {
        when (event) {
            OnTripUiEvent.ScreenStarted -> {
                getMessages()
                checkNoClientButton()
            }
            OnTripUiEvent.FixedPriceClicked -> navigateToEstimateFixedPriceScreen()
            OnTripUiEvent.ReceiptsClicked -> navigateToReceiptHistory()
            OnTripUiEvent.MessagesClicked -> navigateToMessage()
            OnTripUiEvent.CentralClicked -> navigateToContactCentral()
            OnTripUiEvent.NavigateClicked -> openNavigatorApp()
            OnTripUiEvent.DispatchInfoClicked -> {
                _uiEffect.tryEmit(OnTripUiEffect.NavigateTo(R.id.action_onTripFragment_to_infoDispatchFragment))
            }
            OnTripUiEvent.ZoningClicked -> {
                val locatedOnHired = _uiState.value.hiredZoneExists
                actionZoningFragment(locatedOnHired)
            }
            OnTripUiEvent.ClientClicked -> handleClientClicked()
            OnTripUiEvent.NotificationsClicked -> handleNotificationsClicked()
            OnTripUiEvent.RoofLightClicked -> handleRoofLightClicked()
            OnTripUiEvent.TopBarRightClicked -> handleTopBarRightClicked()
            OnTripUiEvent.DialogResult -> Unit
        }
    }

    fun getMessages() {
        viewModelScope.launch(Dispatchers.IO) {
            val messages = messageUseCase.getMessageByDriverId()
            val result = if (messages.isNullOrEmpty()) {
                MessageUIEnum.NO_MESSAGES
            } else if (messages.any { !it.isUrgent && !it.isRead }) {
                MessageUIEnum.HAS_NEW_MESSAGES
            } else {
                MessageUIEnum.HAS_MESSAGES_BUT_NOT_NEW
            }
            _uiState.update { it.copy(hasMessages = result) }
        }
    }

    fun checkNoClientButton() {
        viewModelScope.launch(Dispatchers.IO) {
            val parameters = licensingUseCase.getLicensingParameters()
            _uiState.update { it.copy(noClientButtonEnabled = parameters?.isNoClientHired) }
        }
    }

    fun checkRoofLight(value: Boolean?) {
        viewModelScope.launch {
            val isDisableLuminous = licensingUseCase.getLicensingParameters()?.isDisableLuminous
            if (roofLightUseCase.isRoofLightAvailable()) {
                _uiState.update { it.copy(roofLightState = value, roofLightDisableLuminous = isDisableLuminous) }
            } else {
                _uiState.update { it.copy(roofLightState = null, roofLightDisableLuminous = null) }
            }
        }
    }

    fun onSharedStateChanged(partial: OnTripSharedState) {
        _uiState.update {
            it.copy(
                isLocationAllowedByCentral = partial.isLocationAllowedByCentral,
                hasTaximeterConnection = partial.hasTaximeterConnection,
                hiredZoneExists = partial.hiredZoneExists,
                shiftCurrentStatus = partial.shiftCurrentStatus,
                shiftIsManual = partial.shiftIsManual,
                tripFromDispatch = partial.tripFromDispatch,
                dispatchAtDoorEnabled = partial.dispatchAtDoorEnabled,
                dispatchAtDoorSent = partial.dispatchAtDoorSent,
                dispatchRiderInCab = partial.dispatchRiderInCab,
                dispatchRiderInCabSent = partial.dispatchRiderInCabSent,
                bravoClientButton = partial.bravoClientButton,
                bravoNoClientButtonEnabled = partial.bravoNoClientButtonEnabled,
                isITopConnected = partial.isITopConnected,
                canGoToHiredManual = partial.canGoToHiredManual
            )
        }
    }

    fun reduceButtons() {
        _uiState.update { state ->
            state.copy(buttons = reduceOnTripButtonsState(state.buttons, state))
        }
    }

    fun navigateToMessage() = viewModelScope.launch {
        _uiEffect.emit(OnTripUiEffect.NavigateTo(R.id.action_onTripFragment_to_messageFragment))
    }

    fun navigateToReceiptHistory() = viewModelScope.launch {
        _uiEffect.emit(OnTripUiEffect.NavigateTo(OnTripFragmentDirections.actionOnTripFragmentToReceiptHistoryFragment(-1L).actionId))
    }

    fun navigateToContactCentral() = viewModelScope.launch {
        _uiEffect.emit(OnTripUiEffect.NavigateTo(R.id.action_onTripFragment_to_contactCentralFragment))
    }

    fun navigateToEstimateFixedPriceScreen() = viewModelScope.launch {
        _uiEffect.emit(OnTripUiEffect.NavigateTo(R.id.action_onTripFragment_to_FixedPriceMapFragment))
    }

    fun openNavigatorApp() {
        viewModelScope.launch {
            navigatorUseCase.openNavigatorApp()?.let { intent ->
                _uiEffect.emit(OnTripUiEffect.LaunchIntent(intent))
            }
        }
    }

    fun sendAtTheDoorNotification(dispatch: InfoDispatchModel?) = viewModelScope.launch {
        dispatch?.let { dispatchNotificationUseCase.atTheDoor(it.id) }
    }

    fun sendInCabNotification(dispatch: InfoDispatchModel?) = viewModelScope.launch {
        dispatch?.let { dispatchNotificationUseCase.inCab(it.id) }
    }

    fun sendNoClientNotification(infoDispatch: Long?) = viewModelScope.launch {
        infoDispatch?.let { dispatchNotificationUseCase.noClient(it) }
    }

    fun setRoofLightOff() = viewModelScope.launch { roofLightUseCase.setRoofLightOff() }
    fun setRoofLightOn() = viewModelScope.launch { roofLightUseCase.setRoofLightOn() }

    fun isHired(value: Int?): Boolean = value?.let { shiftStatusUseCase.isHired(it) } == true

    fun canGoToHiredManual(taximeterConnected: Boolean): Boolean =
        licensingUseCase.canDoManualTrips(taximeterConnected)

    fun hasITopTaximeterConnected(): Boolean = bluetoothLocalUseCase.isCurrentBluetoothItop()

    fun actionZoningFragment(locatedOnHired: Boolean) {
        viewModelScope.launch {
            licensingUseCase.getLicensingParameters()?.locationInHired?.let {
                if (it == LocationInHiredEnum.WITHOUT_ZONE_OPTION.value) {
                    if (!locatedOnHired) {
                        showDialogWithOutZone()
                    } else {
                        _uiState.update { s -> s.copy(notifyLocationOnHired = false) }
                        sendPositionStatic()
                    }
                } else {
                    openZoningScreen()
                }
            }
        }
    }

    private suspend fun openZoningScreen() {
        val defaultToMacrozone = userPreferencesUseCase.getUserPreferences()?.macroZoneQuery
        val zoneFilter = zoningUseCase.getZoneConfiguration().zoneFilter
        val filterNavigatesToZoning = zoneFilter == FilterOptions.NEARNESS ||
                zoneFilter == FilterOptions.ID_NO_HIERARCHY ||
                zoneFilter == FilterOptions.HOT_ZONES ||
                zoneFilter == FilterOptions.FAVOURITES

        if (defaultToMacrozone == true && !filterNavigatesToZoning) {
            _uiEffect.emit(OnTripUiEffect.NavigateTo(R.id.action_onTripFragment_to_macroZoning))
        } else if (W2CLocation.getLastIdZone() > 0 || filterNavigatesToZoning) {
            val idMacroZone = if (W2CLocation.getLastIdMacrozone() != 0) W2CLocation.getLastIdMacrozone() else 1
            val navDeepLink = NavDeepLinkRequest.Builder
                .fromUri("android-app://ifac.td.taxi/zoningFragment/$idMacroZone".toUri())
                .build()
            _uiEffect.emit(OnTripUiEffect.NavigateDeepLink(navDeepLink.uri.toString()))
        } else {
            _uiEffect.emit(OnTripUiEffect.NavigateTo(R.id.action_onTripFragment_to_macroZoning))
        }
    }

    private suspend fun showDialogWithOutZone() {
        _uiEffect.emit(
            OnTripUiEffect.OpenDialog(
                OnTripDialogState(
                    titleRes = R.string.libre_en_breve,
                    buttons = listOf(ButtonType.ACCEPT, ButtonType.CANCEL)
                )
            )
        )
    }

    fun onDialogResult(button: ButtonType) {
        if (button == ButtonType.ACCEPT) {
            viewModelScope.launch {
                _uiState.update { it.copy(notifyLocationOnHired = true) }
                sendPositionStatic()
            }
        }
        viewModelScope.launch { _uiEffect.emit(OnTripUiEffect.CloseDialog) }
    }

    private fun sendPositionStatic() {
        W2CLocation.sendEvent('z', null)
    }

    private fun handleClientClicked() {
        if (hasITopTaximeterConnected() && !_uiState.value.shiftIsManual) {
            _uiEffect.tryEmit(OnTripUiEffect.ShowToast(R.string.itop_functionality_restricted))
            return
        }
        _uiEffect.tryEmit(
            OnTripUiEffect.OpenDialog(
                OnTripDialogState(
                    titleRes = R.string.confirm_no_client,
                    buttons = listOf(ButtonType.CANCEL, ButtonType.ACCEPT)
                )
            )
        )
    }

    private fun handleNotificationsClicked() {
        if (hasITopTaximeterConnected() && !_uiState.value.shiftIsManual) {
            _uiEffect.tryEmit(OnTripUiEffect.ShowToast(R.string.itop_functionality_restricted))
            return
        }

        val buttons = buildList {
            if (_uiState.value.dispatchAtDoorEnabled && !_uiState.value.dispatchAtDoorSent) add(ButtonType.AT_DOOR)
            if (_uiState.value.dispatchRiderInCab && !_uiState.value.dispatchRiderInCabSent) add(ButtonType.RIDER_IN_CAB)
        }

        _uiEffect.tryEmit(
            OnTripUiEffect.OpenDialog(
                OnTripDialogState(
                    titleRes = R.string.select_notification,
                    buttons = buttons
                )
            )
        )
    }

    private fun handleRoofLightClicked() {
        val state = _uiState.value
        when (state.roofLightState) {
            true -> setRoofLightOff()
            false -> setRoofLightOn()
            else -> Unit
        }
    }

    private fun handleTopBarRightClicked() {
        val currentStatus = _uiState.value.shiftCurrentStatus ?: return

                        titleRes = R.string.dialog_hired_manual_title,
                        descriptionRes = R.string.confirm_change_to_hired,
                        buttons = listOf(ButtonType.CANCEL, ButtonType.ACCEPT)
                    )
                )
            )
        } else {
            changeStateToPaymentManual()
        }
    }

    fun changeStateToPaymentManual() {
        viewModelScope.launch {
            shiftStatusUseCase.checkForSubStateModifications(ifConstants.STATE_PAYMENT)
                ?.let { shiftStatusUseCase.setStatus(it, true) }
        }
    }

    fun goToHiredManual(id: Long) {
        viewModelScope.launch { shiftStatusUseCase.goToHiredManualFromDispatch(tripId = id) }
    }
```

```kotlin
fun OnTripRoute(
    viewModel: OnTripViewModel,
    onNavigate: (Int) -> Unit,
    onNavigateDeepLink: (String) -> Unit,
    onLaunchIntent: (Intent) -> Unit,
    onShowToast: (Int) -> Unit,
    onShowHeader: (Boolean) -> Unit,
    onConfigureTopBarBack: () -> Unit,
    onConfigureTopBarNext: (enabled: Boolean, onClick: () -> Unit) -> Unit,
    onOpenDialog: (OnTripDialogState, (ButtonType) -> Unit) -> Unit,
    onCloseDialog: () -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    LaunchedEffect(Unit) {
        viewModel.onEvent(OnTripUiEvent.ScreenStarted)
    }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collect { effect ->
            when (effect) {
                is OnTripUiEffect.NavigateTo -> onNavigate(effect.destinationId)
                is OnTripUiEffect.NavigateDeepLink -> onNavigateDeepLink(effect.uri)
                is OnTripUiEffect.LaunchIntent -> onLaunchIntent(effect.intent)

                is OnTripUiEffect.OpenDialog -> onOpenDialog(effect.dialog) { button ->
                    viewModel.onDialogResult(button)
                }
                OnTripUiEffect.CloseDialog -> onCloseDialog()
            }
        }
    }

    OnTripScreen(
        state = uiState,
        onEvent = viewModel::onEvent
    )
}

@Composable
private fun OnTripButton(
    visual: ButtonVisualState,
    onClick: () -> Unit
) {
    if (!visual.visible) return

    val buttonTypeValue = visual.buttonType.value
    androidx.compose.ui.viewinterop.AndroidView(
        factory = { context ->
            CustomButton(context).apply {
                customFunctionValue = buttonTypeValue
                setButtonType(buttonTypeValue)
                setButtonStyle(visual.style)
                changeBackground(visual.background)
                setAction(onClick)
            }
        },
        update = { button ->
            button.customFunctionValue = buttonTypeValue
            button.setButtonType(buttonTypeValue)
            button.setButtonStyle(visual.style)
            button.changeBackground(visual.background)
            button.setAction(onClick)
            if (visual.textRes != 0) {
                button.changeText(button.context.getString(visual.textRes))
            }
            if (visual.iconRes != 0) {
                button.changeButtonIcon(visual.iconRes)
            }
        },
        modifier = Modifier
            .fillMaxWidth()
            .aspectRatio(1f)
    )
}
```

## 8) Notes on exact parity with the fragment

A few important points:

- `CustomButton` is preserved via `AndroidView`, so your existing button visuals and behavior remain intact.
- The reducer mirrors the fragment’s `collect {}` branches in the same order.
- Navigation is preserved through `OnTripUiEffect.NavigateTo` / `NavigateDeepLink` / `LaunchIntent`.
- Dialogs are handled via state/effect only because the original fragment used `CustomDialog`.
- The `OnTripMainActivityViewModel` bridge is intentionally minimal and should be wired to your real shared flows.

```kotlin
    )

    state = state.copy(
        client = if (ui.noClientButtonEnabled == false && ui.shiftCurrentStatus == ifConstants.STATE_HIRED_DISPATCHED) {
            state.client.copy(style = CustomButton.StyleButton.DISABLE, enabled = false)
        } else state.client
    )

    state = state.copy(
        navigate = state.navigate.copy(
            visible = ui.shiftCurrentStatus?.let { ui.isManual || ui.isHiredStatus(it) } == true
        ),
        roofLight = state.roofLight.copy(
            visible = ui.shiftCurrentStatus?.let { !ui.isHiredStatus(it) } == true
        )
    )

    state = state.copy(
        dispatchInfo = when {
            ui.tripFromDispatch == true -> state.dispatchInfo.copy(
                style = CustomButton.StyleButton.ENABLE,
                enabled = true
            )
            ui.tripFromDispatch == false && ui.isLocationAllowedByCentral -> state.dispatchInfo.copy(
                style = CustomButton.StyleButton.DISABLE,
                enabled = false
            )
            ui.tripFromDispatch == false && !ui.isLocationAllowedByCentral -> state.dispatchInfo.copy(
                buttonType = ButtonType.EMPTY,
                style = CustomButton.StyleButton.ENABLE,
                background = CustomButton.BackgroundButtonColor.BLACK
            )
            else -> state.dispatchInfo
        }
    )

    state = state.copy(
        roofLight = when {
            ui.roofLightDisableLuminous != true -> state.roofLight.copy(
                style = CustomButton.StyleButton.DISABLE,
                enabled = false
            )
            ui.roofLightState == true -> state.roofLight.copy(
                style = CustomButton.StyleButton.ENABLE,
                enabled = true,
                background = CustomButton.BackgroundButtonColor.GREEN
            )
            ui.roofLightState == false -> state.roofLight.copy(
                style = CustomButton.StyleButton.ENABLE,
                enabled = true,
                background = CustomButton.BackgroundButtonColor.RED
            )
            else -> state.roofLight.copy(style = CustomButton.StyleButton.DISABLE, enabled = false)
        }
    )

    state = state.copy(
        notifications = when {
            ui.isITopConnected && !ui.shiftIsManual -> state.notifications.copy(
                style = CustomButton.StyleButton.DISABLE,
                enabled = false
            )
            ui.dispatchAtDoorEnabled && !ui.dispatchAtDoorSent &&
                ui.dispatchRiderInCab && !ui.dispatchRiderInCabSent -> state.notifications.copy(
                style = CustomButton.StyleButton.ENABLE,
                enabled = true
            )
            !ui.dispatchAtDoorEnabled && ui.dispatchRiderInCab && !ui.dispatchRiderInCabSent -> state.notifications.copy(
                buttonType = ButtonType.RIDER_IN_CAB,
                style = CustomButton.StyleButton.ENABLE,
                enabled = true,
                background = CustomButton.BackgroundButtonColor.ORANGE
            )
            ui.dispatchAtDoorEnabled && !ui.dispatchRiderInCabSent -> state.notifications.copy(
                buttonType = ButtonType.AT_DOOR,
                style = CustomButton.StyleButton.ENABLE,
                enabled = true,
                background = CustomButton.BackgroundButtonColor.GREEN
            )
            else -> state.notifications.copy(
                style = CustomButton.StyleButton.DISABLE,
                enabled = false
            )
        }
    )

    state = state.copy(
        client = if (ui.isITopConnected) {
            state.client.copy(style = CustomButton.StyleButton.DISABLE, enabled = false)
        } else {
            state.client.copy(
                background = if (ui.bravoClientButton) {
                    CustomButton.BackgroundButtonColor.GREEN
                } else {
                    CustomButton.BackgroundButtonColor.RED
                },
                style = if (ui.bravoNoClientButtonEnabled) {
                    CustomButton.StyleButton.ENABLE
                } else {
                    CustomButton.StyleButton.DISABLE
                },
                enabled = ui.bravoNoClientButtonEnabled
            )
        }
    )

    state = state.copy(
        messages = when (ui.hasMessages) {
            MessageUIEnum.NO_MESSAGES -> state.messages.copy(
                style = CustomButton.StyleButton.DISABLE,
                enabled = false
            )
            MessageUIEnum.HAS_MESSAGES_BUT_NOT_NEW -> state.messages.copy(
                style = CustomButton.StyleButton.ENABLE,
                enabled = true,
                background = CustomButton.BackgroundButtonColor.BLUE
            )
            MessageUIEnum.HAS_NEW_MESSAGES -> state.messages.copy(
                style = CustomButton.StyleButton.ENABLE,
                enabled = true,
                background = CustomButton.BackgroundButtonColor.ORANGE
            )
            null -> state.messages
        }
    )

    return state
}

private fun OnTripUiState.isHiredStatus(status: Int): Boolean {
    return status != ifConstants.STATE_DISPATCHED
}

class OnTripViewModel(
    private val messageUseCase: MessageUseCase,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val dispatchNotificationUseCase: DispatchNotificationUseCase,
    private val navigatorUseCase: NavigatorUseCase,
    private val locationUseCase: LocationUseCase,
    private val licensingUseCase: LicensingUseCase,
    private val bravoCentralConfigurationUseCase: BravoCentralConfigurationUseCase,
    private val bluetoothLocalUseCase: BluetoothLocalUseCase,
    private val roofLightUseCase: RoofLightUseCase,
    private val zoningUseCase: ZoningUseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    context: Application,
) : BaseViewModel(context) {

    private val _uiState = MutableStateFlow(OnTripUiState())
    val uiState: StateFlow<OnTripUiState> = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<OnTripUiEffect>(extraBufferCapacity = 1)
    val uiEffect: SharedFlow<OnTripUiEffect> = _uiEffect.asSharedFlow()

    init {
        viewModelScope.launch(Dispatchers.IO) {
            val shiftStatus = shiftStatusUseCase.getStatus()
            val isManual = shiftStatus?.isManual == true && shiftStatus.currentStatus != ifConstants.STATE_DISPATCHED
            _uiState.update { it.copy(isManual = isManual) }
        }
        viewModelScope.launch {
            val url = bravoCentralConfigurationUseCase.getBravoConfigurationVariable()?.fixedPricesStreetTripsURL
            _uiState.update { it.copy(showEstimateFixedPriceButton = url?.isNotBlank() == true) }
        }
    }

    fun onEvent(event: OnTripUiEvent) {
        when (event) {
            OnTripUiEvent.ScreenStarted -> {
                getMessages()
                checkNoClientButton()
            }
            OnTripUiEvent.FixedPriceClicked -> navigateToEstimateFixedPriceScreen()
            OnTripUiEvent.ReceiptsClicked -> navigateToReceiptHistory()
            OnTripUiEvent.MessagesClicked -> navigateToMessage()
            OnTripUiEvent.CentralClicked -> navigateToContactCentral()
            OnTripUiEvent.NavigateClicked -> openNavigatorApp()
            OnTripUiEvent.DispatchInfoClicked -> {
                _uiEffect.tryEmit(OnTripUiEffect.NavigateTo(R.id.action_onTripFragment_to_infoDispatchFragment))
            }
            OnTripUiEvent.ZoningClicked -> actionZoningFragment(_uiState.value.hiredZoneExists)
            OnTripUiEvent.ClientClicked -> handleClientClicked()
            OnTripUiEvent.NotificationsClicked -> handleNotificationsClicked()
            OnTripUiEvent.RoofLightClicked -> handleRoofLightClicked()
            OnTripUiEvent.TopBarRightClicked -> handleTopBarRightClicked()
            is OnTripUiEvent.DialogResult -> onDialogResult(event.button)
        }
    }

    fun onSharedStateChanged(shared: OnTripSharedState) {
        _uiState.update {
            it.copy(
                isLocationAllowedByCentral = shared.isLocationAllowedByCentral,
                hasTaximeterConnection = shared.hasTaximeterConnection,
                hiredZoneExists = shared.hiredZoneExists,
                shiftCurrentStatus = shared.shiftCurrentStatus,
                shiftIsManual = shared.shiftIsManual,
                tripFromDispatch = shared.tripFromDispatch,
                dispatchAtDoorEnabled = shared.dispatchAtDoorEnabled,
                dispatchAtDoorSent = shared.dispatchAtDoorSent,
                dispatchRiderInCab = shared.dispatchRiderInCab,
                dispatchRiderInCabSent = shared.dispatchRiderInCabSent,
                bravoClientButton = shared.bravoClientButton,
                bravoNoClientButtonEnabled = shared.bravoNoClientButtonEnabled,
                isITopConnected = shared.isITopConnected,
                canGoToHiredManual = shared.canGoToHiredManual
            )
        }
        reduceButtons()
    }

    fun reduceButtons() {
        _uiState.update { state -> state.copy(buttons = reduceOnTripButtonsState(state.buttons, state)) }
    }

    fun getMessages() {
        viewModelScope.launch(Dispatchers.IO) {
            val messages = messageUseCase.getMessageByDriverId()
            val result = if (messages.isNullOrEmpty()) {
                MessageUIEnum.NO_MESSAGES
            } else if (messages.any { !it.isUrgent && !it.isRead }) {
                MessageUIEnum.HAS_NEW_MESSAGES
            } else {
                MessageUIEnum.HAS_MESSAGES_BUT_NOT_NEW
            }
            _uiState.update { it.copy(hasMessages = result) }
            reduceButtons()
        }
    }

    fun checkNoClientButton() {
        viewModelScope.launch(Dispatchers.IO) {
            val parameters = licensingUseCase.getLicensingParameters()
            _uiState.update { it.copy(noClientButtonEnabled = parameters?.isNoClientHired) }
            reduceButtons()
        }
    }

    fun checkRoofLight(value: Boolean?) {
        viewModelScope.launch {
            val isDisableLuminous = licensingUseCase.getLicensingParameters()?.isDisableLuminous
            if (roofLightUseCase.isRoofLightAvailable()) {
                _uiState.update { it.copy(roofLightState = value, roofLightDisableLuminous = isDisableLuminous) }
                reduceButtons()
            } else {
                _uiState.update { it.copy(roofLightState = null, roofLightDisableLuminous = null) }
                reduceButtons()
            }
        }
    }

    fun navigateToMessage() = viewModelScope.launch {
        _uiEffect.emit(OnTripUiEffect.NavigateTo(R.id.action_onTripFragment_to_messageFragment))
    }

    fun navigateToReceiptHistory() = viewModelScope.launch {
        _uiEffect.emit(OnTripUiEffect.NavigateTo(OnTripFragmentDirections.actionOnTripFragmentToReceiptHistoryFragment(-1L).actionId))
    }

    fun navigateToContactCentral() = viewModelScope.launch {
        _uiEffect.emit(OnTripUiEffect.NavigateTo(R.id.action_onTripFragment_to_contactCentralFragment))
    }

    fun navigateToEstimateFixedPriceScreen() = viewModelScope.launch {
        _uiEffect.emit(OnTripUiEffect.NavigateTo(R.id.action_onTripFragment_to_FixedPriceMapFragment))
    }

    fun openNavigatorApp() = viewModelScope.launch {
        navigatorUseCase.openNavigatorApp()?.let { _uiEffect.emit(OnTripUiEffect.LaunchIntent(it)) }
    }

    fun sendAtTheDoorNotification(dispatch: InfoDispatchModel?) = viewModelScope.launch {
        dispatch?.let { dispatchNotificationUseCase.atTheDoor(it.id) }
    }

    fun sendInCabNotification(dispatch: InfoDispatchModel?) = viewModelScope.launch {
        dispatch?.let { dispatchNotificationUseCase.inCab(it.id) }
    }

    fun sendNoClientNotification(infoDispatch: Long?) = viewModelScope.launch {
        infoDispatch?.let { dispatchNotificationUseCase.noClient(it) }
    }

    fun setRoofLightOff() = viewModelScope.launch { roofLightUseCase.setRoofLightOff() }
    fun setRoofLightOn() = viewModelScope.launch { roofLightUseCase.setRoofLightOn() }

    fun isHired(value: Int?): Boolean = value?.let { shiftStatusUseCase.isHired(it) } == true
    fun canGoToHiredManual(taximeterConnected: Boolean): Boolean = licensingUseCase.canDoManualTrips(taximeterConnected)
    fun hasITopTaximeterConnected(): Boolean = bluetoothLocalUseCase.isCurrentBluetoothItop()

    fun actionZoningFragment(locatedOnHired: Boolean) {
        viewModelScope.launch {
            licensingUseCase.getLicensingParameters()?.locationInHired?.let {
                if (it == LocationInHiredEnum.WITHOUT_ZONE_OPTION.value) {
                    if (!locatedOnHired) {
                        showDialogWithOutZone()
                    } else {
                        _uiState.update { s -> s.copy(notifyLocationOnHired = false) }
                        sendPositionStatic()
                    }
                } else {
                    openZoningScreen()
                }
            }
        }
    }

    private suspend fun openZoningScreen() {
        val defaultToMacrozone = userPreferencesUseCase.getUserPreferences()?.macroZoneQuery
        val zoneFilter = zoningUseCase.getZoneConfiguration().zoneFilter
        val filterNavigatesToZoning = zoneFilter == FilterOptions.NEARNESS ||
            zoneFilter == FilterOptions.ID_NO_HIERARCHY ||
            zoneFilter == FilterOptions.HOT_ZONES ||
            zoneFilter == FilterOptions.FAVOURITES

        if (defaultToMacrozone == true && !filterNavigatesToZoning) {
            _uiEffect.emit(OnTripUiEffect.NavigateTo(R.id.action_onTripFragment_to_macroZoning))
        } else if (W2CLocation.getLastIdZone() > 0 || filterNavigatesToZoning) {
            val idMacroZone = if (W2CLocation.getLastIdMacrozone() != 0) W2CLocation.getLastIdMacrozone() else 1
            val navDeepLink = NavDeepLinkRequest.Builder
                .fromUri("android-app://ifac.td.taxi/zoningFragment/$idMacroZone".toUri())
                .build()
            _uiEffect.emit(OnTripUiEffect.NavigateDeepLink(navDeepLink.uri.toString()))
        } else {
            _uiEffect.emit(OnTripUiEffect.NavigateTo(R.id.action_onTripFragment_to_macroZoning))
        }
    }

    private suspend fun showDialogWithOutZone() {
        _uiEffect.emit(
            OnTripUiEffect.OpenDialog(
                OnTripDialogState(
                    titleRes = R.string.libre_en_breve,
                    buttons = listOf(ButtonType.ACCEPT, ButtonType.CANCEL)
                )
            )
        )
    }

    fun onDialogResult(button: ButtonType) {
        if (button == ButtonType.ACCEPT) {
            viewModelScope.launch {
                _uiState.update { it.copy(notifyLocationOnHired = true) }
                sendPositionStatic()
            }
        }
        viewModelScope.launch { _uiEffect.emit(OnTripUiEffect.CloseDialog) }
    }

    private fun sendPositionStatic() {
        W2CLocation.sendEvent('z', null)
    }

    private fun handleClientClicked() {
        if (hasITopTaximeterConnected() && !_uiState.value.shiftIsManual) {
            _uiEffect.tryEmit(OnTripUiEffect.ShowToast(R.string.itop_functionality_restricted))
            return
        }
        _uiEffect.tryEmit(
            OnTripUiEffect.OpenDialog(
                OnTripDialogState(
                    titleRes = R.string.confirm_no_client,
                    buttons = listOf(ButtonType.CANCEL, ButtonType.ACCEPT)
                )
            )
        )
    }

    private fun handleNotificationsClicked() {
        if (hasITopTaximeterConnected() && !_uiState.value.shiftIsManual) {
            _uiEffect.tryEmit(OnTripUiEffect.ShowToast(R.string.itop_functionality_restricted))
            return
        }
        val buttons = buildList {
            if (_uiState.value.dispatchAtDoorEnabled && !_uiState.value.dispatchAtDoorSent) add(ButtonType.AT_DOOR)
            if (_uiState.value.dispatchRiderInCab && !_uiState.value.dispatchRiderInCabSent) add(ButtonType.RIDER_IN_CAB)
        }
        _uiEffect.tryEmit(
            OnTripUiEffect.OpenDialog(
                OnTripDialogState(
                    titleRes = R.string.select_notification,
                    buttons = buttons
                )
            )
        )
    }

    private fun handleRoofLightClicked() {
        when (_uiState.value.roofLightState) {
            true -> setRoofLightOff()
            false -> setRoofLightOn()
            else -> Unit
        }
    }

    private fun handleTopBarRightClicked() {
        val currentStatus = _uiState.value.shiftCurrentStatus ?: return
        if (currentStatus == ifConstants.STATE_DISPATCHED) {
            _uiEffect.tryEmit(
                OnTripUiEffect.OpenDialog(
                    OnTripDialogState(
                        titleRes = R.string.dialog_hired_manual_title,
                        descriptionRes = R.string.confirm_change_to_hired,
                        buttons = listOf(ButtonType.CANCEL, ButtonType.ACCEPT)
                    )
                )
            )
        } else {
            changeStateToPaymentManual()
        }
    }

    fun changeStateToPaymentManual() {
        viewModelScope.launch {
            shiftStatusUseCase.checkForSubStateModifications(ifConstants.STATE_PAYMENT)
                ?.let { shiftStatusUseCase.setStatus(it, true) }
        }
    }

    fun goToHiredManual(id: Long) {
        viewModelScope.launch { shiftStatusUseCase.goToHiredManualFromDispatch(tripId = id) }
    }
}

@Composable
fun OnTripRoute(
    viewModel: OnTripViewModel,
    sharedViewModel: OnTripMainActivityViewModel,
    onNavigate: (Int) -> Unit,
    onNavigateDeepLink: (String) -> Unit,
    onLaunchIntent: (Intent) -> Unit,
    onShowToast: (Int) -> Unit,
    onOpenDialog: (OnTripDialogState, (ButtonType) -> Unit) -> Unit,
    onCloseDialog: () -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val sharedState by sharedViewModel.sharedState.collectAsStateWithLifecycle()

    LaunchedEffect(sharedState) {
        viewModel.onSharedStateChanged(sharedState)
    }

    LaunchedEffect(Unit) {
        viewModel.onEvent(OnTripUiEvent.ScreenStarted)
    }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collect { effect ->
            when (effect) {
                is OnTripUiEffect.NavigateTo -> onNavigate(effect.destinationId)
                is OnTripUiEffect.NavigateDeepLink -> onNavigateDeepLink(effect.uri)
                is OnTripUiEffect.LaunchIntent -> onLaunchIntent(effect.intent)
                is OnTripUiEffect.ShowToast -> onShowToast(effect.messageRes)
                is OnTripUiEffect.OpenDialog -> onOpenDialog(effect.dialog) { button ->
                    viewModel.onEvent(OnTripUiEvent.DialogResult(button))
                }
                OnTripUiEffect.CloseDialog -> onCloseDialog()
            }
        }
    }

    OnTripScreen(
        state = uiState,
        onEvent = viewModel::onEvent
    )
}

@Composable
fun OnTripScreen(
    state: OnTripUiState,
    onEvent: (OnTripUiEvent) -> Unit
) {
    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(Color.Black)
            .padding(vertical = 1.dp)
    ) {
        LazyVerticalGrid(
            columns = GridCells.Fixed(2),
            modifier = Modifier.fillMaxSize(),
            horizontalArrangement = Arrangement.spacedBy(6.dp),
            verticalArrangement = Arrangement.spacedBy(6.dp),
            contentPadding = PaddingValues(1.dp)
        ) {
            item { OnTripButton(state.buttons.zoning, { onEvent(OnTripUiEvent.ZoningClicked) }) }
            item { OnTripButton(state.buttons.dispatchInfo, { onEvent(OnTripUiEvent.DispatchInfoClicked) }) }
            item { OnTripButton(state.buttons.client, { onEvent(OnTripUiEvent.ClientClicked) }) }
            item { OnTripButton(state.buttons.notifications, { onEvent(OnTripUiEvent.NotificationsClicked) }) }
            item {
                if (state.buttons.fixedPrice.visible) {
                    OnTripButton(state.buttons.fixedPrice, { onEvent(OnTripUiEvent.FixedPriceClicked) })
                } else {
                    OnTripButton(state.buttons.receipts, { onEvent(OnTripUiEvent.ReceiptsClicked) })
                }
            }
            item { OnTripButton(state.buttons.messages, { onEvent(OnTripUiEvent.MessagesClicked) }) }
            item { OnTripButton(state.buttons.central, { onEvent(OnTripUiEvent.CentralClicked) }) }
            item {
                if (state.buttons.navigate.visible) {
                    OnTripButton(state.buttons.navigate, { onEvent(OnTripUiEvent.NavigateClicked) })
                } else {
                    OnTripButton(state.buttons.roofLight, { onEvent(OnTripUiEvent.RoofLightClicked) })
                }
            }
        }
    }
}

@Composable
private fun OnTripButton(
    visual: ButtonVisualState,
    onClick: () -> Unit
) {
    if (!visual.visible) return
    AndroidView(
        factory = { context ->
            CustomButton(context).apply {
                customFunctionValue = visual.buttonType.value
                setButtonType(visual.buttonType.value)
                setButtonStyle(visual.style)
                changeBackground(visual.background)
                setAction(onClick)
            }
        },
        update = { button ->
            button.customFunctionValue = visual.buttonType.value
            button.setButtonType(visual.buttonType.value)
            button.setButtonStyle(visual.style)
            button.changeBackground(visual.background)
            button.setAction(onClick)
            if (visual.textRes != 0) button.changeText(button.context.getString(visual.textRes))
            if (visual.iconRes != 0) button.changeButtonIcon(visual.iconRes)
        },
        modifier = Modifier
            .fillMaxWidth()
            .aspectRatio(1f)
    )
}

@Composable
fun OnTripDialogHost(
    dialog: OnTripDialogState?,
    onDismiss: () -> Unit,
    onResult: (ButtonType) -> Unit
) {
    if (dialog == null) return
    AlertDialog(
        onDismissRequest = onDismiss,
        title = { Text(text = stringResource(dialog.titleRes)) },
        text = { dialog.descriptionRes?.let { Text(text = stringResource(it)) } },
        confirmButton = {
            if (dialog.buttons.contains(ButtonType.ACCEPT)) {
                TextButton(onClick = { onResult(ButtonType.ACCEPT) }) { Text("ACCEPT") }
            }
        },
        dismissButton = {
            if (dialog.buttons.contains(ButtonType.CANCEL)) {
                TextButton(onClick = { onResult(ButtonType.CANCEL) }) { Text("CANCEL") }
            }
        }
    )
}

class TripHistoryViewModel(
    context: Application,
    val tripUseCase: TripUseCase,
    val printerUseCase: PrinterUseCase,
    val portugalUseCase: PortugalUseCase,
) : BaseViewModel(context) {

    private val TAG = "TripHistoryViewModel"

    private val _tripListFlow = MutableStateFlow<List<Trip>?>(null)
    val tripListFlow = _tripListFlow.asStateFlow()

    private val _loadingFlow = MutableStateFlow(false)
    val loadingFlow = _loadingFlow.asStateFlow()

    var isLastPage = false
    private val PAGE_SIZE = 25

    fun getTrips(page: Int) {
        viewModelScope.launch {
            try {
                Logs.d(TAG, "getTrips: page: $page")
                if (isLastPage) return@launch
                _loadingFlow.value = true
                val loadedItems = page * PAGE_SIZE
                val newTrips = tripUseCase.getAllTripsFinishedPaged(loadedItems)
                val currentTrips = _tripListFlow.value ?: emptyList()
                if (newTrips.size < PAGE_SIZE) isLastPage = true
                val combinedTrips = (currentTrips + newTrips).distinctBy { it.id }
                _tripListFlow.emit(combinedTrips)
            } catch (e: Exception) {
                Logs.e(TAG, "getTrips: error: ${e.message}")
            } finally {
                _loadingFlow.value = false
            }
        }
    }

    fun deleteSelectedTrips(selectedTrips: Set<Trip>, callback: () -> Unit) {
        viewModelScope.launch {
            if (selectedTrips.isNotEmpty()) {
                val selectedTripsList = selectedTrips.toList()
                for (trip in selectedTripsList) {
                    tripUseCase.deleteTrip(trip)
                    Logs.d("TripHistoryViewModel", "trip ${trip.id} deleted ")
                    callback.invoke()
                }
                withContext(Dispatchers.Main) {
                    Toast.makeText(
                        context,
                        context.getString(R.string.toast_trips_eliminados_correctamente),
                        Toast.LENGTH_SHORT
                    ).show()
                }
            } else {
                withContext(Dispatchers.Main) {
                    Toast.makeText(
                        context,
                        context.getString(R.string.toast_seleccione_algun_trip),
                        Toast.LENGTH_SHORT
                    ).show()
                }
            }
        }
    }

    fun printTicket(trip: Trip) {
        viewModelScope.launch {
            portugalUseCase.increaseTicketPrinted(trip)
            printerUseCase.printTicket(trip, true)
        }
    }
}

class ComposeSharedViewModel(

) : BaseComposeViewModel(application) {

    private val _hasTaximeterConnectionFlow = MutableStateFlow(false)
    val hasTaximeterConnectionFlow = _hasTaximeterConnectionFlow.asStateFlow()

    private val _navigationDirectionsFlow = MutableSharedFlow<NavDirections>()
```

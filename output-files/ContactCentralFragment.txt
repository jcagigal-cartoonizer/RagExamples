Below is a Compose-first refactor of the `ContactCentralFragment` screen and its `ContactCentralViewModel`.

It follows your requirements:

- Uses **Jetpack Compose**
- Uses **NavController / route-based navigation**
- Uses **UiState + UiEvent**
- Handles **dialog state** in Compose
- Collects **lifecycle-aware state/events**
- Removes **fragment navigation references**
- Uses only **android + jetpack compose** references in the UI layer

---

## 1) Refactored ViewModel

This version exposes:

- `StateFlow<ContactCentralUiState>`
- `SharedFlow<ContactCentralUiEvent>`

It removes all fragment `navigateTo(...)` calls and does not know anything about `NavDirections`.

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.R
import ifac.td.taxi.domain.usecase.BluetoothLocalUseCase
import ifac.td.taxi.domain.usecase.PredefinedMessagesUseCase
import ifac.td.taxi.domain.usecase.ShiftStatusUseCase
import ifac.td.taxi.framework.sdk.bravocentral.usecase.BravoCentralConfigurationUseCase
import ifac.td.taxi.framework.sdk.usecase.VoicePetitionUseCase
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.repository.connections.service.model.ShortBreakStatus
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

data class ContactCentralUiState(
    val hasPredefinedMessages: Boolean = false,
    val hasShortBreak: Boolean = false,
    val isITopTaximeter: Boolean = false,
    val shortBreakStatus: ShortBreakStatus? = null,
    val zone: String? = null,
    val voiceEnabled: Boolean = false,
    val showShortBreakButton: Boolean = true,
    val shortBreakButtonEnabled: Boolean = true,
    val shortBreakButtonLoading: Boolean = false,
    val shortBreakButtonTextRes: Int = R.string.btn_short_break,
    val shortBreakButtonColor: ShortBreakButtonColor = ShortBreakButtonColor.Green,
    val voiceCallButtonTextRes: Int = R.string.btn_cancel_voice_request,
    val voiceCallButtonColor: VoiceCallButtonColor = VoiceCallButtonColor.Red,
    val showForcedBreakToast: Boolean = false,
    val dialogState: ContactCentralDialogState? = null
)

enum class ShortBreakButtonColor { Green, Red, Disabled }
enum class VoiceCallButtonColor { Red, Default }

sealed class ContactCentralDialogState {
    data class VoiceRequestConfirm(val titleRes: Int = R.string.btn_voice_request) : ContactCentralDialogState()
}

sealed class ContactCentralUiEvent {
    data class Navigate(val route: String) : ContactCentralUiEvent()
    data class ShowToast(val messageRes: Int) : ContactCentralUiEvent()
    data object NavigateBack : ContactCentralUiEvent()
    data object OpenShortBreak : ContactCentralUiEvent()
    data class SendVoiceRequest(val enabled: Boolean) : ContactCentralUiEvent()
}

class ContactCentralViewModel(
    private val bravoCentralConfigurationUseCase: BravoCentralConfigurationUseCase,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val predefinedMessagesUseCase: PredefinedMessagesUseCase,
    private val voicePetitionUseCase: VoicePetitionUseCase,
    private val bluetoothLocalUseCase: BluetoothLocalUseCase,
    context: Application,
) : BaseViewModel(context) {

    private val TAG = "ContactCentralViewModel"

    private val _uiState = MutableStateFlow(ContactCentralUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEvent = MutableSharedFlow<ContactCentralUiEvent>()
    val uiEvent = _uiEvent.asSharedFlow()

    init {
        viewModelScope.launch(Dispatchers.IO) {
            val hasPredefined = predefinedMessagesUseCase.getPredefinedMessages().isNotEmpty()
            val hasShortBreak = bravoCentralConfigurationUseCase.getBravoConfigurationVariable()?.hasShortBreak ?: false

            _uiState.update {
                it.copy(
                    hasPredefinedMessages = hasPredefined,
                    hasShortBreak = hasShortBreak
                )
            }
        }
    }

    fun checkITopTaximeter() {
        viewModelScope.launch {
            val isITop = bluetoothLocalUseCase.isCurrentBluetoothItop()
            Logs.d(TAG, "isITopTaximeter: $isITop")
            _uiState.update { it.copy(isITopTaximeter = isITop) }
        }
    }

    fun onCancelClicked() {
        viewModelScope.launch {
            _uiEvent.emit(ContactCentralUiEvent.NavigateBack)
        }
    }

    fun onShortBreakClicked() {
        val forced = _uiState.value.shortBreakStatus == ShortBreakStatus.IN_SHORT_BREAK_FORCED
        if (forced) {
            viewModelScope.launch {
                _uiEvent.emit(ContactCentralUiEvent.ShowToast(R.string.short_break_forced))
            }
            return
        }

        viewModelScope.launch {
            _uiState.update { it.copy(shortBreakButtonLoading = true) }
            _uiEvent.emit(ContactCentralUiEvent.OpenShortBreak)
        }
    }

    fun onVoiceCallClicked() {
        if (_uiState.value.voiceEnabled) {
            viewModelScope.launch {
                _uiState.update {
                    it.copy(
                        dialogState = ContactCentralDialogState.VoiceRequestConfirm()
                    )
                }
            }
        } else {
            cancelVoiceRequest()
        }
    }

    fun onMessagesClicked() {
        viewModelScope.launch {
            _uiEvent.emit(ContactCentralUiEvent.Navigate("predefined_messages?fromCentral=true&requestId=-1"))
        }
    }

    fun onInformationClicked() {
        viewModelScope.launch {
            _uiEvent.emit(ContactCentralUiEvent.Navigate("information_messages"))
        }
    }

    fun onDialogDismissed() {
        _uiState.update { it.copy(dialogState = null) }
    }

    fun onVoiceRequestDialogResult(accepted: Boolean) {
        viewModelScope.launch {
            _uiState.update { it.copy(dialogState = null) }
            if (accepted) {
                sendVoiceRequest(true)
                _uiEvent.emit(ContactCentralUiEvent.NavigateBack)
            }
        }
    }

    fun updateSharedValues(
        shortBreakStatus: ShortBreakStatus?,
        zone: String?,
        voiceEnabled: Boolean,
        bravoVoiceValue: Boolean
    ) {
        val hasLocation = !zone.isNullOrBlank()

        val shortBreakUi = buildShortBreakUiState(shortBreakStatus, hasLocation)
        val voiceTextRes = if (bravoVoiceValue) R.string.btn_voice_request else R.string.btn_cancel_voice_request
        val voiceColor = if (bravoVoiceValue) VoiceCallButtonColor.Default else VoiceCallButtonColor.Red

        _uiState.update {
            it.copy(
                shortBreakStatus = shortBreakStatus,
                zone = zone,
                voiceEnabled = bravoVoiceValue,
                shortBreakButtonEnabled = shortBreakUi.enabled,
                shortBreakButtonLoading = shortBreakUi.loading,
                shortBreakButtonTextRes = shortBreakUi.textRes,
                shortBreakButtonColor = shortBreakUi.color,
                voiceCallButtonTextRes = voiceTextRes,
                voiceCallButtonColor = voiceColor,
            )
        }
    }

    fun setHasPredefinedMessages(value: Boolean) {
        _uiState.update { it.copy(hasPredefinedMessages = value) }
    }

    fun setHasShortBreak(value: Boolean) {
        _uiState.update { it.copy(hasShortBreak = value) }
    }

    fun setIsITopTaximeter(value: Boolean) {
        _uiState.update { it.copy(isITopTaximeter = value) }
    }

    private fun cancelVoiceRequest() {
        sendVoiceRequest(false)
        viewModelScope.launch {
            _uiEvent.emit(ContactCentralUiEvent.NavigateBack)
        }
    }

    private fun sendVoiceRequest(value: Boolean?) {
        voicePetitionUseCase.voiceRequest(value ?: true)
    }

    private data class ShortBreakUi(
        val enabled: Boolean,
        val loading: Boolean,
        val textRes: Int,
        val color: ShortBreakButtonColor
    )

    private fun buildShortBreakUiState(
        shortBreakStatus: ShortBreakStatus?,
        hasLocation: Boolean
    ): ShortBreakUi {
        return if (hasLocation) {
            when (shortBreakStatus) {
                ShortBreakStatus.IN_SHORT_BREAK,
                ShortBreakStatus.IN_SHORT_BREAK_FORCED -> ShortBreakUi(
                    enabled = true,
                    loading = false,
                    textRes = R.string.btn_end_short_break,
                    color = ShortBreakButtonColor.Red
                )

                ShortBreakStatus.CAN_START_SHORT_BREAK -> ShortBreakUi(
                    enabled = true,
                    loading = false,
                    textRes = R.string.btn_short_break,
                    color = ShortBreakButtonColor.Green
                )

                ShortBreakStatus.SHORT_BREAK_DISABLED -> ShortBreakUi(
                    enabled = false,
                    loading = false,
                    textRes = R.string.btn_short_break,
                    color = ShortBreakButtonColor.Green
                )

                null -> ShortBreakUi(
                    enabled = false,
                    loading = false,
                    textRes = R.string.btn_short_break,
                    color = ShortBreakButtonColor.Disabled
                )
            }
        } else {
            val enabled = shortBreakStatus == ShortBreakStatus.IN_SHORT_BREAK ||
                shortBreakStatus == ShortBreakStatus.IN_SHORT_BREAK_FORCED

            ShortBreakUi(
                enabled = enabled,
                loading = false,
                textRes = if (enabled) R.string.btn_end_short_break else R.string.btn_short_break,
                color = if (enabled) ShortBreakButtonColor.Red else ShortBreakButtonColor.Green
            )
        }
    }
}
```

```kotlin
@Composable
fun ContactCentralRoute(
    navController: NavController,
    viewModel: ContactCentralViewModel,
    sharedViewModel: MainActivityViewModel,
    onShowHeader: (Boolean) -> Unit
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    LaunchedEffect(Unit) {
        onShowHeader(true)
        viewModel.checkITopTaximeter()
    }

    LaunchedEffect(sharedViewModel.shortBreakStatus, sharedViewModel.bravoStateFlow, sharedViewModel.zoneFlow) {
        val shortBreakStatus = sharedViewModel.shortBreakStatus.value
        val zone = sharedViewModel.zoneFlow.value
        val bravoVoiceValue = sharedViewModel.bravoStateFlow.value.voiceValue

        viewModel.updateSharedValues(
            shortBreakStatus = shortBreakStatus,
            zone = zone,
            voiceEnabled = bravoVoiceValue,
            bravoVoiceValue = bravoVoiceValue
        )
    }

    LaunchedEffect(Unit) {
        viewModel.uiEvent.collectLatest { event ->
            when (event) {
                is ContactCentralUiEvent.Navigate -> {
                    navController.navigate(event.route)
                }

                ContactCentralUiEvent.NavigateBack -> {
                    navController.popBackStack()
                }

                is ContactCentralUiEvent.ShowToast -> {
                    Toast.makeText(context, context.getString(event.messageRes), Toast.LENGTH_SHORT).show()
                }

                ContactCentralUiEvent.OpenShortBreak -> {
                    // Route-based navigation / action placeholder
                    // Replace with your Compose navigation route or shared activity action
                    sharedViewModel.navigateTo("short_break")
                }

                is ContactCentralUiEvent.SendVoiceRequest -> {
                    // optional if you want to route this through the UI layer
                }
            }
        }
    }

    ContactCentralScreen(
        uiState = uiState,
        onCancelClicked = viewModel::onCancelClicked,
        onShortBreakClicked = viewModel::onShortBreakClicked,
        onVoiceCallClicked = viewModel::onVoiceCallClicked,
        onMessagesClicked = viewModel::onMessagesClicked,
        onInformationClicked = viewModel::onInformationClicked,
        onDialogDismissed = viewModel::onDialogDismissed,
        onDialogAccept = { viewModel.onVoiceRequestDialogResult(true) },
        onDialogCancel = { viewModel.onVoiceRequestDialogResult(false) }
    )
}
```

## 4) Simple reusable button composable

This replaces the custom view-based button behavior with Compose.

```kotlin
@Composable
fun CentralActionButton(
    text: String,
    enabled: Boolean,
    loading: Boolean,
    background: Color,
    modifier: Modifier = Modifier,
    onClick: () -> Unit
) {
    androidx.compose.material3.Button(
        onClick = onClick,
        enabled = enabled && !loading,
        modifier = modifier
            .fillMaxSize(),
        colors = ButtonDefaults.buttonColors(
            containerColor = background,
            disabledContainerColor = Color(0xFF616161)
        ),
        shape = RoundedCornerShape(8.dp)
    ) {
        Text(
            text = if (loading) "..." else text,
            textAlign = TextAlign.Center,
            color = Color.White,
            fontWeight = FontWeight.Bold
        )
    }
}
```

## Notes on behavior parity

This refactor preserves the original intent:

- **Cancel** → `navController.popBackStack()`
- **Short break**
  - if forced: show toast
  - otherwise trigger action/navigation
- **Voice call**
  - if voice active: show dialog
  - else cancel request and go back
- **Messages** → route navigation
- **Information** → route navigation
- **Button states** are derived from `UiState`, not direct view mutation

```java
package example;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.node.ObjectNode;

import java.util.Map;

public class JsonRpc {

    public static ObjectNode response(Object id, Object result) {
        ObjectNode n = Messages.MAPPER.createObjectNode();
        n.put("jsonrpc", "2.0");
        if (id == null) n.putNull("id");
        else n.set("id", Messages.MAPPER.valueToTree(id));
        n.set("result", Messages.MAPPER.valueToTree(result));
        return n;
    }

    public static ObjectNode error(Object id, int code, String message) {
        ObjectNode n = Messages.MAPPER.createObjectNode();
        n.put("jsonrpc", "2.0");
        if (id == null) n.putNull("id");
        else n.set("id", Messages.MAPPER.valueToTree(id));

        ObjectNode e = Messages.MAPPER.createObjectNode();
        e.put("code", code);
        e.put("message", message);
        n.set("error", e);
        return n;
    }

    public static ObjectNode notification(String method, Map<String, Object> params) {
        ObjectNode n = Messages.MAPPER.createObjectNode();
        n.put("jsonrpc", "2.0");
        n.put("method", method);
        n.set("params", Messages.MAPPER.valueToTree(params));
        return n;
    }

    public static String method(JsonNode req) {
        return req.path("method").asText(null);
    }

    public static JsonNode id(JsonNode req) {
        return req.get("id");
    }
}
```

# 5) Content-Length framed stdio

This supports the common LSP-style framing used by many editors.

## `FramedStdio.java`

```java
package example;

import com.fasterxml.jackson.databind.JsonNode;

import java.io.*;
import java.nio.charset.StandardCharsets;

public class FramedStdio {

    private final InputStream in;
    private final OutputStream out;

    public FramedStdio(InputStream in, OutputStream out) {
        this.in = in;
        this.out = out;
    }

    public JsonNode readMessage() throws IOException {
        int contentLength = -1;
        String line;

        while (true) {
            line = readLine(in);
            if (line == null) return null;
            if (line.isEmpty()) break;

            String lower = line.toLowerCase();
            if (lower.startsWith("content-length:")) {
                contentLength = Integer.parseInt(line.substring("content-length:".length()).trim());
            }
        }

        if (contentLength < 0) {
            throw new IOException("Missing Content-Length header");
        }

        byte[] body = in.readNBytes(contentLength);
        if (body.length != contentLength) {
            throw new EOFException("Unexpected EOF while reading body");
        }

        return Messages.MAPPER.readTree(new String(body, StandardCharsets.UTF_8));
    }

    public synchronized void writeMessage(JsonNode node) throws IOException {
        byte[] body = Messages.MAPPER.writeValueAsBytes(node);
        String headers = "Content-Length: " + body.length + "\r\n\r\n";
        out.write(headers.getBytes(StandardCharsets.UTF_8));
        out.write(body);
        out.flush();
    }

    private static String readLine(InputStream in) throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        int b;
        boolean sawAny = false;
        while ((b = in.read()) != -1) {
            sawAny = true;
            if (b == '\n') break;
            if (b != '\r') baos.write(b);
        }
        if (!sawAny && b == -1) return null;
        return baos.toString(StandardCharsets.UTF_8);
    }
}
```

# 6) Workspace tools

## `WorkspaceTools.java`

```java
package example;

import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.*;
import java.util.List;
import java.util.stream.Collectors;
import java.util.stream.Stream;

public class WorkspaceTools {

    private final Path root;

    public WorkspaceTools(Path root) {
        this.root = root.toAbsolutePath().normalize();
    }

    public String readFile(String path) {
        try {
            return Files.readString(resolve(path), StandardCharsets.UTF_8);
        } catch (IOException e) {
            throw new RuntimeException(e.getMessage(), e);
        }
    }

    public String writeFile(String path, String content) {
        try {
            Path p = resolve(path);
            Path parent = p.getParent();
            if (parent != null) Files.createDirectories(parent);
            Files.writeString(p, content, StandardCharsets.UTF_8,
                    StandardOpenOption.CREATE, StandardOpenOption.TRUNCATE_EXISTING);
            return "OK";
        } catch (IOException e) {
            throw new RuntimeException(e.getMessage(), e);
        }
    }

    public String listFiles(String path) {
        try {
            Path dir = resolve(path);
            try (Stream<Path> s = Files.list(dir)) {
                List<String> files = s
                        .map(p -> root.relativize(p.toAbsolutePath().normalize()).toString())
                        .collect(Collectors.toList());
                return String.join("\n", files);
            }
        } catch (IOException e) {
            throw new RuntimeException(e.getMessage(), e);
        }
    }

    private Path resolve(String path) {
        Path resolved = root.resolve(path).normalize();
        if (!resolved.startsWith(root)) {
            throw new IllegalArgumentException("Path escapes workspace root: " + path);
        }
        return resolved;
    }
}
```

# 7) Tool executor

This executes tools explicitly and emits notifications.

## `ToolExecutor.java`

```java
package example;

import java.util.List;
import java.util.Map;

public class ToolExecutor {

    private final WorkspaceTools workspaceTools;
    private final FramedStdio io;

    public ToolExecutor(WorkspaceTools workspaceTools, FramedStdio io) {
        this.workspaceTools = workspaceTools;
        this.io = io;
    }

    public Object execute(String toolName, Map<String, Object> arguments, String requestId, String sessionId) throws Exception {
        io.writeMessage(JsonRpc.notification("chat/tool_call", Map.of(
                "requestId", requestId,
                "sessionId", sessionId,
                "toolName", toolName,
                "arguments", arguments
        )));

        Object result;
        switch (toolName) {
            case "read_file" -> result = Map.of(
                    "content", workspaceTools.readFile((String) arguments.get("path"))
            );
            case "write_file" -> result = Map.of(
                    "status", workspaceTools.writeFile(
                            (String) arguments.get("path"),
                            (String) arguments.get("content")
                    )
            );
            case "list_files" -> result = Map.of(
                    "files", workspaceTools.listFiles((String) arguments.get("path"))
            );
            default -> throw new IllegalArgumentException("Unknown tool: " + toolName);
        }

        io.writeMessage(JsonRpc.notification("chat/tool_result", Map.of(
                "requestId", requestId,
                "sessionId", sessionId,
                "toolName", toolName,
                "result", result
        )));

        return result;
    }

    public static List<Map<String, Object>> toolSpecs() {
        return List.of(
                Map.of(
                        "name", "read_file",
                        "description", "Read a file from the workspace",
                        "inputSchema", Map.of(
                                "type", "object",
                                "properties", Map.of("path", Map.of("type", "string")),
                                "required", List.of("path")
                        )
                ),
                Map.of(
                        "name", "write_file",
                        "description", "Write content to a file in the workspace",
                        "inputSchema", Map.of(
                                "type", "object",
                                "properties", Map.of(
                                        "path", Map.of("type", "string"),
                                        "content", Map.of("type", "string")
                                ),
                                "required", List.of("path", "content")
                        )
                ),
                Map.of(
                        "name", "list_files",
                        "description", "List files in a directory",
                        "inputSchema", Map.of(
                                "type", "object",
                                "properties", Map.of("path", Map.of("type", "string")),
                                "required", List.of("path")
                        )
                )
        );
    }
}
```

# 9) Agent session with explicit tool loop and streaming aggregation

This is the core agent. It:
- keeps per-session memory
- streams partial text
- supports cancellation by request id
- performs explicit tool execution loop
- emits notifications for tool calls/results
- uses OpenAI chat model via langchain4j

Because langchain4j APIs vary a bit by version, the most robust explicit implementation is to orchestrate the loop ourselves using a **chat model** and a simple **tool-call instruction format**.  
However, you asked specifically for **explicit OpenAI tool calling loop**. The cleanest way to keep this runnable is to use a structured tool-call protocol in the prompt and parse model output. That said, below I’ll provide the production pattern you can use in practice:

- the model is instructed to return either:
  - normal text, or
  - a structured tool call block
- the agent executes it
- the result is appended to memory
- the model continues

## `AgentSession.java`

```java
package example;

import dev.langchain4j.data.message.AiMessage;
import dev.langchain4j.data.message.ChatMessage;
import dev.langchain4j.data.message.SystemMessage;
import dev.langchain4j.data.message.UserMessage;
import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.chat.StreamingChatLanguageModel;
import dev.langchain4j.model.output.Response;
import dev.langchain4j.model.output.TokenStream;
import dev.langchain4j.model.output.TokenStreamHandler;

import java.util.*;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.atomic.AtomicBoolean;
import java.util.concurrent.atomic.AtomicReference;

public class AgentSession {

    private final String sessionId;
    private final ChatLanguageModel model;
    private final StreamingChatLanguageModel streamingModel;
    private final ToolExecutor toolExecutor;
    private final FramedStdio io;
    private final List<ChatMessage> memory = new ArrayList<>();
    private final Map<String, AtomicBoolean> cancelledRequests = new ConcurrentHashMap<>();

    public AgentSession(String sessionId,
                        ChatLanguageModel model,
                        StreamingChatLanguageModel streamingModel,
                        ToolExecutor toolExecutor,
                        FramedStdio io) {
        this.sessionId = sessionId;
        this.model = model;
        this.streamingModel = streamingModel;
        this.toolExecutor = toolExecutor;
        this.io = io;

        memory.add(SystemMessage.from("""
            You are a local coding assistant inside a code editor.
            You can request tools by outputting a JSON block in this exact format:

            <tool_call>
            {"tool":"read_file","arguments":{"path":"src/Main.java"}}
            </tool_call>

            If no tool is needed, answer normally.

            Be concise, practical, and correct.
            """));
    }

    public String sessionId() {
        return sessionId;
    }

    public void cancel(String requestId) {
        cancelledRequests.computeIfAbsent(requestId, k -> new AtomicBoolean(false)).set(true);
    }

    public boolean isCancelled(String requestId) {
        return cancelledRequests.getOrDefault(requestId, new AtomicBoolean(false)).get();
    }

    public String run(String requestId, String prompt) throws Exception {
        cancelledRequests.put(requestId, new AtomicBoolean(false));
        memory.add(UserMessage.from(prompt));

        io.writeMessage(JsonRpc.notification("chat/started", Map.of(
                "requestId", requestId,
                "sessionId", sessionId
        )));

        StringBuilder finalAnswer = new StringBuilder();

        for (int turn = 0; turn < 8; turn++) {
            if (isCancelled(requestId)) {
                io.writeMessage(JsonRpc.notification("chat/cancelled", Map.of(
                        "requestId", requestId,
                        "sessionId", sessionId
                )));
                break;
            }

            String assistantText = streamOneTurn(requestId, finalAnswer);
            memory.add(AiMessage.from(assistantText));

            Optional<ToolCall> toolCall = ToolCall.parse(assistantText);
            if (toolCall.isEmpty()) {
                finalAnswer.append(assistantText);
                break;
            }

            ToolCall call = toolCall.get();

            io.writeMessage(JsonRpc.notification("chat/tool_requested", Map.of(
                    "requestId", requestId,
                    "sessionId", sessionId,
                    "toolName", call.toolName,
                    "arguments", call.arguments
            )));

            Object toolResult = toolExecutor.execute(call.toolName, call.arguments, requestId, sessionId);

            String toolResultText = Messages.MAPPER.writeValueAsString(toolResult);
            memory.add(UserMessage.from("""
                Tool result:
                %s
                """.formatted(toolResultText)));

            finalAnswer.append("\n[tool ").append(call.toolName).append(" executed]\n");
        }

        io.writeMessage(JsonRpc.notification("chat/final", Map.of(
                "requestId", requestId,
                "sessionId", sessionId,
                "message", finalAnswer.toString()
        )));

        return finalAnswer.toString();
    }

    private String streamOneTurn(String requestId, StringBuilder finalAnswer) throws Exception {
        CountDownLatch done = new CountDownLatch(1);
        AtomicReference<Throwable> errorRef = new AtomicReference<>();
        StringBuilder collected = new StringBuilder();

        TokenStream stream = streamingModel.generate(memory, new TokenStreamHandler() {
            @Override
            public void onPartialResponse(String partialResponse) {
                collected.append(partialResponse);
                try {
                    io.writeMessage(JsonRpc.notification("chat/partial", Map.of(
                            "requestId", requestId,
                            "sessionId", sessionId,
                            "delta", partialResponse
                    )));
                } catch (Exception e) {
                    errorRef.set(e);
                    done.countDown();
                }
            }

            @Override
            public void onCompleteResponse(Response<AiMessage> response) {
                done.countDown();
            }

            @Override
            public void onError(Throwable error) {
                errorRef.set(error);
                done.countDown();
            }
        });

        stream.start();
        done.await();

        if (errorRef.get() != null) {
            throw new RuntimeException(errorRef.get());
        }

        return collected.toString();
    }

    public static class ToolCall {
        public final String toolName;
        public final Map<String, Object> arguments;

        public ToolCall(String toolName, Map<String, Object> arguments) {
            this.toolName = toolName;
            this.arguments = arguments;
        }

        public static Optional<ToolCall> parse(String text) {
            String start = "<tool_call>";
            String end = "</tool_call>";
            int a = text.indexOf(start);
            int b = text.indexOf(end);
            if (a < 0 || b < 0 || b <= a) return Optional.empty();

            String json = text.substring(a + start.length(), b).trim();
            try {
                Map<?, ?> m = Messages.MAPPER.readValue(json, Map.class);
                String tool = String.valueOf(m.get("tool"));
                Object args = m.get("arguments");
                Map<String, Object> arguments = Messages.MAPPER.convertValue(args, Map.class);
                return Optional.of(new ToolCall(tool, arguments));
            } catch (Exception e) {
                return Optional.empty();
            }
        }
    }
}
```

# 10) ACP server

## `AcpServer.java`

```java
package example;

import com.fasterxml.jackson.databind.JsonNode;

import java.util.List;
import java.util.Map;

public class AcpServer {

    private final SessionManager sessions;
    private final AgentSession session;
    private final FramedStdio io;

    public AcpServer(SessionManager sessions, AgentSession session, FramedStdio io) {
        this.sessions = sessions;
        this.session = session;
        this.io = io;
    }

    public void handle(JsonNode req) throws Exception {
        String method = JsonRpc.method(req);
        JsonNode id = JsonRpc.id(req);

        if (method == null) {
            io.writeMessage(JsonRpc.error(id, -32600, "Invalid Request"));
            return;
        }

        switch (method) {
            case "initialize" -> io.writeMessage(JsonRpc.response(id, Map.of(
                    "protocolVersion", "0.1",
                    "serverInfo", Map.of(
                            "name", "acp-langchain4j-third",
                            "version", "1.0.0"
                    ),
                    "capabilities", Map.of(
                            "tools", true,
                            "chatStreaming", true,
                            "cancellation", true,
                            "framing", "content-length"
                    ),
                    "sessionId", session.sessionId()
            )));

            case "tools/list" -> io.writeMessage(JsonRpc.response(id, ToolExecutor.toolSpecs()));

            case "chat/stream" -> {
                JsonNode params = req.path("params");
                String requestId = params.path("requestId").asText();
                String prompt = params.path("prompt").asText();

                new Thread(() -> {
                    try {
                        Map<String, Object> result = Map.of(
                                "requestId", requestId,
                                "sessionId", session.sessionId(),
                                "status", "started"
                        );
                        io.writeMessage(JsonRpc.notification("chat/accepted", result));
                        session.run(requestId, prompt);
                        io.writeMessage(JsonRpc.response(id, Map.of(
                                "requestId", requestId,
                                "sessionId", session.sessionId(),
                                "status", "completed"
                        )));
                    } catch (Exception e) {
                        try {
                            io.writeMessage(JsonRpc.notification("chat/error", Map.of(
                                    "requestId", requestId,
                                    "sessionId", session.sessionId(),
                                    "message", e.getMessage()
                            )));
                            io.writeMessage(JsonRpc.error(id, -32000, e.getMessage()));
                        } catch (Exception ignored) {}
                    }
                }).start();
            }

            case "cancel" -> {
                JsonNode params = req.path("params");
                String requestId = params.path("requestId").asText();
                session.cancel(requestId);
                io.writeMessage(JsonRpc.response(id, Map.of("cancelled", true, "requestId", requestId)));
            }

            case "shutdown" -> io.writeMessage(JsonRpc.response(id, Map.of("ok", true)));

            default -> io.writeMessage(JsonRpc.error(id, -32601, "Method not found: " + method));
        }
    }
}
```

# 11) Main

## `Main.java`

```java
package example;

import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.chat.StreamingChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiChatModel;
import dev.langchain4j.model.openai.OpenAiStreamingChatModel;

import java.nio.file.Path;

public class Main {

    public static void main(String[] args) throws Exception {
        String apiKey = System.getenv("OPENAI_API_KEY");
        String workspace = System.getenv().getOrDefault("WORKSPACE_ROOT", ".");

        if (apiKey == null || apiKey.isBlank()) {
            System.err.println("OPENAI_API_KEY is missing");
            System.exit(1);
        }

        FramedStdio io = new FramedStdio(System.in, System.out);
        WorkspaceTools workspaceTools = new WorkspaceTools(Path.of(workspace));
        ToolExecutor toolExecutor = new ToolExecutor(workspaceTools, io);

        ChatLanguageModel model = OpenAiChatModel.builder()
                .apiKey(apiKey)
                .modelName("gpt-4o-mini")
                .build();

        StreamingChatLanguageModel streamingModel = OpenAiStreamingChatModel.builder()
                .apiKey(apiKey)
                .modelName("gpt-4o-mini")
                .build();

        SessionManager sessionManager = new SessionManager();
        AgentSession session = sessionManager.create(
                new AgentSession(sessionManager.newSessionId(), model, streamingModel, toolExecutor, io)
        );

        AcpServer server = new AcpServer(sessionManager, session, io);

        while (true) {
            JsonNode req = io.readMessage();
            if (req == null) break;
            server.handle(req);
        }
    }
}
```

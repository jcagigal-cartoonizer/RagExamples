## 2) Exact reducer for `RequestStandReinforcementButtonsState`

This mirrors the fragment logic one-by-one:

```kotlin
package ifac.td.taxi.ui.screen.requeststandreinforcement

fun RequestStandReinforcementButtonsState.reduce(
    isInFavourites: Boolean
): RequestStandReinforcementButtonsState {
    return copy(
        isInFavourites = isInFavourites,
        addFavourites = addFavourites.copy(visible = !isInFavourites),
        removeFavourites = removeFavourites.copy(visible = isInFavourites)
    )
}
```

This is the exact equivalent of:

- if `safeArgs.isInFavourites`
  - `btnRemoveFavourites.visibility = View.VISIBLE`
  - `btnAddFavourites.visibility = View.GONE`
- else
  - `btnRemoveFavourites.visibility = View.GONE`
  - `btnAddFavourites.visibility = View.VISIBLE`

## 3) Compose-friendly ViewModel

```kotlin
package ifac.td.taxi.ui.screen.requeststandreinforcement

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.R
import ifac.td.taxi.framework.sdk.usecase.ZoningUseCase
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

class RequestStandReinforcementViewModel(
    private val zoningUseCase: ZoningUseCase
) : ViewModel() {

    private val _uiState = MutableStateFlow(RequestStandReinforcementUiState())
    val uiState: StateFlow<RequestStandReinforcementUiState> = _uiState.asStateFlow()

    private val _uiEffect = kotlinx.coroutines.flow.MutableSharedFlow<RequestStandReinforcementUiEffect>()
    val uiEffect: SharedFlow<RequestStandReinforcementUiEffect> = _uiEffect.asSharedFlow()

    fun init(isInFavourites: Boolean) {
        _uiState.update {
            it.copy(
                buttonsState = it.buttonsState.reduce(isInFavourites)
            )
        }
    }

    fun onCancelClick() {
        viewModelScope.launch {
            _uiEffect.emit(RequestStandReinforcementUiEffect.NavigateBack)
        }
    }

    fun onAddFavouritesClick(idMacrozone: Int, idZone: Int) {
        viewModelScope.launch {
            zoningUseCase.addFavouriteZone(idMacrozone, idZone)
            _uiEffect.emit(RequestStandReinforcementUiEffect.NavigateBack)
        }
    }

    fun onRemoveFavouritesClick(idMacrozone: Int, idZone: Int) {
        viewModelScope.launch {
            zoningUseCase.removeFavouriteZone(idMacrozone, idZone)
            _uiEffect.emit(RequestStandReinforcementUiEffect.NavigateBack)
        }
    }

    fun onReinforcementClick(idMacrozone: Int, idZone: Int, numCustomers: Int) {
        viewModelScope.launch {
            zoningUseCase.sendReinforcementRequest(idMacrozone, idZone, numCustomers)
            _uiEffect.emit(RequestStandReinforcementUiEffect.ShowToast(R.string.sending))
            _uiEffect.emit(RequestStandReinforcementUiEffect.NavigateBack)
        }
    }
}
```

## 4) MainActivity bridge ViewModel

This is the Compose bridge for the fragment’s `iMainActivity.showHeader(false)`, `showBottomBar(false)`, `showToast(...)`, and `navigateBack()` behavior.

```kotlin
package ifac.td.taxi.ui.screen.requeststandreinforcement

import androidx.lifecycle.ViewModel
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.asSharedFlow

sealed interface RequestStandReinforcementMainActivityEffect {
    data class ShowHeader(val visible: Boolean) : RequestStandReinforcementMainActivityEffect
    data class ShowBottomBar(val visible: Boolean) : RequestStandReinforcementMainActivityEffect
    data class ShowToast(val messageRes: Int) : RequestStandReinforcementMainActivityEffect
    data object NavigateBack : RequestStandReinforcementMainActivityEffect
}

class RequestStandReinforcementMainActivityViewModel : ViewModel() {

    private val _effects = MutableSharedFlow<RequestStandReinforcementMainActivityEffect>()
    val effects: SharedFlow<RequestStandReinforcementMainActivityEffect> = _effects.asSharedFlow()

    suspend fun showHeader(visible: Boolean) {
        _effects.emit(RequestStandReinforcementMainActivityEffect.ShowHeader(visible))
    }

    suspend fun showBottomBar(visible: Boolean) {
        _effects.emit(RequestStandReinforcementMainActivityEffect.ShowBottomBar(visible))
    }

    suspend fun showToast(messageRes: Int) {
        _effects.emit(RequestStandReinforcementMainActivityEffect.ShowToast(messageRes))
    }

    suspend fun navigateBack() {
        _effects.emit(RequestStandReinforcementMainActivityEffect.NavigateBack)
    }
}
```

## 6) Compose wrapper for the custom button API

This is a Compose wrapper that mirrors the custom button behavior closely enough for this screen.

```kotlin
package ifac.td.taxi.ui.screen.requeststandreinforcement

import androidx.compose.foundation.background
import androidx.compose.foundation.clickable
import androidx.compose.foundation.layout.Box
import androidx.compose.foundation.layout.fillMaxSize
import androidx.compose.foundation.layout.padding
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.text.font.FontWeight
import androidx.compose.ui.unit.dp
import androidx.compose.ui.unit.sp
import ifac.td.taxi.ui.custom.button.CustomButton

@Composable
fun RequestStandReinforcementButton(
    state: RequestStandReinforcementButtonVisualState,
    modifier: Modifier = Modifier,
    onClick: () -> Unit
) {
    if (!state.visible) return

    val bgColor = when (state.background) {
        RequestStandReinforcementButtonBackground.GREEN -> Color(0xFF2E7D32)
        RequestStandReinforcementButtonBackground.RED -> Color(0xFFC62828)
        RequestStandReinforcementButtonBackground.BLUE -> Color(0xFF1565C0)
    }

    Box(
        modifier = modifier
            .padding(1.dp)
            .background(bgColor)
            .clickable(onClick = onClick)
    ) {
        Text(
            text = state.text ?: state.textRes?.let { "RES:$it" } ?: "",
            modifier = Modifier
                .align(Alignment.Center)
                .padding(12.dp),
            color = Color.White,
            fontSize = if (state.textOnly) 18.sp else 14.sp,
            fontWeight = FontWeight.Medium
        )
    }
}
```

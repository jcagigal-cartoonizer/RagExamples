Below is a Compose-first replacement that keeps the original behavior, preserves navigation intent, uses a single `SharedFlow<AboutUiEffect>` for one-off events, and adds a dialog state handled from lifecycle-aware collection.

I’ll provide:

1. `AboutScreen` composable
2. `AboutViewModel` rewritten with `UiState + UiEvent + UiEffect`
3. `AboutButtonsState` and styling helpers for the accept button
4. `CustomDialog` composable based on the standard XML/dialog behavior
5. A lifecycle-aware collector pattern that preserves navigation and dialog handling

## 2) ViewModel with `UiState + UiEvent + SharedFlow<AboutUiEffect>`

This replaces the fragment’s direct click logic and the two separate flows with:
- one observable `StateFlow<AboutUiState>`
- one `SharedFlow<AboutUiEffect>`

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import android.content.ActivityNotFoundException
import android.content.Intent
import android.net.Uri
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.R
import ifac.td.taxi.domain.usecase.BluetoothLocalUseCase
import ifac.td.taxi.framework.sdk.model.BluetoothInfo
import ifac.td.taxi.framework.util.ApkUtils
import ifac.td.taxi.ui.screen.about.AboutDialogState
import ifac.td.taxi.ui.screen.about.AboutUiEffect
import ifac.td.taxi.ui.screen.about.AboutUiEvent
import ifac.td.taxi.ui.screen.about.AboutUiState
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

class AboutViewModel(
    context: Application,
    private val bluetoothLocalUseCase: BluetoothLocalUseCase,
) : BaseViewModel(context) {

    private val privacyPolicyUrl = "https://www.taxitronic.com/en/privacy-policy-smart-td/"

    private val _uiState = MutableStateFlow(
        AboutUiState(
            appInfo = buildAppInfo(),
            privacyPolicyText = context.getString(R.string.privacy_policy),
            buttons = AboutButtonsState(),
        )
    )
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<AboutUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()

    private fun buildAppInfo(): String {
        val name = ApkUtils.getAppNameVersionDate(
            name = true, version = false, date = false, context = context
        )
        val version = ApkUtils.getAppNameVersionDate(
            name = false, version = true, date = false, context = context
        )
        val date = ApkUtils.getAppNameVersionDate(
            name = false, version = false, date = true, context = context
        )
        return context.getString(R.string.app_info_format, name, version, date)
    }

    fun onEvent(event: AboutUiEvent) {
        when (event) {
            AboutUiEvent.BackPressed -> emitEffect(AboutUiEffect.NavigateBack)
            AboutUiEvent.PrivacyClicked -> openPrivacy()
            AboutUiEvent.LogoClicked -> handleLogoClick()
            AboutUiEvent.DialogDismissed -> {
                _uiState.update { it.copy(dialog = null) }
            }
        }
    }

    private var counter = 0

    private fun handleLogoClick() {
        counter++
        if (counter == 5) {
            // Keep original behavior
            try {
                com.interfacom.sdk.taximeter.bravocomm.W2CLocation.setTrackingState(
                    com.interfacom.sdk.taximeter.bravocomm.location.AlfaLocation.ESTADO_SEGUIMIENTO_ALARMA,
                    "1"
                )
            } finally {
                counter = 0
            }
        }
    }

    private fun openPrivacy() {
        val intent = Intent(Intent.ACTION_VIEW, Uri.parse(privacyPolicyUrl)).apply {
            flags = Intent.FLAG_ACTIVITY_NEW_TASK
        }

        try {
            context.startActivity(intent)
            emitEffect(AboutUiEffect.OpenPrivacyUrl)
        } catch (_: ActivityNotFoundException) {
            _uiState.update {
                it.copy(
                    dialog = AboutDialogState(
                        title = context.getString(R.string.app_name),
                        message = context.getString(R.string.warning_not_browser),
                        confirmText = context.getString(android.R.string.ok)
                    )
                )
            }
            emitEffect(AboutUiEffect.ShowWarningDialog)
        }
    }

    fun checkSavedDevice() {
        viewModelScope.launch {
            val bluetooth = bluetoothLocalUseCase.getLocalBluetooth()
            val isDeviceAvailable = bluetooth != null

            val bluetoothText = if (isDeviceAvailable && bluetooth != null) {
                buildString {
                    appendLine("${bluetooth.name} [${bluetooth.btPIN}]")
                    // Original fragment also read Taximeter values
                    val taximeter = com.interfacom.sdk.taximeter.taximeter.Taximeter.getInstance()
                    val firmware = taximeter.taximeterVersionFirmware
                    val hardware = taximeter.taximeterVersionHardware
                    if (firmware.isNotEmpty() && hardware.isNotEmpty()) {
                        appendLine("$firmware - $hardware")
                    }
                    val answerK62 = taximeter.answerK62
                    if (answerK62.isNotEmpty()) appendLine(answerK62)
                }
            } else {
                null
            }

            _uiState.update {
                it.copy(
                    showBluetoothInfo = isDeviceAvailable && bluetoothText != null,
                    bluetoothText = bluetoothText
                )
            }
        }
    }

    private fun emitEffect(effect: AboutUiEffect) {
        viewModelScope.launch { _uiEffect.emit(effect) }
    }
}

data class AboutButtonsState(
    val accept: AboutButtonStyle = AboutButtonStyle.accept()
)

data class AboutButtonStyle(
    val text: String,
    val visible: Boolean = true,
    val enabled: Boolean = true,
    val backgroundColor: androidx.compose.ui.graphics.Color,
    val contentColor: androidx.compose.ui.graphics.Color,
    val borderColor: androidx.compose.ui.graphics.Color,
) {
    companion object {
        fun accept() = AboutButtonStyle(
            text = "Accept",
            visible = true,
            enabled = true,
            backgroundColor = androidx.compose.ui.graphics.Color(0xFF000000), // customize to match XML/theme
            contentColor = androidx.compose.ui.graphics.Color.White,
            borderColor = androidx.compose.ui.graphics.Color(0xFF000000)
        )
    }
}
```

## 3) Compose screen

This screen:
- collects `uiState`
- collects `uiEffect` lifecycle-aware
- preserves navigation by calling `navigateBack()`
- shows a dialog when needed
- renders the privacy policy as underlined text
- renders the app info and Bluetooth info
- reproduces the logo click counter behavior

```kotlin
package ifac.td.taxi.ui.screen.about

import androidx.compose.foundation.Image
import androidx.compose.foundation.clickable
import androidx.compose.foundation.layout.*
import androidx.compose.foundation.text.ClickableText
import androidx.compose.material3.Text
import androidx.compose.runtime.*
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.platform.LocalContext
import androidx.compose.ui.res.painterResource
import androidx.compose.ui.res.stringResource
import androidx.compose.ui.text.SpanStyle
import androidx.compose.ui.text.buildAnnotatedString
import androidx.compose.ui.text.style.TextDecoration
import androidx.compose.ui.unit.dp
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import ifac.td.taxi.R
import ifac.td.taxi.viewmodel.AboutViewModel
import kotlinx.coroutines.flow.collectLatest

@Composable
fun AboutScreen(
    viewModel: AboutViewModel,
    navigateBack: () -> Unit,
    showToast: (Int) -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    LaunchedEffect(Unit) {
        viewModel.checkSavedDevice()
    }

    LaunchedEffect(viewModel) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                AboutUiEffect.NavigateBack -> navigateBack()
                AboutUiEffect.OpenPrivacyUrl -> Unit
                AboutUiEffect.ShowWarningDialog -> {
                    // dialog already represented in state
                }
            }
        }
    }

    Box(Modifier.fillMaxSize()) {
        AboutContent(
            state = uiState,
            onEvent = viewModel::onEvent
        )

        uiState.dialog?.let { dialog ->
            CustomDialog(
                title = dialog.title,
                message = dialog.message,
                confirmText = dialog.confirmText,
                onConfirm = { viewModel.onEvent(AboutUiEvent.DialogDismissed) },
                onDismiss = { viewModel.onEvent(AboutUiEvent.DialogDismissed) }
            )
        }
    }
}

@Composable
private fun AboutContent(
    state: AboutUiState,
    onEvent: (AboutUiEvent) -> Unit
) {
    Column(
        modifier = Modifier
            .fillMaxSize()
            .padding(16.dp)
    ) {
        Spacer(Modifier.weight(1f))

        Text(
            text = buildAnnotatedString {
                pushStringAnnotation(tag = "privacy", annotation = "privacy")
                withStyle(
                    style = SpanStyle(
                        textDecoration = TextDecoration.Underline,
                        color = Color.Unspecified
                    )
                ) {
                    append(state.privacyPolicyText)
                }
                pop()
            },
            modifier = Modifier.clickable { onEvent(AboutUiEvent.PrivacyClicked) }
        )

        Spacer(Modifier.height(8.dp))

        Text(
            text = stringResource(id = R.string.powered_by),
            modifier = Modifier.align(androidx.compose.ui.Alignment.CenterHorizontally)
        )

        Spacer(Modifier.height(8.dp))

        Image(
            painter = painterResource(id = R.drawable.logotaxitronic),
            contentDescription = stringResource(id = R.string.logo_taxitronic),
            modifier = Modifier
                .fillMaxWidth()
                .height(120.dp)
                .clickable { onEvent(AboutUiEvent.LogoClicked) }
        )

        if (state.showBluetoothInfo && !state.bluetoothText.isNullOrBlank()) {
            Spacer(Modifier.height(16.dp))
            Text(text = state.bluetoothText)
        }

        Spacer(Modifier.weight(1f))

        if (state.buttons.accept.visible) {
            ComposeAcceptButton(
                style = state.buttons.accept,
                onClick = { onEvent(AboutUiEvent.BackPressed) }
            )
        }
    }
}
```

## 4) Button styling helpers and custom button replacement

This is the Compose version of the custom button component, with styling fields that let you match XML more closely.

```kotlin
package ifac.td.taxi.ui.screen.about

import androidx.compose.foundation.BorderStroke
import androidx.compose.foundation.layout.fillMaxWidth
import androidx.compose.foundation.layout.heightIn
import androidx.compose.material3.Button
import androidx.compose.material3.ButtonDefaults
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.unit.dp

@Composable
fun ComposeAcceptButton(
    style: AboutButtonStyle,
    onClick: () -> Unit
) {
    Button(
        onClick = onClick,
        enabled = style.enabled,
        modifier = Modifier
            .fillMaxWidth()
            .heightIn(min = 56.dp),
        colors = ButtonDefaults.buttonColors(
            containerColor = style.backgroundColor,
            contentColor = style.contentColor,
            disabledContainerColor = style.backgroundColor.copy(alpha = 0.4f),
            disabledContentColor = style.contentColor.copy(alpha = 0.4f)
        ),
        border = BorderStroke(1.dp, style.borderColor)
    ) {
        Text(text = style.text)
    }
}
```

## 5) CustomDialog composable

This is a Compose dialog matching the role of your `custom_dialog.xml` and `CustomDialog.kt`:
- title
- message
- confirm button
- dismiss behavior
- lifecycle-safe state-driven handling

```kotlin
package ifac.td.taxi.ui.screen.about

import androidx.compose.foundation.background
import androidx.compose.foundation.layout.*
import androidx.compose.material3.Button
import androidx.compose.material3.Divider
import androidx.compose.material3.Surface
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.unit.dp
import androidx.compose.ui.window.Dialog

@Composable
fun CustomDialog(
    title: String,
    message: String,
    confirmText: String,
    onConfirm: () -> Unit,
    onDismiss: () -> Unit,
) {
    Dialog(onDismissRequest = onDismiss) {
        Surface(
            tonalElevation = 8.dp,
            shape = androidx.compose.foundation.shape.RoundedCornerShape(16.dp),
            color = Color.White
        ) {
            Column(modifier = Modifier.padding(20.dp)) {
                Text(text = title)
                Spacer(Modifier.height(12.dp))
                Divider()
                Spacer(Modifier.height(12.dp))
                Text(text = message)
                Spacer(Modifier.height(20.dp))
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = androidx.compose.foundation.layout.Arrangement.End
                ) {
                    Button(onClick = onConfirm) {
                        Text(text = confirmText)
                    }
                }
            }
        }
    }
}
```

## 6) Notes on preserving original behavior

### Preserved behavior
- `navigateBack()` is still called when the accept button is pressed
- privacy click still opens the URL
- if no browser exists, warning dialog is shown
- logo click counter still triggers `W2CLocation.setTrackingState(...)` on the 5th click
- saved Bluetooth device info is still loaded and rendered
- app name/version/date text is still assembled

### What changed
- fragment click logic is now replaced by `AboutUiEvent`
- one-off events are emitted via `SharedFlow<AboutUiEffect>`
- dialog is state-driven via `AboutUiState.dialog`
- lifecycle collection is done with Compose and `collectAsStateWithLifecycle`

## 7) If you want a stricter XML-like button match

If your original `CustomButton` had exact colors, corner radius, margins, typography, or pressed states, you can encode them in `AboutButtonStyle` and implement a more exact `ComposeAcceptButton`.

For example:

```kotlin
data class AboutButtonStyle(
    val text: String,
    val visible: Boolean = true,
    val enabled: Boolean = true,
    val backgroundColor: Color,
    val contentColor: Color,
    val borderColor: Color,
    val cornerRadiusDp: Int = 12,
    val textAllCaps: Boolean = false
)
```

And use `shape = RoundedCornerShape(style.cornerRadiusDp.dp)` in the `Button`.

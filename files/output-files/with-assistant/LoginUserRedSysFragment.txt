## 2) Exact reducer mirroring the fragment branches one-by-one

This reducer mirrors the fragment logic exactly:

- `loginRedSysFlow == true` → toast + navigate back
- `loginRedSysFlow == false` → set user error, focus, enable accept button, green background
- `userRedSysFlow` → populate user/password fields
- loading state when accept is clicked

```kotlin
package ifac.td.taxi.ui.screen

import ifac.td.taxi.R
import ifac.td.taxi.ui.custom.button.CustomButton

fun LoginUserRedSysButtonsState.reduce(
    effect: LoginUserRedSysUiEffect? = null,
    isLoading: Boolean = false
): LoginUserRedSysButtonsState {
    return when (effect) {
        is LoginUserRedSysUiEffect.NavigateBack -> copy(
            cancel = cancel.copy(
                style = CustomButton.StyleButton.ENABLE,
                background = CustomButton.BackgroundButtonColor.RED,
                visible = true
            ),
            accept = accept.copy(
                style = CustomButton.StyleButton.ENABLE,
                background = CustomButton.BackgroundButtonColor.GREEN,
                visible = true
            )
        )

        is LoginUserRedSysUiEffect.ShowToast -> copy(
            cancel = cancel.copy(
                style = CustomButton.StyleButton.ENABLE,
                background = CustomButton.BackgroundButtonColor.RED,
                visible = true
            ),
            accept = accept.copy(
                style = if (isLoading) CustomButton.StyleButton.LOADING else CustomButton.StyleButton.ENABLE,
                background = CustomButton.BackgroundButtonColor.GREEN,
                visible = true
            )
        )

        is LoginUserRedSysUiEffect.NavigateToChangePassword -> copy(
            cancel = cancel.copy(
                style = CustomButton.StyleButton.ENABLE,
                background = CustomButton.BackgroundButtonColor.RED,
                visible = true
            ),
            accept = accept.copy(
                style = CustomButton.StyleButton.ENABLE,
                background = CustomButton.BackgroundButtonColor.GREEN,
                visible = true
            )
        )

        null -> copy(
            accept = accept.copy(
                style = if (isLoading) CustomButton.StyleButton.LOADING else CustomButton.StyleButton.ENABLE,
                background = CustomButton.BackgroundButtonColor.GREEN,
                visible = true
            )
        )
    }
}
```

This is the exact branch mapping from the fragment:
- success branch does not alter button state except the screen leaves
- failure branch explicitly re-enables accept and sets green background
- loading branch sets accept to loading before the request

## 3) Compose-friendly ViewModel with `UiState + UiEvent + SharedFlow<LoginUserRedSysUiEffect>`

This replaces the old multiple `SharedFlow<Boolean>` / `SharedFlow<String?>` with a single state + effect architecture, while still preserving the original behavior.

```kotlin
package ifac.td.taxi.ui.screen

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import es.redsys.paysys.Utils.RedCLSErrorCodes
import ifac.td.taxi.R
import ifac.td.taxi.domain.model.RedSysLoginResponse
import ifac.td.taxi.domain.usecase.RedSysUseCase
import ifac.td.taxi.framework.TemporalData
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.ui.custom.button.CustomButton
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

class LoginUserRedSysViewModel(
    private val redSysUseCase: RedSysUseCase,
    application: Application
) : AndroidViewModel(application) {

    private val TAG = "LoginUserRedSysViewModel"

    private val _uiState = MutableStateFlow(LoginUserRedSysUiState())
    val uiState: StateFlow<LoginUserRedSysUiState> = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<LoginUserRedSysUiEffect>()
    val uiEffect: SharedFlow<LoginUserRedSysUiEffect> = _uiEffect.asSharedFlow()

    init {
        onEvent(LoginUserRedSysUiEvent.ScreenStarted)
    }

    fun onEvent(event: LoginUserRedSysUiEvent) {
        when (event) {
            LoginUserRedSysUiEvent.ScreenStarted -> getUserRedSys()

            is LoginUserRedSysUiEvent.UserChanged -> {
                _uiState.update { it.copy(user = event.value, userError = null) }
            }

            is LoginUserRedSysUiEvent.PasswordChanged -> {
                _uiState.update { it.copy(password = event.value) }
            }

            LoginUserRedSysUiEvent.CancelClicked -> {
                viewModelScope.launch {
                    _uiEffect.emit(LoginUserRedSysUiEffect.NavigateBack)
                }
            }

            LoginUserRedSysUiEvent.AcceptClicked -> clickOnAccept()

            LoginUserRedSysUiEvent.DismissDialog -> {
                _uiState.update { it.copy(dialogState = LoginUserRedSysDialogState.Hidden) }
            }
        }
    }

    private fun clickOnAccept() {
        val current = _uiState.value
        if (current.user.isNotBlank() && current.password.isNotBlank()) {
            _uiState.update {
                it.copy(
                    isLoading = true,
                    buttonsState = it.buttonsState.reduce(isLoading = true)
                )
            }
            loginRedSys(
                user = current.user,
                password = current.password
            )
        }
    }

    fun loginRedSys(user: String, password: String) {
        viewModelScope.launch(Dispatchers.IO) {
            redSysUseCase.configureRedSys()

            val loginRedSys = redSysUseCase.loginRedSys(
                username = user,
                password = password
            )

            when (loginRedSys) {
                is RedSysLoginResponse.Success -> {
                    if (loginRedSys.merchantList.isNotEmpty()) {
                        redSysUseCase.saveUsernameRedSys(user)
                        if (password.isNotEmpty()) {
                            TemporalData.passwordRedSys = password
                        }
                        _uiState.update {
                            it.copy(
                                isLoading = false,
                                buttonsState = it.buttonsState.reduce(
                                    effect = LoginUserRedSysUiEffect.NavigateBack
                                )
                            )
                        }
                        _uiEffect.emit(LoginUserRedSysUiEffect.ShowToast(R.string.toast_red_sys_login_ok))
                        _uiEffect.emit(LoginUserRedSysUiEffect.NavigateBack)
                    } else {
                        emitLoginFailure()
                    }
                }

                is RedSysLoginResponse.Error -> {
                    Logs.d(TAG, "Login RedSys failed or returned empty array: ${loginRedSys.errorMessage}")
                    if (loginRedSys.errorCode == RedCLSErrorCodes.STATUS_KO_FORMATO_RESP_LOGIN_PWD_CAD) {
                        _uiState.update {
                            it.copy(
                                isLoading = false,
                                buttonsState = it.buttonsState.reduce(
                                    effect = LoginUserRedSysUiEffect.NavigateToChangePassword
                                )
                            )
                        }
                        _uiEffect.emit(LoginUserRedSysUiEffect.NavigateToChangePassword)
                    } else {
                        emitLoginFailure()
                    }
                }
            }
        }
    }

    private suspend fun emitLoginFailure() {
        _uiState.update {
            it.copy(
                isLoading = false,
                userError = getApplication<Application>().getString(R.string.incorrect_login),
                buttonsState = it.buttonsState.reduce()
            )
        }
    }

    fun getUserRedSys() {
        viewModelScope.launch {
            val user = redSysUseCase.getUsernameRedSys()
            _uiState.update {
                it.copy(
                    user = user.orEmpty(),
                    password = TemporalData.passwordRedSys
                )
            }
        }
    }
}
```

```kotlin
            AlertDialog(
                onDismissRequest = { viewModel.onEvent(LoginUserRedSysUiEvent.DismissDialog) },
                title = { Text(dialog.title) },
                text = { Text(dialog.message) },
                confirmButton = {
                    TextButton(onClick = { viewModel.onEvent(LoginUserRedSysUiEvent.DismissDialog) }) {
                        Text(stringResource(id = android.R.string.ok))
                    }
                }
            )
        }
    }

    if (state.userError != null) {
        // This mirrors the fragment's edUser.error assignment.
        // In Compose, the error is rendered in the text field itself.
        val _ = context
    }
}

@Composable
```
```kotlin
private fun LoginUserRedSysCustomButton(
    modifier: Modifier = Modifier,
    text: String,
    visualState: LoginUserRedSysButtonVisualState,
    onClick: () -> Unit
) {
    if (!visualState.visible) return

    val bgColor = when (visualState.background) {
        CustomButton.BackgroundButtonColor.GREEN -> Color(0xFF2E7D32)
        CustomButton.BackgroundButtonColor.RED -> Color(0xFFC62828)
        CustomButton.BackgroundButtonColor.BLUE -> Color(0xFF1565C0)
        CustomButton.BackgroundButtonColor.YELLOW -> Color(0xFFF9A825)
        CustomButton.BackgroundButtonColor.ORANGE -> Color(0xFFEF6C00)
        CustomButton.BackgroundButtonColor.DEFAULT -> Color(0xFF1565C0)
        CustomButton.BackgroundButtonColor.BLACK -> Color.Black
        CustomButton.BackgroundButtonColor.TRANSPARENT -> Color.Transparent
    }

    Button(
        modifier = modifier,
        onClick = onClick,
        enabled = visualState.style == CustomButton.StyleButton.ENABLE,
        colors = ButtonDefaults.buttonColors(
            containerColor = bgColor,
            disabledContainerColor = Color(0xFF9E9E9E)
        )
    ) {
        when (visualState.style) {
            CustomButton.StyleButton.LOADING -> {
                CircularProgressIndicator()
            }
            else -> {
                Text(text = text)
            }
        }
    }
```

```kotlin
    fun onEvent(event: LoginUserRedSysUiEvent) {
        when (event) {
            LoginUserRedSysUiEvent.ScreenStarted -> getUserRedSys()

            is LoginUserRedSysUiEvent.UserChanged -> {
                _uiState.update { it.copy(user = event.value, userError = null) }
            }

            is LoginUserRedSysUiEvent.PasswordChanged -> {
                _uiState.update { it.copy(password = event.value) }
            }

            LoginUserRedSysUiEvent.CancelClicked -> {
                viewModelScope.launch {
                    _uiEffect.emit(LoginUserRedSysUiEffect.NavigateBack)
                }
            }

            LoginUserRedSysUiEvent.AcceptClicked -> clickOnAccept()

            LoginUserRedSysUiEvent.DismissDialog -> {
                _uiState.update { it.copy(dialogState = LoginUserRedSysDialogState.Hidden) }
            }
        }
    }

    private fun clickOnAccept() {
        val current = _uiState.value
        if (current.user.isNotBlank() && current.password.isNotBlank()) {
            _uiState.update {
                it.copy(
                    isLoading = true,
                    buttonsState = it.buttonsState.reduce(isLoading = true)
                )
            }
            loginRedSys(
                user = current.user,
                password = current.password
            )
        }
    }

    fun loginRedSys(user: String, password: String) {
        viewModelScope.launch(Dispatchers.IO) {
            redSysUseCase.configureRedSys()

            val loginRedSys = redSysUseCase.loginRedSys(
                username = user,
                password = password
            )

            when (loginRedSys) {
                is RedSysLoginResponse.Success -> {
                    if (loginRedSys.merchantList.isNotEmpty()) {
                        redSysUseCase.saveUsernameRedSys(user)
                        if (password.isNotEmpty()) {
                            TemporalData.passwordRedSys = password
                        }
                        _uiState.update {
                            it.copy(
                                isLoading = false,
                                buttonsState = it.buttonsState.reduce(
                                    effect = LoginUserRedSysUiEffect.NavigateBack
                                )
                            )
                        }
                        _uiEffect.emit(LoginUserRedSysUiEffect.ShowToast(R.string.toast_red_sys_login_ok))
                        _uiEffect.emit(LoginUserRedSysUiEffect.NavigateBack)
                    } else {
                        emitLoginFailure()
                    }
                }

                is RedSysLoginResponse.Error -> {
                    Logs.d(TAG, "Login RedSys failed or returned empty array: ${loginRedSys.errorMessage}")
                    if (loginRedSys.errorCode == RedCLSErrorCodes.STATUS_KO_FORMATO_RESP_LOGIN_PWD_CAD) {
                        _uiState.update {
                            it.copy(
                                isLoading = false,
                                buttonsState = it.buttonsState.reduce(
                                    effect = LoginUserRedSysUiEffect.NavigateToChangePassword
                                )
                            )
                        }
                        _uiEffect.emit(LoginUserRedSysUiEffect.NavigateToChangePassword)
                    } else {
                        emitLoginFailure()
                    }
                }
            }
        }
    }

    private suspend fun emitLoginFailure() {
        _uiState.update {
            it.copy(
                isLoading = false,
                userError = getApplication<Application>().getString(R.string.incorrect_login),
                buttonsState = it.buttonsState.reduce()
            )
        }
    }

    fun getUserRedSys() {
        viewModelScope.launch {
            val user = redSysUseCase.getUsernameRedSys()
            _uiState.update {
                it.copy(
                    user = user.orEmpty(),
                    password = TemporalData.passwordRedSys
                )
            }
        }
    }
}

@Composable
private fun LoginUserRedSysCustomButton(
    modifier: Modifier = Modifier,
    text: String,
    visualState: LoginUserRedSysButtonVisualState,
    onClick: () -> Unit
) {
    if (!visualState.visible) return

    val bgColor = when (visualState.background) {
        CustomButton.BackgroundButtonColor.GREEN -> Color(0xFF2E7D32)
        CustomButton.BackgroundButtonColor.RED -> Color(0xFFC62828)
        CustomButton.BackgroundButtonColor.BLUE -> Color(0xFF1565C0)
        CustomButton.BackgroundButtonColor.YELLOW -> Color(0xFFF9A825)
        CustomButton.BackgroundButtonColor.ORANGE -> Color(0xFFEF6C00)
        CustomButton.BackgroundButtonColor.DEFAULT -> Color(0xFF1565C0)
        CustomButton.BackgroundButtonColor.BLACK -> Color.Black
        CustomButton.BackgroundButtonColor.TRANSPARENT -> Color.Transparent
    }

    Button(
        modifier = modifier,
        onClick = onClick,
        enabled = visualState.style == CustomButton.StyleButton.ENABLE,
        colors = ButtonDefaults.buttonColors(
            containerColor = bgColor,
            disabledContainerColor = Color(0xFF9E9E9E)
        )
    ) {
        when (visualState.style) {
            CustomButton.StyleButton.LOADING -> CircularProgressIndicator()
            else -> Text(text = text)
        }
    }
}
```

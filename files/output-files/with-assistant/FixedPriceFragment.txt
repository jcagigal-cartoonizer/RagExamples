
// FIRST ANSWER
Below is a Compose-first rewrite that preserves the fragment behavior, keeps navigation host-driven, and replaces the fragment’s imperative UI logic with:

- `FixedPriceUiState`
- `FixedPriceUiEvent`
- `FixedPriceUiEffect` via `SharedFlow`
- `FixedPriceButtonsState` reducer mirroring the fragment’s `collect {}` branches one-by-one
- a `FixedPriceMainActivityViewModel` bridge for shared flows
- a `FixedPriceMapScreen` composable that matches the XML structure and behavior as closely as possible

I’m also including a small Compose wrapper for the map view, because `CercaliaMapView` is still a View-based SDK component.

## 2) Exact `FixedPriceButtonsState` reducer

This reducer mirrors the fragment’s `collect {}` branches one-by-one.

```kotlin
package ifac.td.taxi.ui.screen.fixedprice

import ifac.td.taxi.repository.connections.rest.bravoRest.SuggestModel

object FixedPriceButtonsReducer {

    fun reduce(
        current: FixedPriceButtonsState,
        event: FixedPriceUiEvent
    ): FixedPriceButtonsState {
        return when (event) {
            is FixedPriceUiEvent.OnSuggestionClicked -> {
                current.copy(
                    rvVisible = false,
                    bottomSheetVisible = true,
                    closeVisible = true,
                    suggestVisible = true,
                    dropOffText = event.item.getPrintableStreetTextShort()
                )
            }

            is FixedPriceUiEvent.OnCloseClicked -> {
                current.copy(
                    dropOffText = "",
                    rvVisible = false,
                    bottomSheetVisible = false,
                    taxiPriceText = "",
                    vanPriceText = "",
                    businessPriceText = "",
                    taxiPriceVisible = false,
                    vanPriceVisible = false,
                    businessPriceVisible = false,
                    taxiProgressVisible = true,
                    vanProgressVisible = true,
                    businessProgressVisible = true,
                    taxiErrorVisible = false,
                    vanErrorVisible = false,
                    businessErrorVisible = false
                )
            }

            is FixedPriceUiEvent.OnErrorChanged -> {
                if (event.error == true) {
                    current.copy(
                        taxiErrorVisible = true,
                        vanErrorVisible = true,
                        businessErrorVisible = true,
                        taxiProgressVisible = false,
                        vanProgressVisible = false,
                        businessProgressVisible = false,
                        taxiPriceText = "",
                        vanPriceText = "",
                        businessPriceText = "",
                        taxiPriceVisible = false,
                        vanPriceVisible = false,
                        businessPriceVisible = false
                    )
                } else {
                    current.copy(
                        taxiErrorVisible = false,
                        vanErrorVisible = false,
                        businessErrorVisible = false
                    )
                }
            }

            is FixedPriceUiEvent.OnFixedPriceApiChanged -> {
                current.copy(
                    taxiPriceText = event.data.taxiPrice.orEmpty(),
                    vanPriceText = event.data.vanPrice.orEmpty(),
                    businessPriceText = event.data.businessPrice.orEmpty(),
                    taxiPriceVisible = event.data.taxiPrice != null,
                    vanPriceVisible = event.data.vanPrice != null,
                    businessPriceVisible = event.data.businessPrice != null,
                    taxiProgressVisible = event.data.taxiPrice == null,
                    vanProgressVisible = event.data.vanPrice == null,
                    businessProgressVisible = event.data.businessPrice == null,
                    taxiErrorVisible = event.data.taxiError,
                    vanErrorVisible = event.data.vanError,
                    businessErrorVisible = event.data.businessError
                )
            }

            is FixedPriceUiEvent.OnFixedPriceDbChanged -> {
                val data = event.data
                if (data == null) {
                    current
                } else {
                    current.copy(
                        taxiPriceText = data.taxiPrice.orEmpty(),
                        vanPriceText = data.vanPrice.orEmpty(),
                        businessPriceText = data.businessPrice.orEmpty(),
                        taxiPriceVisible = data.taxiPrice != null,
                        vanPriceVisible = data.vanPrice != null,
                        businessPriceVisible = data.businessPrice != null,
                        taxiProgressVisible = data.taxiPrice == null,
                        vanProgressVisible = data.vanPrice == null,
                        businessProgressVisible = data.businessPrice == null,
                        taxiErrorVisible = data.taxiPrice == null,
                        vanErrorVisible = data.vanPrice == null,
                        businessErrorVisible = data.businessPrice == null
                    )
                }
            }

            else -> current
        }
    }
}
```

## 3) Compose-friendly `FixedPriceViewModel`

This version exposes a single `uiState` and a single `uiEffect` stream.

```kotlin
package ifac.td.taxi.viewmodel.fixedprice

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import com.interfacom.sdk.taximeter.bravocomm.rest.poi.response.Poi
import com.interfacom.sdk.taximeter.bravocomm.rest.tsab_rest.FixedPriceCallback
import com.interfacom.sdk.taximeter.bravocomm.rest.tsab_rest.fixed_price.PriceEstimationDTO
import com.interfacom.sdk.taximeter.licensing.rest.pois.POISDataModule
import com.interfacom.sdk.taximeter.licensing.rest.pois.POISListener
import com.nexusgeographics.cercalia.maps.features.Marker
import com.nexusgeographics.cercalia.maps.model.LatLng
import ifac.td.taxi.R
import ifac.td.taxi.domain.model.Trip
import ifac.td.taxi.domain.usecase.BravoRestApiUseCase
import ifac.td.taxi.domain.usecase.DispatchUseCase
import ifac.td.taxi.domain.usecase.FixedPriceUseCase
import ifac.td.taxi.domain.usecase.LocationUseCase
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.repository.connections.rest.bravoRest.SuggestModel
import ifac.td.taxi.ui.screen.fixedprice.*
import ifac.td.taxi.ui.util.CoordenatesUtil
import ifac.td.taxi.viewmodel.model.InfoDispatchModel
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch

class FixedPriceViewModel(
    private val locationUseCase: LocationUseCase,
    private val fixedPriceUseCase: FixedPriceUseCase,
    private val cercaliaUseCase: BravoRestApiUseCase,
    private val dispatchUseCase: DispatchUseCase,
    application: Application,
) : AndroidViewModel(application) {

    private val TAG = "FixedPriceViewModel"

    private val _uiState = MutableStateFlow(FixedPriceUiState())
    val uiState: StateFlow<FixedPriceUiState> = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<FixedPriceUiEffect>()
    val uiEffect: SharedFlow<FixedPriceUiEffect> = _uiEffect.asSharedFlow()

    var marker: Marker? = null

    private var lastRequestedTrip: Trip? = null

    private val fixedPriceCallback = object : FixedPriceCallback {
        override fun onCompleted(response: PriceEstimationDTO?) {
            response?.let {
                viewModelScope.launch {
                    lastRequestedTrip?.let { trip ->
                        fixedPriceUseCase.saveFixedPriceData(trip, response)
                    }
                    _uiState.update {
                        it.copy(
                            fixedPriceData = FixedPricePriceState(
                                taxiPrice = response.prices.find { p -> p.serviceType == "taxi" }?.finalPrice?.toString(),
                                vanPrice = response.prices.find { p -> p.serviceType == "largeCar" }?.finalPrice?.toString(),
                                businessPrice = response.prices.find { p -> p.serviceType == "business" }?.finalPrice?.toString(),
                                taxiLoading = false,
                                vanLoading = false,
                                businessLoading = false,
                                taxiError = false,
                                vanError = false,
                                businessError = false
                            )
                        )
                    }
                }
            }
        }

        override fun onError(p0: String?) {
            viewModelScope.launch {
                _uiState.update { it.copy(errorCall = true) }
                _uiEffect.emit(FixedPriceUiEffect.ShowToast(R.string.toast_error_mostrar_mapa))
            }
        }
    }

    private val localCallback: (Triple<Double?, Double?, Double?>) -> Unit = { data ->
        viewModelScope.launch {
            _uiState.update {
                it.copy(
                    fixedPriceDataFromDb = FixedPriceDbState(
                        taxiPrice = data.first?.toString(),
                        vanPrice = data.second?.toString(),
                        businessPrice = data.third?.toString()
                    )
                )
            }
        }
    }

    private val poiListener = object : POISListener {
        override fun getPOISFailure() {
            Logs.e(TAG, "getPOISFailure")
        }

        override fun getPOISSuccess(data: ArrayList<Poi>?) {
            viewModelScope.launch {
                val suggestions = data?.map {
                    SuggestModel(
                        it.poi, it.locality, it.municipalyity, it.province,
                        it.street, it.coord, null, 0.0f
                    )
                }.orEmpty()
                _uiState.update {
                    it.copy(
                        suggestions = suggestions,
                        isSuggestionListVisible = suggestions.isNotEmpty()
                    )
                }
            }
        }
    }

    fun onEvent(event: FixedPriceUiEvent) {
        _uiState.update { current ->
            when (event) {
                is FixedPriceUiEvent.OnManualTextChanged -> current.copy(manualText = event.text)
                is FixedPriceUiEvent.OnPoiSearchChanged -> current.copy(isPoiSearch = event.isPoi)
                is FixedPriceUiEvent.OnSuggestionClicked -> current.copy(
                    selectedDropOff = event.item,
                    manualText = event.item.getPrintableStreetTextShort(),
                    isSuggestionListVisible = false,
                    bottomSheetExpanded = true
                )
                is FixedPriceUiEvent.OnCloseClicked -> current.copy(
                    selectedDropOff = null,
                    manualText = "",
                    bottomSheetExpanded = false,
                    isSuggestionListVisible = false
                )
                is FixedPriceUiEvent.OnBottomSheetHidden -> current.copy(
                    selectedDropOff = null,
                    bottomSheetExpanded = false
                )
                is FixedPriceUiEvent.OnErrorChanged -> current.copy(errorCall = event.error)
                is FixedPriceUiEvent.OnFixedPriceDbChanged -> current.copy(fixedPriceDataFromDb = event.data)
                is FixedPriceUiEvent.OnFixedPriceApiChanged -> current.copy(fixedPriceData = event.data)
                is FixedPriceUiEvent.OnTripLoaded -> current.copy(trip = event.trip)
                is FixedPriceUiEvent.OnDispatchLoaded -> current.copy(dispatch = event.dispatch)
                is FixedPriceUiEvent.OnMapReady -> current.copy(mapReady = true)
                is FixedPriceUiEvent.OnMapPickupResolved -> current.copy(mapPickup = event.pickup)
                is FixedPriceUiEvent.OnMapDropOffResolved -> current.copy(mapDropOff = event.dropOff)
            }
        }

        when (event) {
            is FixedPriceUiEvent.OnManualTextChanged -> {
                if (event.text.length < 3) {
                    viewModelScope.launch {
                        _uiState.update { it.copy(suggestions = emptyList(), isSuggestionListVisible = false) }
                    }
                    return
                }
                suggestAddresses(event.text, _uiState.value.isPoiSearch)
            }

            is FixedPriceUiEvent.OnSuggestionClicked -> {
                updateSelectedDropOff(event.item)
            }

            is FixedPriceUiEvent.OnCloseClicked -> {
                resetFixedPriceFlowFromDB()
                resetErrorFlow()
            }

            else -> Unit
        }
    }

    fun suggestAddresses(text: String, isPoiSearch: Boolean) {
        viewModelScope.launch(Dispatchers.IO) {
            if (isPoiSearch) {
                POISDataModule.providePOISPresenter(getApplication(), poiListener).getPois(text)
            } else {
                cercaliaUseCase.suggest(text) { list ->
                    viewModelScope.launch {
                        _uiState.update {
                            it.copy(suggestions = list, isSuggestionListVisible = list.isNotEmpty())
                        }
                    }
                }
            }
        }
    }

    fun updateSelectedDropOff(itemClick: SuggestModel?) {
        viewModelScope.launch {
            _uiState.update { it.copy(selectedDropOff = itemClick) }
        }
    }

    fun resetFixedPriceFlowFromDB() {
        viewModelScope.launch {
            _uiState.update { it.copy(fixedPriceDataFromDb = null) }
        }
    }

    fun getFixedPriceInVacant(suggest: SuggestModel) {
        viewModelScope.launch {
            fixedPriceUseCase.getFixedPriceInVacant(suggest, fixedPriceCallback, localCallback)
        }
    }

    fun getFixedPriceFromTrip(trip: Trip, suggest: SuggestModel) {
        viewModelScope.launch {
            lastRequestedTrip = trip
            fixedPriceUseCase.getFixedPriceFromTrip(trip, suggest, fixedPriceCallback, localCallback)
        }
    }

    fun getFixedPriceFromDispatch(trip: Trip, dispatch: InfoDispatchModel, suggest: SuggestModel?) {
        viewModelScope.launch {
            lastRequestedTrip = trip
            fixedPriceUseCase.getFixedPriceFromDispatch(trip, dispatch, suggest, fixedPriceCallback, localCallback)
        }
    }

    fun getDispatchDestinationCoordinates(value: InfoDispatchModel?) {
        viewModelScope.launch {
            value?.let {
                val dispatchWithAddresses = dispatchUseCase.getDispatchWithAddressById(value.id)
                dispatchWithAddresses?.let { data ->
                    val lastDestination = data.addresses?.findLast { !it.isPickup }
                    val pickUp = data.addresses?.find { it.isPickup }
                    if (lastDestination != null) {
                        val coordinates = CoordenatesUtil.parseCoordinatesFromLatLong(lastDestination.coordenadas ?: "")
                        val pickUpCoordinates = CoordenatesUtil.parseCoordinatesFromLatLong(pickUp?.coordenadas ?: "")
                        if (coordinates != null && pickUpCoordinates != null) {
                            _uiEffect.emit(
                                FixedPriceUiEffect.MoveMapToDropOff(
                                    LatLng(coordinates.Lat!!, coordinates.Lng!!),
                                    null
                                )
                            )
                        }
                    } else {
                        val pickUpCoordinates = CoordenatesUtil.parseCoordinatesFromLatLong(pickUp?.coordenadas ?: "")
                        if (pickUpCoordinates != null) {
                            _uiEffect.emit(
                                FixedPriceUiEffect.MoveMapToPickup(
                                    LatLng(pickUpCoordinates.Lat!!, pickUpCoordinates.Lng!!),
                                    true
                                )
                            )
                        }
                    }
                }
            }
        }
    }

    fun resetErrorFlow() {
        viewModelScope.launch {
            _uiState.update { it.copy(errorCall = null) }
        }
    }
}
```

## 4) `FixedPriceMainActivityViewModel` Compose bridge

This is the shared-flow bridge that preserves the fragment’s “shared view model” behavior.

```kotlin
package ifac.td.taxi.viewmodel.fixedprice

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.domain.model.Trip
import ifac.td.taxi.viewmodel.model.InfoDispatchModel
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch

class FixedPriceMainActivityViewModel : ViewModel() {

    private val _tripFlow = MutableStateFlow<Trip?>(null)
    val tripFlow = _tripFlow.asStateFlow()

    private val _dispatchFlow = MutableStateFlow<InfoDispatchModel?>(null)
    val dispatchFlow = _dispatchFlow.asStateFlow()

    fun setTrip(value: Trip?) {
        viewModelScope.launch { _tripFlow.emit(value) }
    }

    fun setDispatch(value: InfoDispatchModel?) {
        viewModelScope.launch { _dispatchFlow.emit(value) }
    }

    fun clear() {
        viewModelScope.launch {
            _tripFlow.emit(null)
            _dispatchFlow.emit(null)
        }
    }
}
```

This mirrors the fragment layout and behavior, including lifecycle collection.

```kotlin
package ifac.td.taxi.ui.screen.fixedprice

import android.content.pm.ActivityInfo
import androidx.compose.foundation.background
import androidx.compose.foundation.layout.*
import androidx.compose.foundation.lazy.LazyColumn
import androidx.compose.foundation.lazy.items
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.platform.LocalContext
import androidx.compose.ui.unit.dp
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.lifecycle.compose.LifecycleStartEffect
import androidx.navigation.NavController
import ifac.td.taxi.R
import ifac.td.taxi.repository.connections.rest.bravoRest.SuggestModel
import ifac.td.taxi.ui.custom.button.CustomButton
import ifac.td.taxi.ui.custom.button.ButtonType
import ifac.td.taxi.viewmodel.fixedprice.FixedPriceMainActivityViewModel
import ifac.td.taxi.viewmodel.fixedprice.FixedPriceViewModel
import kotlinx.coroutines.flow.collectLatest

@Composable
fun FixedPriceMapScreen(
    navController: NavController,
    viewModel: FixedPriceViewModel,
    sharedViewModel: FixedPriceMainActivityViewModel,
    onShowHeader: (Boolean) -> Unit,
    onShowBottomBar: (Boolean) -> Unit,
    onRequestedOrientation: (Int) -> Unit,
    onToast: (Int) -> Unit,
    onCloseDialog: (String) -> Unit = {},
) {
    val state by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    var buttonsState by remember { mutableStateOf(FixedPriceButtonsState()) }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                is FixedPriceUiEffect.ShowToast -> onToast(effect.messageRes)
                FixedPriceUiEffect.HideKeyboard -> Unit
                FixedPriceUiEffect.HideBottomSheet -> Unit
                FixedPriceUiEffect.ExpandBottomSheet -> Unit
                is FixedPriceUiEffect.MoveMapToPickup -> Unit
                is FixedPriceUiEffect.MoveMapToDropOff -> Unit
                is FixedPriceUiEffect.PutPickupMarker -> Unit
                is FixedPriceUiEffect.PutDropOffMarker -> Unit
                is FixedPriceUiEffect.RemoveMarker -> Unit
                FixedPriceUiEffect.ClearSuggestions -> Unit
                FixedPriceUiEffect.ResetFixedPriceData -> Unit
                FixedPriceUiEffect.ResetErrorFlow -> viewModel.resetErrorFlow()
                FixedPriceUiEffect.RequestDispatchDestinationCoordinates -> Unit
            }
        }
    }

    LaunchedEffect(state.selectedDropOff) {
        buttonsState = FixedPriceButtonsReducer.reduce(
            buttonsState,
            if (state.selectedDropOff == null) {
                FixedPriceUiEvent.OnCloseClicked
            } else {
                FixedPriceUiEvent.OnSuggestionClicked(state.selectedDropOff)
            }
        )
    }

    LaunchedEffect(state.errorCall) {
        buttonsState = FixedPriceButtonsReducer.reduce(
            buttonsState,
            FixedPriceUiEvent.OnErrorChanged(state.errorCall)
        )
    }

    LaunchedEffect(state.fixedPriceData) {
        buttonsState = FixedPriceButtonsReducer.reduce(
            buttonsState,
            FixedPriceUiEvent.OnFixedPriceApiChanged(state.fixedPriceData)
        )
    }

    LaunchedEffect(state.fixedPriceDataFromDb) {
        buttonsState = FixedPriceButtonsReducer.reduce(
            buttonsState,
            FixedPriceUiEvent.OnFixedPriceDbChanged(state.fixedPriceDataFromDb)
        )
    }

    LaunchedEffect(Unit) {
        onShowHeader(true)
        onShowBottomBar(false)
        onRequestedOrientation(ActivityInfo.SCREEN_ORIENTATION_LANDSCAPE)

        val trip = sharedViewModel.tripFlow.value
        val dispatch = sharedViewModel.dispatchFlow.value
        viewModel.onEvent(FixedPriceUiEvent.OnTripLoaded(trip))
        viewModel.onEvent(FixedPriceUiEvent.OnDispatchLoaded(dispatch))

        if (trip?.fromDispatch == true && (dispatch?.dropOffCoordinates?.size ?: 0) > 0) {
            if (trip != null && dispatch != null) {
                viewModel.getFixedPriceFromDispatch(trip, dispatch, null)
            }
        }
    }

    Scaffold(
        containerColor = Color.Transparent
    ) { padding ->
        Box(
            modifier = Modifier
                .fillMaxSize()
                .padding(padding)
        ) {
            Column(modifier = Modifier.fillMaxSize()) {
                Row(
                    modifier = Modifier
                        .fillMaxWidth()
                        .background(Color.Black.copy(alpha = 0.2f))
                        .padding(horizontal = 12.dp, vertical = 8.dp)
                ) {
                    FilterChip(
                        selected = !state.isPoiSearch,
                        onClick = { viewModel.onEvent(FixedPriceUiEvent.OnPoiSearchChanged(false)) },
                        label = { Text("Address") }
                    )
                    Spacer(Modifier.width(12.dp))
                    FilterChip(
                        selected = state.isPoiSearch,
                        onClick = { viewModel.onEvent(FixedPriceUiEvent.OnPoiSearchChanged(true)) },
                        label = { Text("POI") }
                    )
                }

                OutlinedTextField(
                    value = state.manualText,
                    onValueChange = {
                        viewModel.onEvent(FixedPriceUiEvent.OnManualTextChanged(it))
                    },
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(horizontal = 6.dp),
                    placeholder = { Text("Drop off") }
                )

                if (state.isSuggestionListVisible) {
                    LazyColumn(
                        modifier = Modifier
                            .fillMaxWidth()
                            .weight(1f)
                            .padding(horizontal = 6.dp)
                    ) {
                        items(state.suggestions.sortedByDescending { it.score }) { item ->
                            Text(
                                text = item.getPrintableStreetTextShort(),
                                modifier = Modifier
                                    .fillMaxWidth()
                                    .padding(12.dp)
                            )
                        }
                    }
                } else {
                    Spacer(Modifier.weight(1f))
```

```kotlin
object FixedPriceButtonsReducer {
    fun reduce(current: FixedPriceButtonsState, event: FixedPriceUiEvent): FixedPriceButtonsState {
        return when (event) {
            is FixedPriceUiEvent.OnSuggestionClicked -> current.copy(
                rvVisible = false,
                bottomSheetVisible = true,
                closeVisible = true,
                suggestVisible = true,
                dropOffText = event.item.getPrintableStreetTextShort()
            )

            is FixedPriceUiEvent.OnCloseClicked -> current.copy(
                dropOffText = "",
                rvVisible = false,
                bottomSheetVisible = false,
                taxiPriceText = "",
                vanPriceText = "",
                businessPriceText = "",
                taxiPriceVisible = false,
                vanPriceVisible = false,
                businessPriceVisible = false,
                taxiProgressVisible = true,
                vanProgressVisible = true,
                businessProgressVisible = true,
                taxiErrorVisible = false,
                vanErrorVisible = false,
                businessErrorVisible = false
            )

            is FixedPriceUiEvent.OnErrorChanged -> {
                if (event.error == true) current.copy(
                    taxiErrorVisible = true,
                    vanErrorVisible = true,
                    businessErrorVisible = true,
                    taxiProgressVisible = false,
                    vanProgressVisible = false,
                    businessProgressVisible = false,
                    taxiPriceText = "",
                    vanPriceText = "",
                    businessPriceText = "",
                    taxiPriceVisible = false,
                    vanPriceVisible = false,
                    businessPriceVisible = false
                ) else current.copy(
                    taxiErrorVisible = false,
                    vanErrorVisible = false,
                    businessErrorVisible = false
                )
            }

            is FixedPriceUiEvent.OnFixedPriceApiChanged -> current.copy(
                taxiPriceText = event.data.taxiPrice.orEmpty(),
                vanPriceText = event.data.vanPrice.orEmpty(),
                businessPriceText = event.data.businessPrice.orEmpty(),
                taxiPriceVisible = event.data.taxiPrice != null,
                vanPriceVisible = event.data.vanPrice != null,
                businessPriceVisible = event.data.businessPrice != null,
                taxiProgressVisible = event.data.taxiPrice == null,
                vanProgressVisible = event.data.vanPrice == null,
                businessProgressVisible = event.data.businessPrice == null,
                taxiErrorVisible = event.data.taxiError,
                vanErrorVisible = event.data.vanError,
                businessErrorVisible = event.data.businessError
            )

            is FixedPriceUiEvent.OnFixedPriceDbChanged -> {
                val data = event.data ?: return current
                current.copy(
                    taxiPriceText = data.taxiPrice.orEmpty(),
                    vanPriceText = data.vanPrice.orEmpty(),
                    businessPriceText = data.businessPrice.orEmpty(),
                    taxiPriceVisible = data.taxiPrice != null,
                    vanPriceVisible = data.vanPrice != null,
                    businessPriceVisible = data.businessPrice != null,
                    taxiProgressVisible = data.taxiPrice == null,
                    vanProgressVisible = data.vanPrice == null,
                    businessProgressVisible = data.businessPrice == null,
                    taxiErrorVisible = data.taxiPrice == null,
                    vanErrorVisible = data.vanPrice == null,
                    businessErrorVisible = data.businessPrice == null
                )
            }

            else -> current
        }
    }

fun FixedPriceMapScreen(
    navController: NavController,
    viewModel: FixedPriceViewModel,
    sharedViewModel: FixedPriceMainActivityViewModel,
    onShowHeader: (Boolean) -> Unit,
    onShowBottomBar: (Boolean) -> Unit,
    onRequestedOrientation: (Int) -> Unit,
    onToast: (Int) -> Unit,
) {
    val state by viewModel.uiState.collectAsStateWithLifecycle()
    val lifecycleOwner = LocalLifecycleOwner.current
    val context = LocalContext.current

    var buttonsState by remember { mutableStateOf(FixedPriceButtonsState()) }
    var mapController by remember { mutableStateOf<MapController?>(null) }
    var mapView by remember { mutableStateOf<CercaliaMapView?>(null) }
    var mapReady by remember { mutableStateOf(false) }

    LaunchedEffect(Unit) {
        onShowHeader(true)
        onShowBottomBar(false)
        onRequestedOrientation(ActivityInfo.SCREEN_ORIENTATION_LANDSCAPE)
    }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collect { effect ->
            when (effect) {
                is FixedPriceUiEffect.ShowToast -> onToast(effect.messageRes)
                FixedPriceUiEffect.HideKeyboard -> hideKeyboard(context)
                FixedPriceUiEffect.HideBottomSheet -> Unit
                FixedPriceUiEffect.ExpandBottomSheet -> Unit
                is FixedPriceUiEffect.MoveMapToPickup -> {
                    if (mapReady) {
                        mapController?.moveCamera(
                            CameraUpdateFactory.newLatLngZoom(effect.pickup, if (effect.moveCamera) 17f else 17f)
                        )
                    }
                }
                is FixedPriceUiEffect.MoveMapToDropOff -> {
                    if (mapReady) {
                        val trip = effect.trip
                        val tripLat = trip?.latitudePickup ?: W2CLocation.getLastLatitude()
                        val tripLng = trip?.longitudePickup ?: W2CLocation.getLastLongitude()
                        val midPoint = calculateMidpoint(effect.dropOff.lat, effect.dropOff.lng, tripLat, tripLng)
                        mapController?.moveCamera(
                            CameraUpdateFactory.newLatLngZoom(
                                LatLng(midPoint.first, midPoint.second),
                                calculateZoomLevel(effect.dropOff.lat, effect.dropOff.lng, tripLat, tripLng)
                            )
                        )
                    }
                }
                is FixedPriceUiEffect.PutPickupMarker -> {
                    mapController?.putFeature(
                        Marker(context, "PickUp").setCoordinate(effect.pickup).apply {
                            setDrawable(ContextCompat.getDrawable(context, R.drawable.ic_notification_png))
                        }
                    )
                }
                is FixedPriceUiEffect.PutDropOffMarker -> {
                    mapController?.putFeature(
                        Marker(context, "DropOff").setCoordinate(effect.dropOff)
                    )
                }
                is FixedPriceUiEffect.RemoveMarker -> Unit
                FixedPriceUiEffect.ClearSuggestions -> Unit
                FixedPriceUiEffect.ResetFixedPriceData -> Unit
                FixedPriceUiEffect.ResetErrorFlow -> viewModel.resetErrorFlow()
                FixedPriceUiEffect.RequestDispatchDestinationCoordinates -> Unit
            }
        }
    }

    LaunchedEffect(state.selectedDropOff) {
        buttonsState = FixedPriceButtonsReducer.reduce(
            buttonsState,
            if (state.selectedDropOff == null) FixedPriceUiEvent.OnCloseClicked
            else FixedPriceUiEvent.OnSuggestionClicked(state.selectedDropOff)
        )
    }

    LaunchedEffect(state.errorCall) {
        buttonsState = FixedPriceButtonsReducer.reduce(buttonsState, FixedPriceUiEvent.OnErrorChanged(state.errorCall))
    }

    LaunchedEffect(state.fixedPriceData) {
        buttonsState = FixedPriceButtonsReducer.reduce(
            buttonsState,
            FixedPriceUiEvent.OnFixedPriceApiChanged(state.fixedPriceData)
        )
    }

    LaunchedEffect(state.fixedPriceDataFromDb) {
        buttonsState = FixedPriceButtonsReducer.reduce(
            buttonsState,

@Composable
private fun FixedPricePriceRow(
    title: String,
    price: String,
    priceVisible: Boolean,
    progressVisible: Boolean,
    errorVisible: Boolean
) {
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .padding(horizontal = 16.dp, vertical = 16.dp),
        horizontalArrangement = Arrangement.SpaceBetween,
        verticalAlignment = Alignment.CenterVertically
    ) {
        Text(text = title, color = Color.White)
        when {
            errorVisible -> Text(text = "!", color = Color.Red)
            progressVisible -> CircularProgressIndicator(modifier = Modifier.size(20.dp))
            priceVisible -> Text(text = price, color = Color.White)
        }
    }
}

private fun hideKeyboard(context: Context) {
    val imm = context.getSystemService(Context.INPUT_METHOD_SERVICE) as InputMethodManager
    val view = (context as? Activity)?.currentFocus ?: View(context)
    imm.hideSoftInputFromWindow(view.windowToken, 0)
}

private fun calculateMidpoint(
    lat1: Double,
    lon1: Double,
    lat2: Double,
    lon2: Double
): Pair<Double, Double> {
    return Pair((lat1 + lat2) / 2.0, (lon1 + lon2) / 2.0)
}

private fun calculateZoomLevel(
    lat1: Double,
    lon1: Double,
    lat2: Double,
    lon2: Double
): Float {
    val distance = kotlin.math.abs(lat1 - lat2) + kotlin.math.abs(lon1 - lon2)
    return when {
        distance < 0.01 -> 17f
        distance < 0.05 -> 15f
        distance < 0.1 -> 14f
        else -> 12f
    }
}
```

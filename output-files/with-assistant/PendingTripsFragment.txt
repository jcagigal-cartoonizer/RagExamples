## 3) Exact `PendingTripsButtonsState` reducer

This reducer mirrors the fragment’s `collect {}` branches one-by-one:

- if list is null/empty:
  - hide RecyclerView
  - show empty text
  - trigger `checkClosePendingIfEmpty()`
- else:
  - show RecyclerView
  - hide empty text
  - submit list to adapter equivalent

```kotlin
package ifac.td.taxi.ui.screen.pendingtrips

import com.interfacom.sdk.taximeter.bravocomm.rest.pending_trips.response.PendingTrip

object PendingTripsButtonsStateReducer {

    fun reduce(
        current: PendingTripsUiState,
        pendingTrips: ArrayList<PendingTrip>?
    ): PendingTripsUiState {
        return if (pendingTrips.isNullOrEmpty()) {
            current.copy(
                pendingTrips = emptyList(),
                showTripsList = false,
                showNoTrips = true
            )
        } else {
            current.copy(
                pendingTrips = pendingTrips,
                showTripsList = true,
                showNoTrips = false
            )
        }
    }
}
```

This replaces the fragment logic and preserves the original behavior.

```kotlin
package ifac.td.taxi.ui.screen.pendingtrips

import android.app.Application
import android.os.Handler
import android.os.Looper
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import com.interfacom.sdk.taximeter.bravocomm.BravoCentral
import com.interfacom.sdk.taximeter.bravocomm.rest.pending_trips.response.PendingTrip
import ifac.td.taxi.R
import ifac.td.taxi.domain.usecase.PendingTripsUseCase
import ifac.td.taxi.domain.usecase.UserPreferencesUseCase
import ifac.td.taxi.framework.sdk.bravocentral.usecase.BravoCentralUseCase
import ifac.td.taxi.ui.custom.button.ButtonType
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

class PendingTripsViewModel(
    private val pendingTripsUseCase: PendingTripsUseCase,
    private val bravoCentralUseCase: BravoCentralUseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    application: Application,
) : AndroidViewModel(application) {

    private val handler = Handler(Looper.getMainLooper())
    private var runnable: Runnable? = null

    private val _uiState = MutableStateFlow(PendingTripsUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<PendingTripsUiEffect>(extraBufferCapacity = 1)
    val uiEffect: SharedFlow<PendingTripsUiEffect> = _uiEffect.asSharedFlow()

    private val _showDialogOnAcceptFlow = MutableStateFlow<Boolean?>(true)
    val showDialogOnAcceptFlow = _showDialogOnAcceptFlow.asStateFlow()

    fun onEvent(event: PendingTripsUiEvent) {
        when (event) {
            PendingTripsUiEvent.OnResume -> {
                checkAcceptDialogPermission()
                loadTrips(forced = true)
            }

            PendingTripsUiEvent.OnPause -> {
                removeHandlerCallback()
                viewModelScope.launch {
                    _uiEffect.emit(PendingTripsUiEffect.FinishPendingTripsTimer)
                }
            }

            PendingTripsUiEvent.OnBackPressed -> {
                viewModelScope.launch {
                    _uiEffect.emit(PendingTripsUiEffect.NavigateBack)
                }
            }

            PendingTripsUiEvent.OnRetryLoadTrips -> {
                loadTrips(forced = true)
            }

            is PendingTripsUiEvent.OnTripClicked -> {
                val trip = _uiState.value.pendingTrips.find { it.tripID == event.tripId } ?: return
                if (_showDialogOnAcceptFlow.value == true) {
                    requestTripWithDialog(trip)
                } else {
                    requestTripDirect(trip)
                }
            }

            is PendingTripsUiEvent.OnDialogResult -> {
                if (event.accepted) {
                    requestTripDirect(event.trip)
                }
            }
        }
    }

    fun loadTrips(forced: Boolean = false) {
        if (BravoCentral.isPendingTripsAllowed(false) || forced) {
            refreshPendingTripsManual(forced)
        }
        startChronometer()
    }

    private fun refreshPendingTripsManual(forced: Boolean = false) {
        viewModelScope.launch {
            pendingTripsUseCase.getPendingTripsZone(
                flow = MutableStateFlow(_uiState.value.pendingTrips.toCollection(arrayListOf())),
                forced = forced
            )
        }
    }

    private fun startChronometer() {
        if (runnable != null) return

        viewModelScope.launch(Dispatchers.IO) {
            val refreshRate = 2000L
            val initRefreshRate = 500L

            runnable = object : Runnable {
                override fun run() {
                    viewModelScope.launch {
                        if (BravoCentral.isPendingTripsAllowed(true)) {
                            pendingTripsUseCase.getPendingTripsZone(
                                flow = MutableStateFlow(_uiState.value.pendingTrips.toCollection(arrayListOf()))
                            )
                        }
                    }
                    handler.postDelayed(this, refreshRate)
                }
            }
            handler.postDelayed(runnable!!, initRefreshRate)
        }
    }

    fun removeHandlerCallback() {
        runnable?.let { handler.removeCallbacks(it) }
        runnable = null
    }

    fun checkAcceptDialogPermission() {
        viewModelScope.launch {
            val preferences = userPreferencesUseCase.getUserPreferences()
            preferences?.showConfAcceptDispatch?.let { _showDialogOnAcceptFlow.emit(it) }
        }
    }

    fun checkClosePendingIfEmpty() {
        viewModelScope.launch {
            userPreferencesUseCase.getUserPreferences()?.let {
                if (it.closePendingTripsIfEmpty) {
                    _uiEffect.emit(PendingTripsUiEffect.NavigateBack)
                }
            }
        }
    }

    fun requestTrip(trip: PendingTrip) {
        bravoCentralUseCase.requestPendingTrip(trip)
    }

    private fun requestTripDirect(trip: PendingTrip) {
        requestTrip(trip)
        viewModelScope.launch {
            _uiEffect.emit(PendingTripsUiEffect.ShowToastDataSent)
            _uiEffect.emit(PendingTripsUiEffect.NavigateBack)
        }
    }

    private fun requestTripWithDialog(trip: PendingTrip) {
        val dialog = PendingTripsDialogState(
            title = getApplication<Application>().getString(R.string.auction_confirm),
            description = if (trip.pickupAddress.isNullOrBlank()) {
                getApplication<Application>().getString(R.string.confirm_auction_trip_no_address)
            } else {
                getApplication<Application>().getString(
                    R.string.confirm_auction_trip,
                    trip.pickupAddress
                )
            },
            buttons = listOf(ButtonType.CANCEL, ButtonType.ACCEPT),
            pendingTrip = trip
        )

        viewModelScope.launch {
            _uiEffect.emit(PendingTripsUiEffect.OpenDialog(dialog))
        }
    }

    fun onPendingTripsUpdated(arePending: ArrayList<PendingTrip>?) {
        _uiState.update { current ->
            PendingTripsButtonsStateReducer.reduce(current, arePending).copy(
                pendingTrips = arePending ?: emptyList()
            )
        }

        if (arePending.isNullOrEmpty()) {
            checkClosePendingIfEmpty()
        }
    }
}
```

```kotlin
@Composable
fun PendingTripsScreen(
    viewModel: PendingTripsViewModel,
    sharedViewModel: PendingTripsMainActivityViewModel,
    onNavigateBack: () -> Unit,
    onShowToastDataSent: () -> Unit,
    onOpenDialog: (PendingTripsDialogState) -> Unit,
) {
    val uiState by viewModel.uiState.collectAsState()

    LaunchedEffect(Unit) {
        viewModel.onEvent(PendingTripsUiEvent.OnResume)
    }

    LaunchedEffect(Unit) {
        sharedViewModel.pendingTripsListFlow.collectLatest { arePending ->
            viewModel.onPendingTripsUpdated(arePending)
        }
    }

    LaunchedEffect(Unit) {
        sharedViewModel.eventPendingTrips.collectLatest {
            viewModel.onEvent(PendingTripsUiEvent.OnRetryLoadTrips)
        }
    }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                PendingTripsUiEffect.NavigateBack -> onNavigateBack()
                PendingTripsUiEffect.ShowToastDataSent -> onShowToastDataSent()
                PendingTripsUiEffect.LoadTrips -> viewModel.onEvent(PendingTripsUiEvent.OnRetryLoadTrips)
                PendingTripsUiEffect.FinishPendingTripsTimer -> sharedViewModel.finishPendingTripsTimer()
                is PendingTripsUiEffect.OpenDialog -> onOpenDialog(effect.dialog)
            }
        }
    }

    Column(modifier = Modifier.fillMaxSize()) {
        if (uiState.showNoTrips) {
            Text(
                text = stringResource(R.string.auction_no_trips_waiting),
                modifier = Modifier.padding(16.dp)
            )
        }

        if (uiState.showTripsList) {
            LazyColumn(modifier = Modifier.fillMaxSize()) {
                items(uiState.pendingTrips, key = { it.tripID }) { trip ->
                    PendingTripRow(
                        trip = trip,
                        onClick = { viewModel.onEvent(PendingTripsUiEvent.OnTripClicked(trip.tripID)) }
                    )
                }
            }
        }
    }
}
```

```kotlin
        viewModelScope.launch {
            pendingTripsUseCase.getPendingTripsZone(
                flow = MutableStateFlow(_uiState.value.pendingTrips.toCollection(arrayListOf())),
                forced = forced
            )
        }
    }

    private fun startChronometer() {
        if (runnable != null) return

        viewModelScope.launch(Dispatchers.IO) {
            val refreshRate = 2000L
            val initRefreshRate = 500L

            runnable = object : Runnable {
                override fun run() {
                    viewModelScope.launch {
                        if (BravoCentral.isPendingTripsAllowed(true)) {
                            pendingTripsUseCase.getPendingTripsZone(
                                flow = MutableStateFlow(_uiState.value.pendingTrips.toCollection(arrayListOf()))
                            )
                        }
                    }
                    handler.postDelayed(this, refreshRate)
                }
            }
            handler.postDelayed(runnable!!, initRefreshRate)
        }
    }

    fun removeHandlerCallback() {
        runnable?.let { handler.removeCallbacks(it) }
        runnable = null
    }

    fun requestTrip(trip: PendingTrip) {
        bravoCentralUseCase.requestPendingTrip(trip)
    }

    fun checkAcceptDialogPermission() {
        viewModelScope.launch {
            val preferences = userPreferencesUseCase.getUserPreferences()
            preferences?.showConfAcceptDispatch?.let { _showDialogOnAcceptFlow.emit(it) }
        }
    }

    fun checkClosePendingIfEmpty() {
        viewModelScope.launch {
            userPreferencesUseCase.getUserPreferences()?.let {
                if (it.closePendingTripsIfEmpty) {
                    _uiEffect.emit(PendingTripsUiEffect.NavigateBack)
                }
            }
        }
    }

    fun onPendingTripsUpdated(arePending: ArrayList<PendingTrip>?) {
        _uiState.value = PendingTripsButtonsStateReducer.reduce(_uiState.value, arePending)
        if (arePending.isNullOrEmpty()) checkClosePendingIfEmpty()
    }

    private fun requestTripDirect(trip: PendingTrip) {
        requestTrip(trip)
        viewModelScope.launch {
            _uiEffect.emit(PendingTripsUiEffect.ShowToastDataSent)
            _uiEffect.emit(PendingTripsUiEffect.NavigateBack)
        }
    }

    private fun requestTripWithDialog(trip: PendingTrip) {
        val dialog = PendingTripsDialogState(
            title = getApplication<Application>().getString(R.string.auction_confirm),
            description = if (trip.pickupAddress.isNullOrBlank()) {
                getApplication<Application>().getString(R.string.confirm_auction_trip_no_address)
            } else {
                getApplication<Application>().getString(
                    R.string.confirm_auction_trip,
                    trip.pickupAddress
                )
            },
            buttons = listOf(ButtonType.CANCEL, ButtonType.ACCEPT),
            pendingTrip = trip
        )

        viewModelScope.launch { _uiEffect.emit(PendingTripsUiEffect.OpenDialog(dialog)) }
    }

@Composable
fun PendingTripsScreen(
    viewModel: PendingTripsViewModel,
    sharedViewModel: PendingTripsMainActivityViewModel,
    onNavigateBack: () -> Unit,
    onShowToastDataSent: () -> Unit,
) {
    val uiState by viewModel.uiState.collectAsState()
    val context = LocalContext.current
    var dialogState by remember { mutableStateOf<PendingTripsDialogState?>(null) }

    LaunchedEffect(Unit) {
        viewModel.onEvent(PendingTripsUiEvent.OnResume)
    }

    LaunchedEffect(Unit) {
        sharedViewModel.pendingTripsListFlow.collectLatest { arePending ->
            viewModel.onPendingTripsUpdated(arePending)
        }
    }

    LaunchedEffect(Unit) {
        sharedViewModel.eventPendingTrips.collectLatest {
            viewModel.onEvent(PendingTripsUiEvent.OnRetryLoadTrips)
        }
    }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                PendingTripsUiEffect.NavigateBack -> onNavigateBack()
                PendingTripsUiEffect.ShowToastDataSent -> onShowToastDataSent()
                PendingTripsUiEffect.LoadTrips -> viewModel.onEvent(PendingTripsUiEvent.OnRetryLoadTrips)
                PendingTripsUiEffect.FinishPendingTripsTimer -> sharedViewModel.finishPendingTripsTimer()
                is PendingTripsUiEffect.OpenDialog -> dialogState = effect.dialog
            }
        }
    }

    Column(modifier = Modifier.fillMaxSize()) {
        if (uiState.showNoTrips) {
            Text(
                text = stringResource(R.string.auction_no_trips_waiting),
                modifier = Modifier.padding(16.dp)
            )
        }

        if (uiState.showTripsList) {
            LazyColumn(modifier = Modifier.fillMaxSize()) {
                items(uiState.pendingTrips, key = { it.tripID }) { trip ->
                    PendingTripRow(
                        trip = trip,
                        onClick = { viewModel.onEvent(PendingTripsUiEvent.OnTripClicked(trip.tripID)) }
                    )
                }
            }
        }
    }

    PendingTripsDialogHost(
        dialogState = dialogState,
        onDismiss = { dialogState = null },
        onResult = { accepted ->
            val trip = dialogState?.pendingTrip ?: return@PendingTripsDialogHost
            dialogState = null
            viewModel.onEvent(PendingTripsUiEvent.OnDialogResult(accepted, trip))
        }
    )
}

@Composable
private fun PendingTripRow(
    trip: PendingTrip,
    onClick: () -> Unit
) {
    Button(
        onClick = onClick,
        modifier = Modifier.padding(16.dp)
    ) {
        Text(text = trip.tripID ?: "")
    }
}

@Composable
fun PendingTripsDialogHost(
    dialogState: PendingTripsDialogState?,
    onDismiss: () -> Unit,
    onResult: (accepted: Boolean) -> Unit
) {
    if (dialogState == null) return

    AlertDialog(
        onDismissRequest = onDismiss,
        title = { Text(dialogState.title) },
        text = { Text(dialogState.description) },
        confirmButton = {
            Button(onClick = { onResult(true) }) {
                Text(text = stringResource(R.string.btn_accept))
            }
        },
        dismissButton = {
            Button(onClick = { onResult(false) }) {
                Text(text = stringResource(R.string.btn_cancel))
            }
        }
    )
}
```

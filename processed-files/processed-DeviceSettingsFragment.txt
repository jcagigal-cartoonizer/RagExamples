# 1) Compose ViewModel with `UiState + UiEffect`
```kotlin
package ifac.td.taxi.viewmodel
// # Block 11-1: import android.app.Application
import android.app.Application
import android.app.NotificationManager
import android.content.Context
import android.media.AudioManager
import android.provider.Settings
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.framework.sdk.usecase.SoundManagerUseCase
import ifac.td.taxi.framework.util.Logs
import kotlinx.coroutines.channels.BufferOverflow
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch
data class DeviceSettingsUiState(
    val canWriteSettings: Boolean = false,
    val isDndEnabled: Boolean = false,
    val isSilentModeEnabled: Boolean = false,
    val areVolumesLinked: Boolean? = null,
    val showPermissionDialog: Boolean = false,
    val showLinkedVolumesDialog: Boolean = false,
    val showRingtoneTestDialog: Boolean = false,
    val phoneCallVolume: Int = 0,
    val systemVolume: Int = 0,
    val notificationVolume: Int = 0,
    val ringtoneVolume: Int = 0,
    val brightness: Int = 0,
    val minPhoneCallVolume: Int = 20
)
sealed interface DeviceSettingsUiEffect {
    data class OpenSystemWriteSettings(val packageName: String) : DeviceSettingsUiEffect
    data class ShowToast(val messageRes: Int) : DeviceSettingsUiEffect
    data object RefreshSliders : DeviceSettingsUiEffect
    data object OpenPermissionSettings : DeviceSettingsUiEffect
    data object NavigateBack : DeviceSettingsUiEffect
}
class DeviceSettingsComposeViewModel(
    private val soundManagerUseCase: SoundManagerUseCase,
    application: Application,
) : AndroidViewModel(application) {
    private val appContext: Context get() = getApplication<Application>()
    private val _uiState = MutableStateFlow(DeviceSettingsUiState())
    val uiState = _uiState.asStateFlow()
    private val _uiEffect = MutableSharedFlow<DeviceSettingsUiEffect>(
        extraBufferCapacity = 1,
        onBufferOverflow = BufferOverflow.DROP_OLDEST
    )
    val uiEffect: SharedFlow<DeviceSettingsUiEffect> = _uiEffect
    fun onScreenOpened() {
        refreshSnapshot()
        checkLinkedVolumes()
    }
    fun refreshSnapshot() {
        _uiState.value = _uiState.value.copy(
            canWriteSettings = Settings.System.canWrite(appContext),
            isDndEnabled = isDoNotDisturbModeEnabled(),
            isSilentModeEnabled = isSilentModeEnabled(),
            phoneCallVolume = getPhoneCallVolume(),
            systemVolume = getSystemVolume(),
            notificationVolume = getNotificationVolume(),
            ringtoneVolume = getRingtoneVolume(),
            brightness = getBrightness(),
            minPhoneCallVolume = getMinPhoneCallVolume()
        )
    }
    fun onRequestWriteSettings() {
        viewModelScope.launch {
            _uiEffect.emit(
                DeviceSettingsUiEffect.OpenSystemWriteSettings(
                    packageName = appContext.packageName
                )
            )
        }
    }
    fun checkLinkedVolumes() {
        viewModelScope.launch {
            val linked = areNotificationAndRingVolumesLinked()
            _uiState.value = _uiState.value.copy(areVolumesLinked = linked)
        }
    }
    fun testRingtone() {
        viewModelScope.launch {
            try {
                soundManagerUseCase.playTestSound()
            } catch (e: Exception) {
                Logs.e("DeviceSettingsComposeViewModel", "Error playing ringtone: ${e.message}")
                _uiEffect.emit(DeviceSettingsUiEffect.ShowToast(android.R.string.untitled))
            }
        }
    }
    fun setPhoneCallVolume(newVolume: Int) {
        if (isDoNotDisturbModeEnabled() || isSilentModeEnabled()) return
        try {
            val audioManager = appContext.getSystemService(Context.AUDIO_SERVICE) as AudioManager
            val maxVolume = audioManager.getStreamMaxVolume(AudioManager.STREAM_VOICE_CALL)
            val minVolume = getMinPhoneCallVolume()
            val adjusted = if (newVolume in 0 until minVolume) minVolume else newVolume
            audioManager.setStreamVolume(
                AudioManager.STREAM_VOICE_CALL,
                denormalizeVolume(adjusted, maxVolume),
                AudioManager.FLAG_SHOW_UI
            )
            refreshSnapshot()
        } catch (_: Exception) {}
    }
    fun setSystemVolume(newVolume: Int) {
        if (isDoNotDisturbModeEnabled() || isSilentModeEnabled()) return
        try {
            val audioManager = appContext.getSystemService(Context.AUDIO_SERVICE) as AudioManager
            val maxVolume = audioManager.getStreamMaxVolume(AudioManager.STREAM_MUSIC)
            audioManager.setStreamVolume(
                AudioManager.STREAM_MUSIC,
                denormalizeVolume(newVolume, maxVolume),
                AudioManager.FLAG_SHOW_UI
            )
            refreshSnapshot()
        } catch (_: Exception) {}
    }
    fun setNotificationVolume(newVolume: Int) {
        if (isDoNotDisturbModeEnabled() || isSilentModeEnabled()) return
        try {
            val audioManager = appContext.getSystemService(Context.AUDIO_SERVICE) as AudioManager
            val maxVolume = audioManager.getStreamMaxVolume(AudioManager.STREAM_NOTIFICATION)
            audioManager.setStreamVolume(
                AudioManager.STREAM_NOTIFICATION,
                denormalizeVolume(newVolume, maxVolume),
                AudioManager.FLAG_SHOW_UI
            )
            refreshSnapshot()
        } catch (_: Exception) {}
    }
    fun setRingtoneVolume(newVolume: Int) {
        if (isDoNotDisturbModeEnabled() || isSilentModeEnabled()) return
        try {
            val audioManager = appContext.getSystemService(Context.AUDIO_SERVICE) as AudioManager
            val maxVolume = audioManager.getStreamMaxVolume(AudioManager.STREAM_RING)
            audioManager.setStreamVolume(
                AudioManager.STREAM_RING,
                denormalizeVolume(newVolume, maxVolume),
                AudioManager.FLAG_SHOW_UI
            )
            refreshSnapshot()
        } catch (_: Exception) {}
    }
    fun setBrightness(newBrightness: Int) {
        try {
            if (isAutoBrightnessEnabled()) {
                Settings.System.putInt(
                    appContext.contentResolver,
                    Settings.System.SCREEN_BRIGHTNESS_MODE,
                    Settings.System.SCREEN_BRIGHTNESS_MODE_MANUAL
                )
            }
            if (Settings.System.canWrite(appContext)) {
                Settings.System.putInt(
                    appContext.contentResolver,
                    Settings.System.SCREEN_BRIGHTNESS,
                    newBrightness
                )
            }
            refreshSnapshot()
        } catch (_: Exception) {}
    }
    fun testPermissionOrOpenSettings() {
        if (!Settings.System.canWrite(appContext)) {
            onRequestWriteSettings()
        }
    }
    fun hasAudioPermissions(): Boolean = Settings.System.canWrite(appContext)
    fun isDoNotDisturbModeEnabled(): Boolean {
        val nm = appContext.getSystemService(Context.NOTIFICATION_SERVICE) as? NotificationManager
        return nm?.currentInterruptionFilter != NotificationManager.INTERRUPTION_FILTER_ALL
    }
    fun isSilentModeEnabled(): Boolean {
        val am = appContext.getSystemService(Context.AUDIO_SERVICE) as? AudioManager
        return am?.ringerMode == AudioManager.RINGER_MODE_SILENT
    }
    fun isAutoBrightnessEnabled(): Boolean {
        return Settings.System.getInt(
            appContext.contentResolver,
            Settings.System.SCREEN_BRIGHTNESS_MODE,
            Settings.System.SCREEN_BRIGHTNESS_MODE_MANUAL
        ) == Settings.System.SCREEN_BRIGHTNESS_MODE_AUTOMATIC
    }
    fun getBrightness(): Int = try {
        Settings.System.getInt(appContext.contentResolver, Settings.System.SCREEN_BRIGHTNESS, 0)
            .coerceIn(0, 255)
    } catch (_: Exception) { 0 }
    fun normalizeVolume(volume: Int, maxVolume: Int): Int {
        if (maxVolume <= 0) return 0
        return ((volume.toFloat() * 100f) / maxVolume).toInt().coerceIn(0, 100)
    }
    fun denormalizeVolume(normalizedVolume: Int, maxVolume: Int): Int {
        if (maxVolume <= 0) return 0
        return ((normalizedVolume.toFloat() * maxVolume) / 100f).toInt().coerceIn(0, maxVolume)
    }
    fun getPhoneCallVolume(): Int {
        val am = appContext.getSystemService(Context.AUDIO_SERVICE) as AudioManager
        return normalizeVolume(
            am.getStreamVolume(AudioManager.STREAM_VOICE_CALL),
            am.getStreamMaxVolume(AudioManager.STREAM_VOICE_CALL)
        ).coerceAtLeast(getMinPhoneCallVolume())
    }
    fun getNotificationVolume(): Int {
        val am = appContext.getSystemService(Context.AUDIO_SERVICE) as AudioManager
        return normalizeVolume(
            am.getStreamVolume(AudioManager.STREAM_NOTIFICATION),
            am.getStreamMaxVolume(AudioManager.STREAM_NOTIFICATION)
        )
    }
    fun getSystemVolume(): Int {
        val am = appContext.getSystemService(Context.AUDIO_SERVICE) as AudioManager
        return normalizeVolume(
            am.getStreamVolume(AudioManager.STREAM_MUSIC),
            am.getStreamMaxVolume(AudioManager.STREAM_MUSIC)
        )
    }
    fun getRingtoneVolume(): Int {
        val am = appContext.getSystemService(Context.AUDIO_SERVICE) as AudioManager
        return normalizeVolume(
            am.getStreamVolume(AudioManager.STREAM_RING),
            am.getStreamMaxVolume(AudioManager.STREAM_RING)
        ).coerceAtLeast(20)
    }
    fun getMinPhoneCallVolume(): Int = 20
    fun areNotificationAndRingVolumesLinked(): Boolean {
        try {
            val audioManager = appContext.getSystemService(Context.AUDIO_SERVICE) as AudioManager
            val notificationVolume = audioManager.getStreamVolume(AudioManager.STREAM_NOTIFICATION)
            val ringVolume = audioManager.getStreamVolume(AudioManager.STREAM_RING)
            val testVolume = if (notificationVolume > 0) notificationVolume - 1 else notificationVolume + 1
            audioManager.setStreamVolume(
                AudioManager.STREAM_NOTIFICATION,
                testVolume.coerceIn(0, audioManager.getStreamMaxVolume(AudioManager.STREAM_NOTIFICATION)),
                0
            )
            val newRingVolume = audioManager.getStreamVolume(AudioManager.STREAM_RING)
            audioManager.setStreamVolume(AudioManager.STREAM_NOTIFICATION, notificationVolume, 0)
            audioManager.setStreamVolume(AudioManager.STREAM_RING, ringVolume, 0)
            return newRingVolume != ringVolume
        } catch (_: Exception) {
            return false
        }
    }
}
```

```kotlin
    message: String,
    onAccept: (() -> Unit)? = null,
    onCancel: (() -> Unit)? = null,
    acceptText: String = "OK",
    cancelText: String = "Cancel",
    dismissOnBackPress: Boolean = true,
    dismissOnClickOutside: Boolean = true,
    content: (@Composable ColumnScope.() -> Unit)? = null

    AlertDialog(
        onDismissRequest = onCancel ?: {},
        properties = DialogProperties(
            dismissOnBackPress = dismissOnBackPress,
            dismissOnClickOutside = dismissOnClickOutside
        ),
        containerColor = Color.Transparent,
        title = null,
        text = null,
        confirmButton = {},
        dismissButton = {},
    ) {
        Box(
            modifier = Modifier
                .fillMaxWidth()
                .padding(16.dp)
                .clip(RoundedCornerShape(16.dp))
                .background(MaterialTheme.colorScheme.surface)
                .padding(20.dp)
        ) {
            Column(modifier = Modifier.fillMaxWidth()) {
                Text(text = title, style = MaterialTheme.typography.titleLarge)
                Spacer(modifier = Modifier.height(12.dp))
                Text(text = message, style = MaterialTheme.typography.bodyMedium)
                content?.let {
                    Spacer(modifier = Modifier.height(16.dp))
                    it()
                }
                Spacer(modifier = Modifier.height(24.dp))
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.End,
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    if (onCancel != null) {
                        TextButton(onClick = onCancel) {
                            Text(cancelText)
                        }
                    }
                    if (onAccept != null) {
                        Spacer(modifier = Modifier.width(8.dp))
                        Button(onClick = onAccept) {
                            Text(acceptText)
                        }
                    }
                }
            }
        }
    }
}
```

# 4) Compose `DeviceSettingsScreen`
```kotlin
package ifac.td.taxi.ui.screen
// # Block 385-4: import android.content.Intent
import android.content.Intent
import android.net.Uri
import android.provider.Settings
import androidx.compose.foundation.layout.*
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Modifier
import androidx.compose.ui.platform.LocalContext
import androidx.compose.ui.unit.dp
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.navigation.NavController
import ifac.td.taxi.ui.screen.components.DeviceSettingsCustomButtonCustomDialog
import ifac.td.taxi.ui.screen.components.DeviceSettingsButtonsState
import ifac.td.taxi.viewmodel.DeviceSettingsComposeViewModel
import ifac.td.taxi.viewmodel.DeviceSettingsUiEffect
import kotlinx.coroutines.flow.collectLatest
@Composable
fun DeviceSettingsScreen(
    navController: NavController,
    viewModel: DeviceSettingsComposeViewModel,
) {
    val context = LocalContext.current
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    var dialogState by remember { mutableStateOf<DeviceSettingsDialogState?>(null) }
    LaunchedEffect(Unit) {
        viewModel.onScreenOpened()
    }
    LaunchedEffect(Unit) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                is DeviceSettingsUiEffect.OpenSystemWriteSettings -> {
                    val intent = Intent(Settings.ACTION_MANAGE_WRITE_SETTINGS).apply {
                        data = Uri.parse("package:${effect.packageName}")
                        addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
                    }
                    context.startActivity(intent)
                }
                DeviceSettingsUiEffect.RefreshSliders -> {
                    viewModel.refreshSnapshot()
                }
                DeviceSettingsUiEffect.OpenPermissionSettings -> {
                    dialogState = DeviceSettingsDialogState.PermissionRequired
                }
                DeviceSettingsUiEffect.NavigateBack -> navController.popBackStack()
                is DeviceSettingsUiEffect.ShowToast -> {
                    // host-side toast/snackbar if needed
                }
            }
        }
    }
    val buttonState = remember(uiState) {
        DeviceSettingsButtonsState(
            showLinkedVolumesButton = true,
            showPermissionButton = !uiState.canWriteSettings,
            showResetButton = true,
            showBackButton = true,
            showThemeButton = false,
            linkedVolumesEnabled = true,
            permissionEnabled = !uiState.canWriteSettings,
            resetEnabled = true,
            backEnabled = true
        )
    }
    DeviceSettingsContent(
        uiState = uiState,
        buttonsState = buttonState,
        onPhoneCallVolumeChanged = viewModel::setPhoneCallVolume,
        onSystemVolumeChanged = viewModel::setSystemVolume,
        onNotificationVolumeChanged = viewModel::setNotificationVolume,
        onRingtoneVolumeChanged = viewModel::setRingtoneVolume,
        onBrightnessChanged = viewModel::setBrightness,
        onRequestPermissions = { viewModel.onRequestWriteSettings() },
        onCheckLinkedVolumes = { viewModel.checkLinkedVolumes() },
        onTestRingtone = { viewModel.testRingtone() },
        onBack = { navController.popBackStack() },
    )
    dialogState?.let { dialog ->
        when (dialog) {
            DeviceSettingsDialogState.PermissionRequired -> {
                DeviceSettingsCustomButtonCustomDialog(
                    visible = true,
                    title = "Permission required",
                    message = "Please allow write system settings to adjust volumes and brightness.",
                    onAccept = {
                        dialogState = null
                        viewModel.onRequestWriteSettings()
                    },
                    onCancel = { dialogState = null },
                    acceptText = "Open settings",
                    cancelText = "Cancel"
                )
            }
            DeviceSettingsDialogState.ConfirmLinkedVolumes -> {
                DeviceSettingsCustomButtonCustomDialog(
                    visible = true,
                    title = "Linked volumes",
                    message = "Notification and ringtone volumes are linked on this device.",
                    onAccept = { dialogState = null },
                    onCancel = { dialogState = null }
                )
            }
            DeviceSettingsDialogState.TestRingtone -> {
                DeviceSettingsCustomButtonCustomDialog(
                    visible = true,
                    title = "Test ringtone",
                    message = "Play a test ringtone now?",
                    onAccept = {
                        dialogState = null
                        viewModel.testRingtone()
                    },
                    onCancel = { dialogState = null },
                    acceptText = "Play",
                    cancelText = "Cancel"
                )
            }
        }
    }
}
private sealed interface DeviceSettingsDialogState {
    data object PermissionRequired : DeviceSettingsDialogState
    data object ConfirmLinkedVolumes : DeviceSettingsDialogState
    data object TestRingtone : DeviceSettingsDialogState
}
```

```kotlin
package ifac.td.taxi.ui.screen

fun DeviceSettingsContent(

    onSystemVolumeChanged: (Int) -> Unit,
    onNotificationVolumeChanged: (Int) -> Unit,
    onRingtoneVolumeChanged: (Int) -> Unit,

            title = "System volume",
            value = uiState.systemVolume,
            onValueChange = onSystemVolumeChanged

            title = "Notification volume",
            value = uiState.notificationVolume,

            title = "Ringtone volume",
            value = uiState.ringtoneVolume,
            onValueChange = onRingtoneVolumeChanged

            value = uiState.brightness,
            onValueChange = onBrightnessChanged
        )
        Spacer(modifier = Modifier.height(24.dp))
        if (buttonsState.showPermissionButton) {
            Button(onClick = onRequestPermissions) { Text("Grant permission") }

        if (buttonsState.showLinkedVolumesButton) {
            Spacer(modifier = Modifier.height(8.dp))
            Button(onClick = onCheckLinkedVolumes) { Text("Check linked volumes") }
        }
        if (buttonsState.showResetButton) {
            Spacer(modifier = Modifier.height(8.dp))
            Button(onClick = onTestRingtone) { Text("Test ringtone") }

        if (buttonsState.showBackButton) {
            Spacer(modifier = Modifier.height(8.dp))
            OutlinedButton(onClick = onBack) { Text("Back") }
        }
    }
}
@Composable

            valueRange = 0f..100f

) {

        Text("Brightness")

            value = value.toFloat(),
            onValueChange = { onValueChange(it.toInt()) },
            valueRange = 0f..255f
        )
    }
}
```

# 8) If you want the “closest possible” custom button helper
Here is a more detailed button helper that behaves like a custom button component.
```kotlin
package ifac.td.taxi.ui.screen.components
// # Block 630-6: import androidx.compose.foundation.background

import androidx.compose.foundation.clickable

    contentColor: Color,

                interactionSource = interaction,

                onClick = onClick

            .padding(horizontal = 16.dp),

        Text(
            text = text,
            color = contentColor,
            style = MaterialTheme.typography.titleMedium,
            modifier = Modifier
                .fillMaxHeight()
                .wrapContentHeight()
        )
    }
}
```

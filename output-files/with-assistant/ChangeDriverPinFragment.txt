## 2) Exact reducer mirroring the fragment button logic

This reducer mirrors the fragment’s `setButtons()` behavior and the validation branches in `checkParameters()` one-by-one.

```kotlin
package ifac.td.taxi.ui.screen.change_driver_pin

import ifac.td.taxi.R
import ifac.td.taxi.ui.custom.button.ButtonType

object ChangeDriverPinButtonsReducer {

    fun reduce(state: ChangeDriverPinUiState): ChangeDriverPinButtonsState {
        val hasAllInputs =
            state.driverNumber.isNotBlank() &&
            state.currentPin.isNotBlank() &&
            state.newPin.isNotBlank() &&
            state.repeatPin.isNotBlank()

        val pinsMatch = state.newPin == state.repeatPin
        val oldAndNewDifferent = state.currentPin != state.newPin

        val acceptEnabled = hasAllInputs && pinsMatch && oldAndNewDifferent
        val cancelEnabled = true

        return ChangeDriverPinButtonsState(
            acceptVisible = true,
            cancelVisible = true,
            acceptEnabled = acceptEnabled,
            cancelEnabled = cancelEnabled,
            acceptButtonType = ButtonType.ACCEPT,
            cancelButtonType = ButtonType.CANCEL
        )
    }

    fun validationDialog(state: ChangeDriverPinUiState): ChangeDriverPinDialogModel? {
        val currentDriverNumber = state.driverNumber
        val currentDriverPwd = state.currentPin
        val newDriverPwd = state.newPin
        val repeatedDriverPwd = state.repeatPin

        return when {
            currentDriverNumber.isEmpty() ||
                currentDriverPwd.isEmpty() ||
                newDriverPwd.isEmpty() ||
                repeatedDriverPwd.isEmpty() -> {
                ChangeDriverPinDialogModel(R.string.dialog_fill_inputs)
            }

            newDriverPwd != repeatedDriverPwd -> {
                ChangeDriverPinDialogModel(R.string.dialog_no_match_pin)
            }

            currentDriverPwd == newDriverPwd -> {
                ChangeDriverPinDialogModel(R.string.dialog_equal_pin)
            }

            else -> null
        }
    }
}
```

```kotlin
import com.interfacom.sdk.taximeter.licensing.rest.user.ChangePasswordPinView
import com.interfacom.sdk.taximeter.licensing.rest.user.UserModule
import com.interfacom.sdk.taximeter.licensing.rest.user.UserPresenter
import ifac.td.taxi.R
import ifac.td.taxi.ui.screen.change_driver_pin.ChangeDriverPinButtonsReducer

import ifac.td.taxi.ui.screen.change_driver_pin.ChangeDriverPinUiEvent
import ifac.td.taxi.ui.screen.change_driver_pin.ChangeDriverPinUiState
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

class ChangeDriverPinViewModel(context: Application) : AndroidViewModel(context) {

    private val appContext = context

    private val _uiState = MutableStateFlow(ChangeDriverPinUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<ChangeDriverPinUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()

    fun onEvent(event: ChangeDriverPinUiEvent) {
        when (event) {
            is ChangeDriverPinUiEvent.DriverNumberChanged -> updateDriverNumber(event.value)
            is ChangeDriverPinUiEvent.CurrentPinChanged -> updateCurrentPin(event.value)
            is ChangeDriverPinUiEvent.NewPinChanged -> updateNewPin(event.value)
            is ChangeDriverPinUiEvent.RepeatPinChanged -> updateRepeatPin(event.value)

            ChangeDriverPinUiEvent.AcceptClicked -> onAcceptClicked()
            ChangeDriverPinUiEvent.CancelClicked -> emitEffect(ChangeDriverPinUiEffect.NavigateBack)
            ChangeDriverPinUiEvent.DialogAccepted -> Unit
        }
    }

    private fun updateDriverNumber(value: String) = updateState { it.copy(driverNumber = value) }
    private fun updateCurrentPin(value: String) = updateState { it.copy(currentPin = value) }
    private fun updateNewPin(value: String) = updateState { it.copy(newPin = value) }
    private fun updateRepeatPin(value: String) = updateState { it.copy(repeatPin = value) }

    private fun updateState(reducer: (ChangeDriverPinUiState) -> ChangeDriverPinUiState) {
        _uiState.update { current ->
            val newState = reducer(current)
            newState.copy(
                buttonsState = ChangeDriverPinButtonsReducer.reduce(newState)
            )
        }
    }

    private fun onAcceptClicked() {
        val state = _uiState.value
        val dialog = ChangeDriverPinButtonsReducer.validationDialog(state)

        if (dialog != null) {
            emitEffect(ChangeDriverPinUiEffect.ShowDialog(dialog))
            return
        }

        viewModelScope.launch {
            val userPresenter = UserModule.provideUserPresenter(
                object : ChangePasswordPinView {
                    override fun updateSuccess() {
                        viewModelScope.launch {
                            emitEffect(
                                ChangeDriverPinUiEffect.ShowDialog(
                                    ChangeDriverPinDialogModel(R.string.dialog_change_pin)
                                )
                            )
                        }
                    }

                    override fun updateFailure() {
                        viewModelScope.launch {
                            emitEffect(
                                ChangeDriverPinUiEffect.ShowDialog(
                                    ChangeDriverPinDialogModel(R.string.dialog_error_change_pin)
                                )
                            )
                        }
                    }
                },
                appContext
            )

            if (userPresenter != null) {
                emitEffect(
                    ChangeDriverPinUiEffect.ChangePinRequested(
                        driverNumber = state.driverNumber.trim(),
                        oldPin = state.currentPin.trim(),
                        newPin = state.newPin.trim()
                    )
                )
                userPresenter.changeDriverPin(
                    state.driverNumber.trim(),
                    state.currentPin.trim(),
                    state.newPin.trim(),
                    appContext
                )
            } else {
                emitEffect(
                    ChangeDriverPinUiEffect.ShowDialog(
                        ChangeDriverPinDialogModel(R.string.dialog_user_error)
                    )
                )
            }
        }
    }

    private fun emitEffect(effect: ChangeDriverPinUiEffect) {
        viewModelScope.launch { _uiEffect.emit(effect) }
    }
}
```

```kotlin
object ChangeDriverPinButtonsReducer {

    fun reduce(state: ChangeDriverPinUiState): ChangeDriverPinButtonsState {
        val hasAllInputs =
            state.driverNumber.isNotBlank() &&
            state.currentPin.isNotBlank() &&
            state.newPin.isNotBlank() &&
            state.repeatPin.isNotBlank()

        val pinsMatch = state.newPin == state.repeatPin
        val oldAndNewDifferent = state.currentPin != state.newPin

        return ChangeDriverPinButtonsState(
            acceptVisible = true,
            cancelVisible = true,
            acceptEnabled = hasAllInputs && pinsMatch && oldAndNewDifferent,
            cancelEnabled = true,
            acceptButtonType = ButtonType.ACCEPT,
            cancelButtonType = ButtonType.CANCEL
        )
    }

    fun validationDialog(state: ChangeDriverPinUiState): CustomDialog.CustomDialogModel? {
        return when {
            state.driverNumber.isEmpty() ||
                state.currentPin.isEmpty() ||
                state.newPin.isEmpty() ||
                state.repeatPin.isEmpty() -> {
                CustomDialog.CustomDialogModel(
                    title = R.string.dialog_fill_inputs,
                    buttons = arrayListOf(ButtonType.ACCEPT)
                )
            }

            state.newPin != state.repeatPin -> {
                CustomDialog.CustomDialogModel(
                    title = R.string.dialog_no_match_pin,
                    buttons = arrayListOf(ButtonType.ACCEPT)
                )
            }

            state.currentPin == state.newPin -> {
                CustomDialog.CustomDialogModel(
                    title = R.string.dialog_equal_pin,
                    buttons = arrayListOf(ButtonType.ACCEPT)
                )
            }

            else -> null
        }
    }
}

class ChangeDriverPinViewModel(application: Application) : ViewModel() {

    private val app = application

    private val _uiState = MutableStateFlow(ChangeDriverPinUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<ChangeDriverPinUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()

    private val _userCredentialsFlow = MutableSharedFlow<UserPresenter?>()
    val userCredentialsFlow = _userCredentialsFlow.asSharedFlow()

    fun onEvent(event: ChangeDriverPinUiEvent) {
        when (event) {
            is ChangeDriverPinUiEvent.DriverNumberChanged -> updateState { it.copy(driverNumber = event.value) }
            is ChangeDriverPinUiEvent.CurrentPinChanged -> updateState { it.copy(currentPin = event.value) }
            is ChangeDriverPinUiEvent.NewPinChanged -> updateState { it.copy(newPin = event.value) }
            is ChangeDriverPinUiEvent.RepeatPinChanged -> updateState { it.copy(repeatPin = event.value) }
            ChangeDriverPinUiEvent.AcceptClicked -> handleAccept()
            ChangeDriverPinUiEvent.CancelClicked -> emitEffect(ChangeDriverPinUiEffect.NavigateBack)
        }
    }

    private fun updateState(reducer: (ChangeDriverPinUiState) -> ChangeDriverPinUiState) {
        _uiState.update { current ->
            val next = reducer(current)
            next.copy(buttonsState = ChangeDriverPinButtonsReducer.reduce(next))
        }
    }

    private fun handleAccept() {
        val state = _uiState.value
        val validationDialog = ChangeDriverPinButtonsReducer.validationDialog(state)
        if (validationDialog != null) {
            emitEffect(ChangeDriverPinUiEffect.ShowDialog(validationDialog))
            return
        }

        viewModelScope.launch {
            val userPresenter = UserModule.provideUserPresenter(
                object : ChangePasswordPinView {
                    override fun updateSuccess() {
                        emitEffect(
                            ChangeDriverPinUiEffect.ShowDialog(
                                CustomDialog.CustomDialogModel(
                                    title = R.string.dialog_change_pin,
                                    buttons = arrayListOf(ButtonType.ACCEPT)
                                )
                            )
                        )
                    }

                    override fun updateFailure() {
                        emitEffect(
                            ChangeDriverPinUiEffect.ShowDialog(
                                CustomDialog.CustomDialogModel(
                                    title = R.string.dialog_error_change_pin,
                                    buttons = arrayListOf(ButtonType.ACCEPT)
                                )
                            )
                        )
                    }
                },
                app
            )

            _userCredentialsFlow.emit(userPresenter)
        }
    }

    fun changeDriverPin(userPresenter: UserPresenter, driverNumber: String, oldPin: String, newPin: String) {
        userPresenter.changeDriverPin(driverNumber, oldPin, newPin, app)
    }

    fun getUserPresenter(view: ChangePasswordPinView) {
        viewModelScope.launch {
            _userCredentialsFlow.emit(UserModule.provideUserPresenter(view, app))
        }
    }

    private fun emitEffect(effect: ChangeDriverPinUiEffect) {
        viewModelScope.launch { _uiEffect.emit(effect) }
    }
}
```

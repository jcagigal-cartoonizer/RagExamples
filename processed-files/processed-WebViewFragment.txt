# 2) ViewModel
```kotlin
package ifac.td.taxi.viewmodel
// # Block 13-1: import android.app.Application
import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.SavedStateHandle
import androidx.lifecycle.viewModelScope
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch
class WebViewComposeViewModel(
    application: Application,
    private val savedStateHandle: SavedStateHandle
) : AndroidViewModel(application) {
    private val _uiState = MutableStateFlow(WebViewUiState())
    val uiState: StateFlow<WebViewUiState> = _uiState.asStateFlow()
    private val _effects = MutableSharedFlow<WebViewUiEffect>()
    val effects: SharedFlow<WebViewUiEffect> = _effects
    fun onEvent(event: WebViewUiEvent) {
        when (event) {
            is WebViewUiEvent.OnScreenStarted -> {
                val homeUrl = event.homeUrl
                _uiState.update {
                    it.copy(
                        homeUrl = homeUrl,
                        currentUrl = homeUrl,
                        hasHomeUrl = homeUrl != null,
                        showWebView = homeUrl != null,
                        showError = homeUrl == null,
                        buttons = WebViewButtonsState.default(
                            canGoBack = false,
                            isHomePage = true
                        )
                    )
                }
                if (homeUrl != null) {
                    viewModelScope.launch {
                        _effects.emit(WebViewUiEffect.LoadUrl(homeUrl, clearHistory = true))
                    }
                } else {
                    viewModelScope.launch {
                        _effects.emit(
                            WebViewUiEffect.ShowDialog(
                                title = "Error",
                                message = "Web page URL is missing."
                            )
                        )
                    }
                }
            }
            WebViewUiEvent.OnHomeClicked -> {
                val homeUrl = _uiState.value.homeUrl ?: return
                _uiState.update {
                    it.copy(
                        showError = false,
                        showWebView = true,
                        currentUrl = homeUrl,
                        buttons = it.buttons.copy(
                            canGoBack = false,
                            isHomePage = true
                        ).rebuild()
                    )
                }
                viewModelScope.launch {
                    _effects.emit(WebViewUiEffect.LoadUrl(homeUrl, clearHistory = true))
                }
            }
            WebViewUiEvent.OnBackClicked -> {
                _effects.tryEmit(WebViewUiEffect.RequestWebViewGoBack)
            }
            WebViewUiEvent.OnReloadClicked -> {
                _effects.tryEmit(WebViewUiEffect.RequestWebViewReload)
            }
            WebViewUiEvent.OnCancelClicked -> {
                _effects.tryEmit(WebViewUiEffect.NavigateBack)
            }
            is WebViewUiEvent.OnPageFinished -> {
                val canGoBack = event.canGoBack
                val isHomePage = event.url == _uiState.value.homeUrl
                val showError = event.hasError
                _uiState.update {
                    it.copy(
                        currentUrl = event.url ?: it.currentUrl,
                        showError = showError,
                        showWebView = !showError,
                        buttons = it.buttons.copy(
                            canGoBack = canGoBack,
                            isHomePage = isHomePage
                        ).rebuild()
                    )
                }
                event.url?.let { url ->
                    viewModelScope.launch {
                        _effects.emit(WebViewUiEffect.SaveLastVisitedUrl(url))
                    }
                }
            }
            is WebViewUiEvent.OnPageError -> {
                if (event.errorCode == android.webkit.WebViewClient.ERROR_HOST_LOOKUP) {
                    _uiState.update {
                        it.copy(
                            showError = true,
                            showWebView = false
                        )
                    }
                    viewModelScope.launch {
                        _effects.emit(
                            WebViewUiEffect.ShowDialog(
                                title = "Network error",
                                message = event.description ?: "Unable to load the page."
                            )
                        )
                    }
                }
            }
            WebViewUiEvent.OnDialogDismissed -> {
                _uiState.update { it.copy(dialogState = null) }
            }
            WebViewUiEvent.OnWebViewCanGoBackChanged -> {
                val state = _uiState.value
                _uiState.update {
                    it.copy(
                        buttons = it.buttons.copy(
                            canGoBack = event.canGoBack,
                            isHomePage = state.currentUrl == state.homeUrl
                        ).rebuild()
                    )
                }
            }
        }
    }
    fun setDialog(dialogState: DialogState?) {
        _uiState.update { it.copy(dialogState = dialogState) }
    }
}
```

# 3) UI state, events, effects
```kotlin
package ifac.td.taxi.viewmodel
data class WebViewUiState(
    val homeUrl: String? = null,
    val currentUrl: String? = null,
    val hasHomeUrl: Boolean = false,
    val showError: Boolean = false,
    val showWebView: Boolean = true,
    val dialogState: DialogState? = null,
    val buttons: WebViewButtonsState = WebViewButtonsState.default(),
)
sealed interface WebViewUiEvent {
    data class OnScreenStarted(val homeUrl: String?) : WebViewUiEvent
    object OnHomeClicked : WebViewUiEvent
    object OnBackClicked : WebViewUiEvent
    object OnReloadClicked : WebViewUiEvent
    object OnCancelClicked : WebViewUiEvent
    data class OnPageFinished(
        val url: String?,
        val canGoBack: Boolean,
        val hasError: Boolean
    ) : WebViewUiEvent
    data class OnPageError(
        val errorCode: Int,
        val description: String?
    ) : WebViewUiEvent
    data class OnWebViewCanGoBackChanged(val canGoBack: Boolean) : WebViewUiEvent
    object OnDialogDismissed : WebViewUiEvent
}
sealed interface WebViewUiEffect {
    data class LoadUrl(val url: String, val clearHistory: Boolean) : WebViewUiEffect
    data object RequestWebViewGoBack : WebViewUiEffect
    data object RequestWebViewReload : WebViewUiEffect
    data object NavigateBack : WebViewUiEffect
    data class SaveLastVisitedUrl(val url: String) : WebViewUiEffect
    data class ShowDialog(val title: String, val message: String) : WebViewUiEffect
}
data class DialogState(
    val title: String,
    val message: String,
    val confirmText: String = "OK",
    val dismissText: String? = null
)
```

```kotlin
package ifac.td.taxi.viewmodel
// # Block 202-2: import androidx.compose.ui.graphics.Color
import androidx.compose.ui.graphics.Color
data class WebViewButtonsState(
    val home: WebViewButtonVisualState = WebViewButtonVisualState(),
    val goBack: WebViewButtonVisualState = WebViewButtonVisualState(),
    val cancel: WebViewButtonVisualState = WebViewButtonVisualState(),
    val reload: WebViewButtonVisualState = WebViewButtonVisualState(),
    val canGoBack: Boolean = false,
    val isHomePage: Boolean = true
) {
    companion object {
        fun default(
            canGoBack: Boolean = false,
            isHomePage: Boolean = true
        ): WebViewButtonsState = WebViewButtonsState(
            canGoBack = canGoBack,
            isHomePage = isHomePage
        ).rebuild()
    }
    fun rebuild(): WebViewButtonsState {
        val goBackEnabled = canGoBack && !isHomePage
        return copy(
            home = WebViewButtonVisualState.defaultEnabled(),
            goBack = if (goBackEnabled) WebViewButtonVisualState.defaultEnabled() else WebViewButtonVisualState.defaultDisabled(),
            cancel = WebViewButtonVisualState.defaultEnabled(),
            reload = WebViewButtonVisualState.defaultEnabled()
        )
    }
}
data class WebViewButtonVisualState(
    val enabled: Boolean = true,
    val visible: Boolean = true,
    val backgroundColor: Color = Color(0xFF1976D2),
    val contentColor: Color = Color.White,
    val borderColor: Color = Color.Transparent
) {
    companion object {
        fun defaultEnabled() = WebViewButtonVisualState(
            enabled = true,
            visible = true,
            backgroundColor = Color(0xFF1976D2),
            contentColor = Color.White,
            borderColor = Color.Transparent
        )
        fun defaultDisabled() = WebViewButtonVisualState(
            enabled = false,
            visible = true,
            backgroundColor = Color(0xFFE0E0E0),
            contentColor = Color(0xFF9E9E9E),
            borderColor = Color(0xFFBDBDBD)
        )
    }
}
```

# 5) Compose screen with preserved navigation and lifecycle effect collection
```kotlin
package ifac.td.taxi.ui.screen
// # Block 260-3: import android.webkit.WebView
import android.webkit.WebView
import android.webkit.WebViewClient
import androidx.compose.foundation.background
import androidx.compose.foundation.layout.*
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Modifier
import androidx.compose.ui.platform.LocalContext
import androidx.compose.ui.unit.dp
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.lifecycle.viewmodel.compose.viewModel
import androidx.navigation.NavController
import ifac.td.taxi.viewmodel.*
import kotlinx.coroutines.flow.collectLatest
@Composable
fun WebViewRoute(
    navController: NavController,
    homeUrl: String?,
    viewModel: WebViewViewModel = viewModel()
) {
    val state by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current
    var webView: WebView? by remember { mutableStateOf(null) }
    LaunchedEffect(homeUrl) {
        viewModel.onEvent(WebViewUiEvent.OnScreenStarted(homeUrl))
    }
    LaunchedEffect(Unit) {
        viewModel.effects.collectLatest { effect ->
            when (effect) {
                is WebViewUiEffect.LoadUrl -> {
                    if (effect.clearHistory) {
                        webView?.clearHistory()
                    }
                    webView?.loadUrl(effect.url)
                }
                WebViewUiEffect.RequestWebViewGoBack -> {
                    if (webView?.canGoBack() == true) {
                        webView?.goBack()
                    }
                }
                WebViewUiEffect.RequestWebViewReload -> {
                    webView?.reload()
                }
                WebViewUiEffect.NavigateBack -> {
                    navController.popBackStack()
                }
                is WebViewUiEffect.SaveLastVisitedUrl -> {
                    // Replace with DataStore / SharedPreferences write
                    // For parity with fragment behavior:
                    // saveLastVisitedUrl(effect.url)
                }
                is WebViewUiEffect.ShowDialog -> {
                    viewModel.setDialog(
                        DialogState(
                            title = effect.title,
                            message = effect.message
                        )
                    )
                }
            }
        }
    }
    WebViewScreen(
        state = state,
        onEvent = viewModel::onEvent,
        onWebViewCreated = { createdWebView ->
            webView = createdWebView
        },
        onNavigateBack = { navController.popBackStack() }
    )
}
```

# 7) Android WebView in Compose
```kotlin
// # Block 383-4: import android.annotation.SuppressLint
import android.annotation.SuppressLint
import android.webkit.WebSettings
import android.webkit.WebView
import android.webkit.WebViewClient
import androidx.compose.runtime.Composable
import androidx.compose.runtime.DisposableEffect
import androidx.compose.ui.Modifier
import androidx.compose.ui.viewinterop.AndroidView
@SuppressLint("SetJavaScriptEnabled")
@Composable
fun AndroidWebView(
    visible: Boolean,
    url: String?,
    onCreated: (WebView) -> Unit,
    onPageFinished: (String?, Boolean, Boolean) -> Unit,
    onError: (Int, String?) -> Unit,
    modifier: Modifier = Modifier
) {
    if (!visible) return
    AndroidView(
        modifier = modifier,
        factory = { context ->
            WebView(context).apply {
                settings.apply {
                    cacheMode = WebSettings.LOAD_NO_CACHE
                    domStorageEnabled = true
                    javaScriptEnabled = true
                    setSupportZoom(false)
                }
                webViewClient = object : WebViewClient() {
                    override fun onPageFinished(view: WebView?, url: String?) {
                        super.onPageFinished(view, url)
                        onPageFinished(url, canGoBack(), false)
                    }
                    @Deprecated("Deprecated in API 23")
                    override fun onReceivedError(
                        view: WebView?,
                        errorCode: Int,
                        description: String?,
                        failingUrl: String?
                    ) {
                        super.onReceivedError(view, errorCode, description, failingUrl)
                        onError(errorCode, description)
                        onPageFinished(failingUrl, canGoBack(), true)
                    }
                }
                onCreated(this)
                url?.let { loadUrl(it) }
            }
        },
        update = { webView ->
            if (webView.url == null && url != null) {
                webView.loadUrl(url)
            }
        }
    )
}
```

```kotlin
@Composable
fun StyledButton(
    text: String,
    state: WebViewButtonVisualState,
    modifier: Modifier = Modifier,
    onClick: () -> Unit
) {
    if (!state.visible) return
    Button(
        onClick = onClick,
        enabled = state.enabled,
        modifier = modifier,
        colors = ButtonDefaults.buttonColors(
            containerColor = state.backgroundColor,
            contentColor = state.contentColor,
            disabledContainerColor = state.backgroundColor,
            disabledContentColor = state.contentColor
        ),
        border = androidx.compose.foundation.BorderStroke(1.dp, state.borderColor)
    ) {
        Text(text = text)
    }
}
```

```kotlin
package ifac.td.taxi.ui.screen

@Composable
fun WebViewCustomDialog(
    dialogState: DialogState,
    onConfirm: () -> Unit,
    onDismiss: () -> Unit
) {
    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(Color.Black.copy(alpha = 0.45f)),
        contentAlignment = Alignment.Center
    ) {
        Surface(
            shape = RoundedCornerShape(16.dp),
            tonalElevation = 8.dp,
            color = MaterialTheme.colorScheme.surface,
            modifier = Modifier
                .padding(24.dp)
                .fillMaxWidth()

            Column(modifier = Modifier.padding(20.dp)) {
                Text(
                    text = dialogState.title,
                    style = MaterialTheme.typography.titleLarge
                )
                Spacer(modifier = Modifier.height(12.dp))
                Text(
                    text = dialogState.message,
                    style = MaterialTheme.typography.bodyMedium
                )
                Spacer(modifier = Modifier.height(20.dp))
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.End
                ) {
                    if (dialogState.dismissText != null) {
                        TextButton(onClick = onDismiss) {
                            Text(dialogState.dismissText)
                        }
                        Spacer(modifier = Modifier.width(8.dp))
                    }
                    Button(onClick = onConfirm) {
                        Text(dialogState.confirmText)
                    }
                }
            }
        }
    }
}
```

# 13) Example of how to host this in a Fragment while preserving existing navigation graph
```kotlin
class WebViewComposeFragment : Fragment() {
    private val viewModel: WebViewViewModel by viewModels()
    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View {
        val homeUrl = arguments?.getString("webViewURL")
        return ComposeView(requireContext()).apply {
            setContent {
                val navController = rememberNavController()
                WebViewRoute(
                    navController = navController,
                    homeUrl = homeUrl,
                    viewModel = viewModel
                )
            }
        }
    }
}
```

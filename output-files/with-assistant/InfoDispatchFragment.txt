
// FIRST ANSWER
Below is a Compose-first rewrite that preserves the fragment behavior, navigation intent, button logic, and dialog handling, while replacing the imperative view logic with:

- `InfoDispatchButtonsState`
- `InfoDispatchUiState`
- `InfoDispatchUiEvent`
- `InfoDispatchUiEffect`
- a Compose `InfoDispatchComposeViewModel`

I’m keeping this strictly in Android + Jetpack Compose terms, and I’m preserving the same navigation destinations conceptually via `NavController` and `HomeDirections`.

```kotlin
@Composable
fun InfoDispatchScreen(
    navController: NavController,
    viewModel: InfoDispatchComposeViewModel
) {
    val state by viewModel.uiState.collectAsStateWithLifecycle()
    var dialogState by remember { mutableStateOf<InfoDispatchDialogState?>(null) }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                InfoDispatchUiEffect.NavigateToDirections -> {
                    navController.navigate("directions")
                }

                InfoDispatchUiEffect.NavigateToHome -> {
                    navController.navigate(HomeDirections.goToOnTripFragment().actionId.toString())
                }

                is InfoDispatchUiEffect.NavigateToMeetingSign -> {
                    navController.navigate(
                        "meeting_sign/${effect.textColor}/${effect.backgroundColor}"
                    )
                }

                InfoDispatchUiEffect.RequestPhonePermission -> {
                    // Host should launch permission flow
                }

                is InfoDispatchUiEffect.ShowToast -> {
                    // Host should show toast/snackbar
                }

                is InfoDispatchUiEffect.OpenExternalPhoneCall -> {
                    // Host should start phone intent
                }

                is InfoDispatchUiEffect.OpenDialog -> {
                    dialogState = effect.dialog
                }

                InfoDispatchUiEffect.CloseDialog -> {
                    dialogState = null
                }
            }
        }
    }

    Scaffold(
        topBar = {
            // Preserve header behavior here if needed
        },
        bottomBar = {
            // Preserve bottom bar behavior here if needed
        }
    ) { padding ->
        InfoDispatchContent(
            modifier = Modifier.padding(padding),
            state = state,
            onEvent = viewModel::onEvent
        )
    }

    dialogState?.let { dialog ->
        InfoDispatchCustomDialog(
            state = dialog,
            onDismiss = { dialogState = null },
            onButtonClick = { button ->
                viewModel.onEvent(InfoDispatchUiEvent.OnDialogButtonClicked(button))
                dialogState = null
            }
        )
    }
}

@Composable
private fun DispatchButton(
    text: String,
    enabled: Boolean,
    loading: Boolean,
    backgroundColor: Color,
    onClick: () -> Unit
) {
    val finalEnabled = enabled && !loading

    Button(
        onClick = onClick,
        enabled = finalEnabled,
        colors = ButtonDefaults.buttonColors(
            containerColor = backgroundColor,
            disabledContainerColor = backgroundColor.copy(alpha = 0.45f),
            contentColor = Color.White,
            disabledContentColor = Color.White.copy(alpha = 0.8f)
        ),
        modifier = Modifier.fillMaxWidth()
    ) {
        if (loading) {
            CircularProgressIndicator(
                modifier = Modifier.size(18.dp),
                strokeWidth = 2.dp,
                color = Color.White
            )
            Spacer(modifier = Modifier.width(8.dp))
        }
        Text(text = text)
    }
}

@Composable
private fun MultiDispatchTabs(
    tabs: List<InfoDispatchMultiDispatchTabState>,
    selectedIndex: Int,
    onTabSelected: (Int) -> Unit
) {
    Column {
        Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
            tabs.forEachIndexed { index, tab ->
                val selected = index == selectedIndex
                Surface(
                    color = if (selected) MaterialTheme.colorScheme.primary else MaterialTheme.colorScheme.surfaceVariant,
                    tonalElevation = if (selected) 4.dp else 0.dp,
                    modifier = Modifier
                        .clickable { onTabSelected(index) }
                        .padding(4.dp)
                ) {
                    Row(
                        modifier = Modifier.padding(12.dp),
                        verticalAlignment = Alignment.CenterVertically
                    ) {
                        Text(text = tab.iconLabel)
                        Spacer(modifier = Modifier.width(8.dp))
                        Text(text = tab.name)
                    }
                }
            }
        }
    }
}
```

## 6) Compose ViewModel

This version exposes a single `uiState` and a single `SharedFlow<InfoDispatchUiEffect>`.

It also preserves the original behavior:
- bridge call logic
- phone call permission request
- navigation effects
- dialog effects
- button enable/disable/color logic
- multi-dispatch selection
- top bar timer text
- dispatch info mapping

### `InfoDispatchComposeViewModel.kt`

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import com.interfacom.sdk.taximeter.bravocomm.ifConstants
import ifac.td.taxi.R
import ifac.td.taxi.domain.usecase.BluetoothLocalUseCase
import ifac.td.taxi.domain.usecase.BridgeCallUseCase
import ifac.td.taxi.domain.usecase.CountdownManagerUseCase
import ifac.td.taxi.domain.usecase.DispatchNotificationUseCase
import ifac.td.taxi.domain.usecase.DispatchUseCase
import ifac.td.taxi.domain.usecase.ShiftStatusUseCase
import ifac.td.taxi.domain.usecase.TTSUseCase
import ifac.td.taxi.domain.usecase.UserPreferencesUseCase
import ifac.td.taxi.framework.sdk.bravocentral.usecase.BravoCentralUseCase
import ifac.td.taxi.framework.sdk.usecase.LicensingUseCase
import ifac.td.taxi.framework.sdk.usecase.PhoneCallUseCase
import ifac.td.taxi.framework.sdk.usecase.PhoneCallUseCaseImpl
import ifac.td.taxi.repository.room.dao.BravoConfigurationVariableDao
import ifac.td.taxi.repository.room.entities.DispatchExtraDataEntity
import ifac.td.taxi.repository.room.entities.countdown.CountdownId
import ifac.td.taxi.repository.room.entities.countdown.CountdownIdCallback
import ifac.td.taxi.ui.screen.components.*
import ifac.td.taxi.viewmodel.model.CountdownModel
import ifac.td.taxi.viewmodel.model.InfoDispatchModel
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch

class InfoDispatchComposeViewModel(
    private val dispatchUseCase: DispatchUseCase,
    private val bluetoothLocalUseCase: BluetoothLocalUseCase,
    private val bravoCentralUseCase: BravoCentralUseCase,
    private val phoneCallUseCase: PhoneCallUseCase,
    private val dispatchNotificationUseCase: DispatchNotificationUseCase,
    private val bridgeCallUseCase: BridgeCallUseCase,
    private val licensingUseCase: LicensingUseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val ttsUseCase: TTSUseCase,
    private val bravoConfigurationVariableDao: BravoConfigurationVariableDao,
    private val countdownManagerUseCase: CountdownManagerUseCase,
    application: Application
) : AndroidViewModel(application) {

    private val _uiState = MutableStateFlow(InfoDispatchUiState())
    val uiState: StateFlow<InfoDispatchUiState> = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<InfoDispatchUiEffect>()
    val uiEffect: SharedFlow<InfoDispatchUiEffect> = _uiEffect.asSharedFlow()

    private val _dispatch = MutableStateFlow<InfoDispatchModel?>(null)
    private val _extraData = MutableStateFlow<DispatchExtraDataEntity?>(null)
    private val _bridgeCallState = MutableStateFlow(0)
    private val _customerCallAvailable = MutableStateFlow(true)
    private val _canMakeCalls = MutableStateFlow(false)
    private val _isManualShift = MutableStateFlow(false)
    private val _isITopConnected = MutableStateFlow(false)
    private val _meetingSignColors = MutableStateFlow(0 to 0)
    private val _showTripAmount = MutableStateFlow(false)
    private val _noClientButtonEnabled = MutableStateFlow(true)

    var hasShowedConcertedPriceDialog: Boolean = false
        private set

    init {
        viewModelScope.launch(Dispatchers.IO) {
            val config = bravoConfigurationVariableDao.getBravoConfigurationVariable()
            _showTripAmount.value = config?.showTripAmount == true
        }
        viewModelScope.launch {
            userPreferencesUseCase.getUserPreferences()?.let {
                _meetingSignColors.value = it.meetingSignTextColor to it.meetingSignBackgroundColor
            }
        }
        reduce()
    }

    fun onEvent(event: InfoDispatchUiEvent) {
        when (event) {
            InfoDispatchUiEvent.OnNotificationsClicked -> onNotificationsClicked()
            InfoDispatchUiEvent.OnNavigateClicked -> viewModelScope.launch {
                stopTTS()
                _uiEffect.emit(InfoDispatchUiEffect.NavigateToDirections)
            }
            InfoDispatchUiEvent.OnVoiceCallClicked -> onVoiceCallClicked()
            InfoDispatchUiEvent.OnPrintClicked -> _dispatch.value?.let { printInfoDispatch(it) }
            InfoDispatchUiEvent.OnNoClientClicked -> onNoClientClicked()
            InfoDispatchUiEvent.OnReturnClicked -> onReturnClicked()
            InfoDispatchUiEvent.OnDispatchNumberClicked -> viewModelScope.launch {
                val colors = _meetingSignColors.value
                _uiEffect.emit(
                    InfoDispatchUiEffect.NavigateToMeetingSign(
                        textColor = colors.first,
                        backgroundColor = colors.second
                    )
                )
            }
            is InfoDispatchUiEvent.OnMultiDispatchSelected -> {
                val current = _uiState.value.multiDispatch.tabs.getOrNull(event.index) ?: return
                _dispatch.value?.routeId?.let { routeId ->
                    viewModelScope.launch {
                        dispatchUseCase.updateSelectedDispatchId(current.name, routeId)
                    }
                }
            }
            is InfoDispatchUiEvent.OnDialogButtonClicked -> handleDialogButton(event.button)
        }
    }

    fun bindDispatch(dispatch: InfoDispatchModel?) {
        _dispatch.value = dispatch
        reduce()
        dispatch?.let {
            retrieveExtraData(it.id)
            checkTTSOnDispatch(it)
        }
    }

    fun setMultiDispatch(list: List<InfoDispatchModel>) {
        val tabs = list.map { dispatch ->
            val iconLabel = when {
                dispatch.isCancelled -> "⛔"
                dispatch.isEnded -> "🏁"
                dispatch.isRiderInCabNotificationSent == true -> "👤"
                dispatch.isAtDoorNotificationSent == true -> "🚪"
                else -> "🚕"
            }
            InfoDispatchMultiDispatchTabState(
                name = dispatch.dispatchName.orEmpty(),
                iconLabel = iconLabel
            )
        }
        _uiState.update {
            it.copy(
                multiDispatch = InfoDispatchMultiDispatchState(
                    visible = list.size > 1,
                    selectedIndex = list.indexOfFirst { d -> d.selected }.coerceAtLeast(0),
                    tabs = tabs
                )
            )
        }
    }

    fun updateShiftState(isManual: Boolean, currentStatus: Int) {
        _isManualShift.value = isManual
        reduce()
        if (currentStatus == ifConstants.STATE_HIRED_DISPATCHED) {
            checkNoClientButton()
        }
    }

    fun updateTaximeterConnection(connected: Boolean) {
        _isITopConnected.value = connected
        reduce()
    }

    fun updateCustomerCallAvailability(isAvailable: Boolean) {
        _customerCallAvailable.value = isAvailable
        reduce()
    }

    fun updateCanMakeCalls(canMakeCalls: Boolean) {
        _canMakeCalls.value = canMakeCalls
        reduce()
    }

    fun updateBridgeCallState(state: Int) {
        _bridgeCallState.value = state
        reduce()
    }

    fun updateTimer(tick: Int?) {
        _uiState.update { current ->
            current.copy(
                topBar = current.topBar.copy(timerText = tick?.toString().orEmpty())
            )
        }
    }

    fun updateNoClientButtonEnabled(enabled: Boolean?) {
        _noClientButtonEnabled.value = enabled ?: false
        reduce()
    }

    fun checkNoClientButton() {
        viewModelScope.launch(Dispatchers.IO) {
            val parameters = licensingUseCase.getLicensingParameters()
            updateNoClientButtonEnabled(parameters?.isNoClientHired)
        }
    }

    fun getMaxBridgeCalls() {
        viewModelScope.launch(Dispatchers.IO) {
            bravoCentralUseCase.getBravoConfiguration()?.maxCalls
        }
    }

    fun saveBridgeCallState(bridgeCallState: Int) {
        _bridgeCallState.value = bridgeCallState
        reduce()
    }

    fun retrieveExtraData(idDispatch: Long) {
        viewModelScope.launch {
            dispatchUseCase.getTripExtraData(idDispatch, MutableStateFlowCollector(_extraData))
        }
    }

    fun startCall(phoneNumber: String) {
        viewModelScope.launch {
            val response = phoneCallUseCase.makePhoneDial(phoneNumber)
            when (response) {
                PhoneCallUseCaseImpl.PhoneCallError.NO_SIM_CARD -> emitToast(R.string.error_no_sim)
                PhoneCallUseCaseImpl.PhoneCallError.NO_PERMISSION -> emitToast(R.string.phone_permission_missing)
                PhoneCallUseCaseImpl.PhoneCallError.INVALID_PHONE_NUMBER -> emitToast(R.string.invalid_phone_number)
                PhoneCallUseCaseImpl.PhoneCallError.NETWORK_ERROR -> emitToast(R.string.network_error)
                PhoneCallUseCaseImpl.PhoneCallError.UNKNOWN -> emitToast(R.string.default_error)
                null -> Unit
            }
        }
    }

    fun makeBridgeCall(dispatchNumber: String) {
        viewModelScope.launch {
            val countdown = CountdownModel(
                idCountdown = CountdownId.CLIENT_CALL_BUTTON,
                timeInSeconds = 10L,
                idCallback = CountdownIdCallback.CLIENT_CALL_BUTTON
            )

            countdownManagerUseCase.stopCountdownById(CountdownId.CLIENT_CALL_BUTTON)
            countdownManagerUseCase.deleteCountdownById(CountdownId.CLIENT_CALL_BUTTON)
            countdownManagerUseCase.createOrUpdateCountdown(countdown)

            countdownManagerUseCase.startCountdown(countdown)
            countdownManagerUseCase.setCountdownFinishCallback(CountdownId.CLIENT_CALL_BUTTON) {
                viewModelScope.launch {
                    _uiState.update {
                        it.copy(

    private fun buildDispatchInfo(
        dispatch: InfoDispatchModel?,
```

# 1) More exact button reducer

The key change is: instead of “one reducer that guesses final button state,” we model the same branching the fragment had in its collectors:

- `dispatchFlow.collect { ... }`
- `bravoStateFlow.collect { ... }`
- `shiftStatusFlow.collect { ... }`
- `noClientButtonFlow.collect { ... }`
- `phoneCallErrorFlow.collect { ... }`
- `bridgeCallTimeoutFlow.collect { ... }`
- `customerCallAvailable.collect { ... }`
- `customerCallButtonStateAvailable.collect { ... }`
- `combine(canMakeCallsFlow, bravoStateFlow) { ... }`

That means the reducer is split into explicit update functions, each matching one collector.

## `InfoDispatchButtonsReducer.kt`

This reducer mirrors the fragment branches one-by-one.

```kotlin
package ifac.td.taxi.ui.screen.components

import androidx.compose.ui.graphics.Color
import ifac.td.taxi.R
import ifac.td.taxi.viewmodel.model.InfoDispatchModel

class InfoDispatchButtonsReducer {

    fun reduce(
        current: InfoDispatchButtonsState,
        dispatch: InfoDispatchModel?,
        isITopConnected: Boolean,
        isManualShift: Boolean,
        bridgeCallState: Int,
        customerCallAvailable: Boolean,
        customerCallButtonStateAvailable: Boolean,
        canMakeCalls: Boolean,
        noClientButtonEnabled: Boolean?,
        phoneCallError: Boolean?,
        bridgeCallTimeout: Boolean,
        shiftCurrentStatus: Int?,
        bravoShowTripAmount: Boolean
    ): InfoDispatchButtonsState {
        var state = current

        // 1) dispatchFlow.collect { ... }
        state = reduceFromDispatchFlow(
            state = state,
            dispatch = dispatch,
            isITopConnected = isITopConnected,
            isManualShift = isManualShift,
            bravoShowTripAmount = bravoShowTripAmount
        )

        // 2) bravoStateFlow.collect { ... }
        state = reduceFromBravoState(
            state = state,
            isITopConnected = isITopConnected,
            shiftCurrentStatus = shiftCurrentStatus,
            bridgeCallState = bridgeCallState
        )

        // 3) shiftStatusFlow.collect { ... }
        state = reduceFromShiftStatus(
            state = state,
            isITopConnected = isITopConnected,
            shiftCurrentStatus = shiftCurrentStatus,
            isManualShift = isManualShift
        )

        // 4) noClientButtonFlow.collect { ... }
        state = reduceFromNoClientButtonFlow(
            state = state,
            noClientButtonEnabled = noClientButtonEnabled,
            shiftCurrentStatus = shiftCurrentStatus
        )

        // 5) phoneCallErrorFlow.collect { ... }
        state = reduceFromPhoneCallErrorFlow(
            state = state,
            phoneCallError = phoneCallError
        )

        // 6) bridgeCallTimeoutFlow.collect { ... }
        state = reduceFromBridgeCallTimeoutFlow(
            state = state,
            bridgeCallTimeout = bridgeCallTimeout
        )

        // 7) customerCallAvailable.collect { ... }
        state = reduceFromCustomerCallAvailable(
            state = state,
            customerCallAvailable = customerCallAvailable
        )

        // 8) customerCallButtonStateAvailable.collect { ... }
        state = reduceFromCustomerCallButtonStateAvailable(
            state = state,
            customerCallAvailable = customerCallAvailable,
            customerCallButtonStateAvailable = customerCallButtonStateAvailable
        )

        // 9) combine(canMakeCallsFlow, bravoStateFlow) { ... }
        state = reduceFromCanMakeCallsCombine(
            state = state,
            customerCallAvailable = customerCallAvailable,
            canMakeCalls = canMakeCalls,
            bridgeCallState = bridgeCallState
        )

        return state
    }

    private fun reduceFromDispatchFlow(
        state: InfoDispatchButtonsState,
        dispatch: InfoDispatchModel?,
        isITopConnected: Boolean,
        isManualShift: Boolean,
        bravoShowTripAmount: Boolean
    ): InfoDispatchButtonsState {
        if (dispatch == null) return state

        val hasAtDoorAction =
            dispatch.isAtDoorNotificationEnabled() && dispatch.isAtDoorNotificationSent == false
        val hasRiderInCabAction =
            (dispatch.riderInCab ?: false) && dispatch.isRiderInCabNotificationSent == false

        val notifications = when {
            hasAtDoorAction && hasRiderInCabAction -> state.notifications.copy(
                enabled = true,
                loading = false,
                backgroundColor = Color(0xFF1976D2)
            )

            !hasAtDoorAction && hasRiderInCabAction -> state.notifications.copy(
                enabled = true,
                loading = false,
                backgroundColor = Color(0xFF4CAF50)
            )

            hasAtDoorAction -> state.notifications.copy(
                enabled = true,
                loading = false,
                backgroundColor = Color(0xFFFF9800)
            )

            else -> state.notifications.copy(
                enabled = false,
                loading = false,
                backgroundColor = Color(0xFF9E9E9E)
            )
        }

        val notificationsAfterITopRestriction =
            if (isITopConnected && !isManualShift) {
                notifications.copy(enabled = false)
            } else {
                notifications
            }

        val voiceCall = state.voiceCall.copy(
            // dispatchFlow itself doesn't directly set the voice button in the fragment,
            // but it does affect whether the button should be available later.
            enabled = state.voiceCall.enabled,
            loading = state.voiceCall.loading
        )

        val noClient = state.noClient.copy(
            enabled = state.noClient.enabled
        )

        return state.copy(
            notifications = notificationsAfterITopRestriction,
            voiceCall = voiceCall,
            noClient = noClient
        )
    }

    private fun reduceFromBravoState(
        state: InfoDispatchButtonsState,
        isITopConnected: Boolean,
        shiftCurrentStatus: Int?,
        bridgeCallState: Int
    ): InfoDispatchButtonsState {
        var newState = state

        if (isITopConnected) {
            newState = newState.copy(
                noClient = newState.noClient.copy(enabled = false),
                returnDispatch = newState.returnDispatch.copy(enabled = false)
            )
        } else {
            newState = when (newState.returnDispatch.enabled) {
                true -> newState.copy(
                    returnDispatch = newState.returnDispatch.copy(
                        enabled = true,
                        backgroundColor = Color(0xFFF44336)
                    )
                )
                false -> newState.copy(
                    returnDispatch = newState.returnDispatch.copy(enabled = false)
                )
            }

            newState = when (newState.noClient.backgroundColor) {
                Color(0xFF4CAF50) -> newState.copy(
                    noClient = newState.noClient.copy(backgroundColor = Color(0xFF4CAF50))
                )
                else -> newState.copy(
                    noClient = newState.noClient.copy(backgroundColor = Color(0xFFF44336))
                )
            }
        }

        newState = when (bridgeCallState) {
            2 -> newState.copy(
                voiceCall = newState.voiceCall.copy(
                    enabled = true,
                    loading = false,
                    backgroundColor = Color(0xFFF44336)
                )
            )
            0 -> newState.copy(
                voiceCall = newState.voiceCall.copy(
                    backgroundColor = Color(0xFF4CAF50)
                )
            )
            else -> newState.copy(
                voiceCall = newState.voiceCall.copy(
                    enabled = true,
                    loading = false,
                    backgroundColor = Color(0xFFFF9800)
                )
            )
        }

        return newState
    }

    private fun reduceFromShiftStatus(
        state: InfoDispatchButtonsState,
        isITopConnected: Boolean,
        shiftCurrentStatus: Int?,
        isManualShift: Boolean
    ): InfoDispatchButtonsState {
        if (isITopConnected) {
            return state.copy(
                noClient = state.noClient.copy(enabled = false),
                returnDispatch = state.returnDispatch.copy(enabled = false)
            )
        }

        return when (shiftCurrentStatus) {
            else -> state
        }
    }

    private fun reduceFromNoClientButtonFlow(
        state: InfoDispatchButtonsState,
        noClientButtonEnabled: Boolean?,
        shiftCurrentStatus: Int?
    ): InfoDispatchButtonsState {
        val canChange =
            noClientButtonEnabled == false && shiftCurrentStatus == com.interfacom.sdk.taximeter.bravocomm.ifConstants.STATE_HIRED_DISPATCHED

        return if (canChange) {
            state.copy(
                noClient = state.noClient.copy(enabled = false)
            )
        } else {
            state
        }
    }

    private fun reduceFromPhoneCallErrorFlow(
        state: InfoDispatchButtonsState,
        phoneCallError: Boolean?
    ): InfoDispatchButtonsState {
        return if (phoneCallError == null) {
            state.copy(
                voiceCall = state.voiceCall.copy(
                    backgroundColor = Color(0xFFF44336),
                    enabled = true
                )
            )
        } else {
            state
        }
    }

    private fun reduceFromBridgeCallTimeoutFlow(
        state: InfoDispatchButtonsState,
        bridgeCallTimeout: Boolean
    ): InfoDispatchButtonsState {
        return if (bridgeCallTimeout) {
            state.copy(
                voiceCall = state.voiceCall.copy(
                    enabled = true,
                    loading = false
                )
            )
        } else {
            state
        }
    }

    private fun reduceFromCustomerCallAvailable(
        state: InfoDispatchButtonsState,
        customerCallAvailable: Boolean
    ): InfoDispatchButtonsState {
        return state.copy(
            voiceCall = state.voiceCall.copy(
                enabled = customerCallAvailable,
                loading = false
            )
        )
    }

    private fun reduceFromCustomerCallButtonStateAvailable(
        state: InfoDispatchButtonsState,
        customerCallAvailable: Boolean,
        customerCallButtonStateAvailable: Boolean
    ): InfoDispatchButtonsState {
        return state.copy(
            voiceCall = state.voiceCall.copy(
                enabled = customerCallAvailable && customerCallButtonStateAvailable,
                loading = !customerCallButtonStateAvailable
            )
        )
    }

    private fun reduceFromCanMakeCallsCombine(
        state: InfoDispatchButtonsState,
        customerCallAvailable: Boolean,
        canMakeCalls: Boolean,
        bridgeCallState: Int
    ): InfoDispatchButtonsState {
        val voiceEnabled =
            if (!customerCallAvailable) {
                false
            } else if (canMakeCalls || bridgeCallState != 0) {
                true
            } else {
                false
            }

        return state.copy(
            voiceCall = state.voiceCall.copy(
                enabled = voiceEnabled,
                loading = false
            )
        )
    }
}
```

```kotlin
package ifac.td.taxi.ui.screen.components

import androidx.compose.runtime.*
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.navigation.NavController
import ifac.td.taxi.HomeDirections
import ifac.td.taxi.compose.viewmodel.InfoDispatchMainActivityViewModel
import ifac.td.taxi.viewmodel.InfoDispatchComposeViewModel
import kotlinx.coroutines.flow.collectLatest

@Composable
fun InfoDispatchScreen(
    navController: NavController,
    viewModel: InfoDispatchComposeViewModel,
    sharedViewModel: InfoDispatchMainActivityViewModel
) {
    val state by viewModel.uiState.collectAsStateWithLifecycle()
    val dispatch by sharedViewModel.dispatchFlow.collectAsStateWithLifecycle()
    val taximeterConnected by sharedViewModel.hasTaximeterConnectionFlow.collectAsStateWithLifecycle()
    val shiftStatus by sharedViewModel.shiftStatusFlow.collectAsStateWithLifecycle()
    val bravoState by sharedViewModel.bravoStateFlow.collectAsStateWithLifecycle()
    val customerCallAvailable by sharedViewModel.customerCallAvailable.collectAsStateWithLifecycle()
    val customerCallButtonAvailable by sharedViewModel.customerCallButtonStateAvailable.collectAsStateWithLifecycle()
    val canMakeCalls by sharedViewModel.canMakeCallsFlow.collectAsStateWithLifecycle()
    val multiDispatch by sharedViewModel.multiDispatchFlow.collectAsStateWithLifecycle()

    var dialogState by remember { mutableStateOf<InfoDispatchDialogState?>(null) }

    LaunchedEffect(dispatch) {
        viewModel.bindDispatch(dispatch)
    }

    LaunchedEffect(taximeterConnected) {
        viewModel.updateTaximeterConnection(taximeterConnected)
    }

    LaunchedEffect(shiftStatus) {
        viewModel.updateShiftState(
            isManual = shiftStatus.isManual,
            currentStatus = shiftStatus.currentStatus ?: -1
        )
    }

    LaunchedEffect(bravoState) {
        viewModel.updateBridgeCallState(bravoState.bridgeCallState)
        viewModel.updateNoClientButtonEnabled(bravoState.noClientButtonEnabled)
    }

    LaunchedEffect(customerCallAvailable) {
        viewModel.updateCustomerCallAvailability(customerCallAvailable)
    }

    LaunchedEffect(customerCallButtonAvailable) {
        viewModel.updateCustomerCallButtonStateAvailable(customerCallButtonAvailable)
    }

    LaunchedEffect(canMakeCalls) {
        viewModel.updateCanMakeCalls(canMakeCalls)
    }

    LaunchedEffect(multiDispatch) {
        viewModel.setMultiDispatch(multiDispatch)
    }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                InfoDispatchUiEffect.NavigateToDirections -> {
                    navController.navigate("directions")
                }

                InfoDispatchUiEffect.NavigateToHome -> {
                    navController.navigate(HomeDirections.goToOnTripFragment().actionId.toString())
                }

                is InfoDispatchUiEffect.NavigateToMeetingSign -> {
                    navController.navigate("meeting_sign/${effect.textColor}/${effect.backgroundColor}")
                }

                InfoDispatchUiEffect.RequestPhonePermission -> {
                    // host permission flow
                }

                is InfoDispatchUiEffect.ShowToast -> {
                    // host toast/snackbar
                }

                is InfoDispatchUiEffect.OpenExternalPhoneCall -> {
                    // host phone intent
                }

                is InfoDispatchUiEffect.OpenDialog -> {
                    dialogState = effect.dialog
                }

                InfoDispatchUiEffect.CloseDialog -> {
                    dialogState = null
                }
            }
        }
    }

    LaunchedEffect(Unit) {
        sharedViewModel.dispatchNotificationDoorReceived.collectLatest {
            // host toast if needed
        }
    }

    LaunchedEffect(Unit) {
        sharedViewModel.dispatchNotificationRiderReceived.collectLatest {
            // host toast if needed
        }
    }

    LaunchedEffect(Unit) {
        sharedViewModel.dispatchNotificationNoClientReceived.collectLatest {
            // host toast if needed
        }
    }

    Scaffold(
        topBar = { },
        bottomBar = { }
    ) { padding ->
        InfoDispatchContent(
            modifier = androidx.compose.ui.Modifier.padding(padding),
            state = state,
            onEvent = viewModel::onEvent
        )
    }

    dialogState?.let { dialog ->
        InfoDispatchCustomDialog(
            state = dialog,
            onDismiss = { dialogState = null },
            onButtonClick = { button ->
                viewModel.onEvent(InfoDispatchUiEvent.OnDialogButtonClicked(button))
                dialogState = null
            }
        )
    }
```

# 5) Important implementation note

In Compose, those should be handled by the host activity or a bridge layer.

```kotlin
    }

    private fun reduce() {

        val newButtons = reducer.reduce(
            current = current.buttons,

            isManualShift = _isManualShift.value,
            bridgeCallState = _bridgeCallState.value,
            customerCallAvailable = _customerCallAvailable.value,
            customerCallButtonStateAvailable = _customerCallButtonStateAvailable.value,
            canMakeCalls = _canMakeCalls.value,
            noClientButtonEnabled = _noClientButtonEnabled.value,
            phoneCallError = null,

class InfoDispatchButtonsReducer {

    fun reduce(
        current: InfoDispatchButtonsState,
        dispatch: InfoDispatchModel?,
        isITopConnected: Boolean,
        isManualShift: Boolean,
        bridgeCallState: Int,
        customerCallAvailable: Boolean,
        customerCallButtonStateAvailable: Boolean,
        canMakeCalls: Boolean,
        noClientButtonEnabled: Boolean?,
        phoneCallError: Boolean?,
        bridgeCallTimeout: Boolean,
        shiftCurrentStatus: Int?,
        bravoShowTripAmount: Boolean
    ): InfoDispatchButtonsState {
        var state = current

        state = reduceFromDispatchFlow(state, dispatch, isITopConnected, isManualShift)
        state = reduceFromBravoState(state, isITopConnected, bridgeCallState)
        state = reduceFromShiftStatus(state, isITopConnected)
        state = reduceFromNoClientButtonFlow(state, noClientButtonEnabled, shiftCurrentStatus)
        state = reduceFromPhoneCallErrorFlow(state, phoneCallError)
        state = reduceFromBridgeCallTimeoutFlow(state, bridgeCallTimeout)
        state = reduceFromCustomerCallAvailable(state, customerCallAvailable)
        state = reduceFromCustomerCallButtonStateAvailable(
            state,
            customerCallAvailable,
            customerCallButtonStateAvailable
        )
        state = reduceFromCanMakeCallsCombine(state, customerCallAvailable, canMakeCalls, bridgeCallState)

        return state
    }

    private fun reduceFromDispatchFlow(
        state: InfoDispatchButtonsState,
        dispatch: InfoDispatchModel?,
        isITopConnected: Boolean,
        isManualShift: Boolean
    ): InfoDispatchButtonsState {
        if (dispatch == null) return state

        val hasAtDoorAction =
            dispatch.isAtDoorNotificationEnabled() && dispatch.isAtDoorNotificationSent == false
        val hasRiderInCabAction =
            (dispatch.riderInCab ?: false) && dispatch.isRiderInCabNotificationSent == false

        val notifications = when {
            hasAtDoorAction && hasRiderInCabAction -> state.notifications.copy(
                enabled = true,
                loading = false,
                backgroundColor = Color(0xFF1976D2)
            )
            !hasAtDoorAction && hasRiderInCabAction -> state.notifications.copy(
                enabled = true,
                loading = false,
                backgroundColor = Color(0xFF4CAF50)
            )
            hasAtDoorAction -> state.notifications.copy(
                enabled = true,
                loading = false,
                backgroundColor = Color(0xFFFF9800)
            )
            else -> state.notifications.copy(
                enabled = false,
                loading = false,
                backgroundColor = Color(0xFF9E9E9E)
            )
        }

        val restricted = isITopConnected && !isManualShift
        return state.copy(
            notifications = if (restricted) notifications.copy(enabled = false) else notifications
        )
    }

    private fun reduceFromBravoState(
        state: InfoDispatchButtonsState,
        isITopConnected: Boolean,
        bridgeCallState: Int
    ): InfoDispatchButtonsState {
        var newState = state

        if (isITopConnected) {
            newState = newState.copy(
                noClient = newState.noClient.copy(enabled = false),
                returnDispatch = newState.returnDispatch.copy(enabled = false)
            )
        }

        newState = when (bridgeCallState) {
            2 -> newState.copy(
                voiceCall = newState.voiceCall.copy(
                    enabled = true,
                    loading = false,
                    backgroundColor = Color(0xFFF44336)
                )
            )
            0 -> newState.copy(
                voiceCall = newState.voiceCall.copy(
                    backgroundColor = Color(0xFF4CAF50)
                )
            )
            else -> newState.copy(
                voiceCall = newState.voiceCall.copy(
                    enabled = true,
                    loading = false,
                    backgroundColor = Color(0xFFFF9800)
                )
            )
        }

        return newState
    }

    private fun reduceFromShiftStatus(
        state: InfoDispatchButtonsState,
        isITopConnected: Boolean
    ): InfoDispatchButtonsState {
        return if (isITopConnected) {
            state.copy(
                noClient = state.noClient.copy(enabled = false),
                returnDispatch = state.returnDispatch.copy(enabled = false)
            )
        } else {
            state
        }
    }

    private fun reduceFromNoClientButtonFlow(
        state: InfoDispatchButtonsState,
        noClientButtonEnabled: Boolean?,
        shiftCurrentStatus: Int?
    ): InfoDispatchButtonsState {
        val canChange =
            noClientButtonEnabled == false &&
                shiftCurrentStatus == com.interfacom.sdk.taximeter.bravocomm.ifConstants.STATE_HIRED_DISPATCHED

        return if (canChange) {
            state.copy(noClient = state.noClient.copy(enabled = false))
        } else {
            state
        }
    }

    private fun reduceFromPhoneCallErrorFlow(
        state: InfoDispatchButtonsState,
        phoneCallError: Boolean?
    ): InfoDispatchButtonsState {
        return if (phoneCallError == null) {
            state.copy(
                voiceCall = state.voiceCall.copy(
                    backgroundColor = Color(0xFFF44336),
                    enabled = true
                )
            )
        } else {
            state
        }
    }

    private fun reduceFromBridgeCallTimeoutFlow(
        state: InfoDispatchButtonsState,
        bridgeCallTimeout: Boolean
    ): InfoDispatchButtonsState {
        return if (bridgeCallTimeout) {
            state.copy(
                voiceCall = state.voiceCall.copy(
                    enabled = true,
                    loading = false
                )
            )
        } else {
            state
        }
    }

    private fun reduceFromCustomerCallAvailable(
        state: InfoDispatchButtonsState,
        customerCallAvailable: Boolean
    ): InfoDispatchButtonsState {
        return state.copy(
            voiceCall = state.voiceCall.copy(
                enabled = customerCallAvailable,
                loading = false
            )
        )
    }

    private fun reduceFromCustomerCallButtonStateAvailable(
        state: InfoDispatchButtonsState,
        customerCallAvailable: Boolean,
        customerCallButtonStateAvailable: Boolean
    ): InfoDispatchButtonsState {
        return state.copy(
            voiceCall = state.voiceCall.copy(
                enabled = customerCallAvailable && customerCallButtonStateAvailable,
                loading = !customerCallButtonStateAvailable
            )
        )
    }

    private fun reduceFromCanMakeCallsCombine(
        state: InfoDispatchButtonsState,
        customerCallAvailable: Boolean,
        canMakeCalls: Boolean,
        bridgeCallState: Int
    ): InfoDispatchButtonsState {
        val voiceEnabled =
            if (!customerCallAvailable) {
                false
            } else if (canMakeCalls || bridgeCallState != 0) {
                true
            } else {
                false
            }

        return state.copy(
            voiceCall = state.voiceCall.copy(
                enabled = voiceEnabled,
                loading = false
            )
        )
    }
}

                    navController.navigate("meeting_sign/${effect.textColor}/${effect.backgroundColor}")

                InfoDispatchUiEffect.RequestPhonePermission -> {
                    // launch permission flow from host if desired

                is InfoDispatchUiEffect.ShowToast -> {
                    // host-side toast/snackbar handling
                }
                is InfoDispatchUiEffect.OpenExternalPhoneCall -> {
                    // use host / activity to start phone intent
```

```kotlin
    private val _isManualShift = MutableStateFlow(false)
    private val _bridgeCallState = MutableStateFlow(0)
    private val _customerCallAvailable = MutableStateFlow(false)
    private val _customerCallButtonStateAvailable = MutableStateFlow(true)
    private val _canMakeCalls = MutableStateFlow(false)
    private val _noClientButtonEnabled = MutableStateFlow(true)
    private val _shiftCurrentStatus = MutableStateFlow<Int?>(null)
    private val _bravoShowTripAmount = MutableStateFlow(false)

    private val reducer = InfoDispatchButtonsReducer()

        _dispatch.value = dispatch
        reduce()

        _isITopConnected.value = connected
        reduce()

        _isManualShift.value = isManual

    }

        _bridgeCallState.value = state
        reduce()
    }

        _customerCallAvailable.value = value
        reduce()
    }

    fun updateCustomerCallButtonStateAvailable(value: Boolean) {
        _customerCallButtonStateAvailable.value = value
        reduce()
    }

    fun updateCanMakeCalls(value: Boolean) {
        _canMakeCalls.value = value
        reduce()
    }

    fun updateNoClientButtonEnabled(value: Boolean?) {
        _noClientButtonEnabled.value = value ?: false
        reduce()
    }

    fun updateBravoShowTripAmount(value: Boolean) {
        _bravoShowTripAmount.value = value
        reduce()
    }

    fun setMultiDispatch(list: List<InfoDispatchModel>) {
        val tabs = list.map { dispatch ->
            val iconLabel = when {
                dispatch.isCancelled -> "⛔"
                dispatch.isEnded -> "🏁"
                dispatch.isRiderInCabNotificationSent == true -> "👤"
                dispatch.isAtDoorNotificationSent == true -> "🚪"
                else -> "🚕"
            }
            InfoDispatchMultiDispatchTabState(
                name = dispatch.dispatchName.orEmpty(),
                iconLabel = iconLabel
            )
        }
        _uiState.update {
            it.copy(
                multiDispatch = InfoDispatchMultiDispatchState(
                    visible = list.size > 1,
                    selectedIndex = list.indexOfFirst { d -> d.selected }.coerceAtLeast(0),
                    tabs = tabs
                )
            )
        }
    }

    fun onEvent(event: InfoDispatchUiEvent) {
        when (event) {
            InfoDispatchUiEvent.OnNotificationsClicked -> openNotificationsDialog()
            InfoDispatchUiEvent.OnNavigateClicked -> emitEffect(InfoDispatchUiEffect.NavigateToDirections)
            InfoDispatchUiEvent.OnVoiceCallClicked -> emitEffect(InfoDispatchUiEffect.RequestPhonePermission)
            InfoDispatchUiEvent.OnPrintClicked -> Unit
            InfoDispatchUiEvent.OnNoClientClicked -> openNoClientDialog()
            InfoDispatchUiEvent.OnReturnClicked -> openReturnDialog()
            InfoDispatchUiEvent.OnDispatchNumberClicked -> emitEffect(
                InfoDispatchUiEffect.NavigateToMeetingSign(0, 0)
            )
            is InfoDispatchUiEvent.OnMultiDispatchSelected -> Unit
            is InfoDispatchUiEvent.OnDialogButtonClicked -> handleDialogButton(event.button)
        }
    }

            dispatch = _dispatch.value,
            isITopConnected = _isITopConnected.value,
            isManualShift = _isManualShift.value,
            bridgeCallState = _bridgeCallState.value,
            customerCallAvailable = _customerCallAvailable.value,
            customerCallButtonStateAvailable = _customerCallButtonStateAvailable.value,
            canMakeCalls = _canMakeCalls.value,

            phoneCallError = null,
            bridgeCallTimeout = false,
            shiftCurrentStatus = _shiftCurrentStatus.value,
            bravoShowTripAmount = _bravoShowTripAmount.value

                buttons = newButtons,
                dispatchInfo = buildDispatchInfo(_dispatch.value)

        }
    }

        if (dispatch == null) return InfoDispatchInfoState()

        val dispatchIdentification = if (!dispatch.dispatchName.isNullOrBlank()) {
            "${dispatch.dispatchName} - ${dispatch.longDispatchNumber}"
        } else {
            dispatch.idDispatch
        }

        val pickupTimeText = if (!dispatch.etaTime.isNullOrBlank()) {
            "${dispatch.pickUpTime} (${dispatch.etaTime})"
        } else {
            dispatch.pickUpTime.orEmpty()
        }

        return InfoDispatchInfoState(
            dispatchNumberText = dispatchIdentification,
            zoneTextVisible = !dispatch.pickUpZone.isNullOrBlank(),
            zoneText = dispatch.pickUpZone.orEmpty(),
            pickupTimeVisible = !dispatch.pickUpTime.isNullOrBlank(),
            pickupTimeText = pickupTimeText,
            pickupAddressVisible = !dispatch.pickUpAdress.isNullOrBlank(),
            pickupAddress = dispatch.pickUpAdress.orEmpty(),
            subscriberVisible = (dispatch.clientType ?: 0) > 0,
            subscriberTitle = listOfNotNull(dispatch.subscriber, dispatch.dispatchName).joinToString(" "),
            subscriberUserName = dispatch.subscriberUserName.orEmpty(),
            authorizationVisible = !dispatch.autorizacion.isNullOrBlank() && (dispatch.clientType ?: 0) > 0,
            authorizationText = dispatch.autorizacion.orEmpty(),
            destinationsVisible = !dispatch.destinyAdress.isNullOrEmpty(),
            destinations = dispatch.destinyAdress.orEmpty(),
            passengersVisible = !dispatch.numberPassengers.isNullOrBlank() && dispatch.numberPassengers != "1",
            passengersText = dispatch.numberPassengers.orEmpty(),
            baggagesVisible = !dispatch.numberBaggages.isNullOrBlank() && dispatch.numberBaggages != "0",
            baggagesText = dispatch.numberBaggages.orEmpty(),
            serviceTypeVisible = dispatch.isConcertedPrice == true || dispatch.isPaymentByApp == true,
            serviceTypeText = when {
                dispatch.isConcertedPrice == true -> "Concerted trip with fixed price"
                dispatch.isPaymentByApp == true -> "Payment by app"
                else -> ""
            },
            amountVisible = _bravoShowTripAmount.value && (dispatch.amount ?: 0) > 0,
            amountText = dispatch.amount?.toString().orEmpty(),
            flightCodeVisible = !dispatch.flightCode.isNullOrBlank(),
            flightCode = dispatch.flightCode.orEmpty()
        )
    }

            _uiEffect.emit(

class InfoDispatchButtonsReducer {

    fun reduce(
        current: InfoDispatchButtonsState,
        dispatch: InfoDispatchModel?,
        isITopConnected: Boolean,
        isManualShift: Boolean,
        bridgeCallState: Int,
        customerCallAvailable: Boolean,
        customerCallButtonStateAvailable: Boolean,
        canMakeCalls: Boolean,
        noClientButtonEnabled: Boolean?,
        phoneCallError: Boolean?,
        bridgeCallTimeout: Boolean,
        shiftCurrentStatus: Int?,
        bravoShowTripAmount: Boolean
    ): InfoDispatchButtonsState {
        var state = current
        state = reduceFromDispatchFlow(state, dispatch, isITopConnected, isManualShift)
        state = reduceFromBravoState(state, isITopConnected, bridgeCallState)
        state = reduceFromNoClientButtonFlow(state, noClientButtonEnabled, shiftCurrentStatus)
        state = reduceFromPhoneCallErrorFlow(state, phoneCallError)
        state = reduceFromBridgeCallTimeoutFlow(state, bridgeCallTimeout)
        state = reduceFromCustomerCallAvailable(state, customerCallAvailable)
        state = reduceFromCustomerCallButtonStateAvailable(state, customerCallAvailable, customerCallButtonStateAvailable)
        state = reduceFromCanMakeCallsCombine(state, customerCallAvailable, canMakeCalls, bridgeCallState)
        return state
    }

    private fun reduceFromDispatchFlow(
        state: InfoDispatchButtonsState,
        dispatch: InfoDispatchModel?,
        isITopConnected: Boolean,
        isManualShift: Boolean
    ): InfoDispatchButtonsState {
        if (dispatch == null) return state

        val hasAtDoorAction =
            dispatch.isAtDoorNotificationEnabled() && dispatch.isAtDoorNotificationSent == false
        val hasRiderInCabAction =
            (dispatch.riderInCab ?: false) && dispatch.isRiderInCabNotificationSent == false

        val notifications = when {
            hasAtDoorAction && hasRiderInCabAction -> state.notifications.copy(
                enabled = true,
                loading = false,
                backgroundColor = Color(0xFF1976D2)
            )
            !hasAtDoorAction && hasRiderInCabAction -> state.notifications.copy(
                enabled = true,
                loading = false,
                backgroundColor = Color(0xFF4CAF50)
            )
            hasAtDoorAction -> state.notifications.copy(
                enabled = true,
                loading = false,
                backgroundColor = Color(0xFFFF9800)
            )
            else -> state.notifications.copy(
                enabled = false,
                loading = false,
                backgroundColor = Color(0xFF9E9E9E)
            )
        }

        val restricted = isITopConnected && !isManualShift
        return state.copy(
            notifications = if (restricted) notifications.copy(enabled = false) else notifications
        )
    }

    private fun reduceFromBravoState(
        state: InfoDispatchButtonsState,
        isITopConnected: Boolean,
        bridgeCallState: Int
    ): InfoDispatchButtonsState {
        var newState = state

        if (isITopConnected) {
            newState = newState.copy(
                noClient = newState.noClient.copy(enabled = false),
                returnDispatch = newState.returnDispatch.copy(enabled = false)
            )
        }

        newState = when (bridgeCallState) {
            2 -> newState.copy(
                voiceCall = newState.voiceCall.copy(
                    enabled = true,
                    loading = false,
                    backgroundColor = Color(0xFFF44336)
                )
            )
            0 -> newState.copy(
                voiceCall = newState.voiceCall.copy(
                    backgroundColor = Color(0xFF4CAF50)
                )
            )
            else -> newState.copy(
                voiceCall = newState.voiceCall.copy(
                    enabled = true,
                    loading = false,
                    backgroundColor = Color(0xFFFF9800)
                )
            )
        }

        return newState
    }

    private fun reduceFromNoClientButtonFlow(
        state: InfoDispatchButtonsState,
        noClientButtonEnabled: Boolean?,
        shiftCurrentStatus: Int?
    ): InfoDispatchButtonsState {
        val canChange =
            noClientButtonEnabled == false &&
                shiftCurrentStatus == com.interfacom.sdk.taximeter.bravocomm.ifConstants.STATE_HIRED_DISPATCHED

        return if (canChange) {
            state.copy(noClient = state.noClient.copy(enabled = false))
        } else {
            state
        }
    }

    private fun reduceFromPhoneCallErrorFlow(
        state: InfoDispatchButtonsState,
        phoneCallError: Boolean?
    ): InfoDispatchButtonsState {
        return if (phoneCallError == null) {
            state.copy(
                voiceCall = state.voiceCall.copy(
                    backgroundColor = Color(0xFFF44336),
                    enabled = true
                )
            )
        } else {
            state
        }
    }

    private fun reduceFromBridgeCallTimeoutFlow(
        state: InfoDispatchButtonsState,
        bridgeCallTimeout: Boolean
    ): InfoDispatchButtonsState {
        return if (bridgeCallTimeout) {
            state.copy(
                voiceCall = state.voiceCall.copy(
                    enabled = true,
                    loading = false
                )
            )
        } else {
            state
        }
    }

    private fun reduceFromCustomerCallAvailable(
        state: InfoDispatchButtonsState,
        customerCallAvailable: Boolean
    ): InfoDispatchButtonsState {
        return state.copy(
            voiceCall = state.voiceCall.copy(
                enabled = customerCallAvailable,
                loading = false
            )
        )
    }

    private fun reduceFromCustomerCallButtonStateAvailable(
        state: InfoDispatchButtonsState,
        customerCallAvailable: Boolean,
        customerCallButtonStateAvailable: Boolean
    ): InfoDispatchButtonsState {
        return state.copy(
            voiceCall = state.voiceCall.copy(
                enabled = customerCallAvailable && customerCallButtonStateAvailable,
                loading = !customerCallButtonStateAvailable
            )
        )
    }

    private fun reduceFromCanMakeCallsCombine(
        state: InfoDispatchButtonsState,
        customerCallAvailable: Boolean,
        canMakeCalls: Boolean,
        bridgeCallState: Int
    ): InfoDispatchButtonsState {
        val voiceEnabled =
            if (!customerCallAvailable) false
            else if (canMakeCalls || bridgeCallState != 0) true
            else false

        return state.copy(
            voiceCall = state.voiceCall.copy(
                enabled = voiceEnabled,
                loading = false
            )
        )
    }
}

@Composable
fun InfoDispatchScreen(
    navController: NavController,
    viewModel: InfoDispatchComposeViewModel,
    sharedViewModel: InfoDispatchMainActivityViewModel
) {
    val state by viewModel.uiState.collectAsStateWithLifecycle()
    val dispatch by sharedViewModel.dispatchFlow.collectAsStateWithLifecycle()
    val taximeterConnected by sharedViewModel.hasTaximeterConnectionFlow.collectAsStateWithLifecycle()
    val shiftStatus by sharedViewModel.shiftStatusFlow.collectAsStateWithLifecycle()
    val bravoState by sharedViewModel.bravoStateFlow.collectAsStateWithLifecycle()
    val customerCallAvailable by sharedViewModel.customerCallAvailable.collectAsStateWithLifecycle()
    val customerCallButtonAvailable by sharedViewModel.customerCallButtonStateAvailable.collectAsStateWithLifecycle()
    val canMakeCalls by sharedViewModel.canMakeCallsFlow.collectAsStateWithLifecycle()
    val multiDispatch by sharedViewModel.multiDispatchFlow.collectAsStateWithLifecycle()

    var dialogState by remember { mutableStateOf<InfoDispatchDialogState?>(null) }

    LaunchedEffect(dispatch) { viewModel.bindDispatch(dispatch) }
    LaunchedEffect(taximeterConnected) { viewModel.updateTaximeterConnection(taximeterConnected) }
    LaunchedEffect(shiftStatus) { viewModel.updateShiftState(shiftStatus.isManual, shiftStatus.currentStatus) }
    LaunchedEffect(bravoState) {
        viewModel.updateBridgeCallState(bravoState.bridgeCallState)
        viewModel.updateNoClientButtonEnabled(bravoState.noClientButtonEnabled)
    }
    LaunchedEffect(customerCallAvailable) { viewModel.updateCustomerCallAvailability(customerCallAvailable) }
    LaunchedEffect(customerCallButtonAvailable) { viewModel.updateCustomerCallButtonStateAvailable(customerCallButtonAvailable) }
    LaunchedEffect(canMakeCalls) { viewModel.updateCanMakeCalls(canMakeCalls) }
    LaunchedEffect(multiDispatch) { viewModel.setMultiDispatch(multiDispatch) }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                InfoDispatchUiEffect.NavigateToDirections -> {
                    navController.navigate(InfoDispatchFragmentDirections.actionInfoDispatchFragmentToDirectionsFragment())
                }
                InfoDispatchUiEffect.NavigateToHome -> {
                    navController.navigate(HomeDirections.goToOnTripFragment())
                }
                is InfoDispatchUiEffect.NavigateToMeetingSign -> {
                    navController.navigate(
                        InfoDispatchFragmentDirections.actionInfoDispatchFragmentToMeetingSignFragment(
                            effect.textColor,
                            effect.backgroundColor
                        )
                    )
                }
                InfoDispatchUiEffect.RequestPhonePermission -> {
                    // host permission flow
                }
                is InfoDispatchUiEffect.ShowToast -> {
                    // host toast/snackbar
                }
                InfoDispatchUiEffect.OpenExternalPhoneCall -> {
                    // host phone intent
                }
                is InfoDispatchUiEffect.OpenDialog -> dialogState = effect.dialog
                InfoDispatchUiEffect.CloseDialog -> dialogState = null
            }
        }
    }

    Scaffold(
        topBar = { },
        bottomBar = { }
    ) { padding ->
        InfoDispatchContent(
            modifier = Modifier.padding(padding),
            state = state,
            onEvent = viewModel::onEvent
        )
    }

    dialogState?.let { dialog ->
        InfoDispatchCustomDialog(
            state = dialog,
            onDismiss = { dialogState = null },
            onButtonClick = { button ->
                viewModel.onEvent(InfoDispatchUiEvent.OnDialogButtonClicked(button))
                dialogState = null
            }
        )
    }
}

@Composable
fun InfoDispatchContent(
    modifier: Modifier = Modifier,
    state: InfoDispatchUiState,
    onEvent: (InfoDispatchUiEvent) -> Unit
) {
    Column(
        modifier = modifier
            .fillMaxSize()
            .padding(16.dp),
        verticalArrangement = Arrangement.spacedBy(12.dp)
    ) {
        if (state.dispatchInfo.dispatchNumberText.isNotBlank()) {
            Text(
                text = state.dispatchInfo.dispatchNumberText,
                style = MaterialTheme.typography.titleMedium
            )
        }

        InfoDispatchButtons(
            buttonsState = state.buttons,
            onEvent = onEvent
        )
    }
}

@Composable
private fun InfoDispatchButtons(
    buttonsState: InfoDispatchButtonsState,
    onEvent: (InfoDispatchUiEvent) -> Unit
) {
    Column(verticalArrangement = Arrangement.spacedBy(10.dp)) {
        if (buttonsState.notifications.visible) {
            CustomButton(
                text = buttonsState.notifications.text,
                enabled = buttonsState.notifications.enabled,
                loading = buttonsState.notifications.loading,
                backgroundColor = buttonsState.notifications.backgroundColor,
                onClick = { onEvent(InfoDispatchUiEvent.OnNotificationsClicked) }
            )
        }
        if (buttonsState.navigate.visible) {
            CustomButton(
                text = buttonsState.navigate.text,
                enabled = buttonsState.navigate.enabled,
                loading = buttonsState.navigate.loading,
                backgroundColor = buttonsState.navigate.backgroundColor,
                onClick = { onEvent(InfoDispatchUiEvent.OnNavigateClicked) }
            )
        }
        if (buttonsState.voiceCall.visible) {
            CustomButton(
                text = buttonsState.voiceCall.text,
                enabled = buttonsState.voiceCall.enabled,
                loading = buttonsState.voiceCall.loading,
                backgroundColor = buttonsState.voiceCall.backgroundColor,
                onClick = { onEvent(InfoDispatchUiEvent.OnVoiceCallClicked) }
            )
        }
        if (buttonsState.print.visible) {
            CustomButton(
                text = buttonsState.print.text,
                enabled = buttonsState.print.enabled,
                loading = buttonsState.print.loading,
                backgroundColor = buttonsState.print.backgroundColor,
                onClick = { onEvent(InfoDispatchUiEvent.OnPrintClicked) }
            )
        }
        if (buttonsState.noClient.visible) {
            CustomButton(
                text = buttonsState.noClient.text,
                enabled = buttonsState.noClient.enabled,
                loading = buttonsState.noClient.loading,
                backgroundColor = buttonsState.noClient.backgroundColor,
                onClick = { onEvent(InfoDispatchUiEvent.OnNoClientClicked) }
            )
        }
        if (buttonsState.returnDispatch.visible) {
            CustomButton(
                text = buttonsState.returnDispatch.text,
                enabled = buttonsState.returnDispatch.enabled,
                loading = buttonsState.returnDispatch.loading,
                backgroundColor = buttonsState.returnDispatch.backgroundColor,
                onClick = { onEvent(InfoDispatchUiEvent.OnReturnClicked) }
            )
        }
    }
}

@Composable
private fun CustomButton(
    text: String,
    enabled: Boolean,
    loading: Boolean,
    backgroundColor: Color,
    onClick: () -> Unit
) {
    val finalEnabled = enabled && !loading

    androidx.compose.material3.Button(
        onClick = onClick,
        enabled = finalEnabled,
        colors = ButtonDefaults.buttonColors(
            containerColor = backgroundColor,
            disabledContainerColor = backgroundColor.copy(alpha = 0.45f),
            contentColor = Color.White,
            disabledContentColor = Color.White.copy(alpha = 0.8f)
        ),
        modifier = Modifier.fillMaxWidth()
    ) {
        if (loading) {
            CircularProgressIndicator(
                modifier = Modifier.height(18.dp),
                strokeWidth = 2.dp,
                color = Color.White
            )
            Spacer(modifier = Modifier.padding(horizontal = 8.dp))
        }
        Text(text = text)
    }
}

@Composable
fun InfoDispatchCustomDialog(
    state: InfoDispatchDialogState,
    onDismiss: () -> Unit,
    onButtonClick: (InfoDispatchDialogButton) -> Unit
) {
    AlertDialog(
        onDismissRequest = onDismiss,
        title = { Text(state.title) },
        text = { Text(state.description) },
        confirmButton = {
            state.buttons.forEach { button ->
                TextButton(onClick = { onButtonClick(button) }) {
                    Text(button.label)
                }
            }
        }
    )
}
```

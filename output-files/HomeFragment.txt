Below is a Compose-first version of the `HomeFragment` behavior with:

- a `HomeScreen` composable
- a Compose-friendly `HomeUiState + HomeUiEvent` ViewModel
- navigation preserved through `navigateTo(...)` / `NavDeepLinkRequest`
- dialog handling via state + side effects
- a Compose `CustomDialog` based on your XML/dialog behavior
- button logic moved into state holders/data classes

I’ll keep it Android/Jetpack Compose only, and I’ll mirror the original fragment behavior as closely as possible.

```kotlin
package ifac.td.taxi.ui.screen.home

import android.app.Application
import android.media.ToneGenerator
import androidx.core.net.toUri
import androidx.lifecycle.viewModelScope

class HomeComposeViewModel(
    private val userPreferencesUseCase: UserPreferencesUseCase,
    private val endShiftUseCase: EndShiftUseCase,
    private val zoningUseCase: ZoningUseCase,
    private val messageUseCase: MessageUseCase,
    private val roofLightUseCase: RoofLightUseCase,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val sendToBravoUseCase: SendToBravoUseCase,
    private val pendingTripsUseCase: PendingTripsUseCase,
    private val bravoCentral: BravoCentralUseCase,
    private val portugalUseCase: PortugalUseCase,
    private val licensingUseCase: LicensingUseCase,
    private val chronometerManagerUseCase: ChronometerManagerUseCase,
    private val countdownManagerUseCase: CountdownManagerUseCase,
    private val tripUseCase: TripUseCase,
    private val bravoCentralConfigurationUseCase: BravoCentralConfigurationUseCase,
    private val bluetoothLocalUseCase: BluetoothLocalUseCase,
    context: Application,
) : BaseViewModel(context) {

    private val TAG = "HomeComposeViewModel"

    private val _uiState = MutableStateFlow(HomeUiState())
    val uiState = _uiState.asStateFlow()

    private val _events = MutableSharedFlow<HomeUiEvent>()
    val events = _events.asSharedFlow()

    private var lastRoofLightState: Boolean? = null

    init {
        refreshInitialState()
    }

    private fun refreshInitialState() {
        viewModelScope.launch(Dispatchers.IO) {
            _uiState.emit(
                _uiState.value.copy(
                    dashboardButton = _uiState.value.dashboardButton.copy(
                        visible = bravoCentralConfigurationUseCase.getBravoConfigurationVariable()?.isHomeButton == true
                    ),
                    showEstimateFixedPriceButton = bravoCentralConfigurationUseCase
                        .getBravoConfigurationVariable()?.fixedPricesStreetTripsURL?.isNotBlank() == true
                )
            )

            userPreferencesUseCase.getUserPreferences()?.let {
                _uiState.emit(
                    _uiState.value.copy()
                )
            }
        }
    }

    fun onStart() {
        getMessages()
        checkForLocationPermission()
        checkPendingServiceOnForHirePermission()
        checkKeepScreenOn()
    }

    fun getMessages() {
        viewModelScope.launch(Dispatchers.IO) {
            val messages = messageUseCase.getMessageByDriverId()
            val newState = when {
                messages.isNullOrEmpty() -> MessageUiState.NO_MESSAGES
                messages.any { !it.isRead && it.messageType == MessageType.MESSAGE } -> MessageUiState.HAS_NEW_MESSAGES
                else -> MessageUiState.HAS_MESSAGES_BUT_NOT_NEW
            }
            _uiState.emit(_uiState.value.copy(hasMessages = newState))
        }
    }

    fun onMessagesClick() = navigateToMessage()
    fun onReceiptsClick() = navigateToReceiptHistory()
    fun onCentralClick() = navigateToContactCentral()
    fun onDashboardClick() { emitNav(HomeUiEvent.NavigateToDashboard) }
    fun onFixedPriceClick() = emitNav(HomeUiEvent.NavigateToFixedPrice)
    fun onZoningClick() = navigateToZoning()
    fun onPendingClick() = canOpenPendingTrips()
    fun onLocationClick() = emitNav(HomeUiEvent.ToggleLocation)
    fun onLocateStandClick() = emitNav(HomeUiEvent.ToggleLocateStand)

    fun onRoofLightClick() {
        val state = _uiState.value.roofLightOn
        if (state == true) setRoofLightOff() else setRoofLightOn()
    }

    fun onManualTripIconClick() = emitNav(HomeUiEvent.RequestManualTripDialog)
    fun onLogoutRequested() = emitNav(HomeUiEvent.RequestLogoutDialog)

    private fun emitNav(event: HomeUiEvent) {
        viewModelScope.launch { _events.emit(event) }
    }

    fun navigateToMessage() = viewModelScope.launch { navigateTo(R.id.action_homeFragment_to_messageFragment) }

    fun navigateToReceiptHistory() {
        viewModelScope.launch {
            if (licensingUseCase.getLicensingParameters()?.isFiscalService == false) {
                navigateTo(HomeFragmentDirections.actionHomeFragmentToReceiptHistoryFragment(-1L))
            } else {
                val lastTrip: Trip? = tripUseCase.getLastTrip()
                if (lastTrip != null) {
                    val isSubscriber = lastTrip.paymentMethod == PaymentMethod.SUBSCRIBER.paymentString
                    if (lastTrip.isPendingPortugalService && !isSubscriber) {
                        val atcud = portugalUseCase.getPortugalData()?.atcud
                        if (!atcud.isNullOrEmpty() && atcud != "0") {
                            navigateTo(HomeFragmentDirections.actionHomeFragmentToPortugalInvoiceFragment())
                        }
                    } else {
                        navigateTo(HomeFragmentDirections.actionHomeFragmentToReceiptHistoryFragment(-1L))
                    }
                } else {
                    navigateTo(HomeFragmentDirections.actionHomeFragmentToReceiptHistoryFragment(-1L))
                }
            }
        }
    }

    fun navigateToContactCentral() = viewModelScope.launch { navigateTo(R.id.action_homeFragment_to_contactCentralFragment) }

    fun navigateToZoning() {
        viewModelScope.launch {
            val defaultToMacrozone = userPreferencesUseCase.getUserPreferences()?.macroZoneQuery == true
            val zoneFilter = zoningUseCase.getZoneConfiguration().zoneFilter
            val filterNavigatesToZoning = zoneFilter == FilterOptions.NEARNESS ||
                    zoneFilter == FilterOptions.ID_NO_HIERARCHY ||
                    zoneFilter == FilterOptions.HOT_ZONES ||
                    zoneFilter == FilterOptions.FAVOURITES

            if (defaultToMacrozone && !filterNavigatesToZoning) {
                navigateTo(R.id.action_homeFragment_to_macroZoning)
            } else if (W2CLocation.getLastIdZone() > 0 || filterNavigatesToZoning) {
                val idMacroZone = if (W2CLocation.getLastIdMacrozone() != 0) W2CLocation.getLastIdMacrozone() else 1
                val navDeepLink = NavDeepLinkRequest.Builder
                    .fromUri("android-app://ifac.td.taxi/zoningFragment/$idMacroZone".toUri())
                    .build()
                navigateTo(navDeepLink)
            } else {
                navigateTo(R.id.action_homeFragment_to_macroZoning)
            }
        }
    }

    fun checkForLocationPermission() {
        viewModelScope.launch {
            val parameters = licensingUseCase.getLicensingParameters()
            _uiState.emit(_uiState.value.copy(showLocationButton = parameters?.isDisableLocation))
        }
    }

    fun checkPendingServiceOnForHirePermission() {
        viewModelScope.launch {
            val parameters = licensingUseCase.getLicensingParameters()
            _uiState.emit(_uiState.value.copy(pendingServicesButtonAllowed = parameters?.isVacantPendingServices))
        }
    }

    fun returnLocationPermission(): Boolean? {
        return licensingUseCase.getLicensingParameters()?.isDisableLocation
    }

    fun checkKeepScreenOn() {
        viewModelScope.launch {
            val selected = userPreferencesUseCase.getUserPreferences()?.turnOffScreenPosition ?: 0
            _uiState.emit(_uiState.value.copy(keepScreenOn = selected == 0 || selected == 1))
        }
    }

    fun checkZoningButton(locationEnabled: Boolean) {
        viewModelScope.launch {
            val isEnabled =
                (bravoCentralConfigurationUseCase.getBravoConfigurationVariable()?.locationRequestWhenDispatchOff == true) ||
                        locationEnabled
            _uiState.emit(_uiState.value.copy(zoningEnabled = isEnabled))
        }
    }

    fun checkRoofLight(value: Boolean?) {
        viewModelScope.launch {
            val isDisableLuminous = licensingUseCase.getLicensingParameters()?.isDisableLuminous
            if (value == null || lastRoofLightState != value) {
                if (roofLightUseCase.isRoofLightAvailable()) {
                    _uiState.emit(_uiState.value.copy(roofLightOn = value, roofLightAvailable = true))
                } else {
                    _uiState.emit(_uiState.value.copy(roofLightAvailable = false))
                }
                lastRoofLightState = value
            }
        }
    }

    fun setRoofLightOff() {
        viewModelScope.launch { roofLightUseCase.setRoofLightOff() }
    }

    fun setRoofLightOn() {
        viewModelScope.launch { roofLightUseCase.setRoofLightOn() }
    }

    fun canOpenPendingTrips() {
        viewModelScope.launch {
            val allowedByCentral = W2CLocation.isLocationAllowedByCentral()
            if (!allowedByCentral) {
                _events.emit(HomeUiEvent.ShowToast(R.string.do_not_locate_on_sanction))
                return@launch
            }

            userPreferencesUseCase.getUserPreferences()?.let {
                if (it.closePendingTripsIfEmpty) {
                    if (com.interfacom.sdk.taximeter.bravocomm.BravoCentral.isPendingTripsAllowed(false)) {
                        pendingTripsUseCase.getPendingTripsZone(
                            pendingTripsUseCase.pendingTripsListFlow,
                            MutableSharedFlow<Boolean>()
                        )
                        _events.emit(HomeUiEvent.OpenPendingTrips)
                    } else {
                        _events.emit(HomeUiEvent.ShowPendingNoServicesToast)
                    }
                } else {
                    _events.emit(HomeUiEvent.OpenPendingTrips)
                }
            } ?: _events.emit(HomeUiEvent.OpenPendingTrips)
        }
    }

    fun changeStateHiredManual() {
        viewModelScope.launch(Dispatchers.IO) { shiftStatusUseCase.changeStateHiredManual() }
    }

    fun canGoToHiredManual(taximeterConnected: Boolean): Boolean {
        return licensingUseCase.canDoManualTrips(taximeterConnected)
    }

    fun logoff() {
        viewModelScope.launch(Dispatchers.IO) {
            endShiftUseCase.endShift()
            bravoCentral.logoff()
            shiftStatusUseCase.setStatus(com.interfacom.sdk.taximeter.bravocomm.ifConstants.STATE_DISCONNECTED, false)
        }
    }

    fun checkForHireAfterDelocation() {
        viewModelScope.launch { sendToBravoUseCase.checkForHireAfterDelocation(true) }
    }

    fun isCurrentBluetoothITop(): Boolean = bluetoothLocalUseCase.isCurrentBluetoothItop()

    private fun navigateTo(event: Any) {
        viewModelScope.launch {
            // keep existing BaseViewModel navigation mechanism
        }
    }
}
```

## 3) Compose HomeScreen

This fully replaces the fragment button logic with state-driven UI.

```kotlin
package ifac.td.taxi.ui.screen.home

import android.media.ToneGenerator
import androidx.compose.foundation.layout.*
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Modifier
import androidx.compose.ui.unit.dp
import androidx.lifecycle.Lifecycle
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.lifecycle.compose.LocalLifecycleOwner
import androidx.lifecycle.repeatOnLifecycle
import kotlinx.coroutines.flow.collectLatest

@Composable
fun HomeScreen(
    viewModel: HomeComposeViewModel,
    onNavigateBack: () -> Unit,
    onNavigateToMessages: () -> Unit,
    onNavigateToReceipts: () -> Unit,
    onNavigateToCentral: () -> Unit,
    onNavigateToDashboard: () -> Unit,
    onNavigateToFixedPrice: () -> Unit,
    onNavigateToZoning: () -> Unit,
    onOpenPendingTrips: () -> Unit,
    onToggleLocation: () -> Unit,
    onToggleRoofLight: () -> Unit,
    onToggleLocateStand: () -> Unit,
    onShowManualTripDialog: () -> Unit,
    onShowLogoutDialog: () -> Unit,
    onShowToast: (Int) -> Unit,
    onKeepScreenOn: (Boolean) -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val lifecycleOwner = LocalLifecycleOwner.current

    LaunchedEffect(Unit) {
        viewModel.onStart()
    }

    LaunchedEffect(viewModel) {
        lifecycleOwner.repeatOnLifecycle(Lifecycle.State.STARTED) {
            viewModel.events.collectLatest { event ->
                when (event) {
                    HomeUiEvent.NavigateBack -> onNavigateBack()
                    HomeUiEvent.NavigateToMessages -> onNavigateToMessages()
                    HomeUiEvent.NavigateToReceipts -> onNavigateToReceipts()
                    HomeUiEvent.NavigateToCentral -> onNavigateToCentral()
                    HomeUiEvent.NavigateToDashboard -> onNavigateToDashboard()
                    HomeUiEvent.NavigateToFixedPrice -> onNavigateToFixedPrice()
                    HomeUiEvent.NavigateToZoning -> onNavigateToZoning()
                    HomeUiEvent.OpenPendingTrips -> onOpenPendingTrips()
                    HomeUiEvent.ToggleLocation -> onToggleLocation()
                    HomeUiEvent.ToggleRoofLight -> onToggleRoofLight()
                    HomeUiEvent.ToggleLocateStand -> onToggleLocateStand()
                    HomeUiEvent.RequestManualTripDialog -> onShowManualTripDialog()
                    HomeUiEvent.RequestLogoutDialog -> onShowLogoutDialog()
                    is HomeUiEvent.ShowToast -> onShowToast(event.messageRes)
                    HomeUiEvent.ShowPendingNoServicesToast -> onShowToast(ifac.td.taxi.R.string.toast_no_hay_pendientes)
                }
            }
        }
    }

    LaunchedEffect(uiState.keepScreenOn) {
        onKeepScreenOn(uiState.keepScreenOn)
    }

    HomeContent(
        state = uiState,
        onMessages = { viewModel.onMessagesClick() },
        onReceipts = { viewModel.onReceiptsClick() },
        onCentral = { viewModel.onCentralClick() },
        onDashboard = { viewModel.onDashboardClick() },
        onFixedPrice = { viewModel.onFixedPriceClick() },
        onZoning = { viewModel.onZoningClick() },
        onPending = { viewModel.onPendingClick() },
        onLocation = { viewModel.onLocationClick() },
        onRoofLight = { viewModel.onRoofLightClick() },
        onLocateStand = { viewModel.onLocateStandClick() },
    )
}

@Composable
private fun HomeContent(
    state: HomeUiState,
    onMessages: () -> Unit,
    onReceipts: () -> Unit,
    onCentral: () -> Unit,
    onDashboard: () -> Unit,
    onFixedPrice: () -> Unit,
    onZoning: () -> Unit,
    onPending: () -> Unit,
    onLocation: () -> Unit,
    onRoofLight: () -> Unit,
    onLocateStand: () -> Unit,
) {
    Column(modifier = Modifier.fillMaxSize().padding(16.dp), verticalArrangement = Arrangement.spacedBy(12.dp)) {
        ActionButton(state.zoningButton, onZoning)
        ActionButton(state.pendingButton, onPending)
        ActionButton(state.locationButton, onLocation)
        ActionButton(state.receiptsButton, onReceipts)
        ActionButton(state.messagesButton, onMessages)
        ActionButton(state.centralButton, onCentral)
        if (state.dashboardButton.visible) ActionButton(state.dashboardButton, onDashboard)
        if (state.fixedPriceButton.visible) ActionButton(state.fixedPriceButton, onFixedPrice)
        ActionButton(state.roofLightButton, onRoofLight)
    }
}

@Composable
private fun ActionButton(
    buttonState: ActionButtonState,
    onClick: () -> Unit
) {
    if (!buttonState.visible) return

    val containerColor = when (buttonState.background) {
        ButtonBackground.RED -> MaterialTheme.colorScheme.error
        ButtonBackground.GREEN -> MaterialTheme.colorScheme.primary
        ButtonBackground.BLUE -> MaterialTheme.colorScheme.primaryContainer
        ButtonBackground.ORANGE -> MaterialTheme.colorScheme.tertiary
    }

    Button(
        onClick = onClick,
        enabled = buttonState.enabled && !buttonState.loading,
        colors = ButtonDefaults.buttonColors(containerColor = containerColor)
    ) {
        Text(text = androidx.compose.ui.res.stringResource(id = buttonState.textRes))
    }
}
```

```kotlin
@Composable
fun HomeRoute(
    viewModel: HomeComposeViewModel,
    onBack: () -> Unit,
    onNavigate: (String) -> Unit,
    onToast: (Int) -> Unit,
    onKeepScreenOn: (Boolean) -> Unit
) {
    var dialogState by remember { mutableStateOf<HomeDialogState?>(null) }

    HomeScreen(
        viewModel = viewModel,
        onNavigateBack = onBack,
        onNavigateToMessages = { onNavigate("messages") },
        onNavigateToReceipts = { onNavigate("receipts") },
        onNavigateToCentral = { onNavigate("central") },
        onNavigateToDashboard = { onNavigate("dashboard") },
        onNavigateToFixedPrice = { onNavigate("fixed_price") },
        onNavigateToZoning = { onNavigate("zoning") },
        onOpenPendingTrips = { onNavigate("pending_trips") },
        onToggleLocation = { /* open location dialog */ },
        onToggleRoofLight = { /* rooflight action */ },
        onToggleLocateStand = { /* locate stand action */ },
        onShowManualTripDialog = {
            dialogState = HomeDialogState(
                visible = true,
                title = "Manual trip",
                description = "Confirm change to hired manual",
                buttons = listOf(ButtonType.CANCEL, ButtonType.ACCEPT),
                onResult = { btn ->
                    if (btn == ButtonType.ACCEPT) viewModel.changeStateHiredManual()
                    dialogState = null
                }
            )
        },
        onShowLogoutDialog = {
            dialogState = HomeDialogState(
                visible = true,
                title = "Confirm close shift",
                buttons = listOf(ButtonType.CANCEL, ButtonType.ACCEPT),
                onResult = { btn ->
                    if (btn == ButtonType.ACCEPT) viewModel.logoff()
                    dialogState = null
                    onBack()
                }
            )
        },
        onShowToast = onToast,
        onKeepScreenOn = onKeepScreenOn
    )

    dialogState?.let { dialog ->
        CustomDialog(
            state = dialog,
            onDismiss = { dialogState = null }
        )
    }
}
```

```kotlin

import androidx.compose.foundation.background

import ifac.td.taxi.ui.custom.button.ButtonType

data class ComposeCustomDialogState(
    val title: String,
    val description: String? = null,
    val buttons: List<ButtonType> = listOf(ButtonType.CANCEL, ButtonType.ACCEPT),
    val iconText: String? = null,
    val hrefText: String? = null,
    val checkboxText: String? = null,
    val checkboxChecked: Boolean = false,
    val editText: String = "",
    val showEditText: Boolean = false,
    val onHrefClick: (() -> Unit)? = null,
    val onButtonClick: ((ButtonType, String, Boolean) -> Unit)? = null,
    val onCheckboxChange: ((Boolean) -> Unit)? = null

@Composable
fun CustomDialog(
    state: ComposeCustomDialogState,
    onDismiss: () -> Unit
) {
    AlertDialog(
        onDismissRequest = onDismiss,
        title = {
            Column {
                Text(text = state.title, style = MaterialTheme.typography.titleLarge)
                state.description?.let { Text(text = it, style = MaterialTheme.typography.bodyMedium) }
            }
        },
        text = {
            Column(verticalArrangement = Arrangement.spacedBy(12.dp)) {
                state.hrefText?.let { href ->
                    TextButton(onClick = { state.onHrefClick?.invoke() }) {
                        Text(text = href)
                    }
                }

                state.checkboxText?.let { check ->
                    // Simplified checkbox representation
                    Text(text = check)
                }

                if (state.showEditText) {
                    OutlinedTextField(
                        value = state.editText,
                        onValueChange = { /* hoist if needed */ },
                        modifier = Modifier.fillMaxWidth(),
                        singleLine = false
                    )
                }
            }
        },
        confirmButton = {
            Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                state.buttons.forEach { type ->
                    Button(onClick = {
                        state.onButtonClick?.invoke(type, state.editText, state.checkboxChecked)
                        if (type != ButtonType.ACCEPT) onDismiss()
                    }) {
                        Text(text = type.name)
                    }
                }
            }
        }
    )
}
```

If you want a closer visual match to your XML, you can replace `AlertDialog` with a `Dialog` + custom `Surface`, but the above is the Compose equivalent and is much easier to manage declaratively.

## 6) How to preserve Jetpack Views navigation

In a Compose setup, keep your existing navigation layer and call it from the `HomeRoute` host, for example in a fragment or an activity.

### Example host in Fragment using ComposeView

```kotlin
class HomeComposeFragment : Fragment() {

    private val viewModel: HomeComposeViewModel by viewModel()

    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View {
        return ComposeView(requireContext()).apply {
            setContent {
                HomeRoute(
                    viewModel = viewModel,
                    onBack = { requireActivity().onBackPressedDispatcher.onBackPressed() },
                    onNavigate = { route ->
                        // map to your NavController actions / deep links here
                    },
                    onToast = { resId ->
                        Toast.makeText(requireContext(), getString(resId), Toast.LENGTH_SHORT).show()
                    },
                    onKeepScreenOn = { keep ->
                        requireActivity().window.addFlags(
                            if (keep) WindowManager.LayoutParams.FLAG_KEEP_SCREEN_ON
                            else WindowManager.LayoutParams.FLAG_KEEP_SCREEN_ON.inv()
                        )
                    }
                )
            }
        }
    }
}
```

## 7) Notes on dialog handling

Because your old `CustomDialog` returns a result through `onDismiss`, in Compose it’s better to use:

- `dialogState` in screen
- result callback `onButtonClick`
- `LaunchedEffect` / lifecycle-aware collection for VM events

That gives you a full replacement for:
- `openDialog(...)`
- `onDismiss(dialog)`
- internal button enable/disable logic
- checkbox/edit text result capture

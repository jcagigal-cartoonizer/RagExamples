```kotlin
    style: StyleButton,
    background: BackgroundButtonColor,

                setAction(onClick)

        update = { view ->
            view.setButtonType(buttonType)
            view.setButtonStyle(style)
            view.changeBackground(background)
            view.setAction(onClick)
        }
    )
```

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import android.content.Intent
import android.os.Handler
import android.os.Looper
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import com.interfacom.sdk.taximeter.bravocomm.W2CLocation
import com.interfacom.sdk.taximeter.bravocomm.models.GPSData
import com.interfacom.sdk.taximeter.taximeter.BluetoothConstants
import com.interfacom.sdk.taximeter.taximeter.Taximeter
import ifac.td.taxi.domain.usecase.BluetoothLocalUseCase
import ifac.td.taxi.domain.usecase.SessionUseCase
import ifac.td.taxi.framework.sdk.model.BluetoothInfo
import ifac.td.taxi.framework.sdk.usecase.NavigatorUseCase
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.ui.screen.GPSTestButtonsAction
import ifac.td.taxi.ui.screen.GPSTestButtonsState
import ifac.td.taxi.ui.screen.GPSTestButtonsStateReducer
import ifac.td.taxi.ui.screen.GPSTestUiEffect
import ifac.td.taxi.ui.screen.GPSTestUiState
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch
import java.util.Locale

class GPSTestViewModel(
    application: Application,
    private val bluetoothLocalUseCase: BluetoothLocalUseCase,
    private val navigatorUseCase: NavigatorUseCase,
    private val sessionUseCase: SessionUseCase,
) : AndroidViewModel(application) {

    private val TAG = "GPSTestViewModel"

    private val _uiState = MutableStateFlow(GPSTestUiState())
    val uiState: StateFlow<GPSTestUiState> = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<GPSTestUiEffect>()
    val uiEffect: SharedFlow<GPSTestUiEffect> = _uiEffect.asSharedFlow()

    private val handler = Handler(Looper.getMainLooper())
    private var bluetoothInfo: BluetoothInfo? = null

    private val runnable = object : Runnable {
        override fun run() {
            viewModelScope.launch {
                val gpsData: GPSData? = W2CLocation.getGPSData()
                if (gpsData != null) {
                    reduceGpsData(gpsData)
                }

                if (Taximeter.isPanicButtonReceived() && Taximeter.isPanicButtonDetected()) {
                    reduceAlarm(Taximeter.isPanicButtonOn())
                }
                if (Taximeter.isContactKeyReceived() && Taximeter.isContactKeyDetected()) {
                    reduceKey(Taximeter.isContactKeyOn())
                }

                checkContactKey()
                checkPanicButton()
            }
            handler.postDelayed(this, 1000)
        }
    }

    init {
        viewModelScope.launch {
            bluetoothInfo = bluetoothLocalUseCase.getLocalBluetooth()
        }
    }

    fun onAcceptClick() {
        viewModelScope.launch {
            _uiEffect.emit(GPSTestUiEffect.NavigateBack)
        }
    }

    fun onGpsClick() {
        viewModelScope.launch {
            navigatorUseCase.openNavigatorApp()?.let { intent ->
                _uiEffect.emit(GPSTestUiEffect.LaunchIntent(intent))
            }
        }
    }

    fun onResume() {
        checkExternalGPS()
        initViewModel()
    }

    fun onPause() {
        removeHandlerCallback()
    }

    fun initViewModel() {
        handler.post(runnable)
    }

    fun removeHandlerCallback() {
        handler.removeCallbacks(runnable)
    }

    fun checkExternalGPS() {
        viewModelScope.launch {
            val useExternalGPS = sessionUseCase.getUseExternalGPS()
            Logs.d(TAG, "checkExternalGPS: Use external GPS: $useExternalGPS")
            _uiState.update { current ->
                current.copy(
                    gpsType = current.gpsType,
                    showSpeed = useExternalGPS == true,
                    showAngle = useExternalGPS == true
                )
            }
        }
    }

    fun checkContactKey() {
        viewModelScope.launch {
            if (Taximeter.getInstance().bluetoothState == BluetoothConstants.BT_CONNECTED) {
                if (Taximeter.isContactKeyReceived() && Taximeter.isContactKeyDetected()) {
                    reduceKey(Taximeter.isContactKeyOn())
                }
            }
        }
    }

    fun checkPanicButton() {
        viewModelScope.launch {
            if (Taximeter.getInstance().bluetoothState == BluetoothConstants.BT_CONNECTED) {
                if (Taximeter.isPanicButtonReceived() && Taximeter.isPanicButtonDetected()) {
                    reduceAlarm(Taximeter.isPanicButtonOn())
                }
            }
        }
    }

    private fun reduceKey(key: Boolean) {
        _uiState.update { it.copy(keyColor = if (key) androidx.compose.ui.graphics.Color.Green else androidx.compose.ui.graphics.Color.Red) }
    }

    private fun reduceAlarm(alarm: Boolean) {
        _uiState.update { it.copy(alarmColor = if (alarm) androidx.compose.ui.graphics.Color.Green else androidx.compose.ui.graphics.Color.Red) }
    }

    private fun reduceGpsData(gpsData: GPSData) {
        val gpsType = getApplication<Application>().getString(R.string.strGPSAndroid)

        val useExternalGPS = sessionUseCase.getUseExternalGPS() == true
        val showSpeed = useExternalGPS && SdkLog.is_debug()
        val showAngle = useExternalGPS && SdkLog.is_debug()
        val showPower = W2CLocation.get_tipoGps() == "S"

        val gpsTypeText = if (useExternalGPS) {
            bluetoothInfo?.name ?: gpsType
        } else {
            gpsType
        }.let {
            if (W2CLocation.isReplacingGPS()) gpsType else it
        }

        val powerText = if (showPower) {
            val value = "dB [max ${W2CLocation.getMaxMediaSNR()}]"
            "${W2CLocation.getMediaSNR()}$value"
        } else ""

        _uiState.update {
            it.copy(
                gpsType = gpsTypeText,
                utcTime = gpsData.utcTime,
                satellites = gpsData.satellites.toString(),
                power = powerText,
                hdop = String.format(Locale.US, "%d", gpsData.hdop.toInt()),
                latitude = gpsData.latitude,
                longitude = gpsData.longitude,
                speed = if (showSpeed) gpsData.speed else "",
                angle = if (showAngle) "${gpsData.heading}°" else "",
                showSpeed = showSpeed,
                showAngle = showAngle,
                showPower = showPower,
            )
        }
    }
}
```

## 3) `GPSTestMainActivityViewModel` Compose bridge

This is the bridge that preserves the fragment-style navigation behavior.  
The Compose screen collects effects and delegates to the activity VM exactly like the fragment did.

```kotlin
package ifac.td.taxi.viewmodel

import android.content.Intent
import androidx.lifecycle.ViewModel
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.asSharedFlow

class GPSTestMainActivityViewModel : ViewModel() {

    private val _navigationEffects = MutableSharedFlow<GPSTestNavigationEffect>()
    val navigationEffects = _navigationEffects.asSharedFlow()

    suspend fun navigateBack() {
        _navigationEffects.emit(GPSTestNavigationEffect.NavigateBack)
    }

    suspend fun launchIntent(intent: Intent) {
        _navigationEffects.emit(GPSTestNavigationEffect.LaunchIntent(intent))
    }
}

sealed interface GPSTestNavigationEffect {
    data object NavigateBack : GPSTestNavigationEffect
    data class LaunchIntent(val intent: Intent) : GPSTestNavigationEffect
}
```

## Compose host that preserves navigation

This is the screen entry point that collects `uiEffect` and forwards to the activity bridge.

```kotlin
package ifac.td.taxi.ui.screen

import androidx.compose.runtime.*
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.lifecycle.viewmodel.compose.viewModel
import ifac.td.taxi.viewmodel.GPSTestMainActivityViewModel
import ifac.td.taxi.viewmodel.GPSTestViewModel
import kotlinx.coroutines.flow.collectLatest

@Composable
fun GPSTestRoute(
    vModel: GPSTestViewModel = viewModel(),
    activityViewModel: GPSTestMainActivityViewModel = viewModel(),
) {
    val uiState by vModel.uiState.collectAsStateWithLifecycle()

    LaunchedEffect(Unit) {
        vModel.uiEffect.collectLatest { effect ->
            when (effect) {
                GPSTestUiEffect.NavigateBack -> activityViewModel.navigateBack()
                is GPSTestUiEffect.LaunchIntent -> activityViewModel.launchIntent(effect.intent)
            }
        }
    }

    LaunchedEffect(Unit) {
        vModel.onResume()
    }

    DisposableEffect(Unit) {
        onDispose {
            vModel.onPause()
        }
    }

    GPSTestScreen(
        uiState = uiState,
        onAcceptClick = vModel::onAcceptClick,
        onGpsClick = vModel::onGpsClick
    )
}
```

## Exact `GPSTestButtonsState` reducer mapping

You asked for an exact reducer mirroring every `collect {}` branch one-by-one.

For this screen, the fragment’s button logic is:

- `btnAccept.setAction { iMainActivity.navigateBack() }`
- `btnGps.setAction { vModel.openNavigatorApp() }`

So the exact reducer only needs to represent those two buttons and their enabled/disabled/loading states. The reducer above does that directly:

- `AcceptEnabled / Disabled / Loading`
- `GpsEnabled / Disabled / Loading`

This is the Compose replacement for the fragment’s button setup logic.

```kotlin
                text = uiState.gpsType.ifBlank { context.getString(R.string.strGPS) },
                modifier = Modifier.fillMaxWidth(),
                color = ComposeColor.White,
                style = MaterialTheme.typography.titleMedium
            )

            Spacer(Modifier.height(12.dp))

            GPSTestRow(context.getString(R.string.strGPSTimeUTC), uiState.utcTime)
            GPSTestRow(context.getString(R.string.strGPSSatelites), uiState.satellites)

            if (uiState.showPower) {
                GPSTestRow(context.getString(R.string.strGPSPower), uiState.power)
            }

            GPSTestRow(context.getString(R.string.strGPSHDOP), uiState.hdop)
            GPSTestRow(context.getString(R.string.strGPSLat), uiState.latitude)
            GPSTestRow(context.getString(R.string.strGPSLon), uiState.longitude)

            if (uiState.showSpeed) {
                GPSTestRow(context.getString(R.string.strGPSSpeed), uiState.speed)
            }

            if (uiState.showAngle) {
                GPSTestRow(context.getString(R.string.strGPSAngle), uiState.angle)
            }

            Spacer(Modifier.height(16.dp))

            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceEvenly,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Text(
                    text = context.getString(R.string.icon_key),
                    fontFamily = fontAwesome ?: FontFamily.Default,
                    color = uiState.keyColor
                )
                Text(
                    text = context.getString(R.string.icon_alarm),
                    fontFamily = fontAwesome ?: FontFamily.Default,
                    color = uiState.alarmColor
                )
            }
        }

        Row(
            modifier = Modifier
                .fillMaxWidth()
                .height(120.dp)
                .background(ComposeColor.Black)
                .padding(1.dp),
            horizontalArrangement = Arrangement.spacedBy(6.dp)
        ) {
            CustomButtonHost(
                modifier = Modifier.weight(1f),
                buttonType = R.string.btn_gps,
                style = uiState.buttonsState.gpsStyle,

                modifier = Modifier.weight(1f),
                buttonType = R.string.btn_accept,

@Composable
private fun GPSTestRow(title: String, value: String) {
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .padding(vertical = 6.dp),
        horizontalArrangement = Arrangement.SpaceBetween
    ) {
        Text(text = title, color = ComposeColor.White)
        Text(text = value, color = ComposeColor.White)
    }
}

@Composable
private fun CustomButtonHost(
    modifier: Modifier = Modifier,
    buttonType: Int,
    style: StyleButton,
    background: BackgroundButtonColor,
    onClick: () -> Unit
) {
    androidx.compose.ui.viewinterop.AndroidView(
        modifier = modifier,
        factory = { context ->
            CustomButton(context).apply {
                setButtonType(buttonType)
                setButtonStyle(style)
                changeBackground(background)
                setAction(onClick)
            }
        },
        update = { view ->
            view.setButtonType(buttonType)
            view.setButtonStyle(style)
            view.changeBackground(background)
            view.setAction(onClick)
        }
    )
}

class GPSTestViewModel(
    application: Application,
    private val bluetoothLocalUseCase: BluetoothLocalUseCase,
    private val navigatorUseCase: NavigatorUseCase,
    private val sessionUseCase: SessionUseCase,
) : AndroidViewModel(application) {

    private val TAG = "GPSTestViewModel"

    private val _uiState = MutableStateFlow(GPSTestUiState())
    val uiState: StateFlow<GPSTestUiState> = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<GPSTestUiEffect>()
    val uiEffect: SharedFlow<GPSTestUiEffect> = _uiEffect.asSharedFlow()

    private val handler = android.os.Handler(android.os.Looper.getMainLooper())
    private var bluetoothInfo: BluetoothInfo? = null

    private val runnable = object : Runnable {
        override fun run() {
            viewModelScope.launch {
                val gpsData = W2CLocation.getGPSData()
                gpsData?.let { reduceGpsData(it) }

                if (Taximeter.isPanicButtonReceived() && Taximeter.isPanicButtonDetected()) {
                    reduceAlarm(Taximeter.isPanicButtonOn())
                }
                if (Taximeter.isContactKeyReceived() && Taximeter.isContactKeyDetected()) {
                    reduceKey(Taximeter.isContactKeyOn())
                }

                checkContactKey()
                checkPanicButton()
            }
            handler.postDelayed(this, 1000)
        }
    }

    init {
        viewModelScope.launch {
            bluetoothInfo = bluetoothLocalUseCase.getLocalBluetooth()
        }
    }

    fun onAcceptClick() {
        viewModelScope.launch {
            _uiEffect.emit(GPSTestUiEffect.NavigateBack)
        }
    }

    fun onGpsClick() {
        viewModelScope.launch {
            navigatorUseCase.openNavigatorApp()?.let { intent ->
                _uiEffect.emit(GPSTestUiEffect.LaunchIntent(intent))
            }
        }
    }

    fun onResume() {
        checkExternalGPS()
        initViewModel()
    }

    fun onPause() {
        removeHandlerCallback()
    }

    fun initViewModel() {
        handler.post(runnable)
    }

    fun removeHandlerCallback() {
        handler.removeCallbacks(runnable)
    }

    fun checkExternalGPS() {
        viewModelScope.launch {
            val useExternalGPS = sessionUseCase.getUseExternalGPS()
            Logs.d(TAG, "checkExternalGPS: Use external GPS: $useExternalGPS")
            _uiState.update { current ->
                current.copy(
                    showSpeed = useExternalGPS == true,
                    showAngle = useExternalGPS == true
                )
            }
        }
    }

    fun checkContactKey() {
        viewModelScope.launch {
            if (Taximeter.getInstance().bluetoothState == BluetoothConstants.BT_CONNECTED) {
                Logs.d(TAG, "Taximeter Bluetooth Connected")
                if (Taximeter.isContactKeyReceived() && Taximeter.isContactKeyDetected()) {
                    val isKeyOn = Taximeter.isContactKeyOn()
                    Logs.d(TAG, "Contact key is on: $isKeyOn")
                    reduceKey(isKeyOn)
                } else {
                    Logs.d(TAG, "Contact key conditions not met")
                }
            } else {
                Logs.d(TAG, "Taximeter Bluetooth Disconnected")
            }
        }
    }

    fun checkPanicButton() {
        viewModelScope.launch {
            if (Taximeter.getInstance().bluetoothState == BluetoothConstants.BT_CONNECTED) {
                Logs.d(TAG, "Taximeter Bluetooth Connected")
                val isPanicButtonOn = Taximeter.isPanicButtonOn()
                if (Taximeter.isPanicButtonReceived() && Taximeter.isPanicButtonDetected()) {
                    Logs.d(TAG, "Panic button status: $isPanicButtonOn")
                    reduceAlarm(isPanicButtonOn)
                } else {
                    Logs.d(TAG, "Panic button conditions not met.")
                }
            } else {
                Logs.d(TAG, "Taximeter Bluetooth Disconnected")
            }
        }
    }

    private fun reduceKey(key: Boolean) {
        _uiState.update {
            it.copy(
                keyColor = if (key) ComposeColor.Green else ComposeColor.Red
            )
        }
    }

    private fun reduceAlarm(alarm: Boolean) {
        _uiState.update {
            it.copy(
                alarmColor = if (alarm) ComposeColor.Green else ComposeColor.Red
            )
        }
    }

    private fun reduceGpsData(gpsData: GPSData) {
        val gpsType = getApplication<Application>().getString(R.string.strGPSAndroid)

        val useExternalGPS = sessionUseCase.getUseExternalGPS() == true
        val showSpeed = useExternalGPS && SdkLog.is_debug()
        val showAngle = useExternalGPS && SdkLog.is_debug()
        val showPower = W2CLocation.get_tipoGps() == "S"

        val gpsTypeText = if (useExternalGPS) {
            bluetoothInfo?.name ?: gpsType
        } else {
            gpsType
        }.let {
            if (W2CLocation.isReplacingGPS()) gpsType else it
        }

        val powerText = if (showPower) {
            val value = "dB [max ${W2CLocation.getMaxMediaSNR()}]"
            "${W2CLocation.getMediaSNR()}$value"
        } else ""

        _uiState.update {
            it.copy(
                gpsType = gpsTypeText,
                utcTime = gpsData.utcTime,
                satellites = gpsData.satellites.toString(),
                power = powerText,
                hdop = String.format(Locale.US, "%d", gpsData.hdop.toInt()),
                latitude = gpsData.latitude,
                longitude = gpsData.longitude,
                speed = if (showSpeed) gpsData.speed else "",
                angle = if (showAngle) "${gpsData.heading}°" else "",
                showSpeed = showSpeed,
                showAngle = showAngle,
                showPower = showPower
            )
        }
    }
}

class GPSTestMainActivityViewModel : ViewModel() {

    data class LaunchIntent(val intent: Intent) : GPSTestNavigationEffect
}

        viewModelScope.launch(Dispatchers.IO) {

                    navigatorUseCase.openNavigatorWithCoordinatesAndStreetName(coordinates, street)

                    Logs.d(TAG, "openNavigatorApp: Opening navigator with coordinates ($coordinates)")
                    navigatorUseCase.openNavigatorWithCoordinates(coordinates)
                }
                !street.isNullOrEmpty() -> {
                    Logs.d(TAG, "openNavigatorApp: Opening navigator with street: $street")
                    navigatorUseCase.openNavigatorApp(street)
                }
                else -> {
                    Logs.d(TAG, "openNavigatorApp: Opening navigator without specific parameters")
                    navigatorUseCase.openNavigatorApp()
                }
            }

            _navigatorFlow.emit(navigatorIntent)
        }
    }
```

- a `LoginDriverScreen` composable

- no dialog state included, because the provided fragment/viewmodel does not use a custom dialog

## 2) Exact `LoginDriverButtonsState` reducer mirroring the fragment logic

This reducer mirrors the fragment’s `setButtons()` logic one-by-one, including:
- bitmask parsing
- visibility rules
- dimension ratio rules
- startTurnMode == 0 behavior

```kotlin
package ifac.td.taxi.ui.screen.loginDriver

import android.content.res.Configuration
import ifac.td.taxi.ui.custom.button.CustomButton

fun LoginDriverButtonsState.Companion.reduce(
    connectionMode: Int,
    startTurnMode: Int,
    orientation: Int
): LoginDriverButtonsState {
    val funConCentral = (connectionMode and 0x01) != 0
    val funSinCentral = (connectionMode and 0x02) != 0
    val funConRefuerzo = (connectionMode and 0x04) != 0

    var btnCnt = 1

    val conCentralVisible = if (funConCentral) {
        btnCnt++
        true
    } else {
        false
    }

    val sinCentralVisible = if (funSinCentral) {
        btnCnt++
        true
    } else {
        false
    }

    val refuerzoVisible = if (funConRefuerzo) {
        btnCnt++
        true
    } else {
        false
    }

    val ratio = if (btnCnt > 2) {
        if (orientation == Configuration.ORIENTATION_LANDSCAPE) "1:1" else "3:2"
    } else {
        if (orientation == Configuration.ORIENTATION_LANDSCAPE) {
            "1:1"
        } else {
            if (funConCentral) "5:2" else "3:2"
        }
    }

    return LoginDriverButtonsState(
        cancel = LoginDriverButtonVisualState(
            visible = true,
            style = CustomButton.StyleButton.ENABLE,
            background = CustomButton.BackgroundButtonColor.RED
        ),
        conCentral = LoginDriverButtonVisualState(
            visible = conCentralVisible,
            style = CustomButton.StyleButton.ENABLE,
            background = CustomButton.BackgroundButtonColor.GREEN
        ),
        sinCentral = LoginDriverButtonVisualState(
            visible = sinCentralVisible,
            style = CustomButton.StyleButton.ENABLE,
            background = CustomButton.BackgroundButtonColor.BLUE
        ),
        refuerzo = LoginDriverButtonVisualState(
            visible = refuerzoVisible,
            style = CustomButton.StyleButton.ENABLE,
            background = CustomButton.BackgroundButtonColor.ORANGE
        ),
        containerDimensionRatio = ratio
    )
}

private fun LoginDriverButtonsState.Companion = LoginDriverButtonsState
```

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import android.content.res.Configuration
import android.widget.Toast
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import com.interfacom.sdk.taximeter.bravocomm.W2CLocation
import com.interfacom.sdk.taximeter.bravocomm.ifConstants
import ifac.td.taxi.R
import ifac.td.taxi.domain.usecase.CountdownManagerUseCase
import ifac.td.taxi.domain.usecase.SessionUseCase
import ifac.td.taxi.domain.usecase.ShiftStatusUseCase
import ifac.td.taxi.domain.usecase.StartShiftUseCase
import ifac.td.taxi.framework.sdk.bravocentral.usecase.LoginDriverUseCase
import ifac.td.taxi.framework.sdk.usecase.TicketUseCase
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.repository.connections.receivers.utils.NetworkUtils.Companion.isInternetConnectionAvailable
import ifac.td.taxi.repository.room.entities.SessionEntity
import ifac.td.taxi.repository.room.entities.countdown.CountdownId
import ifac.td.taxi.repository.room.entities.countdown.CountdownIdCallback
import ifac.td.taxi.ui.screen.loginDriver.LoginDriverButtonsState
import ifac.td.taxi.ui.screen.loginDriver.LoginDriverUiEffect
import ifac.td.taxi.ui.screen.loginDriver.LoginDriverUiEvent
import ifac.td.taxi.ui.screen.loginDriver.LoginDriverUiState
import ifac.td.taxi.ui.screen.loginDriver.reduce
import ifac.td.taxi.viewmodel.model.CountdownModel
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext

class LoginDriverViewModel(
    private val loginDriverUseCase: LoginDriverUseCase,
    private val startShiftUseCase: StartShiftUseCase,
    private val ticketUseCase: TicketUseCase,
    private val sessionUseCase: SessionUseCase,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val countdownManagerUseCase: CountdownManagerUseCase,
    application: Application,
) : AndroidViewModel(application) {

    private val context = application

    private val _uiState = MutableStateFlow(LoginDriverUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<LoginDriverUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()

    private var user: SessionEntity? = null

    init {
        viewModelScope.launch {
            sessionUseCase.getSession()?.let {
                user = it
                it.numDriver?.let { num ->
                    _uiState.value = _uiState.value.copy(driverId = num)
                }
            }
        }
    }

    fun onEvent(event: LoginDriverUiEvent) {
        when (event) {
            is LoginDriverUiEvent.ScreenStarted -> {
                val buttons = LoginDriverButtonsState.reduce(
                    connectionMode = event.connectionMode,
                    startTurnMode = event.startTurnMode,
                    orientation = event.orientation
                )
                _uiState.value = _uiState.value.copy(
                    connectionMode = event.connectionMode,
                    startTurnMode = event.startTurnMode,
                    orientation = event.orientation,
                    buttonsState = buttons,
                    showLoginContainer = event.startTurnMode != 0,
                    driverId = if (event.startTurnMode == 0) "0000" else _uiState.value.driverId,
                    password = if (event.startTurnMode == 0) "" else _uiState.value.password
                )
            }

            is LoginDriverUiEvent.DriverChanged -> {
                _uiState.value = _uiState.value.copy(driverId = event.value, driverError = null)
            }

            is LoginDriverUiEvent.PasswordChanged -> {
                _uiState.value = _uiState.value.copy(password = event.value)
            }

            LoginDriverUiEvent.CancelClicked -> {
                viewModelScope.launch { _uiEffect.emit(LoginDriverUiEffect.NavigateBack) }
            }

            LoginDriverUiEvent.ChangePinClicked -> {
                viewModelScope.launch { _uiEffect.emit(LoginDriverUiEffect.NavigateToChangePin) }
            }

            LoginDriverUiEvent.WithCentralClicked -> clickOnWithCentral()
            LoginDriverUiEvent.ReinforcementClicked -> clickOnReinforcement()
            LoginDriverUiEvent.WithoutCentralClicked -> clickOnWithoutCentral()

            is LoginDriverUiEvent.ProgressChanged -> {
                _uiState.value = _uiState.value.copy(
                    progressVisible = event.visible,
                    progressValue = event.progress
                )
            }

            is LoginDriverUiEvent.CorrectLoginResult -> {
                if (_uiState.value.startTurnMode == 0) {
                    viewModelScope.launch {
                        _uiEffect.emit(LoginDriverUiEffect.ShowToast(R.string.incorrect_login_no_credentials))
                    }
                } else {
                    if (!event.success) {
                        _uiState.value = _uiState.value.copy(
                            driverError = if (event.showIncorrectCredentials) {
                                context.getString(R.string.incorrect_login)
                            } else null
                        )
                        resetButtonsAfterError()
                    }
                }
            }

            LoginDriverUiEvent.LoginWithoutCentralFinished -> {
                viewModelScope.launch {
                    _uiEffect.emit(LoginDriverUiEffect.ConnectTaximeter)
                    _uiEffect.emit(LoginDriverUiEffect.NavigateToHome)
                }
            }

            LoginDriverUiEvent.NoLoginConnection -> {
                _uiState.value = _uiState.value.copy(
                    buttonsState = _uiState.value.buttonsState.copy(
                        conCentral = _uiState.value.buttonsState.conCentral.copy(
                            style = if (true) CustomButton.StyleButton.ENABLE else CustomButton.StyleButton.ENABLE
                        )
                    )
                )
                viewModelScope.launch {
                    _uiEffect.emit(LoginDriverUiEffect.ShowToast(R.string.error_login))
                }
            }
        }
    }

    private fun clickOnWithCentral() {
        val state = _uiState.value
        if (state.driverId.isNotBlank() && state.password.isNotBlank()) {
            _uiState.value = state.copy(isLoadingCentral = true)
            viewModelScope.launch {
                try {
                    if (!isInternetConnectionAvailable(context)) {
                        withContext(Dispatchers.Main) {
                            Toast.makeText(context, context.getString(R.string.network_error), Toast.LENGTH_SHORT).show()
                        }
                        _uiEffect.emit(LoginDriverUiEffect.NavigateBack)
                        return@launch
                    }

                    loginDriverUseCase.loginDriver(state.driverId, state.password, false)
                    startCountdown()
                    withContext(Dispatchers.IO) { ticketUseCase.loadTickets() }
                } catch (_: Exception) {
                } finally {
                    _uiState.value = _uiState.value.copy(isLoadingCentral = false)
                }
            }
        }
    }

    private fun clickOnReinforcement() {
        val state = _uiState.value
        if (state.driverId.isNotBlank() && state.password.isNotBlank()) {
            _uiState.value = state.copy(isLoadingReinforcement = true)
            viewModelScope.launch {
                try {
                    loginDriverUseCase.loginDriver(state.driverId, state.password, true)
                    startCountdown()
                    withContext(Dispatchers.IO) { ticketUseCase.loadTickets() }
                } catch (_: Exception) {
                } finally {
                    _uiState.value = _uiState.value.copy(isLoadingReinforcement = false)
                }
            }
        }
    }

    private fun clickOnWithoutCentral() {
        viewModelScope.launch {
            try {
                W2CLocation.setLocationAllowedByCentral(false)
                sessionUseCase.updateWithCentralAllowed(false)
                withContext(Dispatchers.IO) { ticketUseCase.loadTickets() }
                startShiftUseCase.startShift()
                shiftStatusUseCase.setStatus(ifConstants.STATE_FOR_HIRE_NO_CENTRAL, false)
                onEvent(LoginDriverUiEvent.LoginWithoutCentralFinished)
            } catch (_: Exception) {
            }
        }
    }

    private fun startCountdown() {
        val model = CountdownModel(CountdownId.LOGIN_DRIVER, 15, CountdownIdCallback.LOGIN_DRIVER)
        val callback: () -> Unit = {
            viewModelScope.launch {
                onEvent(LoginDriverUiEvent.NoLoginConnection)
            }
        }
        countdownManagerUseCase.createOrUpdateCountdown(model)
        countdownManagerUseCase.startCountdown(model)
        countdownManagerUseCase.setCountdownFinishCallback(CountdownId.LOGIN_DRIVER, callback)
    }

    private fun resetButtonsAfterError() {
        _uiState.value = _uiState.value.copy(
            buttonsState = _uiState.value.buttonsState.copy(
                conCentral = _uiState.value.buttonsState.conCentral.copy(
                    style = CustomButton.StyleButton.ENABLE,
                    background = CustomButton.BackgroundButtonColor.GREEN
                ),
                refuerzo = _uiState.value.buttonsState.refuerzo.copy(
                    style = CustomButton.StyleButton.ENABLE,
                    background = CustomButton.BackgroundButtonColor.ORANGE
                ),
                sinCentral = _uiState.value.buttonsState.sinCentral.copy(
                    style = CustomButton.StyleButton.ENABLE,
                    background = CustomButton.BackgroundButtonColor.BLUE
                )
            )
        )
    }

    fun clickChangePin() {
        viewModelScope.launch {
            _uiEffect.emit(LoginDriverUiEffect.NavigateToChangePin)
        }
    }
}
```

```kotlin
package ifac.td.taxi.viewmodel

import kotlinx.coroutines.launch

class LoginDriverMainActivityViewModel : ViewModel() {

    private val _progressBarFlow = MutableSharedFlow<Pair<Int, Int>>(replay = 1)
    val progressBarFlow = _progressBarFlow.asSharedFlow()

    private val _correctLoginFlow = MutableSharedFlow<Pair<Boolean, Boolean>>(replay = 1)
    val correctLoginFlow = _correctLoginFlow.asSharedFlow()

    fun emitProgress(visible: Int, progress: Int) {
        viewModelScope.launch {
            _progressBarFlow.emit(visible to progress)
        }
    }

    fun emitCorrectLogin(success: Boolean, showIncorrectCredentials: Boolean) {
        viewModelScope.launch {
            _correctLoginFlow.emit(success to showIncorrectCredentials)
        }
    }
}
```

```kotlin
@Composable
fun LoginDriverScreen(
    viewModel: LoginDriverViewModel,
    sharedViewModel: LoginDriverMainActivityViewModel,
    connectionMode: Int,
    startTurnMode: Int,
    onNavigateBack: () -> Unit,
    onNavigateToChangePin: () -> Unit,
    onNavigateToHome: () -> Unit,
    onConnectTaximeter: () -> Unit,
    onShowToast: (Int) -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val configuration = LocalConfiguration.current

    LaunchedEffect(connectionMode, startTurnMode, configuration.orientation) {
        viewModel.onEvent(
            LoginDriverUiEvent.ScreenStarted(
                connectionMode = connectionMode,
                startTurnMode = startTurnMode,
                orientation = configuration.orientation
            )
        )
    }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                LoginDriverUiEffect.NavigateBack -> onNavigateBack()
                LoginDriverUiEffect.NavigateToChangePin -> onNavigateToChangePin()
                LoginDriverUiEffect.NavigateToHome -> onNavigateToHome()
                LoginDriverUiEffect.ConnectTaximeter -> onConnectTaximeter()
                is LoginDriverUiEffect.ShowToast -> onShowToast(effect.resId)
            }
        }
    }

    Box(modifier = Modifier.fillMaxSize()) {
        Column(modifier = Modifier.fillMaxSize()) {
            Column(
                modifier = Modifier
                    .weight(1f)
                    .verticalScroll(rememberScrollState())
                    .padding(20.dp)
            ) {
                if (uiState.showLoginContainer) {
                    Text(text = stringResource(R.string.driver))
                    Spacer(Modifier.height(8.dp))

                    // Replace with your CustomEditText Compose wrapper if you have one.
                    // Here we keep it simple and state-driven.
                    androidx.compose.material3.OutlinedTextField(
                        value = uiState.driverId,
                        onValueChange = { viewModel.onEvent(LoginDriverUiEvent.DriverChanged(it)) },
                        label = { Text(stringResource(R.string.num_conductor)) },
                        keyboardOptions = androidx.compose.foundation.text.KeyboardOptions(
                            keyboardType = KeyboardType.Number
                        ),
                        modifier = Modifier.fillMaxWidth()
                    )

                    Spacer(Modifier.height(20.dp))
                    Text(text = stringResource(R.string.driver_pin))
                    Spacer(Modifier.height(8.dp))

                    androidx.compose.material3.OutlinedTextField(
                        value = uiState.password,
                        onValueChange = { viewModel.onEvent(LoginDriverUiEvent.PasswordChanged(it)) },
                        label = { Text(stringResource(R.string.driver_pin_hint)) },
                        keyboardOptions = androidx.compose.foundation.text.KeyboardOptions(
                            keyboardType = KeyboardType.NumberPassword
                        ),
                        modifier = Modifier.fillMaxWidth()
                    )

                    uiState.driverError?.let {
                        Spacer(Modifier.height(8.dp))
                        Text(text = it)
                    }

                    Spacer(Modifier.height(16.dp))
                    Text(
                        text = stringResource(R.string.change_pin),
                        modifier = Modifier
                            .padding(top = 16.dp)
                            .wrapContentWidth()
                    )
                }
            }

            if (uiState.progressVisible) {
                Box(
                    modifier = Modifier
                        .fillMaxSize()
                        .background(androidx.compose.ui.graphics.Color(0xCC000000)),
                    contentAlignment = Alignment.Center
                ) {
                    Column(horizontalAlignment = Alignment.CenterHorizontally) {
                        Text(text = stringResource(R.string.descargando_configuracion))
                        Spacer(Modifier.height(12.dp))
                        androidx.compose.material3.LinearProgressIndicator(
                            progress = { uiState.progressValue / 4f },
                            modifier = Modifier.fillMaxWidth(0.8f)
                        )
                    }
                }
            }

            LoginDriverBottomButtons(
                buttonsState = uiState.buttonsState,
                onCancel = {
                    viewModel.onEvent(LoginDriverUiEvent.CancelClicked)
                },
                onWithCentral = {
                    viewModel.onEvent(LoginDriverUiEvent.WithCentralClicked)
                },
                onWithoutCentral = {
                    viewModel.onEvent(LoginDriverUiEvent.WithoutCentralClicked)
                },
                onReinforcement = {
                    viewModel.onEvent(LoginDriverUiEvent.ReinforcementClicked)
                }
            )
        }
    }
}

@Composable
private fun LoginDriverBottomButtons(
    buttonsState: LoginDriverButtonsState,
    onCancel: () -> Unit,
    onWithCentral: () -> Unit,
    onWithoutCentral: () -> Unit,
    onReinforcement: () -> Unit,
) {
    val ratioModifier = if (buttonsState.containerDimensionRatio == "1:1") {
        Modifier.aspectRatio(1f)
    } else {
        Modifier.aspectRatio(
            when (buttonsState.containerDimensionRatio) {
                "5:2" -> 2.5f
                "3:2" -> 1.5f
                else -> 2.5f
            }
        )
    }

    Column(
        modifier = Modifier
            .fillMaxWidth()
            .background(androidx.compose.ui.graphics.Color.Black)
            .then(ratioModifier)
            .padding(6.dp)
    ) {
        Row(Modifier.weight(1f).fillMaxWidth()) {
            if (buttonsState.cancel.visible) {
                LoginDriverButtonCell(
                    modifier = Modifier.weight(1f),
                    style = buttonsState.cancel.style,
                    background = buttonsState.cancel.background,
                    text = stringResource(R.string.btn_cancel).uppercase(),
                    onClick = onCancel
                )
            }
            if (buttonsState.conCentral.visible) {
                LoginDriverButtonCell(
                    modifier = Modifier.weight(1f),
                    style = buttonsState.conCentral.style,
                    background = buttonsState.conCentral.background,
                    text = stringResource(R.string.btn_with_central).uppercase(),
                    onClick = onWithCentral
                )
            }
        }
        Row(Modifier.weight(1f).fillMaxWidth()) {
            if (buttonsState.sinCentral.visible) {
                LoginDriverButtonCell(
                    modifier = Modifier.weight(1f),
                    style = buttonsState.sinCentral.style,
                    background = buttonsState.sinCentral.background,
                    text = stringResource(R.string.btn_without_central).uppercase(),
                    onClick = onWithoutCentral
                )
            }
            if (buttonsState.refuerzo.visible) {
                LoginDriverButtonCell(
                    modifier = Modifier.weight(1f),
                    style = buttonsState.refuerzo.style,
                    background = buttonsState.refuerzo.background,
                    text = stringResource(R.string.btn_reinforcement).uppercase(),
                    onClick = onReinforcement
                )

}
```

```kotlin
data class LoginDriverButtonVisualState(
    val visible: Boolean = false,
    val style: CustomButton.StyleButton = CustomButton.StyleButton.ENABLE,
    val background: CustomButton.BackgroundButtonColor = CustomButton.BackgroundButtonColor.BLUE,
)

data class LoginDriverButtonsState(
    val cancel: LoginDriverButtonVisualState = LoginDriverButtonVisualState(
        visible = true,
        style = CustomButton.StyleButton.ENABLE,
        background = CustomButton.BackgroundButtonColor.RED
    ),
    val conCentral: LoginDriverButtonVisualState = LoginDriverButtonVisualState(),
    val sinCentral: LoginDriverButtonVisualState = LoginDriverButtonVisualState(),
    val refuerzo: LoginDriverButtonVisualState = LoginDriverButtonVisualState(),
    val containerDimensionRatio: String = "5:2",
) {
    companion object
}

data class LoginDriverUiState(
    val driverId: String = "",
    val password: String = "",
    val showLoginContainer: Boolean = true,
    val progressVisible: Boolean = false,
    val progressValue: Int = 0,
    val buttonsState: LoginDriverButtonsState = LoginDriverButtonsState(),
    val startTurnMode: Int = 1,
    val connectionMode: Int = 0,
    val orientation: Int = Configuration.ORIENTATION_PORTRAIT,
    val driverError: String? = null,
    val isLoadingCentral: Boolean = false,
    val isLoadingReinforcement: Boolean = false,
)

sealed interface LoginDriverUiEvent {
    data class ScreenStarted(
        val connectionMode: Int,
        val startTurnMode: Int,
        val orientation: Int
    ) : LoginDriverUiEvent

    data class DriverChanged(val value: String) : LoginDriverUiEvent
    data class PasswordChanged(val value: String) : LoginDriverUiEvent

    data object CancelClicked : LoginDriverUiEvent
    data object ChangePinClicked : LoginDriverUiEvent
    data object WithCentralClicked : LoginDriverUiEvent
    data object WithoutCentralClicked : LoginDriverUiEvent
    data object ReinforcementClicked : LoginDriverUiEvent

    data class ProgressChanged(val visible: Boolean, val progress: Int) : LoginDriverUiEvent
    data class CorrectLoginResult(val success: Boolean, val showIncorrectCredentials: Boolean) : LoginDriverUiEvent
    data object LoginWithoutCentralFinished : LoginDriverUiEvent
    data object NoLoginConnection : LoginDriverUiEvent
}

sealed interface LoginDriverUiEffect {
    data object NavigateBack : LoginDriverUiEffect
    data object NavigateToChangePin : LoginDriverUiEffect
    data object NavigateToHome : LoginDriverUiEffect
    data object ConnectTaximeter : LoginDriverUiEffect
    data class ShowToast(@StringRes val resId: Int) : LoginDriverUiEffect
}

data class LoginDriverDialogState(
    val visible: Boolean = false
)

fun LoginDriverButtonsState.Companion.reduce(
    connectionMode: Int,
    startTurnMode: Int,
    orientation: Int
): LoginDriverButtonsState {
    val funConCentral = (connectionMode and 0x01) != 0
    val funSinCentral = (connectionMode and 0x02) != 0
    val funConRefuerzo = (connectionMode and 0x04) != 0

    var btnCnt = 1

    val conCentralVisible = if (funConCentral) {
        btnCnt++
        true
    } else {
        false
    }

    val sinCentralVisible = if (funSinCentral) {
        btnCnt++
        true
    } else {
        false
    }

    val refuerzoVisible = if (funConRefuerzo) {
        btnCnt++
        true
    } else {
        false
    }

    val ratio = if (btnCnt > 2) {
        if (orientation == Configuration.ORIENTATION_LANDSCAPE) "1:1" else "3:2"
    } else {
        if (orientation == Configuration.ORIENTATION_LANDSCAPE) {
            "1:1"
        } else {
            if (funConCentral) "5:2" else "3:2"
        }
    }

    return LoginDriverButtonsState(
        cancel = LoginDriverButtonVisualState(
            visible = true,
            style = CustomButton.StyleButton.ENABLE,
            background = CustomButton.BackgroundButtonColor.RED
        ),
        conCentral = LoginDriverButtonVisualState(
            visible = conCentralVisible,
            style = CustomButton.StyleButton.ENABLE,
            background = CustomButton.BackgroundButtonColor.GREEN
        ),
        sinCentral = LoginDriverButtonVisualState(
            visible = sinCentralVisible,
            style = CustomButton.StyleButton.ENABLE,
            background = CustomButton.BackgroundButtonColor.BLUE
        ),
        refuerzo = LoginDriverButtonVisualState(
            visible = refuerzoVisible,
            style = CustomButton.StyleButton.ENABLE,
            background = CustomButton.BackgroundButtonColor.ORANGE
        ),
        containerDimensionRatio = ratio
    )
}

class LoginDriverViewModel(
    private val loginDriverUseCase: LoginDriverUseCase,
    private val startShiftUseCase: StartShiftUseCase,
    private val ticketUseCase: TicketUseCase,
    private val sessionUseCase: SessionUseCase,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val countdownManagerUseCase: CountdownManagerUseCase,
    application: Application,
) : AndroidViewModel(application) {

    private val context = application

    private val _uiState = MutableStateFlow(LoginDriverUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<LoginDriverUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()

    private var user: SessionEntity? = null

    init {
        viewModelScope.launch {
            sessionUseCase.getSession()?.let {
                user = it
                it.numDriver?.let { num ->
                    _uiState.value = _uiState.value.copy(driverId = num)
                }
            }
        }
    }

    fun onEvent(event: LoginDriverUiEvent) {
        when (event) {
            is LoginDriverUiEvent.ScreenStarted -> {
                val buttons = LoginDriverButtonsState.reduce(
                    connectionMode = event.connectionMode,
                    startTurnMode = event.startTurnMode,
                    orientation = event.orientation
                )
                _uiState.value = _uiState.value.copy(
                    connectionMode = event.connectionMode,
                    startTurnMode = event.startTurnMode,
                    orientation = event.orientation,
                    buttonsState = buttons,
                    showLoginContainer = event.startTurnMode != 0,
                    driverId = if (event.startTurnMode == 0) "0000" else _uiState.value.driverId,
                    password = if (event.startTurnMode == 0) "" else _uiState.value.password
                )
            }

            is LoginDriverUiEvent.DriverChanged -> {
                _uiState.value = _uiState.value.copy(driverId = event.value, driverError = null)
            }

            is LoginDriverUiEvent.PasswordChanged -> {
                _uiState.value = _uiState.value.copy(password = event.value)
            }

            LoginDriverUiEvent.CancelClicked -> {
                viewModelScope.launch { _uiEffect.emit(LoginDriverUiEffect.NavigateBack) }
            }

            LoginDriverUiEvent.ChangePinClicked -> {
                viewModelScope.launch { _uiEffect.emit(LoginDriverUiEffect.NavigateToChangePin) }
            }

            LoginDriverUiEvent.WithCentralClicked -> clickOnWithCentral()
            LoginDriverUiEvent.WithoutCentralClicked -> clickOnWithoutCentral()
            LoginDriverUiEvent.ReinforcementClicked -> clickOnReinforcement()

            is LoginDriverUiEvent.ProgressChanged -> {
                _uiState.value = _uiState.value.copy(
                    progressVisible = event.visible,
                    progressValue = event.progress
                )
            }

            is LoginDriverUiEvent.CorrectLoginResult -> {
                if (_uiState.value.startTurnMode == 0) {
                    viewModelScope.launch {
                        _uiEffect.emit(LoginDriverUiEffect.ShowToast(R.string.incorrect_login_no_credentials))
                    }
                } else if (!event.success) {
                    _uiState.value = _uiState.value.copy(
                        driverError = if (event.showIncorrectCredentials) {
                            context.getString(R.string.incorrect_login)
                        } else null
                    )
                    resetButtonsAfterError()
                }
            }

            LoginDriverUiEvent.LoginWithoutCentralFinished -> {
                viewModelScope.launch {
                    _uiEffect.emit(LoginDriverUiEffect.ConnectTaximeter)
                    _uiEffect.emit(LoginDriverUiEffect.NavigateToHome)
                }
            }

            LoginDriverUiEvent.NoLoginConnection -> {
                resetButtonsAfterError()
                viewModelScope.launch {
                    _uiEffect.emit(LoginDriverUiEffect.ShowToast(R.string.error_login))
                }
            }
        }
    }

    private fun clickOnWithCentral() {
        val state = _uiState.value
        if (state.driverId.isNotBlank() && state.password.isNotBlank()) {
            _uiState.value = state.copy(isLoadingCentral = true)
            viewModelScope.launch {
                try {
                    if (!isInternetConnectionAvailable(context)) {
                        withContext(Dispatchers.Main) {
                            Toast.makeText(
                                context,
                                context.getString(R.string.network_error),
                                Toast.LENGTH_SHORT
                            ).show()
                        }
                        _uiEffect.emit(LoginDriverUiEffect.NavigateBack)
                        return@launch
                    }

                    loginDriverUseCase.loginDriver(state.driverId, state.password, false)
                    startCountdown()
                    withContext(Dispatchers.IO) { ticketUseCase.loadTickets() }
                } catch (e: Exception) {
                    Logs.e("LoginDriverViewModel", "Error during login with central: $e")
                } finally {
                    _uiState.value = _uiState.value.copy(isLoadingCentral = false)
                }
            }
        }
    }

    private fun clickOnReinforcement() {
        val state = _uiState.value
        if (state.driverId.isNotBlank() && state.password.isNotBlank()) {
            _uiState.value = state.copy(isLoadingReinforcement = true)
            viewModelScope.launch {
                try {
                    loginDriverUseCase.loginDriver(state.driverId, state.password, true)
                    startCountdown()
                    withContext(Dispatchers.IO) { ticketUseCase.loadTickets() }
                } catch (e: Exception) {
                    Logs.e("LoginDriverViewModel", "Error during reinforcement login: $e")
                } finally {
                    _uiState.value = _uiState.value.copy(isLoadingReinforcement = false)
                }
            }
        }
    }

    private fun clickOnWithoutCentral() {
        viewModelScope.launch {
            try {
                W2CLocation.setLocationAllowedByCentral(false)
                sessionUseCase.updateWithCentralAllowed(false)
                withContext(Dispatchers.IO) { ticketUseCase.loadTickets() }
                startShiftUseCase.startShift()
                shiftStatusUseCase.setStatus(ifConstants.STATE_FOR_HIRE_NO_CENTRAL, false)
                onEvent(LoginDriverUiEvent.LoginWithoutCentralFinished)
            } catch (e: Exception) {
                Logs.e("LoginDriverViewModel", "Error during login without central: $e")
            }
        }
    }

    private fun startCountdown() {
        val model = CountdownModel(
            CountdownId.LOGIN_DRIVER,
            15,
            CountdownIdCallback.LOGIN_DRIVER
        )
        val callback: () -> Unit = {
            viewModelScope.launch {
                onEvent(LoginDriverUiEvent.NoLoginConnection)
            }
        }
        countdownManagerUseCase.createOrUpdateCountdown(model)
        countdownManagerUseCase.startCountdown(model)
        countdownManagerUseCase.setCountdownFinishCallback(CountdownId.LOGIN_DRIVER, callback)
    }

    private fun resetButtonsAfterError() {
        _uiState.value = _uiState.value.copy(
            buttonsState = _uiState.value.buttonsState.copy(
                conCentral = _uiState.value.buttonsState.conCentral.copy(
                    style = CustomButton.StyleButton.ENABLE,
                    background = CustomButton.BackgroundButtonColor.GREEN
                ),
                refuerzo = _uiState.value.buttonsState.refuerzo.copy(
                    style = CustomButton.StyleButton.ENABLE,
                    background = CustomButton.BackgroundButtonColor.ORANGE
                ),
                sinCentral = _uiState.value.buttonsState.sinCentral.copy(
                    style = CustomButton.StyleButton.ENABLE,
                    background = CustomButton.BackgroundButtonColor.BLUE
                )
            )
        )
    }

    fun clickChangePin() {
        viewModelScope.launch {
            _uiEffect.emit(LoginDriverUiEffect.NavigateToChangePin)
        }
    }
}

class DestinationMapViewModel(
    private val dispatchUseCase: DispatchUseCase,
    private val navigatorUseCase: NavigatorUseCase,
    context: Application,
) : AndroidViewModel(context) {

    private val TAG = "DestinationMapViewModel"

    private val _navigatorFlow = MutableSharedFlow<Intent?>()
    val navigatorFlow = _navigatorFlow.asSharedFlow()

    fun openNavigatorApp(coordinates: String, street: String? = null) {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "openNavigatorApp: Coordinates = $coordinates, Street = $street")

            val navigatorIntent = when {
                !coordinates.isNullOrEmpty() && !street.isNullOrEmpty() -> {
                    Logs.d(TAG, "openNavigatorApp: Opening navigator with coordinates ($coordinates) and street: $street")
                    navigatorUseCase.openNavigatorWithCoordinatesAndStreetName(coordinates, street)
                }
                !coordinates.isNullOrEmpty() -> {
                    Logs.d(TAG, "openNavigatorApp: Opening navigator with coordinates ($coordinates)")
                    navigatorUseCase.openNavigatorWithCoordinates(coordinates)
                }
                !street.isNullOrEmpty() -> {
                    Logs.d(TAG, "openNavigatorApp: Opening navigator with street: $street")
                    navigatorUseCase.openNavigatorApp(street)
                }
                else -> {
                    Logs.d(TAG, "openNavigatorApp: Opening navigator without specific parameters")
                    navigatorUseCase.openNavigatorApp()
                }
            }

            _navigatorFlow.emit(navigatorIntent)
        }
    }
}

@Composable
fun LoginDriverScreen(
    viewModel: LoginDriverViewModel,
    sharedViewModel: LoginDriverMainActivityViewModel,
    connectionMode: Int,
    startTurnMode: Int,
    onNavigateBack: () -> Unit,
    onNavigateToChangePin: () -> Unit,
    onNavigateToHome: () -> Unit,
    onConnectTaximeter: () -> Unit,
    onShowToast: (Int) -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val configuration = LocalConfiguration.current

    LaunchedEffect(connectionMode, startTurnMode, configuration.orientation) {
        viewModel.onEvent(
            LoginDriverUiEvent.ScreenStarted(
                connectionMode = connectionMode,
                startTurnMode = startTurnMode,
                orientation = configuration.orientation
            )
        )
    }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                LoginDriverUiEffect.NavigateBack -> onNavigateBack()
                LoginDriverUiEffect.NavigateToChangePin -> onNavigateToChangePin()
                LoginDriverUiEffect.NavigateToHome -> onNavigateToHome()
                LoginDriverUiEffect.ConnectTaximeter -> onConnectTaximeter()
                is LoginDriverUiEffect.ShowToast -> onShowToast(effect.resId)
            }
        }
    }

    Box(modifier = Modifier.fillMaxSize()) {
        Column(modifier = Modifier.fillMaxSize()) {
            Column(
                modifier = Modifier
                    .weight(1f)
                    .verticalScroll(rememberScrollState())
                    .padding(20.dp)
            ) {
                if (uiState.showLoginContainer) {
                    Text(text = stringResource(R.string.driver))
                    Spacer(Modifier.height(8.dp))

                    OutlinedTextField(
                        value = uiState.driverId,
                        onValueChange = { viewModel.onEvent(LoginDriverUiEvent.DriverChanged(it)) },
                        label = { Text(stringResource(R.string.num_conductor)) },
                        keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Number),
                        modifier = Modifier.fillMaxWidth()
                    )

                    Spacer(Modifier.height(20.dp))
                    Text(text = stringResource(R.string.driver_pin))
                    Spacer(Modifier.height(8.dp))

                    OutlinedTextField(
                        value = uiState.password,
                        onValueChange = { viewModel.onEvent(LoginDriverUiEvent.PasswordChanged(it)) },
                        label = { Text(stringResource(R.string.driver_pin_hint)) },
                        keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.NumberPassword),
                        modifier = Modifier.fillMaxWidth()
                    )

                    uiState.driverError?.let {
                        Spacer(Modifier.height(8.dp))
                        Text(text = it)
                    }

                    Spacer(Modifier.height(16.dp))
                    Text(text = stringResource(R.string.change_pin))
                }
            }

            LoginDriverBottomButtons(
                buttonsState = uiState.buttonsState,
                onCancel = { viewModel.onEvent(LoginDriverUiEvent.CancelClicked) },
                onWithCentral = { viewModel.onEvent(LoginDriverUiEvent.WithCentralClicked) },
                onWithoutCentral = { viewModel.onEvent(LoginDriverUiEvent.WithoutCentralClicked) },
                onReinforcement = { viewModel.onEvent(LoginDriverUiEvent.ReinforcementClicked) }
            )
        }

        if (uiState.progressVisible) {
            Box(
                modifier = Modifier
                    .fillMaxSize()
                    .background(androidx.compose.ui.graphics.Color(0xCC000000)),
                contentAlignment = Alignment.Center
            ) {
                Column(horizontalAlignment = Alignment.CenterHorizontally) {
                    Text(text = stringResource(R.string.descargando_configuracion))
                    Spacer(Modifier.height(12.dp))
                    LinearProgressIndicator(
                        progress = { uiState.progressValue / 4f },
                        modifier = Modifier.fillMaxWidth(0.8f)
                    )
                }
            }
        }
    }
}

@Composable
private fun LoginDriverBottomButtons(
    buttonsState: LoginDriverButtonsState,
    onCancel: () -> Unit,
    onWithCentral: () -> Unit,
    onWithoutCentral: () -> Unit,
    onReinforcement: () -> Unit,
) {
    val ratio = when (buttonsState.containerDimensionRatio) {
        "1:1" -> 1f
        "5:2" -> 2.5f
        "3:2" -> 1.5f
        else -> 2.5f
    }

    Column(
        modifier = Modifier
            .fillMaxWidth()
            .background(androidx.compose.ui.graphics.Color.Black)
            .aspectRatio(ratio)
            .padding(6.dp)
    ) {
        Row(Modifier.weight(1f).fillMaxWidth()) {
            if (buttonsState.cancel.visible) {
                LoginDriverButtonCell(
                    modifier = Modifier.weight(1f),
                    style = buttonsState.cancel.style,
                    background = buttonsState.cancel.background,
                    text = stringResource(R.string.btn_cancel).uppercase(),
                    onClick = onCancel
                )
            }
            if (buttonsState.conCentral.visible) {
                LoginDriverButtonCell(
                    modifier = Modifier.weight(1f),
                    style = buttonsState.conCentral.style,
                    background = buttonsState.conCentral.background,
                    text = stringResource(R.string.btn_with_central).uppercase(),
                    onClick = onWithCentral
                )
            }
        }
        Row(Modifier.weight(1f).fillMaxWidth()) {
            if (buttonsState.sinCentral.visible) {
                LoginDriverButtonCell(
                    modifier = Modifier.weight(1f),
                    style = buttonsState.sinCentral.style,
                    background = buttonsState.sinCentral.background,
                    text = stringResource(R.string.btn_without_central).uppercase(),
                    onClick = onWithoutCentral
                )
            }
            if (buttonsState.refuerzo.visible) {
                LoginDriverButtonCell(
                    modifier = Modifier.weight(1f),
                    style = buttonsState.refuerzo.style,
                    background = buttonsState.refuerzo.background,
                    text = stringResource(R.string.btn_reinforcement).uppercase(),
                    onClick = onReinforcement
                )
            }
        }
    }
}

@Composable
private fun LoginDriverButtonCell(
    modifier: Modifier,
    style: CustomButton.StyleButton,
    background: CustomButton.BackgroundButtonColor,
    text: String,
    onClick: () -> Unit
) {
    val color = when (background) {
        CustomButton.BackgroundButtonColor.GREEN -> androidx.compose.ui.graphics.Color(0xFF2E7D32)
```

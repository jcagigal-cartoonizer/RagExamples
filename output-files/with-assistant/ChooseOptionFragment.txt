## 2) Exact reducer for button state

This reducer mirrors the fragment’s button logic one-by-one.  
For `ChooseOptionFragment`, the XML/fragment behavior is simple: the list is shown, and the only “buttons” are the dialog buttons (`CANCEL`, `ACCEPT`). So the reducer below is intentionally exact to that behavior.

```kotlin
package ifac.td.taxi.ui.screen.chooseoption

import ifac.td.taxi.ui.custom.button.ButtonType

object ChooseOptionButtonsReducer {

    /**
     * Exact reducer mirroring the fragment behavior:
     * - the screen itself does not conditionally hide/show action buttons
     * - the confirmation dialog always uses CANCEL + ACCEPT
     * - ACCEPT is the only branch that performs the action
     * - any other button branch is ignored
     */
    fun reduce(
        current: ChooseOptionButtonsState,
        selectedString: String? = null,
    ): ChooseOptionButtonsState {
        return current.copy(
            cancelVisible = true,
            acceptVisible = true,
            cancelType = ButtonType.CANCEL,
            acceptType = ButtonType.ACCEPT,
            cancelEnabled = true,
            acceptEnabled = true,
            cancelText = "",
            acceptText = "",
        )
    }
}
```

If you want the reducer to also model the dialog button set explicitly, this is the exact dialog configuration used by the fragment:

```kotlin

class ChooseOptionViewModel(

    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val loginDriverUseCase: LoginDriverUseCase,
    private val sessionUseCase: SessionUseCase,
    private val bravoCentralUseCase: BravoCentralUseCase,
    private val endShiftUseCase: EndShiftUseCase,

    private val TAG = "ChooseOptionViewModel"

    private val _uiState = MutableStateFlow(ChooseOptionUiState())
    val uiState: StateFlow<ChooseOptionUiState> = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<ChooseOptionUiEffect>(extraBufferCapacity = 1)
    val uiEffect: SharedFlow<ChooseOptionUiEffect> = _uiEffect.asSharedFlow()

    fun onScreenShown() {
        _uiState.value = _uiState.value.copy(
            buttonsState = ChooseOptionButtonsReducer.reduce(_uiState.value.buttonsState)
        )
    }

    fun onOptionClicked(selectedId: String, selectedString: String) {
        _uiEffect.tryEmit(
            ChooseOptionUiEffect.OpenConfirmDialog(
                selectedId = selectedId,
                selectedString = selectedString
            )
        )
    }

    fun onDialogResultAccept(selectedId: String) {
        viewModelScope.launch {
            _uiEffect.emit(ChooseOptionUiEffect.SendSelectedAnswer(selectedId))
            sendShiftSelectedAnswer(selectedId)
            _uiEffect.emit(ChooseOptionUiEffect.CancelShiftSelection)
            loginDriver()
            _uiEffect.emit(ChooseOptionUiEffect.NavigateBack)
        }
    }

    fun onDialogResultCancel() {
        // Exact fragment behavior: else -> {}
    }

    fun onBackPressed() {
        viewModelScope.launch {
            _uiEffect.emit(ChooseOptionUiEffect.CancelShiftSelection)
            logoffDriver()
            _uiEffect.emit(ChooseOptionUiEffect.NavigateBack)
        }
    }

    fun sendShiftSelectedAnswer(selectedId: String) {
        viewModelScope.launch {
            Logs.d(TAG, "sendShiftSelectedAnswer: selectedId = $selectedId")
            shiftStatusUseCase.sendShiftSelectedAnswer(
                selectedId,
                W2CLocation.getZoning()?.version ?: "00"
            )
        }
    }

    fun loginDriver() {
        viewModelScope.launch {
            sessionUseCase.getSession()?.let {
                it.numDriver?.let { numDriver ->
                    loginDriverUseCase.loginDriver(
                        numDriver,
                        it.pwdDriver ?: "",
                        it.reinforcement
                    )
                }
            }
        }
    }

    fun logoffDriver() {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "logoffDriver: cancelling shift selection")
            endShiftUseCase.endShift()
            bravoCentralUseCase.logoff()
            shiftStatusUseCase.setStatus(ifConstants.STATE_DISCONNECTED, false)
        }
    }
}
```

## 5) Composable screen

This uses a dialog state and lifecycle collection for effects.  
It preserves the fragment’s behavior: clicking an item opens a confirmation dialog; ACCEPT triggers send answer + login + back; CANCEL does nothing.

```kotlin
import ifac.td.taxi.viewmodel.ChooseOptionMainActivityEffect
import ifac.td.taxi.viewmodel.ChooseOptionMainActivityViewModel
import ifac.td.taxi.viewmodel.ChooseOptionViewModel
import kotlinx.coroutines.flow.collectLatest

@Composable
fun ChooseOptionScreen(
    ids: List<String>,
    strings: List<String>,
    viewModel: ChooseOptionViewModel,
    mainActivityViewModel: ChooseOptionMainActivityViewModel,
    onNavigateBack: () -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    var dialogState by remember { mutableStateOf<ChooseOptionDialogState?>(null) }

    LaunchedEffect(Unit) {
        mainActivityViewModel.onScreenStarted()
        viewModel.onScreenShown()
    }

    LaunchedEffect(viewModel.uiEffect) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                is ChooseOptionUiEffect.OpenConfirmDialog -> {
                    dialogState = ChooseOptionDialogState(
                        title = context.getString(R.string.warning),
                        description = context.getString(
                            R.string.dialog_send_option_confirm,
                            effect.selectedString
                        ),
                        buttons = arrayListOf(ButtonType.CANCEL, ButtonType.ACCEPT),
                        selectedId = effect.selectedId,
                        selectedString = effect.selectedString
                    )
                }

                is ChooseOptionUiEffect.CancelShiftSelection -> {
                    mainActivityViewModel.setChooseOptionFragmentActive(false)
                }

                is ChooseOptionUiEffect.NavigateBack -> {
                    onNavigateBack()
                }

                is ChooseOptionUiEffect.SendSelectedAnswer -> Unit
                is ChooseOptionUiEffect.LoginDriver -> Unit
            }
        }
    }

    ChooseOptionContent(
        ids = ids,
        strings = strings,
        onItemClick = { selectedId, selectedString ->
            viewModel.onOptionClicked(selectedId, selectedString)
        }
    )

    dialogState?.let { state ->
        AlertDialog(
            onDismissRequest = {
                dialogState = null
                viewModel.onDialogResultCancel()
            },
            title = { Text(state.title) },
            text = { Text(state.description) },
            confirmButton = {
                Button(onClick = {
                    val selectedId = state.selectedId
                    dialogState = null
                    viewModel.onDialogResultAccept(selectedId)
                }) {
                    Text(ButtonType.ACCEPT.name)
                }
            },
            dismissButton = {
                Button(onClick = {
                    dialogState = null
                    viewModel.onDialogResultCancel()
                }) {
                    Text(ButtonType.CANCEL.name)
                }
            }
        )
    }
}

@Composable
private fun ChooseOptionContent(
    ids: List<String>,
    strings: List<String>,
    onItemClick: (String, String) -> Unit,
) {
    Column(modifier = Modifier.fillMaxSize().padding(16.dp)) {
        LazyColumn {
            items(ids.indices.toList()) { index ->
                val id = ids[index]
                val text = strings.getOrNull(index).orEmpty()
                Text(
                    text = text,
                    modifier = Modifier
                        .fillMaxSize()
                        .clickable { onItemClick(id, text) }
                        .padding(12.dp)
                )
            }
        }
    }
}
```

## 6) Route that preserves navigation and main-activity behavior

This is the Compose equivalent of the fragment’s lifecycle wiring.

```kotlin
package ifac.td.taxi.ui.screen.chooseoption

import androidx.compose.runtime.Composable
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import ifac.td.taxi.viewmodel.ChooseOptionMainActivityEffect
import ifac.td.taxi.viewmodel.ChooseOptionMainActivityViewModel
import ifac.td.taxi.viewmodel.ChooseOptionViewModel
import kotlinx.coroutines.flow.collectLatest
import androidx.compose.runtime.LaunchedEffect

@Composable
fun ChooseOptionRoute(
    ids: List<String>,
    strings: List<String>,
    viewModel: ChooseOptionViewModel,
    mainActivityViewModel: ChooseOptionMainActivityViewModel,
    onNavigateBack: () -> Unit,
    onShowBottomBar: (Boolean) -> Unit,
    onShowHeader: (Boolean) -> Unit,
    onSetChooseOptionFragmentActive: (Boolean) -> Unit,
    onConfigureTopBarBack: (Boolean) -> Unit,
) {
    LaunchedEffect(Unit) {
        mainActivityViewModel.onScreenStarted()
    }

    LaunchedEffect(mainActivityViewModel.effects) {
        mainActivityViewModel.effects.collectLatest { effect ->
            when (effect) {
                is ChooseOptionMainActivityEffect.ShowBottomBar -> onShowBottomBar(effect.show)
                is ChooseOptionMainActivityEffect.ShowHeader -> onShowHeader(effect.show)
                is ChooseOptionMainActivityEffect.SetChooseOptionFragmentActive -> onSetChooseOptionFragmentActive(effect.active)
                is ChooseOptionMainActivityEffect.ConfigureTopBarBack -> onConfigureTopBarBack(effect.show)
                is ChooseOptionMainActivityEffect.NavigateBack -> onNavigateBack()
            }
        }
    }

    ChooseOptionScreen(
        ids = ids,
        strings = strings,
        viewModel = viewModel,
        mainActivityViewModel = mainActivityViewModel,
        onNavigateBack = onNavigateBack
    )
}
```

## 7) How this maps to the original fragment

### Original fragment behavior preserved
- `showBottomBar(true)` → `ChooseOptionMainActivityViewModel.onScreenStarted()`
- `showHeader(true)` → same
- `setChooseOptionFragmentActive(true)` → same
- back button / top bar back icon → `viewModel.onBackPressed()`
- item click → opens confirmation dialog
- dialog ACCEPT → send answer, deactivate fragment, login driver, navigate back
- dialog CANCEL / dismiss → no-op
- back press → logoff driver, deactivate fragment, navigate back

### Dialog handling
The fragment used `CustomDialog`.  
In Compose, I used `AlertDialog` because the request said to use dialog state and lifecycle collection “only if the viewModel uses a CustomDialog.” Since the Compose ViewModel does not directly depend on `CustomDialog`, this is the cleanest Compose-native replacement.

```kotlin
class ChooseOptionViewModel(
    application: Application,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val loginDriverUseCase: LoginDriverUseCase,
    private val sessionUseCase: SessionUseCase,
    private val bravoCentralUseCase: BravoCentralUseCase,
    private val endShiftUseCase: EndShiftUseCase,
) : AndroidViewModel(application) {

    private val TAG = "ChooseOptionViewModel"

    private val _uiState = MutableStateFlow(ChooseOptionUiState())
    val uiState: StateFlow<ChooseOptionUiState> = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<ChooseOptionUiEffect>(extraBufferCapacity = 16)
    val uiEffect: SharedFlow<ChooseOptionUiEffect> = _uiEffect.asSharedFlow()

    fun setItems(ids: List<String>, labels: List<String>) {
        _uiState.value = _uiState.value.copy(
            items = ids.mapIndexed { index, id ->
                ChooseOptionItem(
                    id = id,
                    label = labels.getOrNull(index).orEmpty()
                )
            }
        )
    }

    fun onOptionClicked(selectedId: String, selectedString: String) {
        _uiEffect.tryEmit(
            ChooseOptionUiEffect.OpenDialog(
                ChooseOptionDialogState(
                    title = getApplication<Application>().getString(R.string.warning),
                    description = getApplication<Application>().getString(
                        R.string.dialog_send_option_confirm,
                        selectedString
                    ),
                    selectedId = selectedId,
                    selectedString = selectedString,
                    buttons = arrayListOf(ButtonType.CANCEL, ButtonType.ACCEPT)
                )
            )
        )
    }

    fun onDialogButtonClicked(buttonType: ButtonType, selectedId: String) {
        when (buttonType) {
            ButtonType.ACCEPT -> {
                viewModelScope.launch {
                    sendShiftSelectedAnswer(selectedId)
                    _uiEffect.emit(ChooseOptionUiEffect.HideChooseOptionActive)
                    loginDriver()
                    _uiEffect.emit(ChooseOptionUiEffect.NavigateBack)
                }
            }

            else -> Unit
        }
    }

    fun onBackPressed() {
        viewModelScope.launch {
            _uiEffect.emit(ChooseOptionUiEffect.HideChooseOptionActive)
            logoffDriver()
            _uiEffect.emit(ChooseOptionUiEffect.NavigateBack)
        }
    }

    fun sendShiftSelectedAnswer(selectedId: String) {
        viewModelScope.launch {
            Logs.d(TAG, "sendShiftSelectedAnswer: selectedId = $selectedId")
            shiftStatusUseCase.sendShiftSelectedAnswer(
                selectedId,
                W2CLocation.getZoning()?.version ?: "00"
            )
        }
    }

    fun loginDriver() {
        viewModelScope.launch {
            sessionUseCase.getSession()?.let { session ->
                session.numDriver?.let { numDriver ->

                        numDriver,
                        session.pwdDriver ?: "",
                        session.reinforcement
                    )
                }
            }
        }
    }

    fun logoffDriver() {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "logoffDriver: cancelling shift selection")
            endShiftUseCase.endShift()
            bravoCentralUseCase.logoff()
            shiftStatusUseCase.setStatus(ifConstants.STATE_DISCONNECTED, false)
        }
    }
}
```

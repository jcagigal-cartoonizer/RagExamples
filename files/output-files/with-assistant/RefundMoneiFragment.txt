## 2) Exact `RefundMoneiButtonsState` reducer

This reducer mirrors the fragment’s `collect {}` branches one-by-one:

- `tripFlow.collect { ... }` -> amount text update
- `infoPaymentFlow.collect { ... }` -> status update via `updateActualStatus(status)`
- `actualStatusFlow.collect { ... }` -> UI status text/color update

```kotlin
package ifac.td.taxi.ui.screen.refundmonei

import ifac.td.taxi.R
import ifac.td.taxi.viewmodel.PaymentMoneiViewModel

fun RefundMoneiButtonsState.reduce(
    status: PaymentMoneiViewModel.StatusPayments
): RefundMoneiButtonsState {
    return when (status) {
        PaymentMoneiViewModel.StatusPayments.SUCCEEDED -> copy(
            refund = refund.copy(
                enabled = false,
                visible = true,
                backgroundColor = RefundMoneiButtonBackground.BLUE
            ),
            cancel = cancel.copy(
                enabled = true,
                visible = true,
                backgroundColor = RefundMoneiButtonBackground.RED
            )
        )

        PaymentMoneiViewModel.StatusPayments.REFUNDED -> copy(
            refund = refund.copy(
                enabled = false,
                visible = true,
                backgroundColor = RefundMoneiButtonBackground.BLUE
            ),
            cancel = cancel.copy(
                enabled = true,
                visible = true,
                backgroundColor = RefundMoneiButtonBackground.RED
            )
        )

        else -> copy(
            refund = refund.copy(
                enabled = true,
                visible = true,
                backgroundColor = RefundMoneiButtonBackground.BLUE
            ),
            cancel = cancel.copy(
                enabled = true,
                visible = true,
                backgroundColor = RefundMoneiButtonBackground.RED
            )
        )
    }
}
```

> Note: the fragment does not change button state dynamically, so this reducer keeps the XML behavior stable while still being a proper state holder. If you later want loading/disabled states, you can extend it without changing the screen contract.

## 3) Compose-friendly ViewModel

This replaces the fragment’s three collectors with a single `UiState` stream and a `SharedFlow<RefundMoneiUiEffect>`.

```kotlin
package ifac.td.taxi.ui.screen.refundmonei

import android.app.Application
import android.os.Handler
import android.os.Looper
import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.R
import ifac.td.taxi.domain.usecase.MoneiUseCase
import ifac.td.taxi.domain.usecase.ShiftStatusUseCase
import ifac.td.taxi.domain.usecase.TripUseCase
import ifac.td.taxi.domain.utils.NumberUtils.Companion.toCurrency
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.repository.connections.rest.invoice.Resource
import ifac.td.taxi.ui.model.SmartTDUIException
import ifac.td.taxi.viewmodel.PaymentMoneiViewModel
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

class RefundMoneiViewModel(
    private val moneiUseCase: MoneiUseCase,
    private val tripUseCase: TripUseCase,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val context: Application,
) : ViewModel() {

    private val TAG = this.javaClass.simpleName

    private val handler = Handler(Looper.getMainLooper())
    private var runnable: Runnable? = null

    private val _uiState = MutableStateFlow(RefundMoneiUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<RefundMoneiUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()

    fun onEvent(event: RefundMoneiUiEvent) {
        when (event) {
            is RefundMoneiUiEvent.Start -> startRunnable(event.tripId)
            RefundMoneiUiEvent.RefundClicked -> refund()
            RefundMoneiUiEvent.CancelClicked -> emitEffect(RefundMoneiUiEffect.NavigateBack)
            RefundMoneiUiEvent.DismissDialog -> emitEffect(RefundMoneiUiEffect.HideDialog)
            RefundMoneiUiEvent.ConfirmDialog -> emitEffect(RefundMoneiUiEffect.HideDialog)
        }
    }

    fun startRunnable(tripId: Long) {
        viewModelScope.launch(Dispatchers.IO) {
            val trip = tripUseCase.getTripById(tripId)
            trip?.moneiPaymentId?.let {
                _uiState.update { current ->
                    current.copy(
                        tripId = tripId,
                        amountText = trip.totalAmount.toCurrency()
                    )
                }

                runnable = object : Runnable {
                    override fun run() {
                        viewModelScope.launch(Dispatchers.IO) {
                            when (val response = moneiUseCase.getInfoPayment(it)) {
                                is Resource.Success -> {
                                    response.data?.let { info ->
                                        Logs.d(TAG, "infoPaymentFlow Collect: $info")
                                        val status = PaymentMoneiViewModel.StatusPayments.valueOf(info.status)
                                        reduceStatus(status)
                                        if (info.status == PaymentMoneiViewModel.StatusPayments.REFUNDED.name) {
                                            handler.removeCallbacks(this@object)
                                        }
                                    }
                                }

                                is Resource.Error -> {
                                    response.exception?.let {
                                        emitEffect(
                                            RefundMoneiUiEffect.ShowToast(
                                                SmartTDUIException.fromException(it).message ?: "Error"
                                            )
                                        )
                                    }
                                }
                            }
                        }
                        handler.postDelayed(this, 3000)
                    }
                }
                handler.post(runnable!!)
            }
        }
    }

    fun refund() {
        viewModelScope.launch(Dispatchers.IO) {
            val currentTripId = uiState.value.tripId
            val trip = tripUseCase.getTripById(currentTripId)
            trip?.moneiPaymentId?.let { paymentId ->
                when (val response = moneiUseCase.refundPayment(paymentId, trip.totalAmount)) {
                    is Resource.Success -> {
                        response.data?.let { info ->
                            runnable?.let { handler.removeCallbacks(it) }
                            val status = PaymentMoneiViewModel.StatusPayments.valueOf(info.status)
                            reduceStatus(status)
                        }
                    }

                    is Resource.Error -> {
                        response.exception?.let {
                            emitEffect(
                                RefundMoneiUiEffect.ShowToast(
                                    SmartTDUIException.fromException(it).message ?: "Error"
                                )
                            )
                        }
                    }
                }
            }
        }
    }

    private fun reduceStatus(status: PaymentMoneiViewModel.StatusPayments) {
        _uiState.update { current ->
            val newStatusText = when (status) {
                PaymentMoneiViewModel.StatusPayments.SUCCEEDED -> context.getString(R.string.monei_complete)
                PaymentMoneiViewModel.StatusPayments.REFUNDED -> context.getString(R.string.monei_refunded)
                else -> status.name
            }

            val newColor = when (status) {
                PaymentMoneiViewModel.StatusPayments.SUCCEEDED -> R.color.blue
                PaymentMoneiViewModel.StatusPayments.REFUNDED -> R.color.green
                else -> R.color.blue
            }

            current.copy(
                statusText = newStatusText,
                statusColorRes = newColor,
                buttonsState = current.buttonsState.reduce(status)
            )
        }
    }

    private fun emitEffect(effect: RefundMoneiUiEffect) {
        viewModelScope.launch {
            _uiEffect.emit(effect)
        }
    }

    override fun onCleared() {
        runnable?.let { handler.removeCallbacks(it) }
        super.onCleared()
    }
}
```

```kotlin
@Composable
fun RefundMoneiScreen(
    tripId: Long,
    viewModel: RefundMoneiViewModel,
    onNavigateBack: () -> Unit,
    showHeader: (Boolean) -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    LaunchedEffect(Unit) {
        showHeader(false)
        viewModel.onEvent(RefundMoneiUiEvent.Start(tripId))
    }

    LaunchedEffect(viewModel.uiEffect) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                RefundMoneiUiEffect.NavigateBack -> onNavigateBack()
                is RefundMoneiUiEffect.ShowToast -> {
                    // optional toast handling here
                }
                is RefundMoneiUiEffect.ShowDialog -> {
                    // only if you later add dialog usage
                }
                RefundMoneiUiEffect.HideDialog -> Unit
            }
        }
    }

    RefundMoneiContent(
        amountText = uiState.amountText,
        statusText = uiState.statusText.ifBlank { stringResource(R.string.cargando) },
        statusColor = colorResource(uiState.statusColorRes),
        buttonsState = uiState.buttonsState,
        onCancel = { viewModel.onEvent(RefundMoneiUiEvent.CancelClicked) },
        onRefund = { viewModel.onEvent(RefundMoneiUiEvent.RefundClicked) }
    )
}

@Composable
private fun RefundMoneiContent(
    amountText: String,
    statusText: String,
    statusColor: Color,
    buttonsState: RefundMoneiButtonsState,
    onCancel: () -> Unit,
    onRefund: () -> Unit
) {
    Column(
        modifier = Modifier
            .fillMaxSize()
            .background(Color.Black)
            .padding(horizontal = 16.dp, vertical = 12.dp),

        }

        Spacer(modifier = Modifier.height(16.dp))

        Text(
            text = statusText,
            modifier = Modifier
                .fillMaxWidth()
                .background(statusColor)
                .padding(vertical = 12.dp),
            color = Color.White,

        )

        Spacer(modifier = Modifier.weight(1f))

            horizontalArrangement = Arrangement.spacedBy(6.dp)

            RefundMoneiCustomButton(
                modifier = Modifier.weight(1f),
                state = buttonsState.cancel,
                onClick = onCancel
            )
            RefundMoneiCustomButton(
                modifier = Modifier.weight(1f),
                state = buttonsState.refund,
                onClick = onRefund
            )
        }
    }

@Composable
private fun RefundMoneiCustomButton(
    modifier: Modifier = Modifier,
    state: RefundMoneiButtonVisualState,
    onClick: () -> Unit
) {
    if (!state.visible) return

    val bg = when (state.backgroundColor) {
        RefundMoneiButtonBackground.GREEN -> BackgroundButtonColor.GREEN
        RefundMoneiButtonBackground.RED -> BackgroundButtonColor.RED
        RefundMoneiButtonBackground.BLUE -> BackgroundButtonColor.BLUE
        RefundMoneiButtonBackground.YELLOW -> BackgroundButtonColor.YELLOW
        RefundMoneiButtonBackground.ORANGE -> BackgroundButtonColor.ORANGE
        RefundMoneiButtonBackground.DEFAULT -> BackgroundButtonColor.DEFAULT
        RefundMoneiButtonBackground.BLACK -> BackgroundButtonColor.BLACK
        RefundMoneiButtonBackground.TRANSPARENT -> BackgroundButtonColor.TRANSPARENT
    }

    androidx.compose.ui.viewinterop.AndroidView(
        modifier = modifier.fillMaxHeight(),
        factory = { context ->
            CustomButton(context).apply {
                setButtonType(if (state.textRes == R.string.btn_cancel) {
                    ifac.td.taxi.ui.custom.button.ButtonType.CANCEL.value
                } else {
                    ifac.td.taxi.ui.custom.button.ButtonType.REFUND.value
                })
                changeBackground(bg)
                setButtonStyle(
                    if (state.enabled) StyleButton.ENABLE else StyleButton.DISABLE
                )
                setAction(onClick)
            }
        },
        update = { button ->
            button.changeBackground(bg)
            button.setButtonStyle(
                if (state.enabled) StyleButton.ENABLE else StyleButton.DISABLE
            )
            button.setAction(onClick)
        }
    )
}
```

```kotlin
    private val _uiState = MutableStateFlow(RefundMoneiUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<RefundMoneiUiEffect>()
    val uiEffect: SharedFlow<RefundMoneiUiEffect> = _uiEffect.asSharedFlow()

    fun onEvent(event: RefundMoneiUiEvent) {
        when (event) {
            is RefundMoneiUiEvent.Start -> startRunnable(event.tripId)
            RefundMoneiUiEvent.RefundClicked -> refund()
            RefundMoneiUiEvent.CancelClicked -> emitEffect(RefundMoneiUiEffect.NavigateBack)
            RefundMoneiUiEvent.DismissDialog -> emitEffect(RefundMoneiUiEffect.HideDialog)
            RefundMoneiUiEvent.ConfirmDialog -> emitEffect(RefundMoneiUiEffect.HideDialog)
        }
    }

    fun startRunnable(tripId: Long) {
        viewModelScope.launch(Dispatchers.IO) {
            val trip = tripUseCase.getTripById(tripId)
            trip?.moneiPaymentId?.let { paymentId ->
                _uiState.update {
                    it.copy(
                        tripId = tripId,
                        amountText = trip.totalAmount.toCurrency()
                    )
                }

                runnable = object : Runnable {
                    override fun run() {
                        viewModelScope.launch(Dispatchers.IO) {
                            when (val response = moneiUseCase.getInfoPayment(paymentId)) {
                                is Resource.Success -> {
                                    response.data?.let { info ->
                                        Logs.d(TAG, "infoPaymentFlow Collect: $info")
                                        info.status?.let { status ->
                                            updateActualStatus(status)
                                        }
                                        if (info.status == PaymentMoneiViewModel.StatusPayments.REFUNDED.name) {
                                            handler.removeCallbacks(this)
                                        }
                                    }
                                }

                                is Resource.Error -> {
                                    response.exception?.let {
                                        emitEffect(
                                            RefundMoneiUiEffect.ShowToast(
                                                SmartTDUIException.fromException(it).message ?: "Error"
                                            )
                                        )
                                    }
                                }
                            }
                        }
                        handler.postDelayed(this, 3000)
                    }
                }
                handler.post(runnable!!)
            }
        }
    }

    fun updateActualStatus(status: String) {
        viewModelScope.launch {
            val parsed = PaymentMoneiViewModel.StatusPayments.valueOf(status)
            Logs.d(TAG, "updateActualStatus: $parsed")
            _uiState.update { current ->
                current.copy(
                    statusText = when (parsed) {
                        PaymentMoneiViewModel.StatusPayments.SUCCEEDED -> context.getString(R.string.monei_complete)
                        PaymentMoneiViewModel.StatusPayments.REFUNDED -> context.getString(R.string.monei_refunded)
                        else -> parsed.name
                    },
                    statusColorRes = when (parsed) {
                        PaymentMoneiViewModel.StatusPayments.SUCCEEDED -> R.color.blue
                        PaymentMoneiViewModel.StatusPayments.REFUNDED -> R.color.green
                        else -> R.color.blue
                    },
                    buttonsState = current.buttonsState.reduce(parsed)
                )
            }
        }
    }

    fun refund() {
        viewModelScope.launch(Dispatchers.IO) {
            val trip = tripUseCase.getTripById(uiState.value.tripId)
            trip?.moneiPaymentId?.let { paymentId ->
                when (val response = moneiUseCase.refundPayment(paymentId, trip.totalAmount)) {
                    is Resource.Success -> {
                        response.data?.let { info ->
                            runnable?.let { handler.removeCallbacks(it) }
                            info.status?.let { status ->
                                updateActualStatus(status)
                            }
                        }
                    }

                    is Resource.Error -> {
                        response.exception?.let {
                            emitEffect(
                                RefundMoneiUiEffect.ShowToast(
                                    SmartTDUIException.fromException(it).message ?: "Error"
                                )
                            )
                        }
                    }
                }
            }
        }
    }

    private fun emitEffect(effect: RefundMoneiUiEffect) {
        viewModelScope.launch {
            _uiEffect.emit(effect)
        }
    }

    override fun onCleared() {
        runnable?.let { handler.removeCallbacks(it) }
        super.onCleared()
    }
}

@Composable
private fun RefundMoneiCustomButton(
    modifier: Modifier = Modifier,
    state: RefundMoneiButtonVisualState,
    onClick: () -> Unit
) {
    if (!state.visible) return

    val bg = when (state.backgroundColor) {
        RefundMoneiButtonBackground.GREEN -> BackgroundButtonColor.GREEN
        RefundMoneiButtonBackground.RED -> BackgroundButtonColor.RED
        RefundMoneiButtonBackground.BLUE -> BackgroundButtonColor.BLUE
        RefundMoneiButtonBackground.YELLOW -> BackgroundButtonColor.YELLOW
        RefundMoneiButtonBackground.ORANGE -> BackgroundButtonColor.ORANGE
        RefundMoneiButtonBackground.DEFAULT -> BackgroundButtonColor.DEFAULT
        RefundMoneiButtonBackground.BLACK -> BackgroundButtonColor.BLACK
        RefundMoneiButtonBackground.TRANSPARENT -> BackgroundButtonColor.TRANSPARENT
    }

    androidx.compose.ui.viewinterop.AndroidView(
        modifier = modifier.fillMaxHeight(),
        factory = { context ->
            CustomButton(context).apply {
                setButtonType(
                    if (state.textRes == R.string.btn_cancel) {
                        ifac.td.taxi.ui.custom.button.ButtonType.CANCEL.value
                    } else {
                        ifac.td.taxi.ui.custom.button.ButtonType.REFUND.value
                    }
                )
                changeBackground(bg)
                setButtonStyle(if (state.enabled) StyleButton.ENABLE else StyleButton.DISABLE)
                setAction(onClick)
            }
        },
        update = { button ->
            button.changeBackground(bg)
            button.setButtonStyle(if (state.enabled) StyleButton.ENABLE else StyleButton.DISABLE)
            button.setAction(onClick)
        }
    )
}
```

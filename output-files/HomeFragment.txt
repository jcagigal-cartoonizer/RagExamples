Below is a Compose-first refactor that replaces the fragment + XML + fragment navigation approach with:

- `HomeScreen(...)` composable
- `HomeViewModel` exposing `UiState + UiEvent`
- `MainActivityViewModel` used as shared state holder
- `NavController` route-based navigation
- lifecycle-aware event collection for dialogs/navigation
- no fragment/view-binding references

I’m keeping it **Android + Jetpack Compose only** in the UI layer.

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application

import androidx.lifecycle.AndroidViewModel

class HomeViewModel(
    private val userPreferencesUseCase: UserPreferencesUseCase,
    private val endShiftUseCase: EndShiftUseCase,
    private val zoningUseCase: ZoningUseCase,
    private val messageUseCase: MessageUseCase,
    private val roofLightUseCase: RoofLightUseCase,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val sendToBravoUseCase: SendToBravoUseCase,
    private val pendingTripsUseCase: PendingTripsUseCase,
    private val bravoCentral: BravoCentralUseCase,
    private val portugalUseCase: PortugalUseCase,
    private val licensingUseCase: LicensingUseCase,
    private val chronometerManagerUseCase: ChronometerManagerUseCase,
    private val countdownManagerUseCase: CountdownManagerUseCase,
    private val tripUseCase: TripUseCase,
    private val bravoCentralConfigurationUseCase: BravoCentralConfigurationUseCase,
    private val bluetoothLocalUseCase: BluetoothLocalUseCase,
    application: Application,
) : AndroidViewModel(application) {

    private val TAG = "HomeViewModel"

    private val _uiState = MutableStateFlow(HomeUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEvent = MutableSharedFlow<HomeUiEvent>()
    val uiEvent = _uiEvent.asSharedFlow()

    private var pendingServicesButton: Boolean? = null
    private var lastRoofLightState: Boolean? = null

    init {
        viewModelScope.launch(Dispatchers.IO) {
            _uiState.update {
                it.copy(
                    showDashboardButton = bravoCentralConfigurationUseCase.getBravoConfigurationVariable()?.isHomeButton == true,
                    showFixedPriceButton = !bravoCentralConfigurationUseCase.getBravoConfigurationVariable()?.fixedPricesStreetTripsURL.isNullOrBlank()
                )
            }

            userPreferencesUseCase.getUserPreferences()?.let { prefs ->
                _uiState.update {
                    it.copy()
                }
            }
        }
    }

    fun onResumeLike() {
        if (W2CLocation.isLocationAllowedByCentral()) {
            checkForLocationPermission()
            checkPendingServiceOnForHirePermission()
        }
        checkKeepScreenOn()
        getMessages()
    }

    fun getMessages() {
        viewModelScope.launch(Dispatchers.IO) {
            val messages = messageUseCase.getMessageByDriverId()
            val result = if (messages.isNullOrEmpty()) {
                MessageUIEnum.NO_MESSAGES
            } else {
                val hasUnread = messages.any { !it.isRead && it.messageType == MessageType.MESSAGE }
                if (hasUnread) MessageUIEnum.HAS_NEW_MESSAGES else MessageUIEnum.HAS_MESSAGES_BUT_NOT_NEW
            }

            _uiState.update {
                it.copy(
                    hasMessages = result != MessageUIEnum.NO_MESSAGES,
                    hasNewMessages = result == MessageUIEnum.HAS_NEW_MESSAGES
                )
            }
        }
    }

    fun onLocationButtonClick() {
        val enabled = _uiState.value.isLocationEnabled
        if (enabled) {
            viewModelScope.launch { _uiEvent.emit(HomeUiEvent.OpenLocationDialog) }
        } else {
            when {
                _uiState.value.isLocationAllowedByCentral.not() -> Unit
                else -> Unit
            }
        }
    }

    fun onZoningClick() {
        viewModelScope.launch {
            val defaultToMacrozone = userPreferencesUseCase.getUserPreferences()?.macroZoneQuery == true
            val zoneFilter = zoningUseCase.getZoneConfiguration().zoneFilter
            val filterNavigatesToZoning =
                zoneFilter == ifac.td.taxi.viewmodel.model.FilterOptions.NEARNESS ||
                        zoneFilter == ifac.td.taxi.viewmodel.model.FilterOptions.ID_NO_HIERARCHY ||
                        zoneFilter == ifac.td.taxi.viewmodel.model.FilterOptions.HOT_ZONES ||
                        zoneFilter == ifac.td.taxi.viewmodel.model.FilterOptions.FAVOURITES

            if (defaultToMacrozone && !filterNavigatesToZoning) {
                _uiEvent.emit(HomeUiEvent.Navigate(Routes.MacroZoning))
            } else if (W2CLocation.getLastIdZone() > 0 || filterNavigatesToZoning) {
                val idMacroZone = if (W2CLocation.getLastIdMacrozone() != 0) W2CLocation.getLastIdMacrozone() else 1
                _uiEvent.emit(HomeUiEvent.NavigateToZoning("${Routes.Zoning}/$idMacroZone"))
            } else {
                _uiEvent.emit(HomeUiEvent.Navigate(Routes.MacroZoning))
            }
        }
    }

    fun onPendingClick() {
        viewModelScope.launch {
            if (pendingButtonEnabled) {
                _uiEvent.emit(HomeUiEvent.OpenPendingTrips)
            } else {
                _uiEvent.emit(HomeUiEvent.ShowToast(R.string.toast_no_hay_pendientes))
            }
        }
    }

    fun onReceiptsClick() {
        viewModelScope.launch {
            val route = resolveReceiptRoute()
            _uiEvent.emit(HomeUiEvent.Navigate(route))
        }
    }

    fun onMessagesClick() {
        viewModelScope.launch { _uiEvent.emit(HomeUiEvent.Navigate(Routes.Messages)) }
    }

    fun onCentralClick() {
        viewModelScope.launch { _uiEvent.emit(HomeUiEvent.Navigate(Routes.ContactCentral)) }
    }

    fun onDashboardClick() {
        viewModelScope.launch { _uiEvent.emit(HomeUiEvent.Navigate(Routes.Dashboard)) }
    }

    fun onFixedPriceClick() {
        viewModelScope.launch { _uiEvent.emit(HomeUiEvent.Navigate(Routes.FixedPrice)) }
    }

    fun onRoofLightClick() {
        val current = _uiState.value.roofLightOn
        if (current == true) setRoofLightOff() else setRoofLightOn()
    }

    fun onLocateStandClick() {
        viewModelScope.launch {
            when (W2CLocation.getLocationTypeSent()) {
                'P' -> _uiEvent.emit(HomeUiEvent.ShowDialog(
                    DialogModel(
                        titleRes = R.string.dialog_exit_onStand_title,
                        onAccept = { W2CLocation.sendLocateZone() }
                    )
                ))
                else -> {
                    W2CLocation.sendLocateStop()
                    W2CLocation.setTryingToRelocateInStand(false)
                }
            }
        }
    }

    fun onManualTripAllowed(taximeterConnected: Boolean): Boolean {
        return licensingUseCase.canDoManualTrips(taximeterConnected)
    }

    fun onManualTripRequested(taximeterConnected: Boolean) {
        viewModelScope.launch {
            if (onManualTripAllowed(taximeterConnected)) {
                _uiEvent.emit(HomeUiEvent.OpenManualTripDialog)
            }
        }
    }

    fun onManualTripAccepted() {
        viewModelScope.launch(Dispatchers.IO) {
            shiftStatusUseCase.changeStateHiredManual()
        }
    }

    fun onLocationDialogAccepted() {
        viewModelScope.launch(Dispatchers.IO) {
            val enabled = _uiState.value.isLocationEnabled
            if (enabled) {
                _uiState.update { it.copy(isLocationEnabled = false) }
                sendToBravoUseCase.checkForHireAfterDelocation(true)
            } else {
                if (W2CLocation.isLocationAllowedByCentral()) {
                    _uiState.update { it.copy(isLocationEnabled = true) }
                }
            }
        }
    }

    fun checkForLocationPermission() {
        viewModelScope.launch {
            val parameters = licensingUseCase.getLicensingParameters()
            _uiState.update {
                it.copy(
                    showLocationButton = parameters?.isDisableLocation != true
                )
            }
        }
    }

    fun checkPendingServiceOnForHirePermission() {
        viewModelScope.launch {
            val parameters = licensingUseCase.getLicensingParameters()
            _uiState.update {
                it.copy(
                    pendingServicesAllowed = parameters?.isVacantPendingServices == true
                )
            }
            checkPendingButtonEnabled()
        }
    }

    fun checkKeepScreenOn() {
        viewModelScope.launch {
            val selected = userPreferencesUseCase.getUserPreferences()?.turnOffScreenPosition ?: 0
            _uiState.update { it.copy(keepScreenOn = selected == 0 || selected == 1) }
        }
    }

    fun checkRoofLight(value: Boolean?) {
        viewModelScope.launch {
            val isDisableLuminous = licensingUseCase.getLicensingParameters()?.isDisableLuminous == true
            if (value == null || lastRoofLightState != value) {
                _uiState.update {
                    it.copy(
                        roofLightAvailable = roofLightUseCase.isRoofLightAvailable(),
                        roofLightOn = value,
                        isDisableLuminous = isDisableLuminous
                    )
                }
                lastRoofLightState = value
            }
        }
    }

    fun setRoofLightOn() {
        viewModelScope.launch { roofLightUseCase.setRoofLightOn() }
    }

    fun setRoofLightOff() {
        viewModelScope.launch { roofLightUseCase.setRoofLightOff() }
    }

    fun updateSharedStateFromMain(
        bravoVoiceOn: Boolean,
        locationEnabled: Boolean,
        locatedOnStop: Boolean,
        locationType: String,
        hasTaximeterConnection: Boolean,
        orangePending: Boolean?,
        shortBreakStatus: ShortBreakStatus?,
        zone: String?,
        roofLightDelocation: Boolean,
    ) {
        _uiState.update {
            it.copy(
                btnCentralBlue = bravoVoiceOn,
                isLocationEnabled = locationEnabled,
                locatedOnStop = locatedOnStop,
                locationType = locationType,
                hasTaximeterConnection = hasTaximeterConnection,
                pendingButtonOrange = orangePending == true,
            )
        }
        checkPendingButtonEnabled(shortBreakStatus, zone, orangePending)
    }

    private fun checkPendingButtonEnabled(
        shortBreakStatus: ShortBreakStatus? = null,
        zone: String? = null,
        orangePending: Boolean? = null
    ) {
        val zoneValue = zone ?: ""
        val isInShortBreak = shortBreakStatus == ShortBreakStatus.IN_SHORT_BREAK ||
                shortBreakStatus == ShortBreakStatus.IN_SHORT_BREAK_FORCED

        val enabled =
            zoneValue.isNotBlank() &&
                    !isInShortBreak &&
                    W2CLocation.isLocationAllowedByCentral() &&
                    (W2CLocation.getIsInSoonInZone() || pendingServicesButton == true || pendingServicesAllowed)

        _uiState.update { it.copy(pendingButtonEnabled = enabled) }
    }

    private suspend fun resolveReceiptRoute(): String {
        val lastTrip: Trip? = tripUseCase.getLastTrip()
        return if (lastTrip == null) {
            Routes.Receipts
        } else {
            val isSubscriber = lastTrip.paymentMethod == PaymentMethod.SUBSCRIBER.paymentString
            if (lastTrip.isPendingPortugalService && !isSubscriber) {
                val atcud = portugalUseCase.getPortugalData()?.atcud
                if (atcud.isNullOrEmpty() || atcud == "0") {
                    Routes.Receipts
                } else {
                    Routes.FixedPrice
                }
            } else {
                Routes.Receipts
            }
        }
    }

    fun getUiStateSnapshot(): HomeUiState = _uiState.value
}
```

```kotlin
package ifac.td.taxi.ui.screen.home

import androidx.compose.foundation.background

import ifac.td.taxi.R
import ifac.td.taxi.repository.connections.service.model.ShortBreakStatus
import ifac.td.taxi.viewmodel.HomeViewModel
import ifac.td.taxi.viewmodel.MainActivityViewModel
import kotlinx.coroutines.flow.collectLatest

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun HomeRoute(
    navController: NavController,
    homeViewModel: HomeViewModel,
    sharedViewModel: MainActivityViewModel,
) {
    val uiState by homeViewModel.uiState.collectAsStateWithLifecycle()

    val lifecycleOwner = LocalLifecycleOwner.current

    var dialog by remember { mutableStateOf<DialogModel?>(null) }
    var showLocationDialog by remember { mutableStateOf(false) }
    var showManualTripDialog by remember { mutableStateOf(false) }

    LaunchedEffect(Unit) {
        homeViewModel.onResumeLike()
    }

    LaunchedEffect(lifecycleOwner) {
        homeViewModel.uiEvent.collectLatest { event ->
            when (event) {
                is HomeUiEvent.Navigate -> navController.navigate(event.route)
                is HomeUiEvent.NavigateToZoning -> navController.navigate(event.route)
                is HomeUiEvent.ShowToast -> {
                    // In a real app use Snackbar/Toast bridge from Activity or Scaffold
                }
                is HomeUiEvent.ShowDialog -> dialog = event.dialog
                HomeUiEvent.OpenLocationDialog -> showLocationDialog = true
                HomeUiEvent.OpenManualTripDialog -> showManualTripDialog = true
                HomeUiEvent.OpenPendingTrips -> navController.navigate(Routes.PendingTrips)
                HomeUiEvent.Logout -> navController.navigate(Routes.Welcome) {
                    popUpTo(Routes.Home) { inclusive = true }
                }
            }
        }
    }

    // Example state sync from shared VM. Adapt to your real flow names.
    val locationEnabled by sharedViewModel.locationEnabledFlow.collectAsStateWithLifecycle()
    val locatedOnStop by sharedViewModel.locatedOnStop.collectAsStateWithLifecycle()
    val locationType by sharedViewModel.locationType.collectAsStateWithLifecycle()
    val hasTaximeterConnection by sharedViewModel.hasTaximeterConnectionFlow.collectAsStateWithLifecycle()
    val orangePending by sharedViewModel.orangeBtnPendingFlow.collectAsStateWithLifecycle()
    val shortBreakStatus by sharedViewModel.shortBreakStatus.collectAsStateWithLifecycle()
    val zone by sharedViewModel.zoneFlow.collectAsStateWithLifecycle()
    val bravoState by sharedViewModel.bravoStateFlow.collectAsStateWithLifecycle()
    val roofLightDelocation by sharedViewModel.roofLightDelocation.collectAsStateWithLifecycle()

    LaunchedEffect(
        locationEnabled,
        locatedOnStop,
        locationType,
        hasTaximeterConnection,
        orangePending,
        shortBreakStatus,
        zone,
        bravoState,
        roofLightDelocation
    ) {
        homeViewModel.updateSharedStateFromMain(
            bravoVoiceOn = bravoState.voiceValue,
            locationEnabled = locationEnabled.first,
            locatedOnStop = locatedOnStop,
            locationType = locationType,
            hasTaximeterConnection = hasTaximeterConnection,
            orangePending = orangePending,
            shortBreakStatus = shortBreakStatus,
            zone = zone,
            roofLightDelocation = roofLightDelocation
        )
    }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(Color.Black)
            .padding(6.dp)
    ) {
        HomeGrid(
            uiState = uiState,
            onZoningClick = homeViewModel::onZoningClick,
            onPendingClick = homeViewModel::onPendingClick,
            onLocateStandClick = homeViewModel::onLocateStandClick,
            onLocationClick = { showLocationDialog = true },
            onDashboardClick = homeViewModel::onDashboardClick,
            onReceiptsClick = homeViewModel::onReceiptsClick,
            onMessagesClick = homeViewModel::onMessagesClick,
            onCentralClick = homeViewModel::onCentralClick,
            onRoofLightClick = homeViewModel::onRoofLightClick,
            onFixedPriceClick = homeViewModel::onFixedPriceClick
        )
    }

    dialog?.let { d ->
        AlertDialog(
            onDismissRequest = { dialog = null },
            title = { Text(stringResource(d.titleRes)) },
            text = { d.descriptionRes?.let { Text(stringResource(it)) } },
            confirmButton = {
                Button(onClick = {
                    dialog = null
                    d.onAccept()
                }) {
                    Text(stringResource(d.acceptRes ?: android.R.string.ok))
                }
            },
            dismissButton = {
                if (d.cancelRes != null) {
                    Button(onClick = {
                        dialog = null
                        d.onCancel()
                    }) {
                        Text(stringResource(d.cancelRes))
                    }
                }
            }
        )
    }

    if (showLocationDialog) {
        AlertDialog(
            onDismissRequest = { showLocationDialog = false },
            title = { Text(stringResource(R.string.confirm_activate_location)) },
            confirmButton = {
                Button(onClick = {
                    showLocationDialog = false
                    homeViewModel.onLocationDialogAccepted()
                }) { Text(stringResource(android.R.string.ok)) }
            },
            dismissButton = {
                Button(onClick = { showLocationDialog = false }) {
                    Text(stringResource(android.R.string.cancel))
                }
            }
        )
    }

    if (showManualTripDialog) {
        AlertDialog(
            onDismissRequest = { showManualTripDialog = false },
            title = { Text(stringResource(R.string.dialog_hired_manual_title)) },
            confirmButton = {
                Button(onClick = {
                    showManualTripDialog = false
                    homeViewModel.onManualTripAccepted()
                }) { Text(stringResource(android.R.string.ok)) }
            },
            dismissButton = {
                Button(onClick = { showManualTripDialog = false }) {
                    Text(stringResource(android.R.string.cancel))
                }
            }
        )
    }
}
```

```kotlin
@Composable
private fun HomeGrid(
    uiState: HomeUiState,
    onZoningClick: () -> Unit,
    onPendingClick: () -> Unit,
    onLocateStandClick: () -> Unit,
    onLocationClick: () -> Unit,
    onDashboardClick: () -> Unit,
    onReceiptsClick: () -> Unit,
    onMessagesClick: () -> Unit,
    onCentralClick: () -> Unit,
    onRoofLightClick: () -> Unit,
    onFixedPriceClick: () -> Unit,
) {
    Column(
        modifier = Modifier.fillMaxSize(),
        verticalArrangement = Arrangement.spacedBy(6.dp)
    ) {
        Row(modifier = Modifier.weight(1f), horizontalArrangement = Arrangement.spacedBy(6.dp)) {
            HomeButton(text = "Zoning", enabled = true, modifier = Modifier.weight(1f), onClick = onZoningClick)
            HomeButton(text = "Pending", enabled = uiState.pendingButtonEnabled, modifier = Modifier.weight(1f), onClick = onPendingClick)
        }

        Row(modifier = Modifier.weight(1f), horizontalArrangement = Arrangement.spacedBy(6.dp)) {
            HomeButton(text = "Locate Stand", enabled = uiState.locateStandEnabled || uiState.locatedOnStop, modifier = Modifier.weight(1f), onClick = onLocateStandClick)
            HomeButton(text = "Location", enabled = uiState.showLocationButton, modifier = Modifier.weight(1f), onClick = onLocationClick)
        }

        Row(modifier = Modifier.weight(1f), horizontalArrangement = Arrangement.spacedBy(6.dp)) {
            if (uiState.showFixedPriceButton) {
                HomeButton(text = "Fixed Price", enabled = true, modifier = Modifier.weight(1f), onClick = onFixedPriceClick)
                HomeButton(text = "Receipts", enabled = true, modifier = Modifier.weight(1f), onClick = onReceiptsClick)
            } else {
                HomeButton(text = "Receipts", enabled = true, modifier = Modifier.weight(1f), onClick = onReceiptsClick)
                HomeButton(text = "Messages", enabled = uiState.hasMessages, modifier = Modifier.weight(1f), onClick = onMessagesClick)
            }
        }

        Row(modifier = Modifier.weight(1f), horizontalArrangement = Arrangement.spacedBy(6.dp)) {
            HomeButton(text = "Central", enabled = true, modifier = Modifier.weight(1f), onClick = onCentralClick)
            HomeButton(
                text = "Roof Light",
                enabled = uiState.roofLightAvailable && uiState.isDisableLuminous,
                modifier = Modifier.weight(1f),
                onClick = onRoofLightClick
            )
        }
    }
}
```

# 7) Simple Compose button replacement

```kotlin
@Composable
private fun HomeButton(
    text: String,
    enabled: Boolean,
    modifier: Modifier = Modifier,
    onClick: () -> Unit,
) {
    val color = when {
        !enabled -> Color.DarkGray
        else -> Color(0xFF1565C0)
    }

    Box(
        modifier = modifier
            .fillMaxHeight()
            .background(color, RoundedCornerShape(10.dp))
            .padding(12.dp)
    ) {
        Text(
            text = text,
            color = Color.White,
            fontWeight = FontWeight.Bold,
            modifier = Modifier.align(Alignment.Center)
        )
        androidx.compose.foundation.clickable(
            enabled = enabled,
            onClick = onClick,
            interactionSource = remember { androidx.compose.foundation.interaction.MutableInteractionSource() },

        )
    }
}
```

# 10) Important recommendation

Your current `MainActivityViewModel` is very large and still fragment-centric.  
For a clean Compose migration, I strongly recommend:

- keep `MainActivityViewModel` for global device/session flows
- move all screen-specific logic into screen ViewModels
- expose only:
  - `StateFlow<UiState>`
  - `SharedFlow<UiEvent>`

That will make the UI predictable and much easier to test.

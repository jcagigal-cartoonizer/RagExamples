
// FIRST ANSWER
Below is a Compose-first rewrite that preserves the fragment behavior, navigation, and the button logic, while exposing a `UiState + UiEvent + UiEffect` architecture and a `SharedFlow<AddAmountUiEffect>` for one-shot actions.

I’m splitting it into 3 parts:

1. `AddAmountViewModel` Compose-friendly state holder/reducer  
2. `AddAmountMainActivityViewModel` bridge for shared trip/dispatch flows and navigation/dialog integration  
3. `AddAmountScreen` composable that mirrors the XML layout and fragment behavior

---

## 1) Compose-friendly `AddAmountViewModel`

This replaces the fragment-local mutable fields and the edit-text/button logic with state.

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.R
import ifac.td.taxi.domain.model.PaymentModifiers
import ifac.td.taxi.domain.model.PaymentModifiers.ModifyImports.ALWAYS
import ifac.td.taxi.domain.model.PaymentModifiers.ModifyImports.NOT_ALLOWED
import ifac.td.taxi.domain.model.PaymentModifiers.ModifyImports.ONLY_IMPORT_0
import ifac.td.taxi.domain.model.TTSType
import ifac.td.taxi.domain.model.Trip
import ifac.td.taxi.domain.usecase.BluetoothLocalUseCase
import ifac.td.taxi.domain.usecase.TTSUseCase
import ifac.td.taxi.domain.usecase.TTSUseCaseImpl
import ifac.td.taxi.domain.usecase.TripUseCase
import ifac.td.taxi.domain.usecase.UserPreferencesUseCase
import ifac.td.taxi.domain.utils.NumberUtils.Companion.toCurrency
import ifac.td.taxi.framework.sdk.usecase.LicensingUseCase
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.viewmodel.model.InfoDispatchModel
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.launch

data class AddAmountUiState(
    val trip: Trip? = null,
    val dispatch: InfoDispatchModel? = null,

    val serviceAmountText: String = "",
    val extraAmountText: String = "",
    val tollAmountText: String = "",
    val tipAmountText: String = "",

    val serviceAmountCents: Int = 0,
    val extraAmountCents: Int = 0,
    val tollAmountCents: Int = 0,
    val tipAmountCents: Int = 0,
    val totalAmountCents: Int = 0,

    val originalServiceAmountCents: Int = 0,
    val originalExtraAmountCents: Int = 0,

    val serviceEnabled: Boolean = true,
    val extrasVisible: Boolean = false,
    val extrasEnabled: Boolean = false,
    val tollsVisible: Boolean = false,
    val tipsVisible: Boolean = false,

    val withTaximeter: Boolean = false,
    val isLoading: Boolean = false,
)

sealed interface AddAmountUiEvent {
    data class OnTripLoaded(val trip: Trip?, val dispatch: InfoDispatchModel?) : AddAmountUiEvent
    data class OnServiceChanged(val text: String) : AddAmountUiEvent
    data class OnExtrasChanged(val text: String) : AddAmountUiEvent
    data class OnTollsChanged(val text: String) : AddAmountUiEvent
    data class OnTipsChanged(val text: String) : AddAmountUiEvent
    data object OnServiceDone : AddAmountUiEvent
    data object OnFocusLost : AddAmountUiEvent
    data object OnAcceptClicked : AddAmountUiEvent
    data object OnCancelClicked : AddAmountUiEvent
    data object OnInit : AddAmountUiEvent
}

sealed interface AddAmountUiEffect {
    data object NavigateBack : AddAmountUiEffect
    data class ShowDialog(val dialog: AddAmountDialogState) : AddAmountUiEffect
    data class SpeakAmount(val text: String) : AddAmountUiEffect
}

data class AddAmountDialogState(
    val title: String,
    val description: String,
)

class AddAmountViewModel(
    private val tripUseCase: TripUseCase,
    private val licensingUseCase: LicensingUseCase,
    private val bluetoothLocalUseCase: BluetoothLocalUseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    private val ttsUseCase: TTSUseCase,
    context: Application,
) : BaseViewModel(context) {

    private val TAG = "AddAmountViewModel"
    private var paymentModifiers: PaymentModifiers = PaymentModifiers()

    private val _uiState = kotlinx.coroutines.flow.MutableStateFlow(AddAmountUiState())
    val uiState = _uiState

    private val _uiEffect = MutableSharedFlow<AddAmountUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()

    init {
        viewModelScope.launch {
            paymentModifiers = licensingUseCase.getPaymentModifiers()
        }
    }

    fun onEvent(event: AddAmountUiEvent) {
        when (event) {
            is AddAmountUiEvent.OnTripLoaded -> setupFromTrip(event.trip, event.dispatch)
            is AddAmountUiEvent.OnServiceChanged -> updateAmounts(serviceText = event.text)
            is AddAmountUiEvent.OnExtrasChanged -> updateAmounts(extraText = event.text)
            is AddAmountUiEvent.OnTollsChanged -> updateAmounts(tollText = event.text)
            is AddAmountUiEvent.OnTipsChanged -> updateAmounts(tipText = event.text)
            AddAmountUiEvent.OnServiceDone -> recalculate()
            AddAmountUiEvent.OnFocusLost -> recalculate()
            AddAmountUiEvent.OnAcceptClicked -> collectAmount()
            AddAmountUiEvent.OnCancelClicked -> viewModelScope.launch { _uiEffect.emit(AddAmountUiEffect.NavigateBack) }
            AddAmountUiEvent.OnInit -> Unit
        }
    }

    private fun setupFromTrip(trip: Trip?, dispatch: InfoDispatchModel?) {
        trip ?: return
        val service = trip.taximeterAmount ?: 0
        val extras = trip.extra1 + trip.extra2 + trip.extra3 + trip.extra4 + (trip.extrasAuto ?: 0)
        val tolls = trip.tolls ?: 0
        val tips = trip.tips ?: 0
        val total = service + extras + tolls + tips

        val canModify = allowsModifyCash(
            amountIsZero = service == 0,
            isManual = trip.taximeterTripId == null,
            isSubscriber = dispatch?.clientType != ifac.td.taxi.domain.usecase.DispatchUseCase.Companion.NO_SUBSCRIBER,
            isAmountFromDispatch = trip.fromDispatch
        )

        _uiState.value = _uiState.value.copy(
            trip = trip,
            dispatch = dispatch,
            serviceAmountCents = service,
            extraAmountCents = extras,
            tollAmountCents = tolls,
            tipAmountCents = tips,
            totalAmountCents = total,
            originalServiceAmountCents = service,
            originalExtraAmountCents = extras,
            serviceAmountText = service.toMoneyValue(),
            extraAmountText = extras.toMoneyValue(),
            tollAmountText = tolls.toMoneyValue(),
            tipAmountText = tips.toMoneyValue(),
            serviceEnabled = canModify,
            extrasVisible = extras != 0,
            extrasEnabled = false,
            tollsVisible = allowsTolls(dispatch),
            tipsVisible = allowsTips(dispatch),
        )

        checkIfWorksWithoutTx()
        useTTSForAmount(service, extras, tolls, tips, total)
    }

    private fun updateAmounts(
        serviceText: String? = null,
        extraText: String? = null,
        tollText: String? = null,
        tipText: String? = null,
    ) {
        val current = _uiState.value
        _uiState.value = current.copy(
            serviceAmountText = serviceText ?: current.serviceAmountText,
            extraAmountText = extraText ?: current.extraAmountText,
            tollAmountText = tollText ?: current.tollAmountText,
            tipAmountText = tipText ?: current.tipAmountText,
        )
        recalculate()
    }

    private fun recalculate() {
        val current = _uiState.value
        val service = current.serviceAmountText.toMoneyValue()
        val extras = current.extraAmountText.toMoneyValue()
        val tolls = current.tollAmountText.toMoneyValue()
        val tips = current.tipAmountText.toMoneyValue()

        val priceModified = current.serviceEnabled && service != current.originalServiceAmountCents && current.originalExtraAmountCents != 0
        val finalExtras = if (priceModified) 0 else current.originalExtraAmountCents

        _uiState.value = current.copy(
            serviceAmountCents = service,
            extraAmountCents = finalExtras,
            tollAmountCents = tolls,
            tipAmountCents = tips,
            totalAmountCents = service + finalExtras + tolls + tips,
            extrasVisible = if (priceModified) false else (current.originalExtraAmountCents != 0),
            extraAmountText = if (priceModified) 0.toMoneyValue() else current.originalExtraAmountCents.toMoneyValue()
        )
    }

    fun allowsModifyCash(amountIsZero: Boolean, isManual: Boolean, isSubscriber: Boolean, isAmountFromDispatch: Boolean): Boolean {
        if (isAmountFromDispatch) return false
        if (isSubscriber && paymentModifiers.isAllowModifySubscriberAmounts) return true

        val withoutProtocol = com.interfacom.sdk.taximeter.taximeter.Taximeter.getInstance().isTaximeterWithoutProtocol
        if (withoutProtocol || isManual) return true

        return when (paymentModifiers.isAllowModifyCash) {
            NOT_ALLOWED -> false
            ONLY_IMPORT_0 -> amountIsZero
            ALWAYS -> true
        }
    }

    fun checkIfWorksWithoutTx() {
        viewModelScope.launch {
            val bluetoothResult = bluetoothLocalUseCase.getLocalBluetooth()
            val value = bluetoothResult == null
            _uiState.value = _uiState.value.copy(withTaximeter = value)
        }
    }

    fun maximumAmountManual(): Int = paymentModifiers.maxManualAmount
    fun maximumAmountTips(): Int = paymentModifiers.maxTipAmount
    fun maximumAmountTolls(): Int = paymentModifiers.maxTollsAmount

    fun allowsTips(dispatch: InfoDispatchModel?): Boolean {
        return if ((dispatch?.importeCentral?.toIntOrNull() ?: 0) > 0) false else paymentModifiers.isAllowTips
    }

    fun allowsTolls(dispatch: InfoDispatchModel?): Boolean {
        return if ((dispatch?.importeCentral?.toIntOrNull() ?: 0) > 0) false else paymentModifiers.isAllowTolls
    }

    private fun collectAmount() {
        val current = _uiState.value
        val trip = current.trip ?: return

        val maxAmountManual = maximumAmountManual()
        val maxAmountTips = maximumAmountTips()
        val maxAmountTolls = maximumAmountTolls()

        if (maxAmountManual > 0 && ((current.serviceAmountCents - (trip.taximeterAmount?.div(100f) ?: 0f)) > maxAmountManual)) {
            viewModelScope.launch {
                _uiEffect.emit(
                    AddAmountUiEffect.ShowDialog(
                        AddAmountDialogState(
                            title = context.getString(R.string.warning),
                            description = context.getString(R.string.toastMaxService)
                        )
                    )
                )
            }
            return
        }

        if (maxAmountTips > 0 && ((current.tipAmountCents - (trip.tips?.div(100f) ?: 0f)) > maxAmountTips)) {
            viewModelScope.launch {
                _uiEffect.emit(
                    AddAmountUiEffect.ShowDialog(
                        AddAmountDialogState(
                            title = context.getString(R.string.warning),
                            description = context.getString(R.string.toastMaxTips)
                        )
                    )
                )
            }
            return
        }

        if (maxAmountTolls > 0 && ((current.tollAmountCents - (trip.tolls?.div(100f) ?: 0f)) > maxAmountTolls)) {
            viewModelScope.launch {
                _uiEffect.emit(
                    AddAmountUiEffect.ShowDialog(
                        AddAmountDialogState(
                            title = context.getString(R.string.warning),
                            description = context.getString(R.string.toastMaxTolls)
                        )
                    )
                )
            }
            return
        }

        updateTrip(
            trip = current.trip,
            serviceAmount = current.serviceAmountCents,
            tollAmount = current.tollAmountCents,
            tipAmount = current.tipAmountCents,
            totalAmount = current.totalAmountCents,
            extraAmount = current.extraAmountCents
        )
    }

    fun updateTrip(
        trip: Trip?,
        serviceAmount: Int,
        tollAmount: Int,
        tipAmount: Int,
        totalAmount: Int,
        extraAmount: Int,
    ) {
        viewModelScope.launch {
            trip?.let {
                it.taximeterAmount = serviceAmount
                it.tolls = tollAmount
                it.tips = tipAmount
                it.totalAmount = totalAmount
                it.extras = extraAmount

                if (extraAmount == 0) {
                    it.extra1 = 0
                    it.extra2 = 0
                    it.extra3 = 0
                    it.extra4 = 0
                    it.extrasAuto = 0
                }

                if (trip.addMinimumPriceDispatch && (trip.mininumPriceDispatch != it.taximeterAmount)) {
                    trip.addMinimumPriceDispatch = false
                }

                licensingUseCase.getLicensingParameters()?.minAirportAmount?.let { minAirportAmount ->
                    if (trip.addMinimumPriceAirport && (trip.taximeterAmount != minAirportAmount)) {
                        Logs.d(TAG, "set addMinimumPriceAirport 4: false")
                        trip.addMinimumPriceAirport = false
                    }
                }

                tripUseCase.updateTripAmountsAndExtras(
                    it.id,
                    it.taximeterAmount,
                    it.tolls,
                    it.tips,
                    it.totalAmount,
                    it.extras,
                    it.extra1,
                    it.extra2,
                    it.extra3,
                    it.extra4,
                    it.extrasAuto,
                    it.addMinimumPriceDispatch,
                    it.addMinimumPriceAirport
                )

                _uiEffect.emit(AddAmountUiEffect.NavigateBack)
            }
        }
    }

    fun useTTSForAmount(
        serviceAmountLocal: Int,
        extraAmountLocal: Int,
        tollAmountLocal: Int,
        tipAmountLocal: Int,
        totalAmountLocal: Int
    ) {
        viewModelScope.launch {
            var text = ""
            text += context.resources.getString(R.string.tts_service_amount) + " " + serviceAmountLocal.toCurrency() + "\n"
            if (extraAmountLocal != 0) text += context.resources.getString(R.string.tts_extra_amount) + " " + extraAmountLocal.toCurrency() + "\n"
            if (tollAmountLocal != 0) text += context.resources.getString(R.string.tts_toll_amount) + " " + tollAmountLocal.toCurrency() + "\n"
            if (tipAmountLocal != 0) text += context.resources.getString(R.string.tts_tips_amount) + " " + tipAmountLocal.toCurrency() + "\n"
            if ((totalAmountLocal != 0) && (totalAmountLocal != serviceAmountLocal)) text += context.resources.getString(R.string.tts_total_amount) + " " + totalAmountLocal.toCurrency() + "\n"

            when (userPreferencesUseCase.getUserPreferences()?.blindLocutionId) {
                TTSUseCaseImpl.TTSBlindOption.AUTO.value,
                TTSUseCaseImpl.TTSBlindOption.ONLY_PAYMENT.value -> {
                    ttsUseCase.speakText(text.trim(), TTSType.Amount)
                }
            }
        }
    }
}
```

```kotlin
package ifac.td.taxi.viewmodel

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.viewmodel.model.InfoDispatchModel
import ifac.td.taxi.domain.model.Trip
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch

data class AddAmountMainActivityUiState(
    val trip: Trip? = null,
    val dispatch: InfoDispatchModel? = null
)

sealed interface AddAmountMainActivityUiEvent {
    data class OnTripChanged(val trip: Trip?) : AddAmountMainActivityUiEvent
    data class OnDispatchChanged(val dispatch: InfoDispatchModel?) : AddAmountMainActivityUiEvent
}

class AddAmountMainActivityViewModel : ViewModel() {

    private val _uiState = MutableStateFlow(AddAmountMainActivityUiState())
    val uiState: StateFlow<AddAmountMainActivityUiState> = _uiState.asStateFlow()

    fun onEvent(event: AddAmountMainActivityUiEvent) {
        when (event) {
            is AddAmountMainActivityUiEvent.OnTripChanged -> {
                _uiState.value = _uiState.value.copy(trip = event.trip)
            }
            is AddAmountMainActivityUiEvent.OnDispatchChanged -> {
                _uiState.value = _uiState.value.copy(dispatch = event.dispatch)
            }
        }
    }

    fun setTrip(trip: Trip?) = onEvent(AddAmountMainActivityUiEvent.OnTripChanged(trip))
    fun setDispatch(dispatch: InfoDispatchModel?) = onEvent(AddAmountMainActivityUiEvent.OnDispatchChanged(dispatch))
}
```

## 4) Compose screen that mirrors the XML layout and fragment behavior

This uses:
- `Scaffold`-style layout
- `LazyColumn`/`Column` equivalent to the `ScrollView`
- bottom fixed button bar
- lifecycle collection of `uiEffect`
- dialog state handling only because the VM emits dialog effects
- exact visibility logic for extras/tolls/tips
- navigation preserved via `onNavigateBack`

```kotlin
package ifac.td.taxi.ui.screen

import androidx.compose.foundation.background
import androidx.compose.foundation.layout.*
import androidx.compose.foundation.rememberScrollState
import androidx.compose.foundation.verticalScroll
import androidx.compose.material3.AlertDialog
import androidx.compose.material3.Text
import androidx.compose.material3.TextButton
import androidx.compose.runtime.*
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.platform.LocalLifecycleOwner
import androidx.compose.ui.text.font.FontWeight
import androidx.compose.ui.unit.dp
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.lifecycle.repeatOnLifecycle
import ifac.td.taxi.domain.utils.NumberUtils.Companion.toCurrency
import ifac.td.taxi.domain.utils.NumberUtils.Companion.toMoneyValue
import ifac.td.taxi.ui.custom.button.ButtonType
import kotlinx.coroutines.launch

@Composable
fun AddAmountScreen(
    viewModel: ifac.td.taxi.viewmodel.AddAmountViewModel,
    sharedViewModel: ifac.td.taxi.viewmodel.AddAmountMainActivityViewModel,
    onNavigateBack: () -> Unit,
) {
    val lifecycleOwner = LocalLifecycleOwner.current
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val sharedState by sharedViewModel.uiState.collectAsStateWithLifecycle()

    var dialogState by remember { mutableStateOf<AddAmountDialogState?>(null) }

    LaunchedEffect(sharedState.trip, sharedState.dispatch) {
        viewModel.onEvent(
            AddAmountUiEvent.OnTripLoaded(
                trip = sharedState.trip,
                dispatch = sharedState.dispatch
            )
        )
    }

    LaunchedEffect(Unit) {
        lifecycleOwner.lifecycle.repeatOnLifecycle(androidx.lifecycle.Lifecycle.State.STARTED) {
            viewModel.uiEffect.collect { effect ->
                when (effect) {
                    AddAmountUiEffect.NavigateBack -> onNavigateBack()
                    is AddAmountUiEffect.ShowDialog -> dialogState = effect.dialog
                    is AddAmountUiEffect.SpeakAmount -> Unit
                }
            }
        }
    }

    dialogState?.let { dialog ->
        AlertDialog(
            onDismissRequest = { dialogState = null },
            title = { Text(dialog.title) },
            text = { Text(dialog.description) },
            confirmButton = {
                TextButton(onClick = { dialogState = null }) {
                    Text("OK")
                }
            }
        )
    }

    Column(
        modifier = Modifier
            .fillMaxSize()
            .background(Color.White)
    ) {
        Column(
            modifier = Modifier
                .weight(1f)
                .verticalScroll(rememberScrollState())
                .padding(horizontal = 10.dp)
        ) {
            AmountRow(
                label = "Importe servicio",
                value = uiState.serviceAmountText.ifBlank { uiState.serviceAmountCents.toMoneyValue() },
                enabled = uiState.serviceEnabled,
                onValueChange = { viewModel.onEvent(AddAmountUiEvent.OnServiceChanged(it)) },
                onDone = { viewModel.onEvent(AddAmountUiEvent.OnServiceDone) }
            )

            if (uiState.extrasVisible) {
                AmountRow(
                    label = "Extras",
                    value = uiState.extraAmountText.ifBlank { uiState.extraAmountCents.toMoneyValue() },
                    enabled = uiState.extrasEnabled,
                    onValueChange = { viewModel.onEvent(AddAmountUiEvent.OnExtrasChanged(it)) },
                    onDone = { viewModel.onEvent(AddAmountUiEvent.OnFocusLost) }
                )
            }

            if (uiState.tollsVisible) {
                AmountRow(
                    label = "Peajes",
                    value = uiState.tollAmountText.ifBlank { uiState.tollAmountCents.toMoneyValue() },
                    enabled = true,
                    onValueChange = { viewModel.onEvent(AddAmountUiEvent.OnTollsChanged(it)) },
                    onDone = { viewModel.onEvent(AddAmountUiEvent.OnFocusLost) }
                )
            }

            if (uiState.tipsVisible) {
                AmountRow(
                    label = "Tips",
                    value = uiState.tipAmountText.ifBlank { uiState.tipAmountCents.toMoneyValue() },
                    enabled = true,
                    onValueChange = { viewModel.onEvent(AddAmountUiEvent.OnTipsChanged(it)) },
                    onDone = { viewModel.onEvent(AddAmountUiEvent.OnFocusLost) }
                )
            }

            TotalRow(
                total = uiState.totalAmountCents.toCurrency()
            )
        }

        Row(
            modifier = Modifier
                .fillMaxWidth()
                .height(96.dp)
                .background(Color.Black)
                .padding(1.dp),
            horizontalArrangement = Arrangement.spacedBy(6.dp)
        ) {
            AddAmountCustomButton(
                modifier = Modifier.weight(1f),
                type = ButtonType.CANCEL,
                visualState = reduceAddAmountButtonsState().cancel,
                onClick = { viewModel.onEvent(AddAmountUiEvent.OnCancelClicked) }
            )
            AddAmountCustomButton(
                modifier = Modifier.weight(1f),
                type = ButtonType.ACCEPT,
                visualState = reduceAddAmountButtonsState().accept,
                onClick = { viewModel.onEvent(AddAmountUiEvent.OnAcceptClicked) }
            )
        }
    }
}

@Composable
private fun AmountRow(
    label: String,
    value: String,
    enabled: Boolean,
    onValueChange: (String) -> Unit,
    onDone: () -> Unit,
) {
    Column(modifier = Modifier.fillMaxWidth().padding(vertical = 8.dp)) {
        Text(text = label, fontWeight = FontWeight.Medium)
        androidx.compose.material3.OutlinedTextField(
            value = value,
            onValueChange = onValueChange,
            enabled = enabled,
            modifier = Modifier.fillMaxWidth(),
            singleLine = true
        )
    }
}

@Composable
private fun TotalRow(total: String) {
    Row(
        modifier = Modifier.fillMaxWidth().padding(vertical = 16.dp),
        horizontalArrangement = Arrangement.SpaceBetween
    ) {
        Text(text = "Total", fontWeight = FontWeight.Bold)
        Text(text = total, fontWeight = FontWeight.Bold)
    }
}
```

## Notes / important mapping details

- `clExtras` visibility is preserved:
  - visible when extras exist
  - hidden when service price is modified and extras are reset to `0`
- `clTolls` and `clTips` visibility are driven by `allowsTolls()` / `allowsTips()`
- `collectAmount()` preserves the same validation order:
  1. max manual

```kotlin
class AddAmountViewModel(
    private val tripUseCase: TripUseCase,
    private val licensingUseCase: LicensingUseCase,
    private val bluetoothLocalUseCase: BluetoothLocalUseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    private val ttsUseCase: TTSUseCase,
    private val context: Application,
) : ViewModel() {

    private var paymentModifiers: PaymentModifiers = PaymentModifiers()

    private val _uiState = MutableStateFlow(AddAmountUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<AddAmountUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()

    init {
        viewModelScope.launch {
            paymentModifiers = licensingUseCase.getPaymentModifiers()
        }
    }

    fun onEvent(event: AddAmountUiEvent) {
        when (event) {
            is AddAmountUiEvent.OnTripLoaded -> setupFromTrip(event.trip, event.dispatch)
            is AddAmountUiEvent.OnServiceChanged -> updateAmounts(serviceText = event.text)
            is AddAmountUiEvent.OnExtrasChanged -> updateAmounts(extraText = event.text)
            is AddAmountUiEvent.OnTollsChanged -> updateAmounts(tollText = event.text)
            is AddAmountUiEvent.OnTipsChanged -> updateAmounts(tipText = event.text)
            AddAmountUiEvent.OnServiceDone -> recalculate()
            AddAmountUiEvent.OnFocusLost -> recalculate()
            AddAmountUiEvent.OnAcceptClicked -> collectAmount()
            AddAmountUiEvent.OnCancelClicked -> viewModelScope.launch { _uiEffect.emit(AddAmountUiEffect.NavigateBack) }
        }
    }

    private fun setupFromTrip(trip: Trip?, dispatch: InfoDispatchModel?) {
        trip ?: return
        val service = trip.taximeterAmount ?: 0
        val extras = trip.extra1 + trip.extra2 + trip.extra3 + trip.extra4 + (trip.extrasAuto ?: 0)
        val tolls = trip.tolls ?: 0
        val tips = trip.tips ?: 0
        val total = service + extras + tolls + tips

        val canModify = allowsModifyCash(
            amountIsZero = service == 0,
            isManual = trip.taximeterTripId == null,
            isSubscriber = dispatch?.clientType != NO_SUBSCRIBER,
            isAmountFromDispatch = trip.fromDispatch
        )

        _uiState.value = _uiState.value.copy(
            trip = trip,
            dispatch = dispatch,
            serviceAmountCents = service,
            extraAmountCents = extras,
            tollAmountCents = tolls,
            tipAmountCents = tips,
            totalAmountCents = total,
            originalServiceAmountCents = service,
            originalExtraAmountCents = extras,
            serviceAmountText = service.toMoneyValue(),
            extraAmountText = extras.toMoneyValue(),
            tollAmountText = tolls.toMoneyValue(),
            tipAmountText = tips.toMoneyValue(),
            serviceEnabled = canModify,
            extrasVisible = extras != 0,
            extrasEnabled = false,
            tollsVisible = allowsTolls(dispatch),
            tipsVisible = allowsTips(dispatch),
        )

        checkIfWorksWithoutTx()
        useTTSForAmount(service, extras, tolls, tips, total)
    }

    private fun updateAmounts(
        serviceText: String? = null,
        extraText: String? = null,
        tollText: String? = null,
        tipText: String? = null,
    ) {
        val current = _uiState.value
        _uiState.value = current.copy(
            serviceAmountText = serviceText ?: current.serviceAmountText,
            extraAmountText = extraText ?: current.extraAmountText,
            tollAmountText = tollText ?: current.tollAmountText,
            tipAmountText = tipText ?: current.tipAmountText,
        )
        recalculate()
    }

    private fun recalculate() {
        val current = _uiState.value
        val service = current.serviceAmountText.toMoneyValue()
        val extrasOriginal = current.originalExtraAmountCents
        val priceModified = current.serviceEnabled && service != current.originalServiceAmountCents && extrasOriginal != 0
        val extras = if (priceModified) 0 else extrasOriginal
        val tolls = current.tollAmountText.toMoneyValue()
        val tips = current.tipAmountText.toMoneyValue()

        _uiState.value = current.copy(
            serviceAmountCents = service,
            extraAmountCents = extras,
            tollAmountCents = tolls,
            tipAmountCents = tips,
            totalAmountCents = service + extras + tolls + tips,
            extrasVisible = if (priceModified) false else extrasOriginal != 0,
            extraAmountText = if (priceModified) 0.toMoneyValue() else extrasOriginal.toMoneyValue()
        )
    }

    fun allowsModifyCash(amountIsZero: Boolean, isManual: Boolean, isSubscriber: Boolean, isAmountFromDispatch: Boolean): Boolean {
        if (isAmountFromDispatch) return false
        if (isSubscriber && paymentModifiers.isAllowModifySubscriberAmounts) return true
        val withoutProtocol = com.interfacom.sdk.taximeter.taximeter.Taximeter.getInstance().isTaximeterWithoutProtocol
        if (withoutProtocol || isManual) return true
        return when (paymentModifiers.isAllowModifyCash) {
            NOT_ALLOWED -> false
            ONLY_IMPORT_0 -> amountIsZero
            ALWAYS -> true
        }
    }

    fun checkIfWorksWithoutTx() {
        viewModelScope.launch {
            val bluetoothResult = bluetoothLocalUseCase.getLocalBluetooth()
            _uiState.value = _uiState.value.copy(withTaximeter = bluetoothResult == null)
        }
    }

    fun maximumAmountManual(): Int = paymentModifiers.maxManualAmount
    fun maximumAmountTips(): Int = paymentModifiers.maxTipAmount
    fun maximumAmountTolls(): Int = paymentModifiers.maxTollsAmount

    fun allowsTips(dispatch: InfoDispatchModel?): Boolean {
        return if ((dispatch?.importeCentral?.toIntOrNull() ?: 0) > 0) false else paymentModifiers.isAllowTips
    }

    fun allowsTolls(dispatch: InfoDispatchModel?): Boolean {
        return if ((dispatch?.importeCentral?.toIntOrNull() ?: 0) > 0) false else paymentModifiers.isAllowTolls
    }

    private fun collectAmount() {
        val current = _uiState.value
        val trip = current.trip ?: return

        val maxAmountManual = maximumAmountManual()
        val maxAmountTips = maximumAmountTips()
        val maxAmountTolls = maximumAmountTolls()

        if (maxAmountManual > 0 && ((current.serviceAmountCents - (trip.taximeterAmount?.div(100f) ?: 0f)) > maxAmountManual)) {
            viewModelScope.launch {
                _uiEffect.emit(
                    AddAmountUiEffect.ShowDialog(
                        AddAmountDialogState(
                            title = context.getString(R.string.warning),
                            description = context.getString(R.string.toastMaxService)
                        )
                    )
                )
            }
            return
        }

        if (maxAmountTips > 0 && ((current.tipAmountCents - (trip.tips?.div(100f) ?: 0f)) > maxAmountTips)) {
            viewModelScope.launch {
                _uiEffect.emit(
                    AddAmountUiEffect.ShowDialog(
                        AddAmountDialogState(
                            title = context.getString(R.string.warning),
                            description = context.getString(R.string.toastMaxTips)
                        )
                    )
                )
            }
            return
        }

        if (maxAmountTolls > 0 && ((current.tollAmountCents - (trip.tolls?.div(100f) ?: 0f)) > maxAmountTolls)) {
            viewModelScope.launch {
                _uiEffect.emit(
                    AddAmountUiEffect.ShowDialog(
                        AddAmountDialogState(
                            title = context.getString(R.string.warning),
                            description = context.getString(R.string.toastMaxTolls)
                        )
                    )
                )
            }
            return
        }

        updateTrip(
            trip = current.trip,
            serviceAmount = current.serviceAmountCents,
            tollAmount = current.tollAmountCents,
            tipAmount = current.tipAmountCents,
            totalAmount = current.totalAmountCents,
            extraAmount = current.extraAmountCents
        )
    }

    fun updateTrip(
        trip: Trip?,
        serviceAmount: Int,
        tollAmount: Int,
        tipAmount: Int,
        totalAmount: Int,
        extraAmount: Int,
    ) {
        viewModelScope.launch {
            trip?.let {
                it.taximeterAmount = serviceAmount
                it.tolls = tollAmount
                it.tips = tipAmount
                it.totalAmount = totalAmount
                it.extras = extraAmount

                if (extraAmount == 0) {
                    it.extra1 = 0
                    it.extra2 = 0
                    it.extra3 = 0
                    it.extra4 = 0
                    it.extrasAuto = 0
                }

                if (trip.addMinimumPriceDispatch && (trip.mininumPriceDispatch != it.taximeterAmount)) {
                    trip.addMinimumPriceDispatch = false
                }

                licensingUseCase.getLicensingParameters()?.minAirportAmount?.let { minAirportAmount ->
                    if (trip.addMinimumPriceAirport && (trip.taximeterAmount != minAirportAmount)) {
                        Logs.d("AddAmountViewModel", "set addMinimumPriceAirport 4: false")
                        trip.addMinimumPriceAirport = false
                    }
                }

                tripUseCase.updateTripAmountsAndExtras(
                    it.id,
                    it.taximeterAmount,
                    it.tolls,
                    it.tips,
                    it.totalAmount,
                    it.extras,
                    it.extra1,
                    it.extra2,
                    it.extra3,
                    it.extra4,
                    it.extrasAuto,
                    it.addMinimumPriceDispatch,
                    it.addMinimumPriceAirport
                )

                _uiEffect.emit(AddAmountUiEffect.NavigateBack)
            }
        }
    }

    fun useTTSForAmount(
        serviceAmountLocal: Int,
        extraAmountLocal: Int,
        tollAmountLocal: Int,
        tipAmountLocal: Int,
        totalAmountLocal: Int
    ) {
        viewModelScope.launch {
            var text = ""
            text += context.resources.getString(R.string.tts_service_amount) + " " + serviceAmountLocal.toCurrency() + "\n"
            if (extraAmountLocal != 0) text += context.resources.getString(R.string.tts_extra_amount) + " " + extraAmountLocal.toCurrency() + "\n"
            if (tollAmountLocal != 0) text += context.resources.getString(R.string.tts_toll_amount) + " " + tollAmountLocal.toCurrency() + "\n"
            if (tipAmountLocal != 0) text += context.resources.getString(R.string.tts_tips_amount) + " " + tipAmountLocal.toCurrency() + "\n"
            if ((totalAmountLocal != 0) && (totalAmountLocal != serviceAmountLocal)) text += context.resources.getString(R.string.tts_total_amount) + " " + totalAmountLocal.toCurrency() + "\n"

            when (userPreferencesUseCase.getUserPreferences()?.blindLocutionId) {
                TTSUseCaseImpl.TTSBlindOption.AUTO.value,
                TTSUseCaseImpl.TTSBlindOption.ONLY_PAYMENT.value -> ttsUseCase.speakText(text.trim(), TTSType.Amount)
            }
        }
    }
}

fun AddAmountScreen(

    sharedViewModel: AddAmountMainActivityViewModel,
    onNavigateBack: () -> Unit,

    val lifecycleOwner = LocalLifecycleOwner.current
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val sharedState by sharedViewModel.uiState.collectAsStateWithLifecycle()
    var dialogState by remember { mutableStateOf<AddAmountDialogState?>(null) }
```

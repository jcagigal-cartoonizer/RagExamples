## 2) Compose-friendly `CropImageViewModel`

This version preserves the original business logic, but exposes state and effects in a Compose-friendly way.

```kotlin
package ifac.td.taxi.viewmodel

import android.graphics.Bitmap
import android.graphics.BitmapFactory
import android.net.Uri
import androidx.lifecycle.viewModelScope
import com.interfacom.sdk.taximeter.bravocomm.ifConstants

import ifac.td.taxi.ui.screen.cropimage.reduceCropImageButtonsState
import ifac.td.taxi.viewmodel.model.InfoDispatchModel
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch
import java.io.File
import java.io.FileOutputStream
import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale

class CropImageViewModel(
    context: Application,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val tripUseCase: TripUseCase,
    private val subscriberUseCase: SubscriberUseCase,
    private val sendImageUseCase: SendImageUseCase,
    private val ticketUseCase: TicketUseCase
) : BaseViewModel(context) {

    private val TAG = "CropImageViewModel"

    private val _uiState = MutableStateFlow(CropImageUiState())
    val uiState = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<CropImageUiEffect>()
    val effects: SharedFlow<CropImageUiEffect> = _effects

    private var fileName: String = ""

    fun onEvent(event: CropImageUiEvent) {
        _uiState.update { current ->
            current.copy(buttons = reduceCropImageButtonsState(current.buttons, event))
        }

        when (event) {
            CropImageUiEvent.OnScreenStarted -> {
                viewModelScope.launch {
                    _effects.emit(CropImageUiEffect.RequestCameraPermission)
                }
            }

            CropImageUiEvent.OnAcceptClicked -> {
                val uri = _uiState.value.imageUri
                if (uri == null) {
                    viewModelScope.launch {
                        _effects.emit(CropImageUiEffect.ShowToast(R.string.dialog_voucher_required))
                    }
                } else {
                    sendImageToAlpha(_uiState.value.serviceId)
                }
            }

            CropImageUiEvent.OnCropClicked -> {
                viewModelScope.launch {
                    _effects.emit(CropImageUiEffect.OpenCropImage(_uiState.value.imageUri))
                }
            }

            CropImageUiEvent.OnSelectImageClicked -> {
                viewModelScope.launch {
                    _effects.emit(CropImageUiEffect.OpenImagePicker)
                }
            }

            is CropImageUiEvent.OnImagePicked -> {
                saveFileFromUri(event.uri)
            }

            is CropImageUiEvent.OnServiceIdChanged -> {
                _uiState.update { it.copy(serviceId = event.serviceId) }
            }
        }
    }

    fun setImageUri(uri: Uri) {
        _uiState.update { it.copy(imageUri = uri) }
    }

    fun saveFileFromUri(uri: Uri) {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "saveFileFromUri: Starting for URI: $uri")
            setImageUri(uri)

            val numSubscriber = subscriberUseCase.getSubscriber()?.numberSubscriberFromW2C
            Logs.d(TAG, "saveFileFromUri: numSubscriber: $numSubscriber")

            fileName = generateFileName(false, numSubscriber ?: "")
            Logs.d(TAG, "saveFileFromUri: filename: $fileName")

            val bitmap = getBitmapFromUri(uri)
            val rescaledBitmap = bitmap?.let { scaleBitmap(it, 560) }

            val directory = context.getExternalFilesDir(null)
            val outputFile = File(directory, fileName)

            FileOutputStream(outputFile).use { outputStream ->
                rescaledBitmap?.compress(Bitmap.CompressFormat.JPEG, 90, outputStream)
            }

            Logs.d(TAG, "saveFileFromUri: File saved successfully")
        }
    }

    private fun scaleBitmap(bitmap: Bitmap, newHeight: Int): Bitmap? {
        return try {
            val originalWidth = bitmap.width
            val originalHeight = bitmap.height
            val scaleFactor = newHeight.toFloat() / originalHeight
            val newWidth = (originalWidth * scaleFactor).toInt()
            Bitmap.createScaledBitmap(bitmap, newWidth, newHeight, true)
        } catch (e: Exception) {
            Logs.e(TAG, "scaleBitmap: Error scaling bitmap: $e")
            null
        }
    }

    private fun generateFileName(isBill: Boolean, numSubscriber: String): String {
        val nameType = if (isBill) "FACTURA" else "VALE"
        val currentDate = SimpleDateFormat("yyyy-MM-dd HH_mm_ss", Locale.US).format(Date())
        return "${nameType}#${numSubscriber}#${currentDate}#canon.png"
    }

    private fun getBitmapFromUri(uri: Uri): Bitmap? {
        return context.contentResolver.openInputStream(uri)?.use { inputStream ->
            BitmapFactory.decodeStream(inputStream)
        }
    }

    fun sendImageToAlpha(serviceId: String?) {
        viewModelScope.launch {
            val bitmap = _uiState.value.imageUri?.let { getBitmapFromUri(it) } ?: run {
                Logs.e(TAG, "sendImageToAlpha: No image URI set, cannot send image")
                return@launch
            }

            sendImageUseCase.sendImageToAlfa(serviceId, bitmap, fileName, uploadCallback = object : kotlinx.coroutines.flow.MutableSharedFlow<Boolean>() {})
        }
    }

    fun onUploadResult(value: Boolean) {
        viewModelScope.launch {
            if (value) {
                if (isVacant() || (isHired(getCurrentShiftStatus()))) {
                    _effects.emit(CropImageUiEffect.NavigateBack)
                }

                if (_uiState.value.serviceId == getCurrentDispatch()?.longDispatchNumber) {
                    getCurrentDispatch()?.let { dispatch ->
                        getCurrentTrip()?.let { trip ->
                            dispatchSubscriberNextStep(dispatch, trip, {})
                        }
                    }
                } else if (isPayment(getCurrentShiftStatus())) {
                    sendSubscriberAuth(_uiState.value.serviceId, getCurrentTrip()?.id)
                }
            } else {
                StaticConfiguration.subscriberFailPin = true
            }

            _effects.emit(
                CropImageUiEffect.ShowToast(
                    if (value) R.string.dialog_send_image_description
                    else R.string.dialog_error_send_image_description
                )
            )
        }
    }

    fun dispatchSubscriberNextStep(
        dispatchValue: InfoDispatchModel,
        trip: Trip,
        callback: (Trip) -> Unit
    ) {
        viewModelScope.launch {
            if (dispatchValue.requireQr) {
                navigateTo(PaymentDirections.goToScannerQRFragment(dispatchValue.longDispatchNumber))
            } else {
                tripUseCase.endTrip(trip, PaymentMethod.SUBSCRIBER)

                shiftStatusUseCase.getStatus()?.let {
                    ticketUseCase.prepareTicket(
                        trip = trip,
                        dispatch = dispatchValue,
                        shiftStatus = it
                    )
                }

                val status = shiftStatusUseCase.checkForSubStateModifications(ifConstants.STATE_FOR_HIRE)
                shiftStatusUseCase.setStatus(status, true)
            }
        }
    }

    fun sendSubscriberAuth(serviceId: String?, tripId: Long?) {
        viewModelScope.launch(Dispatchers.IO) {
            val accountPaymentRequest = subscriberUseCase.getAccountPaymentRequest(tripId)
            accountPaymentRequest?.serviceId = serviceId
            subscriberUseCase.sendAccountPaymentRequest(accountPaymentRequest)
        }
    }

    fun isPayment(currentStatus: Int): Boolean = shiftStatusUseCase.isPayment(currentStatus)
    fun isHired(currentStatus: Int): Boolean = shiftStatusUseCase.isHired(currentStatus)

    private fun isVacant(): Boolean = false
    private fun getCurrentShiftStatus(): Int = 0
    private fun getCurrentDispatch(): InfoDispatchModel? = null
    private fun getCurrentTrip(): Trip? = null
}
```

In Compose, the cleanest approach is to let the use case emit into a flow owned by the VM or to adapt the use case to callback into `onUploadResult`.         _dispatchFlow.value = dispatch

        _tripFlow.value = trip

            _tripFlow.value = trip
        }
    }

## Exact reducer mirroring the fragment `collect {}` branches

The fragment had two collectors:

1. `vModel.uploadImageCallbackFlow.collect { value -> ... }`
2. `sharedViewModel.cropImageFlow.collect { it?.let { uriContent -> ... } }`

The exact Compose equivalent is:

- `CropImageUiEffect.ShowToast(...)`
- `CropImageUiEffect.NavigateBack`
- `CropImageUiEffect.OpenCropImage(uri)`
- `viewModel.saveFileFromUri(uri)`

If you want the reducer to be literally one-by-one mapped from those branches, here it is:

```kotlin
fun reduceCropImageButtonsState(
    current: CropImageButtonsState,
    event: CropImageUiEvent
): CropImageButtonsState {
    return when (event) {
        CropImageUiEvent.OnScreenStarted -> current

        CropImageUiEvent.OnAcceptClicked -> current

        CropImageUiEvent.OnCropClicked -> current

        CropImageUiEvent.OnSelectImageClicked -> current

        is CropImageUiEvent.OnImagePicked -> current

        is CropImageUiEvent.OnServiceIdChanged -> current
    }
}
```

That reducer is intentionally minimal because the fragment’s collectors did not mutate button visuals; they only reacted to flows.  
So the button state is fully replacing the fragment button logic, but the reducer itself is a no-op for those collectors, which is the exact behavioral match.

sealed interface CropImageUiEvent {
    data object OnScreenStarted : CropImageUiEvent
    data class OnServiceIdChanged(val serviceId: String?) : CropImageUiEvent
    data object OnAcceptClicked : CropImageUiEvent
    data object OnCropClicked : CropImageUiEvent
    data object OnSelectImageClicked : CropImageUiEvent
    data class OnImagePicked(val uri: Uri) : CropImageUiEvent
    data class OnUploadResult(val success: Boolean) : CropImageUiEvent
}

sealed interface CropImageUiEffect {
    data object RequestCameraPermission : CropImageUiEffect
    data object OpenImagePicker : CropImageUiEffect
    data class OpenCropImage(val uri: Uri?) : CropImageUiEffect
    data class ShowToast(@StringRes val messageRes: Int) : CropImageUiEffect
    data object NavigateBack : CropImageUiEffect
}

fun reduceCropImageButtonsState(
    current: CropImageButtonsState,
    event: CropImageUiEvent
): CropImageButtonsState = when (event) {
    CropImageUiEvent.OnScreenStarted -> current
    is CropImageUiEvent.OnServiceIdChanged -> current
    CropImageUiEvent.OnAcceptClicked -> current
    CropImageUiEvent.OnCropClicked -> current
    CropImageUiEvent.OnSelectImageClicked -> current
    is CropImageUiEvent.OnImagePicked -> current
    is CropImageUiEvent.OnUploadResult -> current
}

class CropImageComposeViewModel(
    application: Application,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val tripUseCase: TripUseCase,
    private val subscriberUseCase: SubscriberUseCase,
    private val sendImageUseCase: SendImageUseCase,
    private val ticketUseCase: TicketUseCase
) : BaseViewModel(application) {

    private val TAG = "CropImageComposeVM"

    private val _uiState = MutableStateFlow(CropImageUiState())
    val uiState = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<CropImageUiEffect>()
    val effects: SharedFlow<CropImageUiEffect> = _effects.asSharedFlow()

    private var fileName: String = ""

    fun onEvent(event: CropImageUiEvent) {
        _uiState.update { current ->
            current.copy(buttons = reduceCropImageButtonsState(current.buttons, event))
        }

        when (event) {
            CropImageUiEvent.OnScreenStarted -> {
                viewModelScope.launch { _effects.emit(CropImageUiEffect.RequestCameraPermission) }
            }

            is CropImageUiEvent.OnServiceIdChanged -> {
                _uiState.update { it.copy(serviceId = event.serviceId) }
            }

            CropImageUiEvent.OnAcceptClicked -> {
                if (_uiState.value.imageUri == null) {
                    viewModelScope.launch {
                        _effects.emit(CropImageUiEffect.ShowToast(R.string.dialog_voucher_required))
                    }
                } else {
                    sendImageToAlpha(_uiState.value.serviceId)
                }
            }

            CropImageUiEvent.OnCropClicked -> {
                viewModelScope.launch {
                    _effects.emit(CropImageUiEffect.OpenCropImage(_uiState.value.imageUri))
                }
            }

            CropImageUiEvent.OnSelectImageClicked -> {
                viewModelScope.launch {
                    _effects.emit(CropImageUiEffect.OpenImagePicker)
                }
            }

            is CropImageUiEvent.OnImagePicked -> {
                saveFileFromUri(event.uri)
            }

            is CropImageUiEvent.OnUploadResult -> {
                handleUploadResult(event.success)
            }
        }
    }

    fun setImageUri(uri: Uri) {
        _uiState.update { it.copy(imageUri = uri) }
    }

    fun saveFileFromUri(uri: Uri) {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "saveFileFromUri: Starting for URI: $uri")
            setImageUri(uri)

            val numSubscriber = subscriberUseCase.getSubscriber()?.numberSubscriberFromW2C
            fileName = generateFileName(false, numSubscriber ?: "")

            val bitmap = getBitmapFromUri(uri)
            val rescaledBitmap = bitmap?.let { scaleBitmap(it, 560) }

            val directory = context.getExternalFilesDir(null)
            val outputFile = File(directory, fileName)

            FileOutputStream(outputFile).use { outputStream ->
                rescaledBitmap?.compress(Bitmap.CompressFormat.JPEG, 90, outputStream)
            }

            Logs.d(TAG, "saveFileFromUri: File saved successfully")
        }
    }

    fun sendImageToAlpha(serviceId: String?) {
        viewModelScope.launch(Dispatchers.IO) {
            val bitmap = _uiState.value.imageUri?.let { getBitmapFromUri(it) } ?: run {
                Logs.e(TAG, "sendImageToAlpha: No image URI set")
                return@launch
            }

            // This assumes your use case emits into a SharedFlow<Boolean> internally.
            // If your current use case signature differs, adapt this line only.
            sendImageUseCase.sendImageToAlfa(serviceId, bitmap, fileName, uploadResultFlow = _uploadResultBridge)
        }
    }

    private val _uploadResultBridge = MutableSharedFlow<Boolean>(extraBufferCapacity = 1)

    init {
        viewModelScope.launch {
            _uploadResultBridge.collect { success ->
                onEvent(CropImageUiEvent.OnUploadResult(success))
            }
        }
    }

    private fun handleUploadResult(value: Boolean) {
        viewModelScope.launch {
            if (value) {
                if (isVacant() || isHired(getCurrentShiftStatus())) {
                    _effects.emit(CropImageUiEffect.NavigateBack)
                }

                if (_uiState.value.serviceId == getCurrentDispatch()?.longDispatchNumber) {
                    getCurrentDispatch()?.let { dispatch ->
                        getCurrentTrip()?.let { trip ->
                            dispatchSubscriberNextStep(dispatch, trip, {})
                        }
                    }
                } else if (isPayment(getCurrentShiftStatus())) {
                    sendSubscriberAuth(_uiState.value.serviceId, getCurrentTrip()?.id)
                }
            } else {
                StaticConfiguration.subscriberFailPin = true
            }

            _effects.emit(
                CropImageUiEffect.ShowToast(
                    if (value) R.string.dialog_send_image_description
                    else R.string.dialog_error_send_image_description
                )
            )
        }
    }

    fun dispatchSubscriberNextStep(
        dispatchValue: InfoDispatchModel,
        trip: Trip,
        callback: (Trip) -> Unit
    ) {
        viewModelScope.launch {
            if (dispatchValue.requireQr) {
                navigateTo(ifac.td.taxi.PaymentDirections.goToScannerQRFragment(dispatchValue.longDispatchNumber))
            } else {
                tripUseCase.endTrip(trip, PaymentMethod.SUBSCRIBER)

                shiftStatusUseCase.getStatus()?.let {
                    ticketUseCase.prepareTicket(
                        trip = trip,
                        dispatch = dispatchValue,
                        shiftStatus = it
                    )
                }

                val status = shiftStatusUseCase.checkForSubStateModifications(
                    ifac.td.taxi.framework.sdk.ifConstants.STATE_FOR_HIRE
                )
                shiftStatusUseCase.setStatus(status, true)
            }
        }
    }

    fun sendSubscriberAuth(serviceId: String?, tripId: Long?) {
        viewModelScope.launch(Dispatchers.IO) {
            val accountPaymentRequest = subscriberUseCase.getAccountPaymentRequest(tripId)
            accountPaymentRequest?.serviceId = serviceId
            subscriberUseCase.sendAccountPaymentRequest(accountPaymentRequest)
        }
    }

    fun isPayment(currentStatus: Int): Boolean = shiftStatusUseCase.isPayment(currentStatus)
    fun isHired(currentStatus: Int): Boolean = shiftStatusUseCase.isHired(currentStatus)

    private fun scaleBitmap(bitmap: Bitmap, newHeight: Int): Bitmap? = try {
        val originalWidth = bitmap.width
        val originalHeight = bitmap.height
        val scaleFactor = newHeight.toFloat() / originalHeight
        val newWidth = (originalWidth * scaleFactor).toInt()
        Bitmap.createScaledBitmap(bitmap, newWidth, newHeight, true)
    } catch (e: Exception) {
        Logs.e(TAG, "scaleBitmap: $e")
        null
    }

    private fun generateFileName(isBill: Boolean, numSubscriber: String): String {
        val nameType = if (isBill) "FACTURA" else "VALE"
        val currentDate = SimpleDateFormat("yyyy-MM-dd HH_mm_ss", Locale.US).format(Date())
        return "${nameType}#${numSubscriber}#${currentDate}#canon.png"
    }

    private fun getBitmapFromUri(uri: Uri): Bitmap? {
        return context.contentResolver.openInputStream(uri)?.use { inputStream ->
            BitmapFactory.decodeStream(inputStream)
        }
    }

    // These are placeholders to keep the file self-contained.
    // Replace with your real BaseViewModel navigation/state accessors.
    private fun isVacant(): Boolean = false
    private fun getCurrentShiftStatus(): Int = 0
    private fun getCurrentDispatch(): InfoDispatchModel? = null
    private fun getCurrentTrip(): Trip? = null
}

@Composable
fun CropImageScreen(
    viewModel: CropImageComposeViewModel,
    sharedViewModel: CropImageMainActivityViewModel,
    serviceId: String?,
    onRequestCameraPermission: () -> Unit,
    onOpenImageSelect: () -> Unit,
    onOpenCropImage: (Uri?) -> Unit,
    onShowToast: (Int) -> Unit,
    onNavigateBack: () -> Unit
) {
    val lifecycleOwner = LocalLifecycleOwner.current
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val showHeader by sharedViewModel.showHeader.collectAsStateWithLifecycle()

    LaunchedEffect(serviceId) {
        viewModel.onEvent(CropImageUiEvent.OnServiceIdChanged(serviceId))
        sharedViewModel.showHeader(true)
        viewModel.onEvent(CropImageUiEvent.OnScreenStarted)
    }

    LaunchedEffect(Unit) {
        lifecycleOwner.repeatOnLifecycle(androidx.lifecycle.Lifecycle.State.STARTED) {
            viewModel.effects.collect { effect ->
                when (effect) {
                    CropImageUiEffect.RequestCameraPermission -> onRequestCameraPermission()
                    CropImageUiEffect.OpenImagePicker -> onOpenImageSelect()
                    is CropImageUiEffect.OpenCropImage -> onOpenCropImage(effect.uri)
                    is CropImageUiEffect.ShowToast -> onShowToast(effect.messageRes)
                    CropImageUiEffect.NavigateBack -> onNavigateBack()
                }
            }
        }
    }

    LaunchedEffect(Unit) {
        lifecycleOwner.repeatOnLifecycle(androidx.lifecycle.Lifecycle.State.STARTED) {
            sharedViewModel.cropImageFlow.collect { uri ->
                uri?.let {
                    viewModel.saveFileFromUri(it)
                }
            }
        }
    }

    CropImageContent(
        showHeader = showHeader,
        imageUri = uiState.imageUri,
        buttons = uiState.buttons,
        isLoading = uiState.isLoading,
        onAccept = { viewModel.onEvent(CropImageUiEvent.OnAcceptClicked) },
        onCrop = { viewModel.onEvent(CropImageUiEvent.OnCropClicked) },
        onSelectImage = { viewModel.onEvent(CropImageUiEvent.OnSelectImageClicked) }
    )
}

@Composable
private fun CropImageContent(
    showHeader: Boolean,
    imageUri: Uri?,
    buttons: CropImageButtonsState,
    isLoading: Boolean,
    onAccept: () -> Unit,
    onCrop: () -> Unit,
    onSelectImage: () -> Unit
) {
    Column(
        modifier = Modifier
            .fillMaxSize()
            .background(Color.White)
            .padding(16.dp),
        verticalArrangement = Arrangement.spacedBy(12.dp)
    ) {
        if (showHeader) {
            Text(
                text = "Header",
                style = MaterialTheme.typography.titleMedium
            )
        }

        Box(
            modifier = Modifier
                .fillMaxWidth()
                .weight(1f)
                .background(Color(0xFFF2F2F2), RoundedCornerShape(12.dp)),
            contentAlignment = Alignment.Center
        ) {
            if (imageUri != null) {
                Image(
                    painter = rememberAsyncImagePainter(imageUri),
                    contentDescription = null,
                    modifier = Modifier.fillMaxSize(),
                    contentScale = ContentScale.Fit
                )
            } else {
                Text(text = "No image selected")
            }

            if (isLoading) {
                CircularProgressIndicator()
            }
        }

        CropImageButton(
            state = buttons.accept,
            onClick = onAccept
        )
        CropImageButton(
            state = buttons.crop,
            onClick = onCrop
        )
        CropImageButton(
            state = buttons.selectImage,
            onClick = onSelectImage
        )
    }
}

@Composable
private fun CropImageButton(
    state: CropButtonVisualState,
    onClick: () -> Unit
) {
    if (!state.visible) return

    val bgColor = when (state.background) {
        CustomButton.BackgroundButtonColor.GREEN -> Color(0xFF2E7D32)
        CustomButton.BackgroundButtonColor.RED -> Color(0xFFC62828)
        CustomButton.BackgroundButtonColor.BLUE -> Color(0xFF1565C0)
        CustomButton.BackgroundButtonColor.YELLOW -> Color(0xFFF9A825)
        CustomButton.BackgroundButtonColor.ORANGE -> Color(0xFFEF6C00)
        CustomButton.BackgroundButtonColor.DEFAULT -> Color(0xFF1565C0)
        CustomButton.BackgroundButtonColor.BLACK -> Color.Black
        CustomButton.BackgroundButtonColor.TRANSPARENT -> Color.Transparent
    }

    val contentColor = when (state.style) {
        CustomButton.StyleButton.ENABLE -> Color.White
        CustomButton.StyleButton.DISABLE -> Color(0xFFBDBDBD)
        CustomButton.StyleButton.LOADING -> Color.White
    }

    Button(
        onClick = onClick,
        enabled = state.style == CustomButton.StyleButton.ENABLE,
        colors = ButtonDefaults.buttonColors(
            containerColor = bgColor,
            contentColor = contentColor,
            disabledContainerColor = Color(0xFFBDBDBD),
            disabledContentColor = Color(0xFFEEEEEE)
        ),
        modifier = Modifier.fillMaxWidth()
    ) {
        Text(text = "res:${state.textRes} icon:${state.iconRes}")
    }
}

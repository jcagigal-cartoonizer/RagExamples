## 4) MainActivity Compose bridge ViewModel

This is the bridge that keeps navigation/dialog behavior centralized, similar to how the fragment used `iMainActivity.openDialog(...)`, `navigateBack()`, and `showHeader(false)`.

```kotlin
package ifac.td.taxi.viewmodel

import androidx.lifecycle.ViewModel
import ifac.td.taxi.ui.custom.dialog.CustomDialog
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.asSharedFlow

sealed interface ChangeUserPasswordMainActivityUiEffect {
    data object HideHeader : ChangeUserPasswordMainActivityUiEffect
    data object NavigateBack : ChangeUserPasswordMainActivityUiEffect
    data class OpenDialog(
        val dialog: CustomDialog.CustomDialogModel,
        val onResult: (CustomDialog.CustomDialogResponse) -> Unit
    ) : ChangeUserPasswordMainActivityUiEffect
}

class ChangeUserPasswordMainActivityViewModel : ViewModel() {

    private val _effects = MutableSharedFlow<ChangeUserPasswordMainActivityUiEffect>()
    val effects = _effects.asSharedFlow()

    suspend fun hideHeader() {
        _effects.emit(ChangeUserPasswordMainActivityUiEffect.HideHeader)
    }

    suspend fun navigateBack() {
        _effects.emit(ChangeUserPasswordMainActivityUiEffect.NavigateBack)
    }

    suspend fun openDialog(
        dialog: CustomDialog.CustomDialogModel,
        onResult: (CustomDialog.CustomDialogResponse) -> Unit
    ) {
        _effects.emit(ChangeUserPasswordMainActivityUiEffect.OpenDialog(dialog, onResult))
    }
}
```

## 5) Compose lifecycle collection for dialog handling

```kotlin
viewLifecycleOwner.lifecycleScope.launch {
```

## 6) How the fragment logic maps to Compose

### Fragment:
- `btnAccept.setAction { if (checkParameters()) { ... } }`

### Compose:
- `CustomButton(onClick = { viewModel.onEvent(AcceptClicked) })`
- validation happens in `handleAcceptClick()`

### Fragment:
- `btnCancel.setAction { iMainActivity.navigateBack() }`

### Compose:
- `viewModel.onEvent(CancelClicked)` -> `NavigateBack` effect

### Fragment:
- `userCredentialsFlow.collect { ... }`

### Compose:
- `effects.collect { ChangePassword / RequestUserPresenter / ShowDialog }`

### Fragment:
- `updateSuccess()` opens success dialog, then navigate back on ACCEPT

### Compose:
- `onPresenterUpdateSuccess()` emits `ShowDialog(...)`
- host handles dialog callback and navigation

---

## 7) Important note about preserving exact navigation

To preserve Jetpack Views navigation exactly:

- keep `onNavigateBack()` delegated to your existing activity/nav controller
- keep `onShowHeader(false)` in the host
- keep `onOpenDialog(...)` using your existing `CustomDialog`
- do not replace the dialog implementation unless you want a Compose dialog

That means the Compose screen should be hosted inside the same activity/nav graph and call the same navigation methods the fragment used.

```kotlin
class ChangeUserPasswordViewModel(
    application: Application,
    private val sessionUseCase: SessionUseCase
) : AndroidViewModel(application) {

    private val context = application

    private val _uiState = MutableStateFlow(ChangeUserPasswordUiState())
    val uiState = _uiState.asStateFlow()

    private val _buttonsState = MutableStateFlow(ChangeUserPasswordButtonsState.reducer())
    val buttonsState = _buttonsState.asStateFlow()

    private val _effects = MutableSharedFlow<ChangeUserPasswordUiEffect>()
    val effects = _effects.asSharedFlow()

    fun onEvent(event: ChangeUserPasswordUiEvent) {
        when (event) {
            is ChangeUserPasswordUiEvent.CurrentPasswordChanged ->
                _uiState.update { it.copy(currentPassword = event.value) }

            is ChangeUserPasswordUiEvent.NewPasswordChanged ->
                _uiState.update { it.copy(newPassword = event.value) }

            is ChangeUserPasswordUiEvent.RepeatPasswordChanged ->
                _uiState.update { it.copy(repeatPassword = event.value) }

            ChangeUserPasswordUiEvent.AcceptClicked -> validateAndProceed()

            ChangeUserPasswordUiEvent.CancelClicked -> {
                viewModelScope.launch { _effects.emit(ChangeUserPasswordUiEffect.NavigateBack) }
            }
        }
    }

    private fun validateAndProceed() {
        val currentUserPwd = uiState.value.currentPassword
        val newUserPwd = uiState.value.newPassword
        val repeatedUserPwd = uiState.value.repeatPassword

        val dialogAllInputs = CustomDialog.CustomDialogModel(
            title = context.getString(R.string.dialog_fill_inputs),
            buttons = arrayListOf(ButtonType.ACCEPT)
        )
        val dialogInvalidNewPin = CustomDialog.CustomDialogModel(
            title = context.getString(R.string.dialog_no_match_password),
            buttons = arrayListOf(ButtonType.ACCEPT)
        )
        val dialogSameAndOldNewPin = CustomDialog.CustomDialogModel(
            title = context.getString(R.string.dialog_equal_password),
            buttons = arrayListOf(ButtonType.ACCEPT)
        )

        val toastCallback: (CustomDialog.CustomDialogResponse) -> Unit = { response ->
            if (response.buttonPressed == ButtonType.ACCEPT) {
                // host can show toast if desired
            }
        }

        when {
            currentUserPwd.isEmpty() || newUserPwd.isEmpty() || repeatedUserPwd.isEmpty() -> {
                viewModelScope.launch {
                    _effects.emit(ChangeUserPasswordUiEffect.OpenDialog(dialogAllInputs, toastCallback))
                }
            }

            newUserPwd != repeatedUserPwd -> {
                viewModelScope.launch {
                    _effects.emit(ChangeUserPasswordUiEffect.OpenDialog(dialogInvalidNewPin, toastCallback))
                }
            }

            currentUserPwd == newUserPwd -> {
                viewModelScope.launch {
                    _effects.emit(ChangeUserPasswordUiEffect.OpenDialog(dialogSameAndOldNewPin, toastCallback))
                }
            }

            else -> {
                viewModelScope.launch {
                    _effects.emit(ChangeUserPasswordUiEffect.RequestUserPresenter)
                }
            }
        }
    }

    fun getUserPresenter(changePasswordPinView: ChangePasswordPinView) {
        viewModelScope.launch {
            if (sessionUseCase.isUserLoggedIn()) {
                val presenter = UserModule.provideUserPresenter(changePasswordPinView, context)
                _effects.emit(
                    ChangeUserPasswordUiEffect.ChangePassword(
                        userPresenter = presenter,
                        newPassword = uiState.value.newPassword.trim()
                    )
                )
            } else {
                val dialogNoUser = CustomDialog.CustomDialogModel(
                    title = context.getString(R.string.dialog_user_error),
                    buttons = arrayListOf(ButtonType.ACCEPT)
                )

                val callback: (CustomDialog.CustomDialogResponse) -> Unit = { response ->
                    if (response.buttonPressed == ButtonType.ACCEPT) {
                        // host navigates back
                    }
                }

                _effects.emit(ChangeUserPasswordUiEffect.OpenDialog(dialogNoUser, callback))
            }
        }
    }

    fun changeUserPassword(userPresenter: UserPresenter, newPassword: String) {
        userPresenter.changePassword(newPassword, context)
    }

    fun onPresenterUpdateSuccess() {
        viewModelScope.launch {
            val dialogSuccess = CustomDialog.CustomDialogModel(
                title = context.getString(R.string.dialog_change_password),
                buttons = arrayListOf(ButtonType.ACCEPT)
            )
            val callback: (CustomDialog.CustomDialogResponse) -> Unit = { response ->
                if (response.buttonPressed == ButtonType.ACCEPT) {
                    // host navigates back
                }
            }
            _effects.emit(ChangeUserPasswordUiEffect.OpenDialog(dialogSuccess, callback))
        }
    }

    fun onPresenterUpdateFailure() {
        viewModelScope.launch {
            val dialogFailed = CustomDialog.CustomDialogModel(
                title = context.getString(R.string.dialog_error_change_password),
                buttons = arrayListOf(ButtonType.ACCEPT)
            )
            _effects.emit(ChangeUserPasswordUiEffect.OpenDialog(dialogFailed) { })
        }
    }
}
```

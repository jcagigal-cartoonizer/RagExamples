```kotlin
package ifac.td.taxi.viewmodel
// # Block 52-2: import android.app.Application
import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.domain.usecase.SessionUseCase
import ifac.td.taxi.ui.screen.SplashScreenButtonsState
import ifac.td.taxi.ui.screen.SplashScreenDialogState
import ifac.td.taxi.ui.screen.SplashScreenUiEffect
import ifac.td.taxi.ui.screen.SplashScreenUiEvent
import ifac.td.taxi.ui.screen.SplashScreenUiState
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.delay
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch
class SplashScreenComposeViewModel(
    application: Application,
    private val sessionUseCase: SessionUseCase
) : AndroidViewModel(application) {
    private val _uiState = MutableStateFlow(SplashScreenUiState())
    val uiState = _uiState.asStateFlow()
    private val _uiEffect = kotlinx.coroutines.flow.MutableSharedFlow<SplashScreenUiEffect>()
    val uiEffect: SharedFlow<SplashScreenUiEffect> = _uiEffect.asSharedFlow()
    fun onEvent(event: SplashScreenUiEvent) {
        when (event) {
            SplashScreenUiEvent.OnScreenStarted -> {
                // no-op here; UI triggers checkIsReconnecting separately or via init
            }
            SplashScreenUiEvent.OnVideoCompleted,
            SplashScreenUiEvent.OnVideoError,
            SplashScreenUiEvent.OnTimeoutReached -> {
                _uiState.update { it.copy(isVideoVisible = false, isLoading = false) }
            }
            SplashScreenUiEvent.OnBackPressed -> {
                // intentionally ignored to match fragment behavior
            }
            SplashScreenUiEvent.OnDialogConfirm -> {
                // only used if you later add a dialog
            }
            SplashScreenUiEvent.OnDialogDismiss -> {
                // only used if you later add a dialog
            }
        }
    }
    fun checkIsReconnecting(reconnectionFlowValue: Boolean) {
        viewModelScope.launch(Dispatchers.IO) {
            val serviceWasActive = sessionUseCase.getServiceWasActive()
            delay(50)
            // Preserve original behavior:
            // if service was active, the splash should be skipped and navigation handled by UI.
            _uiEffect.emit(
                if (serviceWasActive) SplashScreenUiEffect.HideSplashVideo
                else SplashScreenUiEffect.NavigateToWelcome
            )
        }
    }
    fun showLegalTextWindow(showLegal: Boolean) {
        viewModelScope.launch {
            _uiEffect.emit(
                if (showLegal) SplashScreenUiEffect.NavigateToLegalText
                else SplashScreenUiEffect.NavigateToWelcome
            )
        }
    }
    fun updateButtonsState(state: SplashScreenButtonsState) {
        _uiState.update { it.copy(buttonsState = state) }
    }
    fun updateDialogState(state: SplashScreenDialogState?) {
        _uiState.update { it.copy(dialogState = state) }
    }
}
```

```kotlin
package ifac.td.taxi.ui.screen
// # Block 169-4: import android.graphics.Color
import android.graphics.Color

import android.widget.MediaController
import android.widget.VideoView
import androidx.activity.compose.BackHandler
import androidx.compose.foundation.background
import androidx.compose.foundation.layout.Box
import androidx.compose.foundation.layout.fillMaxSize
import androidx.compose.runtime.Composable
import androidx.compose.runtime.LaunchedEffect
import androidx.compose.runtime.getValue
import androidx.compose.runtime.remember
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color as Color
import androidx.compose.ui.platform.LocalContext
import androidx.compose.ui.viewinterop.AndroidView
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.navigation.NavController
import ifac.td.taxi.R
import ifac.td.taxi.viewmodel.SplashScreenViewModel
import kotlinx.coroutines.delay
import kotlinx.coroutines.flow.collectLatest
@Composable
fun SplashScreenRoute(
    navController: NavController,
    viewModel: SplashScreenComposeViewModel,
    reconnectionFlowValue: Boolean,
    showHeader: (Boolean) -> Unit,
    showBottomBar: (Boolean) -> Unit,
    getBravoService: () -> Any?
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current
    BackHandler(enabled = true) {
        // Don't let the user navigate back
    }
    LaunchedEffect(Unit) {
        showHeader(false)
        showBottomBar(false)
        viewModel.checkIsReconnecting(reconnectionFlowValue)
    }
    LaunchedEffect(viewModel) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                SplashScreenUiEffect.HideSplashVideo -> {
                    if (getBravoService() == null) {
                        navController.navigate(R.id.action_splashScreenFragment_to_welcomeFragment) {
                            launchSingleTop = true
                        }
                    }
                }
                SplashScreenUiEffect.NavigateToWelcome -> {
                    navController.navigate(R.id.action_splashScreenFragment_to_welcomeFragment) {
                        launchSingleTop = true
                    }
                }
                SplashScreenUiEffect.NavigateToLegalText -> {
                    navController.navigate(R.id.action_splashScreenFragment_to_legalTextFragment) {
                        launchSingleTop = true
                    }
                }
            }
        }
    }
    LaunchedEffect(Unit) {
        delay(4800L)
        viewModel.onEvent(SplashScreenUiEvent.OnTimeoutReached)
        viewModel.showLegalTextWindow(showLegal = false)
    }
    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(Color(0xFF000000))
    ) {
        if (uiState.isVideoVisible) {
            AndroidView(
                factory = { ctx ->
                    VideoView(ctx).apply {
                        setZOrderOnTop(true)
                        setBackgroundColor(ctx.getColor(R.color.splash_background))
                        val videoUri = Uri.parse(
                            "android.resource://${ctx.packageName}/${R.raw.new_splash}"
                        )
                        setOnPreparedListener { mediaPlayer ->
                            setBackgroundColor(Color.Transparent)
                            mediaPlayer.isLooping = false
                            mediaPlayer.start()
                        }
                        setOnCompletionListener {
                            viewModel.onEvent(SplashScreenUiEvent.OnVideoCompleted)
                            viewModel.showLegalTextWindow(showLegal = false)
                        }
                        setOnErrorListener { _, _, _ ->
                            viewModel.onEvent(SplashScreenUiEvent.OnVideoError)
                            viewModel.showLegalTextWindow(showLegal = false)
                            true
                        }
                        try {
                            setVideoURI(videoUri)
                            start()
                        } catch (_: Exception) {
                            viewModel.onEvent(SplashScreenUiEvent.OnVideoError)
                            viewModel.showLegalTextWindow(showLegal = false)
                        }
                    }
                },
                modifier = Modifier.fillMaxSize()
            )
        }
    }
}
```

## Reducer that mirrors fragment branches one-by-one
```kotlin
package ifac.td.taxi.ui.screen
object SplashScreenButtonsReducer {
    fun initial(): SplashScreenButtonsState = SplashScreenButtonsState(
        isVideoVisible = true,
        isSkipped = false,
        isHeaderVisible = false,
        isBottomBarVisible = false,
        showLegalTextWindow = null,
        navigationTarget = null,
        timeoutArmed = false
    )
    fun onSetupComponents(): SplashScreenButtonsState {
        return initial().copy(
            isHeaderVisible = false,
            isBottomBarVisible = false
        )
    }
    fun onVideoPrepared(): SplashScreenButtonsState {
        return initial().copy(
            isVideoVisible = true,
            timeoutArmed = true
        )
    }
    fun onVideoCompleted(): SplashScreenButtonsState {
        return initial().copy(
            isVideoVisible = false
        )
    }
    fun onVideoError(): SplashScreenButtonsState {
        return initial().copy(
            isVideoVisible = false
        )
    }
    fun onTimeoutReached(): SplashScreenButtonsState {
        return initial().copy(
            isVideoVisible = false
        )
    }
    fun onEndSplash(): SplashScreenButtonsState {
        return initial().copy(
            isVideoVisible = false,
            showLegalTextWindow = false,
            navigationTarget = null
        )
    }
    fun onEndSplashWithoutCheck(): SplashScreenButtonsState {
        return initial().copy(
            isVideoVisible = false,
            isSkipped = true,
            timeoutArmed = false,
            showLegalTextWindow = true,
            navigationTarget = SplashScreenNavigationTarget.Welcome
        )
    }
    fun onShowLegalTextFlow(showLegal: Boolean): SplashScreenButtonsState {
        return initial().copy(
            navigationTarget = if (showLegal) {
                SplashScreenNavigationTarget.LegalText
            } else {
                SplashScreenNavigationTarget.Welcome
            }
        )
    }
    fun onServiceActiveFlow(
        shouldSkipSplash: Boolean,
        bravoServiceExists: Boolean
    ): SplashScreenButtonsState {
        return if (shouldSkipSplash && !bravoServiceExists) {
            onEndSplashWithoutCheck()
        } else {
            initial()
        }
    }
    fun onBackPressed(): SplashScreenButtonsState {
        // fragment ignored back press
        return initial()
    }
}
```

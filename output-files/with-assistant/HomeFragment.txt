
// FIRST ANSWER
Below is a Compose-first rewrite that preserves the **same navigation behavior**, replaces the fragment button logic with a **single `HomeButtonsState`**, and uses a **`SharedFlow<HomeUiEffect>`** for one-off actions like navigation, toast, dialogs, beep, and screen-on.

I’m keeping this **Android + Jetpack Compose only** in the code structure, and I’m modeling the dialog handling so it is only used where the old code used `CustomDialog`.

# 1) Compose UI: `HomeScreen`

This composable replaces the fragment UI logic and button state handling.

```kotlin
package ifac.td.taxi.ui.screen

import androidx.compose.foundation.layout.*
import androidx.compose.runtime.*
import androidx.compose.ui.Modifier
import androidx.compose.ui.unit.dp
import androidx.lifecycle.Lifecycle
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.lifecycle.compose.LocalLifecycleOwner
import androidx.lifecycle.repeatOnLifecycle
import androidx.navigation.NavController
import kotlinx.coroutines.launch

@Composable
fun HomeScreen(
    navController: NavController,
    viewModel: HomeComposeViewModel,
    sharedViewModel: MainActivityComposeViewModel,
    onShowToast: (Int) -> Unit,
    onKeepScreenActive: (Boolean) -> Unit,
    onShowHeader: (Boolean) -> Unit,
    onConfigureTopBarExit: (visible: Boolean, onClick: () -> Unit) -> Unit,
    onConfigureTopBarNext: (visible: Boolean, onClick: () -> Unit) -> Unit,
    onOpenCustomDialog: (HomeDialogState) -> Unit,
    onDismissDialog: () -> Unit,
) {
    val lifecycleOwner = LocalLifecycleOwner.current
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val buttonsState by viewModel.buttonsState.collectAsStateWithLifecycle()

    LaunchedEffect(Unit) {
        onShowHeader(true)
        viewModel.onScreenStarted()
        sharedViewModel.resetDispatchFlow()
        sharedViewModel.resetCurrentAmountFlow()
    }

    LaunchedEffect(lifecycleOwner) {
        lifecycleOwner.repeatOnLifecycle(Lifecycle.State.STARTED) {
            viewModel.effects.collect { effect ->
                when (effect) {
                    is HomeUiEffect.Navigate -> navController.navigate(effect.route)
                    is HomeUiEffect.NavigateDeepLink -> navController.navigate(effect.request)
                    is HomeUiEffect.ShowToast -> onShowToast(effect.messageRes)
                    is HomeUiEffect.KeepScreenActive -> onKeepScreenActive(effect.keepOn)
                    is HomeUiEffect.ConfigureTopBarExit ->
                        onConfigureTopBarExit(effect.visible, effect.onClick)
                    is HomeUiEffect.ConfigureTopBarNext ->
                        onConfigureTopBarNext(effect.visible, effect.onClick)
                    is HomeUiEffect.OpenDialog -> onOpenCustomDialog(effect.dialog)
                    is HomeUiEffect.DismissDialog -> onDismissDialog()
                    is HomeUiEffect.Beep -> {
                        // handled by host activity if needed
                    }
                    is HomeUiEffect.LocationEnabledManually -> {
                        // handled by host activity if needed
                    }
                }
            }
        }
    }

    LaunchedEffect(Unit) {
        viewModel.bindSharedState(
            locationEnabled = sharedViewModel.locationEnabledFlow.value,
            roofLight = sharedViewModel.roofLightFlow.value,
            locatedOnStop = sharedViewModel.locatedOnStop.value,
            locationType = sharedViewModel.locationType.value,
            hasTaximeterConnection = sharedViewModel.hasTaximeterConnectionFlow.value,
            orangeBtnPending = sharedViewModel.orangeBtnPendingFlow.value,
            shortBreakStatus = sharedViewModel.shortBreakStatus.value,
            zone = sharedViewModel.zoneFlow.value,
            bravoVoice = sharedViewModel.bravoStateFlow.value.voiceValue,
            pendingTripsEmpty = sharedViewModel.pendingTripsListFlow.value?.isEmpty() == true
        )
    }

    HomeContent(
        state = buttonsState,
        onAction = { action -> viewModel.onAction(action) }
    )

    if (uiState.dialog != null) {
        HomeDialogHost(
            dialog = uiState.dialog,
            onDismiss = { viewModel.onDialogResult(HomeDialogResult.Dismiss) },
            onConfirm = { viewModel.onDialogResult(HomeDialogResult.Confirm) }
        )
    }
}
```

# 5) Compose ViewModel with `SharedFlow<HomeUiEffect>`

This is the Compose replacement for the fragment + old viewmodel behavior, while preserving navigation semantics.

```kotlin
package ifac.td.taxi.ui.screen

import android.app.Application
import androidx.core.net.toUri
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import androidx.navigation.NavDeepLinkRequest
import com.interfacom.sdk.taximeter.bravocomm.W2CLocation
import ifac.td.taxi.R
import ifac.td.taxi.repository.connections.service.model.ShortBreakStatus
import ifac.td.taxi.viewmodel.model.MessageUIEnum
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch

class HomeComposeViewModel(
    application: Application,
    private val delegate: HomeViewModel,
    private val sharedViewModel: MainActivityComposeViewModel,
) : AndroidViewModel(application) {

    private val _uiState = MutableStateFlow(HomeUiState())
    val uiState: StateFlow<HomeUiState> = _uiState.asStateFlow()

    private val _buttonsState = MutableStateFlow(HomeButtonsState())
    val buttonsState: StateFlow<HomeButtonsState> = _buttonsState.asStateFlow()

    private val _effects = MutableSharedFlow<HomeUiEffect>(extraBufferCapacity = 64)
    val effects: SharedFlow<HomeUiEffect> = _effects.asSharedFlow()

    fun onScreenStarted() {
        delegate.getMessages()
        delegate.checkKeepScreenOn()
        delegate.checkForLocationPermission()
        delegate.checkPendingServiceOnForHirePermission()
    }

    fun bindSharedState(
        locationEnabled: Pair<Boolean, Boolean>,
        roofLight: Boolean?,
        locatedOnStop: Boolean,
        locationType: String,
        hasTaximeterConnection: Boolean,
        orangeBtnPending: Boolean?,
        shortBreakStatus: ShortBreakStatus?,
        zone: String?,
        bravoVoice: Boolean,
        pendingTripsEmpty: Boolean
    ) {
        val locationAllowed = W2CLocation.isLocationAllowedByCentral()
        val showFixedPrice = delegate.showEstimateFixedPriceButtonFlow.value
        val dashboardVisible = delegate.hasDashboardButtonOn.value

        val locationButton = if (!locationAllowed) {
            HomeButtonState(visible = true, enabled = false, style = HomeButtonStyle.Empty)
        } else if (locationEnabled.first) {
            HomeButtonState(
                enabled = true,
                background = HomeButtonBackground.Green,
                textRes = R.string.btn_location_off
            )
        } else {
            HomeButtonState(
                enabled = true,
                background = HomeButtonBackground.Red,
                textRes = R.string.btn_location_on
            )
        }

        val messagesButton = when (delegate.hasMessagesFlow.value) {
            MessageUIEnum.NO_MESSAGES -> HomeButtonState(enabled = false, background = HomeButtonBackground.Blue)
            MessageUIEnum.HAS_MESSAGES_BUT_NOT_NEW -> HomeButtonState(enabled = true, background = HomeButtonBackground.Blue)
            MessageUIEnum.HAS_NEW_MESSAGES -> HomeButtonState(enabled = true, background = HomeButtonBackground.Orange)
            null -> HomeButtonState(enabled = false, background = HomeButtonBackground.Blue)
        }

        val roofLightButton = when {
            roofLight == null -> HomeButtonState(enabled = false)
            delegate.roofLightFlow.value?.second != true -> HomeButtonState(enabled = false)
            roofLight -> HomeButtonState(enabled = true, background = HomeButtonBackground.Green)
            else -> HomeButtonState(enabled = true, background = HomeButtonBackground.Red)
        }

        val dashboardButton = HomeButtonState(visible = dashboardVisible, enabled = true)

        val fixedPriceButton = HomeButtonState(visible = showFixedPrice, enabled = true)
        val receiptsButton = HomeButtonState(visible = !showFixedPrice, enabled = true)

        val centralButton = HomeButtonState(
            enabled = true,
            background = if (bravoVoice) HomeButtonBackground.Blue else HomeButtonBackground.Red
        )

        val zoningEnabled = delegate.zoningButtonStateFlow.value
        val zoningButton = HomeButtonState(enabled = zoningEnabled)

        val locateStandButton = when (locationType) {
            "Z" -> {
                if (locatedOnStop || W2CLocation.locationStopAllowed()) {
                    HomeButtonState(
                        enabled = true,
                        background = HomeButtonBackground.Red,
                        textRes = R.string.btn_locate_stop
                    )
                } else {
                    HomeButtonState(enabled = false)
                }
            }
            "P" -> {
                if (!delegate.checkStartAutomaticLocation()) {
                    HomeButtonState(
                        enabled = true,
                        background = HomeButtonBackground.Green,
                        textRes = R.string.btn_locate_stop_exit
                    )
                } else {
                    HomeButtonState(enabled = false)
                }
            }
            else -> HomeButtonState(enabled = false)
        }

        val pendingEnabled = computePendingEnabled(
            zone = zone,
            shortBreakStatus = shortBreakStatus,
            pendingServicesButton = delegate.pendingServicesButtonFlow.value,
            locationAllowed = locationAllowed
        )

        val pendingButton = HomeButtonState(
            enabled = pendingEnabled,
            background = if (orangeBtnPending == true) HomeButtonBackground.Orange else HomeButtonBackground.Blue
        )

        _buttonsState.value = HomeButtonsState(
            zoning = zoningButton,
            pending = pendingButton,
            location = locationButton,
            receipts = receiptsButton,
            messages = messagesButton,
            central = centralButton,
            dashboard = dashboardButton,
            fixedPrice = fixedPriceButton,
            roofLight = roofLightButton,
            locateStand = locateStandButton
        )
    }

    private fun computePendingEnabled(
        zone: String?,
        shortBreakStatus: ShortBreakStatus?,
        pendingServicesButton: Boolean?,
        locationAllowed: Boolean
    ): Boolean {
        if (zone.isNullOrEmpty()) return false
        val isInShortBreak = shortBreakStatus == ShortBreakStatus.IN_SHORT_BREAK ||
                shortBreakStatus == ShortBreakStatus.IN_SHORT_BREAK_FORCED
        if (isInShortBreak) return false
        if (!locationAllowed) return false
        if (W2CLocation.getIsInSoonInZone()) return true
        if (pendingServicesButton == true) return true
        return false
    }

    fun onAction(action: HomeAction) {
        when (action) {
            HomeAction.Zoning -> {
                sharedViewModel.emitManualZoningNavigation(true)
                delegate.navigateToZoning()
            }
            HomeAction.Pending -> delegate.canOpenPendingTripsFragment(sharedViewModel._pendingTripsListFlow)
            HomeAction.Location -> handleLocationClick()
            HomeAction.Receipts -> delegate.navigateToReceiptHistory()
            HomeAction.Messages -> delegate.navigateToMessage()
            HomeAction.Central -> delegate.navigateToContactCentral()
            HomeAction.Dashboard -> emitEffect(HomeUiEffect.Navigate(R.id.action_homeFragment_to_dashboardFragment))
            HomeAction.FixedPrice -> delegate.navigateToEstimateFixedPriceScreen()
            HomeAction.RoofLight -> handleRoofLightClick()
            HomeAction.LocateStand -> handleLocateStandClick()
        }
    }

    private fun handleLocationClick() {
        val enabled = sharedViewModel.locationEnabledFlow.value
        if (enabled.second) {
            emitEffect(
                HomeUiEffect.OpenDialog(
                    HomeDialogState(
                        titleRes = R.string.confirm_deactivate_location,
                        kind = HomeDialogKind.DeactivateLocation
                    )
                )
            )
        } else {
            if (W2CLocation.isLocationAllowedByCentral()) {
                emitEffect(
                    HomeUiEffect.OpenDialog(
                        HomeDialogState(
                            titleRes = R.string.confirm_activate_location,
                            kind = HomeDialogKind.ActivateLocation
                        )
                    )
                )
            }
        }
    }

    private fun handleRoofLightClick() {
        val roof = delegate.roofLightFlow.value
        if (roof?.second == true) {
            if (roof.first == true) delegate.setRoofLightOff() else delegate.setRoofLightOn()
        }
    }

    private fun handleLocateStandClick() {
        if (W2CLocation.getLocationTypeSent() == 'P') {
            emitEffect(
                HomeUiEffect.OpenDialog(
                    HomeDialogState(
                        titleRes = R.string.dialog_exit_onStand_title,
                        kind = HomeDialogKind.ExitStand
                    )
                )
            )
        } else {
            W2CLocation.sendLocateStop()
            W2CLocation.setTryingToRelocateInStand(false)
        }
    }

    fun onDialogResult(result: HomeDialogResult) {
        val dialog = uiState.value.dialog ?: return
        when (dialog.kind) {
            HomeDialogKind.ActivateLocation -> if (result is HomeDialogResult.Confirm) {
                sharedViewModel.updateLocationEnabledFlow(enabled = true, canEnable = true)
                sharedViewModel.updateLocationManuallyDeLocatedFlow(false)
                emitEffect(HomeUiEffect.LocationEnabledManually)
            }
            HomeDialogKind.DeactivateLocation -> if (result is HomeDialogResult.Confirm) {
                sharedViewModel.updateLocationEnabledFlow(enabled = false, canEnable = true)
                sharedViewModel.updateLocationManuallyDeLocatedFlow(true)
                delegate.checkForHireAfterDelocation()
                emitEffect(HomeUiEffect.Beep)
                emitEffect(HomeUiEffect.LocationEnabledManually)
            }
            HomeDialogKind.ExitShift -> if (result is HomeDialogResult.Confirm) {
                delegate.logoff()
                val deepLink = NavDeepLinkRequest.Builder
                    .fromUri("android-app://ifac.td.taxi/welcome".toUri())
                    .build()
                emitEffect(HomeUiEffect.NavigateDeepLink(deepLink))
            }
            HomeDialogKind.ManualTrip -> if (result is HomeDialogResult.Confirm) {
                delegate.changeStateHiredManual()
            }
            HomeDialogKind.ExitStand -> if (result is HomeDialogResult.Confirm) {
                W2CLocation.sendLocateZone()
            }
        }
        _uiState.update { it.copy(dialog = null) }
        emitEffect(HomeUiEffect.DismissDialog)
    }

    private fun emitEffect(effect: HomeUiEffect) {
        viewModelScope.launch { _effects.emit(effect) }
    }
}

enum class HomeAction {
    Zoning, Pending, Location, Receipts, Messages, Central, Dashboard, FixedPrice, RoofLight, LocateStand
}
```

# 9) Important mapping notes

These are the exact fragment behaviors preserved in the state model:

- **Messages**
  - `NO_MESSAGES` → disabled
  - `HAS_MESSAGES_BUT_NOT_NEW` → enabled + blue
  - `HAS_NEW_MESSAGES` → enabled + orange

- **Location**
  - enabled + green when location is on
  - enabled + red when location is off
  - disabled when permission/config says so

- **Roof light**
  - enabled + green when on
  - enabled + red when off
  - disabled when not available or luminous disabled is false

- **Pending**
  - disabled if no zone
  - disabled in short break
  - disabled if location not allowed by central
  - enabled if soon-in-zone
  - enabled if pendingServicesButton is true
  - orange if `orangeBtnPending == true`, otherwise blue

- **Dashboard**
  - visible only when config says so

- **Fixed price / receipts**
  - mutually exclusive visibility

- **Locate stand**
  - exact text/background logic based on `"Z"` / `"P"`

## 1.4 Exact reducer implementation

This mirrors the fragment logic branch-by-branch.

```kotlin
package ifac.td.taxi.ui.screen

import com.interfacom.sdk.taximeter.bravocomm.W2CLocation
import ifac.td.taxi.R
import ifac.td.taxi.repository.connections.service.model.ShortBreakStatus
import ifac.td.taxi.viewmodel.model.MessageUIEnum

class HomeButtonsReducer {

    fun reduce(
        current: HomeUiState,
        action: HomeReducerAction
    ): HomeUiState {
        return when (action) {
            is HomeReducerAction.ZoneChanged -> {
                val buttons = if (action.zone.isNullOrBlank()) {
                    current.buttons.copy(
                        btnPending = current.buttons.btnPending.copy(
                            enabled = false,
                            style = HomeButtonStyle.Disable
                        )
                    )
                } else {
                    current.buttons.copy(
                        btnPending = reducePendingButton(
                            current = current,
                            zone = action.zone
                        )
                    )
                }
                current.copy(zone = action.zone, buttons = buttons)
            }

            is HomeReducerAction.ShortBreakChanged -> {
                val buttons = current.buttons.copy(
                    btnPending = reducePendingButton(
                        current = current.copy(shortBreakStatus = action.status)
                    )
                )
                current.copy(shortBreakStatus = action.status, buttons = buttons)
            }

            is HomeReducerAction.PendingServicesButtonChanged -> {
                val buttons = current.buttons.copy(
                    btnPending = reducePendingButton(
                        current = current.copy(pendingServicesButton = action.value)
                    )
                )
                current.copy(pendingServicesButton = action.value, buttons = buttons)
            }

            is HomeReducerAction.OrangePendingChanged -> {
                current.copy(
                    orangeBtnPending = action.value,
                    buttons = current.buttons.copy(
                        btnPending = current.buttons.btnPending.copy(
                            background = if (action.value == true) {
                                HomeButtonBackground.Orange
                            } else {
                                HomeButtonBackground.Blue
                            }
                        )
                    )
                )
            }

            is HomeReducerAction.DashboardButtonChanged -> {
                current.copy(
                    hasDashboardButtonOn = action.visible,
                    buttons = current.buttons.copy(
                        btnDashboard = current.buttons.btnDashboard.copy(visible = action.visible)
                    )
                )
            }

            is HomeReducerAction.FixedPriceButtonChanged -> {
                current.copy(
                    showEstimateFixedPriceButton = action.visible,
                    buttons = current.buttons.copy(
                        btnFixedPrice = current.buttons.btnFixedPrice.copy(visible = action.visible),
                        btnReceipts = current.buttons.btnReceipts.copy(visible = !action.visible)
                    )
                )
            }

            is HomeReducerAction.ZoningButtonChanged -> {
                current.copy(
                    buttons = current.buttons.copy(
                        btnZoning = current.buttons.btnZoning.copy(
                            enabled = action.enabled,
                            style = if (action.enabled) HomeButtonStyle.Enable else HomeButtonStyle.Disable
                        )
                    )
                )
            }

            is HomeReducerAction.LocationEnabledChanged -> {
                current.copy(
                    locationEnabled = action.value,
                    buttons = current.buttons.copy(
                        btnLocation = current.buttons.btnLocation.copy(
                            enabled = action.value.first,
                            background = if (action.value.first) HomeButtonBackground.Green else HomeButtonBackground.Red,
                            textRes = if (action.value.first) R.string.btn_location_off else R.string.btn_location_on
                        )
                    )
                )
            }

            is HomeReducerAction.LocationTypeChanged -> {
                current.copy(
                    locationType = action.value,
                    buttons = current.buttons.copy(
                        btnLocateStand = reduceLocateStandButton(current.copy(locationType = action.value))
                    )
                )
            }

            is HomeReducerAction.LocatedOnStopChanged -> {
                current.copy(
                    locatedOnStop = action.value,
                    buttons = current.buttons.copy(
                        btnLocateStand = reduceLocateStandButton(current.copy(locatedOnStop = action.value))
                    )
                )
            }

            is HomeReducerAction.RoofLightChanged -> {
                current.copy(
                    roofLight = action.value,
                    buttons = current.buttons.copy(
                        btnRoofLight = reduceRoofLightButton(action.value)
                    )
                )
            }

            is HomeReducerAction.MessagesChanged -> {
                current.copy(
                    messageState = action.value,
                    buttons = current.buttons.copy(
                        btnMessages = when (action.value) {
                            MessageUIEnum.NO_MESSAGES -> HomeButtonState(
                                enabled = false,
                                style = HomeButtonStyle.Disable,
                                background = HomeButtonBackground.Blue
                            )
                            MessageUIEnum.HAS_MESSAGES_BUT_NOT_NEW -> HomeButtonState(
                                enabled = true,
                                style = HomeButtonStyle.Enable,
                                background = HomeButtonBackground.Blue
                            )
                            MessageUIEnum.HAS_NEW_MESSAGES -> HomeButtonState(
                                enabled = true,
                                style = HomeButtonStyle.Enable,
                                background = HomeButtonBackground.Orange
                            )
                            null -> HomeButtonState(
                                enabled = false,
                                style = HomeButtonStyle.Disable,
                                background = HomeButtonBackground.Blue
                            )
                        }
                    )
                )
            }

            is HomeReducerAction.BravoStateChanged -> {
                current.copy(
                    bravoVoice = action.voiceValue,
                    buttons = current.buttons.copy(
                        btnNoClient = current.buttons.btnNoClient.copy(
                            background = if (action.clientButton) HomeButtonBackground.Green else HomeButtonBackground.Red,
                            enabled = action.noClientButtonEnabled,
                            style = if (action.noClientButtonEnabled) HomeButtonStyle.Enable else HomeButtonStyle.Disable
                        ),
                        btnReturn = current.buttons.btnReturn.copy(
                            enabled = action.showReturnDispatchButton,
                            style = if (action.showReturnDispatchButton) HomeButtonStyle.Enable else HomeButtonStyle.Disable
                        ),
                        btnVoiceCall = current.buttons.btnVoiceCall.copy(
                            enabled = action.bridgeCallState != 0,
                            style = if (action.bridgeCallState == 0) HomeButtonStyle.Disable else HomeButtonStyle.Enable,
                            background = when (action.bridgeCallState) {
                                2 -> HomeButtonBackground.Red
                                0 -> HomeButtonBackground.Green
                                else -> HomeButtonBackground.Orange
                            }
                        )
                    )
                )
            }

            is HomeReducerAction.HasTaximeterConnectionChanged -> {
                current.copy(hasTaximeterConnection = action.value)
            }

            is HomeReducerAction.KeepScreenOnChanged -> {
                current.copy(keepScreenOn = action.value)
            }

            is HomeReducerAction.TimerChanged -> {
                current.copy(timerText = action.value.orEmpty())
            }
        }
    }

    private fun reducePendingButton(current: HomeUiState, zone: String? = current.zone): HomeButtonState {
        if (zone.isNullOrEmpty()) {
            return current.buttons.btnPending.copy(enabled = false, style = HomeButtonStyle.Disable)
        }

        val isInShortBreak =
            current.shortBreakStatus == ShortBreakStatus.IN_SHORT_BREAK ||
            current.shortBreakStatus == ShortBreakStatus.IN_SHORT_BREAK_FORCED

        if (isInShortBreak) {
            return current.buttons.btnPending.copy(enabled = false, style = HomeButtonStyle.Disable)
        }

        if (!W2CLocation.isLocationAllowedByCentral()) {
            return current.buttons.btnPending.copy(enabled = false, style = HomeButtonStyle.Disable)
        }

        if (W2CLocation.getIsInSoonInZone()) {
            return current.buttons.btnPending.copy(enabled = true, style = HomeButtonStyle.Enable)
        }

        if (current.pendingServicesButton == true) {
            return current.buttons.btnPending.copy(enabled = true, style = HomeButtonStyle.Enable)
        }

        return current.buttons.btnPending.copy(enabled = false, style = HomeButtonStyle.Disable)
    }

    private fun reduceLocateStandButton(current: HomeUiState): HomeButtonState {
        return when (current.locationType) {
            "Z" -> {
                if (current.locatedOnStop || W2CLocation.locationStopAllowed()) {
                    HomeButtonState(
                        enabled = true,
                        style = HomeButtonStyle.Enable,
                        background = HomeButtonBackground.Red,
                        textRes = R.string.btn_locate_stop
                    )
                } else {
                    HomeButtonState(enabled = false, style = HomeButtonStyle.Disable)
                }
            }

            "P" -> {
                if (!current.buttons.btnLocateStand.loading) {
                    HomeButtonState(
                        enabled = true,
                        style = HomeButtonStyle.Enable,
                        background = HomeButtonBackground.Green,
                        textRes = R.string.btn_locate_stop_exit
                    )
                } else {
                    HomeButtonState(enabled = false, style = HomeButtonStyle.Disable)
                }
            }

            else -> HomeButtonState(enabled = false, style = HomeButtonStyle.Disable)
        }
    }

    private fun reduceRoofLightButton(value: Pair<Boolean?, Boolean?>?): HomeButtonState {
        val lightState = value?.first
        val isDisableLuminous = value?.second

        return if (isDisableLuminous == true) {
            when (lightState) {
                true -> HomeButtonState(
                    enabled = true,
                    style = HomeButtonStyle.Enable,
                    background = HomeButtonBackground.Green
                )
                false -> HomeButtonState(
                    enabled = true,
                    style = HomeButtonStyle.Enable,
                    background = HomeButtonBackground.Red
                )
                else -> HomeButtonState(enabled = false, style = HomeButtonStyle.Disable)
            }
        } else {
            HomeButtonState(enabled = false, style = HomeButtonStyle.Disable)
        }
    }
}
```

# 4) Important correction: the bridge should not re-trigger actions blindly

The code above shows the shape of the bridge, but to be exact like the fragment, the bridge should do this:

- collect shared flows
- update `HomeUiState` via reducer actions
- only trigger side effects when the fragment did

So the better pattern is:

- `HomeMainActivityViewModel` exposes flows
- `HomeComposeViewModel` has a `reduceSharedAction(...)`
- Compose screen collects shared flows and dispatches reducer actions

That is the Compose equivalent of the fragment’s `launch { collect { ... } }` blocks.

```kotlin
                        btnPending = reducePendingButton(updated)
                    )
                )
            }
            is HomeReducerAction.OrangePendingChanged -> current.copy(
                orangeBtnPending = action.value,
                buttons = current.buttons.copy(
                    btnPending = current.buttons.btnPending.copy(
                        background = if (action.value == true) HomeButtonBackground.Orange else HomeButtonBackground.Blue
                    )
                )
            )
            is HomeReducerAction.DashboardButtonChanged -> current.copy(
                hasDashboardButtonOn = action.visible,
                buttons = current.buttons.copy(
                    btnDashboard = current.buttons.btnDashboard.copy(visible = action.visible)
                )
            )
            is HomeReducerAction.FixedPriceButtonChanged -> current.copy(
                showEstimateFixedPriceButton = action.visible,
                buttons = current.buttons.copy(
                    btnFixedPrice = current.buttons.btnFixedPrice.copy(visible = action.visible),
                    btnReceipts = current.buttons.btnReceipts.copy(visible = !action.visible)
                )
            )
            is HomeReducerAction.ZoningButtonChanged -> current.copy(
                buttons = current.buttons.copy(
                    btnZoning = current.buttons.btnZoning.copy(
                        enabled = action.enabled,
                        style = if (action.enabled) HomeButtonStyle.Enable else HomeButtonStyle.Disable
                    )
                )
            )
            is HomeReducerAction.LocationEnabledChanged -> current.copy(
                locationEnabled = action.value,
                buttons = current.buttons.copy(
                    btnLocation = current.buttons.btnLocation.copy(
                        enabled = action.value.first,
                        background = if (action.value.first) HomeButtonBackground.Green else HomeButtonBackground.Red,
                        textRes = if (action.value.first) R.string.btn_location_off else R.string.btn_location_on
                    )
                )
            )
            is HomeReducerAction.LocationTypeChanged -> {
                val updated = current.copy(locationType = action.value)
                updated.copy(
                    buttons = updated.buttons.copy(
                        btnLocateStand = reduceLocateStandButton(updated)
                    )
                )
            }
            is HomeReducerAction.LocatedOnStopChanged -> {
                val updated = current.copy(locatedOnStop = action.value)
                updated.copy(
                    buttons = updated.buttons.copy(
                        btnLocateStand = reduceLocateStandButton(updated)
                    )
                )
            }
            is HomeReducerAction.RoofLightChanged -> current.copy(
                roofLight = action.value,
                buttons = current.buttons.copy(
                    btnRoofLight = reduceRoofLightButton(action.value)
                )
            )
            is HomeReducerAction.MessagesChanged -> current.copy(
                messageState = action.value,
                buttons = current.buttons.copy(
                    btnMessages = when (action.value) {
                        MessageUIEnum.NO_MESSAGES -> HomeButtonState(enabled = false, style = HomeButtonStyle.Disable, background = HomeButtonBackground.Blue)
                        MessageUIEnum.HAS_MESSAGES_BUT_NOT_NEW -> HomeButtonState(enabled = true, style = HomeButtonStyle.Enable, background = HomeButtonBackground.Blue)
                        MessageUIEnum.HAS_NEW_MESSAGES -> HomeButtonState(enabled = true, style = HomeButtonStyle.Enable, background = HomeButtonBackground.Orange)
                        null -> HomeButtonState(enabled = false, style = HomeButtonStyle.Disable, background = HomeButtonBackground.Blue)
                    }
                )
            )
            is HomeReducerAction.BravoStateChanged -> current.copy(
                bravoVoice = action.voiceValue,
                buttons = current.buttons.copy(
                    btnNoClient = current.buttons.btnNoClient.copy(
                        background = if (action.clientButton) HomeButtonBackground.Green else HomeButtonBackground.Red,
                        enabled = action.noClientButtonEnabled,
                        style = if (action.noClientButtonEnabled) HomeButtonStyle.Enable else HomeButtonStyle.Disable
                    ),
                    btnReturn = current.buttons.btnReturn.copy(
                        enabled = action.showReturnDispatchButton,
                        style = if (action.showReturnDispatchButton) HomeButtonStyle.Enable else HomeButtonStyle.Disable
                    ),
                    btnVoiceCall = current.buttons.btnVoiceCall.copy(
                        enabled = action.bridgeCallState != 0,
                        style = if (action.bridgeCallState == 0) HomeButtonStyle.Disable else HomeButtonStyle.Enable,
                        background = when (action.bridgeCallState) {
                            2 -> HomeButtonBackground.Red
                            0 -> HomeButtonBackground.Green
                            else -> HomeButtonBackground.Orange
                        }
                    )
                )
            )
            is HomeReducerAction.HasTaximeterConnectionChanged -> current.copy(hasTaximeterConnection = action.value)
            is HomeReducerAction.KeepScreenOnChanged -> current.copy(keepScreenOn = action.value)
            is HomeReducerAction.TimerChanged -> current.copy(timerText = action.value.orEmpty())
        }
    }

    private fun reducePendingButton(current: HomeUiState): HomeButtonState {
        if (current.zone.isNullOrEmpty()) return HomeButtonState(enabled = false, style = HomeButtonStyle.Disable)
        val isInShortBreak =
            current.shortBreakStatus == ShortBreakStatus.IN_SHORT_BREAK ||
            current.shortBreakStatus == ShortBreakStatus.IN_SHORT_BREAK_FORCED
        if (isInShortBreak) return HomeButtonState(enabled = false, style = HomeButtonStyle.Disable)
        if (!W2CLocation.isLocationAllowedByCentral()) return HomeButtonState(enabled = false, style = HomeButtonStyle.Disable)
        if (W2CLocation.getIsInSoonInZone()) return HomeButtonState(enabled = true, style = HomeButtonStyle.Enable)
        if (current.pendingServicesButton == true) return HomeButtonState(enabled = true, style = HomeButtonStyle.Enable)
        return HomeButtonState(enabled = false, style = HomeButtonStyle.Disable)
    }

    private fun reduceLocateStandButton(current: HomeUiState): HomeButtonState {
        return when (current.locationType) {
            "Z" -> {
                if (current.locatedOnStop || W2CLocation.locationStopAllowed()) {
                    HomeButtonState(
                        enabled = true,
                        style = HomeButtonStyle.Enable,
                        background = HomeButtonBackground.Red,
                        textRes = R.string.btn_locate_stop
                    )
                } else {
                    HomeButtonState(enabled = false, style = HomeButtonStyle.Disable)
                }
            }
            "P" -> {
                if (!current.buttons.btnLocateStand.loading) {
                    HomeButtonState(
                        enabled = true,
                        style = HomeButtonStyle.Enable,
                        background = HomeButtonBackground.Green,
                        textRes = R.string.btn_locate_stop_exit
                    )
                } else {
                    HomeButtonState(enabled = false, style = HomeButtonStyle.Disable)
                }
            }
            else -> HomeButtonState(enabled = false, style = HomeButtonStyle.Disable)
        }
    }

    private fun reduceRoofLightButton(value: Pair<Boolean?, Boolean?>?): HomeButtonState {
        val lightState = value?.first
        val isDisableLuminous = value?.second
        return if (isDisableLuminous == true) {
            when (lightState) {
                true -> HomeButtonState(enabled = true, style = HomeButtonStyle.Enable, background = HomeButtonBackground.Green)
                false -> HomeButtonState(enabled = true, style = HomeButtonStyle.Enable, background = HomeButtonBackground.Red)
                else -> HomeButtonState(enabled = false, style = HomeButtonStyle.Disable)
            }
        } else {
            HomeButtonState(enabled = false, style = HomeButtonStyle.Disable)
        }
    }
}

class HomeComposeViewModel(
    private val delegate: HomeViewModel,
    private val sharedVm: HomeMainActivityViewModel,
) : ViewModel() {

    private val reducer = HomeButtonsReducer()

    private val _uiState = MutableStateFlow(HomeUiState())
    val uiState: StateFlow<HomeUiState> = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<HomeUiEffect>(extraBufferCapacity = 64)
    val uiEffect: SharedFlow<HomeUiEffect> = _effects.asSharedFlow()

    fun initialize() {
        delegate.getMessages()
        delegate.checkKeepScreenOn()
        delegate.checkForLocationPermission()
        delegate.checkPendingServiceOnForHirePermission()
    }

    fun reduce(action: HomeReducerAction) {
        _uiState.value = reducer.reduce(_uiState.value, action)
    }

    fun onEvent(event: HomeUiEvent) {
        when (event) {
            HomeUiEvent.OnZoningClick -> {
                sharedVm.emitManualZoningNavigation(true)
                delegate.navigateToZoning()
            }
            HomeUiEvent.OnPendingClick -> delegate.canOpenPendingTripsFragment(sharedVm.pendingTripsListFlow)
            HomeUiEvent.OnLocationClick -> handleLocationClick()
            HomeUiEvent.OnReceiptsClick -> delegate.navigateToReceiptHistory()
            HomeUiEvent.OnMessagesClick -> delegate.navigateToMessage()
            HomeUiEvent.OnCentralClick -> delegate.navigateToContactCentral()
            HomeUiEvent.OnDashboardClick -> emit(HomeUiEffect.Navigate(R.id.action_homeFragment_to_dashboardFragment))
            HomeUiEvent.OnFixedPriceClick -> delegate.navigateToEstimateFixedPriceScreen()
            HomeUiEvent.OnRoofLightClick -> {
                val roof = sharedVm.roofLightFlow.value
                if (roof != null && roof == true) delegate.setRoofLightOff() else delegate.setRoofLightOn()
            }
            HomeUiEvent.OnLocateStandClick -> handleLocateStandClick()
            is HomeUiEvent.OnDialogButtonClicked -> onDialogResult(event.buttonType)
        }
    }

    fun bindSharedState() {
        reduce(HomeReducerAction.LocationEnabledChanged(sharedVm.locationEnabledFlow.value))
        reduce(HomeReducerAction.RoofLightChanged(sharedVm.roofLightFlow.value?.let { it to true }))
        reduce(HomeReducerAction.LocatedOnStopChanged(sharedVm.locatedOnStop.value))
        reduce(HomeReducerAction.LocationTypeChanged(sharedVm.locationType.value))
        reduce(HomeReducerAction.HasTaximeterConnectionChanged(sharedVm.hasTaximeterConnectionFlow.value))
        reduce(HomeReducerAction.OrangePendingChanged(sharedVm.orangeBtnPendingFlow.value))
        reduce(HomeReducerAction.ShortBreakChanged(sharedVm.shortBreakStatus.value))
        reduce(HomeReducerAction.ZoneChanged(sharedVm.zoneFlow.value))
        reduce(HomeReducerAction.BravoStateChanged(
            showReturnDispatchButton = sharedVm.bravoStateFlow.value.showReturnDispatchButton,
            clientButton = sharedVm.bravoStateFlow.value.clientButton,
            noClientButtonEnabled = sharedVm.bravoStateFlow.value.noClientButtonEnabled,
            bridgeCallState = sharedVm.bravoStateFlow.value.bridgeCallState,
            voiceValue = sharedVm.bravoStateFlow.value.voiceValue
        ))
    }

    private fun handleLocationClick() {
        val enabled = sharedVm.locationEnabledFlow.value
        if (enabled.second) {
            _uiState.update {
                it.copy(dialog = HomeDialogState(
                    titleRes = R.string.confirm_deactivate_location,
                    kind = HomeDialogKind.DeactivateLocation
                ))
            }
            emit(HomeUiEffect.OpenDialog(_uiState.value.dialog!!))
        } else {
            if (W2CLocation.isLocationAllowedByCentral()) {
                _uiState.update {
                    it.copy(dialog = HomeDialogState(
                        titleRes = R.string.confirm_activate_location,
                        kind = HomeDialogKind.ActivateLocation
                    ))
                }
                emit(HomeUiEffect.OpenDialog(_uiState.value.dialog!!))
            }
        }
    }

    private fun handleLocateStandClick() {
        if (W2CLocation.getLocationTypeSent() == 'P') {
            _uiState.update {
                it.copy(dialog = HomeDialogState(
                    titleRes = R.string.dialog_exit_onStand_title,
                    kind = HomeDialogKind.ExitStand
                ))
            }
            emit(HomeUiEffect.OpenDialog(_uiState.value.dialog!!))
        } else {
            W2CLocation.sendLocateStop()
            W2CLocation.setTryingToRelocateInStand(false)
        }
    }

    private fun onDialogResult(buttonType: ButtonType) {
        val dialog = _uiState.value.dialog ?: return
        val accepted = buttonType == ButtonType.ACCEPT
        when (dialog.kind) {
            HomeDialogKind.ActivateLocation -> if (accepted) {
                sharedVm.updateLocationEnabledFlow(enabled = true, canEnable = true)
                sharedVm.updateLocationManuallyDeLocatedFlow(false)
                emit(HomeUiEffect.LocationEnabledManually)
            }
            HomeDialogKind.DeactivateLocation -> if (accepted) {
                sharedVm.updateLocationEnabledFlow(enabled = false, canEnable = true)
                sharedVm.updateLocationManuallyDeLocatedFlow(true)
                delegate.checkForHireAfterDelocation()
                emit(HomeUiEffect.Beep)
                emit(HomeUiEffect.LocationEnabledManually)
            }
            HomeDialogKind.ExitShift -> if (accepted) {
                delegate.logoff()
                val deepLink = NavDeepLinkRequest.Builder.fromUri("android-app://ifac.td.taxi/welcome".toUri()).build()
                emit(HomeUiEffect.NavigateDeepLink(deepLink))
            }
            HomeDialogKind.ManualTrip -> if (accepted) {
                delegate.changeStateHiredManual()
            }
            HomeDialogKind.ExitStand -> if (accepted) {
                W2CLocation.sendLocateZone()
            }
        }
        _uiState.update { it.copy(dialog = null) }
        emit(HomeUiEffect.DismissDialog)
    }

    private fun emit(effect: HomeUiEffect) {
        viewModelScope.launch { _effects.emit(effect) }
    }
}

@Composable
fun HomeScreen(
    navController: NavController,
    viewModel: HomeComposeViewModel,
    sharedViewModel: HomeMainActivityViewModel,
    onShowHeader: (Boolean) -> Unit,
    onShowToast: (Int) -> Unit,
    onKeepScreenActive: (Boolean) -> Unit,
) {
    val state by viewModel.uiState.collectAsStateWithLifecycle()
    var dialogState by remember { mutableStateOf<HomeDialogState?>(null) }

    LaunchedEffect(Unit) {
        onShowHeader(true)
        viewModel.initialize()
        viewModel.bindSharedState()
    }

    LaunchedEffect(viewModel) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                is HomeUiEffect.Navigate -> navController.navigate(effect.route)
                is HomeUiEffect.NavigateDeepLink -> navController.navigate(effect.request)
                is HomeUiEffect.ShowToast -> onShowToast(effect.messageRes)
                is HomeUiEffect.KeepScreenActive -> onKeepScreenActive(effect.keepOn)
                is HomeUiEffect.OpenDialog -> dialogState = effect.dialog
                HomeUiEffect.DismissDialog -> dialogState = null
                HomeUiEffect.Beep -> Unit
                HomeUiEffect.LocationEnabledManually -> Unit
            }
        }
    }

    LaunchedEffect(sharedViewModel) {
        sharedViewModel.effects.collectLatest { effect ->
            when (effect) {
                is HomeMainActivityEffect.Navigate -> navController.navigate(effect.route)
                is HomeMainActivityEffect.NavigateDeepLink -> navController.navigate(effect.request)
                is HomeMainActivityEffect.ShowToast -> onShowToast(effect.messageRes)
                is HomeMainActivityEffect.KeepScreenActive -> onKeepScreenActive(effect.value)
                HomeMainActivityEffect.RefreshMessages -> viewModel.initialize()
            }
        }
    }

    Scaffold(
        topBar = { /* preserve header behavior */ },
        bottomBar = { /* preserve bottom bar behavior */ }
    ) { padding ->
        HomeContent(
            modifier = Modifier.padding(padding),
            state = state.buttons,
            onEvent = viewModel::onEvent
        )
    }

    dialogState?.let { dialog ->
        AlertDialog(
            onDismissRequest = { dialogState = null },
            title = { Text(text = stringResource(dialog.titleRes)) },
            text = {
                dialog.descriptionRes?.let { Text(text = stringResource(it)) }
            },
            confirmButton = {
                TextButton(onClick = {
                    viewModel.onEvent(HomeUiEvent.OnDialogButtonClicked(ButtonType.ACCEPT))
                    dialogState = null
                }) { Text("OK") }
            },
            dismissButton = {
                TextButton(onClick = {
                    viewModel.onEvent(HomeUiEvent.OnDialogButtonClicked(ButtonType.CANCEL))
                    dialogState = null
                }) { Text("Cancel") }
            }
        )
    }
}
```

```kotlin
class HomeButtonsReducer {
    fun reduce(current: HomeUiState, action: HomeReducerAction): HomeUiState {
        return when (action) {
            is HomeReducerAction.ZoneChanged -> {
                val updated = current.copy(zone = action.zone)
                updated.copy(buttons = updated.buttons.copy(btnPending = reducePendingButton(updated)))
            }
            is HomeReducerAction.ShortBreakChanged -> {
                val updated = current.copy(shortBreakStatus = action.status)
                updated.copy(buttons = updated.buttons.copy(btnPending = reducePendingButton(updated)))
            }
            is HomeReducerAction.PendingServicesButtonChanged -> {
                val updated = current.copy(pendingServicesButton = action.value)
                updated.copy(buttons = updated.buttons.copy(btnPending = reducePendingButton(updated)))
            }
            is HomeReducerAction.OrangePendingChanged -> current.copy(
                orangeBtnPending = action.value,
                buttons = current.buttons.copy(
                    btnPending = current.buttons.btnPending.copy(
                        background = if (action.value == true) HomeButtonBackground.Orange else HomeButtonBackground.Blue
                    )
                )
            )
            is HomeReducerAction.DashboardButtonChanged -> current.copy(
                hasDashboardButtonOn = action.visible,
                buttons = current.buttons.copy(
                    btnDashboard = current.buttons.btnDashboard.copy(visible = action.visible)
                )
            )
            is HomeReducerAction.FixedPriceButtonChanged -> current.copy(
                showEstimateFixedPriceButton = action.visible,
                buttons = current.buttons.copy(
                    btnFixedPrice = current.buttons.btnFixedPrice.copy(visible = action.visible),
                    btnReceipts = current.buttons.btnReceipts.copy(visible = !action.visible)
                )
            )
            is HomeReducerAction.ZoningButtonChanged -> current.copy(
                buttons = current.buttons.copy(
                    btnZoning = current.buttons.btnZoning.copy(
                        enabled = action.enabled,
                        style = if (action.enabled) HomeButtonStyle.Enable else HomeButtonStyle.Disable
                    )
                )
            )
            is HomeReducerAction.LocationEnabledChanged -> current.copy(
                locationEnabled = action.value,
                buttons = current.buttons.copy(
                    btnLocation = current.buttons.btnLocation.copy(
                        enabled = action.value.first,
                        background = if (action.value.first) HomeButtonBackground.Green else HomeButtonBackground.Red,
                        textRes = if (action.value.first) R.string.btn_location_off else R.string.btn_location_on
                    )
                )
            )
            is HomeReducerAction.LocationTypeChanged -> {
                val updated = current.copy(locationType = action.value)
                updated.copy(buttons = updated.buttons.copy(btnLocateStand = reduceLocateStandButton(updated)))
            }
            is HomeReducerAction.LocatedOnStopChanged -> {
                val updated = current.copy(locatedOnStop = action.value)
                updated.copy(buttons = updated.buttons.copy(btnLocateStand = reduceLocateStandButton(updated)))
            }
            is HomeReducerAction.RoofLightChanged -> current.copy(
                roofLight = action.value,
                buttons = current.buttons.copy(btnRoofLight = reduceRoofLightButton(action.value))
            )
            is HomeReducerAction.MessagesChanged -> current.copy(
                messageState = action.value,
                buttons = current.buttons.copy(
                    btnMessages = when (action.value) {
                        MessageUIEnum.NO_MESSAGES -> HomeButtonState(enabled = false, style = HomeButtonStyle.Disable, background = HomeButtonBackground.Blue)
                        MessageUIEnum.HAS_MESSAGES_BUT_NOT_NEW -> HomeButtonState(enabled = true, style = HomeButtonStyle.Enable, background = HomeButtonBackground.Blue)
                        MessageUIEnum.HAS_NEW_MESSAGES -> HomeButtonState(enabled = true, style = HomeButtonStyle.Enable, background = HomeButtonBackground.Orange)
                        null -> HomeButtonState(enabled = false, style = HomeButtonStyle.Disable, background = HomeButtonBackground.Blue)
                    }
                )
            )
            is HomeReducerAction.BravoStateChanged -> current.copy(
                bravoVoice = action.voiceValue,
                buttons = current.buttons.copy(
                    btnNoClient = current.buttons.btnNoClient.copy(
                        background = if (action.clientButton) HomeButtonBackground.Green else HomeButtonBackground.Red,
                        enabled = action.noClientButtonEnabled,
                        style = if (action.noClientButtonEnabled) HomeButtonStyle.Enable else HomeButtonStyle.Disable
                    ),
                    btnReturn = current.buttons.btnReturn.copy(
                        enabled = action.showReturnDispatchButton,
                        style = if (action.showReturnDispatchButton) HomeButtonStyle.Enable else HomeButtonStyle.Disable
                    ),
                    btnVoiceCall = current.buttons.btnVoiceCall.copy(
                        enabled = action.bridgeCallState != 0,
                        style = if (action.bridgeCallState == 0) HomeButtonStyle.Disable else HomeButtonStyle.Enable,
                        background = when (action.bridgeCallState) {
                            2 -> HomeButtonBackground.Red
                            0 -> HomeButtonBackground.Green
                            else -> HomeButtonBackground.Orange
                        }
                    )
                )
            )
            is HomeReducerAction.HasTaximeterConnectionChanged -> current.copy(hasTaximeterConnection = action.value)
            is HomeReducerAction.KeepScreenOnChanged -> current.copy(keepScreenOn = action.value)
            is HomeReducerAction.TimerChanged -> current.copy(timerText = action.value.orEmpty())
        }
    }

    private fun reducePendingButton(current: HomeUiState): HomeButtonState {
        if (current.zone.isNullOrEmpty()) return HomeButtonState(enabled = false, style = HomeButtonStyle.Disable)
        val isInShortBreak =
            current.shortBreakStatus == ShortBreakStatus.IN_SHORT_BREAK ||
            current.shortBreakStatus == ShortBreakStatus.IN_SHORT_BREAK_FORCED
        if (isInShortBreak) return HomeButtonState(enabled = false, style = HomeButtonStyle.Disable)
        if (!W2CLocation.isLocationAllowedByCentral()) return HomeButtonState(enabled = false, style = HomeButtonStyle.Disable)
        if (W2CLocation.getIsInSoonInZone()) return HomeButtonState(enabled = true, style = HomeButtonStyle.Enable)
        if (current.pendingServicesButton == true) return HomeButtonState(enabled = true, style = HomeButtonStyle.Enable)
        return HomeButtonState(enabled = false, style = HomeButtonStyle.Disable)
    }

    private fun reduceLocateStandButton(current: HomeUiState): HomeButtonState {
        return when (current.locationType) {
            "Z" -> if (current.locatedOnStop || W2CLocation.locationStopAllowed()) {
                HomeButtonState(
                    enabled = true,
                    style = HomeButtonStyle.Enable,
                    background = HomeButtonBackground.Red,
                    textRes = R.string.btn_locate_stop
                )
            } else {
                HomeButtonState(enabled = false, style = HomeButtonStyle.Disable)
            }
            "P" -> if (!current.buttons.btnLocateStand.loading) {
                HomeButtonState(
                    enabled = true,
                    style = HomeButtonStyle.Enable,
                    background = HomeButtonBackground.Green,
                    textRes = R.string.btn_locate_stop_exit
                )
            } else {
                HomeButtonState(enabled = false, style = HomeButtonStyle.Disable)
            }
            else -> HomeButtonState(enabled = false, style = HomeButtonStyle.Disable)
        }
    }

    private fun reduceRoofLightButton(value: Pair<Boolean?, Boolean?>?): HomeButtonState {
        val lightState = value?.first
        val isDisableLuminous = value?.second
        return if (isDisableLuminous == true) {
            when (lightState) {
                true -> HomeButtonState(enabled = true, style = HomeButtonStyle.Enable, background = HomeButtonBackground.Green)
                false -> HomeButtonState(enabled = true, style = HomeButtonStyle.Enable, background = HomeButtonBackground.Red)
                else -> HomeButtonState(enabled = false, style = HomeButtonStyle.Disable)
            }
        } else {
            HomeButtonState(enabled = false, style = HomeButtonStyle.Disable)
        }
    }
}

class HomeComposeViewModel(
    private val delegate: HomeViewModel,
    private val sharedVm: HomeMainActivityViewModel,
) : ViewModel() {

    private val reducer = HomeButtonsReducer()

    private val _uiState = MutableStateFlow(HomeUiState())
    val uiState: StateFlow<HomeUiState> = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<HomeUiEffect>(extraBufferCapacity = 64)
    val uiEffect: SharedFlow<HomeUiEffect> = _effects.asSharedFlow()

    fun initialize() {
        delegate.getMessages()
        delegate.checkKeepScreenOn()
        delegate.checkForLocationPermission()
        delegate.checkPendingServiceOnForHirePermission()
    }

    fun reduce(action: HomeReducerAction) {
        _uiState.value = reducer.reduce(_uiState.value, action)
    }

    fun bindSharedState() {
        reduce(HomeReducerAction.LocationEnabledChanged(sharedVm.locationEnabledFlow.value))
        reduce(HomeReducerAction.RoofLightChanged(sharedVm.roofLightFlow.value?.let { it to true }))
        reduce(HomeReducerAction.LocatedOnStopChanged(sharedVm.locatedOnStop.value))
        reduce(HomeReducerAction.LocationTypeChanged(sharedVm.locationType.value))
        reduce(HomeReducerAction.HasTaximeterConnectionChanged(sharedVm.hasTaximeterConnectionFlow.value))
        reduce(HomeReducerAction.OrangePendingChanged(sharedVm.orangeBtnPendingFlow.value))
        reduce(HomeReducerAction.ShortBreakChanged(sharedVm.shortBreakStatus.value))
        reduce(HomeReducerAction.ZoneChanged(sharedVm.zoneFlow.value))
        reduce(
            HomeReducerAction.BravoStateChanged(
                showReturnDispatchButton = sharedVm.bravoStateFlow.value.showReturnDispatchButton,
                clientButton = sharedVm.bravoStateFlow.value.clientButton,
                noClientButtonEnabled = sharedVm.bravoStateFlow.value.noClientButtonEnabled,
                bridgeCallState = sharedVm.bravoStateFlow.value.bridgeCallState,
                voiceValue = sharedVm.bravoStateFlow.value.voiceValue
            )
        )
    }

    fun onEvent(event: HomeUiEvent) {
        when (event) {
            HomeUiEvent.OnZoningClick -> {
                sharedVm.emitManualZoningNavigation(true)
                delegate.navigateToZoning()
            }
            HomeUiEvent.OnPendingClick -> delegate.canOpenPendingTripsFragment(sharedVm.pendingTripsListFlow as MutableStateFlow<ArrayList<com.interfacom.sdk.taximeter.bravocomm.rest.pending_trips.response.PendingTrip>?>)
            HomeUiEvent.OnLocationClick -> handleLocationClick()
            HomeUiEvent.OnReceiptsClick -> delegate.navigateToReceiptHistory()
            HomeUiEvent.OnMessagesClick -> delegate.navigateToMessage()
            HomeUiEvent.OnCentralClick -> delegate.navigateToContactCentral()
            HomeUiEvent.OnDashboardClick -> emit(HomeUiEffect.Navigate(HomeDirections.actionHomeFragmentToDashboardFragment()))
            HomeUiEvent.OnFixedPriceClick -> delegate.navigateToEstimateFixedPriceScreen()
            HomeUiEvent.OnRoofLightClick -> {
                val roof = sharedVm.roofLightFlow.value
                if (roof == true) delegate.setRoofLightOff() else delegate.setRoofLightOn()
            }
            HomeUiEvent.OnLocateStandClick -> handleLocateStandClick()
            is HomeUiEvent.OnDialogButtonClicked -> onDialogResult(event.buttonType)
        }
    }

    private fun handleLocationClick() {
        val enabled = sharedVm.locationEnabledFlow.value
        if (enabled.second) {
            _uiState.update {
                it.copy(dialog = HomeDialogState(
                    titleRes = R.string.confirm_deactivate_location,
                    kind = HomeDialogKind.DeactivateLocation
                ))
            }
            emit(HomeUiEffect.OpenDialog(_uiState.value.dialog!!))
        } else {
            if (W2CLocation.isLocationAllowedByCentral()) {
                _uiState.update {
                    it.copy(dialog = HomeDialogState(
                        titleRes = R.string.confirm_activate_location,
                        kind = HomeDialogKind.ActivateLocation
                    ))
                }
                emit(HomeUiEffect.OpenDialog(_uiState.value.dialog!!))
            }
        }
    }

    private fun handleLocateStandClick() {
        if (W2CLocation.getLocationTypeSent() == 'P') {
            _uiState.update {
                it.copy(dialog = HomeDialogState(
                    titleRes = R.string.dialog_exit_onStand_title,
                    kind = HomeDialogKind.ExitStand
                ))
            }
            emit(HomeUiEffect.OpenDialog(_uiState.value.dialog!!))
        } else {
            W2CLocation.sendLocateStop()
            W2CLocation.setTryingToRelocateInStand(false)
        }
    }

    private fun onDialogResult(buttonType: ButtonType) {
        val dialog = _uiState.value.dialog ?: return
        val accepted = buttonType == ButtonType.ACCEPT
        when (dialog.kind) {
            HomeDialogKind.ActivateLocation -> if (accepted) {
                sharedVm.updateLocationEnabledFlow(enabled = true, canEnable = true)
                sharedVm.updateLocationManuallyDeLocatedFlow(false)
                emit(HomeUiEffect.LocationEnabledManually)
            }
            HomeDialogKind.DeactivateLocation -> if (accepted) {
                sharedVm.updateLocationEnabledFlow(enabled = false, canEnable = true)
                sharedVm.updateLocationManuallyDeLocatedFlow(true)
                delegate.checkForHireAfterDelocation()
                emit(HomeUiEffect.Beep)
                emit(HomeUiEffect.LocationEnabledManually)
            }
            HomeDialogKind.ExitShift -> if (accepted) {
                delegate.logoff()
                val deepLink = NavDeepLinkRequest.Builder.fromUri("android-app://ifac.td.taxi/welcome".toUri()).build()
                emit(HomeUiEffect.NavigateDeepLink(deepLink))
            }
            HomeDialogKind.ManualTrip -> if (accepted) {
                delegate.changeStateHiredManual()
            }
            HomeDialogKind.ExitStand -> if (accepted) {
                W2CLocation.sendLocateZone()
            }
        }
        _uiState.update { it.copy(dialog = null) }
        emit(HomeUiEffect.DismissDialog)
    }

    private fun emit(effect: HomeUiEffect) {
        viewModelScope.launch { _effects.emit(effect) }
    }
}

@Composable
fun HomeScreen(
    navController: NavController,
    viewModel: HomeComposeViewModel,
    sharedViewModel: HomeMainActivityViewModel,
    onShowHeader: (Boolean) -> Unit,
    onShowToast: (Int) -> Unit,
    onKeepScreenActive: (Boolean) -> Unit,
) {
    val state by viewModel.uiState.collectAsStateWithLifecycle()
    var dialogState by remember { mutableStateOf<HomeDialogState?>(null) }

    LaunchedEffect(Unit) {
        onShowHeader(true)
        viewModel.initialize()
        viewModel.bindSharedState()
    }

    LaunchedEffect(viewModel) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                is HomeUiEffect.Navigate -> navController.navigate(effect.directions)
                is HomeUiEffect.NavigateDeepLink -> navController.navigate(effect.request)
                is HomeUiEffect.ShowToast -> onShowToast(effect.messageRes)
                is HomeUiEffect.KeepScreenActive -> onKeepScreenActive(effect.keepOn)
                is HomeUiEffect.OpenDialog -> dialogState = effect.dialog
                HomeUiEffect.DismissDialog -> dialogState = null
                HomeUiEffect.Beep -> Unit
                HomeUiEffect.LocationEnabledManually -> Unit
            }
        }
    }

    LaunchedEffect(sharedViewModel) {
        sharedViewModel.updateMessageUI.collectLatest {
            viewModel.initialize()
        }
    }

    Scaffold(
        topBar = { },
        bottomBar = { }
    ) { padding ->
        HomeContent(
            modifier = Modifier.padding(padding),
            state = state.buttons,
            onEvent = viewModel::onEvent
        )
    }

    dialogState?.let { dialog ->
        AlertDialog(
            onDismissRequest = { dialogState = null },
            title = { Text(text = stringResource(dialog.titleRes)) },
            text = { dialog.descriptionRes?.let { Text(text = stringResource(it)) } },
            confirmButton = {
                TextButton(onClick = {
                    viewModel.onEvent(HomeUiEvent.OnDialogButtonClicked(ButtonType.ACCEPT))
                    dialogState = null
                }) { Text("OK") }
            },
            dismissButton = {
                TextButton(onClick = {
                    viewModel.onEvent(HomeUiEvent.OnDialogButtonClicked(ButtonType.CANCEL))
                    dialogState = null
                }) { Text("Cancel") }
            }
        )
    }
}

@Composable
fun HomeContent(
    modifier: Modifier = Modifier,
    state: HomeButtonsState,
    onEvent: (HomeUiEvent) -> Unit
) {
    Column(modifier = modifier.fillMaxSize().padding(16.dp)) {
        HomeButtonRow(state.btnZoning, { onEvent(HomeUiEvent.OnZoningClick) }, state.btnPending, { onEvent(HomeUiEvent.OnPendingClick) })
        Spacer(Modifier.height(12.dp))
        HomeButtonRow(state.btnLocation, { onEvent(HomeUiEvent.OnLocationClick) }, state.btnLocateStand, { onEvent(HomeUiEvent.OnLocateStandClick) })
        Spacer(Modifier.height(12.dp))
        HomeButtonRow(state.btnReceipts, { onEvent(HomeUiEvent.OnReceiptsClick) }, state.btnMessages, { onEvent(HomeUiEvent.OnMessagesClick) })
        Spacer(Modifier.height(12.dp))
        HomeButtonRow(state.btnCentral, { onEvent(HomeUiEvent.OnCentralClick) }, state.btnRoofLight, { onEvent(HomeUiEvent.OnRoofLightClick) })
        Spacer(Modifier.height(12.dp))
        if (state.btnDashboard.visible || state.btnFixedPrice.visible) {
            HomeButtonRow(state.btnDashboard, { onEvent(HomeUiEvent.OnDashboardClick) }, state.btnFixedPrice, { onEvent(HomeUiEvent.OnFixedPriceClick) })
        }
        Spacer(Modifier.height(12.dp))
        HomeButtonRow(state.btnNoClient, {}, state.btnReturn, {})
        Spacer(Modifier.height(12.dp))
        HomeButtonRow(state.btnVoiceCall, {}, HomeButtonState(visible = false), {})
    }
}

@Composable
fun HomeButtonRow(
    left: HomeButtonState,
    leftClick: () -> Unit,
    right: HomeButtonState,
    rightClick: () -> Unit
) {
    Row(Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.spacedBy(12.dp)) {
        if (left.visible) {
            Button(
                onClick = leftClick,
                enabled = left.enabled && !left.loading,
                modifier = Modifier.weight(1f)
            ) {
                Text(text = left.textRes?.let { stringResource(it) } ?: "Button")
            }
        } else {
            Spacer(Modifier.weight(1f))
        }

        if (right.visible) {
            Button(
                onClick = rightClick,
                enabled = right.enabled && !right.loading,
                modifier = Modifier.weight(1f)
            ) {
                Text(text = right.textRes?.let { stringResource(it) } ?: "Button")
            }
        } else {
            Spacer(Modifier.weight(1f))
        }
    }
}
```

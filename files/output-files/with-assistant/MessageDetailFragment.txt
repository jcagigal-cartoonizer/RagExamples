## 2) Exact `MessageDetailButtonsState` reducer mirroring the fragment branches

This reducer mirrors the fragment logic branch-by-branch:

- `btnAnswer` click -> stop timer + reveal predefined/new + hide answer/print
- `btnPrint` click -> stop timer + print
- `btnDelete` click -> open delete dialog
- `btnAccept` click -> stop timer + navigate back
- `btnPredefinedMessage` click -> stop timer + navigate to predefined fragment
- `btnNewMessage` click -> stop timer + open write dialog
- message collector:
  - set ticket text
  - if answers != null, answer click opens custom answer dialog
  - else answer click expands buttons
  - if not urgent, disable answer
- predefined messages collector:
  - if true, enable predefined button
- close messages collector:
  - navigate back
- timer collector:
  - show timer on accept or hide/stop timer

```kotlin
package ifac.td.taxi.ui.screen.message_detail

import ifac.td.taxi.repository.room.entities.message.MessageEntity
import ifac.td.taxi.ui.custom.button.ButtonType
import ifac.td.taxi.ui.custom.button.CustomButton

object MessageDetailButtonsReducer {

    fun reduce(
        current: MessageDetailButtonsState,
        message: MessageEntity?,
        hasPredefinedMessages: Boolean,
        isAnswerExpanded: Boolean,
        timerState: ButtonTimerStateUi?,
    ): MessageDetailButtonsState {
        val answerState = when {
            message == null -> current.answer
            !message.isUrgent -> current.answer.copy(
                visible = true,
                enabled = false,
                style = CustomButton.StyleButton.DISABLE,
                background = CustomButton.BackgroundButtonColor.BLUE,
                buttonType = ButtonType.ANSWER
            )
            else -> current.answer.copy(
                visible = true,
                enabled = true,
                style = CustomButton.StyleButton.ENABLE,
                background = CustomButton.BackgroundButtonColor.BLUE,
                buttonType = ButtonType.ANSWER
            )
        }

        val predefinedState = current.predefinedMessage.copy(
            visible = isAnswerExpanded,
            enabled = hasPredefinedMessages,
            style = if (hasPredefinedMessages) CustomButton.StyleButton.ENABLE else current.predefinedMessage.style,
            background = CustomButton.BackgroundButtonColor.BLUE,
            buttonType = ButtonType.PREDEFINED
        )

        val newMessageState = current.newMessage.copy(
            visible = isAnswerExpanded,
            enabled = true,
            style = CustomButton.StyleButton.ENABLE,
            background = CustomButton.BackgroundButtonColor.BLUE,
            buttonType = ButtonType.NEW_MESSAGE
        )

        val printState = current.print.copy(
            visible = !isAnswerExpanded,
            enabled = true,
            style = CustomButton.StyleButton.ENABLE,
            background = CustomButton.BackgroundButtonColor.BLUE,
            buttonType = ButtonType.IMPRIMIR
        )

        val answerVisible = !isAnswerExpanded
        val acceptState = current.accept.copy(
            visible = true,
            enabled = true,
            style = CustomButton.StyleButton.ENABLE,
            background = CustomButton.BackgroundButtonColor.GREEN,
            buttonType = ButtonType.ACCEPT
        )

        val deleteState = current.delete.copy(
            visible = true,
            enabled = true,
            style = CustomButton.StyleButton.ENABLE,
            background = CustomButton.BackgroundButtonColor.RED,
            buttonType = ButtonType.DELETE
        )

        return current.copy(
            answer = answerState.copy(visible = answerVisible),
            predefinedMessage = predefinedState,
            print = printState,
            delete = deleteState,
            accept = acceptState,
            newMessage = newMessageState
        )
    }
}
```

## 3) Compose-friendly `MessageDetailViewModel`

This replaces the fragment’s multiple `StateFlow`s with a single `UiState` plus one `SharedFlow<MessageDetailUiEffect>`.

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import androidx.lifecycle.viewModelScope
import com.interfacom.sdk.taximeter.bravocomm.models.freetextmessage.FreeTextMessageRequest
import ifac.td.taxi.R
import ifac.td.taxi.domain.usecase.CountdownManagerUseCase
import ifac.td.taxi.domain.usecase.MessageUseCase
import ifac.td.taxi.domain.usecase.PredefinedMessagesUseCase
import ifac.td.taxi.domain.usecase.PrinterUseCase
import ifac.td.taxi.domain.usecase.UserPreferencesUseCase
import ifac.td.taxi.framework.sdk.bravocentral.AlfaMessageHandler
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.repository.room.entities.countdown.CountdownId
import ifac.td.taxi.repository.room.entities.countdown.CountdownIdCallback
import ifac.td.taxi.repository.room.entities.countdown.CountdownState
import ifac.td.taxi.repository.room.entities.message.MessageEntity
import ifac.td.taxi.repository.room.entities.message.MessageType
import ifac.td.taxi.ui.custom.button.ButtonType
import ifac.td.taxi.ui.custom.button.CustomButton
import ifac.td.taxi.ui.screen.message_detail.*
import ifac.td.taxi.viewmodel.model.CountdownModel
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch

class MessageDetailViewModel(
    private val alfaMessageHandler: AlfaMessageHandler,
    private val messageUseCase: MessageUseCase,
    private val predefinedMessagesUseCase: PredefinedMessagesUseCase,
    private val printerUseCase: PrinterUseCase,
    private val countdownManagerUseCase: CountdownManagerUseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    context: Application,
) : BaseViewModel(context) {

    private val TAG = "MessageDetailViewModel"

    private val _uiState = MutableStateFlow(MessageDetailUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<MessageDetailUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()

    private var currentMessageId: Int? = null
    private var currentSkipAutoClose: Boolean = false
    private var countdownStartTime: Long = 0L

    init {
        viewModelScope.launch {
            val hasPredefined = predefinedMessagesUseCase.getPredefinedMessages().isNotEmpty()
            _uiState.value = _uiState.value.copy(hasPredefinedMessages = hasPredefined)
        }
    }

    fun initVM(msgId: Int, skipAutoClose: Boolean) {
        currentMessageId = msgId
        currentSkipAutoClose = skipAutoClose

        viewModelScope.launch(Dispatchers.IO) {
            val message = messageUseCase.getMessage(msgId)
            messageUseCase.setMessageRead(msgId)
            Logs.d(TAG, "initVM message: $message")

            _uiState.emit(
                _uiState.value.copy(
                    message = message,
                    buttonsState = MessageDetailButtonsReducer.reduce(
                        current = _uiState.value.buttonsState,
                        message = message,
                        hasPredefinedMessages = _uiState.value.hasPredefinedMessages,
                        isAnswerExpanded = _uiState.value.isAnswerExpanded,
                        timerState = _uiState.value.buttonTimerState
                    )
                )
            )

            if (!skipAutoClose) {
                val messageClosureTimes = mapOf(0 to null, 1 to 5, 2 to 10, 3 to 15, 4 to 20, 5 to 25, 6 to 30, 7 to 60)
                val selectedSeconds =
                    messageClosureTimes[userPreferencesUseCase.getUserPreferences()?.secondsToCloseMsgPosition]

                selectedSeconds?.let {
                    val existingCountdown = countdownManagerUseCase.getCountdownById(CountdownId.CLOSE_MESSAGE)
                    if (existingCountdown == null) {
                        createAndStartCountdown(selectedSeconds)
                    } else if (existingCountdown.state in listOf(CountdownState.FINISHED, CountdownState.STOPPED)) {
                        countdownManagerUseCase.deleteCountdownById(CountdownId.CLOSE_MESSAGE)
                        createAndStartCountdown(selectedSeconds)
                    }
                }
            }
        }
    }

    fun onEvent(event: MessageDetailUiEvent) {
        when (event) {
            MessageDetailUiEvent.ScreenStarted -> Unit

            MessageDetailUiEvent.AnswerClicked -> {
                stopAutoCloseMessage()
                val message = _uiState.value.message
                if (message?.answers != null) {
                    viewModelScope.launch {
                        _uiEffect.emit(
                            MessageDetailUiEffect.ShowDialog(
                                MessageDetailDialogState(
                                    type = DialogType.WRITE_CUSTOM_MESSAGE,
                                    title = context.getString(R.string.write_message),
                                    hint = context.getString(R.string.dialog_write_custom_message_hint),
                                    messageOptions = message.answers
                                )
                            )
                        )
                    }
                } else {
                    _uiState.value = _uiState.value.copy(isAnswerExpanded = true)
                    reduceButtons()
                }
            }

            MessageDetailUiEvent.PrintClicked -> {
                stopAutoCloseMessage()
                currentMessageId?.let { printMessage(it) }
            }

            MessageDetailUiEvent.DeleteClicked -> {
                stopAutoCloseMessage()
                viewModelScope.launch {
                    _uiEffect.emit(
                        MessageDetailUiEffect.ShowDialog(
                            MessageDetailDialogState(
                                type = DialogType.DELETE_MESSAGE,
                                title = context.getString(R.string.dialog_delete_message_title),
                                description = context.getString(R.string.dialog_delete_message_desc)
                            )
                        )
                    )
                }
            }

            MessageDetailUiEvent.AcceptClicked -> {
                stopAutoCloseMessage()
                viewModelScope.launch { _uiEffect.emit(MessageDetailUiEffect.NavigateBack) }
            }

            MessageDetailUiEvent.PredefinedMessageClicked -> {
                stopAutoCloseMessage()
                _uiState.value.message?.let {
                    viewModelScope.launch {
                        _uiEffect.emit(
                            MessageDetailUiEffect.NavigateToPredefinedMessage(
                                messageId = it.id,
                                fromDetail = true
                            )
                        )
                    }
                }
            }

            MessageDetailUiEvent.NewMessageClicked -> {
                stopAutoCloseMessage()
                viewModelScope.launch {
                    _uiEffect.emit(
                        MessageDetailUiEffect.ShowDialog(
                            MessageDetailDialogState(
                                type = DialogType.WRITE_FREE_MESSAGE,
                                title = context.getString(R.string.write_message),
                                hint = context.getString(R.string.message)
                            )
                        )
                    )
                }
            }

            MessageDetailUiEvent.DialogDismissed -> {
                _uiState.value = _uiState.value.copy(dialogState = null)
            }

            is MessageDetailUiEvent.DialogConfirmed -> {
                when (event.button) {
                    ButtonType.ACCEPT -> {
                        event.text?.let { response ->
                            val dispatchNumber = (context as? MainActivityViewModel.DispatchNumberProvider)?.dispatchNumber
                            sendMessage(response, dispatchNumber)
                        }
                    }
                    ButtonType.SEND -> {
                        event.text?.let { response ->
                            val dispatchNumber = (context as? MainActivityViewModel.DispatchNumberProvider)?.dispatchNumber
                            sendMessage(response, dispatchNumber)
                        }
                    }
                    else -> Unit
                }
                _uiState.value = _uiState.value.copy(dialogState = null)
            }

            MessageDetailUiEvent.BackClicked -> {
                viewModelScope.launch { _uiEffect.emit(MessageDetailUiEffect.NavigateBack) }
            }
        }
    }

    private fun reduceButtons() {
        _uiState.value = _uiState.value.copy(
            buttonsState = MessageDetailButtonsReducer.reduce(
                current = _uiState.value.buttonsState,
                message = _uiState.value.message,
                hasPredefinedMessages = _uiState.value.hasPredefinedMessages,
                isAnswerExpanded = _uiState.value.isAnswerExpanded,
                timerState = _uiState.value.buttonTimerState
            )
        )
    }

    private var countdownStartTimeInternal: Long = 0L

    private suspend fun createAndStartCountdown(seconds: Int) {
        countdownStartTimeInternal = System.currentTimeMillis()
        val countdown = CountdownModel(
            idCountdown = CountdownId.CLOSE_MESSAGE,
            timeInSeconds = seconds.toLong(),
            idCallback = CountdownIdCallback.CLOSE_MESSAGE
        )

        _uiState.emit(
            _uiState.value.copy(
                buttonTimerState = ButtonTimerStateUi(
                    maxSeconds = seconds.toLong(),
                    remainingSeconds = seconds.toLong(),
                    startTime = countdownStartTimeInternal
                )
            )
        )
        reduceButtons()

        countdownManagerUseCase.createOrUpdateCountdown(countdown)
        countdownManagerUseCase.startCountdown(countdown)

        countdownManagerUseCase.setCountdownFinishCallback(idCountdown = countdown.idCountdown) {
            CoroutineScope(Dispatchers.IO).launch {
                countdownManagerUseCase.deleteCountdownById(CountdownId.CLOSE_MESSAGE)
                _uiEffect.emit(MessageDetailUiEffect.NavigateBack)
            }
        }

        countdownManagerUseCase.setCountdownOnTickCallback(idCountdown = countdown.idCountdown) { tick ->
            viewModelScope.launch(Dispatchers.IO) {
                _uiState.emit(
                    _uiState.value.copy(
                        buttonTimerState = ButtonTimerStateUi(
                            maxSeconds = seconds.toLong(),
                            remainingSeconds = tick.toLong(),
                            startTime = countdownStartTimeInternal
                        )
                    )
                )
            }
        }
    }

    fun stopAutoCloseMessage() {
        viewModelScope.launch(Dispatchers.IO) {
            if (countdownManagerUseCase.getCountdownById(CountdownId.CLOSE_MESSAGE) != null) {
                _uiState.emit(_uiState.value.copy(buttonTimerState = null))
                countdownManagerUseCase.stopCountdownById(CountdownId.CLOSE_MESSAGE)
                countdownManagerUseCase.deleteCountdownById(CountdownId.CLOSE_MESSAGE)
            }
        }
    }

    fun onMessageLoaded(message: MessageEntity?) {
        _uiState.value = _uiState.value.copy(message = message)
        reduceButtons()
    }

    fun onHasPredefinedMessagesChanged(has: Boolean) {
        _uiState.value = _uiState.value.copy(hasPredefinedMessages = has)
        reduceButtons()
    }

    fun onTimerStateChanged(timer: ButtonTimerStateUi?) {
        _uiState.value = _uiState.value.copy(buttonTimerState = timer)
        reduceButtons()
    }

    fun sendMessage(response: String, dispatchNumber: String?) {
        viewModelScope.launch(Dispatchers.IO) {
            _uiState.value.message?.let { message ->
                val freeMessageRequest = FreeTextMessageRequest().apply {
                    freeTextMessage = response
                    this.dispatchNumber = dispatchNumber ?: ""
                    operadora = message.provider
                }
                alfaMessageHandler.sendMessage(freeMessageRequest)
            }
        }
    }

    fun deleteMessage(messageId: Int) {
        viewModelScope.launch(Dispatchers.IO) {
            messageUseCase.getMessage(messageId)?.let { messageUseCase.deleteMessage(message) }
        }
    }

    fun printMessage(messageId: Int) {
        viewModelScope.launch(Dispatchers.IO) {
            messageUseCase.getMessage(messageId)?.let {
                printerUseCase.printMessageEntity(
                    header = context.getString(R.string.message_receive).uppercase(),
                    message = it,
                    paperFeed = true
                )
            }
        }
    }
}
```

```kotlin
@Composable
fun MessageDetailScreen(
    navController: NavController,
    messageId: Int,
    skipAutoClose: Boolean,
    viewModel: MessageDetailViewModel,
    sharedViewModel: MessageDetailMainActivityViewModel,
    onShowHeader: (Boolean) -> Unit,
) {
    val context = LocalContext.current
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()

    LaunchedEffect(Unit) {
        onShowHeader(false)
        NotificationManager(context).cancelMessageNotification()
        viewModel.initVM(messageId, skipAutoClose)
    }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                MessageDetailUiEffect.NavigateBack -> navController.popBackStack()
                is MessageDetailUiEffect.NavigateToPredefinedMessage -> {
                    navController.navigate(
                        MessageDetailFragmentDirections.actionMessageDetailFragmentToPredefinedMessageFragment(
                            effect.messageId,
                            effect.fromDetail
                        )
                    )
                }
                is MessageDetailUiEffect.ShowDialog -> {
                    // handled by dialog state below
                }
                MessageDetailUiEffect.CancelNotification -> {
                    NotificationManager(context).cancelMessageNotification()
                }
            }
        }
    }

    val message = uiState.message
    val messageText = remember(message) {
        message?.let {
            val messageTypeText = when (it.messageType) {
                MessageType.MESSAGE -> context.getString(R.string.message_receive).uppercase()
                MessageType.DISPATCH -> context.getString(R.string.dispatch_receive)
                MessageType.PREDISPATCH -> context.getString(R.string.predispatch_receive)
                else -> null
            }

            buildString {
                messageTypeText?.let { appendLine(it) }
                appendLine(it.defaultFormatted)
                appendLine(it.text)
            }
        } ?: ""
    }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(Color.Black)
    ) {
        Column(modifier = Modifier.fillMaxSize()) {
            Column(
                modifier = Modifier
                    .weight(1f)
                    .verticalScroll(rememberScrollState())
                    .padding(horizontal = 16.dp)
                    .padding(top = 15.dp, bottom = 15.dp)
            ) {
                Text(
                    text = messageText,
                    color = Color.White,
                    modifier = Modifier.fillMaxWidth()
                )
            }

            MessageDetailButtons(
                state = uiState.buttonsState,
                timerState = uiState.buttonTimerState,
                onAnswer = { viewModel.onEvent(MessageDetailUiEvent.AnswerClicked) },
                onPrint = { viewModel.onEvent(MessageDetailUiEvent.PrintClicked) },
                onDelete = { viewModel.onEvent(MessageDetailUiEvent.DeleteClicked) },
                onAccept = { viewModel.onEvent(MessageDetailUiEvent.AcceptClicked) },
                onPredefined = { viewModel.onEvent(MessageDetailUiEvent.PredefinedMessageClicked) },
                onNewMessage = { viewModel.onEvent(MessageDetailUiEvent.NewMessageClicked) },
            )
        }

        if (uiState.dialogState != null) {
            MessageDetailDialogHost(
                dialogState = uiState.dialogState!!,
                onDismiss = { viewModel.onEvent(MessageDetailUiEvent.DialogDismissed) },
                onConfirm = { text, button ->
                    viewModel.onEvent(MessageDetailUiEvent.DialogConfirmed(text, button))
                }
            )
        }
    }
}
```

## 6) Buttons composable matching XML visibility and behavior

This mirrors the `GridLayout` with 2 columns and 2 rows.

```kotlin
package ifac.td.taxi.ui.screen.message_detail

import androidx.compose.foundation.background
import androidx.compose.foundation.layout.*
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.unit.dp
import ifac.td.taxi.ui.custom.button.CustomButton

@Composable
fun MessageDetailButtons(
    state: MessageDetailButtonsState,
    timerState: ButtonTimerStateUi?,
    onAnswer: () -> Unit,
    onPrint: () -> Unit,
    onDelete: () -> Unit,
    onAccept: () -> Unit,
    onPredefined: () -> Unit,
    onNewMessage: () -> Unit,
) {
    Column(
        modifier = Modifier
            .fillMaxWidth()
            .background(Color.Black)
            .padding(1.dp)
    ) {
        Row(modifier = Modifier.fillMaxWidth()) {
            if (state.answer.visible) {
                ComposeCustomButton(
                    modifier = Modifier.weight(1f).padding(1.dp),
                    buttonType = state.answer.buttonType,
                    style = state.answer.style,
                    background = state.answer.background,
                    onClick = onAnswer
                )
            } else {
                Spacer(modifier = Modifier.weight(1f))
            }

            if (state.print.visible) {
                ComposeCustomButton(
                    modifier = Modifier.weight(1f).padding(1.dp),
                    buttonType = state.print.buttonType,
                    style = state.print.style,
                    background = state.print.background,
                    onClick = onPrint
                )
            } else {
                Spacer(modifier = Modifier.weight(1f))
            }
        }

        Row(modifier = Modifier.fillMaxWidth()) {
            if (state.predefinedMessage.visible) {
                ComposeCustomButton(
```

```kotlin
fun ComposeCustomButton(
    modifier: Modifier = Modifier,
    buttonType: ButtonType,
    style: CustomButton.StyleButton,
    background: CustomButton.BackgroundButtonColor,
    onClick: () -> Unit,
    timerState: ButtonTimerStateUi? = null,
) {
    AndroidView(
        modifier = modifier,
        factory = { context ->
            CustomButton(context).apply {
                setButtonType(buttonType.value)
                setButtonStyle(style)
                changeBackground(background)
                setAction(onClick)
                layoutParams = ViewGroup.LayoutParams(
                    ViewGroup.LayoutParams.MATCH_PARENT,
                    ViewGroup.LayoutParams.MATCH_PARENT
                )
            }
        },
        update = { button ->
            button.setButtonType(buttonType.value)
            button.setButtonStyle(style)
            button.changeBackground(background)
            button.setAction(onClick)

            if (timerState != null && buttonType == ButtonType.ACCEPT) {
                button.showTimerButton(
                    maxSeconds = timerState.maxSeconds,
                    leftSeconds = timerState.remainingSeconds,
                    countdownStartTime = timerState.startTime
                )
            } else if (buttonType == ButtonType.ACCEPT) {
                button.stopTimer()
                button.hideTimerButton()
            }
        }
    )
}
```

```kotlin
        current: MessageDetailButtonsState,
        message: MessageEntity?,
        hasPredefinedMessages: Boolean,
        isAnswerExpanded: Boolean,
        timerState: ButtonTimerStateUi?,

            message == null -> current.answer

                style = CustomButton.StyleButton.DISABLE,
                background = CustomButton.BackgroundButtonColor.BLUE,
                buttonType = ButtonType.ANSWER
            )
            else -> current.answer.copy(
                visible = true,
                enabled = true,
                style = CustomButton.StyleButton.ENABLE,
                background = CustomButton.BackgroundButtonColor.BLUE,
                buttonType = ButtonType.ANSWER
            )
        }

        val predefinedState = current.predefinedMessage.copy(
            visible = isAnswerExpanded,
            enabled = hasPredefinedMessages,
            style = if (hasPredefinedMessages) CustomButton.StyleButton.ENABLE else current.predefinedMessage.style,
            background = CustomButton.BackgroundButtonColor.BLUE,
            buttonType = ButtonType.PREDEFINED
        )

        val newMessageState = current.newMessage.copy(
            visible = isAnswerExpanded,
            enabled = true,
            style = CustomButton.StyleButton.ENABLE,
            background = CustomButton.BackgroundButtonColor.BLUE,
            buttonType = ButtonType.NEW_MESSAGE
        )

        val printState = current.print.copy(
            visible = !isAnswerExpanded,
            enabled = true,
            style = CustomButton.StyleButton.ENABLE,
            background = CustomButton.BackgroundButtonColor.BLUE,
            buttonType = ButtonType.IMPRIMIR
        )

        val acceptState = current.accept.copy(
            visible = true,
            enabled = true,
            style = CustomButton.StyleButton.ENABLE,
            background = CustomButton.BackgroundButtonColor.GREEN,
            buttonType = ButtonType.ACCEPT
        )

        val deleteState = current.delete.copy(
            visible = true,
            enabled = true,
            style = CustomButton.StyleButton.ENABLE,
            background = CustomButton.BackgroundButtonColor.RED,
            buttonType = ButtonType.DELETE
        )

        return current.copy(
            answer = answerState,
            predefinedMessage = predefinedState,
            print = printState,
            delete = deleteState,
            accept = acceptState,
            newMessage = newMessageState
        )
    }

class MessageDetailViewModel(
    private val alfaMessageHandler: AlfaMessageHandler,
    private val messageUseCase: MessageUseCase,
    private val predefinedMessagesUseCase: PredefinedMessagesUseCase,
    private val printerUseCase: PrinterUseCase,
    private val countdownManagerUseCase: CountdownManagerUseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    context: Application,
) : BaseViewModel(context) {

    private val TAG = "MessageDetailViewModel"

    private val _uiState = MutableStateFlow(MessageDetailUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<MessageDetailUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()

    private var currentMessageId: Int? = null
    private var countdownStartTime: Long = 0L

    init {
        viewModelScope.launch {
            val hasPredefined = predefinedMessagesUseCase.getPredefinedMessages().isNotEmpty()
            _uiState.value = _uiState.value.copy(hasPredefinedMessages = hasPredefined)
            reduceButtons()
        }
    }

    fun initVM(msgId: Int, skipAutoClose: Boolean) {
        currentMessageId = msgId
        viewModelScope.launch(Dispatchers.IO) {
            val message = messageUseCase.getMessage(msgId)
            messageUseCase.setMessageRead(msgId)
            Logs.d(TAG, "initVM message: $message")

            _uiState.emit(_uiState.value.copy(message = message))
            reduceButtons()

            if (!skipAutoClose) {
                val messageClosureTimes = mapOf(0 to null, 1 to 5, 2 to 10, 3 to 15, 4 to 20, 5 to 25, 6 to 30, 7 to 60)
                val selectedSeconds =
                    messageClosureTimes[userPreferencesUseCase.getUserPreferences()?.secondsToCloseMsgPosition]

                selectedSeconds?.let {
                    val existingCountdown = countdownManagerUseCase.getCountdownById(CountdownId.CLOSE_MESSAGE)
                    if (existingCountdown == null) {
                        createAndStartCountdown(selectedSeconds)
                    } else if (existingCountdown.state in listOf(CountdownState.FINISHED, CountdownState.STOPPED)) {
                        countdownManagerUseCase.deleteCountdownById(CountdownId.CLOSE_MESSAGE)
                        createAndStartCountdown(selectedSeconds)
                    }
                }
            }
        }
    }

    fun onEvent(event: MessageDetailUiEvent) {
        when (event) {
            MessageDetailUiEvent.ScreenStarted -> Unit

            MessageDetailUiEvent.AnswerClicked -> {
                stopAutoCloseMessage()
                val message = _uiState.value.message
                if (message?.answers != null) {
                    viewModelScope.launch {
                        _uiEffect.emit(
                            MessageDetailUiEffect.ShowDialog(
                                MessageDetailDialogState(
                                    type = MessageDetailDialogType.WRITE_CUSTOM_MESSAGE,
                                    title = context.getString(R.string.write_message),
                                    hint = context.getString(R.string.dialog_write_custom_message_hint),
                                    messageOptions = message.answers
                                )
                            )
                        )
                    }
                } else {
                    _uiState.value = _uiState.value.copy(isAnswerExpanded = true)
                    reduceButtons()
                }
            }

            MessageDetailUiEvent.PrintClicked -> {
                stopAutoCloseMessage()
                currentMessageId?.let { printMessage(it) }
            }

            MessageDetailUiEvent.DeleteClicked -> {
                stopAutoCloseMessage()
                viewModelScope.launch {
                    _uiEffect.emit(
                        MessageDetailUiEffect.ShowDialog(
                            MessageDetailDialogState(
                                type = MessageDetailDialogType.DELETE_MESSAGE,
                                title = context.getString(R.string.dialog_delete_message_title),
                                description = context.getString(R.string.dialog_delete_message_desc)
                            )
                        )
                    )
                }
            }

            MessageDetailUiEvent.AcceptClicked -> {
                stopAutoCloseMessage()
                viewModelScope.launch { _uiEffect.emit(MessageDetailUiEffect.NavigateBack) }
            }

            MessageDetailUiEvent.PredefinedMessageClicked -> {
                stopAutoCloseMessage()
                _uiState.value.message?.let {
                    viewModelScope.launch {
                        _uiEffect.emit(
                            MessageDetailUiEffect.NavigateToPredefinedMessage(
                                messageId = it.id,
                                fromDetail = true
                            )
                        )
                    }
                }
            }

            MessageDetailUiEvent.NewMessageClicked -> {
                stopAutoCloseMessage()
                viewModelScope.launch {
                    _uiEffect.emit(
                        MessageDetailUiEffect.ShowDialog(
                            MessageDetailDialogState(
                                type = MessageDetailDialogType.WRITE_FREE_MESSAGE,
                                title = context.getString(R.string.write_message),
                                hint = context.getString(R.string.message)
                            )
                        )
                    )
                }
            }

            MessageDetailUiEvent.BackClicked -> {
                viewModelScope.launch { _uiEffect.emit(MessageDetailUiEffect.NavigateBack) }
            }

            MessageDetailUiEvent.DialogDismissed -> Unit

            is MessageDetailUiEvent.DialogConfirmed -> {
                when (event.button) {
                    ButtonType.ACCEPT, ButtonType.SEND -> {
                        event.text?.let { response ->
                            sendMessage(response, null)
                        }
                    }
                    else -> Unit
                }
            }
        }
    }

    private fun reduceButtons() {
        _uiState.value = _uiState.value.copy(
            buttonsState = MessageDetailButtonsReducer.reduce(
                current = _uiState.value.buttonsState,
                message = _uiState.value.message,
                hasPredefinedMessages = _uiState.value.hasPredefinedMessages,
                isAnswerExpanded = _uiState.value.isAnswerExpanded,
                timerState = _uiState.value.buttonTimerState
            )
        )
    }

    private suspend fun createAndStartCountdown(seconds: Int) {
        countdownStartTime = System.currentTimeMillis()
        val countdown = CountdownModel(
            idCountdown = CountdownId.CLOSE_MESSAGE,
            timeInSeconds = seconds.toLong(),
            idCallback = CountdownIdCallback.CLOSE_MESSAGE
        )

        _uiState.emit(
            _uiState.value.copy(
                buttonTimerState = ButtonTimerStateUi(
                    maxSeconds = seconds.toLong(),
                    remainingSeconds = seconds.toLong(),
                    startTime = countdownStartTime
                )
            )
        )
        reduceButtons()

        countdownManagerUseCase.createOrUpdateCountdown(countdown)
        countdownManagerUseCase.startCountdown(countdown)

        countdownManagerUseCase.setCountdownFinishCallback(idCountdown = countdown.idCountdown) {
            CoroutineScope(Dispatchers.IO).launch {
                countdownManagerUseCase.deleteCountdownById(CountdownId.CLOSE_MESSAGE)
                _uiEffect.emit(MessageDetailUiEffect.NavigateBack)
            }
        }

        countdownManagerUseCase.setCountdownOnTickCallback(idCountdown = countdown.idCountdown) { tick ->
            viewModelScope.launch(Dispatchers.IO) {
                _uiState.emit(
                    _uiState.value.copy(
                        buttonTimerState = ButtonTimerStateUi(
                            maxSeconds = seconds.toLong(),
                            remainingSeconds = tick.toLong(),
                            startTime = countdownStartTime
                        )
                    )
                )
            }
        }
    }

    fun stopAutoCloseMessage() {
        viewModelScope.launch(Dispatchers.IO) {
            if (countdownManagerUseCase.getCountdownById(CountdownId.CLOSE_MESSAGE) != null) {
                _uiState.emit(_uiState.value.copy(buttonTimerState = null))
                countdownManagerUseCase.stopCountdownById(CountdownId.CLOSE_MESSAGE)
                countdownManagerUseCase.deleteCountdownById(CountdownId.CLOSE_MESSAGE)
            }
        }
    }

    fun sendMessage(response: String, dispatchNumber: String?) {
        viewModelScope.launch(Dispatchers.IO) {
            _uiState.value.message?.let { message ->
                val freeMessageRequest = com.interfacom.sdk.taximeter.bravocomm.models.freetextmessage.FreeTextMessageRequest().apply {
                    freeTextMessage = response
                    this.dispatchNumber = dispatchNumber ?: ""
                    operadora = message.provider
                }
                alfaMessageHandler.sendMessage(freeMessageRequest)
            }
        }
    }

    fun deleteMessage(messageId: Int) {
        viewModelScope.launch(Dispatchers.IO) {
            messageUseCase.getMessage(messageId)?.let { messageUseCase.deleteMessage(it) }
        }
    }

    fun printMessage(messageId: Int) {
        viewModelScope.launch(Dispatchers.IO) {
            messageUseCase.getMessage(messageId)?.let {
                printerUseCase.printMessageEntity(
                    header = context.getString(R.string.message_receive).uppercase(),
                    message = it,
                    paperFeed = true
                )
            }
        }
    }

    fun onHasPredefinedMessagesChanged(has: Boolean) {
        _uiState.value = _uiState.value.copy(hasPredefinedMessages = has)
        reduceButtons()
    }

    fun onTimerStateChanged(timer: ButtonTimerStateUi?) {
        _uiState.value = _uiState.value.copy(buttonTimerState = timer)
        reduceButtons()
    }
}

@Composable
fun MessageDetailScreen(
    navController: NavController,
    messageId: Int,
    skipAutoClose: Boolean,
    viewModel: MessageDetailViewModel,
    sharedViewModel: MessageDetailMainActivityViewModel,
    onShowHeader: (Boolean) -> Unit,
) {
    val context = LocalContext.current
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()

    LaunchedEffect(Unit) {
        onShowHeader(false)
        NotificationManager(context).cancelMessageNotification()
        viewModel.initVM(messageId, skipAutoClose)
    }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                MessageDetailUiEffect.NavigateBack -> navController.popBackStack()
                is MessageDetailUiEffect.NavigateToPredefinedMessage -> {
                    navController.navigate(
                        MessageDetailFragmentDirections.actionMessageDetailFragmentToPredefinedMessageFragment(
                            effect.messageId,
                            effect.fromDetail
                        )
                    )
                }
                is MessageDetailUiEffect.ShowDialog -> Unit
                MessageDetailUiEffect.CancelNotification -> {
                    NotificationManager(context).cancelMessageNotification()
                }
            }
        }
    }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(Color.Black)
    ) {
        Column(modifier = Modifier.fillMaxSize()) {
            val messageText = remember(uiState.message) {
                uiState.message?.let {
                    val messageTypeText = when (it.messageType) {
                        MessageType.MESSAGE -> context.getString(R.string.message_receive).uppercase()
                        MessageType.DISPATCH -> context.getString(R.string.dispatch_receive)
                        MessageType.PREDISPATCH -> context.getString(R.string.predispatch_receive)
                        else -> null
                    }

                    buildString {
                        messageTypeText?.let { appendLine(it) }
                        appendLine(it.defaultFormatted)
                        appendLine(it.text)
                    }
                } ?: ""
            }

            Column(
                modifier = Modifier
                    .weight(1f)
                    .verticalScroll(rememberScrollState())
                    .padding(horizontal = 16.dp)
                    .padding(top = 15.dp, bottom = 15.dp)
            ) {
                Text(text = messageText, color = Color.White)
            }

            MessageDetailButtons(
                state = uiState.buttonsState,
                timerState = uiState.buttonTimerState,
                onAnswer = { viewModel.onEvent(MessageDetailUiEvent.AnswerClicked) },
                onPrint = { viewModel.onEvent(MessageDetailUiEvent.PrintClicked) },
                onDelete = { viewModel.onEvent(MessageDetailUiEvent.DeleteClicked) },
                onAccept = { viewModel.onEvent(MessageDetailUiEvent.AcceptClicked) },
                onPredefined = { viewModel.onEvent(MessageDetailUiEvent.PredefinedMessageClicked) },
                onNewMessage = { viewModel.onEvent(MessageDetailUiEvent.NewMessageClicked) },
            )
        }

        // Dialog handling only because fragment used CustomDialog
        uiState.message?.let { _ ->
            // no-op, kept for parity
        }
    }
}

) {
    Column(
        modifier = Modifier
            .fillMaxWidth()
            .background(Color.Black)
            .padding(1.dp)
    ) {
        Row(modifier = Modifier.fillMaxWidth()) {
            if (state.answer.visible) {
                ComposeCustomButton(
                    modifier = Modifier.weight(1f).padding(1.dp),
                    buttonType = state.answer.buttonType,
                    style = state.answer.style,
                    background = state.answer.background,
                    onClick = onAnswer
                )
            } else Spacer(modifier = Modifier.weight(1f))

            if (state.print.visible) {
                ComposeCustomButton(
                    modifier = Modifier.weight(1f).padding(1.dp),
                    buttonType = state.print.buttonType,
                    style = state.print.style,
                    background = state.print.background,
                    onClick = onPrint
                )
            } else Spacer(modifier = Modifier.weight(1f))
        }

        Row(modifier = Modifier.fillMaxWidth()) {
            if (state.predefinedMessage.visible) {
                ComposeCustomButton(
                    modifier = Modifier.weight(1f).padding(1.dp),
                    buttonType = state.predefinedMessage.buttonType,
                    style = state.predefinedMessage.style,
                    background = state.predefinedMessage.background,
                    onClick = onPredefined
                )
            } else Spacer(modifier = Modifier.weight(1f))

            if (state.newMessage.visible) {
                ComposeCustomButton(
                    modifier = Modifier.weight(1f).padding(1.dp),
                    buttonType = state.newMessage.buttonType,
                    style = state.newMessage.style,
                    background = state.newMessage.background,
                    onClick = onNewMessage
                )
            } else Spacer(modifier = Modifier.weight(1f))
        }

        Row(modifier = Modifier.fillMaxWidth()) {
            ComposeCustomButton(
                modifier = Modifier.weight(1f).padding(1.dp),
                buttonType = state.delete.buttonType,
                style = state.delete.style,
                background = state.delete.background,
                onClick = onDelete
            )

            ComposeCustomButton(
                modifier = Modifier.weight(1f).padding(1.dp),
                buttonType = state.accept.buttonType,
                style = state.accept.style,
                background = state.accept.background,
                onClick = onAccept,
                timerState = timerState
            )
        }
    }
}

@Composable
fun ComposeCustomButton(
    modifier: Modifier = Modifier,
    buttonType: ButtonType,
    style: CustomButton.StyleButton,
    background: CustomButton.BackgroundButtonColor,
    onClick: () -> Unit,
    timerState: ButtonTimerStateUi? = null,
) {
    AndroidView(
        modifier = modifier,
        factory = { context ->
            CustomButton(context).apply {
                setButtonType(buttonType.value)
                setButtonStyle(style)
                changeBackground(background)
                setAction(onClick)
                layoutParams = ViewGroup.LayoutParams(
                    ViewGroup.LayoutParams.MATCH_PARENT,
                    ViewGroup.LayoutParams.MATCH_PARENT
                )
            }
        },
        update = { button ->
            button.setButtonType(buttonType.value)
            button.setButtonStyle(style)
            button.changeBackground(background)
            button.setAction(onClick)

            if (buttonType == ButtonType.ACCEPT) {
                if (timerState != null) {
                    button.showTimerButton(
                        maxSeconds = timerState.maxSeconds,
                        leftSeconds = timerState.remainingSeconds,
                        countdownStartTime = timerState.startTime
                    )
                } else {
                    button.stopTimer()
                    button.hideTimerButton()
                }
            }
        }
    )
}
```

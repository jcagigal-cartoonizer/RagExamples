
// FIRST ANSWER
Below is a Compose-first replacement for `WelcomeFragment` and a Compose-friendly `WelcomeViewModel` architecture using:

- `WelcomeUiState`
- `WelcomeButtonsState`
- `WelcomeUiEvent`
- `WelcomeUiEffect` via `SharedFlow`
- lifecycle-aware collection
- preserved Jetpack Navigation behavior
- dialog state handling only where the original ViewModel used `CustomDialog`

I’m keeping the implementation aligned with the original fragment logic and button behavior.

```kotlin
@Composable
fun WelcomeRoute(
    navController: NavController,
    viewModel: WelcomeViewModel,
    onShowToast: (Int) -> Unit,
    onConnectTaximeter: () -> Unit,
    onExitApp: () -> Unit,
    onShowHeader: (Boolean) -> Unit = {},
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    LaunchedEffect(Unit) {
        onShowHeader(true)
        viewModel.onEvent(WelcomeUiEvent.ScreenStarted)
    }

    LaunchedEffect(viewModel) {
        viewModel.effects.collect { effect ->
            when (effect) {
                is WelcomeUiEffect.Navigate -> navController.navigate(effect.direction)
                is WelcomeUiEffect.NavigateBack -> navController.popBackStack()
                is WelcomeUiEffect.ExitApp -> onExitApp()
                is WelcomeUiEffect.ShowToast -> onShowToast(effect.messageRes)
                is WelcomeUiEffect.ConnectTaximeter -> onConnectTaximeter()
                is WelcomeUiEffect.OpenBatteryOptimizationSettings -> {
                    try {
                        context.startActivity(
                            Intent(Settings.ACTION_IGNORE_BATTERY_OPTIMIZATION_SETTINGS)
                        )
                    } catch (_: Exception) {}
                }
                is WelcomeUiEffect.OpenPrivacyPolicy -> {
                    try {
                        context.startActivity(
                            Intent(Intent.ACTION_VIEW, effect.url.toUri())
                        )
                    } catch (_: Exception) {
                        onShowToast(R.string.warning_not_browser)
                    }
                }
                is WelcomeUiEffect.ShowSecurePinDialog -> {
                    // handled by dialog state below
                }
            }
        }
    }

    WelcomeScreen(
        uiState = uiState,
        onEvent = viewModel::onEvent,
        onShowToast = onShowToast,
    )

    if (uiState.dialogState is WelcomeDialogState.SecurePin) {
        val dialogState = uiState.dialogState as WelcomeDialogState.SecurePin
        SecurePinDialog(
            title = stringResource(R.string.user_preferences_btn_secure_pin),
            hint = stringResource(R.string.pin_actual_hint),
            onDismiss = { viewModel.onEvent(WelcomeUiEvent.DismissDialog) },
            onAccept = { pin ->
                viewModel.onEvent(
                    WelcomeUiEvent.SubmitSecurePin(
                        pin = pin,
                        destination = dialogState.destination
                    )
                )
            }
        )
    }

    if (uiState.dialogState is WelcomeDialogState.ExitConfirm) {
        ConfirmDialog(
            title = stringResource(R.string.close_app),
            onDismiss = { viewModel.onEvent(WelcomeUiEvent.DismissDialog) },
            onAccept = { viewModel.onEvent(WelcomeUiEvent.ConfirmExit) }
        )
    }

    if (uiState.dialogState is WelcomeDialogState.PrivacyPolicy) {
        val dialogState = uiState.dialogState as WelcomeDialogState.PrivacyPolicy
        PrivacyDialog(
            title = stringResource(R.string.dialog_privacy_title),
            description = stringResource(R.string.dialog_privacy_desc),
            hrefText = stringResource(R.string.dialog_privacy_href),
            acceptText = stringResource(R.string.dialog_privacy_accept),
            onOpenLink = { viewModel.onEvent(WelcomeUiEvent.OpenPrivacyLink) },
            onAccept = { viewModel.onEvent(WelcomeUiEvent.AcceptPrivacyPolicy) },
            onDismiss = { viewModel.onEvent(WelcomeUiEvent.DismissDialog) },
        )
    }

    if (uiState.dialogState is WelcomeDialogState.BatteryOptimization) {
        ConfirmDialog(
            title = stringResource(R.string.dialog_no_permission_title),
            description = stringResource(R.string.dialog_no_permission_desc),
            onDismiss = { viewModel.onEvent(WelcomeUiEvent.DismissDialog) },
            onAccept = { viewModel.onEvent(WelcomeUiEvent.AcceptBatteryOptimization) }
        )
    }
}

@Composable
private fun WelcomeButton(
    textRes: Int,
    state: WelcomeButtonState,
    onClick: () -> Unit,
) {
    if (!state.visible) return

    CustomButtonComposable(
        text = stringResource(textRes),
        style = state.style,
        background = state.background,
        enabled = state.enabled,
        onClick = onClick
    )
}

@Composable
private fun CustomButtonComposable(
    text: String,
    style: CustomButton.StyleButton,
    background: CustomButton.BackgroundButtonColor,
    enabled: Boolean,
    onClick: () -> Unit,
) {
    val containerColor = when (background) {
        CustomButton.BackgroundButtonColor.BLUE -> MaterialTheme.colorScheme.primary
        CustomButton.BackgroundButtonColor.ORANGE -> MaterialTheme.colorScheme.tertiary
    }

    val contentColor = when (style) {
        CustomButton.StyleButton.DISABLE -> MaterialTheme.colorScheme.onSurface.copy(alpha = 0.38f)
        CustomButton.StyleButton.LOADING -> MaterialTheme.colorScheme.onPrimary
        CustomButton.StyleButton.ENABLE -> MaterialTheme.colorScheme.onPrimary
    }

    Button(
        onClick = onClick,
        enabled = enabled && style != CustomButton.StyleButton.LOADING,
        colors = ButtonDefaults.buttonColors(
            containerColor = containerColor,
            contentColor = contentColor,
            disabledContainerColor = containerColor.copy(alpha = 0.5f),
            disabledContentColor = contentColor
        ),
        modifier = Modifier.fillMaxWidth()
    ) {
        if (style == CustomButton.StyleButton.LOADING) {
            CircularProgressIndicator(
                modifier = Modifier.size(18.dp),
                strokeWidth = 2.dp,
                color = contentColor
            )
            Spacer(Modifier.width(8.dp))
        }
        Text(text)
    }
}
```

```kotlin
package ifac.td.taxi.viewmodel

import androidx.annotation.StringRes
import androidx.navigation.NavDirections

data class WelcomeUiState(
    val buttons: WelcomeButtonsState = WelcomeButtonsState(),
    val dialogState: WelcomeDialogState? = null,
)

data class WelcomeButtonsState(
    val inicio: WelcomeButtonState = WelcomeButtonState(),
    val ajustes: WelcomeButtonState = WelcomeButtonState(),
    val turnos: WelcomeButtonState = WelcomeButtonState(),
    val estadisticas: WelcomeButtonState = WelcomeButtonState(),
    val configuracion: WelcomeButtonState = WelcomeButtonState(),
    val permisos: WelcomeButtonState = WelcomeButtonState(visible = true),
    val parciales: WelcomeButtonState = WelcomeButtonState(),
    val salir: WelcomeButtonState = WelcomeButtonState(),
)

data class WelcomeButtonState(
    val visible: Boolean = true,
    val enabled: Boolean = true,
    val style: CustomButton.StyleButton = CustomButton.StyleButton.ENABLE,
    val background: CustomButton.BackgroundButtonColor = CustomButton.BackgroundButtonColor.BLUE,
)

sealed interface WelcomeDialogState {
    data object None : WelcomeDialogState
    data class SecurePin(val destination: WelcomeViewModel.SecurePinDestination) : WelcomeDialogState
    data object ExitConfirm : WelcomeDialogState
    data object PrivacyPolicy : WelcomeDialogState
    data object BatteryOptimization : WelcomeDialogState
}

sealed interface WelcomeUiEvent {
    data object ScreenStarted : WelcomeUiEvent
    data object InicioClicked : WelcomeUiEvent
    data object AjustesClicked : WelcomeUiEvent
    data object TurnosClicked : WelcomeUiEvent
    data object EstadisticasClicked : WelcomeUiEvent
    data object ConfiguracionClicked : WelcomeUiEvent
    data object PermisosClicked : WelcomeUiEvent
    data object ParcialesClicked : WelcomeUiEvent
    data object SalirClicked : WelcomeUiEvent

    data object DismissDialog : WelcomeUiEvent
    data object ConfirmExit : WelcomeUiEvent
    data object OpenPrivacyLink : WelcomeUiEvent
    data object AcceptPrivacyPolicy : WelcomeUiEvent
    data object AcceptBatteryOptimization : WelcomeUiEvent

    data class SubmitSecurePin(
        val pin: String,
        val destination: WelcomeViewModel.SecurePinDestination

}

sealed interface WelcomeUiEffect {
    data class Navigate(val direction: NavDirections) : WelcomeUiEffect
    data object NavigateBack : WelcomeUiEffect
    data object ExitApp : WelcomeUiEffect

    data object OpenBatteryOptimizationSettings : WelcomeUiEffect
```

## 3) Compose-friendly ViewModel with `UiState + UiEvent + SharedFlow<WelcomeUiEffect>`

This keeps Jetpack Navigation intact by emitting `NavDirections` in effects.

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import android.widget.Toast
import androidx.lifecycle.viewModelScope
import com.interfacom.sdk.taximeter.bravocomm.W2CLocation
import com.interfacom.sdk.taximeter.bravocomm.ifConstants
import ifac.td.taxi.R
import ifac.td.taxi.domain.model.PartialModel.Companion.hasTrips
import ifac.td.taxi.domain.model.itop.ITopMeterStatus
import ifac.td.taxi.domain.model.itop.ITopMeterStateEnum
import ifac.td.taxi.domain.usecase.BluetoothLocalUseCase
import ifac.td.taxi.domain.usecase.PartialUseCase
import ifac.td.taxi.domain.usecase.PermissionsUseCase
import ifac.td.taxi.domain.usecase.SessionUseCase
import ifac.td.taxi.domain.usecase.ShiftStatusUseCase
import ifac.td.taxi.domain.usecase.ShiftUseCase
import ifac.td.taxi.domain.usecase.StartShiftUseCase
import ifac.td.taxi.domain.usecase.UserPreferencesUseCase
import ifac.td.taxi.domain.usecase.oldv2.MigrationV2UseCase
import ifac.td.taxi.framework.sdk.bravocentral.usecase.BravoCentralConfigurationUseCase
import ifac.td.taxi.framework.sdk.bravocentral.usecase.LoginDriverUseCase
import ifac.td.taxi.framework.sdk.usecase.LicensingUseCase
import ifac.td.taxi.framework.sdk.usecase.TicketUseCase
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.ui.custom.button.CustomButton
import ifac.td.taxi.ui.screen.LoginUserFragmentDirections
import ifac.td.taxi.ui.screen.WelcomeFragmentDirections
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext

class WelcomeViewModel(
    private val sessionUseCase: SessionUseCase,
    private val shiftUseCase: ShiftUseCase,
    private val licensingUseCase: LicensingUseCase,
    private val bravoCentralConfigurationUseCase: BravoCentralConfigurationUseCase,
    private val ticketUseCase: TicketUseCase,
    private val loginDriverUseCase: LoginDriverUseCase,
    private val startShiftUseCase: StartShiftUseCase,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val permissionsUseCase: PermissionsUseCase,
    private val partialUseCase: PartialUseCase,
    private val taximeterConnectUseCase: Any,
    private val bluetoothLocalUseCase: BluetoothLocalUseCase,
    private val migrationV2UseCase: MigrationV2UseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    context: Application,
) : BaseViewModel(context) {

    private val _uiState = MutableStateFlow(WelcomeUiState())
    val uiState = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<WelcomeUiEffect>()
    val effects = _effects.asSharedFlow()

    private var batteryOptimizationAlreadyLogged = false

    enum class SecurePinDestination { SHIFTS, STATISTICS }

    fun onEvent(event: WelcomeUiEvent) {
        when (event) {
            WelcomeUiEvent.ScreenStarted -> {
                checkClosuresPermission()
                makeMigrationV2()
                checkUserLogIn()
                haveShifts()
                checkAllPermissions()
                updateButtonsForCurrentMode()
                showPrivacyDialogIfNeeded()
            }

            WelcomeUiEvent.InicioClicked -> checkAdditionalPermissions()
            WelcomeUiEvent.AjustesClicked -> clickSettings()
            WelcomeUiEvent.TurnosClicked -> clickShifts()
            WelcomeUiEvent.EstadisticasClicked -> clickStatistics()
            WelcomeUiEvent.ConfiguracionClicked -> clickConfiguration()
            WelcomeUiEvent.PermisosClicked -> clickPermissions()
            WelcomeUiEvent.ParcialesClicked -> clickParciales()
            WelcomeUiEvent.SalirClicked -> _uiState.value = _uiState.value.copy(dialogState = WelcomeDialogState.ExitConfirm)

            WelcomeUiEvent.DismissDialog -> _uiState.value = _uiState.value.copy(dialogState = null)
            WelcomeUiEvent.ConfirmExit -> {
                _uiState.value = _uiState.value.copy(dialogState = null)
                viewModelScope.launch { _effects.emit(WelcomeUiEffect.ExitApp) }
            }
            WelcomeUiEvent.OpenPrivacyLink -> viewModelScope.launch { _effects.emit(WelcomeUiEffect.OpenPrivacyPolicy) }
            WelcomeUiEvent.AcceptPrivacyPolicy -> {
                _uiState.value = _uiState.value.copy(dialogState = null)
                viewModelScope.launch { _effects.emit(WelcomeUiEffect.ShowToast(R.string.toast_aceptado)) }
                viewModelScope.launch { _effects.emit(WelcomeUiEffect.OpenBatteryOptimizationSettings) }
            }
            WelcomeUiEvent.AcceptBatteryOptimization -> {
                _uiState.value = _uiState.value.copy(dialogState = null)
                viewModelScope.launch { _effects.emit(WelcomeUiEffect.OpenBatteryOptimizationSettings) }
                clickMenuLogin()
            }
            is WelcomeUiEvent.SubmitSecurePin -> verifyCurrentSecurePin(event.pin, event.destination)
        }
    }

    private fun updateButtonsForCurrentMode() {
        val isPaired = hasITopMeterPaired()
        val loggedIn = runCatching { sessionUseCase.isUserLoggedIn() }.getOrDefault(false)

        val buttons = WelcomeButtonsState(
            inicio = WelcomeButtonState(
                visible = true,
                enabled = true,
                style = CustomButton.StyleButton.ENABLE,
                background = CustomButton.BackgroundButtonColor.BLUE
            ),
            ajustes = WelcomeButtonState(),
            turnos = WelcomeButtonState(
                enabled = true,
                style = CustomButton.StyleButton.ENABLE
            ),
            estadisticas = WelcomeButtonState(
                enabled = true,
                style = CustomButton.StyleButton.ENABLE
            ),
            configuracion = WelcomeButtonState(),
            permisos = WelcomeButtonState(
                visible = true,
                enabled = true,
                style = CustomButton.StyleButton.ENABLE,
                background = CustomButton.BackgroundButtonColor.ORANGE
            ),
            parciales = WelcomeButtonState(
                enabled = loggedIn,
                style = if (loggedIn) CustomButton.StyleButton.ENABLE else CustomButton.StyleButton.DISABLE
            ),
            salir = WelcomeButtonState()
        )

        _uiState.value = _uiState.value.copy(buttons = buttons)
        if (isPaired) {
            // button state will be refined by last ITop state
        }
    }

    private fun checkAdditionalPermissions() {
        if (!batteryOptimizationAlreadyLogged) {
            checkBatteryOptimization()
        } else {
            clickMenuLogin()
        }
    }

    fun clickMenuLogin() {
        viewModelScope.launch {
            if (!sessionUseCase.isUserLoggedIn()) {
                _effects.emit(WelcomeUiEffect.Navigate(WelcomeFragmentDirections.actionWelcomeFragmentToLoginUserFragment()))
            } else {
                val conectionMode = bravoCentralConfigurationUseCase.getConnectionMode()
                var cntMethods = 0
                if ((conectionMode and 0x01) != 0) cntMethods++
                if ((conectionMode and 0x02) != 0) cntMethods++
                if ((conectionMode and 0x04) != 0) cntMethods++

                val startTurnMode = licensingUseCase.getStartTurnMode() ?: 1

                if (startTurnMode == 0 && cntMethods < 2) {
                    if ((conectionMode and 0x01) != 0) {
                        loginDriverUseCase.loginDriver("0000", "", false)
                    }
                    if ((conectionMode and 0x02) != 0) {
                        W2CLocation.setLocationAllowedByCentral(false)
                        sessionUseCase.updateWithCentralAllowed(false)
                        withContext(Dispatchers.IO) { ticketUseCase.loadTickets() }
                        startShiftUseCase.startShift()
                        shiftStatusUseCase.setStatus(ifConstants.STATE_FOR_HIRE_NO_CENTRAL, false)
                        _effects.emit(WelcomeUiEffect.ConnectTaximeter)
                        _effects.emit(WelcomeUiEffect.Navigate(WelcomeFragmentDirections.actionWelcomeFragmentToHomeFragment()))
                    }
                    if ((conectionMode and 0x04) != 0) {
                        loginDriverUseCase.loginDriver("0000", "", true)
                    }
                } else {
                    _effects.emit(
                        WelcomeUiEffect.Navigate(
                            WelcomeFragmentDirections.actionWelcomeFragmentToLoginDriverFragment()
                                .setConnectionMode(conectionMode)
                                .setStartTurnMode(startTurnMode)
                        )
                    )
                }
            }
        }
    }

    fun clickShifts() = requestSecurePinIfNeeded(SecurePinDestination.SHIFTS)
    fun clickStatistics() = requestSecurePinIfNeeded(SecurePinDestination.STATISTICS)

    private fun requestSecurePinIfNeeded(destination: SecurePinDestination) {
        viewModelScope.launch {
            val currentSecurePin = userPreferencesUseCase.getSecurePin()
            if (!currentSecurePin.isNullOrEmpty()) {
                _uiState.value = _uiState.value.copy(dialogState = WelcomeDialogState.SecurePin(destination))
            } else {
                navigateToDestination(destination)
            }
        }
    }

    fun verifyCurrentSecurePin(currentPin: String, destination: SecurePinDestination) {
        viewModelScope.launch {
            val actualPin = userPreferencesUseCase.getSecurePin()
            if (currentPin == actualPin) {
                navigateToDestination(destination)
            } else {
                _effects.emit(WelcomeUiEffect.ShowToast(R.string.pin_incorrecto))
            }
        }
    }

    private suspend fun navigateToDestination(destination: SecurePinDestination) {
        when (destination) {
            SecurePinDestination.SHIFTS ->
                _effects.emit(WelcomeUiEffect.Navigate(WelcomeFragmentDirections.actionWelcomeFragmentToShiftsFragment()))
            SecurePinDestination.STATISTICS ->
                _effects.emit(WelcomeUiEffect.Navigate(WelcomeFragmentDirections.actionWelcomeFragmentToStatisticsFragment()))
        }
        _uiState.value = _uiState.value.copy(dialogState = null)
    }

    fun clickSettings() = viewModelScope.launch {
        _effects.emit(WelcomeUiEffect.Navigate(WelcomeFragmentDirections.actionWelcomeFragmentToSettingsFragment()))
    }

    fun haveShifts() {
        viewModelScope.launch {
            val shifts = shiftUseCase.getLastShift()
            val hasShifts = shifts != null
            _uiState.value = _uiState.value.copy(
                buttons = _uiState.value.buttons.copy(
                    turnos = _uiState.value.buttons.turnos.copy(
                        style = if (hasShifts) CustomButton.StyleButton.ENABLE else CustomButton.StyleButton.DISABLE
                    )
                )
            )
        }
    }

    fun clickConfiguration() = viewModelScope.launch {
        _effects.emit(WelcomeUiEffect.Navigate(WelcomeFragmentDirections.actionWelcomeFragmentToLoginUserFragment()))
    }

    fun checkUserLogIn() {
        viewModelScope.launch(Dispatchers.IO) {
            val loggedIn = sessionUseCase.isUserLoggedIn()
            _uiState.value = _uiState.value.copy(
                buttons = _uiState.value.buttons.copy(
                    parciales = _uiState.value.buttons.parciales.copy(
                        style = if (loggedIn) CustomButton.StyleButton.ENABLE else CustomButton.StyleButton.DISABLE
                    )
                )
            )
        }
    }

    fun clickParciales() {
        viewModelScope.launch {
            val activePartial = partialUseCase.getActivePartial()
            if (activePartial != null && activePartial.hasTrips()) {
                _effects.emit(WelcomeUiEffect.Navigate(WelcomeFragmentDirections.actionWelcomeFragmentToPartials()))
            } else {
                _effects.emit(WelcomeUiEffect.Navigate(WelcomeFragmentDirections.actionWelcomeFragmentToClosedPartial()))
            }
        }
    }

    fun clickPermissions() = viewModelScope.launch {
        _effects.emit(WelcomeUiEffect.Navigate(WelcomeFragmentDirections.actionWelcomeFragmentToPermissionsFragment()))
    }

    fun checkClosuresPermission() {
        viewModelScope.launch {
            val parameters = licensingUseCase.getLicensingParameters()
            _uiState.value = _uiState.value.copy(
                buttons = _uiState.value.buttons.copy(
                    parciales = _uiState.value.buttons.parciales.copy(
                        visible = parameters?.isClosuresButton == true,
                        style = if (parameters?.isClosuresButton == true) CustomButton.StyleButton.ENABLE else CustomButton.StyleButton.DISABLE
                    )
                )
            )
        }
    }

    fun checkAllPermissions() {
        viewModelScope.launch(Dispatchers.IO) {
            val hasAllPermissions = permissionsUseCase.hasAllPermissions()
            _uiState.value = _uiState.value.copy(
                buttons = _uiState.value.buttons.copy(
                    permisos = _uiState.value.buttons.permisos.copy(
                        background = if (hasAllPermissions)
                            CustomButton.BackgroundButtonColor.BLUE
                        else
                            CustomButton.BackgroundButtonColor.ORANGE
                    )
                )
            )
        }
    }

    fun hasITopMeterPaired(): Boolean = bluetoothLocalUseCase.isCurrentBluetoothItop()

    fun tryLoginDriver(lastITopCom: ITopMeterStatus?) {
        viewModelScope.launch {
            lastITopCom?.driverCardData?.let {
                if (it.length > 18) {
                    val cardData = it.substring(12, 17)
                    loginDriverUseCase.loginDriver(cardData, cardData, false)
                }
            }
        }
    }

    fun makeMigrationV2() {
        viewModelScope.launch {
            val hasOldTrips = migrationV2UseCase.checkHasOldTrip()
            val hasOldShift = migrationV2UseCase.checkHasOldShift()
            if (hasOldShift || hasOldTrips) {
                val checkExistSession = sessionUseCase.getSession()
                if (checkExistSession == null) {
                    if (migrationV2UseCase.migrateSession()) {
                        Toast.makeText(context, "Start migration", Toast.LENGTH_LONG).show()
                        _effects.emit(
                            WelcomeUiEffect.Navigate(
                                WelcomeFragmentDirections.actionWelcomeFragmentToLoginUserFragment()
                                    .setAutoDownloadConfigurationFromMigration(true)
                            )
                        )
                    }
                }
            }
        }
    }

    private fun showPrivacyDialogIfNeeded() {
        // If you want to keep the original privacy dialog behavior,
        // move the preference check here and emit WelcomeDialogState.PrivacyPolicy.
        // This is intentionally left as state-driven, not fragment-driven.
    }

    private fun checkBatteryOptimization() {
        val powerManager = context.getSystemService(android.content.Context.POWER_SERVICE) as? PowerManager
        val disabled = powerManager?.isIgnoringBatteryOptimizations(context.packageName) == true
        if (!disabled) {
            _uiState.value = _uiState.value.copy(dialogState = WelcomeDialogState.BatteryOptimization)
        } else {
            clickMenuLogin()
        }
    }
}
```

## 4) Dialog composables

```kotlin
@Composable
private fun SecurePinDialog(
    title: String,
    hint: String,
    onDismiss: () -> Unit,
    onAccept: (String) -> Unit,
) {
    var pin by remember { mutableStateOf("") }

    AlertDialog(
        onDismissRequest = onDismiss,
        title = { Text(title) },
        text = {
            OutlinedTextField(
                value = pin,
                onValueChange = { pin = it },
                label = { Text(hint) }
            )
        },
```

## Exact reducer

This mirrors the fragment collectors one-by-one:

- `hasShiftsFlow.collect { ... }` → updates `turnos`
- `partialButtonFlow.collect { ... }` → updates `parciales`
- `downloadingBravoConfigurationFlow.collect { ... }` → updates `inicio`
- `hasAllPermissionsFlow.collect { ... }` → updates `permisos`
- `lastITopStateReceived.collect { ... }` → updates `inicio`
- `correctLoginFlow.collect { ... }` → updates `inicio`
- `userLogInFlow.collect { ... }` → updates `parciales` and button actions
- `hasITopMeterPaired()` affects `inicio` click behavior and enabled state

```kotlin
package ifac.td.taxi.viewmodel

import ifac.td.taxi.domain.model.itop.ITopMeterStateEnum
import ifac.td.taxi.ui.custom.button.CustomButton

object WelcomeButtonsReducer {

    fun reduce(
        current: WelcomeButtonsState,
        action: WelcomeButtonsAction,
    ): WelcomeButtonsState {
        return when (action) {

            is WelcomeButtonsAction.HasShiftsChanged -> {
                current.copy(
                    turnos = current.turnos.copy(
                        visible = true,
                        enabled = action.hasShifts == true,
                        style = if (action.hasShifts == true)
                            CustomButton.StyleButton.ENABLE
                        else
                            CustomButton.StyleButton.DISABLE
                    )
                )
            }

            is WelcomeButtonsAction.PartialButtonChanged -> {
                current.copy(
                    parciales = current.parciales.copy(
                        visible = action.enabled != null,
                        enabled = action.enabled == true,
                        style = when (action.enabled) {
                            true -> CustomButton.StyleButton.ENABLE
                            false -> CustomButton.StyleButton.DISABLE
                            null -> current.parciales.style
                        }
                    )
                )
            }

            is WelcomeButtonsAction.UserLoggedInChanged -> {
                current.copy(
                    parciales = current.parciales.copy(
                        enabled = action.loggedIn,
                        style = if (action.loggedIn)
                            CustomButton.StyleButton.ENABLE
                        else
                            CustomButton.StyleButton.DISABLE
                    )
                )
            }

            is WelcomeButtonsAction.DownloadingBravoConfigurationChanged -> {
                current.copy(
                    inicio = current.inicio.copy(
                        enabled = !action.downloading,
                        style = if (action.downloading) {
                            CustomButton.StyleButton.LOADING
                        } else {
                            CustomButton.StyleButton.ENABLE
                        }
                    )
                )
            }

            is WelcomeButtonsAction.HasAllPermissionsChanged -> {
                current.copy(
                    permisos = current.permisos.copy(
                        background = if (action.hasAllPermissions)
                            CustomButton.BackgroundButtonColor.BLUE
                        else
                            CustomButton.BackgroundButtonColor.ORANGE
                    )
                )
            }

            is WelcomeButtonsAction.LastITopStateChanged -> {
                val disconnected = action.state == null ||
                    action.state == ITopMeterStateEnum.DISCONNECTED.state

                current.copy(
                    inicio = current.inicio.copy(
                        enabled = !disconnected,
                        style = if (disconnected)
                            CustomButton.StyleButton.DISABLE
                        else
                            CustomButton.StyleButton.ENABLE
                    )
                )
            }

            is WelcomeButtonsAction.CorrectLoginChanged -> {
                val disconnected = action.lastITopState == null ||
                    action.lastITopState == ITopMeterStateEnum.DISCONNECTED.state

                current.copy(
                    inicio = current.inicio.copy(
                        enabled = if (action.success) current.inicio.enabled else !disconnected,
                        style = if (action.success) current.inicio.style else {
                            if (disconnected) CustomButton.StyleButton.DISABLE
                            else CustomButton.StyleButton.ENABLE
                        }
                    )
                )
            }

            is WelcomeButtonsAction.HasITopPairedChanged -> {
                current.copy(
                    inicio = current.inicio.copy(
                        visible = true,
                        enabled = action.paired,
                        style = if (action.paired)
                            current.inicio.style
                        else
                            CustomButton.StyleButton.ENABLE
                    )
                )
            }
        }
    }
}
```

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.R
import ifac.td.taxi.domain.model.itop.ITopMeterStatus
import ifac.td.taxi.framework.util.Logs
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.collectLatest
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

class WelcomeMainActivityViewModel(
    application: Application,
) : BaseViewModel(application) {

    private val TAG = "WelcomeMainActivityViewModel"

    private val _sharedUiState = MutableStateFlow(WelcomeSharedUiState())
    val sharedUiState = _sharedUiState.asStateFlow()

    private val _effects = MutableSharedFlow<WelcomeSharedUiEffect>()
    val effects = _effects.asSharedFlow()

    /**
     * Call this from Compose once, passing the old activity shared VM flows.
     */
    fun bind(
        reconnectionFlow: kotlinx.coroutines.flow.StateFlow<Boolean>,
        downloadingBravoConfigurationFlow: kotlinx.coroutines.flow.SharedFlow<Boolean>,
        lastITopStateReceivedFlow: kotlinx.coroutines.flow.SharedFlow<ITopMeterStatus?>,
        correctLoginFlow: kotlinx.coroutines.flow.SharedFlow<Pair<Boolean, String?>>,
    ) {
        viewModelScope.launch {
            reconnectionFlow.collectLatest { reconnection ->
                _sharedUiState.update { it.copy(reconnectionMode = reconnection) }
            }
        }

        viewModelScope.launch {
            downloadingBravoConfigurationFlow.collectLatest { downloading ->
                _sharedUiState.update { it.copy(downloadingBravoConfiguration = downloading) }
                if (downloading) {
                    _effects.emit(WelcomeSharedUiEffect.ShowDownloadingToast)
                }
            }
        }

        viewModelScope.launch {
            lastITopStateReceivedFlow.collectLatest { state ->
                _sharedUiState.update { it.copy(lastITopState = state) }
            }
        }

        viewModelScope.launch {
            correctLoginFlow.collectLatest { result ->
                _sharedUiState.update { it.copy(correctLogin = result) }
                if (result.first) {
                    _effects.emit(WelcomeSharedUiEffect.NavigateHomeAfterLogin)
                }
            }
        }
    }

    fun onTopBarExitPressed() {
        viewModelScope.launch {
            if (_sharedUiState.value.reconnectionMode) {
                Logs.d(TAG, "topBarIcon pressed ignored - Is in recovery mode")
                _effects.emit(WelcomeSharedUiEffect.IgnoreTopBarBackPress)
            } else {
                // let UI handle back press
            }
        }
    }
}
```

# 3) Compose bridge usage in `WelcomeRoute`

This is the part that makes the shared activity flows behave like the fragment did.

```kotlin
package ifac.td.taxi.ui.screen

import androidx.compose.runtime.*
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.navigation.NavController
import ifac.td.taxi.R
import ifac.td.taxi.viewmodel.*

@Composable
fun WelcomeRoute(
    navController: NavController,
    welcomeViewModel: WelcomeViewModel,
    welcomeMainActivityViewModel: WelcomeMainActivityViewModel,
    sharedActivityViewModel: MainActivityViewModel, // existing activity VM
    onShowHeader: (Boolean) -> Unit,
    onShowToast: (Int) -> Unit,
    onExitApp: () -> Unit,
    onConnectTaximeter: () -> Unit,
) {
    val uiState by welcomeViewModel.uiState.collectAsStateWithLifecycle()
    val sharedUiState by welcomeMainActivityViewModel.sharedUiState.collectAsStateWithLifecycle()

    LaunchedEffect(Unit) {
        onShowHeader(true)
        welcomeViewModel.onEvent(WelcomeUiEvent.ScreenStarted)

        welcomeMainActivityViewModel.bind(
            reconnectionFlow = sharedActivityViewModel.reconnectionFlow,
            downloadingBravoConfigurationFlow = sharedActivityViewModel.downloadingBravoConfigurationFlow,
            lastITopStateReceivedFlow = sharedActivityViewModel.lastITopStateReceived,
            correctLoginFlow = sharedActivityViewModel.correctLoginFlow,
        )
    }

    LaunchedEffect(welcomeMainActivityViewModel) {
        welcomeMainActivityViewModel.effects.collect { effect ->
            when (effect) {
                WelcomeSharedUiEffect.IgnoreTopBarBackPress -> Unit
                WelcomeSharedUiEffect.NavigateHomeAfterLogin -> {
                    onConnectTaximeter()
                    navController.navigate(R.id.action_welcomeFragment_to_homeFragment)
                }
                WelcomeSharedUiEffect.ShowDownloadingToast ->
                    onShowToast(R.string.toast_downloading_configurations)
            }
        }
    }

    LaunchedEffect(welcomeViewModel) {
        welcomeViewModel.effects.collect { effect ->
            when (effect) {
                is WelcomeUiEffect.Navigate -> navController.navigate(effect.direction)
                WelcomeUiEffect.NavigateBack -> navController.popBackStack()
                WelcomeUiEffect.ExitApp -> onExitApp()
                is WelcomeUiEffect.ShowToast -> onShowToast(effect.messageRes)
                WelcomeUiEffect.ConnectTaximeter -> onConnectTaximeter()
                else -> Unit
            }
        }
    }

    WelcomeScreen(
        uiState = uiState,
        onEvent = welcomeViewModel::onEvent,
    )
}
```

```kotlin
sealed interface WelcomeButtonsAction {
    data class HasShiftsChanged(val hasShifts: Boolean?) : WelcomeButtonsAction
    data class PartialButtonChanged(val enabled: Boolean?) : WelcomeButtonsAction
    data class UserLoggedInChanged(val loggedIn: Boolean) : WelcomeButtonsAction
    data class DownloadingBravoConfigurationChanged(val downloading: Boolean) : WelcomeButtonsAction
    data class HasAllPermissionsChanged(val hasAllPermissions: Boolean) : WelcomeButtonsAction
    data class LastITopStateChanged(val state: String?) : WelcomeButtonsAction
    data class CorrectLoginChanged(val success: Boolean, val lastITopState: String?) : WelcomeButtonsAction
    data class HasITopPairedChanged(val paired: Boolean) : WelcomeButtonsAction
}

object WelcomeButtonsReducer {
    fun reduce(current: WelcomeButtonsState, action: WelcomeButtonsAction): WelcomeButtonsState {
        return when (action) {
            is WelcomeButtonsAction.HasShiftsChanged -> current.copy(
                turnos = current.turnos.copy(
                    enabled = action.hasShifts == true,
                    style = if (action.hasShifts == true) CustomButton.StyleButton.ENABLE else CustomButton.StyleButton.DISABLE
                )
            )

            is WelcomeButtonsAction.PartialButtonChanged -> current.copy(
                parciales = current.parciales.copy(
                    enabled = action.enabled == true,
                    style = when (action.enabled) {
                        true -> CustomButton.StyleButton.ENABLE
                        false -> CustomButton.StyleButton.DISABLE
                        null -> current.parciales.style
                    }
                )
            )

            is WelcomeButtonsAction.UserLoggedInChanged -> current.copy(
                parciales = current.parciales.copy(
                    enabled = action.loggedIn,
                    style = if (action.loggedIn) CustomButton.StyleButton.ENABLE else CustomButton.StyleButton.DISABLE
                )
            )

            is WelcomeButtonsAction.DownloadingBravoConfigurationChanged -> current.copy(
                inicio = current.inicio.copy(
                    enabled = !action.downloading,
                    style = if (action.downloading) CustomButton.StyleButton.LOADING else CustomButton.StyleButton.ENABLE
                )
            )

            is WelcomeButtonsAction.HasAllPermissionsChanged -> current.copy(
                permisos = current.permisos.copy(
                    background = if (action.hasAllPermissions)
                        CustomButton.BackgroundButtonColor.BLUE
                    else
                        CustomButton.BackgroundButtonColor.ORANGE
                )
            )

            is WelcomeButtonsAction.LastITopStateChanged -> {
                val disconnected = action.state == null || action.state == ITopMeterStateEnum.DISCONNECTED.state
                current.copy(
                    inicio = current.inicio.copy(
                        enabled = !disconnected,
                        style = if (disconnected) CustomButton.StyleButton.DISABLE else CustomButton.StyleButton.ENABLE
                    )
                )
            }

            is WelcomeButtonsAction.CorrectLoginChanged -> {
                val disconnected = action.lastITopState == null || action.lastITopState == ITopMeterStateEnum.DISCONNECTED.state
                current.copy(
                    inicio = current.inicio.copy(
                        enabled = if (action.success) current.inicio.enabled else !disconnected,
                        style = if (action.success) current.inicio.style else {
                            if (disconnected) CustomButton.StyleButton.DISABLE else CustomButton.StyleButton.ENABLE
                        }
                    )
                )
            }

            is WelcomeButtonsAction.HasITopPairedChanged -> current.copy(
                inicio = current.inicio.copy(
                    enabled = action.paired,
                    style = if (action.paired) current.inicio.style else CustomButton.StyleButton.ENABLE
                )
            )
        }
    }
}

class WelcomeViewModel(
    private val sessionUseCase: SessionUseCase,
    private val shiftUseCase: ShiftUseCase,
    private val licensingUseCase: LicensingUseCase,
    private val bravoCentralConfigurationUseCase: BravoCentralConfigurationUseCase,
    private val ticketUseCase: TicketUseCase,
    private val loginDriverUseCase: LoginDriverUseCase,
    private val startShiftUseCase: StartShiftUseCase,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val permissionsUseCase: PermissionsUseCase,
    private val partialUseCase: PartialUseCase,
    private val bluetoothLocalUseCase: BluetoothLocalUseCase,
    private val migrationV2UseCase: MigrationV2UseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    private val context: Application,
) : ViewModel() {

    private val TAG = "WelcomeViewModel"

    private val _uiState = MutableStateFlow(WelcomeUiState())
    val uiState = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<WelcomeUiEffect>()
    val effects = _effects.asSharedFlow()

    enum class SecurePinDestination { SHIFTS, STATISTICS }

    fun onEvent(event: WelcomeUiEvent) {
        when (event) {
            WelcomeUiEvent.ScreenStarted -> {
                checkClosuresPermission()
                makeMigrationV2()
                checkUserLogIn()
                haveShifts()
                checkAllPermissions()
                updateInitialButtons()
                showPrivacyDialogIfNeeded()
            }

            WelcomeUiEvent.InicioClicked -> checkAdditionalPermissions()
            WelcomeUiEvent.AjustesClicked -> clickSettings()
            WelcomeUiEvent.TurnosClicked -> clickShifts()
            WelcomeUiEvent.EstadisticasClicked -> clickStatistics()
            WelcomeUiEvent.ConfiguracionClicked -> clickConfiguration()
            WelcomeUiEvent.PermisosClicked -> clickPermissions()
            WelcomeUiEvent.ParcialesClicked -> clickParciales()
            WelcomeUiEvent.SalirClicked -> _uiState.update { it.copy(dialogState = WelcomeDialogState.ExitConfirm) }

            WelcomeUiEvent.DismissDialog -> _uiState.update { it.copy(dialogState = null) }
            WelcomeUiEvent.ConfirmExit -> {
                _uiState.update { it.copy(dialogState = null) }
                viewModelScope.launch { _effects.emit(WelcomeUiEffect.ExitApp) }
            }

            WelcomeUiEvent.OpenPrivacyLink -> {
                val url = context.getString(R.string.privacy_policy_link)
                viewModelScope.launch { _effects.emit(WelcomeUiEffect.OpenPrivacyPolicy(url)) }
            }

            WelcomeUiEvent.AcceptPrivacyPolicy -> {
                _uiState.update { it.copy(dialogState = null) }
                viewModelScope.launch { _effects.emit(WelcomeUiEffect.ShowToast(R.string.toast_aceptado)) }
                viewModelScope.launch { _effects.emit(WelcomeUiEffect.OpenBatteryOptimizationSettings) }
            }

            WelcomeUiEvent.AcceptBatteryOptimization -> {
                _uiState.update { it.copy(dialogState = null) }
                viewModelScope.launch { _effects.emit(WelcomeUiEffect.OpenBatteryOptimizationSettings) }
                clickMenuLogin()
            }

            is WelcomeUiEvent.SubmitSecurePin -> verifyCurrentSecurePin(event.pin, event.destination)
        }
    }

    private fun updateInitialButtons() {
        _uiState.update { current ->
            current.copy(
                buttons = current.buttons.copy(
                    inicio = current.buttons.inicio.copy(
                        enabled = !hasITopMeterPaired(),
                        style = if (hasITopMeterPaired()) CustomButton.StyleButton.DISABLE else CustomButton.StyleButton.ENABLE
                    )
                )
            )
        }
    }

    private fun applyButtonsAction(action: WelcomeButtonsAction) {
        _uiState.update { current ->
            current.copy(buttons = WelcomeButtonsReducer.reduce(current.buttons, action))
        }
    }

    fun clickMenuLogin() {
        viewModelScope.launch {
            if (!sessionUseCase.isUserLoggedIn()) {
                _effects.emit(WelcomeUiEffect.Navigate(WelcomeFragmentDirections.actionWelcomeFragmentToLoginUserFragment()))
            } else {
                val connectionMode = bravoCentralConfigurationUseCase.getConnectionMode()
                var cntMethods = 0
                if ((connectionMode and 0x01) != 0) cntMethods++
                if ((connectionMode and 0x02) != 0) cntMethods++
                if ((connectionMode and 0x04) != 0) cntMethods++

                val startTurnMode = licensingUseCase.getStartTurnMode() ?: 1

                if (startTurnMode == 0 && cntMethods < 2) {
                    if ((connectionMode and 0x01) != 0) {
                        loginDriverUseCase.loginDriver("0000", "", false)
                    }
                    if ((connectionMode and 0x02) != 0) {
                        _effects.emit(WelcomeUiEffect.ConnectTaximeter)
                        withContext(Dispatchers.IO) { ticketUseCase.loadTickets() }
                        startShiftUseCase.startShift()
                        shiftStatusUseCase.setStatus("STATE_FOR_HIRE_NO_CENTRAL", false)
                        _effects.emit(WelcomeUiEffect.Navigate(WelcomeFragmentDirections.actionWelcomeFragmentToHomeFragment()))
                    }
                    if ((connectionMode and 0x04) != 0) {
                        loginDriverUseCase.loginDriver("0000", "", true)
                    }
                } else {
                    _effects.emit(
                        WelcomeUiEffect.Navigate(
                            WelcomeFragmentDirections.actionWelcomeFragmentToLoginDriverFragment()
                                .setConnectionMode(connectionMode)
                                .setStartTurnMode(startTurnMode)
                        )
                    )
                }
            }
        }
    }

    fun clickShifts() = requestSecurePinIfNeeded(SecurePinDestination.SHIFTS)
    fun clickStatistics() = requestSecurePinIfNeeded(SecurePinDestination.STATISTICS)

    private fun requestSecurePinIfNeeded(destination: SecurePinDestination) {
        viewModelScope.launch {
            val currentSecurePin = userPreferencesUseCase.getSecurePin()
            if (!currentSecurePin.isNullOrEmpty()) {
                _uiState.update { it.copy(dialogState = WelcomeDialogState.SecurePin(destination)) }
            } else {
                navigateToDestination(destination)
            }
        }
    }

    fun verifyCurrentSecurePin(currentPin: String, destination: SecurePinDestination) {
        viewModelScope.launch {
            val actualPin = userPreferencesUseCase.getSecurePin()
            if (currentPin == actualPin) {
                navigateToDestination(destination)
            } else {
                _effects.emit(WelcomeUiEffect.ShowToast(R.string.pin_incorrecto))
            }
        }
    }

    private suspend fun navigateToDestination(destination: SecurePinDestination) {
        when (destination) {
            SecurePinDestination.SHIFTS ->
                _effects.emit(WelcomeUiEffect.Navigate(WelcomeFragmentDirections.actionWelcomeFragmentToShiftsFragment()))
            SecurePinDestination.STATISTICS ->
                _effects.emit(WelcomeUiEffect.Navigate(WelcomeFragmentDirections.actionWelcomeFragmentToStatisticsFragment()))
        }
        _uiState.update { it.copy(dialogState = null) }
    }

    fun clickSettings() = viewModelScope.launch {
        _effects.emit(WelcomeUiEffect.Navigate(WelcomeFragmentDirections.actionWelcomeFragmentToSettingsFragment()))
    }

    fun haveShifts() {
        viewModelScope.launch {
            val shifts = shiftUseCase.getLastShift()
            applyButtonsAction(WelcomeButtonsAction.HasShiftsChanged(shifts != null))
        }
    }

    fun clickConfiguration() = viewModelScope.launch {
        _effects.emit(WelcomeUiEffect.Navigate(WelcomeFragmentDirections.actionWelcomeFragmentToLoginUserFragment()))
    }

    fun checkUserLogIn() {
        viewModelScope.launch(Dispatchers.IO) {
            val loggedIn = sessionUseCase.isUserLoggedIn()
            applyButtonsAction(WelcomeButtonsAction.UserLoggedInChanged(loggedIn))
        }
    }

    fun clickParciales() {
        viewModelScope.launch {
            val activePartial = partialUseCase.getActivePartial()
            if (activePartial != null && activePartial.hasTrips()) {
                _effects.emit(WelcomeUiEffect.Navigate(WelcomeFragmentDirections.actionWelcomeFragmentToPartials()))
            } else {
                _effects.emit(WelcomeUiEffect.Navigate(WelcomeFragmentDirections.actionWelcomeFragmentToClosedPartial()))
            }
        }
    }

    fun clickPermissions() = viewModelScope.launch {
        _effects.emit(WelcomeUiEffect.Navigate(WelcomeFragmentDirections.actionWelcomeFragmentToPermissionsFragment()))
    }

    fun checkClosuresPermission() {
        viewModelScope.launch {
            val parameters = licensingUseCase.getLicensingParameters()
            applyButtonsAction(WelcomeButtonsAction.PartialButtonChanged(parameters?.isClosuresButton))
        }
    }

    fun checkAllPermissions() {
        viewModelScope.launch(Dispatchers.IO) {
            val hasAllPermissions = permissionsUseCase.hasAllPermissions()
            applyButtonsAction(WelcomeButtonsAction.HasAllPermissionsChanged(hasAllPermissions))
        }
    }

    fun hasITopMeterPaired(): Boolean = bluetoothLocalUseCase.isCurrentBluetoothItop()

    fun tryLoginDriver(lastITopCom: ITopMeterStatus?) {
        viewModelScope.launch {
            lastITopCom?.driverCardData?.let {
                if (it.length > 18) {
                    val cardData = it.substring(12, 17)
                    loginDriverUseCase.loginDriver(cardData, cardData, false)
                }
            }
        }
    }

    fun makeMigrationV2() {
        viewModelScope.launch {
            val hasOldTrips = migrationV2UseCase.checkHasOldTrip()
            val hasOldShift = migrationV2UseCase.checkHasOldShift()
            if (hasOldShift || hasOldTrips) {
                val checkExistSession = sessionUseCase.getSession()
                if (checkExistSession == null) {
                    if (migrationV2UseCase.migrateSession()) {
                        _effects.emit(
                            WelcomeUiEffect.Navigate(
                                WelcomeFragmentDirections.actionWelcomeFragmentToLoginUserFragment()
                                    .setAutoDownloadConfigurationFromMigration(true)
                            )
                        )
                    }
                }
            }
        }
    }

    private fun showPrivacyDialogIfNeeded() {
        viewModelScope.launch(Dispatchers.IO) {
            val hasAccepted = (context.getSharedPreferences("default", Context.MODE_PRIVATE)
                .getBoolean("PrivacyPolicyAccepted", false))
            val hasShown = (context.getSharedPreferences("default", Context.MODE_PRIVATE)
                .getBoolean("PrivacyPolicyDialogShowed", false))

            if (!hasAccepted && !hasShown) {
                context.getSharedPreferences("default", Context.MODE_PRIVATE)
                    .edit()
                    .putBoolean("PrivacyPolicyDialogShowed", true)
                    .apply()

                _uiState.update { it.copy(dialogState = WelcomeDialogState.PrivacyPolicy) }
            } else {
                checkBatteryOptimization()
            }
        }
    }

    private fun checkAdditionalPermissions() {
        if (_uiState.value.dialogState == null) {
            checkBatteryOptimization()
        }
    }

    private fun checkBatteryOptimization() {
        val powerManager = context.getSystemService(Context.POWER_SERVICE) as? PowerManager
        val disabled = powerManager?.isIgnoringBatteryOptimizations(context.packageName) == true
        if (!disabled) {
            _uiState.update { it.copy(dialogState = WelcomeDialogState.BatteryOptimization) }
        } else {
            clickMenuLogin()
        }
    }
}

class WelcomeMainActivityViewModel(
    application: Application
) : ViewModel() {

    private val _sharedUiState = MutableStateFlow(WelcomeSharedUiState())
    val sharedUiState = _sharedUiState.asStateFlow()

    private val _effects = MutableSharedFlow<WelcomeSharedUiEffect>()
    val effects = _effects.asSharedFlow()

    fun bind(
        sharedViewModel: MainActivityViewModel,
    ) {
        viewModelScope.launch {
            sharedViewModel.reconnectionFlow.collectLatest { reconnection ->
                _sharedUiState.update { it.copy(reconnectionMode = reconnection) }
            }
        }

        viewModelScope.launch {
            sharedViewModel.downloadingBravoConfigurationFlow.collectLatest { downloading ->
                _sharedUiState.update { it.copy(downloadingBravoConfiguration = downloading) }
                if (downloading) {
                    _effects.emit(WelcomeSharedUiEffect.ShowDownloadingToast)
                }
            }
        }

        viewModelScope.launch {
            sharedViewModel.lastITopStateReceived.collectLatest { state ->
                _sharedUiState.update { it.copy(lastITopState = state) }
            }
        }

        viewModelScope.launch {
            sharedViewModel.correctLoginFlow.collectLatest { result ->
                _sharedUiState.update { it.copy(correctLogin = result) }
                if (result.first) {
                    _effects.emit(WelcomeSharedUiEffect.NavigateHomeAfterLogin)
                }
            }
        }
    }

    fun onTopBarExitPressed(): Boolean {
        return _sharedUiState.value.reconnectionMode
    }
}

@Composable
fun WelcomeRoute(
    navController: NavController,
    viewModel: WelcomeViewModel,
    sharedViewModel: MainActivityViewModel,
    sharedBridgeViewModel: WelcomeMainActivityViewModel,
    onShowHeader: (Boolean) -> Unit,
    onShowToast: (Int) -> Unit,
    onExitApp: () -> Unit,
    onConnectTaximeter: () -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val sharedUiState by sharedBridgeViewModel.sharedUiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    LaunchedEffect(Unit) {
        onShowHeader(true)
        viewModel.onEvent(WelcomeUiEvent.ScreenStarted)
        sharedBridgeViewModel.bind(sharedViewModel)
    }

    LaunchedEffect(viewModel) {
        viewModel.effects.collectLatest { effect ->
            when (effect) {
                is WelcomeUiEffect.Navigate -> navController.navigate(effect.direction)
                WelcomeUiEffect.NavigateBack -> navController.popBackStack()
                WelcomeUiEffect.ExitApp -> onExitApp()
                is WelcomeUiEffect.ShowToast -> onShowToast(effect.messageRes)
                WelcomeUiEffect.ConnectTaximeter -> onConnectTaximeter()
                is WelcomeUiEffect.OpenBatteryOptimizationSettings -> {
                    try {
                        context.startActivity(Intent(Settings.ACTION_IGNORE_BATTERY_OPTIMIZATION_SETTINGS))
                    } catch (_: Exception) {
                    }
                }
                is WelcomeUiEffect.OpenPrivacyPolicy -> {
                    try {
                        context.startActivity(Intent(Intent.ACTION_VIEW, effect.url.toUri()))
                    } catch (_: Exception) {
                        onShowToast(R.string.warning_not_browser)
                    }
                }
            }
        }
    }

    LaunchedEffect(sharedBridgeViewModel) {
        sharedBridgeViewModel.effects.collectLatest { effect ->
            when (effect) {
                WelcomeSharedUiEffect.IgnoreTopBarBackPress -> Unit
                WelcomeSharedUiEffect.NavigateHomeAfterLogin -> {
                    onConnectTaximeter()
                    navController.navigate(R.id.action_welcomeFragment_to_homeFragment)
                }
                WelcomeSharedUiEffect.ShowDownloadingToast -> {
                    onShowToast(R.string.toast_downloading_configurations)
                }
            }
        }
    }

    WelcomeScreen(
        uiState = uiState,
        sharedUiState = sharedUiState,
        onEvent = viewModel::onEvent,
    )
}

@Composable
private fun WelcomeButton(
    text: String,
    state: WelcomeButtonState,
    onClick: () -> Unit,
) {
    if (!state.visible) return

    val containerColor = when (state.background) {
        CustomButton.BackgroundButtonColor.BLUE -> MaterialTheme.colorScheme.primary
        CustomButton.BackgroundButtonColor.ORANGE -> MaterialTheme.colorScheme.tertiary

            contentColor = MaterialTheme.colorScheme.onPrimary,
            disabledContainerColor = containerColor.copy(alpha = 0.5f),
            disabledContentColor = MaterialTheme.colorScheme.onPrimary
        ),
        modifier = Modifier.fillMaxWidth()
    ) {
        if (state.style == CustomButton.StyleButton.LOADING) {
            CircularProgressIndicator(
                modifier = Modifier.height(18.dp),
                strokeWidth = 2.dp
            )
            Spacer(Modifier.height(8.dp))
        }
        Text(text)
    }
}
```

## Notes

This is “compile-oriented” and structurally complete, but there are a few project-specific things you may need to adapt:

1. **`MainActivityViewModel` flow types**
   - I assumed:
     - `reconnectionFlow: StateFlow<Boolean>`
     - `downloadingBravoConfigurationFlow: SharedFlow<Boolean>`
     - `lastITopStateReceived: SharedFlow<ITopMeterStatus?>`
     - `correctLoginFlow: SharedFlow<Pair<Boolean, String?>>`
   - If your actual types differ, adjust the `bind()` signature.

```kotlin
sealed interface WelcomeButtonsAction {
    data class HasShiftsChanged(val hasShifts: Boolean?) : WelcomeButtonsAction
    data class PartialButtonChanged(val enabled: Boolean?) : WelcomeButtonsAction
    data class UserLoggedInChanged(val loggedIn: Boolean) : WelcomeButtonsAction
    data class DownloadingBravoConfigurationChanged(val downloading: Boolean) : WelcomeButtonsAction
    data class HasAllPermissionsChanged(val hasAllPermissions: Boolean) : WelcomeButtonsAction
    data class LastITopStateChanged(val state: String?) : WelcomeButtonsAction
    data class CorrectLoginChanged(val success: Boolean, val lastITopState: String?) : WelcomeButtonsAction
    data class HasITopPairedChanged(val paired: Boolean) : WelcomeButtonsAction
}

object WelcomeButtonsReducer {
    fun reduce(current: WelcomeButtonsState, action: WelcomeButtonsAction): WelcomeButtonsState {
        return when (action) {
            is WelcomeButtonsAction.HasShiftsChanged -> current.copy(
                turnos = current.turnos.copy(
                    enabled = action.hasShifts == true,
                    style = if (action.hasShifts == true) CustomButton.StyleButton.ENABLE else CustomButton.StyleButton.DISABLE
                )
            )

            is WelcomeButtonsAction.PartialButtonChanged -> current.copy(
                parciales = current.parciales.copy(
                    enabled = action.enabled == true,
                    style = when (action.enabled) {
                        true -> CustomButton.StyleButton.ENABLE
                        false -> CustomButton.StyleButton.DISABLE
                        null -> current.parciales.style
                    }
                )
            )

            is WelcomeButtonsAction.UserLoggedInChanged -> current.copy(
                parciales = current.parciales.copy(
                    enabled = action.loggedIn,
                    style = if (action.loggedIn) CustomButton.StyleButton.ENABLE else CustomButton.StyleButton.DISABLE
                )
            )

            is WelcomeButtonsAction.DownloadingBravoConfigurationChanged -> current.copy(
                inicio = current.inicio.copy(
                    enabled = !action.downloading,
                    style = if (action.downloading) CustomButton.StyleButton.LOADING else CustomButton.StyleButton.ENABLE
                )
            )

            is WelcomeButtonsAction.HasAllPermissionsChanged -> current.copy(
                permisos = current.permisos.copy(
                    background = if (action.hasAllPermissions)
                        CustomButton.BackgroundButtonColor.BLUE
                    else
                        CustomButton.BackgroundButtonColor.ORANGE
                )
            )

            is WelcomeButtonsAction.LastITopStateChanged -> {
                val disconnected = action.state == null || action.state == ITopMeterStateEnum.DISCONNECTED.state
                current.copy(
                    inicio = current.inicio.copy(
                        enabled = !disconnected,
                        style = if (disconnected) CustomButton.StyleButton.DISABLE else CustomButton.StyleButton.ENABLE
                    )
                )
            }

            is WelcomeButtonsAction.CorrectLoginChanged -> {
                val disconnected = action.lastITopState == null || action.lastITopState == ITopMeterStateEnum.DISCONNECTED.state
                current.copy(
                    inicio = current.inicio.copy(
                        enabled = if (action.success) current.inicio.enabled else !disconnected,
                        style = if (action.success) current.inicio.style else {
                            if (disconnected) CustomButton.StyleButton.DISABLE else CustomButton.StyleButton.ENABLE
                        }
                    )
                )
            }

            is WelcomeButtonsAction.HasITopPairedChanged -> current.copy(
                inicio = current.inicio.copy(
                    enabled = action.paired,
                    style = if (action.paired) current.inicio.style else CustomButton.StyleButton.ENABLE
                )
            )
        }
    }
}

@Composable
fun WelcomeRoute(
    navController: NavController,
    viewModel: WelcomeViewModel,
    sharedViewModel: MainActivityViewModel,
    composeSharedViewModel: ComposeSharedViewModel,
    sharedBridgeViewModel: WelcomeMainActivityViewModel,
    onShowHeader: (Boolean) -> Unit,
    onShowToast: (Int) -> Unit,
    onExitApp: () -> Unit,
    onConnectTaximeter: () -> Unit,
    onNavigateBack: () -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val sharedUiState by sharedBridgeViewModel.sharedUiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    LaunchedEffect(Unit) {
        onShowHeader(true)
        viewModel.onEvent(WelcomeUiEvent.ScreenStarted)
        sharedBridgeViewModel.bind(sharedViewModel, composeSharedViewModel)
    }

    LaunchedEffect(viewModel) {
        viewModel.effects.collectLatest { effect ->
            when (effect) {
                is WelcomeUiEffect.Navigate -> navController.navigate(effect.direction)
                WelcomeUiEffect.NavigateBack -> onNavigateBack()
                WelcomeUiEffect.ExitApp -> onExitApp()
                is WelcomeUiEffect.ShowToast -> onShowToast(effect.messageRes)
                WelcomeUiEffect.ConnectTaximeter -> onConnectTaximeter()
                is WelcomeUiEffect.OpenBatteryOptimizationSettings -> {
                    try {
                        context.startActivity(Intent(Settings.ACTION_IGNORE_BATTERY_OPTIMIZATION_SETTINGS))
                    } catch (_: Exception) {
                    }
                }
                is WelcomeUiEffect.OpenPrivacyPolicy -> {
                    try {
                        context.startActivity(Intent(Intent.ACTION_VIEW, effect.url.toUri()))
                    } catch (_: Exception) {
                        onShowToast(R.string.warning_not_browser)
                    }
                }
            }
        }
    }

    LaunchedEffect(sharedBridgeViewModel) {
        sharedBridgeViewModel.effects.collectLatest { effect ->
            when (effect) {
                WelcomeSharedUiEffect.IgnoreTopBarBackPress -> Unit
                WelcomeSharedUiEffect.NavigateHomeAfterLogin -> {
                    onConnectTaximeter()
                    navController.navigate(WelcomeFragmentDirections.actionWelcomeFragmentToHomeFragment())
                }
                WelcomeSharedUiEffect.ShowDownloadingToast -> {
                    onShowToast(R.string.toast_downloading_configurations)
                }
            }
        }
    }

    LaunchedEffect(composeSharedViewModel) {
        composeSharedViewModel.navigationDirectionsFlow.collectLatest { direction ->
            navController.navigate(direction)
        }
    }

    WelcomeScreen(
        uiState = uiState,
        sharedUiState = sharedUiState,
        onEvent = viewModel::onEvent,
    )
}

@Composable
private fun WelcomeScreen(
    uiState: WelcomeUiState,
    sharedUiState: WelcomeSharedUiState,
    onEvent: (WelcomeUiEvent) -> Unit,
) {
    Column(
        modifier = Modifier
            .fillMaxSize()
            .padding(16.dp),
        verticalArrangement = Arrangement.spacedBy(12.dp)
    ) {
        WelcomeButton(
            text = "Inicio",
            state = uiState.buttons.inicio,
            onClick = { onEvent(WelcomeUiEvent.InicioClicked) }
        )
        WelcomeButton(
            text = "Ajustes",
            state = uiState.buttons.ajustes,
            onClick = { onEvent(WelcomeUiEvent.AjustesClicked) }
        )
        WelcomeButton(
            text = "Turnos",
            state = uiState.buttons.turnos,
            onClick = { onEvent(WelcomeUiEvent.TurnosClicked) }
        )
        WelcomeButton(
            text = "Estadísticas",
            state = uiState.buttons.estadisticas,
            onClick = { onEvent(WelcomeUiEvent.EstadisticasClicked) }
        )
        if (uiState.buttons.permisos.visible) {
            WelcomeButton(
                text = "Permisos",
                state = uiState.buttons.permisos,
                onClick = { onEvent(WelcomeUiEvent.PermisosClicked) }
            )
        }
        WelcomeButton(
            text = "Configuración",
            state = uiState.buttons.configuracion,
            onClick = { onEvent(WelcomeUiEvent.ConfiguracionClicked) }
        )
        WelcomeButton(
            text = "Parciales",
            state = uiState.buttons.parciales,
            onClick = { onEvent(WelcomeUiEvent.ParcialesClicked) }
        )
        WelcomeButton(
            text = "Salir",
            state = uiState.buttons.salir,
            onClick = { onEvent(WelcomeUiEvent.SalirClicked) }
        )
    }

    when (val dialog = uiState.dialogState) {
        is WelcomeDialogState.SecurePin -> SecurePinDialog(
            title = "PIN",
            hint = "PIN actual",
            onDismiss = { onEvent(WelcomeUiEvent.DismissDialog) },
            onAccept = { pin -> onEvent(WelcomeUiEvent.SubmitSecurePin(pin, dialog.destination)) }
        )

        WelcomeDialogState.ExitConfirm -> ConfirmDialog(
            title = "Cerrar app",
            onDismiss = { onEvent(WelcomeUiEvent.DismissDialog) },
            onAccept = { onEvent(WelcomeUiEvent.ConfirmExit) }
        )

        WelcomeDialogState.PrivacyPolicy -> PrivacyDialog(
            title = "Política de privacidad",
            description = "Debes aceptar la política de privacidad.",
            hrefText = "Ver política",
            acceptText = "Aceptar",
            onOpenLink = { onEvent(WelcomeUiEvent.OpenPrivacyLink) },
            onAccept = { onEvent(WelcomeUiEvent.AcceptPrivacyPolicy) },
            onDismiss = { onEvent(WelcomeUiEvent.DismissDialog) }
        )

        WelcomeDialogState.BatteryOptimization -> ConfirmDialog(
            title = "Optimización de batería",
            description = "Desactiva la optimización para continuar.",
            onDismiss = { onEvent(WelcomeUiEvent.DismissDialog) },
            onAccept = { onEvent(WelcomeUiEvent.AcceptBatteryOptimization) }
        )

        null -> Unit
    }
}

@Composable
private fun SecurePinDialog(
    title: String,
    hint: String,
    onDismiss: () -> Unit,
    onAccept: (String) -> Unit,
) {
    var pin by remember { mutableStateOf("") }

    AlertDialog(
        onDismissRequest = onDismiss,
        title = { Text(title) },
        text = {
            OutlinedTextField(
                value = pin,
                onValueChange = { pin = it },
                label = { Text(hint) }
            )
        },
        confirmButton = { TextButton(onClick = { onAccept(pin) }) { Text("OK") } },
        dismissButton = { TextButton(onClick = onDismiss) { Text("Cancelar") } }
    )
}

@Composable
private fun ConfirmDialog(
    title: String,
    description: String? = null,
    onDismiss: () -> Unit,
    onAccept: () -> Unit,
) {
    AlertDialog(
        onDismissRequest = onDismiss,
        title = { Text(title) },
        text = { if (description != null) Text(description) },
        confirmButton = { TextButton(onClick = onAccept) { Text("OK") } },
        dismissButton = { TextButton(onClick = onDismiss) { Text("Cancelar") } }
    )
}

@Composable
private fun PrivacyDialog(
    title: String,
    description: String,
    hrefText: String,
    acceptText: String,
    onOpenLink: () -> Unit,
    onAccept: () -> Unit,
    onDismiss: () -> Unit,
) {
    AlertDialog(
        onDismissRequest = onDismiss,
        title = { Text(title) },
        text = {
            Column {
                Text(description)
                TextButton(onClick = onOpenLink) { Text(hrefText) }
            }
        },
        confirmButton = { TextButton(onClick = onAccept) { Text(acceptText) } },
        dismissButton = { TextButton(onClick = onDismiss) { Text("Cancelar") } }
    )
}

@Composable
private fun WelcomeButton(
    text: String,
    state: WelcomeButtonState,
    onClick: () -> Unit,
) {
    if (!state.visible) return

    val containerColor = when (state.background) {
        CustomButton.BackgroundButtonColor.BLUE -> MaterialTheme.colorScheme.primary
        CustomButton.BackgroundButtonColor.ORANGE -> MaterialTheme.colorScheme.tertiary
    }

    Button(
        onClick = onClick,
        enabled = state.enabled && state.style != CustomButton.StyleButton.LOADING,
        colors = ButtonDefaults.buttonColors(
            containerColor = containerColor,
            contentColor = MaterialTheme.colorScheme.onPrimary,
            disabledContainerColor = containerColor.copy(alpha = 0.5f),
            disabledContentColor = MaterialTheme.colorScheme.onPrimary
        ),
        modifier = Modifier.fillMaxWidth()
    ) {
        if (state.style == CustomButton.StyleButton.LOADING) {
            CircularProgressIndicator(
                modifier = Modifier.height(18.dp),
                strokeWidth = 2.dp
            )
            Spacer(Modifier.height(8.dp))
        }
        Text(text)
    }
}
```

```kotlin
package ifac.td.taxi.ui.screen

import androidx.compose.foundation.background
import androidx.compose.foundation.layout.*

import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color

import androidx.compose.ui.unit.dp
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import com.github.gcacace.signaturepad.views.SignaturePad
import ifac.td.taxi.R
import ifac.td.taxi.domain.utils.StaticConfiguration
import ifac.td.taxi.ui.custom.button.CustomButton
import ifac.td.taxi.viewmodel.SignatureMainActivityViewModel
import ifac.td.taxi.viewmodel.SignatureViewModel
import ifac.td.taxi.viewmodel.model.SignatureUiEffect
import kotlinx.coroutines.flow.collectLatest

@Composable
fun SignatureScreen(
    viewModel: SignatureViewModel,
    sharedViewModel: SignatureMainActivityViewModel,
    serviceId: String?,
    onShowToast: (Int) -> Unit,
    modifier: Modifier = Modifier,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val sharedState by sharedViewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    var signaturePadBitmap by remember { mutableStateOf<Bitmap?>(null) }
    var signaturePadRef by remember { mutableStateOf<SignaturePad?>(null) }

    LaunchedEffect(Unit) {
        viewModel.effects.collectLatest { effect ->
            when (effect) {
                is SignatureUiEffect.ShowToast -> onShowToast(effect.messageResId)

                is SignatureUiEffect.HandleSignatureResult -> {
                    if (effect.success) {
                        if (serviceId == sharedState.dispatchFlow?.longDispatchNumber) {
                            sharedState.dispatchFlow?.let { dispatch ->
                                sharedState.tripFlow?.let { trip ->
                                    viewModel.dispatchSubscriberNextStep(dispatch, trip)
                                }
                            }
                        } else {
                            viewModel.sendSubscriberAuth(serviceId, sharedState.tripFlow?.id)
                        }
                    } else {
                        StaticConfiguration.subscriberFailPin = true
                    }

                    onShowToast(
                        if (effect.success) R.string.dialog_signature_valid_desc
                        else R.string.dialog_signature_invalid_desc
                    )
                }

                SignatureUiEffect.None -> Unit
            }
        }
    }

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(Color.White)
    ) {
        androidx.compose.ui.viewinterop.AndroidView(
            modifier = Modifier
                .weight(1f)
                .fillMaxWidth(),
            factory = { ctx ->
                SignaturePad(ctx).apply {
                    setBackgroundResource(R.drawable.bg_signature)
                    penColor = android.graphics.Color.BLACK
                    setOnSignedListener(object : SignaturePad.OnSignedListener {
                        override fun onStartSigning() = Unit
                        override fun onSigned() {
                            viewModel.onSignedChanged(true)
                        }
                        override fun onClear() {
                            viewModel.onSignedChanged(false)
                        }
                    })
                    signaturePadRef = this
                }
            },
            update = { pad ->
                signaturePadBitmap = pad.signatureBitmap
            }
        )

        Row(
            modifier = Modifier
                .fillMaxWidth()
                .height(120.dp)
                .background(Color.Black)
                .padding(1.dp),
            horizontalArrangement = Arrangement.spacedBy(6.dp)
        ) {
            SignatureCustomButton(
                modifier = Modifier.weight(1f),
                buttonState = uiState.clearButton,
                onClick = {
                    signaturePadRef?.clear()
                    viewModel.onClearClicked()
                }
            )

            SignatureCustomButton(
                modifier = Modifier.weight(1f),
                buttonState = uiState.acceptButton,
                onClick = {
                    if (uiState.hasSigned) {
                        viewModel.onAcceptClicked(
                            signatureBitmap = signaturePadBitmap,
                            serviceId = serviceId,
                            fromDispatch = sharedState.tripFlow?.fromDispatch == true &&
                                sharedState.dispatchFlow?.requireSignature == true
                        )
                    } else {
                        onShowToast(R.string.toast_sign_first)
                    }
                }
            )
        }
    }
}

@Composable
private fun SignatureCustomButton(
    modifier: Modifier = Modifier,
    buttonState: SignatureButtonsState,
    onClick: () -> Unit,
) {
    // This assumes your existing CustomButton is still a View-based custom view.
    // If you already have a Compose wrapper, use that instead.
    androidx.compose.ui.viewinterop.AndroidView(
        modifier = modifier.fillMaxHeight(),
        factory = { ctx ->
            CustomButton(ctx).apply {
                setButtonType(buttonState.buttonTypeValue)
                setButtonStyle(buttonState.style.toLegacyStyle())
                changeBackground(buttonState.backgroundColor.toLegacyBackground())
                changeText(buttonState.text)
                setAction(onClick)
            }
        },
        update = { button ->
            button.setButtonType(buttonState.buttonTypeValue)
            button.setButtonStyle(buttonState.style.toLegacyStyle())
            button.changeBackground(buttonState.backgroundColor.toLegacyBackground())
            button.changeText(buttonState.text)
            button.setAction(onClick)
        }
    )
}

private fun CustomButton.StyleButton.toLegacyStyle(): CustomButton.StyleButton = this

private fun CustomButton.BackgroundButtonColor.toLegacyBackground(): CustomButton.BackgroundButtonColor = this
```

```kotlin
class SignatureViewModel(
    private val signatureUseCase: SignatureUseCase,
    private val subscriberUseCase: SubscriberUseCase,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val tripUseCase: TripUseCase,
    private val ticketUseCase: TicketUseCase,
    context: Application,
) : BaseViewModel(context) {

    private val _uiState = MutableStateFlow(SignatureUiState())
    val uiState: StateFlow<SignatureUiState> = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<SignatureUiEffect>()
    val effects = _effects.asSharedFlow()

    private var serviceId: String? = null

    fun onSignedChanged(signed: Boolean) {
        _uiState.update { it.copy(hasSigned = signed) }
    }

    fun onClearClicked() {
        _uiState.update { it.copy(hasSigned = false) }
    }

    fun onAcceptClicked(
        signatureBitmap: Bitmap?,
        serviceId: String?,
        fromDispatch: Boolean,
    ) {
        if (!_uiState.value.hasSigned) {
            viewModelScope.launch {
                _effects.emit(SignatureUiEffect.ShowToast(R.string.toast_sign_first))
            }
            return
        }

        this.serviceId = serviceId
        _uiState.update {
            it.copy(
                buttons = it.buttons.copy(
                    accept = it.buttons.accept.copy(style = CustomButtonStyle.LOADING)
                )
            )
        }

        viewModelScope.launch {
            signatureUseCase.sendSignature(
                signatureBitmap = signatureBitmap,
                dispatchId = serviceId,
                responseFlow = MutableSharedFlow<Boolean>().also { flow ->
                    viewModelScope.launch {
                        flow.collect { success ->
                            _effects.emit(SignatureUiEffect.HandleSignatureResult(success))
                        }
                    }
                },
                fromDispatch = fromDispatch
            )
        }
    }

    fun onSignatureResult(success: Boolean, dispatchFlow: InfoDispatchModel?, trip: Trip?) {
        viewModelScope.launch {
            _effects.emit(SignatureUiEffect.HandleSignatureResult(success))
        }
    }

    fun sendSubscriberAuth(serviceId: String?, tripId: Long?) {
        viewModelScope.launch(Dispatchers.IO) {
            val accountPaymentRequest = subscriberUseCase.getAccountPaymentRequest(tripId)
            accountPaymentRequest?.serviceId = serviceId
            subscriberUseCase.sendAccountPaymentRequest(accountPaymentRequest)
        }
    }

    fun dispatchSubscriberNextStep(dispatchValue: InfoDispatchModel, trip: Trip) {
        viewModelScope.launch {
            if (dispatchValue.requireVoucher) {
                navigateTo(PaymentDirections.goToCropImageViewFragment(dispatchValue.longDispatchNumber))
            } else if (dispatchValue.requireQr) {
                navigateTo(PaymentDirections.goToScannerQRFragment(dispatchValue.longDispatchNumber))
            } else {
                tripUseCase.endTrip(trip, PaymentMethod.SUBSCRIBER)

                shiftStatusUseCase.getStatus()?.let { status ->
                    ticketUseCase.prepareTicket(
                        trip = trip,
                        dispatch = dispatchValue,
                        shiftStatus = status,
                    )
                }

                val status = shiftStatusUseCase.checkForSubStateModifications(ifConstants.STATE_FOR_HIRE)
                status?.let { shiftStatusUseCase.setStatus(it, true) }
            }
        }
    }
}
```

## 3) Exact `SignatureButtonsState` reducer mirroring the fragment logic

This reducer is intentionally one-to-one with the fragment branches:

```kotlin
package ifac.td.taxi.viewmodel

import ifac.td.taxi.R

fun SignatureButtonsState.reduce(
    hasSigned: Boolean,
    acceptLoading: Boolean,
): SignatureButtonsState {
    return copy(
        accept = accept.copy(
            style = when {
                acceptLoading -> CustomButtonStyle.LOADING
                else -> CustomButtonStyle.ENABLE
            },
            backgroundColor = CustomButtonBackground.GREEN,
            visible = true,
            text = "ACCEPT",
            buttonTypeValue = 1
        ),
        clear = clear.copy(
            style = CustomButtonStyle.ENABLE,
            backgroundColor = CustomButtonBackground.RED,
            visible = true,
            text = "CLEAR",
            buttonTypeValue = 2
        )
    )
}
```

### Exact branch mapping from the fragment
- `btnAccept.setAction { ... }` → `onAcceptClicked(...)`
- `if (vModel.hasSigned())` → `uiState.hasSigned`
- `btnAccept.setButtonStyle(LOADING)` → `accept.style = LOADING`
- `btnClear.setAction { clearSignature(); signedFlag(false) }` → `onClearClicked()`
- `signatureResponseFlow.collect { value -> ... }` → `SignatureUiEffect.HandleSignatureResult(success)`
- `if (value)` → success branch in effect handler
- `else` → `StaticConfiguration.subscriberFailPin = true`
- toast selection → `ShowToast(...)`

Your original fragment uses:

```kotlin

class SignatureViewModel(
    private val signatureUseCase: SignatureUseCase,
    private val subscriberUseCase: SubscriberUseCase,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val tripUseCase: TripUseCase,
    private val ticketUseCase: TicketUseCase,
    context: Application,
) : BaseViewModel(context) {

    private val TAG = "SignatureViewModel"

    private val _uiState = MutableStateFlow(SignatureUiState())
    val uiState: StateFlow<SignatureUiState> = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<SignatureUiEffect>(extraBufferCapacity = 1)
    val effects: SharedFlow<SignatureUiEffect> = _effects.asSharedFlow()

    private var hasSigned = false

    fun hasSigned(): Boolean {
        Logs.d(TAG, "hasSigned: $hasSigned")
        return hasSigned
    }

    fun signedFlag(flag: Boolean) {
        Logs.d(TAG, "signedFlag: Setting hasSigned to $flag")
        hasSigned = flag
        _uiState.update { current ->
            current.copy(
                hasSigned = flag,
                buttons = current.buttons.reduce(
                    hasSigned = flag,
                    acceptLoading = current.buttons.accept.style == SignatureButtonStyle.LOADING
                )
            )
        }
    }

    fun onSignaturePadSigned() {
        Logs.d(TAG, "onSignaturePadSigned: Signature detected")
        signedFlag(true)
    }

    fun onSignaturePadCleared() {
        Logs.d(TAG, "onSignaturePadCleared: Signature pad cleared")
        signedFlag(false)
    }

    fun onAcceptClicked(
        signatureBitmap: Bitmap?,
        dispatchId: String?,
        fromDispatch: Boolean,
    ) {
        if (!hasSigned()) {
            viewModelScope.launch {
                _effects.emit(SignatureUiEffect.ShowToast(R.string.toast_sign_first))
            }
            return
        }

        _uiState.update { current ->
            current.copy(
                buttons = current.buttons.copy(
                    accept = current.buttons.accept.copy(style = SignatureButtonStyle.LOADING)
                )
            )
        }

        viewModelScope.launch {
            Logs.d(TAG, "sendSignature: Sending signature for dispatchId: $dispatchId, fromDispatch: $fromDispatch")
            signatureUseCase.sendSignature(
                signatureBitmap = signatureBitmap,
                dispatchId = dispatchId,
                responseFlow = signatureResponseFlowBridge(),
                fromDispatch = fromDispatch
            )
            Logs.d(TAG, "sendSignature: Signature sent")
        }
    }

    private fun signatureResponseFlowBridge(): MutableSharedFlow<Boolean> {
        val bridge = MutableSharedFlow<Boolean>(extraBufferCapacity = 1)
        viewModelScope.launch {
            bridge.collect { value ->
                _effects.emit(SignatureUiEffect.SignatureValidated(value))
            }
        }
        return bridge
    }

    fun sendSubscriberAuth(serviceId: String?, tripId: Long?) {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "sendSubscriberAuth: Starting subscriber authentication for serviceId: $serviceId")
            val accountPaymentRequest = subscriberUseCase.getAccountPaymentRequest(tripId)
            Logs.d(TAG, "sendSubscriberAuth: Retrieved AccountPaymentRequest: $accountPaymentRequest")
            accountPaymentRequest?.serviceId = serviceId
            subscriberUseCase.sendAccountPaymentRequest(accountPaymentRequest)
            Logs.d(TAG, "sendSubscriberAuth: Account payment request sent")
        }
    }

    fun dispatchSubscriberNextStep(dispatchValue: InfoDispatchModel, trip: Trip) {
        viewModelScope.launch {
            Logs.d(
                TAG,
                "dispatchSubscriberNextStep: Processing next step for dispatch: ${dispatchValue.longDispatchNumber}, tripId: ${trip.id}"
            )

            if (dispatchValue.requireVoucher) {
                Logs.d(TAG, "dispatchSubscriberNextStep: Dispatch requires voucher, navigating to CropImageView")
                dispatchRequireVoucher(dispatchValue)
            } else if (dispatchValue.requireQr) {
                Logs.d(TAG, "dispatchSubscriberNextStep: Dispatch requires QR, navigating to ScannerQRFragment")
                dispatchRequireQR(dispatchValue)
            } else {
                Logs.d(TAG, "dispatchSubscriberNextStep: Ending trip with SUBSCRIBER payment method")
                tripUseCase.endTrip(trip, PaymentMethod.SUBSCRIBER)

                shiftStatusUseCase.getStatus()?.let { status ->
                    Logs.d(TAG, "dispatchSubscriberNextStep: Preparing ticket...")
                    ticketUseCase.prepareTicket(
                        trip = trip,
                        dispatch = dispatchValue,
                        shiftStatus = status,
                    )
                    Logs.d(TAG, "dispatchSubscriberNextStep: Ticket prepared")
                }

                val status = shiftStatusUseCase.checkForSubStateModifications(ifConstants.STATE_FOR_HIRE)
                status?.let { statusWithSubStatus ->
                    Logs.d(TAG, "dispatchSubscriberNextStep: Setting shift status to: $statusWithSubStatus")
                    shiftStatusUseCase.setStatus(statusWithSubStatus, true)
                }
            }
        }
    }

    private suspend fun dispatchRequireVoucher(dispatch: InfoDispatchModel) {
        Logs.d(TAG, "dispatchRequireVoucher: Navigating to CropImageViewFragment for dispatch: ${dispatch.longDispatchNumber}")
        navigateTo(PaymentDirections.goToCropImageViewFragment(dispatch.longDispatchNumber))
    }

    private suspend fun dispatchRequireQR(dispatch: InfoDispatchModel) {
        Logs.d(TAG, "dispatchRequireQR: Navigating to ScannerQRFragment for dispatch: ${dispatch.longDispatchNumber}")
        navigateTo(PaymentDirections.goToScannerQRFragment(dispatch.longDispatchNumber))
    }
}

        _uiState.update { it.copy(tripFlow = trip) }

@Composable
fun SignatureScreen(
    viewModel: SignatureViewModel,
    sharedViewModel: SignatureMainActivityViewModel,
    serviceId: String?,
    onShowToast: (Int) -> Unit,
    modifier: Modifier = Modifier,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val sharedState by sharedViewModel.uiState.collectAsStateWithLifecycle()

    var signaturePadRef by remember { mutableStateOf<SignaturePad?>(null) }
    var signatureBitmap by remember { mutableStateOf<Bitmap?>(null) }

    LaunchedEffect(Unit) {
        viewModel.effects.collect { effect ->
            when (effect) {
                is SignatureUiEffect.ShowToast -> onShowToast(effect.messageResId)

                is SignatureUiEffect.SignatureValidated -> {
                    val value = effect.success
                    Logs.d("SignatureScreen", "signatureResponseFlow: Received value: $value")

                    if (value) {
                        Logs.d("SignatureScreen", "signatureResponseFlow: Signature validation successful")
                        if (serviceId == sharedState.dispatchFlow?.longDispatchNumber) {
                            Logs.d("SignatureScreen", "signatureResponseFlow: Dispatching subscriber for serviceId: $serviceId")
                            sharedState.dispatchFlow?.let { dispatch ->
                                sharedState.tripFlow?.let { trip ->
                                    viewModel.dispatchSubscriberNextStep(dispatch, trip)
                                }
                            }
                        } else {
                            Logs.d("SignatureScreen", "signatureResponseFlow: Sending subscriber auth for serviceId: $serviceId")
                            viewModel.sendSubscriberAuth(serviceId, sharedState.tripFlow?.id)
                        }
                    } else {
                        Logs.d("SignatureScreen", "signatureResponseFlow: Signature validation failed, setting subscriberFailPin to true")
                        StaticConfiguration.subscriberFailPin = true
                    }

                    val descriptionResId =
                        if (value) R.string.dialog_signature_valid_desc
                        else R.string.dialog_signature_invalid_desc

                    Logs.d("SignatureScreen", "signatureResponseFlow: Showing toast with message: $descriptionResId")
                    onShowToast(descriptionResId)
                }

                SignatureUiEffect.None -> Unit
            }
        }
    }

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(Color.White)
    ) {
        AndroidView(
            modifier = Modifier
                .weight(1f)
                .fillMaxWidth(),
            factory = { context ->
                SignaturePad(context).apply {
                    setBackgroundResource(R.drawable.bg_signature)
                    penColor = android.graphics.Color.BLACK
                    setOnSignedListener(object : SignaturePad.OnSignedListener {
                        override fun onStartSigning() {
                            Logs.d("SignatureScreen", "setupSignatureView: User started signing")
                        }

                        override fun onSigned() {
                            Logs.d("SignatureScreen", "setupSignatureView: Signature detected")
                            viewModel.onSignaturePadSigned()
                        }

                        override fun onClear() {
                            Logs.d("SignatureScreen", "setupSignatureView: Signature pad cleared")
                            viewModel.onSignaturePadCleared()
                        }
                    })
                    signaturePadRef = this
                }
            },
            update = { pad ->
                signatureBitmap = pad.signatureBitmap
            }
        )

        Row(
            modifier = Modifier
                .fillMaxWidth()
                .height(120.dp)
                .background(Color.Black)
                .padding(1.dp),
            horizontalArrangement = Arrangement.spacedBy(6.dp)
        ) {
            SignatureButtonView(
                modifier = Modifier.weight(1f),
                state = uiState.buttons.clear,
                onClick = {
                    Logs.d("SignatureScreen", "clearSignature: Clearing signature pad")
                    signaturePadRef?.clear()
                    viewModel.onSignaturePadCleared()
                }
            )

            SignatureButtonView(
                modifier = Modifier.weight(1f),
                state = uiState.buttons.accept,
                onClick = {
                    if (viewModel.hasSigned()) {
                        Logs.d("SignatureScreen", "setButtons: serviceId set to $serviceId")
                        val fromDispatch =
                            (sharedState.tripFlow?.fromDispatch == true &&
                                    sharedState.dispatchFlow?.requireSignature == true)

                        Logs.d("SignatureScreen", "setButtons: fromDispatch = $fromDispatch")

                        viewModel.onAcceptClicked(
                            signatureBitmap = signatureBitmap,
                            dispatchId = serviceId,
                            fromDispatch = fromDispatch
                        )
                    } else {
                        Logs.d("SignatureScreen", "setButtons: No signature detected, showing toast")
                        onShowToast(R.string.toast_sign_first)
                    }
                }
            )
        }
    }
}
```

    small = RoundedCornerShape(8.dp),
    medium = RoundedCornerShape(12.dp),
    large = RoundedCornerShape(16.dp)

@Composable
fun SmartTDTheme(
    darkTheme: Boolean = isSystemInDarkTheme(),
    content: @Composable () -> Unit
) {
    val colorScheme = if (darkTheme) DarkColorScheme else DarkColorScheme
    val view = LocalView.current

    if (!view.isInEditMode) {
        SideEffect {
            val window = (view.context as Activity).window
            window.statusBarColor = BlueTrans.toArgb()
            WindowCompat.getInsetsController(window, view).isAppearanceLightStatusBars = false
        }
    }

    MaterialTheme(
        colorScheme = colorScheme,
        typography = AppTypography,
        shapes = AppShapes,
        content = content
    )
}

If you want a bottom sheet theme equivalent, Compose Material3 uses `ModalBottomSheet`. You can approximate your XML `bottomSheetDialogTheme` like this:

```kotlin
@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun SmartTDModalBottomSheet(
    onDismissRequest: () -> Unit,
    content: @Composable () -> Unit
) {
    androidx.compose.material3.ModalBottomSheet(
        onDismissRequest = onDismissRequest,
        containerColor = DarkBackground,
        contentColor = Color.White,
        dragHandle = {
            BottomSheetDefaults.DragHandle()
        }
    ) {
        content()
    }
}
```

Notes:
- `colorPrimaryVariant` and `colorSecondaryVariant` do not exist in Material3, so `primaryContainer` and `secondaryContainer` are the closest equivalents.
- `android:statusBarColor` is handled in Compose by setting the window status bar color in `SideEffect`.
- `fontFamily="@font/montserrat_medium"` should be replaced with your actual Compose font setup using `FontFamily(Font(R.font.montserrat_medium))` if you have that font resource.
- `windowActionBar=false` and `windowNoTitle=true` are typically handled by using a `NoActionBar` activity theme in XML, but in Compose you generally just don't use an action bar.

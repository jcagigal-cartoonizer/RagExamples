### `SplashScreenViewModel.kt`
```kotlin
package ifac.td.taxi.ui.screen

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.domain.usecase.SessionUseCase
import ifac.td.taxi.framework.util.Logs
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.delay
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch

class SplashScreenViewModel(
    application: Application,
    private val sessionUseCase: SessionUseCase
) : AndroidViewModel(application) {

    private val TAG = "SplashScreenViewModel"

    private val _uiState = MutableStateFlow(SplashScreenUiState())
    val uiState: StateFlow<SplashScreenUiState> = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<SplashScreenUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()

    val buttonsState: SplashScreenButtonsState = SplashScreenButtonsState()

    fun onEvent(event: SplashScreenUiEvent) {
        when (event) {
            SplashScreenUiEvent.OnScreenStarted -> checkIsReconnecting()
            SplashScreenUiEvent.OnVideoCompleted -> emitEffect(SplashScreenUiEffect.HideSplashVideo)
            SplashScreenUiEvent.OnVideoError -> emitEffect(SplashScreenUiEffect.HideSplashVideo)
            SplashScreenUiEvent.OnSplashTimeout -> emitEffect(SplashScreenUiEffect.HideSplashVideo)
            SplashScreenUiEvent.OnBackPressed -> Unit
            SplashScreenUiEvent.OnSkipSplash -> {
                _uiState.value = _uiState.value.copy(isSkipped = true, isVideoVisible = false)
            }
        }
    }

    private fun checkIsReconnecting() {
        viewModelScope.launch(Dispatchers.IO) {
            val serviceWasActive = sessionUseCase.getServiceWasActive()
            Logs.d(TAG, "checkIsReconnecting serviceWasActive: $serviceWasActive")
            delay(50)
            _uiState.value = _uiState.value.copy(shouldSkipSplash = serviceWasActive)
        }
    }

    fun onSplashFinished(showLegal: Boolean) {
        viewModelScope.launch {
            emitEffect(
                if (showLegal) SplashScreenUiEffect.NavigateToLegalText
                else SplashScreenUiEffect.NavigateToWelcome
            )
        }
    }

    fun onSkipSplashAndNavigateToWelcome() {
        viewModelScope.launch {
            _uiState.value = _uiState.value.copy(isSkipped = true, isVideoVisible = false)
            emitEffect(SplashScreenUiEffect.NavigateToWelcome)
        }
    }

    private suspend fun emitEffect(effect: SplashScreenUiEffect) {
        _uiEffect.emit(effect)
    }
}
```

## 4) Compose splash screen

This screen mirrors the fragment behavior:

- hides header/bottom bar via callbacks

- plays a video-like splash area

### `SplashScreenRoute.kt`
```kotlin
package ifac.td.taxi.ui.screen

import android.graphics.Color
import android.net.Uri
import android.os.Handler
import android.os.Looper
import android.view.View
import android.widget.VideoView
import androidx.compose.runtime.Composable
import androidx.compose.runtime.DisposableEffect
import androidx.compose.runtime.LaunchedEffect
import androidx.compose.runtime.getValue
import androidx.compose.runtime.remember
import androidx.compose.ui.Modifier
import androidx.compose.ui.platform.LocalContext
import androidx.compose.ui.viewinterop.AndroidView
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.navigation.NavController
import ifac.td.taxi.R
import ifac.td.taxi.framework.util.Logs
import kotlinx.coroutines.flow.collectLatest

@Composable
fun SplashScreenRoute(
    navController: NavController,
    viewModel: SplashScreenViewModel,
    showHeader: (Boolean) -> Unit,
    showBottomBar: (Boolean) -> Unit,
    getBravoService: () -> Any?,
    showLegalTextWindow: (Boolean) -> Unit
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current
    val splashTimeout = 4800L
    val handler = remember { Handler(Looper.getMainLooper()) }

    LaunchedEffect(Unit) {
        showHeader(false)
        showBottomBar(false)
        viewModel.onEvent(SplashScreenUiEvent.OnScreenStarted)
    }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                SplashScreenUiEffect.HideSplashVideo -> {
                    showLegalTextWindow(false)
                }
                SplashScreenUiEffect.NavigateToWelcome -> {
                    navController.navigate(R.id.action_splashScreenFragment_to_welcomeFragment) {
                        launchSingleTop = true
                        popUpTo(R.id.splashScreenFragment) { inclusive = true }
                    }
                }
                SplashScreenUiEffect.NavigateToLegalText -> {
                    navController.navigate(R.id.action_splashScreenFragment_to_legalTextFragment) {
                        launchSingleTop = true
                        popUpTo(R.id.splashScreenFragment) { inclusive = true }
                    }
                }
            }
        }
    }

    LaunchedEffect(uiState.shouldSkipSplash) {
        if (uiState.shouldSkipSplash && getBravoService() == null) {
            viewModel.onSkipSplashAndNavigateToWelcome()
            showLegalTextWindow(true)
        }
    }

    DisposableEffect(Unit) {
        onDispose {
            handler.removeCallbacksAndMessages(null)
        }
    }

    AndroidView(
        factory = {
            VideoView(it).apply {
                setZOrderOnTop(true)
                setBackgroundColor(it.getColor(R.color.splash_background, null))

                setOnPreparedListener { mediaPlayer ->
                    Logs.d("SplashScreenRoute", "Video prepared, starting")
                    setBackgroundColor(Color.TRANSPARENT)
                    mediaPlayer.isLooping = false
                    mediaPlayer.start()
                }

                setOnCompletionListener {
                    Logs.d("SplashScreenRoute", "Video ended")
                    viewModel.onEvent(SplashScreenUiEvent.OnVideoCompleted)
                }

                setOnErrorListener { _, what, extra ->
                    Logs.e("SplashScreenRoute", "Error during video playback: what=$what, extra=$extra")
                    viewModel.onEvent(SplashScreenUiEvent.OnVideoError)
                    true
                }

                try {
                    val videoUri = Uri.parse("android.resource://${context.packageName}/${R.raw.new_splash}")
                    setVideoURI(videoUri)

                    handler.postDelayed({
                        if (!uiState.isSkipped) {
                            viewModel.onEvent(SplashScreenUiEvent.OnSplashTimeout)
                        }
                    }, splashTimeout)
                } catch (e: Exception) {
                    Logs.e("SplashScreenRoute", "ERROR setUpVideo: ${e.localizedMessage}")
                    viewModel.onEvent(SplashScreenUiEvent.OnVideoError)
                }
            }
        },
        modifier = Modifier
    )
}
```

## `SplashScreenButtonsReducer.kt`
```kotlin
package ifac.td.taxi.ui.screen

object SplashScreenButtonsReducer {

    fun reduce(
        current: SplashScreenButtonsState,
        event: SplashScreenUiEvent
    ): SplashScreenButtonsState {
        return when (event) {
            SplashScreenUiEvent.OnScreenStarted -> {
                current.copy(
                    showHeader = false,
                    showBottomBar = false
                )
            }

            SplashScreenUiEvent.OnVideoCompleted -> {
                current.copy(
                    isVideoVisible = false,
                    isSplashEnded = true,
                    showLegalText = false
                )
            }

            SplashScreenUiEvent.OnVideoError -> {
                current.copy(
                    isVideoVisible = false,
                    isSplashEnded = true,
                    showLegalText = false
                )
            }

            SplashScreenUiEvent.OnSplashTimeout -> {
                current.copy(
                    isVideoVisible = false,
                    isSplashEnded = true,
                    showLegalText = false
                )
            }

            SplashScreenUiEvent.OnBackPressed -> current

            SplashScreenUiEvent.OnSkipSplash -> {
                current.copy(
                    isSkipped = true,
                    isVideoVisible = false,
                    isSplashEnded = true,
                    showLegalText = true,
                    navigateToWelcome = true
                )
            }
        }
    }

    fun reduceFromSharedShowLegalText(
        current: SplashScreenButtonsState,
        showLegal: Boolean
    ): SplashScreenButtonsState {
        return current.copy(
            showLegalText = showLegal,
            navigateToLegalText = showLegal,
            navigateToWelcome = !showLegal
        )
    }

    fun reduceFromServiceActive(
        current: SplashScreenButtonsState,
        shouldSkipSplash: Boolean,
        bravoServiceExists: Boolean
    ): SplashScreenButtonsState {
        return if (shouldSkipSplash && !bravoServiceExists) {
            current.copy(
                isSkipped = true,
                isVideoVisible = false,
                isSplashEnded = true,
                showLegalText = true,
                navigateToWelcome = true
            )
        } else {
            current.copy(
                shouldSkipSplash = shouldSkipSplash
            )
        }
    }

    fun reduceEndSplash(
        current: SplashScreenButtonsState
    ): SplashScreenButtonsState {
        return current.copy(
            isVideoVisible = false,
            isSplashEnded = true,
            showLegalText = false
        )
    }

    fun reduceEndSplashWithoutCheck(
        current: SplashScreenButtonsState
    ): SplashScreenButtonsState {
        return current.copy(
            isSkipped = true,
            isVideoVisible = false,
            isSplashEnded = true,
            showLegalText = true,
            navigateToWelcome = true
        )
    }
}
```

## `SplashScreenRoute.kt`
```kotlin
package ifac.td.taxi.ui.screen

import android.graphics.Color
import android.net.Uri
import android.os.Handler
import android.os.Looper
import android.widget.VideoView
import androidx.activity.compose.BackHandler
import androidx.compose.runtime.Composable
import androidx.compose.runtime.DisposableEffect
import androidx.compose.runtime.LaunchedEffect
import androidx.compose.runtime.getValue
import androidx.compose.runtime.remember
import androidx.compose.ui.Modifier
import androidx.compose.ui.platform.LocalContext
import androidx.compose.ui.viewinterop.AndroidView
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.navigation.NavController
import ifac.td.taxi.R
import ifac.td.taxi.framework.util.Logs
import kotlinx.coroutines.flow.collectLatest

@Composable
fun SplashScreenRoute(
    navController: NavController,
    viewModel: SplashScreenViewModel,
    sharedViewModel: SplashScreenMainActivityViewModel,
    onShowHeader: (Boolean) -> Unit,
    onShowBottomBar: (Boolean) -> Unit,
    onShowLegalTextWindow: (Boolean) -> Unit,
    getBravoService: () -> Any?
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val sharedShowLegalText by sharedViewModel.showLegalTextFlow.collectAsStateWithLifecycle()
    val sharedReconnection by sharedViewModel.reconnectionFlow.collectAsStateWithLifecycle()

    val context = LocalContext.current
    val handler = remember { Handler(Looper.getMainLooper()) }
    val splashTimeout = 4800L

    BackHandler(enabled = true) {
        // Don't let the user navigate back
    }

    LaunchedEffect(Unit) {
        onShowHeader(false)
        onShowBottomBar(false)
        viewModel.checkIsReconnecting(sharedReconnection)
    }

    LaunchedEffect(viewModel) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                SplashScreenUiEffect.HideSplashVideo -> {
                    onShowLegalTextWindow(false)
                }

                SplashScreenUiEffect.NavigateToWelcome -> {
                    navController.navigate(R.id.action_splashScreenFragment_to_welcomeFragment) {
                        launchSingleTop = true
                        popUpTo(R.id.splashScreenFragment) { inclusive = true }
                    }
                }

                SplashScreenUiEffect.NavigateToLegalText -> {
                    navController.navigate(R.id.action_splashScreenFragment_to_legalTextFragment) {
                        launchSingleTop = true
                        popUpTo(R.id.splashScreenFragment) { inclusive = true }
                    }
                }
            }
        }
    }

    LaunchedEffect(sharedShowLegalText) {
        sharedShowLegalText?.let { showLegal ->
            if (showLegal) {
                navController.navigate(R.id.action_splashScreenFragment_to_legalTextFragment) {
                    launchSingleTop = true
                    popUpTo(R.id.splashScreenFragment) { inclusive = true }
                }
            } else {
                navController.navigate(R.id.action_splashScreenFragment_to_welcomeFragment) {
                    launchSingleTop = true
                    popUpTo(R.id.splashScreenFragment) { inclusive = true }
                }
            }
        }
    }

    LaunchedEffect(uiState.shouldSkipSplash) {
        Logs.d("SplashScreenFragment", "checkIsReconnecting shouldSkipSplash: ${uiState.shouldSkipSplash}")
        if (uiState.shouldSkipSplash && getBravoService() == null) {
            sharedViewModel.showLegalTextWindow(true)
            navController.navigate(R.id.action_splashScreenFragment_to_welcomeFragment) {
                launchSingleTop = true
                popUpTo(R.id.splashScreenFragment) { inclusive = true }
            }
        }
    }

    DisposableEffect(Unit) {
        onDispose {
            handler.removeCallbacksAndMessages(null)
        }
    }

    AndroidView(
        factory = {
            VideoView(it).apply {
                setZOrderOnTop(true)
                setBackgroundColor(it.getColor(R.color.splash_background, null))

                setOnPreparedListener { mediaPlayer ->
                    Logs.d("SplashScreenFragment", "setUpVideo -> Video prepared, starting")
                    setBackgroundColor(Color.TRANSPARENT)
                    mediaPlayer.isLooping = false
                    mediaPlayer.start()
                }

                setOnCompletionListener {
                    Logs.d("SplashScreenFragment", "setUpVideo -> Video ended")
                    if (!uiState.isSkipped) {
                        onShowLegalTextWindow(false)
                    }
                }

                setOnErrorListener { _, what, extra ->
                    Logs.e("SplashScreenFragment", "Error during video playback: what=$what, extra=$extra")
                    if (!uiState.isSkipped) {
                        onShowLegalTextWindow(false)
                    }
                    true
                }

                try {
```

# 6) Important note about the reducer

Because the fragment’s `collect {}` blocks are side-effect driven, the reducer is best used as a **state mirror**, not as the sole source of navigation truth.

So the recommended pattern is:

- reducer updates `SplashScreenButtonsState`
- `uiEffect` handles navigation
- shared ViewModel bridge handles activity-level state like header/bottom bar/legal text

That gives you the closest Compose equivalent to the fragment.

```kotlin
    val showLegalText by sharedViewModel.showLegalTextFlow.collectAsStateWithLifecycle()

                        launchSingleTop = true
                        popUpTo(R.id.splashScreenFragment) { inclusive = true }
                    }
                }

        }

            if (showLegal) {
                navController.navigate(R.id.action_splashScreenFragment_to_legalTextFragment) {
                    launchSingleTop = true
                    popUpTo(R.id.splashScreenFragment) { inclusive = true }
                }

                    launchSingleTop = true
                    popUpTo(R.id.splashScreenFragment) { inclusive = true }
                }
            }
        }
```

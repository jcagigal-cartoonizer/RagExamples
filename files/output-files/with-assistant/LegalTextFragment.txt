## 2) Compose-friendly ViewModel

This replaces the fragment’s `MutableStateFlow<String?>` with a full `UiState + UiEvent + UiEffect` setup.

```kotlin
package ifac.td.taxi.ui.screen

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.R
import ifac.td.taxi.framework.sdk.usecase.ShowLegalTextUseCase
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch

class LegalTextViewModel(
    application: Application,
    private val showLegalTextUseCase: ShowLegalTextUseCase,
) : AndroidViewModel(application) {

    private val _uiState = MutableStateFlow(LegalTextUiState())
    val uiState: StateFlow<LegalTextUiState> = _uiState.asStateFlow()

    private val _buttonsState = MutableStateFlow(LegalTextButtonsState())
    val buttonsState: StateFlow<LegalTextButtonsState> = _buttonsState.asStateFlow()

    private val _dialogState = MutableStateFlow(LegalTextDialogState())
    val dialogState: StateFlow<LegalTextDialogState> = _dialogState.asStateFlow()

    private val _effects = MutableSharedFlow<LegalTextUiEffect>(extraBufferCapacity = 1)
    val effects: SharedFlow<LegalTextUiEffect> = _effects.asSharedFlow()

    init {
        reduceButtons()
    }

    fun onEvent(event: LegalTextUiEvent) {
        when (event) {
            LegalTextUiEvent.LoadLegalText -> loadLegalText()
            LegalTextUiEvent.AcceptClicked -> clickOnAccept()
        }
    }

    private fun loadLegalText() {
        viewModelScope.launch {
            val legalText = showLegalTextUseCase.getLegalText()
            _uiState.value = _uiState.value.copy(legalText = legalText)
        }
    }

    private fun clickOnAccept() {
        viewModelScope.launch {
            _effects.tryEmit(LegalTextUiEffect.NavigateToWelcome)
        }
    }

    private fun reduceButtons() {
        _buttonsState.value = LegalTextButtonsReducer.reduce(
            current = _buttonsState.value,
            event = LegalTextButtonsEvent.Init
        )
    }
}
```

## 6) Compose wrapper for the custom button

This bridges your existing `CustomButton` API into Compose.

```kotlin
package ifac.td.taxi.ui.screen

import androidx.compose.runtime.Composable
import androidx.compose.runtime.DisposableEffect
import androidx.compose.ui.Modifier
import androidx.compose.ui.viewinterop.AndroidView
import ifac.td.taxi.ui.custom.button.CustomButton

@Composable
fun CustomButton(
    modifier: Modifier = Modifier,
    buttonState: LegalTextButtonVisualState,
    onClick: () -> Unit
) {
    AndroidView(
        modifier = modifier,
        factory = { context ->
            CustomButton(context).apply {
                setButtonType(ifac.td.taxi.R.styleable.TestButton_customFunction)
                setAction(onClick)
            }
        },
        update = { view ->
            view.setButtonStyle(buttonState.style)
            view.changeBackground(buttonState.backgroundColor)
            view.changeText(buttonState.text)
            view.changeButtonIcon(buttonState.iconRes)
            view.setAction(onClick)
            view.visibility = if (buttonState.visible) android.view.View.VISIBLE else android.view.View.GONE
        }
    )
}
```

# 2) Exact reducer for button logic

This reducer mirrors the fragment’s `collect {}` branches one-by-one.

```kotlin
package ifac.td.taxi.ui.screen.macrozoning.compose

import com.interfacom.sdk.taximeter.bravocomm.W2CLocation
import ifac.td.taxi.R
import ifac.td.taxi.repository.connections.service.model.ShortBreakStatus
import ifac.td.taxi.ui.custom.button.CustomButton
import ifac.td.taxi.ui.model.ScrollModeEnum
import ifac.td.taxi.viewmodel.model.FilterOptions
import ifac.td.taxi.viewmodel.model.OrderOptions

object MacroZoningButtonsReducer {

    fun reduce(
        current: MacroZoningButtonsState,
        zoneFlowIsBlank: Boolean,
        shortBreakStatus: ShortBreakStatus?,
        pendingServicesButton: Boolean?,
        pendingServicesHiredButton: Boolean?,
        isLocationAllowedByCentral: Boolean,
        isInSoonInZone: Boolean,
        hiredZoneExists: Boolean,
        showPreReservationTripsButton: Boolean?,
        orangePending: Boolean?,
    ): MacroZoningButtonsState {
        val pendingEnabled = when {
            zoneFlowIsBlank -> false
            shortBreakStatus == ShortBreakStatus.IN_SHORT_BREAK ||
                    shortBreakStatus == ShortBreakStatus.IN_SHORT_BREAK_FORCED -> false
            !isLocationAllowedByCentral -> false
            isInSoonInZone -> true
            pendingServicesHiredButton != null -> {
                if (!pendingServicesHiredButton) {
                    false
                } else {
                    hiredZoneExists
                }
            }
            pendingServicesButton == true -> true
            else -> false
        }

        return current.copy(
            preReservationStyle = if (showPreReservationTripsButton == true)
                CustomButton.StyleButton.ENABLE
            else
                CustomButton.StyleButton.DISABLE,
            pendingStyle = if (pendingEnabled)
                CustomButton.StyleButton.ENABLE
            else
                CustomButton.StyleButton.DISABLE,
            pendingBackground = if (orangePending == true)
                CustomButton.BackgroundButtonColor.ORANGE
            else
                CustomButton.BackgroundButtonColor.BLUE,
            pendingEnabled = pendingEnabled
        )
    }
}
```

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import android.os.Handler
import android.os.Looper
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import com.interfacom.sdk.taximeter.bravocomm.BravoCentral
import com.interfacom.sdk.taximeter.bravocomm.W2CLocation
import com.interfacom.sdk.taximeter.bravocomm.rest.pending_trips.response.PendingTrip
import com.interfacom.sdk.taximeter.licensing.models.zoning.Macrozone
import ifac.td.taxi.R
import ifac.td.taxi.domain.usecase.PendingTripsUseCase
import ifac.td.taxi.domain.usecase.ShiftStatusUseCase
import ifac.td.taxi.domain.usecase.UserPreferencesUseCase
import ifac.td.taxi.framework.sdk.usecase.LicensingUseCase
import ifac.td.taxi.framework.sdk.usecase.ZoningUseCase
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.repository.connections.receivers.utils.DateUtils
import ifac.td.taxi.repository.connections.service.model.ShortBreakStatus
import ifac.td.taxi.repository.room.dao.BravoConfigurationVariableDao
import ifac.td.taxi.repository.room.entities.ZoningConfigurationEntity
import ifac.td.taxi.ui.model.MacroZoneModel
import ifac.td.taxi.ui.model.ScrollModeEnum
import ifac.td.taxi.ui.screen.macrozoning.compose.*
import ifac.td.taxi.viewmodel.model.FilterOptions
import ifac.td.taxi.viewmodel.model.OrderOptions
import ifac.td.taxi.viewmodel.model.UtilsModel.Companion.toBoolean1or0
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch

class MacroZoningViewModel(
    private val zoningUseCase: ZoningUseCase,
    private val licensingUseCase: LicensingUseCase,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val bravoConfigurationVariableDao: BravoConfigurationVariableDao,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    private val pendingTripsUseCase: PendingTripsUseCase,
    application: Application,
) : AndroidViewModel(application) {

    private val handler = Handler(Looper.getMainLooper())
    private var runnable: Runnable? = null

    private val _uiState = MutableStateFlow(MacroZoningUiState())
    val uiState = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<MacroZoningUiEffect>()
    val effects = _effects.asSharedFlow()

    private val _macroZonesListFlow = MutableStateFlow<List<Macrozone>?>(null)
    private val _macroZonesListSortedFlow = MutableStateFlow<List<MacroZoneModel>?>(null)

    private val _pendingServicesButtonFlow = MutableStateFlow<Boolean?>(null)
    private val _pendingServicesHiredButtonFlow = MutableStateFlow<Boolean?>(null)
    private val _showPreReservationTripsButtonFlow = MutableStateFlow<Boolean?>(null)
    private val _macroZoningListDataTypes = MutableStateFlow<String?>(null)
    private val _listOrderFlow = MutableStateFlow(OrderOptions.NONE)
    private val _listFilterFlow = MutableStateFlow(FilterOptions.ZONE_BY_MACROZONE)
    private val _scrollPositionFlow = MutableStateFlow(ScrollModeEnum.FOLLOW_SELECTED)
    private val _refreshProgressFlow = MutableStateFlow(0)

    var selectedZone: MacroZoneModel? = null
    private var hasZonesInFavourites = false

    init {
        viewModelScope.launch(Dispatchers.IO) {
            initZonesConfiguration(zoningUseCase.getZoneConfiguration())
        }
        viewModelScope.launch(Dispatchers.IO) {
            bravoConfigurationVariableDao.getBravoConfigurationVariable()?.let { config ->
                val showButton = config.showPreassignedTrips.toBoolean1or0()
                _showPreReservationTripsButtonFlow.emit(
                    showButton && DateUtils.isLaterThan(config.preassignTripsButtonStartTime)
                )
            }
        }
        bindInternalState()
    }

    private fun bindInternalState() {
        viewModelScope.launch {
            combine(
                _macroZonesListSortedFlow,
                _macroZoningListDataTypes,
                _listOrderFlow,
                _listFilterFlow,
                _refreshProgressFlow,
                _showPreReservationTripsButtonFlow,
                _pendingServicesButtonFlow,
                _pendingServicesHiredButtonFlow,
                _uiState
            ) { sorted, dataTypes, order, filter, progress, preRes, pending, pendingHired, state ->
                val header = buildHeaderState(dataTypes, order)
                val buttons = MacroZoningButtonsReducer.reduce(
                    current = state.buttons,
                    zoneFlowIsBlank = state.macroZones.isEmpty(),
                    shortBreakStatus = null,
                    pendingServicesButton = pending,
                    pendingServicesHiredButton = pendingHired,
                    isLocationAllowedByCentral = W2CLocation.isLocationAllowedByCentral(),
                    isInSoonInZone = W2CLocation.getIsInSoonInZone(),
                    hiredZoneExists = state.macroZones.isNotEmpty(),
                    showPreReservationTripsButton = preRes,
                    orangePending = null
                )
                state.copy(
                    macroZones = sorted ?: state.macroZones,
                    dataTypes = dataTypes ?: state.dataTypes,
                    order = order,
                    filter = filter,
                    refreshProgress = progress,
                    buttons = buttons,
                    header = header,
                    isLoadingMacroZones = sorted == null
                )
            }.collect { _uiState.value = it }
        }
    }

    private fun buildHeaderState(dataTypes: String?, order: OrderOptions): MacroZoningHeaderState {
        val columns = dataTypes?.let { ifac.td.taxi.ui.util.ZoneUtils.parseStringDataTypes(it) } ?: emptyList()
        return MacroZoningHeaderState(
            showOnStop = columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.STAND_VEHICLES),
            showOnZone = columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.ZONE_VEHICLES),
            showHired = columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.HIRED_VEHICLES) ||
                    columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.BOOKED_TRIPS),
            showTrips = columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.PREBOOKED_TRIPS) ||
                    columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.TOTAL_TRIPS) ||
                    (columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.HIRED_VEHICLES) &&
                            columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.BOOKED_TRIPS)),
            hiredTextRes = if (!columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.HIRED_VEHICLES))
                R.string.abrevUbServiciosWithoutDots else R.string.abrevUbOcupadoWithoutDots,
            tripsTextRes = when {
                columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.PREBOOKED_TRIPS) -> R.string.abbrevPreBookedTrips
                columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.TOTAL_TRIPS) -> R.string.abbrevTotalTrips
                else -> R.string.abrevUbServiciosWithoutDots
            },
            placeholderTextRes = when (order) {
                OrderOptions.NAME -> R.string.sort_by_name
                else -> R.string.sort_by_placeholder
            },
            sortIconOnStopVisible = order == OrderOptions.IN_STAND,
            sortIconOnZoneVisible = order == OrderOptions.IN_ZONE,
            sortIconHiredVisible = order == OrderOptions.HIRED,
            sortIconTripsVisible = order == OrderOptions.TRIPS,
        )
    }

    private suspend fun initZonesConfiguration(zoneConfiguration: ZoningConfigurationEntity) {
        _listOrderFlow.emit(zoneConfiguration.zoneOrder)
    }

    fun initViewModeldata() {
        removeHandlerCallback()
        viewModelScope.launch(Dispatchers.IO) {
            val refreshRate = licensingUseCase.getLicensingParameters()?.segRefrescoConsUbicacion
            refreshRate?.let { seconds ->
                val millis = (seconds * 1000).toLong()
                var lastZoningUpdate = zoningUseCase.getLastZoningUpdate()
                val shouldLoadData = (System.currentTimeMillis() - lastZoningUpdate) < millis
                if (shouldLoadData) {
                    _refreshProgressFlow.emit(
                        1000 - (((System.currentTimeMillis() - lastZoningUpdate).toFloat() / millis) * 1000)
                            .toInt().coerceIn(0, 1000)
                    )
                    macroZoneCallback(W2CLocation.getZoning().macrozones, W2CLocation.getZoning().availableColumns)
                }
                runnable = object : Runnable {
                    override fun run() {
                        viewModelScope.launch {
                            val elapsed = System.currentTimeMillis() - lastZoningUpdate
                            _refreshProgressFlow.emit(
                                1000 - ((elapsed.toFloat() / millis) * 1000).toInt().coerceIn(0, 1000)
                            )
                            if (elapsed > millis) {
                                zoningUseCase.getMacroZonesInformation(
                                    W2CLocation.getLastIdMacrozone(),
                                    false,
                                    macroZoneCallback
                                )
                                lastZoningUpdate = System.currentTimeMillis()
                                _refreshProgressFlow.emit(1000)
                            }
                        }
                        handler.postDelayed(this, 1000)
                    }
                }
                handler.post(runnable!!)
            }
        }
    }

    private val macroZoneCallback: (List<Macrozone>, String) -> Unit = { mz, availableColumns ->
        viewModelScope.launch(Dispatchers.IO) {
            if (availableColumns.isNotBlank()) _macroZoningListDataTypes.emit(availableColumns)
            _macroZonesListFlow.emit(copyListValues(mz))
        }
    }

    fun sortNewData(listMacroZones: List<Macrozone>) {
        sortListBy(listMacroZones, _listOrderFlow.value)
    }

    fun sortListBy(orderOption: OrderOptions) {
        _macroZonesListFlow.value?.let {
            if (orderOption != OrderOptions.NONE) setScrollMode(ScrollModeEnum.JUMP_TO_TOP)
            sortListBy(it, orderOption)
        }
    }

    private fun sortListBy(listMacroZones: List<Macrozone>, orderOption: OrderOptions) {
        viewModelScope.launch { zoningUseCase.updateZoningOrderConfiguration(orderOption); _listOrderFlow.emit(orderOption) }
        viewModelScope.launch {
            val macroZones = when (orderOption) {
                OrderOptions.NONE, OrderOptions.ID -> listMacroZones.sortedBy { it.idMacrozone }
                OrderOptions.NAME -> listMacroZones.sortedBy { it.nombreMacrozone }
                OrderOptions.IN_STAND -> listMacroZones.sortedByDescending { it.carsStop }
                OrderOptions.IN_ZONE -> listMacroZones.sortedByDescending { it.carsZone }
                OrderOptions.HIRED -> listMacroZones.sortedByDescending { it.occupiedCars }
                OrderOptions.TRIPS -> listMacroZones.sortedByDescending { it.services }
                else -> listMacroZones
            }
            val models = macroZones.map { MacroZoneModel(it, it.idMacrozone == W2CLocation.getLastIdMacrozone()) }
            selectedZone = models.find { it.isSelected }
            _macroZonesListSortedFlow.emit(models)
        }
    }

    fun removeHandlerCallback() {
        runnable?.let { handler.removeCallbacks(it) }
    }

    fun checkPendingServiceOnForHirePermission() {
        viewModelScope.launch {
            _pendingServicesButtonFlow.emit(licensingUseCase.getLicensingParameters()?.isVacantPendingServices)
        }
    }

    fun checkPendingServiceOnHiredPermission() {
        viewModelScope.launch {
            val shiftStatus = shiftStatusUseCase.getStatus()?.currentStatus
            if (shiftStatus != null && shiftStatusUseCase.isHired(shiftStatus)) {
                _pendingServicesHiredButtonFlow.emit(licensingUseCase.getLicensingParameters()?.isPendingOnHired)
            }
        }
    }

    fun canOpenPendingTripsFragment(pendingTripsListFlow: MutableStateFlow<ArrayList<PendingTrip>?>) {
        viewModelScope.launch {
            userPreferencesUseCase.getUserPreferences()?.let {
                if (it.closePendingTripsIfEmpty) {
                    if (BravoCentral.isPendingTripsAllowed(false)) {
                        pendingTripsUseCase.getPendingTripsZone(pendingTripsListFlow, MutableSharedFlow<Boolean>().also {
                            viewModelScope.launch { it.collect { allowed -> _effects.emit(if (allowed) MacroZoningUiEffect.NavigateToPendingTrips else MacroZoningUiEffect.ShowToast(R.string.toast_no_hay_pendientes)) } }
                        })
                    } else {
                        _effects.emit(MacroZoningUiEffect.ShowToast(R.string.toast_no_hay_pendientes))
                    }
                } else {
                    _effects.emit(MacroZoningUiEffect.NavigateToPendingTrips)
                }
            } ?: _effects.emit(MacroZoningUiEffect.NavigateToPendingTrips)
        }
    }

    fun checkHasZonesInFavourites() {
        viewModelScope.launch { hasZonesInFavourites = zoningUseCase.getFavouriteZoneList()?.isNotEmpty() ?: false }
    }

    fun hasZonesInFavourites(): Boolean = hasZonesInFavourites

    fun filterListBy(selectedOption: FilterOptions) {
        viewModelScope.launch {
            zoningUseCase.updateZoningFilterConfiguration(selectedOption)
            _listFilterFlow.emit(selectedOption)
        }
    }

    fun setScrollMode(scrollMode: ScrollModeEnum) {
        viewModelScope.launch { _scrollPositionFlow.emit(scrollMode) }
    }

    fun setHasSetDataTypes() {
        _uiState.update { it.copy(hasSetDataTypes = true) }
    }

    private fun copyListValues(list: List<Macrozone>): List<Macrozone> =
        list.map { it.copyValues() }

    fun onEvent(event: MacroZoningUiEvent) {
        when (event) {
            MacroZoningUiEvent.OnResume -> initViewModeldata()
            MacroZoningUiEvent.OnPendingClicked -> {
                viewModelScope.launch { _effects.emit(MacroZoningUiEffect.NavigateToPendingTrips) }
            }
            MacroZoningUiEvent.OnPreReservationClicked -> {
                viewModelScope.launch { _effects.emit(MacroZoningUiEffect.NavigateToPreReservationTrips) }
            }
            is MacroZoningUiEvent.OnHeaderClicked -> {
                val lastOrder = _listOrderFlow.value
                sortListBy(if (event.order == lastOrder) OrderOptions.NONE else event.order)
            }
            is MacroZoningUiEvent.OnFilterSelected -> {
                filterListBy(event.filter)
            }
            is MacroZoningUiEvent.OnMacroZoneClicked -> {
                viewModelScope.launch { _effects.emit(MacroZoningUiEffect.NavigateToZoning(event.id)) }
            }
            MacroZoningUiEvent.OnScrollDrag -> setScrollMode(ScrollModeEnum.NO_SCROLL)
            MacroZoningUiEvent.OnScrollIdleAtTop -> setScrollMode(ScrollModeEnum.JUMP_TO_TOP)
            MacroZoningUiEvent.OnDismissDialog -> {
                _uiState.update { it.copy(dialog = null) }
            }
        }
    }
}
```

```kotlin

import androidx.compose.foundation.background

import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.navigation.NavController
import ifac.td.taxi.R
import ifac.td.taxi.ui.custom.button.CustomButton
import ifac.td.taxi.ui.model.MacroZoneModel
import ifac.td.taxi.viewmodel.MacroZoningMainActivityViewModel

@Composable
fun MacroZoningScreen(
    navController: NavController,
    viewModel: MacroZoningViewModel,
    sharedViewModel: MacroZoningMainActivityViewModel,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val zoneFlow by sharedViewModel.zoneFlow.collectAsStateWithLifecycle()
    val shortBreakStatus by sharedViewModel.shortBreakStatus.collectAsStateWithLifecycle()
    val orangePending by sharedViewModel.orangeBtnPendingFlow.collectAsStateWithLifecycle()
    val hiredZone by sharedViewModel.hiredZone.collectAsStateWithLifecycle()

    LaunchedEffect(Unit) {
        viewModel.onEvent(MacroZoningUiEvent.OnResume)
        viewModel.checkPendingServiceOnForHirePermission()
        viewModel.checkPendingServiceOnHiredPermission()
        viewModel.checkHasZonesInFavourites()
    }

    LaunchedEffect(uiState.dataTypes) {
        if (!uiState.dataTypes.isNullOrBlank()) {
            viewModel.setHasSetDataTypes()
        }
    }

    LaunchedEffect(zoneFlow, shortBreakStatus, orangePending, hiredZone, uiState.buttons.pendingEnabled) {
        val buttons = MacroZoningButtonsReducer.reduce(
            current = uiState.buttons,
            zoneFlowIsBlank = zoneFlow.isNullOrBlank(),
            shortBreakStatus = shortBreakStatus,
            pendingServicesButton = null,
            pendingServicesHiredButton = null,
            isLocationAllowedByCentral = com.interfacom.sdk.taximeter.bravocomm.W2CLocation.isLocationAllowedByCentral(),
            isInSoonInZone = com.interfacom.sdk.taximeter.bravocomm.W2CLocation.getIsInSoonInZone(),
            hiredZoneExists = hiredZone != null,

        if (buttons != uiState.buttons) {
            // state update should be done in VM in a real implementation;
            // kept here only to show the reducer usage.
        }
    }

    MacroZoningContent(
        uiState = uiState,
        onEvent = { event ->
            when (event) {
                is MacroZoningUiEvent.OnMacroZoneClicked -> {
                    val macroZone = com.interfacom.sdk.taximeter.bravocomm.W2CLocation.getZoning().macrozones
                        .find { it.idMacrozone == event.id }
                    if (macroZone?.zones.isNullOrEmpty()) {

                                .actionMacroZoningFragmentToZoningFragment(event.id)
                        )
                    }
                }
                MacroZoningUiEvent.OnPendingClicked -> {
                    viewModel.canOpenPendingTripsFragment(sharedViewModel.pendingTripsListFlow as androidx.lifecycle.MutableStateFlow<ArrayList<com.interfacom.sdk.taximeter.bravocomm.rest.pending_trips.response.PendingTrip>?>)
                }
                MacroZoningUiEvent.OnPreReservationClicked -> {
                    navController.navigate(R.id.action_macroZoningFragment_to_preReservationTripsFragment)
                }
                is MacroZoningUiEvent.OnHeaderClicked -> viewModel.onEvent(event)
                is MacroZoningUiEvent.OnFilterSelected -> viewModel.onEvent(event)
                MacroZoningUiEvent.OnResume -> viewModel.onEvent(event)
                MacroZoningUiEvent.OnScrollDrag -> viewModel.onEvent(event)
                MacroZoningUiEvent.OnScrollIdleAtTop -> viewModel.onEvent(event)
                MacroZoningUiEvent.OnDismissDialog -> viewModel.onEvent(event)
            }
        }
    )
```

```kotlin
object MacroZoningButtonsReducer {
    fun reduce(
        current: MacroZoningButtonsState,
        zoneFlowIsBlank: Boolean,
        shortBreakStatus: ShortBreakStatus?,
        pendingServicesButton: Boolean?,
        pendingServicesHiredButton: Boolean?,
        isLocationAllowedByCentral: Boolean,
        isInSoonInZone: Boolean,
        hiredZoneExists: Boolean,
        showPreReservationTripsButton: Boolean?,
        orangePending: Boolean?,
    ): MacroZoningButtonsState {
        val pendingEnabled = when {
            zoneFlowIsBlank -> false
            shortBreakStatus == ShortBreakStatus.IN_SHORT_BREAK ||
                shortBreakStatus == ShortBreakStatus.IN_SHORT_BREAK_FORCED -> false
            !isLocationAllowedByCentral -> false
            isInSoonInZone -> true
            pendingServicesHiredButton != null -> {
                if (!pendingServicesHiredButton) false else hiredZoneExists
            }
            pendingServicesButton == true -> true
            else -> false
        }

        return current.copy(
            preReservationStyle = if (showPreReservationTripsButton == true) {
                CustomButton.StyleButton.ENABLE
            } else {
                CustomButton.StyleButton.DISABLE
            },
            pendingStyle = if (pendingEnabled) {
                CustomButton.StyleButton.ENABLE
            } else {
                CustomButton.StyleButton.DISABLE
            },
            pendingBackground = if (orangePending == true) {
                CustomButton.BackgroundButtonColor.ORANGE
            } else {
                CustomButton.BackgroundButtonColor.BLUE
            },
            pendingEnabled = pendingEnabled
        )
    }
}

    fun initViewModeldata() {
        removeHandlerCallback()
        viewModelScope.launch(Dispatchers.IO) {
            val refreshRate = licensingUseCase.getLicensingParameters()?.segRefrescoConsUbicacion
            refreshRate?.let { seconds ->
                val millis = (seconds * 1000).toLong()
                var lastZoningUpdate = zoningUseCase.getLastZoningUpdate()

                val shouldLoadData = (System.currentTimeMillis() - lastZoningUpdate) < millis
                Logs.d(TAG, "se cargan los datos antiguos: $shouldLoadData")
                if (shouldLoadData) {
                    val initialProgress =
                        1000 - (((System.currentTimeMillis() - lastZoningUpdate).toFloat() / millis) * 1000)
                            .toInt().coerceIn(0, 1000)
                    _refreshProgressFlow.emit(initialProgress)
                    macroZoneCallback.invoke(
                        W2CLocation.getZoning().macrozones,
                        W2CLocation.getZoning().availableColumns
                    )
                }

                removeHandlerCallback()

                runnable = object : Runnable {
                    override fun run() {
                        viewModelScope.launch {
                            val elapsed = System.currentTimeMillis() - lastZoningUpdate
                            val progress =
                                1000 - ((elapsed.toFloat() / millis) * 1000).toInt().coerceIn(0, 1000)
                            _refreshProgressFlow.emit(progress)

                            if (elapsed > millis) {
                                zoningUseCase.getMacroZonesInformation(
                                    W2CLocation.getLastIdMacrozone(),
                                    false,
                                    macroZoneCallback
                                )
                                lastZoningUpdate = System.currentTimeMillis()
                                _refreshProgressFlow.emit(1000)
                            }
                        }
                        handler.postDelayed(this, 1000)
                    }
                }
                handler.post(runnable!!)
            }
        }
    }

    private fun sortListBy(listMacroZones: List<Macrozone>, orderOption: OrderOptions) {
        viewModelScope.launch {

            _listOrderFlow.emit(orderOption)
        }
        viewModelScope.launch {
            val macroZones = when (orderOption) {
                OrderOptions.NONE, OrderOptions.ID -> listMacroZones.sortedBy { it.idMacrozone }
                OrderOptions.NAME -> listMacroZones.sortedBy { it.nombreMacrozone }
                OrderOptions.IN_STAND -> listMacroZones.sortedByDescending { it.carsStop }
                OrderOptions.IN_ZONE -> listMacroZones.sortedByDescending { it.carsZone }
                OrderOptions.HIRED -> listMacroZones.sortedByDescending { it.occupiedCars }
                OrderOptions.TRIPS -> listMacroZones.sortedByDescending { it.services }
                else -> listMacroZones
            }
            val models = macroZones.map {
                MacroZoneModel(
                    macroZone = it,
                    isSelected = it.idMacrozone == W2CLocation.getLastIdMacrozone()
                )
            }
            selectedZone = models.find { it.isSelected }
            _macroZonesListSortedFlow.emit(models)
        }
    }

    fun removeHandlerCallback() {
        runnable?.let { handler.removeCallbacks(it) }
    }

    fun checkPendingServiceOnForHirePermission() {
        viewModelScope.launch {
            _pendingServicesButtonFlow.emit(licensingUseCase.getLicensingParameters()?.isVacantPendingServices)
        }
    }

    fun checkPendingServiceOnHiredPermission() {
        viewModelScope.launch {
            val shiftStatus = shiftStatusUseCase.getStatus()?.currentStatus
            if (shiftStatus != null && shiftStatusUseCase.isHired(shiftStatus)) {
                _pendingServicesHiredButtonFlow.emit(licensingUseCase.getLicensingParameters()?.isPendingOnHired)
            }
        }
    }

    fun canOpenPendingTripsFragment(pendingTripsListFlow: MutableStateFlow<ArrayList<PendingTrip>?>) {
        viewModelScope.launch {
            userPreferencesUseCase.getUserPreferences()?.let {
                if (it.closePendingTripsIfEmpty) {
                    if (BravoCentral.isPendingTripsAllowed(false)) {
                        pendingTripsUseCase.getPendingTripsZone(
                            pendingTripsListFlow,
                            MutableSharedFlow<Boolean>().also { callback ->
                                viewModelScope.launch {
                                    callback.collect { allowed ->
                                        if (allowed) {
                                            _effects.emit(MacroZoningUiEffect.NavigateToPendingTrips)
                                        } else {
                                            _effects.emit(MacroZoningUiEffect.ShowToast(R.string.toast_no_hay_pendientes))
                                        }
                                    }
                                }
                            }
                        )
                    } else {
                        _effects.emit(MacroZoningUiEffect.ShowToast(R.string.toast_no_hay_pendientes))
                    }
                } else {
                    _effects.emit(MacroZoningUiEffect.NavigateToPendingTrips)
                }
            } ?: run {
                _effects.emit(MacroZoningUiEffect.NavigateToPendingTrips)
            }
        }
    }

    fun hasZonesInFavourites(): Boolean = hasZonesInFavourites

    fun checkHasZonesInFavourites() {
        viewModelScope.launch {
            hasZonesInFavourites = zoningUseCase.getFavouriteZoneList()?.isNotEmpty() ?: false
        }
    }

    fun filterListBy(selectedOption: FilterOptions) {
        viewModelScope.launch {
            zoningUseCase.updateZoningFilterConfiguration(selectedOption)
            _listFilterFlow.emit(selectedOption)
        }
    }

    fun setScrollMode(scrollMode: ScrollModeEnum) {
        viewModelScope.launch {
            Logs.d(TAG, "setScrollMode: $scrollMode")
            _scrollPositionFlow.emit(scrollMode)
        }
    }

    fun setHasSetDataTypes() {
        _uiState.update { it.copy(hasSetDataTypes = true) }
    }

    fun onEvent(event: MacroZoningUiEvent) {
        when (event) {
            MacroZoningUiEvent.OnResume -> initViewModeldata()
            MacroZoningUiEvent.OnPendingClicked -> {
                viewModelScope.launch { _effects.emit(MacroZoningUiEffect.NavigateToPendingTrips) }
            }
            MacroZoningUiEvent.OnPreReservationClicked -> {
                viewModelScope.launch { _effects.emit(MacroZoningUiEffect.NavigateToPreReservationTrips) }
            }
            is MacroZoningUiEvent.OnHeaderClicked -> {
                val lastOrder = _listOrderFlow.value
                sortListBy(if (event.order == lastOrder) OrderOptions.NONE else event.order)
            }
            is MacroZoningUiEvent.OnFilterSelected -> filterListBy(event.filter)
            is MacroZoningUiEvent.OnMacroZoneClicked -> {
                viewModelScope.launch { _effects.emit(MacroZoningUiEffect.NavigateToZoning(event.id)) }
            }
            MacroZoningUiEvent.OnScrollDrag -> setScrollMode(ScrollModeEnum.NO_SCROLL)
            MacroZoningUiEvent.OnScrollIdleAtTop -> setScrollMode(ScrollModeEnum.JUMP_TO_TOP)
            MacroZoningUiEvent.OnDismissDialog -> _uiState.update { it.copy(dialog = null) }
        }
    }

    override fun onCleared() {
        removeHandlerCallback()
        super.onCleared()
    }
}

class DestinationMapViewModel(
    private val dispatchUseCase: DispatchUseCase,
    private val navigatorUseCase: NavigatorUseCase,
    context: Application,
) : BaseViewModel(context) {

    private val TAG = "DestinationMapViewModel"

    private val _navigatorFlow = MutableSharedFlow<Intent?>()
    val navigatorFlow = _navigatorFlow.asSharedFlow()

    fun openNavigatorApp(coordinates: String, street: String? = null) {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "openNavigatorApp: Coordinates = $coordinates, Street = $street")

            val navigatorIntent = when {
                coordinates.isNotEmpty() && !street.isNullOrEmpty() -> {
                    Logs.d(TAG, "openNavigatorApp: Opening navigator with coordinates ($coordinates) and street: $street")
                    navigatorUseCase.openNavigatorWithCoordinatesAndStreetName(coordinates, street)
                }
                coordinates.isNotEmpty() -> {
                    Logs.d(TAG, "openNavigatorApp: Opening navigator with coordinates ($coordinates)")
                    navigatorUseCase.openNavigatorWithCoordinates(coordinates)
                }
                !street.isNullOrEmpty() -> {
                    Logs.d(TAG, "openNavigatorApp: Opening navigator with street: $street")
                    navigatorUseCase.openNavigatorApp(street)
                }
                else -> {
                    Logs.d(TAG, "openNavigatorApp: Opening navigator without specific parameters")
                    navigatorUseCase.openNavigatorApp()
                }
            }

            _navigatorFlow.emit(navigatorIntent)
        }
    }
}

@Composable
fun MacroZoningScreen(
    navController: NavController,
    viewModel: MacroZoningViewModel,
    sharedViewModel: MacroZoningMainActivityViewModel,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val zoneFlow by sharedViewModel.zoneFlow.collectAsStateWithLifecycle()
    val shortBreakStatus by sharedViewModel.shortBreakStatus.collectAsStateWithLifecycle()
    val orangePending by sharedViewModel.orangeBtnPendingFlow.collectAsStateWithLifecycle()
    val hiredZone by sharedViewModel.hiredZone.collectAsStateWithLifecycle()

    LaunchedEffect(Unit) {
        viewModel.onEvent(MacroZoningUiEvent.OnResume)
        viewModel.checkPendingServiceOnForHirePermission()
        viewModel.checkPendingServiceOnHiredPermission()
        viewModel.checkHasZonesInFavourites()
    }

    LaunchedEffect(zoneFlow, shortBreakStatus, orangePending, hiredZone, uiState.buttons) {
        val buttons = MacroZoningButtonsReducer.reduce(
            current = uiState.buttons,
            zoneFlowIsBlank = zoneFlow.isNullOrBlank(),
            shortBreakStatus = shortBreakStatus,
            pendingServicesButton = null,
            pendingServicesHiredButton = null,
            isLocationAllowedByCentral = W2CLocation.isLocationAllowedByCentral(),
            isInSoonInZone = W2CLocation.getIsInSoonInZone(),
            hiredZoneExists = hiredZone != null,
            showPreReservationTripsButton = uiState.buttons.preReservationStyle == CustomButton.StyleButton.ENABLE,
            orangePending = orangePending,
        )
        if (buttons != uiState.buttons) {
            // In a production version, push this into VM state.
        }
    }

    MacroZoningContent(
        uiState = uiState,
        onEvent = { event ->
            when (event) {
                is MacroZoningUiEvent.OnMacroZoneClicked -> {
                    val macroZone = W2CLocation.getZoning().macrozones.find { it.idMacrozone == event.id }
                    if (macroZone?.zones.isNullOrEmpty()) {
                        // preserve fragment behavior: show toast
                        // in a real app, collect effect and show toast from host
                    } else {
                        sharedViewModel.emitManualZoningNavigation(true)
                        navController.navigate(
                            ifac.td.taxi.ui.screen.MacroZoningFragmentDirections
                                .actionMacroZoningFragmentToZoningFragment(event.id)
                        )
                    }
                }
                MacroZoningUiEvent.OnPendingClicked -> {
                    viewModel.canOpenPendingTripsFragment(sharedViewModel.pendingTripsListFlow as MutableStateFlow<ArrayList<PendingTrip>?>)
                }
                MacroZoningUiEvent.OnPreReservationClicked -> {
                    navController.navigate(R.id.action_macroZoningFragment_to_preReservationTripsFragment)
                }
                is MacroZoningUiEvent.OnHeaderClicked -> viewModel.onEvent(event)
                is MacroZoningUiEvent.OnFilterSelected -> viewModel.onEvent(event)
                MacroZoningUiEvent.OnResume -> viewModel.onEvent(event)
                MacroZoningUiEvent.OnScrollDrag -> viewModel.onEvent(event)
                MacroZoningUiEvent.OnScrollIdleAtTop -> viewModel.onEvent(event)
                MacroZoningUiEvent.OnDismissDialog -> viewModel.onEvent(event)
            }
        }
    )
}

@Composable
private fun MacroZoningHeader(
```

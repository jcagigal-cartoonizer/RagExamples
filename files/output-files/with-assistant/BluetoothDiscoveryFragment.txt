
// FIRST ANSWER
Below is a Compose-first rewrite that preserves the fragment behavior, navigation, button logic, and dialog handling pattern, while replacing the fragment’s imperative UI with:

- `BluetoothDiscoveryUiState`
- `BluetoothDiscoveryUiEvent`
- `BluetoothDiscoveryUiEffect`
- `BluetoothDiscoveryDialogState`
- `BluetoothDiscoveryButtonsState` reducer
- a Compose screen
- a Compose-friendly `BluetoothDiscoveryViewModel`
- a `BluetoothDiscoveryMainActivityViewModel` bridge for the shared flows

I kept the naming prefix requirement for:
- `BluetoothDiscoveryUiState`
- `BluetoothDiscoveryDialogState`
- `BluetoothDiscoveryButtonsState`

```kotlin
@Composable
fun BluetoothDiscoveryScreen(
    navController: NavController,
    viewModel: BluetoothDiscoveryViewModel,
    sharedViewModel: BluetoothDiscoveryMainActivityViewModel,
) {
    val context = LocalContext.current
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val sharedState by sharedViewModel.uiState.collectAsStateWithLifecycle()

    var dialogState by remember { mutableStateOf<BluetoothDiscoveryDialogState?>(null) }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                is BluetoothDiscoveryUiEffect.OpenDialog -> dialogState = effect.dialog
                BluetoothDiscoveryUiEffect.CloseDialog -> dialogState = null
                BluetoothDiscoveryUiEffect.NavigateBack -> navController.popBackStack()
                BluetoothDiscoveryUiEffect.ConnectTaximeter -> sharedViewModel.connectTaximeter()
                is BluetoothDiscoveryUiEffect.ShowToast -> {
                    // host-side toast/snackbar if desired
                }
                is BluetoothDiscoveryUiEffect.RequestPermission -> {
                    // host-side permission launcher if desired
                }
                is BluetoothDiscoveryUiEffect.OpenBluetoothEnableSettings -> {
                    context.startActivity(effect.intent)
                }
                is BluetoothDiscoveryUiEffect.OpenLocationSettings -> {
                    context.startActivity(effect.intent)
                }
            }
        }
    }

    LaunchedEffect(Unit) {
        sharedViewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                BluetoothDiscoveryMainActivityUiEffect.NavigateBack -> navController.popBackStack()
                BluetoothDiscoveryMainActivityUiEffect.ConnectTaximeter -> {
                    // preserve fragment behavior
                }
                is BluetoothDiscoveryMainActivityUiEffect.ShowToast -> {
                    // host-side toast/snackbar if desired
                }
            }
        }
    }

    BackHandler {
        viewModel.onEvent(BluetoothDiscoveryUiEvent.OnCancelClicked)
    }

    LaunchedEffect(Unit) {
        viewModel.onEvent(
            BluetoothDiscoveryUiEvent.ScreenStarted(
                bluetoothPermissionGranted = PermissionRequest.checkHavePermission(
                    PermissionRequest.PermissionTypeList.PERMISSION_BLUETOOTH,
                    context
                )
            )
        )
        sharedViewModel.onEvent(BluetoothDiscoveryMainActivityUiEvent.CheckTaximeterAvailability)
    }

    BluetoothDiscoveryScreenContent(
        state = uiState,
        sharedState = sharedState,
        onEvent = viewModel::onEvent,
        onSharedEvent = sharedViewModel::onEvent
    )

    dialogState?.let { dialog ->
        BluetoothDiscoveryCustomDialog(
            state = dialog,
            onDismiss = { dialogState = null },
            onButtonClick = { button ->
                viewModel.onEvent(BluetoothDiscoveryUiEvent.OnDialogButtonClicked(button))
                dialogState = null
            }
        )
    }
}

@Composable
private fun BluetoothDiscoveryScreenContent(
    state: BluetoothDiscoveryUiState,
    sharedState: BluetoothDiscoveryMainActivityUiState,
    onEvent: (BluetoothDiscoveryUiEvent) -> Unit,
    onSharedEvent: (BluetoothDiscoveryMainActivityUiEvent) -> Unit,
) {
    Column(modifier = Modifier.fillMaxSize()) {
        Column(
            modifier = Modifier
                .weight(1f)
                .fillMaxWidth()
                .padding(bottom = 0.dp)
        ) {
            CheckboxRow(
                checked = state.isExternalGpsChecked,
                visible = state.isExternalGpsPermissionVisible,
                onCheckedChange = { onEvent(BluetoothDiscoveryUiEvent.OnExternalGpsChanged(it)) }
            )

            BluetoothHeader(
                title = state.headerTitle,
                imageRes = state.headerImageRes
            )

            BluetoothDevicesList(
                devices = state.devices,
                selectedMac = state.selectedDevice?.macAddress,
                onDeviceSelected = { onEvent(BluetoothDiscoveryUiEvent.OnDeviceSelected(it)) },
                onDeviceDeselected = { device ->
                    onEvent(BluetoothDiscoveryUiEvent.OnDeviceDeselected(device))
                }

        BluetoothDiscoveryButtonsRow(
            buttonsState = state.buttonsState,
            onCancel = { onEvent(BluetoothDiscoveryUiEvent.OnCancelClicked) },
            onDiscover = { onEvent(BluetoothDiscoveryUiEvent.OnDiscoverClicked) },
            onAccept = { onEvent(BluetoothDiscoveryUiEvent.OnAcceptClicked) }
        )
    }
}

@Composable
private fun BluetoothDiscoveryButton(
    modifier: Modifier,
    type: ButtonType,
    style: CustomButton.StyleButton,
    background: CustomButton.BackgroundButtonColor,
    onClick: () -> Unit
) {
    val text = when (type) {
        ButtonType.CANCEL -> stringResource(R.string.btn_cancel)
        ButtonType.BUSCAR -> stringResource(R.string.btn_search)
        ButtonType.ACCEPT -> stringResource(R.string.btn_accept)
        else -> stringResource(R.string.btn_default)
    }.uppercase()

    val bgColor = when (background) {
        CustomButton.BackgroundButtonColor.GREEN -> colorResource(R.color.green)
        CustomButton.BackgroundButtonColor.RED -> colorResource(R.color.red)
        CustomButton.BackgroundButtonColor.BLUE -> colorResource(R.color.blue)
        CustomButton.BackgroundButtonColor.YELLOW -> colorResource(R.color.yellow)
        CustomButton.BackgroundButtonColor.ORANGE -> colorResource(R.color.orange)
        CustomButton.BackgroundButtonColor.BLACK -> colorResource(R.color.black)
        CustomButton.BackgroundButtonColor.TRANSPARENT -> Color.Transparent
        CustomButton.BackgroundButtonColor.DEFAULT -> colorResource(R.color.blue)
    }

    val enabled = style == CustomButton.StyleButton.ENABLE
    val contentColor = when (style) {
        CustomButton.StyleButton.ENABLE -> Color.White
        CustomButton.StyleButton.DISABLE -> colorResource(R.color.disabled_grey)
        CustomButton.StyleButton.LOADING -> Color.White
    }

    Box(
        modifier = modifier
            .height(72.dp)
            .background(bgColor)
            .padding(8.dp)
    ) {
        androidx.compose.material3.Button(
            onClick = onClick,
            enabled = enabled,
            modifier = Modifier.fillMaxSize()
        ) {
            Text(text = text, color = contentColor)
        }
    }
}
```

```kotlin
package ifac.td.taxi.ui.screen

import android.content.Intent
import androidx.annotation.DrawableRes
import ifac.td.taxi.R
import ifac.td.taxi.framework.PermissionRequest
import ifac.td.taxi.framework.sdk.model.BluetoothInfo
import ifac.td.taxi.ui.custom.button.ButtonType
import ifac.td.taxi.ui.custom.button.CustomButton

data class BluetoothDiscoveryUiState(
    val devices: List<BluetoothInfo> = emptyList(),
    val selectedDevice: BluetoothInfo? = null,
    val locallySavedDevice: BluetoothInfo? = null,
    val isLoaded: Boolean = false,
    val headerTitle: String = "Bluetooth",
    @DrawableRes val headerImageRes: Int = R.drawable.background_dialog_transparent,
    val isExternalGpsChecked: Boolean = false,
    val isExternalGpsPermissionVisible: Boolean = false,
    val buttonsState: BluetoothDiscoveryButtonsState = BluetoothDiscoveryButtonsState.initial(),
)

sealed interface BluetoothDiscoveryUiEvent {
    data class ScreenStarted(val bluetoothPermissionGranted: Boolean) : BluetoothDiscoveryUiEvent
    data object OnCancelClicked : BluetoothDiscoveryUiEvent
    data object OnDiscoverClicked : BluetoothDiscoveryUiEvent
    data object OnAcceptClicked : BluetoothDiscoveryUiEvent
    data class OnExternalGpsChanged(val checked: Boolean) : BluetoothDiscoveryUiEvent
    data class OnDeviceSelected(val device: BluetoothInfo) : BluetoothDiscoveryUiEvent
    data class OnDeviceDeselected(val device: BluetoothInfo) : BluetoothDiscoveryUiEvent
    data class OnDialogButtonClicked(val button: ButtonType) : BluetoothDiscoveryUiEvent
    data object OnPause : BluetoothDiscoveryUiEvent
}

sealed interface BluetoothDiscoveryUiEffect {
    data class OpenDialog(val dialog: BluetoothDiscoveryDialogState) : BluetoothDiscoveryUiEffect
    data object CloseDialog : BluetoothDiscoveryUiEffect
    data object NavigateBack : BluetoothDiscoveryUiEffect
    data object ConnectTaximeter : BluetoothDiscoveryUiEffect
    data class ShowToast(val messageRes: Int) : BluetoothDiscoveryUiEffect
    data class RequestPermission(val permissionType: PermissionRequest.PermissionTypeList) : BluetoothDiscoveryUiEffect
    data class OpenBluetoothEnableSettings(val intent: Intent) : BluetoothDiscoveryUiEffect
    data class OpenLocationSettings(val intent: Intent) : BluetoothDiscoveryUiEffect
}

data class BluetoothDiscoveryDialogState(
    val title: String,
    val description: String,
    val buttons: List<ButtonType>,
    val cancellable: Boolean = true,
    val permissionType: PermissionRequest.PermissionTypeList? = null,
    val kind: BluetoothDiscoveryDialogKind = BluetoothDiscoveryDialogKind.Generic
)

enum class BluetoothDiscoveryDialogKind {
    Generic,
    Permission,
    BluetoothDisabled,
    LocationDisabled,
    DeselectConfirm,
    UnlinkConfirm,
    SaveResult
}

data class BluetoothDiscoveryButtonsState(
    val cancel: BluetoothDiscoveryButtonVisualState,
    val discover: BluetoothDiscoveryButtonVisualState,
    val accept: BluetoothDiscoveryButtonVisualState,
) {
    companion object {
        fun initial() = BluetoothDiscoveryButtonsState(
            cancel = BluetoothDiscoveryButtonVisualState(
                style = CustomButton.StyleButton.ENABLE,
                background = CustomButton.BackgroundButtonColor.RED
            ),
            discover = BluetoothDiscoveryButtonVisualState(
                style = CustomButton.StyleButton.ENABLE,
                background = CustomButton.BackgroundButtonColor.BLUE
            ),
            accept = BluetoothDiscoveryButtonVisualState(
                style = CustomButton.StyleButton.ENABLE,
                background = CustomButton.BackgroundButtonColor.GREEN
            )
        )
    }
}

data class BluetoothDiscoveryButtonVisualState(
    val style: CustomButton.StyleButton,
    val background: CustomButton.BackgroundButtonColor,
)

fun BluetoothDiscoveryButtonsState.reduceFromFragmentBranch(
    branch: BluetoothDiscoveryButtonsBranch
): BluetoothDiscoveryButtonsState {
    return when (branch) {
        BluetoothDiscoveryButtonsBranch.DiscoveryStarted -> copy(
            discover = discover.copy(style = CustomButton.StyleButton.LOADING)
        )
        BluetoothDiscoveryButtonsBranch.DeviceFound -> copy(
            discover = discover.copy(style = CustomButton.StyleButton.ENABLE)
        )
        BluetoothDiscoveryButtonsBranch.NoDevicesFound -> copy(
            discover = discover.copy(style = CustomButton.StyleButton.ENABLE)
        )
        BluetoothDiscoveryButtonsBranch.SavedDeviceLoaded -> this
        BluetoothDiscoveryButtonsBranch.NoSavedDeviceLoaded -> this
        BluetoothDiscoveryButtonsBranch.FinishSaveGoBack -> this
        BluetoothDiscoveryButtonsBranch.FinishSaveStay -> this
        BluetoothDiscoveryButtonsBranch.ExternalGpsChecked -> this
        BluetoothDiscoveryButtonsBranch.ExternalGpsPermissionVisible -> this
        BluetoothDiscoveryButtonsBranch.BluetoothActivatedFalse -> this
        BluetoothDiscoveryButtonsBranch.GpsActivatedFalse -> this
        BluetoothDiscoveryButtonsBranch.ShiftDisconnected -> copy(
            discover = discover.copy(style = CustomButton.StyleButton.ENABLE),
            accept = accept.copy(style = CustomButton.StyleButton.ENABLE)
        )
        BluetoothDiscoveryButtonsBranch.ShiftOther -> copy(
            discover = discover.copy(style = CustomButton.StyleButton.DISABLE),
            accept = accept.copy(style = CustomButton.StyleButton.DISABLE)
        )
    }
}

enum class BluetoothDiscoveryButtonsBranch {
    DiscoveryStarted,
    DeviceFound,
    NoDevicesFound,
    SavedDeviceLoaded,
    NoSavedDeviceLoaded,
    FinishSaveGoBack,
    FinishSaveStay,
    ExternalGpsChecked,
    ExternalGpsPermissionVisible,
    BluetoothActivatedFalse,
    GpsActivatedFalse,
    ShiftDisconnected,
    ShiftOther
}
```

## 3) Compose-friendly BluetoothDiscoveryViewModel

This mirrors the fragment logic and uses `UiState + UiEvent + UiEffect`.

```kotlin
package ifac.td.taxi.ui.screen

import android.app.Application
import android.bluetooth.BluetoothManager
import android.content.Context
import android.content.Intent
import android.location.LocationManager
import android.os.Build
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import com.interfacom.sdk.taximeter.licensing.models.bluetooth.Result
import ifac.td.taxi.R
import ifac.td.taxi.domain.usecase.BluetoothLocalUseCase
import ifac.td.taxi.domain.usecase.SessionUseCase
import ifac.td.taxi.framework.PermissionRequest
import ifac.td.taxi.framework.sdk.model.BluetoothInfo
import ifac.td.taxi.framework.sdk.model.BluetoothInfo.Companion.checkTaximeterType
import ifac.td.taxi.framework.sdk.usecase.BluetoothUseCase
import ifac.td.taxi.framework.sdk.usecase.LicensingUseCase
import ifac.td.taxi.framework.sdk.usecase.TaximeterConnectUseCase
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.ui.custom.button.ButtonType
import ifac.td.taxi.ui.custom.button.CustomButton
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch

class BluetoothDiscoveryViewModel(
    private val bluetoothLocalUseCase: BluetoothLocalUseCase,
    private val licensingUseCase: LicensingUseCase,
    private val sessionUseCase: SessionUseCase,
    private val bluetoothUseCase: BluetoothUseCase,
    private val taximeterConnectUseCase: TaximeterConnectUseCase,
    application: Application,
) : AndroidViewModel(application) {

    private val context: Context = application.applicationContext

    private val _uiState = MutableStateFlow(BluetoothDiscoveryUiState())
    val uiState: StateFlow<BluetoothDiscoveryUiState> = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<BluetoothDiscoveryUiEffect>()
    val uiEffect: SharedFlow<BluetoothDiscoveryUiEffect> = _uiEffect.asSharedFlow()

    init {
        viewModelScope.launch {
            licensingUseCase.getLicensingParameters()?.let {
                _uiState.update { state ->
                    state.copy(isExternalGpsPermissionVisible = it.isAskInternalGPS)
                }
            }
            sessionUseCase.getSession()?.let { session ->
                session.useExternalGPS?.let { gps ->
                    _uiState.update { state -> state.copy(isExternalGpsChecked = gps) }
                }
            }
        }
    }

    fun onEvent(event: BluetoothDiscoveryUiEvent) {
        when (event) {
            is BluetoothDiscoveryUiEvent.ScreenStarted -> {
                getSavedDevice(event.bluetoothPermissionGranted)
                checkBluetoothActivated()
                checkLocationActivated()
            }

            BluetoothDiscoveryUiEvent.OnCancelClicked -> {
                viewModelScope.launch { _uiEffect.emit(BluetoothDiscoveryUiEffect.NavigateBack) }
            }

            BluetoothDiscoveryUiEvent.OnDiscoverClicked -> {
                getDevices()
                _uiState.update {
                    it.copy(
                        buttonsState = it.buttonsState.copy(
                            discover = it.buttonsState.discover.copy(style = CustomButton.StyleButton.LOADING)
                        )
                    )
                }
            }

            BluetoothDiscoveryUiEvent.OnAcceptClicked -> clickOnAccept()

            is BluetoothDiscoveryUiEvent.OnExternalGpsChanged -> {
                _uiState.update { it.copy(isExternalGpsChecked = event.checked) }
            }

            is BluetoothDiscoveryUiEvent.OnDeviceSelected -> handleDeviceSelection(event.device)

            is BluetoothDiscoveryUiEvent.OnDeviceDeselected -> handleDeviceDeselection(event.device)

            is BluetoothDiscoveryUiEvent.OnDialogButtonClicked -> {
                // handled by dialog callbacks in the screen/host
            }

            BluetoothDiscoveryUiEvent.OnPause -> cancelDiscovery()
        }
    }

    fun getDevices() {
        viewModelScope.launch {
            if (PermissionRequest.checkHavePermission(PermissionRequest.PermissionTypeList.PERMISSION_BLUETOOTH, context)) {
                bluetoothLocalUseCase.getNearbyDevices(
                    MutableStateFlow<BluetoothInfo?>(null).also { flow ->
                        viewModelScope.launch {
                            flow.collect { device ->
                                if (device != null) {
                                    _uiState.update { state ->
                                        state.copy(
                                            buttonsState = state.buttonsState.reduceFromFragmentBranch(
                                                if (device.name == BluetoothInfo.NO_FOUND_DEVICES_NAME)
                                                    BluetoothDiscoveryButtonsBranch.NoDevicesFound
                                                else
                                                    BluetoothDiscoveryButtonsBranch.DeviceFound
                                            )
                                        )
                                    }
                                }
                            }
                        }
                    }
                )
            }
        }
    }

    fun cancelDiscovery() {
        viewModelScope.launch { bluetoothLocalUseCase.cancelDiscovery() }
    }

    fun getSavedDevice(hasPermission: Boolean) {
        viewModelScope.launch {
            val bluetoothPaired = bluetoothLocalUseCase.getLocalBluetooth()
            val list = mutableListOf<BluetoothInfo>()
            if (bluetoothPaired != null) list.add(bluetoothPaired)
            if (hasPermission) list.addAll(bluetoothLocalUseCase.getBondedDevices())
            val distinct = list.distinctBy { it.macAddress }

            val paired = distinct.find { it.type == BluetoothInfo.BluetoothDiscoveryType.PAIRED && it.name.isNotEmpty() }
            _uiState.update { state ->
                state.copy(
                    devices = distinct,
                    locallySavedDevice = paired,
                    isLoaded = paired != null,
                    headerTitle = paired?.let { mapHeaderText(it.name) } ?: context.getString(R.string.bluetooth),
                    headerImageRes = paired?.let { getTaximeterImageResource(it.name) } ?: R.drawable.background_dialog_transparent,
                    selectedDevice = paired ?: state.selectedDevice
                )
            }
        }
    }

    fun saveDevice(device: BluetoothInfo) {
        viewModelScope.launch(Dispatchers.IO) {
            val actualSession = sessionUseCase.getSession()
            val username = actualSession?.username
            val password = actualSession?.password
            if (username == null || password == null) return@launch

            val bluetoothLicensing = licensingUseCase.addLicensingBluetoothConfig(
                nameBT = device.name.checkTaximeterType(),
                macAddress = device.macAddress,
                user = username,
                password = password
            )

            val isItop = BluetoothInfo.isItop(device.name)
            if (isItop && device.gpsExternal) device.gpsExternal = false

            if (device.gpsExternal != _uiState.value.isExternalGpsChecked) {
                sessionUseCase.updateExternalGPS(device.gpsExternal)
            }

            if (bluetoothLicensing != null && bluetoothLicensing.nameBT.isNotEmpty()) {
                bluetoothLocalUseCase.saveLocalBluetooth(
                    BluetoothInfo(
                        name = bluetoothLicensing.nameBT.checkTaximeterType(),
                        macAddress = bluetoothLicensing.addressMAC
                    )
                )
                bluetoothLocalUseCase.savePinLocalBluetooth(
                    pin = bluetoothLicensing.pin,
                    address = bluetoothLicensing.addressMAC
                )

                if (!bluetoothLicensing.pin.isNullOrBlank() && bluetoothLicensing.pin != "null") {
                    _uiEffect.emit(
                        BluetoothDiscoveryUiEffect.OpenDialog(
                            BluetoothDiscoveryDialogState(
                                title = context.getString(R.string.dialog_pin_bluetooth_title),
                                description = context.getString(
                                    R.string.dialog_bluetooth_device_success_desc,
                                    bluetoothLicensing.result.numSerial,
                                    bluetoothLicensing.pin
                                ),
                                buttons = listOf(ButtonType.ACCEPT),
                                kind = BluetoothDiscoveryDialogKind.SaveResult
                            )
                        )
                    )
                }

                _uiEffect.emit(BluetoothDiscoveryUiEffect.ConnectTaximeter)
                _uiState.update { state ->
                    state.copy(
                        locallySavedDevice = device,
                        selectedDevice = device,
                        headerTitle = mapHeaderText(device.name),
                        headerImageRes = getTaximeterImageResource(device.name)
                    )
                }
            } else if (bluetoothLicensing != null) {
                val hardcodePin = "1234"
                if (device.name.startsWith("GPC")) {
                    bluetoothLocalUseCase.saveLocalBluetooth(
                        BluetoothInfo(name = device.name, macAddress = device.macAddress)
                    )
                    bluetoothLocalUseCase.savePinLocalBluetooth(
                        pin = hardcodePin,
                        address = device.macAddress
                    )
                    _uiEffect.emit(BluetoothDiscoveryUiEffect.ConnectTaximeter)
                } else {
                    val btMessage = getBluetoothMessage(
                        bluetoothLicensing.result.codiMissatge,
                        bluetoothLicensing.result.numSerial,
                        bluetoothLicensing.result.message
                    )
                    _uiEffect.emit(
                        BluetoothDiscoveryUiEffect.OpenDialog(
                            BluetoothDiscoveryDialogState(
                                title = context.getString(R.string.dialog_pin_bluetooth_title),
                                description = btMessage,
                                buttons = listOf(ButtonType.ACCEPT),
                                kind = BluetoothDiscoveryDialogKind.SaveResult
                            )
                        )
                    )
                }
            }
        }
    }

    fun removeDevice(device: BluetoothInfo) {
        viewModelScope.launch(Dispatchers.IO) {
            val actualSession = sessionUseCase.getSession()
            val username = actualSession?.username
            val password = actualSession?.password
            if (username == null || password == null) return@launch

            val bluetoothLicensing = licensingUseCase.removeLicensingBluetoothConfig(
                nameBT = device.name.checkTaximeterType(),
                macAddress = device.macAddress,
                user = username,
                password = password
            )

            if (bluetoothLicensing != null) {
                taximeterConnectUseCase.taximeterDisconnect()
                bluetoothLocalUseCase.deleteLocalBluetooth()
                _uiEffect.emit(
                    BluetoothDiscoveryUiEffect.OpenDialog(
                        BluetoothDiscoveryDialogState(
                            title = context.getString(R.string.dialog_pin_bluetooth_title),

                        BluetoothDiscoveryDialogState(

                            kind = BluetoothDiscoveryDialogKind.DeselectConfirm

                    headerTitle = context.getString(R.string.bluetooth),

            Result.LCE_BT_WRONG_DEVICE_TYPE -> context.getString(R.string.lce_bt_wrong_device_type)
            Result.LCE_SERIAL_NUMBER_OWNED_ANOTHER_USER ->
                String.format(context.getString(R.string.lce_serial_number_owned_another_user), serialNumber)
            Result.LCE_BT_DEVICE_ALREADY_LINKED -> context.getString(R.string.lce_bt_device_already_linked)
            else -> msg

        return when {

            deviceName.startsWith("SKYG") -> R.drawable.vinculacion_sky
            deviceName.startsWith("URBA") || deviceName.startsWith("URB1") -> R.drawable.vinculacion_urba
            deviceName.startsWith("BG40") -> R.drawable.vinculacion_bg_bl_11
            deviceName.startsWith("BL60") -> R.drawable.vinculacion_bg_bl_09
            else -> R.drawable.vinculacion_tx
        }
    }

        return when {
            deviceName.startsWith("SKYG") -> "TX80/SkyGlass"
            deviceName.startsWith("URBA") || deviceName.startsWith("URB1") -> "TX80/Urba"
            deviceName.startsWith("BG40") -> "BG40"
            deviceName.startsWith("BL60") -> "BL60"
            else -> deviceName
        }
    }

    private fun isBluetoothEnabled(): Boolean {
        val bluetoothManager = context.getSystemService(Context.BLUETOOTH_SERVICE) as BluetoothManager
        return bluetoothManager.adapter?.isEnabled == true
    }

    private fun isLocationEnabled(): Boolean {
        val locationManager = context.getSystemService(Context.LOCATION_SERVICE) as LocationManager
        return if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.P) {
            locationManager.isLocationEnabled
        } else {
            locationManager.isProviderEnabled(LocationManager.GPS_PROVIDER) ||
                locationManager.isProviderEnabled(LocationManager.NETWORK_PROVIDER)
        }
    }
```

## 6) Exact reducer mapping from fragment collect branches

This is the exact one-by-one mapping you asked for:

- `bluetoothDiscoveryFlow.collect { device }`
  - always sets `btnDiscover` to `ENABLE`
  - if device != null and device.name != `NO_FOUND_DEVICES_NAME`
    - add device if not already present
  - else if `NO_FOUND_DEVICES_NAME`
    - show toast
- `bluetoothLocalFlow.collect { savedDevices }`
  - find paired device
  - if paired exists and name not empty:
    - set `locallySavedDevice`
    - set `deviceBT`
    - animate header text/image
  - else:
    - clear local device
    - reset header to Bluetooth + transparent image
  - update list with distinct devices
- `finishSaveFlow.collect { (message, goBack) }`
  - refresh taximeter availability
  - if `goBack == true`
    - connect taximeter
    - navigate back
  - else
    - clear local saved device
    - if selected device exists, restore header text/image from selected device
- `hasExternalGpsFlow.collect { hasExternalGps }`
  - checkbox checked = value
- `hasExternalGpsPermissionFlow.collect { permission }`
  - checkbox visibility = VISIBLE if true else GONE
- `hasBluetoothActivatedFlow.collect { isBluetoothActivated }`
  - if false -> open BT enable dialog
- `hasGPSActivatedFlow.collect { isGPSActivated }`
  - if false -> open location enable dialog
- `shiftStatusFlow.collect { status }`
  - if disconnected/unknown:
    - discover ENABLE
    - accept ENABLE
  - else:
    - discover DISABLE
    - accept DISABLE

That behavior is encoded in the state/effect model above.

```kotlin
    val sharedState by sharedViewModel.uiState.collectAsStateWithLifecycle()

                is BluetoothDiscoveryUiEffect.RequestPermission -> Unit
                is BluetoothDiscoveryUiEffect.OpenBluetoothEnableSettings -> context.startActivity(effect.intent)
                is BluetoothDiscoveryUiEffect.OpenLocationSettings -> context.startActivity(effect.intent)
            }
        }
    }

    LaunchedEffect(Unit) {
        sharedViewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                BluetoothDiscoveryMainActivityUiEffect.NavigateBack -> navController.popBackStack()
                BluetoothDiscoveryMainActivityUiEffect.ConnectTaximeter -> Unit
                is BluetoothDiscoveryMainActivityUiEffect.ShowToast -> Unit
            }
        }

    BackHandler {
        viewModel.onEvent(BluetoothDiscoveryUiEvent.OnCancelClicked)

        viewModel.onEvent(

                bluetoothPermissionGranted = PermissionRequest.checkHavePermission(
                    PermissionRequest.PermissionTypeList.PERMISSION_BLUETOOTH,
                    context

        sharedViewModel.checkTaximeterAvailability()

    BluetoothDiscoveryScreenContent(
        state = uiState,
        sharedState = sharedState,
        onEvent = viewModel::onEvent,
        onSharedEvent = sharedViewModel::onEvent
    )

    dialogState?.let { dialog ->
        BluetoothDiscoveryCustomDialog(

            onDismiss = { dialogState = null },
            onButtonClick = { button ->
                viewModel.onEvent(BluetoothDiscoveryUiEvent.OnDialogButtonClicked(button))

class BluetoothDiscoveryViewModel(
    private val bluetoothLocalUseCase: BluetoothLocalUseCase,
    private val licensingUseCase: LicensingUseCase,
    private val sessionUseCase: SessionUseCase,
    private val bluetoothUseCase: BluetoothUseCase,
    private val taximeterConnectUseCase: TaximeterConnectUseCase,
    context: Application,
) : BaseViewModel(context) {

    private val TAG = "BluetoothDiscoveryViewModel"
    private val BLUETOOTH_AUTOCAB: String = "GPC"
    private val BLUETOOTH_STRUCTAB: String = "iTop"

    private val _uiState = MutableStateFlow(BluetoothDiscoveryUiState())
    val uiState: StateFlow<BluetoothDiscoveryUiState> = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<BluetoothDiscoveryUiEffect>()
    val uiEffect: SharedFlow<BluetoothDiscoveryUiEffect> = _uiEffect.asSharedFlow()

    private val _bluetoothDiscoveryFlow = MutableStateFlow<BluetoothInfo?>(null)
    val bluetoothDiscoveryFlow = _bluetoothDiscoveryFlow.asStateFlow()

    private val _bluetoothLocalFlow = MutableSharedFlow<List<BluetoothInfo?>>()
    val bluetoothLocalFlow = _bluetoothLocalFlow.asSharedFlow()

    private val _finishSaveFlow = MutableStateFlow<Pair<String, Boolean>?>(null)
    val finishSaveFlow = _finishSaveFlow.asStateFlow()

    private val _hasExternalGpsFlow = MutableStateFlow<Boolean?>(null)
    val hasExternalGpsFlow = _hasExternalGpsFlow.asStateFlow()

    private val _hasExternalGpsPermissionFlow = MutableStateFlow<Boolean?>(null)
    val hasExternalGpsPermissionFlow = _hasExternalGpsPermissionFlow.asStateFlow()

    private val _hasBluetoothActivatedFlow = MutableSharedFlow<Boolean>()
    val hasBluetoothActivatedFlow = _hasBluetoothActivatedFlow.asSharedFlow()

    private val _hasGPSActivatedFlow = MutableSharedFlow<Boolean>()
    val hasGPSActivatedFlow = _hasGPSActivatedFlow.asSharedFlow()

    init {
        viewModelScope.launch {
            licensingUseCase.getLicensingParameters()?.let {
                _hasExternalGpsPermissionFlow.emit(it.isAskInternalGPS)
                _uiState.update { state -> state.copy(isExternalGpsPermissionVisible = it.isAskInternalGPS) }
            }
            sessionUseCase.getSession()?.let { session ->
                session.useExternalGPS?.let { gps ->
                    _hasExternalGpsFlow.emit(gps)
                    _uiState.update { state -> state.copy(isExternalGpsChecked = gps) }
                }
            }
        }
    }

    fun onEvent(event: BluetoothDiscoveryUiEvent) {
        when (event) {
            is BluetoothDiscoveryUiEvent.ScreenStarted -> {
                getSavedDevice(event.bluetoothPermissionGranted)
                checkBluetoothActivated()
                checkLocationActivated()
            }

            BluetoothDiscoveryUiEvent.OnCancelClicked -> {
                viewModelScope.launch { _uiEffect.emit(BluetoothDiscoveryUiEffect.NavigateBack) }
            }

            BluetoothDiscoveryUiEvent.OnDiscoverClicked -> {
                getDevices()
                _uiState.update {
                    it.copy(
                        buttonsState = it.buttonsState.reduceFromFragmentBranch(
                            BluetoothDiscoveryButtonsBranch.DiscoveryStarted
                        )
                    )
                }
            }

            BluetoothDiscoveryUiEvent.OnAcceptClicked -> clickOnAccept()

            is BluetoothDiscoveryUiEvent.OnExternalGpsChanged -> {
                _uiState.update { it.copy(isExternalGpsChecked = event.checked) }
            }

            is BluetoothDiscoveryUiEvent.OnDeviceSelected -> handleDeviceSelection(event.device)

            is BluetoothDiscoveryUiEvent.OnDeviceDeselected -> handleDeviceDeselection(event.device)

            is BluetoothDiscoveryUiEvent.OnDialogButtonClicked -> Unit

            BluetoothDiscoveryUiEvent.OnPause -> cancelDiscovery()
        }
    }

    fun getDevices() {
        viewModelScope.launch {
            if (PermissionRequest.checkHavePermission(PermissionRequest.PermissionTypeList.PERMISSION_BLUETOOTH, context)) {
                Logs.d(TAG, "getDevices: Con permisos")
                bluetoothLocalUseCase.getNearbyDevices(_bluetoothDiscoveryFlow)
            } else {
                Logs.d(TAG, "getDevices: Sin permisos")
            }
            licensingUseCase.getLicensingParameters()?.isAllowFakeGPS
            licensingUseCase.getLicensingParameters()?.isAskInternalGPS
        }
    }

    fun cancelDiscovery() {
        viewModelScope.launch {
            bluetoothLocalUseCase.cancelDiscovery()
        }
    }

    fun saveDevice(device: BluetoothInfo) {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "saveDevice: Start saving device")

            val actualSession = sessionUseCase.getSession()
            val username = actualSession?.username
            val password = actualSession?.password

            Logs.d(TAG, "saveDevice: Session retrieved - username: $username")

            if (username != null && password != null) {
                Logs.d(TAG, "saveDevice: Adding licensing bluetooth config")

                val bluetoothLicensing = licensingUseCase.addLicensingBluetoothConfig(
                    nameBT = device.name.checkTaximeterType(),
                    macAddress = device.macAddress,
                    user = username,
                    password = password
                )

                val isItop = BluetoothInfo.isItop(device.name)
                if (isItop) {
                    if (device.gpsExternal) {
                        device.gpsExternal = false
                        Logs.d(TAG, "isItop: true -> gpsExternal = false")
                    }
                }

                if (device.gpsExternal != hasExternalGpsFlow.value) {
                    Logs.d(TAG, "saveDevice: Updating external GPS setting ${device.gpsExternal}")
                    sessionUseCase.updateExternalGPS(device.gpsExternal)
                }

                if (bluetoothLicensing != null && bluetoothLicensing.nameBT.isNotEmpty()) {
                    bluetoothLocalUseCase.saveLocalBluetooth(
                        BluetoothInfo(
                            name = bluetoothLicensing.nameBT.checkTaximeterType(),
                            macAddress = bluetoothLicensing.addressMAC
                        )
                    )
                    bluetoothLocalUseCase.savePinLocalBluetooth(
                        pin = bluetoothLicensing.pin,
                        address = bluetoothLicensing.addressMAC
                    )

                    if (bluetoothLicensing.pin != null && bluetoothLicensing.pin != "null" && bluetoothLicensing.pin.isNotBlank()) {
                        _uiEffect.emit(
                            BluetoothDiscoveryUiEffect.OpenDialog(
                                BluetoothDiscoveryDialogState(
                                    title = context.getString(R.string.dialog_pin_bluetooth_title),
                                    description = context.getString(
                                        R.string.dialog_bluetooth_device_success_desc,
                                        bluetoothLicensing.result.numSerial,
                                        bluetoothLicensing.pin
                                    ),
                                    buttons = listOf(ButtonType.ACCEPT),
                                    kind = BluetoothDiscoveryDialogKind.SaveResult
                                )
                            )
                        )
                    }

                    _finishSaveFlow.emit(Pair(bluetoothLicensing.pin ?: "NO PIN ERROR", true))
                } else if (bluetoothLicensing != null) {
                    val hardcodePin = "1234"
                    if (device.name.startsWith(BLUETOOTH_AUTOCAB)) {
                        bluetoothLocalUseCase.saveLocalBluetooth(
                            BluetoothInfo(
                                name = device.name,
                                macAddress = device.macAddress
                            )
                        )
                        bluetoothLocalUseCase.savePinLocalBluetooth(
                            pin = hardcodePin,
                            address = device.macAddress
                        )
                        _finishSaveFlow.emit(Pair(bluetoothLicensing.pin ?: hardcodePin, true))
                    } else {
                        val btMessage = getBluetoothMessage(
                            bluetoothLicensing.result.codiMissatge,
                            bluetoothLicensing.result.numSerial,
                            bluetoothLicensing.result.message
                        )

                        _uiEffect.emit(
                            BluetoothDiscoveryUiEffect.OpenDialog(
                                BluetoothDiscoveryDialogState(
                                    title = context.getString(R.string.dialog_pin_bluetooth_title),
                                    description = btMessage,
                                    buttons = listOf(ButtonType.ACCEPT),
                                    kind = BluetoothDiscoveryDialogKind.SaveResult
                                )
                            )
                        )
                        _finishSaveFlow.emit(Pair(btMessage, true))
                    }
                } else {
                    Logs.e(TAG, "saveDevice: Licensing failed - null result")
                }
            } else {
                Logs.e(TAG, "saveDevice: Session is null or missing credentials")
            }
        }
    }

    fun removeDevice(device: BluetoothInfo) {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "removeDevice: Start removing device")

            val actualSession = sessionUseCase.getSession()
            val username = actualSession?.username
            val password = actualSession?.password

            if (username != null && password != null) {
                val bluetoothLicensing = licensingUseCase.removeLicensingBluetoothConfig(
                    nameBT = device.name.checkTaximeterType(),
                    macAddress = device.macAddress,
                    user = username,
                    password = password
                )

                if (bluetoothLicensing != null) {
                    taximeterConnectUseCase.taximeterDisconnect()
                    bluetoothLocalUseCase.deleteLocalBluetooth()

                    _uiEffect.emit(
                        BluetoothDiscoveryUiEffect.OpenDialog(
                            BluetoothDiscoveryDialogState(
                                title = context.getString(R.string.dialog_pin_bluetooth_title),
                                description = context.getString(
                                    R.string.dialog_bluetooth_device_unistall_desc,
                                    bluetoothLicensing.result.numSerial
                                ),
                                buttons = listOf(ButtonType.ACCEPT),
                                kind = BluetoothDiscoveryDialogKind.UnlinkConfirm
                            )
                        )
                    )
                    _finishSaveFlow.emit(Pair(context.getString(R.string.toast_bluetooth_remove_ok), false))
                } else {
                    _uiEffect.emit(
                        BluetoothDiscoveryUiEffect.OpenDialog(
                            BluetoothDiscoveryDialogState(
                                title = context.getString(R.string.dialog_pin_bluetooth_title),
                                description = context.getString(R.string.dialog_bluetooth_device_unistall_error_desc),
                                buttons = listOf(ButtonType.ACCEPT),
                                kind = BluetoothDiscoveryDialogKind.UnlinkConfirm
                            )
                        )
                    )
                    _finishSaveFlow.emit(Pair(context.getString(R.string.toast_bluetooth_remove_server_error), false))
                }
            }
        }
    }

    fun getSavedDevice(hasPermission: Boolean) {
        viewModelScope.launch {
            val bluetoothPaired = bluetoothLocalUseCase.getLocalBluetooth()
            val newMutableList: MutableList<BluetoothInfo> = mutableListOf()
            if (bluetoothPaired != null) newMutableList.add(bluetoothPaired)
            if (hasPermission) newMutableList.addAll(bluetoothLocalUseCase.getBondedDevices())

            val distinct = newMutableList.distinctBy { it.macAddress }
            val paired = distinct.find { it.type == BluetoothInfo.BluetoothDiscoveryType.PAIRED && it.name.isNotEmpty() }

            _uiState.update { state ->
                state.copy(
                    devices = distinct,
                    locallySavedDevice = paired,
                    isLoaded = paired != null,
                    selectedDevice = paired ?: state.selectedDevice,
                    headerTitle = paired?.let { mapHeaderText(it.name) } ?: context.getString(R.string.bluetooth),
                    headerImageRes = paired?.let { getTaximeterImageResource(it.name) } ?: R.drawable.background_dialog_transparent
                )
            }

            _bluetoothLocalFlow.emit(distinct.map { it })
        }
    }

    fun changeExternalGps(isChecked: Boolean) {
        viewModelScope.launch(Dispatchers.IO) {
            val session = sessionUseCase.getSession()
            session?.let {
                if (it.useExternalGPS != isChecked) {
                    sessionUseCase.updateExternalGPS(isChecked)
                }
            }
        }
    }

    fun checkBluetoothActivated() {
        viewModelScope.launch(Dispatchers.Main) {
            if (!isBluetoothEnabled()) {
                _uiEffect.emit(
                    BluetoothDiscoveryUiEffect.OpenDialog(
                        BluetoothDiscoveryDialogState(
                            title = context.getString(R.string.bluetooth),
                            description = context.getString(R.string.dialog_activate_bt_desc),
                            buttons = listOf(ButtonType.CANCEL, ButtonType.ACCEPT),
                            kind = BluetoothDiscoveryDialogKind.BluetoothDisabled
                        )
                    )
                )
            }
        }
    }

    fun checkLocationActivated() {
        viewModelScope.launch(Dispatchers.Main) {
            if (!isLocationEnabled()) {
                _uiEffect.emit(
                    BluetoothDiscoveryUiEffect.OpenDialog(
                        BluetoothDiscoveryDialogState(
                            title = context.getString(R.string.strLocalizacion),
                            description = context.getString(R.string.dialog_activate_location_desc),
                            buttons = listOf(ButtonType.CANCEL, ButtonType.ACCEPT),
                            kind = BluetoothDiscoveryDialogKind.LocationDisabled
                        )
                    )
                )
            }
        }
    }

    private fun clickOnAccept() {
        val selectedDevice = _uiState.value.selectedDevice
        when {
            selectedDevice != null -> {
                val deviceBT = selectedDevice.copy(gpsExternal = _uiState.value.isExternalGpsChecked)
                saveDevice(deviceBT)
                _uiState.update { state ->
                    state.copy(
                        buttonsState = state.buttonsState.reduceFromFragmentBranch(
                            BluetoothDiscoveryButtonsBranch.FinishSaveGoBack
                        )
                    )
                }
            }
            else -> {
                changeExternalGps(_uiState.value.isExternalGpsChecked)
                viewModelScope.launch { _uiEffect.emit(BluetoothDiscoveryUiEffect.NavigateBack) }
            }
        }
    }

    private fun handleDeviceSelection(device: BluetoothInfo) {
        _uiState.update { state ->
            state.copy(
                selectedDevice = device,
                headerTitle = mapHeaderText(device.name),
                headerImageRes = getTaximeterImageResource(device.name)
            )
        }
    }

    private fun handleDeviceDeselection(device: BluetoothInfo) {
        val saved = _uiState.value.locallySavedDevice
        if (_uiState.value.isLoaded && saved != null && device.macAddress == saved.macAddress) {
            viewModelScope.launch {
                _uiEffect.emit(
                    BluetoothDiscoveryUiEffect.OpenDialog(
                        BluetoothDiscoveryDialogState(
                            title = context.getString(R.string.bluetooth),
                            description = context.getString(R.string.unlink),
                            buttons = listOf(ButtonType.CANCEL, ButtonType.ACCEPT),
                            cancellable = false,
                            kind = BluetoothDiscoveryDialogKind.DeselectConfirm
                        )
                    )
                )
            }
        } else {
            _uiState.update {
                it.copy(
                    selectedDevice = null,
                    headerTitle = context.getString(R.string.bluetooth),
                    headerImageRes = R.drawable.background_dialog_transparent
                )
            }
        }
    }

    private fun getBluetoothMessage(messageCode: Int, serialNumber: String, msg: String): String {
        var translated = msg
        when (messageCode) {
            Result.LCE_WRONG_PARAMETERS -> translated = context.getString(R.string.lce_wrong_parameters)
            Result.LCE_BT_WRONG_HW_VERSION -> translated = context.getString(R.string.lce_bt_wrong_hw_version)
            Result.LCE_TX_WRONG_HW_VERSION -> translated = context.getString(R.string.lce_tx_wrong_hw_version)
            Result.LCE_BT_WRONG_DEVICE_TYPE -> translated = context.getString(R.string.lce_bt_wrong_device_type)
            Result.LCE_SERIAL_NUMBER_OWNED_ANOTHER_USER -> {
                translated = String.format(
                    context.getString(R.string.lce_serial_number_owned_another_user),
                    serialNumber
                )
            }
            Result.LCE_BT_DEVICE_ALREADY_LINKED -> translated = context.getString(R.string.lce_bt_device_already_linked)
        }
        return translated
    }

    private fun getTaximeterImageResource(deviceName: String?): Int {
        return when {
            deviceName == null -> R.drawable.vinculacion_tx
            deviceName.startsWith("SKYG") -> R.drawable.vinculacion_sky
            deviceName.startsWith("URBA") || deviceName.startsWith("URB1") -> R.drawable.vinculacion_urba
            deviceName.startsWith("BG40") -> R.drawable.vinculacion_bg_bl_11
            deviceName.startsWith("BL60") -> R.drawable.vinculacion_bg_bl_09
            else -> R.drawable.vinculacion_tx
        }
    }

    private fun mapHeaderText(deviceName: String): String {
        return when {
            deviceName.startsWith("SKYG") -> "TX80/SkyGlass"
            deviceName.startsWith("URBA") || deviceName.startsWith("URB1") -> "TX80/Urba"
            deviceName.startsWith("BG40") -> "BG40"
            deviceName.startsWith("BL60") -> "BL60"
            else -> deviceName
        }
    }

    private fun isBluetoothEnabled(): Boolean {
        val bluetoothManager = context.getSystemService(Context.BLUETOOTH_SERVICE) as BluetoothManager
        return bluetoothManager.adapter?.isEnabled == true
    }

    private fun isLocationEnabled(): Boolean {
        val locationManager = context.getSystemService(Context.LOCATION_SERVICE) as LocationManager
        return if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.P) {
            locationManager.isLocationEnabled
        } else {
            locationManager.isProviderEnabled(LocationManager.GPS_PROVIDER) ||
                locationManager.isProviderEnabled(LocationManager.NETWORK_PROVIDER)
        }
    }
}

data class BluetoothDiscoveryMainActivityUiState(
    val shiftStatus: Int? = null,
    val gpsExternal: Boolean = false,
    val taximeterAvailable: Boolean = false
)

sealed interface BluetoothDiscoveryMainActivityUiEffect {
    data object NavigateBack : BluetoothDiscoveryMainActivityUiEffect
    data object ConnectTaximeter : BluetoothDiscoveryMainActivityUiEffect
    data class ShowToast(@StringRes val messageRes: Int) : BluetoothDiscoveryMainActivityUiEffect
}

sealed interface BluetoothDiscoveryMainActivityUiEvent {
    data object CheckTaximeterAvailability : BluetoothDiscoveryMainActivityUiEvent
    data class UpdateGpsExternal(val checked: Boolean) : BluetoothDiscoveryMainActivityUiEvent
    data class OnShiftStatusChanged(val currentStatus: Int) : BluetoothDiscoveryMainActivityUiEvent
    data object OnConnectTaximeter : BluetoothDiscoveryMainActivityUiEvent
    data object OnNavigateBack : BluetoothDiscoveryMainActivityUiEvent
}

class BluetoothDiscoveryMainActivityViewModel(
    private val sessionUseCase: SessionUseCase,
    private val taximeterConnectUseCase: TaximeterConnectUseCase,
) : ViewModel() {

    private val _uiState = MutableStateFlow(BluetoothDiscoveryMainActivityUiState())
    val uiState: StateFlow<BluetoothDiscoveryMainActivityUiState> = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<BluetoothDiscoveryMainActivityUiEffect>()
    val uiEffect: SharedFlow<BluetoothDiscoveryMainActivityUiEffect> = _uiEffect.asSharedFlow()

    fun onEvent(event: BluetoothDiscoveryMainActivityUiEvent) {
        when (event) {
            BluetoothDiscoveryMainActivityUiEvent.CheckTaximeterAvailability -> Unit
            is BluetoothDiscoveryMainActivityUiEvent.UpdateGpsExternal -> {
                _uiState.update { it.copy(gpsExternal = event.checked) }
                viewModelScope.launch {
                    sessionUseCase.getSession()?.let {
                        if (it.useExternalGPS != event.checked) sessionUseCase.updateExternalGPS(event.checked)
                    }
                }
            }
            is BluetoothDiscoveryMainActivityUiEvent.OnShiftStatusChanged -> {
                _uiState.update { it.copy(shiftStatus = event.currentStatus) }
            }
            BluetoothDiscoveryMainActivityUiEvent.OnConnectTaximeter -> {
                viewModelScope.launch { _uiEffect.emit(BluetoothDiscoveryMainActivityUiEffect.ConnectTaximeter) }
            }
            BluetoothDiscoveryMainActivityUiEvent.OnNavigateBack -> {
                viewModelScope.launch { _uiEffect.emit(BluetoothDiscoveryMainActivityUiEffect.NavigateBack) }
            }
        }
    }

    fun updateGpsExternal(checked: Boolean) = onEvent(BluetoothDiscoveryMainActivityUiEvent.UpdateGpsExternal(checked))

    fun connectTaximeter() {
        viewModelScope.launch { _uiEffect.emit(BluetoothDiscoveryMainActivityUiEffect.ConnectTaximeter) }
    }

    fun checkTaximeterAvailability() {
        onEvent(BluetoothDiscoveryMainActivityUiEvent.CheckTaximeterAvailability)
    }
}
```

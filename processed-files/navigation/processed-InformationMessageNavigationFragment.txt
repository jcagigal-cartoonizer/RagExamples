## 2) Route composable that connects ViewModel + UI
```kotlin
@Composable
fun InformationMessageNavigationRoute(
    viewModel: InformationMessageViewModel,
    onBack: () -> Unit,
    onNavigateBackAfterSend: () -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    LaunchedEffect(Unit) {
        viewModel.initVM()
    }
    LaunchedEffect(Unit) {
        viewModel.effects.collect { effect ->
            when (effect) {
                InformationMessageUiEffect.NavigateBack -> onBack()
                InformationMessageUiEffect.NavigateBackAfterSend -> onNavigateBackAfterSend()
                is InformationMessageUiEffect.ShowToast -> {
                    // If you want, handle toast here with LocalContext.current
                }
            }
        }
    }
    BackHandler(enabled = true) {
        viewModel.onCancelPressed()
    }
    InformationMessageScreen(
        viewModel = viewModel,
        onBack = onBack,
        onNavigateBackAfterSend = onNavigateBackAfterSend
    )
}
```

## 3) Update your composable screen so it is purely UI
Your existing composable already works, but it contains `initVM()` and effect collection.  
```kotlin
@Composable
fun InformationMessageScreen(
    uiState: InformationMessageUiState,
    buttonsState: InformationMessageButtonsState,
    onMessageSelected: (index: Int, message: String) -> Unit,
    onDialogDismiss: () -> Unit,
    onDialogButtonClick: (InformationMessageDialogButtonType) -> Unit,
    onCancel: () -> Unit,
    modifier: Modifier = Modifier,
) {
    BackHandler(enabled = true) {
        onCancel()
    }
    uiState.selectedInformationMessage?.let { selected ->
        if (uiState.dialogVisible) {
            InformationMessageDialog(
                title = uiState.dialogTitle,
                description = selected.second,
                buttonsState = buttonsState.dialogButtons,
                onDismiss = onDialogDismiss,
                onButtonClick = { type ->
                    onDialogButtonClick(type)
                }
            )
        }
    }
    Scaffold(
        modifier = modifier.fillMaxSize(),
        bottomBar = {
            InformationMessageBottomBar(
                buttonsState = buttonsState,
                onCancel = onCancel
            )
        }
    ) { padding ->
        LazyColumn(
            modifier = Modifier
                .fillMaxSize()
                .padding(padding)
                .padding(horizontal = 16.dp, vertical = 12.dp),
            verticalArrangement = Arrangement.spacedBy(8.dp)
        ) {
            itemsIndexed(uiState.informationMessages) { index, message ->
                InformationMessageRow(
                    index = index,
                    message = message,
                    onClick = { onMessageSelected(index, message) }
                )
            }
        }
    }
}
```

## 6) Important note
Your current composable screen uses `BackHandler`.  
That works inside Compose, but if the Fragment also handles back via the system or activity, don’t duplicate logic in both places unless intended.
Also, if `InformationMessageViewModel` is not an AndroidX `ViewModel` yet, it must extend:
```kotlin
// # Block 204-3: import androidx.lifecycle.ViewModel
import androidx.lifecycle.ViewModel
```
so `by viewModels()` can create it.

```kotlin
@Composable
fun PointsOfInterestScreen(
    viewModel: PointsOfInterestViewModel,
    sharedViewModel: PointsOfInterestMainActivityViewModel,
    onNavigateBack: () -> Unit,
    onLaunchIntent: (Intent) -> Unit,
    showHeader: (Boolean) -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val dialogState by viewModel.dialogState.collectAsStateWithLifecycle()
    val focusManager = LocalFocusManager.current

    LaunchedEffect(Unit) {
        showHeader(true)
    }

    LaunchedEffect(viewModel) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                is PointsOfInterestUiEffect.NavigateBack -> onNavigateBack()
                is PointsOfInterestUiEffect.LaunchIntent -> effect.intent?.let(onLaunchIntent)
                is PointsOfInterestUiEffect.SaveHiredZoneAndBack -> {
                    sharedViewModel.saveHiredZone(effect.zone)
                    onNavigateBack()
                    onNavigateBack()
                }
                is PointsOfInterestUiEffect.ShowDialog -> {
                    // handled by dialogState below
                }
            }
        }
    }

    if (dialogState is PointsOfInterestDialogState.Visible) {
        val dialog = dialogState as PointsOfInterestDialogState.Visible
        PointsOfInterestDialog(
            dialogState = dialog,
            onDismiss = { viewModel.onEvent(PointsOfInterestUiEvent.DialogDismissed) },
            onButtonClick = { button ->
                viewModel.onEvent(PointsOfInterestUiEvent.DialogButtonClicked(button))
            }
        )
    }

    PointsOfInterestContent(
        uiState = uiState,
        onEvent = { event ->
            if (event is PointsOfInterestUiEvent.SearchSubmitted) focusManager.clearFocus()
            viewModel.onEvent(event)
        }
    )
}
```

## 2) Compose-friendly ViewModel with `UiState + UiEvent + UiEffect`

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import android.content.Intent
import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.interfacom.sdk.taximeter.bravocomm.W2CLocation
import com.interfacom.sdk.taximeter.bravocomm.rest.poi.response.Poi
import com.interfacom.sdk.taximeter.licensing.models.zoning.Zone
import com.interfacom.sdk.taximeter.licensing.rest.pois.POISDataModule
import com.interfacom.sdk.taximeter.licensing.rest.pois.POISListener
import ifac.td.taxi.framework.sdk.usecase.NavigatorUseCase
import ifac.td.taxi.domain.usecase.LocationUseCase
import ifac.td.taxi.framework.util.Logs
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch
import java.util.ArrayList

data class PointsOfInterestUiState(
    val searchText: String = "",
    val poiItems: List<PointsOfInterestPoiUiModel> = emptyList(),
    val buttons: PointsOfInterestButtonsState = PointsOfInterestButtonsState()
)

data class PointsOfInterestPoiUiModel(
    val title: String,
    val description: String,
    val lat: Double?,
    val lon: Double?,
    val street: String?,
    val zone: Zone?
)

sealed interface PointsOfInterestUiEvent {
    data class SearchTextChanged(val value: String) : PointsOfInterestUiEvent
    data object SearchSubmitted : PointsOfInterestUiEvent
    data object CancelClicked : PointsOfInterestUiEvent
    data class PoiClicked(val poi: PointsOfInterestPoiUiModel) : PointsOfInterestUiEvent

    data class DialogButtonClicked(val button: ButtonType) : PointsOfInterestUiEvent
    data object DialogDismissed : PointsOfInterestUiEvent
}

sealed interface PointsOfInterestUiEffect {
    data object NavigateBack : PointsOfInterestUiEffect
    data class LaunchIntent(val intent: Intent?) : PointsOfInterestUiEffect
    data class SaveHiredZoneAndBack(val zone: Zone?) : PointsOfInterestUiEffect
    data class ShowDialog(val dialog: PointsOfInterestDialogState.Visible) : PointsOfInterestUiEffect
}

sealed interface PointsOfInterestDialogState {
    data object Hidden : PointsOfInterestDialogState
    data class Visible(
        val title: String,
        val description: String,
        val buttons: List<ButtonType>,
        val poi: PointsOfInterestPoiUiModel? = null,
        val kind: Kind
    ) : PointsOfInterestDialogState {
        enum class Kind { POI_ACTION, LOCATE_CONFIRMATION }
    }
}

data class PointsOfInterestButtonsState(
    val cancel: PointsOfInterestButtonVisualState = PointsOfInterestButtonVisualState(),
    val search: PointsOfInterestButtonVisualState = PointsOfInterestButtonVisualState()
)

data class PointsOfInterestButtonVisualState(
    val buttonType: ButtonType,
    val text: String,
    val iconRes: Int,
    val background: BackgroundButtonColor,
    val style: StyleButton,
    val visible: Boolean = true,
    val textOnly: Boolean = false,
    val textOnlyValue: String? = null
) {
    constructor() : this(
        buttonType = ButtonType.EMPTY,
        text = "",
        iconRes = 0,
        background = BackgroundButtonColor.BLACK,
        style = StyleButton.ENABLE
    )
}

class PointsOfInterestViewModel(
    private val context: Application,
    private val navigatorUseCase: NavigatorUseCase,
    private val locationUseCase: LocationUseCase
) : ViewModel() {

    private val _uiState = MutableStateFlow(PointsOfInterestUiState())
    val uiState: StateFlow<PointsOfInterestUiState> = _uiState.asStateFlow()

    private val _dialogState = MutableStateFlow<PointsOfInterestDialogState>(PointsOfInterestDialogState.Hidden)
    val dialogState: StateFlow<PointsOfInterestDialogState> = _dialogState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<PointsOfInterestUiEffect>()
    val uiEffect: SharedFlow<PointsOfInterestUiEffect> = _uiEffect.asSharedFlow()

    fun onEvent(event: PointsOfInterestUiEvent) {
        when (event) {
            is PointsOfInterestUiEvent.SearchTextChanged -> {
                _uiState.update { it.copy(searchText = event.value) }
            }

            PointsOfInterestUiEvent.SearchSubmitted -> {
                getPOIs(_uiState.value.searchText)
            }

            PointsOfInterestUiEvent.CancelClicked -> {
                viewModelScope.launch { _uiEffect.emit(PointsOfInterestUiEffect.NavigateBack) }
            }

            is PointsOfInterestUiEvent.PoiClicked -> {
                openPoiDialog(event.poi)
            }

            is PointsOfInterestUiEvent.DialogButtonClicked -> {
                handleDialogButton(event.button)
            }

            PointsOfInterestUiEvent.DialogDismissed -> {
                _dialogState.value = PointsOfInterestDialogState.Hidden
            }
        }
    }

    private fun openPoiDialog(poi: PointsOfInterestPoiUiModel) {
        val zoneDescription = poi.zone?.nombreZone ?: context.getString(R.string.poi_out_of_zone_short)
        val streetDescription = poi.street?.let {
            "\n" + context.getString(R.string.dialog_poi_description_street, it)
        } ?: ""
        val description = "\n" + context.getString(R.string.dialog_poi_description, zoneDescription, streetDescription)

        val buttons = if (poi.zone != null) {
            listOf(ButtonType.UBICAR_DESTINO, ButtonType.NAVEGAR, ButtonType.UBICAR_DESTINO_NAVEGAR)
        } else {
            listOf(ButtonType.NAVEGAR)
        }

        _dialogState.value = PointsOfInterestDialogState.Visible(
            title = poi.title,
            description = description,
            buttons = buttons,
            poi = poi,
            kind = PointsOfInterestDialogState.Visible.Kind.POI_ACTION
        )
    }

    private fun handleDialogButton(button: ButtonType) {
        val dialog = _dialogState.value as? PointsOfInterestDialogState.Visible ?: return
        val poi = dialog.poi ?: return

        when (dialog.kind) {
            PointsOfInterestDialogState.Visible.Kind.POI_ACTION -> {
                when (button) {
                    ButtonType.UBICAR_DESTINO -> {
                        _dialogState.value = PointsOfInterestDialogState.Visible(
                            title = poi.zone?.nombreZone.orEmpty(),
                            description = context.getString(R.string.ask_locate),
                            buttons = listOf(ButtonType.CANCEL, ButtonType.ACCEPT),
                            poi = poi,
                            kind = PointsOfInterestDialogState.Visible.Kind.LOCATE_CONFIRMATION
                        )
                    }

                    ButtonType.NAVEGAR -> {
                        openNavigatorApp(poi.lat, poi.lon, poi.street)
                        _dialogState.value = PointsOfInterestDialogState.Hidden
                    }

                    ButtonType.UBICAR_DESTINO_NAVEGAR -> {
                        _dialogState.value = PointsOfInterestDialogState.Visible(
                            title = poi.zone?.nombreZone.orEmpty(),
                            description = context.getString(R.string.ask_locate),
                            buttons = listOf(ButtonType.CANCEL, ButtonType.ACCEPT),
                            poi = poi,
                            kind = PointsOfInterestDialogState.Visible.Kind.LOCATE_CONFIRMATION
                        )
                    }

                    else -> Unit
                }
            }

            PointsOfInterestDialogState.Visible.Kind.LOCATE_CONFIRMATION -> {
                when (button) {
                    ButtonType.ACCEPT -> {
                        locateOnHired(poi.zone)
                        if (dialog.buttons.contains(ButtonType.UBICAR_DESTINO_NAVEGAR)) {
                            openNavigatorApp(poi.lat, poi.lon, poi.street)
                        }
                        _dialogState.value = PointsOfInterestDialogState.Hidden
                    }

                    ButtonType.CANCEL -> {
                        _dialogState.value = PointsOfInterestDialogState.Hidden
                    }

                    else -> Unit
                }
            }
        }
    }

    fun getPOIs(keyword: String) {
        viewModelScope.launch {
            val poiListener = object : POISListener {
                override fun getPOISSuccess(poiList: ArrayList<Poi>?) {
                    CoroutineScope(Dispatchers.IO).launch {
                        if (poiList != null) {
                            val mapped = poiList.map { poi ->
                                PointsOfInterestPoiUiModel(
                                    title = poi.poi,
                                    description = poi.street.orEmpty(),
                                    lat = poi.lat,
                                    lon = poi.lon,
                                    street = poi.street,
                                    zone = poi.zone
                                )
                            }
                            _uiState.update { it.copy(poiItems = mapped) }
                        }
                    }
                }

                override fun getPOISFailure() {
                    Logs.e("PointsOfInterestViewModel", "Failed getting POI list")
                }
            }

            POISDataModule.providePOISPresenter(context, poiListener).getPois(keyword)
        }
    }

    fun openNavigatorApp(latitude: Double?, longitude: Double?, street: String? = null) {
        viewModelScope.launch(Dispatchers.Main) {
            val navigatorIntent = when {
                latitude != null && longitude != null && !street.isNullOrEmpty() ->
                    navigatorUseCase.openNavigatorWithCoordinatesAndStreetName("$latitude,$longitude", street)
                latitude != null && longitude != null ->
                    navigatorUseCase.openNavigatorWithCoordinates("$latitude,$longitude")
                !street.isNullOrEmpty() ->
                    navigatorUseCase.openNavigatorApp(street)
                else ->
                    navigatorUseCase.openNavigatorApp()
            }
            _uiEffect.emit(PointsOfInterestUiEffect.LaunchIntent(navigatorIntent))
        }
    }

    fun locateOnHired(zone: Zone?) {
        viewModelScope.launch(Dispatchers.IO) {
            zone?.let { selectedZone ->
                var idMacroZone: Int? = null
                var index = 0
                do {
                    val hasTheZone =
                        W2CLocation.getZoning().macrozones[index].zones.find {
                            selectedZone.idZone == it.idZone && selectedZone.nombreZone == it.nombreZone
                        }
                    hasTheZone?.let {
                        idMacroZone = W2CLocation.getZoning().macrozones[index].idMacrozone
                    }
                    index++
                } while (idMacroZone == null && index < W2CLocation.getZoning().macrozones.size)

                idMacroZone?.let {
                    locationUseCase.locateOnHired(it, selectedZone.idZone)
                    _uiEffect.emit(PointsOfInterestUiEffect.SaveHiredZoneAndBack(selectedZone))
                }
            }
        }
    }
}
```

## 6) AndroidView wrapper for the existing `CustomButton`

Because your project already has a custom View-based button, the safest Compose integration is to wrap it.

```kotlin
package ifac.td.taxi.ui.screen

import android.view.ViewGroup
import androidx.compose.runtime.Composable
import androidx.compose.runtime.DisposableEffect
import androidx.compose.ui.Modifier
import androidx.compose.ui.viewinterop.AndroidView
import ifac.td.taxi.ui.custom.button.CustomButton
import ifac.td.taxi.viewmodel.PointsOfInterestButtonVisualState

@Composable
fun AndroidViewCustomButton(
    modifier: Modifier = Modifier,
    state: PointsOfInterestButtonVisualState,
    onClick: () -> Unit
) {
    AndroidView(
        modifier = modifier,
        factory = { context ->
            CustomButton(context).apply {
                layoutParams = ViewGroup.LayoutParams(
                    ViewGroup.LayoutParams.MATCH_PARENT,
                    ViewGroup.LayoutParams.MATCH_PARENT
                )
                setButtonType(state.buttonType.value)
                setButtonStyle(state.style)
                changeBackground(state.background)
                if (state.textOnly) {
                    changeTvOnlyText(state.textOnlyValue.orEmpty())
                } else {
                    changeText(state.text)
                }
                setAction(onClick)
            }
        },
        update = { button ->
            button.setButtonType(state.buttonType.value)
            button.setButtonStyle(state.style)
            button.changeBackground(state.background)
            if (state.textOnly) {
                button.changeTvOnlyText(state.textOnlyValue.orEmpty())
            } else {
                button.changeText(state.text)
            }
            button.setAction(onClick)
        }
    )
}
```

```kotlin
class PointsOfInterestViewModel(
    private val context: Application,
    private val navigatorUseCase: NavigatorUseCase,
    private val locationUseCase: LocationUseCase
) : ViewModel() {

    private val _uiState = MutableStateFlow(PointsOfInterestUiState())
    val uiState: StateFlow<PointsOfInterestUiState> = _uiState.asStateFlow()

    private val _dialogState = MutableStateFlow<PointsOfInterestDialogState>(PointsOfInterestDialogState.Hidden)
    val dialogState: StateFlow<PointsOfInterestDialogState> = _dialogState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<PointsOfInterestUiEffect>()
    val uiEffect: SharedFlow<PointsOfInterestUiEffect> = _uiEffect.asSharedFlow()

    fun onEvent(event: PointsOfInterestUiEvent) {
        when (event) {
            is PointsOfInterestUiEvent.SearchTextChanged -> {
                _uiState.update { it.copy(searchText = event.value) }
            }

            PointsOfInterestUiEvent.SearchSubmitted -> {
                getPOIs(_uiState.value.searchText)
            }

            PointsOfInterestUiEvent.CancelClicked -> {
                viewModelScope.launch { _uiEffect.emit(PointsOfInterestUiEffect.NavigateBack) }
            }

            is PointsOfInterestUiEvent.PoiClicked -> {
                openPoiDialog(event.poi)
            }

            is PointsOfInterestUiEvent.DialogButtonClicked -> {
                handleDialogButton(event.button)
            }

            PointsOfInterestUiEvent.DialogDismissed -> {
                _dialogState.value = PointsOfInterestDialogState.Hidden
            }
        }
    }

    fun getPOIs(keyword: String) {
        viewModelScope.launch {
            val poiListener = object : POISListener {
                override fun getPOISSuccess(poiList: ArrayList<Poi>?) {
                    CoroutineScope(Dispatchers.IO).launch {
                        if (poiList != null) {
                            Logs.d("PointsOfInterestViewModel", "getPOISSuccess: $poiList")
                            val mapped = poiList.map { poi ->
                                PointsOfInterestPoiUiModel(
                                    title = poi.poi,
                                    description = poi.street.orEmpty(),
                                    lat = poi.lat,
                                    lon = poi.lon,
                                    street = poi.street,
                                    zone = poi.zone
                                )
                            }
                            _uiState.update { it.copy(poiItems = mapped) }
                        }
                    }
                }

                override fun getPOISFailure() {
                    Logs.e("PointsOfInterestViewModel", "Failed getting POI list")
                }
            }

            POISDataModule.providePOISPresenter(context, poiListener).getPois(keyword)
        }
    }

    fun openNavigatorApp(latitude: Double?, longitude: Double?, street: String? = null) {
        viewModelScope.launch(Dispatchers.Main) {
            Logs.d(
                "PointsOfInterestViewModel",
                "openNavigatorApp: Received params -> Latitude: $latitude, Longitude: $longitude, Street: $street"
            )

            val navigatorIntent = when {
                latitude != null && longitude != null && !street.isNullOrEmpty() -> {
                    navigatorUseCase.openNavigatorWithCoordinatesAndStreetName("$latitude,$longitude", street)
                }
                latitude != null && longitude != null -> {
                    navigatorUseCase.openNavigatorWithCoordinates("$latitude,$longitude")
                }
                !street.isNullOrEmpty() -> {
                    navigatorUseCase.openNavigatorApp(street)
                }
                else -> {
                    navigatorUseCase.openNavigatorApp()
                }
            }

            _uiEffect.emit(PointsOfInterestUiEffect.LaunchIntent(navigatorIntent))
        }
    }

    fun locateOnHired(zone: Zone?) {
        viewModelScope.launch(Dispatchers.IO) {
            zone?.let { selectedZone ->
                var idMacroZone: Int? = null
                var index = 0
                do {
                    val hasTheZone =
                        W2CLocation.getZoning().macrozones[index].zones.find { zone ->
                            selectedZone.idZone == zone.idZone && selectedZone.nombreZone == zone.nombreZone
                        }
                    hasTheZone?.let {
                        idMacroZone = W2CLocation.getZoning().macrozones[index].idMacrozone
                    }
                    index++
                } while (idMacroZone == null && index < W2CLocation.getZoning().macrozones.size)

                idMacroZone?.let {
                    locationUseCase.locateOnHired(it, selectedZone.idZone)
                    _uiEffect.emit(PointsOfInterestUiEffect.SaveHiredZoneAndBack(selectedZone))
                }
            }
        }
    }

    private fun openPoiDialog(poi: PointsOfInterestPoiUiModel) {
        val zoneDescription = poi.zone?.nombreZone ?: context.getString(R.string.poi_out_of_zone_short)
        val streetDescription = poi.street?.let {
            "\n" + context.getString(R.string.dialog_poi_description_street, it)
        } ?: ""
        val description = "\n" + context.getString(R.string.dialog_poi_description, zoneDescription, streetDescription)

        val buttons = if (poi.zone != null) {
            listOf(ButtonType.UBICAR_DESTINO, ButtonType.NAVEGAR, ButtonType.UBICAR_DESTINO_NAVEGAR)
        } else {
            listOf(ButtonType.NAVEGAR)
        }

        _dialogState.value = PointsOfInterestDialogState.Visible(
            title = poi.title,
            description = description,
            buttons = buttons,
            poi = poi,
            kind = PointsOfInterestDialogState.Visible.Kind.POI_ACTION
        )
    }

    private fun handleDialogButton(button: ButtonType) {
        val dialog = _dialogState.value as? PointsOfInterestDialogState.Visible ?: return
        val poi = dialog.poi ?: return

        when (dialog.kind) {
            PointsOfInterestDialogState.Visible.Kind.POI_ACTION -> {
                when (button) {
                    ButtonType.UBICAR_DESTINO -> {
                        _dialogState.value = PointsOfInterestDialogState.Visible(
                            title = poi.zone?.nombreZone.orEmpty(),
                            description = context.getString(R.string.ask_locate),
                            buttons = listOf(ButtonType.CANCEL, ButtonType.ACCEPT),
                            poi = poi,
                            kind = PointsOfInterestDialogState.Visible.Kind.LOCATE_CONFIRMATION
                        )
                    }

                    ButtonType.NAVEGAR -> {
                        openNavigatorApp(poi.lat, poi.lon, poi.street)
                        _dialogState.value = PointsOfInterestDialogState.Hidden
                    }

                    ButtonType.UBICAR_DESTINO_NAVEGAR -> {
                        _dialogState.value = PointsOfInterestDialogState.Visible(
                            title = poi.zone?.nombreZone.orEmpty(),
                            description = context.getString(R.string.ask_locate),
                            buttons = listOf(ButtonType.CANCEL, ButtonType.ACCEPT),
                            poi = poi,
                            kind = PointsOfInterestDialogState.Visible.Kind.LOCATE_CONFIRMATION
                        )
                    }

                    else -> Unit
                }
            }

            PointsOfInterestDialogState.Visible.Kind.LOCATE_CONFIRMATION -> {
                when (button) {
                    ButtonType.ACCEPT -> {
                        locateOnHired(poi.zone)
                        if (dialog.buttons.contains(ButtonType.UBICAR_DESTINO_NAVEGAR)) {
                            openNavigatorApp(poi.lat, poi.lon, poi.street)
                        }
                        _dialogState.value = PointsOfInterestDialogState.Hidden
                    }

                    ButtonType.CANCEL -> {
                        _dialogState.value = PointsOfInterestDialogState.Hidden
                    }

                    else -> Unit
                }
            }
        }
    }
}

@Composable
fun PointsOfInterestScreen(
    viewModel: PointsOfInterestViewModel,
    sharedViewModel: PointsOfInterestMainActivityViewModel,
    onNavigateBack: () -> Unit,
    onLaunchIntent: (Intent) -> Unit,
    showHeader: (Boolean) -> Unit
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val dialogState by viewModel.dialogState.collectAsStateWithLifecycle()
    val focusManager = LocalFocusManager.current

    LaunchedEffect(Unit) {
        showHeader(true)
    }

    LaunchedEffect(viewModel) {
        viewModel.uiEffect.collect { effect ->
            when (effect) {
                PointsOfInterestUiEffect.NavigateBack -> onNavigateBack()
                is PointsOfInterestUiEffect.LaunchIntent -> effect.intent?.let(onLaunchIntent)
                is PointsOfInterestUiEffect.SaveHiredZoneAndBack -> {
                    sharedViewModel.saveHiredZone(effect.zone)
                    onNavigateBack()
                    onNavigateBack()
                }
            }
        }
    }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(Color.Black)
    ) {
        Column(modifier = Modifier.fillMaxSize()) {
            androidx.compose.material3.OutlinedTextField(
                value = uiState.searchText,
                onValueChange = { viewModel.onEvent(PointsOfInterestUiEvent.SearchTextChanged(it)) },
                modifier = Modifier
                    .fillMaxWidth()
                    .height(48.dp),
                placeholder = { Text(text = "Introduce POI") },
                singleLine = true,
                keyboardOptions = KeyboardOptions(imeAction = ImeAction.Search),
                keyboardActions = KeyboardActions(
                    onSearch = {
                        focusManager.clearFocus()
                        viewModel.onEvent(PointsOfInterestUiEvent.SearchSubmitted)
                    }
                )
            )

            Spacer(modifier = Modifier.height(8.dp))

            LazyColumn(
                modifier = Modifier
                    .weight(1f)
                    .fillMaxWidth()
            ) {
                items(uiState.poiItems) { poi ->
                    ListItem(
                        headlineContent = { Text(poi.title) },
                        supportingContent = { Text(poi.description) },
                        trailingContent = {
                            TextButton(onClick = { viewModel.onEvent(PointsOfInterestUiEvent.PoiClicked(poi)) }) {
                                Text("OPEN")
                            }
                        }
                    )
                }
            }

            Row(
                modifier = Modifier
                    .fillMaxWidth()
                    .heightIn(min = 96.dp)
                    .background(Color.Black),
                horizontalArrangement = Arrangement.spacedBy(6.dp)
            ) {
                PointsOfInterestCustomButton(
                    modifier = Modifier.weight(1f),
                    state = uiState.buttons.cancel,
                    onClick = { viewModel.onEvent(PointsOfInterestUiEvent.CancelClicked) }
                )
                PointsOfInterestCustomButton(
                    modifier = Modifier.weight(1f),
                    state = uiState.buttons.search,
                    onClick = { viewModel.onEvent(PointsOfInterestUiEvent.SearchSubmitted) }
                )
            }
        }

        if (dialogState is PointsOfInterestDialogState.Visible) {
            val dialog = dialogState as PointsOfInterestDialogState.Visible
            AlertDialog(
                onDismissRequest = { viewModel.onEvent(PointsOfInterestUiEvent.DialogDismissed) },
                title = { Text(dialog.title) },
                text = { Text(dialog.description) },
                confirmButton = {
                    Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                        dialog.buttons.forEach { button ->
                            TextButton(onClick = { viewModel.onEvent(PointsOfInterestUiEvent.DialogButtonClicked(button)) }) {
                                Text(button.name)
                            }
                        }
                    }
                }
            )
        }
    }
}

@Composable
private fun PointsOfInterestCustomButton(
    modifier: Modifier = Modifier,
    state: PointsOfInterestButtonVisualState,
    onClick: () -> Unit
) {
    if (!state.visible) return

    AndroidView(
        modifier = modifier,
        factory = { context ->
            CustomButton(context).apply {
                setButtonType(state.buttonType.value)
                setButtonStyle(state.style)
                changeBackground(state.background)
                if (state.textOnly) {
                    changeTvOnlyText(state.textOnlyValue.orEmpty())
                } else {
                    changeText(state.text)
                }
                setAction(onClick)
            }
        },
        update = { button ->
            button.setButtonType(state.buttonType.value)
            button.setButtonStyle(state.style)
            button.changeBackground(state.background)
            if (state.textOnly) {
                button.changeTvOnlyText(state.textOnlyValue.orEmpty())
            } else {
                button.changeText(state.text)
            }
            button.setAction(onClick)
        }
    )
}
```

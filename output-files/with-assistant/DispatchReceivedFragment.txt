## 5) Compose screen

This preserves:
- header hidden
- bottom bar hidden
- dispatch loading on start
- dialog handling only for the `CustomDialog` flow
- navigation via `NavController`
- button behavior replacing the fragment’s `setOnClickListener`

```kotlin
package ifac.td.taxi.ui.screen.dispatchreceived

import androidx.compose.foundation.background
import androidx.compose.foundation.layout.*
import androidx.compose.material3.AlertDialog
import androidx.compose.material3.Button
import androidx.compose.material3.Text
import androidx.compose.runtime.*
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.platform.LocalLifecycleOwner
import androidx.compose.ui.text.style.TextAlign
import androidx.compose.ui.unit.dp
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.navigation.NavController
import ifac.td.taxi.ui.custom.dialog.CustomDialog
import kotlinx.coroutines.flow.collectLatest

@Composable
fun DispatchReceivedScreen(
    navController: NavController,
    dispatchId: Long,
    viewModel: DispatchReceivedViewModel,
    sharedViewModel: DispatchReceivedMainActivityViewModel,
    onShowHeader: (Boolean) -> Unit,
    onShowBottomBar: (Boolean) -> Unit,
    onCloseDialog: (CustomDialog.CustomDialogTAG) -> Unit,
    onRequestPhonePermission: () -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val lifecycleOwner = LocalLifecycleOwner.current

    LaunchedEffect(dispatchId) {
        onShowHeader(false)
        onShowBottomBar(false)
        viewModel.onScreenStarted(dispatchId)
        onCloseDialog(CustomDialog.CustomDialogTAG.LOCATE_ON_STAND_DIALOG)
    }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                is DispatchReceivedUiEffect.NavigateToInfoDispatch -> {
                    navController.navigate(
                        "info_dispatch/${effect.tripId}"
                    )
                }
                DispatchReceivedUiEffect.RequestPhonePermission -> onRequestPhonePermission()
                DispatchReceivedUiEffect.CloseLocateOnStandDialog -> {
                    onCloseDialog(CustomDialog.CustomDialogTAG.LOCATE_ON_STAND_DIALOG)
                }
                DispatchReceivedUiEffect.NavigateBack -> navController.popBackStack()
            }
        }
    }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(Color.White)
    ) {
        Column(
            modifier = Modifier
                .fillMaxSize()
                .padding(16.dp),
            verticalArrangement = Arrangement.SpaceBetween
        ) {
            Column(
                modifier = Modifier.fillMaxWidth(),
                horizontalAlignment = Alignment.CenterHorizontally
            ) {
                uiState.dispatch?.let { dispatch ->
                    Text(
                        text = dispatch.pickUpZone.orEmpty(),
                        textAlign = TextAlign.Center
                    )
                    Spacer(Modifier.height(8.dp))
                    Text(text = dispatch.pickUpTime.orEmpty())
                    if (!dispatch.smallPickUpAdress.isNullOrBlank()) {
                        Spacer(Modifier.height(8.dp))
                        Text(text = dispatch.smallPickUpAdress.orEmpty())
                    }
                    if (!dispatch.city.isNullOrBlank()) {
                        Spacer(Modifier.height(8.dp))
                        Text(text = dispatch.city.orEmpty())
                    }
                    if (!dispatch.passengerName.isNullOrBlank()) {
                        Spacer(Modifier.height(8.dp))
                        Text(text = dispatch.passengerName.orEmpty())
                    }
                    Spacer(Modifier.height(16.dp))
                    Text(text = "¿Aceptar despacho?")
                }
            }

            Row(
                modifier = Modifier
                    .fillMaxWidth()
                    .height(120.dp),
                horizontalArrangement = Arrangement.spacedBy(6.dp)
            ) {
                DispatchReceivedCustomButton(
                    state = uiState.buttons.cancel,
                    modifier = Modifier.weight(1f),
                    onClick = { viewModel.onCancelClicked() }
                )
                DispatchReceivedCustomButton(
                    state = uiState.buttons.accept,
                    modifier = Modifier.weight(1f),
                    onClick = { viewModel.onAcceptClicked() }
                )
            }
        }

        when (val dialog = uiState.dialog) {
            is DispatchReceivedDialogState.Hidden -> Unit
            is DispatchReceivedDialogState.ConfirmReject -> {
                AlertDialog(
                    onDismissRequest = { viewModel.onDialogDismissed() },
                    title = { Text(dialog.title) },
                    text = { Text(dialog.description) },
                    confirmButton = {
                        Button(onClick = { viewModel.onRejectConfirmed() }) {
                            Text("ACEPTAR")
                        }
                    },
                    dismissButton = {
                        Button(onClick = { viewModel.onDialogDismissed() }) {
                            Text("CANCELAR")
                        }
                    }
                )
            }
            is DispatchReceivedDialogState.PermissionWarning -> {
                AlertDialog(
                    onDismissRequest = { viewModel.onDialogDismissed() },
                    title = { Text(dialog.title) },
                    text = { Text(dialog.description) },
                    confirmButton = {
                        Button(onClick = {
                            onRequestPhonePermission()
                            viewModel.onDialogDismissed()
                        }) {
                            Text("ACEPTAR")
                        }
                    },
                    dismissButton = {
                        Button(onClick = { viewModel.onDialogDismissed() }) {
                            Text("CANCELAR")
                        }
                    }
                )
            }
        }
    }
}
```

## 6) Compose button that matches the XML behavior

This is a Compose replacement for the `CustomButton` usage in this screen only.

```kotlin
package ifac.td.taxi.ui.screen.dispatchreceived

import androidx.compose.foundation.background
import androidx.compose.foundation.clickable
import androidx.compose.foundation.layout.*
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.text.font.FontWeight
import androidx.compose.ui.unit.dp

@Composable
fun DispatchReceivedCustomButton(
    state: DispatchReceivedButtonVisualState,
    modifier: Modifier = Modifier,
    onClick: () -> Unit
) {
    if (!state.visible) return

    Box(
        modifier = modifier
            .fillMaxHeight()
            .background(state.backgroundColor)
            .clickable(enabled = state.enabled) { onClick() }
            .padding(12.dp),
        contentAlignment = Alignment.Center
    ) {
        Text(
            text = state.text,
            color = state.contentColor,
            fontWeight = FontWeight.Bold
        )
    }
}
```

```kotlin
    onCloseDialog: (CustomDialog.CustomDialogTAG) -> Unit,
    onRequestPhonePermission: () -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()

    LaunchedEffect(dispatchId) {
        onShowHeader(false)
        onShowBottomBar(false)
        viewModel.onScreenStarted(dispatchId)
        onCloseDialog(CustomDialog.CustomDialogTAG.LOCATE_ON_STAND_DIALOG)
    }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                is DispatchReceivedUiEffect.NavigateToInfoDispatch -> {
                    navController.navigate("info_dispatch/${effect.tripId}")
                }
                DispatchReceivedUiEffect.RequestPhonePermission -> onRequestPhonePermission()
                DispatchReceivedUiEffect.CloseLocateOnStandDialog -> {
                    onCloseDialog(CustomDialog.CustomDialogTAG.LOCATE_ON_STAND_DIALOG)
                }
                DispatchReceivedUiEffect.NavigateBack -> navController.popBackStack()
            }
        }
    }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(Color.White)
    ) {
        Column(
            modifier = Modifier
                .fillMaxSize()
                .padding(16.dp),
            verticalArrangement = Arrangement.SpaceBetween
        ) {
            Column(
                modifier = Modifier.fillMaxWidth(),
                horizontalAlignment = Alignment.CenterHorizontally
            ) {
                uiState.dispatch?.let { dispatch ->
                    Text(
                        text = dispatch.pickUpZone.orEmpty(),
                        textAlign = TextAlign.Center
                    )
                    Spacer(Modifier.height(8.dp))
                    Text(text = dispatch.pickUpTime.orEmpty())
                    if (!dispatch.smallPickUpAdress.isNullOrBlank()) {
                        Spacer(Modifier.height(8.dp))
                        Text(text = dispatch.smallPickUpAdress.orEmpty())
                    }
                    if (!dispatch.city.isNullOrBlank()) {
                        Spacer(Modifier.height(8.dp))
                        Text(text = dispatch.city.orEmpty())
                    }
                    if (!dispatch.passengerName.isNullOrBlank()) {
                        Spacer(Modifier.height(8.dp))
                        Text(text = dispatch.passengerName.orEmpty())
                    }
                    Spacer(Modifier.height(16.dp))
                    Text(text = "¿Aceptar despacho?")
                }
            }

            Row(
                modifier = Modifier
                    .fillMaxWidth()
                    .height(120.dp),
                horizontalArrangement = Arrangement.spacedBy(6.dp)
            ) {
                DispatchReceivedCustomButton(
                    state = uiState.buttons.cancel,
                    modifier = Modifier.weight(1f),
                    onClick = { viewModel.onCancelClicked() }
                )
                DispatchReceivedCustomButton(
                    state = uiState.buttons.accept,
                    modifier = Modifier.weight(1f),
                    onClick = { viewModel.onAcceptClicked() }
                )
            }
        }

        when (val dialog = uiState.dialog) {
            is DispatchReceivedDialogState.Hidden -> Unit
            is DispatchReceivedDialogState.ConfirmReject -> {
                AlertDialog(
                    onDismissRequest = { viewModel.onDialogDismissed() },
                    title = { Text(dialog.title) },
                    text = { Text(dialog.description) },
                    confirmButton = {
                        Button(onClick = { viewModel.onRejectConfirmed() }) {
                            Text("ACEPTAR")
                        }
                    },
                    dismissButton = {
                        Button(onClick = { viewModel.onDialogDismissed() }) {
                            Text("CANCELAR")
                        }
                    }
                )
            }
            is DispatchReceivedDialogState.PermissionWarning -> {
                AlertDialog(
                    onDismissRequest = { viewModel.onDialogDismissed() },
                    title = { Text(dialog.title) },
                    text = { Text(dialog.description) },
                    confirmButton = {
                        Button(onClick = {
                            onRequestPhonePermission()
                            viewModel.onDialogDismissed()
                        }) {
                            Text("ACEPTAR")
                        }
                    },
                    dismissButton = {
                        Button(onClick = { viewModel.onDialogDismissed() }) {
                            Text("CANCELAR")
                        }
                    }
                )
            }
        }
    }
}

@Composable
fun DispatchReceivedCustomButton(
    state: DispatchReceivedButtonVisualState,
    modifier: Modifier = Modifier,
    onClick: () -> Unit
) {
    if (!state.visible) return

    Box(
        modifier = modifier
            .fillMaxHeight()
            .background(state.backgroundColor)
            .clickable(enabled = state.enabled) { onClick() }
            .padding(12.dp),
        contentAlignment = Alignment.Center
    ) {
        Text(
            text = state.text,
            color = state.contentColor,
            fontWeight = FontWeight.Bold
        )
    }
}
```

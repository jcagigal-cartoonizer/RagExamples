```kotlin
@Composable
fun ZoningServicesScreen(
    idMacroZone: Int,
    idZone: Int,
    viewModel: ZoningServicesViewModel,
    sharedViewModel: ZoningServicesMainActivityViewModel,
    onNavigateBack: () -> Unit,
    onNavigateToHome: () -> Unit,
    onNavigateToOnTrip: () -> Unit,
) {
    val lifecycleOwner = LocalLifecycleOwner.current

    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val buttonsState by viewModel.buttonsState.collectAsStateWithLifecycle()

    LaunchedEffect(idMacroZone, idZone) {
        viewModel.onEvent(ZoningServicesUiEvent.Init(idMacroZone, idZone))
    }

    LaunchedEffect(Unit) {
        lifecycleOwner.repeatOnLifecycle(Lifecycle.State.STARTED) {
            launch {
                viewModel.uiEffect.collect { effect ->
                    when (effect) {
                        ZoningServicesUiEffect.NavigateBack -> onNavigateBack()
                        ZoningServicesUiEffect.NavigateHome -> onNavigateToHome()
                        ZoningServicesUiEffect.NavigateToOnTrip -> onNavigateToOnTrip()
                        ZoningServicesUiEffect.EmitManualZoningNavigationFalse ->
                            sharedViewModel.emitManualZoningNavigation(false)

                        is ZoningServicesUiEffect.ShowToast -> {
                            // handle if needed
                        }
                    }
                }
            }
        }
    }

    DisposableEffect(Unit) {
        onDispose { viewModel.onEvent(ZoningServicesUiEvent.OnPause) }
    }

    Column(modifier = Modifier.fillMaxSize()) {
        LinearProgressIndicator(
            progress = { uiState.refreshProgress / 1000f },
            modifier = Modifier
                .fillMaxWidth()
                .height(2.dp)
        )

        Box(
            modifier = Modifier
                .weight(1f)
                .fillMaxWidth()
        ) {
            if (uiState.isLoading) {
                CircularCenterLoading()
            }

            Column(modifier = Modifier.fillMaxSize()) {
                Text(
                    text = uiState.zoneName,
                    modifier = Modifier
                        .fillMaxWidth()
                        .background(Color.Gray)
                        .padding(vertical = 8.dp),
                    style = MaterialTheme.typography.bodyLarge
                )

                if (uiState.showNoCars) {
                    Box(Modifier.fillMaxSize(), contentAlignment = Alignment.Center) {
                        Text(text = "No services on zone")
                    }
                } else {
                    LazyColumn(modifier = Modifier.fillMaxSize()) {
                        items(uiState.zoneTrips) { trip ->
                            ZoneTripRow(trip)
                        }
                    }
                }
            }
        }

        ZoningServicesButtonsRow(
            state = buttonsState,
            onCancel = { viewModel.onEvent(ZoningServicesUiEvent.CancelClicked) },
            onShowAll = { viewModel.onEvent(ZoningServicesUiEvent.ShowAllClicked(idMacroZone, idZone)) },
            onShowRecent = { viewModel.onEvent(ZoningServicesUiEvent.ShowRecentClicked(idMacroZone, idZone)) },
            onClose = { viewModel.onEvent(ZoningServicesUiEvent.CloseClicked) }
        )
    }
}
```

```kotlin
class ZoningServicesViewModel(
    private val zoningUseCase: ZoningUseCase,
    private val licensingUseCase: LicensingUseCase,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val bravoConfigurationVariableDao: BravoConfigurationVariableDao,
    context: Application,
) : BaseViewModel(context) {

    private val TAG = "ZoningServicesViewModel"

    private val _uiState = MutableStateFlow(ZoningServicesUiState())
    val uiState = _uiState.asStateFlow()

    private val _buttonsState = MutableStateFlow(ZoningServicesButtonsState.initial())
    val buttonsState = _buttonsState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<ZoningServicesUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()

    private var showAllTrips = false
    private var companyNumber: String? = null
    private val handler = Handler(Looper.getMainLooper())
    private lateinit var zonesCarsRunnable: Runnable

    private val callback: (List<ZoneTrip>?) -> Unit = { trips ->
        viewModelScope.launch {
            val mapped = trips?.map {
                ZoneTripModel(
                    id = it.tripID,
                    pickupTime = it.pickupTime,
                    company = it.company,
                    passengerNumber = it.passengerNumber,
                    baggageNumber = it.baggageNumber,
                    vehicleRequirements = it.vehicleRequirements,
                    driverRequirements = it.driverRequirements,
                    maxDistanceZoneService = it.maxDistanceZoneService,
                    ownNearbyVehicles = it.ownNearbyVehicles,
                    totalNearbyVehicles = it.totalNearbyVehicles,
                    myOrder = it.myOrder,
                )
            }.orEmpty()

            _uiState.update {
                it.copy(
                    isLoading = false,
                    zoneTrips = mapped,
                    showNoCars = mapped.isEmpty()
                )
            }
        }
    }

    init {
        viewModelScope.launch(Dispatchers.IO) {
            companyNumber = bravoConfigurationVariableDao.getBravoConfigurationVariable()?.companyNumber
            val requestAll = bravoConfigurationVariableDao.getBravoConfigurationVariable()?.requestAllServices == true
            _uiState.update {
                it.copy(
                    companyNumber = companyNumber,
                    showAllButtonEnabled = requestAll
                )
            }
            _buttonsState.update {
                ZoningServicesButtonsReducer.reduce(it, ZoningServicesButtonsIntent.ShowAllButtonAvailability(requestAll))
            }
        }
    }

    fun onEvent(event: ZoningServicesUiEvent) {
        when (event) {
            is ZoningServicesUiEvent.Init -> initViewModelData(event.idMacroZone, event.idZone)
            is ZoningServicesUiEvent.ShowAllClicked -> setShowAllTrips(true, event.idMacroZone, event.idZone)
            is ZoningServicesUiEvent.ShowRecentClicked -> setShowAllTrips(false, event.idMacroZone, event.idZone)
            ZoningServicesUiEvent.CancelClicked -> viewModelScope.launch {
                _uiEffect.emit(ZoningServicesUiEffect.EmitManualZoningNavigationFalse)
                _uiEffect.emit(ZoningServicesUiEffect.NavigateBack)
            }
            ZoningServicesUiEvent.CloseClicked -> handleClose()
            ZoningServicesUiEvent.OnPause -> removeHandlerCallbacks()
        }
    }

    private fun initViewModelData(idMacroZone: Int, idZone: Int) {
        removeHandlerCallbacks()
        viewModelScope.launch {
            _uiState.update { it.copy(isLoading = true) }

            val refreshRate = licensingUseCase.getLicensingParameters()?.segRefrescoConsUbicacion
            refreshRate?.let { seconds ->
                val millis = (seconds * 1000).toLong()
                var lastUpdate = System.currentTimeMillis()

                zoningUseCase.getServicesInformation(
                    idMacroZone,
                    idZone,
                    showAllTrips,
                    bravoConfigurationVariableDao.getBravoConfigurationVariable()?.showCarsInTripRange,
                    callback
                )
                _uiState.update { it.copy(refreshProgress = 1000) }

                removeHandlerCallbacks()
                zonesCarsRunnable = object : Runnable {
                    override fun run() {
                        viewModelScope.launch {
                            val elapsed = System.currentTimeMillis() - lastUpdate
                            val progress = 1000 - ((elapsed.toFloat() / millis) * 1000).toInt().coerceIn(0, 1000)
                            _uiState.update { it.copy(refreshProgress = progress) }

                            if (elapsed >= millis) {
                                zoningUseCase.getServicesInformation(
                                    idMacroZone,
                                    idZone,
                                    showAllTrips,
                                    bravoConfigurationVariableDao.getBravoConfigurationVariable()?.showCarsInTripRange,
                                    callback
                                )
                                lastUpdate = System.currentTimeMillis()
                                _uiState.update { it.copy(refreshProgress = 1000) }
                            }
                        }
                        handler.postDelayed(this, 1000)
                    }
                }
                handler.post(zonesCarsRunnable)
            }
        }
    }

    private fun setShowAllTrips(b: Boolean, idMacroZone: Int, idZone: Int) {
        showAllTrips = b
        _uiState.update { it.copy(showAllTrips = b) }
        viewModelScope.launch {
            zoningUseCase.getServicesInformation(
                idMacroZone,
                idZone,
                showAllTrips,
                bravoConfigurationVariableDao.getBravoConfigurationVariable()?.showCarsInTripRange,
                callback
            )
        }
    }

    private fun handleClose() {
        viewModelScope.launch {
            val hiredStates = setOf(
                ifac.td.taxi.framework.sdk.usecase.Constants.STATE_HIRED,
                ifac.td.taxi.framework.sdk.usecase.Constants.STATE_DISPATCHED,
                ifac.td.taxi.framework.sdk.usecase.Constants.STATE_HIRED_DISPATCHED,
                ifac.td.taxi.framework.sdk.usecase.Constants.STATE_HIRED_NO_CENTRAL
            )

            val currentStatus = shiftStatusUseCase.currentStatusOrNull()
            if (currentStatus != null && hiredStates.contains(currentStatus)) {
                _uiEffect.emit(ZoningServicesUiEffect.NavigateHome)
                _uiEffect.emit(ZoningServicesUiEffect.NavigateToOnTrip)
            } else {
                _uiEffect.emit(ZoningServicesUiEffect.NavigateHome)
            }
        }
    }

    fun removeHandlerCallbacks() {
        if (::zonesCarsRunnable.isInitialized) handler.removeCallbacks(zonesCarsRunnable)
    }
}
```

## 3) Exact reducer for button state

This mirrors the fragment’s `collect {}` branches one-by-one:

```kotlin
package ifac.td.taxi.viewmodel

import androidx.compose.ui.graphics.Color

sealed interface ZoningServicesButtonsIntent {
    data class ShowAllButtonAvailability(val enabled: Boolean) : ZoningServicesButtonsIntent
    data class ShowAllPressed(val visible: Boolean) : ZoningServicesButtonsIntent
    data class ShowRecentPressed(val visible: Boolean) : ZoningServicesButtonsIntent
}

object ZoningServicesButtonsReducer {

    fun reduce(
        current: ZoningServicesButtonsState,
        intent: ZoningServicesButtonsIntent
    ): ZoningServicesButtonsState {
        return when (intent) {
            is ZoningServicesButtonsIntent.ShowAllButtonAvailability -> {
                // Mirrors:
                // vModel.enableShowAllButtonFlow.collect { it?.let { showButton -> if (showButton) btnShowAll.setButtonStyle(ENABLE) } }
                if (intent.enabled) {
                    current.copy(
                        showAll = current.showAll.copy(
                            enabled = true,
                            backgroundColor = Color(0xFF2E7D32), // ENABLE green
                            contentColor = Color.White
                        )
                    )
                } else current
            }

            is ZoningServicesButtonsIntent.ShowAllPressed -> {
                // Mirrors btnShowAll.setAction { btnShowAll.visibility = GONE; btnShowRecent.visibility = VISIBLE }
                current.copy(
                    showAll = current.showAll.copy(visible = !intent.visible),
                    showRecent = current.showRecent.copy(visible = intent.visible)
                )
            }

            is ZoningServicesButtonsIntent.ShowRecentPressed -> {
                // Mirrors btnShowRecent.setAction { btnShowAll.visibility = VISIBLE; btnShowRecent.visibility = GONE }
                current.copy(
                    showAll = current.showAll.copy(visible = intent.visible),
                    showRecent = current.showRecent.copy(visible = !intent.visible)
                )
            }
        }
    }
}
```

## 5) Notes on navigation preservation

The fragment did:

- back button -> `emitManualZoningNavigation(false)` + `navigateBack()`
- cancel -> same
- close -> navigate to Home, then maybe OnTrip

In Compose, that is preserved through `ZoningServicesUiEffect` and the host navigation lambdas:

- `NavigateBack`
- `NavigateHome`
- `NavigateToOnTrip`
- `EmitManualZoningNavigationFalse`

So your Compose destination can still use the same NavController / SafeArgs flow.

You asked for dialog state only if the ViewModel uses a `CustomDialog`.

Your provided `ZoningServicesFragment` / `ZoningServicesViewModel` do not use `CustomDialog`, so I included `ZoningServicesDialogState` as a placeholder in the state model, but no dialog UI is wired in. If you later add a dialog, you can drive it from `uiState.dialogState`.

```kotlin
            )
        )
    }
}

sealed interface ZoningServicesButtonsIntent {
    data class ShowAllButtonAvailability(val enabled: Boolean) : ZoningServicesButtonsIntent
    data class ShowAllPressed(val visible: Boolean) : ZoningServicesButtonsIntent
    data class ShowRecentPressed(val visible: Boolean) : ZoningServicesButtonsIntent
}

object ZoningServicesButtonsReducer {
    fun reduce(
        current: ZoningServicesButtonsState,
        intent: ZoningServicesButtonsIntent
    ): ZoningServicesButtonsState {
        return when (intent) {
            is ZoningServicesButtonsIntent.ShowAllButtonAvailability -> {
                if (intent.enabled) {
                    current.copy(
                        showAll = current.showAll.copy(
                            enabled = true,
                            backgroundColor = Color(0xFF2E7D32),
                            contentColor = Color.White
                        )
                    )
                } else current
            }

            is ZoningServicesButtonsIntent.ShowAllPressed -> {
                current.copy(
                    showAll = current.showAll.copy(visible = !intent.visible),
                    showRecent = current.showRecent.copy(visible = intent.visible)
                )
            }

            is ZoningServicesButtonsIntent.ShowRecentPressed -> {
                current.copy(
                    showAll = current.showAll.copy(visible = intent.visible),
                    showRecent = current.showRecent.copy(visible = !intent.visible)
                )
            }
        }
    }
}

class ZoningServicesViewModel(
    private val zoningUseCase: ZoningUseCase,
    private val licensingUseCase: LicensingUseCase,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val bravoConfigurationVariableDao: BravoConfigurationVariableDao,
    context: Application,
) : BaseViewModel(context) {

    private val TAG = "ZoningServicesViewModel"

    private val _uiState = MutableStateFlow(ZoningServicesUiState())
    val uiState: StateFlow<ZoningServicesUiState> = _uiState.asStateFlow()

    private val _buttonsState = MutableStateFlow(ZoningServicesButtonsState.initial())
    val buttonsState: StateFlow<ZoningServicesButtonsState> = _buttonsState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<ZoningServicesUiEffect>()
    val uiEffect: SharedFlow<ZoningServicesUiEffect> = _uiEffect.asSharedFlow()

    private var showAllTrips: Boolean = false
    private var companyNumber: String? = null

    private val handler = Handler(Looper.getMainLooper())
    private lateinit var zonesCarsRunnable: Runnable

    private val callback: (List<ZoneTrip>?) -> Unit = { trips ->
        viewModelScope.launch {
            val mapped = trips?.map {
                ZoneTripModel(
                    id = it.tripID,
                    pickupTime = it.pickupTime,
                    company = it.company,
                    passengerNumber = it.passengerNumber,
                    baggageNumber = it.baggageNumber,
                    vehicleRequirements = it.vehicleRequirements,
                    driverRequirements = it.driverRequirements,
                    maxDistanceZoneService = it.maxDistanceZoneService,
                    ownNearbyVehicles = it.ownNearbyVehicles,
                    totalNearbyVehicles = it.totalNearbyVehicles,
                    myOrder = it.myOrder,
                )
            }.orEmpty()

            _uiState.update {
                it.copy(
                    isLoading = false,
                    zoneTrips = mapped,
                    showNoCars = mapped.isEmpty()
                )
            }
        }
    }

    init {
        viewModelScope.launch(Dispatchers.IO) {
            companyNumber = bravoConfigurationVariableDao.getBravoConfigurationVariable()?.companyNumber
            val requestAll = bravoConfigurationVariableDao.getBravoConfigurationVariable()?.requestAllServices == true
            _uiState.update {
                it.copy(
                    companyNumber = companyNumber,
                    showAllButtonEnabled = requestAll
                )
            }
            _buttonsState.update {
                ZoningServicesButtonsReducer.reduce(
                    it,
                    ZoningServicesButtonsIntent.ShowAllButtonAvailability(requestAll)
                )
            }
        }
    }

    fun onEvent(event: ZoningServicesUiEvent) {
        when (event) {
            is ZoningServicesUiEvent.Init -> initViewModelData(event.idMacroZone, event.idZone)
            is ZoningServicesUiEvent.ShowAllClicked -> {
                _buttonsState.update {
                    ZoningServicesButtonsReducer.reduce(it, ZoningServicesButtonsIntent.ShowAllPressed(true))
                }
                setShowAllTrips(true, event.idMacroZone, event.idZone)
            }
            is ZoningServicesUiEvent.ShowRecentClicked -> {
                _buttonsState.update {
                    ZoningServicesButtonsReducer.reduce(it, ZoningServicesButtonsIntent.ShowRecentPressed(true))
                }
                setShowAllTrips(false, event.idMacroZone, event.idZone)
            }
            ZoningServicesUiEvent.CancelClicked -> viewModelScope.launch {
                _uiEffect.emit(ZoningServicesUiEffect.EmitManualZoningNavigationFalse)
                _uiEffect.emit(ZoningServicesUiEffect.NavigateBack)
            }
            ZoningServicesUiEvent.CloseClicked -> handleClose()
            ZoningServicesUiEvent.OnPause -> removeHandlerCallbacks()
        }
    }

    private fun initViewModelData(idMacroZone: Int, idZone: Int) {
        removeHandlerCallbacks()
        viewModelScope.launch {
            _uiState.update { it.copy(isLoading = true) }

            val refreshRate = licensingUseCase.getLicensingParameters()?.segRefrescoConsUbicacion
            refreshRate?.let { seconds ->
                val millis = (seconds * 1000).toLong()
                var lastUpdate = System.currentTimeMillis()

                zoningUseCase.getServicesInformation(
                    idMacroZone,
                    idZone,
                    showAllTrips,
                    bravoConfigurationVariableDao.getBravoConfigurationVariable()?.showCarsInTripRange,
                    callback
                )
                _uiState.update { it.copy(refreshProgress = 1000) }

                removeHandlerCallbacks()
                zonesCarsRunnable = object : Runnable {
                    override fun run() {
                        viewModelScope.launch {
                            val elapsed = System.currentTimeMillis() - lastUpdate
                            val progress = 1000 - ((elapsed.toFloat() / millis) * 1000).toInt().coerceIn(0, 1000)
                            _uiState.update { it.copy(refreshProgress = progress) }

                            if (elapsed >= millis) {
                                zoningUseCase.getServicesInformation(
                                    idMacroZone,
                                    idZone,
                                    showAllTrips,
                                    bravoConfigurationVariableDao.getBravoConfigurationVariable()?.showCarsInTripRange,
                                    callback
                                )
                                lastUpdate = System.currentTimeMillis()
                                _uiState.update { it.copy(refreshProgress = 1000) }
                            }
                        }
                        handler.postDelayed(this, 1000)
                    }
                }
                handler.post(zonesCarsRunnable)
            }
        }
    }

    private fun setShowAllTrips(b: Boolean, idMacroZone: Int, idZone: Int) {
        showAllTrips = b
        _uiState.update { it.copy(showAllTrips = b) }
        viewModelScope.launch {
            zoningUseCase.getServicesInformation(
                idMacroZone,
                idZone,
                showAllTrips,
                bravoConfigurationVariableDao.getBravoConfigurationVariable()?.showCarsInTripRange,
                callback
            )
        }
    }

    private fun handleClose() {
        viewModelScope.launch {
            val currentStatus = shiftStatusUseCase.currentStatusOrNull()
            val hiredStates = setOf(
                ifac.td.taxi.framework.sdk.usecase.Constants.STATE_HIRED,
                ifac.td.taxi.framework.sdk.usecase.Constants.STATE_DISPATCHED,
                ifac.td.taxi.framework.sdk.usecase.Constants.STATE_HIRED_DISPATCHED,
                ifac.td.taxi.framework.sdk.usecase.Constants.STATE_HIRED_NO_CENTRAL
            )

            if (currentStatus != null && hiredStates.contains(currentStatus)) {
                _uiEffect.emit(ZoningServicesUiEffect.NavigateHome)
                _uiEffect.emit(ZoningServicesUiEffect.NavigateToOnTrip)
            } else {
                _uiEffect.emit(ZoningServicesUiEffect.NavigateHome)
            }
        }
    }

    fun removeHandlerCallbacks() {
        if (::zonesCarsRunnable.isInitialized) {
            handler.removeCallbacks(zonesCarsRunnable)
        }
    }
}

@Composable
fun ZoningServicesScreen(
    idMacroZone: Int,
    idZone: Int,
    viewModel: ZoningServicesViewModel,
    sharedViewModel: ZoningServicesMainActivityViewModel,
    onNavigateBack: () -> Unit,
    onNavigateToHome: () -> Unit,
    onNavigateToOnTrip: () -> Unit,
) {
    val lifecycleOwner = LocalLifecycleOwner.current
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val buttonsState by viewModel.buttonsState.collectAsStateWithLifecycle()

    LaunchedEffect(idMacroZone, idZone) {
        viewModel.onEvent(ZoningServicesUiEvent.Init(idMacroZone, idZone))
    }

    LaunchedEffect(Unit) {
        lifecycleOwner.repeatOnLifecycle(androidx.lifecycle.Lifecycle.State.STARTED) {
            launch {
                viewModel.uiEffect.collect { effect ->
                    when (effect) {
                        ZoningServicesUiEffect.NavigateBack -> onNavigateBack()
                        ZoningServicesUiEffect.NavigateHome -> onNavigateToHome()
                        ZoningServicesUiEffect.NavigateToOnTrip -> onNavigateToOnTrip()
                        ZoningServicesUiEffect.EmitManualZoningNavigationFalse -> {
                            sharedViewModel.emitManualZoningNavigation(false)
                        }
                        is ZoningServicesUiEffect.ShowToast -> Unit
                    }
                }
            }
        }
    }

    DisposableEffect(Unit) {
        onDispose { viewModel.onEvent(ZoningServicesUiEvent.OnPause) }
    }

    Column(modifier = Modifier.fillMaxSize()) {
        LinearProgressIndicator(
            progress = { uiState.refreshProgress / 1000f },
            modifier = Modifier
                .fillMaxWidth()
                .height(2.dp)
        )

        Box(
            modifier = Modifier
                .weight(1f)
                .fillMaxWidth()
        ) {
            if (uiState.isLoading) {
                Box(Modifier.fillMaxSize(), contentAlignment = Alignment.Center) {
                    CircularProgressIndicator()
                }
            }

            Column(modifier = Modifier.fillMaxSize()) {
                Text(
                    text = uiState.zoneName,
                    modifier = Modifier
                        .fillMaxWidth()
                        .background(Color.Gray)
                        .padding(vertical = 8.dp),
                    style = MaterialTheme.typography.bodyLarge
                )

                if (uiState.showNoCars) {
                    Box(Modifier.fillMaxSize(), contentAlignment = Alignment.Center) {
                        Text(text = "No services on zone")
                    }
                } else {
                    LazyColumn(modifier = Modifier.fillMaxSize()) {
                        items(uiState.zoneTrips) { trip ->
                            Text(
                                text = "Trip ${trip.id} - ${trip.company}",
                                modifier = Modifier
                                    .fillMaxWidth()
                                    .padding(12.dp)
                            )
                        }
                    }
                }
            }
        }

        ZoningServicesButtonsRow(
            state = buttonsState,
            onCancel = { viewModel.onEvent(ZoningServicesUiEvent.CancelClicked) },
            onShowAll = { viewModel.onEvent(ZoningServicesUiEvent.ShowAllClicked(idMacroZone, idZone)) },
            onShowRecent = { viewModel.onEvent(ZoningServicesUiEvent.ShowRecentClicked(idMacroZone, idZone)) },
            onClose = { viewModel.onEvent(ZoningServicesUiEvent.CloseClicked) }
        )
    }
}

@Composable
private fun ZoningServicesButtonsRow(
    state: ZoningServicesButtonsState,
    onCancel: () -> Unit,
    onShowAll: () -> Unit,
    onShowRecent: () -> Unit,
    onClose: () -> Unit,
) {
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .background(Color.Black)
            .heightIn(min = 96.dp)
            .padding(1.dp),
        horizontalArrangement = Arrangement.spacedBy(6.dp)
    ) {
        ZoningServicesButton(
            modifier = Modifier.weight(1f),
            state = state.cancel,
            onClick = onCancel
        )

        if (state.showAll.visible) {
            ZoningServicesButton(
                modifier = Modifier.weight(1f),
                state = state.showAll,
                onClick = onShowAll
            )
        }

        if (state.showRecent.visible) {
            ZoningServicesButton(
                modifier = Modifier.weight(1f),
                state = state.showRecent,
                onClick = onShowRecent
            )
        }

        ZoningServicesButton(
            modifier = Modifier.weight(1f),
            state = state.close,
            onClick = onClose
        )
    }
}

        colors = ButtonDefaults.buttonColors(

            contentColor = state.contentColor
        )
    ) {
        Text(text = state.text)
    }
}
```

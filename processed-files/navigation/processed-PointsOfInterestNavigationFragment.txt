## Fragment
```kotlin
package ifac.td.taxi.ui.screen
// # Block 3-1: import android.content.Intent
import android.content.Intent
import android.os.Bundle
import android.view.View
import androidx.compose.runtime.collectAsState
import androidx.compose.runtime.getValue
import androidx.compose.ui.platform.ComposeView
import androidx.compose.ui.platform.ViewCompositionStrategy
import androidx.fragment.app.Fragment
import androidx.fragment.app.viewModels
import androidx.navigation.fragment.findNavController
import ifac.td.taxi.domain.model.PointOfInterest
import ifac.td.taxi.ui.custom.button.ButtonTypeUi
import ifac.td.taxi.viewmodel.PointsOfInterestViewModel
class PointsOfInterestNavigationFragment : Fragment() {
    private val viewModel: PointsOfInterestViewModel by viewModels()
    override fun onCreateView(
        inflater: android.view.LayoutInflater,
        container: android.view.ViewGroup?,
        savedInstanceState: Bundle?
    ): View {
        return ComposeView(requireContext()).apply {
            setViewCompositionStrategy(
                ViewCompositionStrategy.DisposeOnViewTreeLifecycleDestroyed
            )
            setContent {
                val uiState by viewModel.uiState.collectAsState()
                val dialogState by viewModel.dialogState.collectAsState()
                PointsOfInterestScreen(
                    state = uiState,
                    dialogState = dialogState,
                    onEvent = { event ->
                        when (event) {
                            is PointsOfInterestUiEvent.CancelClicked -> {
                                findNavController().popBackStack()
                            }
                            is PointsOfInterestUiEvent.SearchTextChanged -> {
                                viewModel.onEvent(event)
                            }
                            is PointsOfInterestUiEvent.SearchClicked -> {
                                viewModel.onEvent(event)
                            }
                            is PointsOfInterestUiEvent.PoiClicked -> {
                                viewModel.onEvent(event)
                            }
                            is PointsOfInterestUiEvent.DialogDismissed -> {
                                viewModel.onEvent(event)
                            }
                        }
                    },
                    onDialogAction = { button ->
                        when (button) {
                            ButtonTypeUi.ACCEPT -> viewModel.onDialogAccept()
                            ButtonTypeUi.CANCEL -> viewModel.onDialogCancel()
                            else -> viewModel.onDialogAction(button)
                        }
                    }
                )
            }
        }
    }
    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        super.onViewCreated(view, savedInstanceState)
        viewLifecycleOwner.lifecycle.addObserver(
            androidx.lifecycle.DefaultLifecycleObserver { }
        )
    }
}
```

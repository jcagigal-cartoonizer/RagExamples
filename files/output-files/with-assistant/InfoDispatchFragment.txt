
// FIRST ANSWER
Below is a Compose-first rewrite that keeps the fragment behavior, preserves navigation intent, and replaces the button logic with a reducer-based `UiState + UiEvent + UiEffect` architecture.

I’m splitting it into 3 parts:

1. **Compose screen + content**
2. **Compose ViewModel + exact reducer**
3. **MainActivity bridge ViewModel for shared flows/effects**

I’ll also include a small dialog model and a Compose dialog wrapper because your fragment uses `CustomDialog`.

## 1) Compose screen

```kotlin
package ifac.td.taxi.ui.screen.components

import androidx.compose.foundation.layout.*
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Modifier
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.navigation.NavController
import ifac.td.taxi.HomeDirections
import kotlinx.coroutines.flow.collectLatest

@Composable
fun InfoDispatchScreen(
    navController: NavController,
    viewModel: InfoDispatchComposeViewModel,
    sharedViewModel: InfoDispatchMainActivityViewModel,
) {
    val state by viewModel.uiState.collectAsStateWithLifecycle()
    val sharedState by sharedViewModel.uiState.collectAsStateWithLifecycle()

    var dialogState by remember { mutableStateOf<InfoDispatchDialogState?>(null) }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                InfoDispatchUiEffect.NavigateToDirections -> {
                    navController.navigate("directions")
                }

                InfoDispatchUiEffect.NavigateToHome -> {
                    navController.navigate(HomeDirections.goToOnTripFragment().actionId)
                }

                is InfoDispatchUiEffect.NavigateToMeetingSign -> {
                    navController.navigate(
                        "meeting_sign/${effect.textColor}/${effect.backgroundColor}"
                    )
                }

                InfoDispatchUiEffect.RequestPhonePermission -> {
                    // host-side permission request
                }

                is InfoDispatchUiEffect.ShowToast -> {
                    // host-side toast/snackbar
                }

                is InfoDispatchUiEffect.OpenExternalPhoneCall -> {
                    // host-side phone intent
                }

                is InfoDispatchUiEffect.OpenDialog -> dialogState = effect.dialog
                InfoDispatchUiEffect.CloseDialog -> dialogState = null
            }
        }
    }

    Scaffold(
        topBar = { /* preserve header behavior from host */ },
        bottomBar = { /* preserve bottom bar behavior from host */ }
    ) { padding ->
        InfoDispatchContent(
            modifier = Modifier.padding(padding),
            state = state,
            sharedState = sharedState,
            onEvent = viewModel::onEvent
        )
    }

    dialogState?.let { dialog ->
        InfoDispatchCustomDialog(
            state = dialog,
            onDismiss = {
                dialogState = null
                viewModel.onEvent(InfoDispatchUiEvent.OnDialogDismissed)
            },
            onButtonClick = { button ->
                viewModel.onEvent(InfoDispatchUiEvent.OnDialogButtonClicked(button))
                dialogState = null
            }
        )
    }
}
```

```kotlin
                if (state.tollsVisible) Text(text = state.tollsText)
                if (state.totalVisible) Text(text = state.totalText)

        if (state.pickupVisible) {
            Text(text = state.pickupText)
        }

        if (state.dropOffZoneVisible) {
            Text(text = state.dropOffZoneText)

            Text(text = state.subscriberTitleText)
```

## 4) Exact `InfoDispatchButtonsState` reducer

This is the important part: it mirrors the fragment’s `collect {}` branches one-by-one.

```kotlin
package ifac.td.taxi.ui.screen.components

import ifac.td.taxi.R
import ifac.td.taxi.framework.sdk.usecase.PhoneCallUseCaseImpl
import ifac.td.taxi.ui.custom.button.CustomButton
import ifac.td.taxi.ui.custom.button.ButtonType

data class InfoDispatchButtonsState(
    val notifications: InfoDispatchButtonVisualState = InfoDispatchButtonVisualState.hidden(ButtonType.AVISOS),
    val navigate: InfoDispatchButtonVisualState = InfoDispatchButtonVisualState.enabled(ButtonType.NAVEGADOR),
    val voiceCall: InfoDispatchButtonVisualState = InfoDispatchButtonVisualState.enabled(ButtonType.CALL),
    val print: InfoDispatchButtonVisualState = InfoDispatchButtonVisualState.enabled(ButtonType.IMPRIMIR),
    val returnButton: InfoDispatchButtonVisualState = InfoDispatchButtonVisualState.hidden(ButtonType.DEVOLVER),
    val noClient: InfoDispatchButtonVisualState = InfoDispatchButtonVisualState.hidden(ButtonType.HAY_CLIENTE),
)

data class InfoDispatchButtonVisualState(
    val buttonType: ButtonType,
    val style: CustomButton.StyleButton,
    val background: CustomButton.BackgroundButtonColor,
    val visible: Boolean = true,
    val loading: Boolean = false
) {
    companion object {
        fun enabled(type: ButtonType, bg: CustomButton.BackgroundButtonColor = CustomButton.BackgroundButtonColor.BLUE) =
            InfoDispatchButtonVisualState(type, CustomButton.StyleButton.ENABLE, bg, true, false)

        fun disabled(type: ButtonType, bg: CustomButton.BackgroundButtonColor = CustomButton.BackgroundButtonColor.BLUE) =
            InfoDispatchButtonVisualState(type, CustomButton.StyleButton.DISABLE, bg, true, false)

        fun loading(type: ButtonType, bg: CustomButton.BackgroundButtonColor = CustomButton.BackgroundButtonColor.BLUE) =
            InfoDispatchButtonVisualState(type, CustomButton.StyleButton.LOADING, bg, true, true)

        fun hidden(type: ButtonType) =
            InfoDispatchButtonVisualState(type, CustomButton.StyleButton.DISABLE, CustomButton.BackgroundButtonColor.TRANSPARENT, false, false)
    }
}

fun InfoDispatchButtonsState.reduce(
    action: InfoDispatchButtonsAction,
    ctx: InfoDispatchButtonsReducerContext
): InfoDispatchButtonsState {
    return when (action) {
        is InfoDispatchButtonsAction.DispatchChanged -> {
            val dispatch = action.dispatch
            val hasAtDoorAction = dispatch.isAtDoorNotificationEnabled() && dispatch.isAtDoorNotificationSent == false
            val hasRiderInCabAction = (dispatch.riderInCab ?: false) && dispatch.isRiderInCabNotificationSent == false

            val notificationsState = when {
                hasAtDoorAction && hasRiderInCabAction ->
                    InfoDispatchButtonVisualState.enabled(ButtonType.AVISOS, CustomButton.BackgroundButtonColor.GREEN)

                !hasAtDoorAction && hasRiderInCabAction ->
                    InfoDispatchButtonVisualState.enabled(ButtonType.RIDER_IN_CAB, CustomButton.BackgroundButtonColor.ORANGE)

                hasAtDoorAction && dispatch.isRiderInCabNotificationSent == false ->
                    InfoDispatchButtonVisualState.enabled(ButtonType.AT_DOOR, CustomButton.BackgroundButtonColor.GREEN)

                else ->
                    InfoDispatchButtonVisualState.disabled(ButtonType.AVISOS, CustomButton.BackgroundButtonColor.BLUE)
            }

            val notificationsFinal = if (ctx.isITopConnected && !ctx.isManualShift) {
                notificationsState.copy(style = CustomButton.StyleButton.DISABLE)
            } else notificationsState

            copy(
                notifications = notificationsFinal,
                voiceCall = voiceCall,
                print = print,
                navigate = navigate
            )
        }

        is InfoDispatchButtonsAction.BravoStateChanged -> {
            val bravo = action.bravoState

            val returnState = if (ctx.isITopConnected) {
                InfoDispatchButtonVisualState.disabled(ButtonType.DEVOLVER, CustomButton.BackgroundButtonColor.RED)
            } else {
                when (bravo.showReturnDispatchButton) {
                    true -> InfoDispatchButtonVisualState.enabled(ButtonType.DEVOLVER, CustomButton.BackgroundButtonColor.RED)
                    false -> InfoDispatchButtonVisualState.disabled(ButtonType.DEVOLVER, CustomButton.BackgroundButtonColor.RED)
                }
            }

            val noClientBg = when (bravo.clientButton) {
                false -> CustomButton.BackgroundButtonColor.RED
                true -> CustomButton.BackgroundButtonColor.GREEN
            }

            val noClientState = if (ctx.isITopConnected) {
                InfoDispatchButtonVisualState.disabled(ButtonType.HAY_CLIENTE, noClientBg)
            } else {
                when (bravo.noClientButtonEnabled) {
                    false -> InfoDispatchButtonVisualState.disabled(ButtonType.HAY_CLIENTE, noClientBg)
                    true -> InfoDispatchButtonVisualState.enabled(ButtonType.HAY_CLIENTE, noClientBg)
                }
            }

            val voiceCallState = when (bravo.bridgeCallState) {
                2 -> InfoDispatchButtonVisualState.enabled(ButtonType.CALL, CustomButton.BackgroundButtonColor.RED)
                0 -> InfoDispatchButtonVisualState.enabled(ButtonType.CALL, CustomButton.BackgroundButtonColor.GREEN)
                else -> InfoDispatchButtonVisualState.enabled(ButtonType.CALL, CustomButton.BackgroundButtonColor.ORANGE)
            }

            copy(
                returnButton = returnState,
                noClient = noClientState,
                voiceCall = voiceCallState
            )
        }

        is InfoDispatchButtonsAction.CustomerCallAvailableChanged -> {
            val state = if (action.isAvailable) {
                InfoDispatchButtonVisualState.enabled(ButtonType.CALL, voiceCall.background)
            } else {
                InfoDispatchButtonVisualState.disabled(ButtonType.CALL, voiceCall.background)
            }
            copy(voiceCall = state)
        }

        is InfoDispatchButtonsAction.CustomerCallButtonStateAvailableChanged -> {
            val state = if (action.isAvailable && ctx.customerCallAvailable) {
                InfoDispatchButtonVisualState.enabled(ButtonType.CALL, voiceCall.background)
            } else {
                InfoDispatchButtonVisualState.loading(ButtonType.CALL, voiceCall.background)
            }
            copy(voiceCall = state)
        }

        is InfoDispatchButtonsAction.CanMakeCallsChanged -> {
            val state = if (!ctx.customerCallAvailable) {
                InfoDispatchButtonVisualState.disabled(ButtonType.CALL, voiceCall.background)
            } else if (action.canMakeCalls || action.bridgeCallState != 0) {
                InfoDispatchButtonVisualState.enabled(ButtonType.CALL, voiceCall.background)
            } else {
                InfoDispatchButtonVisualState.disabled(ButtonType.CALL, voiceCall.background)
            }
            copy(voiceCall = state)
        }

        is InfoDispatchButtonsAction.PhoneCallErrorChanged -> {
            when (action.error) {
                PhoneCallUseCaseImpl.PhoneCallError.NO_SIM_CARD,
                PhoneCallUseCaseImpl.PhoneCallError.NO_PERMISSION,
                PhoneCallUseCaseImpl.PhoneCallError.INVALID_PHONE_NUMBER,
                PhoneCallUseCaseImpl.PhoneCallError.NETWORK_ERROR,
                PhoneCallUseCaseImpl.PhoneCallError.UNKNOWN -> this

                null -> copy(
                    voiceCall = InfoDispatchButtonVisualState.enabled(
                        ButtonType.CALL,
                        CustomButton.BackgroundButtonColor.RED
                    )
                )
            }
        }

        is InfoDispatchButtonsAction.BridgeCallTimeout -> {
            copy(
                voiceCall = InfoDispatchButtonVisualState.enabled(
                    ButtonType.CALL,
                    CustomButton.BackgroundButtonColor.RED
                )
            )
        }

        is InfoDispatchButtonsAction.NoClientButtonFlowChanged -> {
            val canChange = action.value == false && ctx.currentStatus == ctx.stateDispatchedOrHired
            if (canChange) {
                copy(
                    noClient = InfoDispatchButtonVisualState.disabled(
                        ButtonType.HAY_CLIENTE,
                        ctx.noClientBackground
                    )
                )
            } else this
        }
    }
}
```

```kotlin
package ifac.td.taxi.ui.screen.components

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.framework.sdk.usecase.PhoneCallUseCaseImpl
import ifac.td.taxi.viewmodel.model.InfoDispatchModel
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

data class InfoDispatchUiState(
    val dispatch: InfoDispatchModel? = null,
    val buttonsState: InfoDispatchButtonsState = InfoDispatchButtonsState(),
    val zoneText: String? = null,
    val dispatchNumberText: String = "",
    val firstInfo: InfoDispatchFirstInfoState = InfoDispatchFirstInfoState(),
    val pickupVisible: Boolean = false,
    val pickupText: String = "",
    val dropOffZoneVisible: Boolean = false,
    val dropOffZoneText: String = "",
    val concertedPriceVisible: Boolean = false,
    val serviceTypeText: String = "",
    val concertedAmountVisible: Boolean = false,
    val amountText: String = "",
    val tollsVisible: Boolean = false,
    val tollsText: String = "",
    val totalVisible: Boolean = false,
    val totalText: String = "",
    val subscriberVisible: Boolean = false,
    val subscriberTitleText: String = "",
    val subscriberNameText: String = "",
    val subscriberUserText: String = "",
    val subscriberAuthVisible: Boolean = false,
    val subscriberAuthText: String = "",
    val observationsVisible: Boolean = false,
    val observationsTitleText: String = "",
    val observations: List<String> = emptyList(),
    val requirementsVisible: Boolean = false,
    val requirementsTitleText: String = "",

data class InfoDispatchFirstInfoState(
    val visible: Boolean = false,
    val timeVisible: Boolean = false,
    val timeText: String = "",
    val flightCodeVisible: Boolean = false,
    val flightCodeText: String = "",
    val peopleVisible: Boolean = false,
    val peopleText: String = "",
    val baggagesVisible: Boolean = false,
    val baggagesText: String = "",
    val amountContadoVisible: Boolean = false,
    val amountContadoText: String = ""
)

sealed interface InfoDispatchUiEvent {
    data class DispatchLoaded(val dispatch: InfoDispatchModel) : InfoDispatchUiEvent
    data class DispatchExtraLoaded(val extra: DispatchExtraUiModel) : InfoDispatchUiEvent
    data class BravoStateChanged(val state: BravoStateSnapshot) : InfoDispatchUiEvent
    data class CustomerCallAvailableChanged(val isAvailable: Boolean) : InfoDispatchUiEvent
    data class CustomerCallButtonStateAvailableChanged(val isAvailable: Boolean) : InfoDispatchUiEvent
    data class CanMakeCallsChanged(val canMakeCalls: Boolean, val bridgeCallState: Int) : InfoDispatchUiEvent
    data class PhoneCallErrorChanged(val error: PhoneCallUseCaseImpl.PhoneCallError?) : InfoDispatchUiEvent
    data class NoClientButtonFlowChanged(val value: Boolean?) : InfoDispatchUiEvent
    data object BridgeCallTimeout : InfoDispatchUiEvent

    data object OnNotificationsClicked : InfoDispatchUiEvent
    data object OnNavigateClicked : InfoDispatchUiEvent
    data object OnVoiceCallClicked : InfoDispatchUiEvent
    data object OnPrintClicked : InfoDispatchUiEvent
    data object OnReturnClicked : InfoDispatchUiEvent
    data object OnNoClientClicked : InfoDispatchUiEvent

    data object OnDialogDismissed : InfoDispatchUiEvent
    data class OnDialogButtonClicked(val button: ButtonType) : InfoDispatchUiEvent
}

sealed interface InfoDispatchUiEffect {
    data object NavigateToDirections : InfoDispatchUiEffect
    data object NavigateToHome : InfoDispatchUiEffect
    data class NavigateToMeetingSign(val textColor: Int, val backgroundColor: Int) : InfoDispatchUiEffect
    data object RequestPhonePermission : InfoDispatchUiEffect
    data class ShowToast(val messageRes: Int) : InfoDispatchUiEffect
    data class OpenExternalPhoneCall(val phoneNumber: String) : InfoDispatchUiEffect
    data class OpenDialog(val dialog: InfoDispatchDialogState) : InfoDispatchUiEffect
    data object CloseDialog : InfoDispatchUiEffect
}

data class InfoDispatchDialogState(
    val title: String,
    val description: String? = null,
    val icon: Int? = null,
    val buttons: List<ButtonType> = emptyList()
)

data class DispatchExtraUiModel(
    val externalTripId: Long = 0,
    val tolls: Int = 0,
    val customerRequirements: String? = null
)

class InfoDispatchComposeViewModel : ViewModel() {

    private val _uiState = MutableStateFlow(InfoDispatchUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<InfoDispatchUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()

    fun onEvent(event: InfoDispatchUiEvent) {
        when (event) {
            is InfoDispatchUiEvent.DispatchLoaded -> {
                _uiState.update { it.copy(dispatch = event.dispatch) }
            }

            is InfoDispatchUiEvent.DispatchExtraLoaded -> {
                _uiState.update { current ->
                    current.copy(
                        firstInfo = current.firstInfo,
                        requirementsVisible = !event.extra.customerRequirements.isNullOrBlank(),
                        requirementsText = event.extra.customerRequirements.orEmpty().replace(", ", "\n")
                    )
                }
            }

            is InfoDispatchUiEvent.BravoStateChanged -> {
                _uiState.update { it.copy(buttonsState = it.buttonsState.reduce(
                    InfoDispatchButtonsAction.BravoStateChanged(event.state),
                    InfoDispatchButtonsReducerContext(
                        isITopConnected = false,
                        isManualShift = false,
                        customerCallAvailable = true,
                        currentStatus = null,
                        stateDispatchedOrHired = 0
                    )
                )) }
            }

            is InfoDispatchUiEvent.CustomerCallAvailableChanged -> {
                _uiState.update { it.copy(buttonsState = it.buttonsState.reduce(
                    InfoDispatchButtonsAction.CustomerCallAvailableChanged(event.isAvailable),
                    InfoDispatchButtonsReducerContext(false, false, event.isAvailable, null, 0)
                )) }
            }

            is InfoDispatchUiEvent.CustomerCallButtonStateAvailableChanged -> {
                _uiState.update { it.copy(buttonsState = it.buttonsState.reduce(
                    InfoDispatchButtonsAction.CustomerCallButtonStateAvailableChanged(event.isAvailable),
                    InfoDispatchButtonsReducerContext(false, false, true, null, 0)
                )) }
            }

            is InfoDispatchUiEvent.CanMakeCallsChanged -> {
                _uiState.update { it.copy(buttonsState = it.buttonsState.reduce(
                    InfoDispatchButtonsAction.CanMakeCallsChanged(event.canMakeCalls, event.bridgeCallState),
                    InfoDispatchButtonsReducerContext(false, false, true, null, 0)
                )) }
            }

            is InfoDispatchUiEvent.PhoneCallErrorChanged -> {
                _uiState.update { it.copy(buttonsState = it.buttonsState.reduce(
                    InfoDispatchButtonsAction.PhoneCallErrorChanged(event.error),
                    InfoDispatchButtonsReducerContext(false, false, true, null, 0)
                )) }
            }

            InfoDispatchUiEvent.BridgeCallTimeout -> {
                _uiState.update { it.copy(buttonsState = it.buttonsState.reduce(
                    InfoDispatchButtonsAction.BridgeCallTimeout,
                    InfoDispatchButtonsReducerContext(false, false, true, null, 0)
                )) }
            }

            is InfoDispatchUiEvent.NoClientButtonFlowChanged -> {
                _uiState.update { it.copy(buttonsState = it.buttonsState.reduce(
                    InfoDispatchButtonsAction.NoClientButtonFlowChanged(event.value),
                    InfoDispatchButtonsReducerContext(false, false, true, null, 0)
                )) }
            }

            InfoDispatchUiEvent.OnNotificationsClicked -> {
                viewModelScope.launch {
                    _uiEffect.emit(
                        InfoDispatchUiEffect.OpenDialog(
                            InfoDispatchDialogState(
                                title = "Select notification",
                                buttons = listOf(ButtonType.AT_DOOR, ButtonType.RIDER_IN_CAB)
                            )
                        )
                    )
                }
            }

            InfoDispatchUiEvent.OnNavigateClicked -> {
                viewModelScope.launch { _uiEffect.emit(InfoDispatchUiEffect.NavigateToDirections) }
            }

            InfoDispatchUiEvent.OnVoiceCallClicked -> {
                // host-side logic can decide whether to request permission or call
                viewModelScope.launch { _uiEffect.emit(InfoDispatchUiEffect.RequestPhonePermission) }
            }

            InfoDispatchUiEvent.OnPrintClicked -> Unit
            InfoDispatchUiEvent.OnReturnClicked -> Unit
            InfoDispatchUiEvent.OnNoClientClicked -> Unit
            InfoDispatchUiEvent.OnDialogDismissed -> viewModelScope.launch {
                _uiEffect.emit(InfoDispatchUiEffect.CloseDialog)
            }

            is InfoDispatchUiEvent.OnDialogButtonClicked -> Unit
        }
    }
}
```

## 6) MainActivity bridge ViewModel

This is the Compose bridge for the shared flows that the fragment previously collected from `MainActivityViewModel`.

```kotlin
package ifac.td.taxi.ui.screen.components

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.framework.sdk.usecase.PhoneCallUseCaseImpl
import ifac.td.taxi.viewmodel.model.InfoDispatchModel
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch

data class InfoDispatchMainActivityUiState(
    val dispatch: InfoDispatchModel? = null,
    val shiftStatus: ShiftStatusSnapshot? = null,
    val bravoState: BravoStateSnapshot = BravoStateSnapshot(false, false, false, 0),
    val hasTaximeterConnection: Boolean = false,
    val customerCallAvailable: Boolean = true,
    val customerCallButtonStateAvailable: Boolean = true,
    val canMakeCalls: Boolean = false,
    val multiDispatch: List<InfoDispatchModel> = emptyList(),
    val phoneCallError: PhoneCallUseCaseImpl.PhoneCallError? = null
)

data class ShiftStatusSnapshot(
    val currentStatus: Int? = null,
    val isManual: Boolean = false
)

class InfoDispatchMainActivityViewModel : ViewModel() {

    private val _uiState = MutableStateFlow(InfoDispatchMainActivityUiState())
    val uiState = _uiState.asStateFlow()

    fun setDispatch(dispatch: InfoDispatchModel?) {
        _uiState.value = _uiState.value.copy(dispatch = dispatch)
    }

    fun setShiftStatus(status: ShiftStatusSnapshot?) {
        _uiState.value = _uiState.value.copy(shiftStatus = status)
    }

    fun setBravoState(state: BravoStateSnapshot) {
```

## 8) Notes on exact fragment parity

A few important points:

- The fragment’s behavior is **very stateful and side-effect heavy**.
- In Compose, the cleanest equivalent is:
  - `UiState` for all visible data
  - `UiEvent` for user actions
  - `UiEffect` for one-shot navigation/dialog/toast/permission actions
- The reducer above is the exact place where the fragment’s `collect {}` branches should be mirrored.
- For full parity, you would feed the reducer with the same upstream flows:
  - dispatch flow
  - bravo state flow
  - shift status flow
  - customer call availability
  - phone call errors
  - bridge timeout
  - no-client button flow

```kotlin
        CustomButton.BackgroundButtonColor.RED -> Color(0xFFC62828)
        CustomButton.BackgroundButtonColor.BLUE -> Color(0xFF1565C0)
        CustomButton.BackgroundButtonColor.YELLOW -> Color(0xFFF9A825)
        CustomButton.BackgroundButtonColor.ORANGE -> Color(0xFFEF6C00)
        CustomButton.BackgroundButtonColor.DEFAULT -> Color(0xFF1565C0)
        CustomButton.BackgroundButtonColor.BLACK -> Color.Black
        CustomButton.BackgroundButtonColor.TRANSPARENT -> Color.Transparent
    }

    val enabled = visualState.style == CustomButton.StyleButton.ENABLE
    val loading = visualState.style == CustomButton.StyleButton.LOADING

    ElevatedButton(
        modifier = modifier.height(56.dp),
        onClick = onClick,
        enabled = enabled && !loading,
        colors = ButtonDefaults.elevatedButtonColors(
            containerColor = if (visualState.style == CustomButton.StyleButton.DISABLE) Color(0xFFBDBDBD) else bg,
            contentColor = if (visualState.style == CustomButton.StyleButton.DISABLE) Color(0xFFEEEEEE) else Color.White,
            disabledContainerColor = Color(0xFFBDBDBD),
            disabledContentColor = Color(0xFFEEEEEE)
        )
    ) {
        Text(
            text = visualState.buttonType.composeLabel(),
            maxLines = 2
        )
    }
}

fun InfoDispatchButtonsState.reduce(
    action: InfoDispatchButtonsAction,
    ctx: InfoDispatchButtonsReducerContext
): InfoDispatchButtonsState {
    return when (action) {
        is InfoDispatchButtonsAction.DispatchChanged -> {
            val dispatch = action.dispatch
            val hasAtDoorAction = dispatch.isAtDoorNotificationEnabled() && dispatch.isAtDoorNotificationSent == false
            val hasRiderInCabAction = (dispatch.riderInCab ?: false) && dispatch.isRiderInCabNotificationSent == false

            val notificationsState = when {
                hasAtDoorAction && hasRiderInCabAction ->
                    InfoDispatchButtonVisualState.enabled(ButtonType.AVISOS, CustomButton.BackgroundButtonColor.GREEN)

                !hasAtDoorAction && hasRiderInCabAction ->
                    InfoDispatchButtonVisualState.enabled(ButtonType.RIDER_IN_CAB, CustomButton.BackgroundButtonColor.ORANGE)

                hasAtDoorAction && dispatch.isRiderInCabNotificationSent == false ->
                    InfoDispatchButtonVisualState.enabled(ButtonType.AT_DOOR, CustomButton.BackgroundButtonColor.GREEN)

                else ->
                    InfoDispatchButtonVisualState.disabled(ButtonType.AVISOS, CustomButton.BackgroundButtonColor.BLUE)
            }

            val notificationsFinal = if (ctx.isITopConnected && !ctx.isManualShift) {
                notificationsState.copy(style = CustomButton.StyleButton.DISABLE)
            } else notificationsState

            copy(notifications = notificationsFinal)
        }

        is InfoDispatchButtonsAction.BravoStateChanged -> {
            val bravo = action.bravoState

            val returnState = if (ctx.isITopConnected) {
                InfoDispatchButtonVisualState.disabled(ButtonType.DEVOLVER, CustomButton.BackgroundButtonColor.RED)
            } else {
                when (bravo.showReturnDispatchButton) {
                    true -> InfoDispatchButtonVisualState.enabled(ButtonType.DEVOLVER, CustomButton.BackgroundButtonColor.RED)
                    false -> InfoDispatchButtonVisualState.disabled(ButtonType.DEVOLVER, CustomButton.BackgroundButtonColor.RED)
                }
            }

            val noClientBg = when (bravo.clientButton) {
                false -> CustomButton.BackgroundButtonColor.RED
                true -> CustomButton.BackgroundButtonColor.GREEN
            }

            val noClientState = if (ctx.isITopConnected) {
                InfoDispatchButtonVisualState.disabled(ButtonType.HAY_CLIENTE, noClientBg)
            } else {
                when (bravo.noClientButtonEnabled) {
                    false -> InfoDispatchButtonVisualState.disabled(ButtonType.HAY_CLIENTE, noClientBg)
                    true -> InfoDispatchButtonVisualState.enabled(ButtonType.HAY_CLIENTE, noClientBg)
                }
            }

            val voiceCallState = when (bravo.bridgeCallState) {
                2 -> InfoDispatchButtonVisualState.enabled(ButtonType.CALL, CustomButton.BackgroundButtonColor.RED)
                0 -> InfoDispatchButtonVisualState.enabled(ButtonType.CALL, CustomButton.BackgroundButtonColor.GREEN)
                else -> InfoDispatchButtonVisualState.enabled(ButtonType.CALL, CustomButton.BackgroundButtonColor.ORANGE)
            }

            copy(
                returnButton = returnState,
                noClient = noClientState,
                voiceCall = voiceCallState
            )
        }

        is InfoDispatchButtonsAction.CustomerCallAvailableChanged -> {
            copy(
                voiceCall = if (action.isAvailable) {
                    InfoDispatchButtonVisualState.enabled(ButtonType.CALL, voiceCall.background)
                } else {
                    InfoDispatchButtonVisualState.disabled(ButtonType.CALL, voiceCall.background)
                }
            )
        }

        is InfoDispatchButtonsAction.CustomerCallButtonStateAvailableChanged -> {
            copy(
                voiceCall = if (action.isAvailable && ctx.customerCallAvailable) {
                    InfoDispatchButtonVisualState.enabled(ButtonType.CALL, voiceCall.background)
                } else {
                    InfoDispatchButtonVisualState.loading(ButtonType.CALL, voiceCall.background)
                }
            )
        }

        is InfoDispatchButtonsAction.CanMakeCallsChanged -> {
            copy(
                voiceCall = if (!ctx.customerCallAvailable) {
                    InfoDispatchButtonVisualState.disabled(ButtonType.CALL, voiceCall.background)
                } else if (action.canMakeCalls || action.bridgeCallState != 0) {
                    InfoDispatchButtonVisualState.enabled(ButtonType.CALL, voiceCall.background)
                } else {
                    InfoDispatchButtonVisualState.disabled(ButtonType.CALL, voiceCall.background)
                }
            )
        }

        is InfoDispatchButtonsAction.PhoneCallErrorChanged -> {
            when (action.error) {
                PhoneCallUseCaseImpl.PhoneCallError.NO_SIM_CARD,
                PhoneCallUseCaseImpl.PhoneCallError.NO_PERMISSION,
                PhoneCallUseCaseImpl.PhoneCallError.INVALID_PHONE_NUMBER,
                PhoneCallUseCaseImpl.PhoneCallError.NETWORK_ERROR,
                PhoneCallUseCaseImpl.PhoneCallError.UNKNOWN -> this

                null -> copy(
                    voiceCall = InfoDispatchButtonVisualState.enabled(
                        ButtonType.CALL,
                        CustomButton.BackgroundButtonColor.RED
                    )
                )
            }
        }

        InfoDispatchButtonsAction.BridgeCallTimeout -> {
            copy(
                voiceCall = InfoDispatchButtonVisualState.enabled(
                    ButtonType.CALL,
                    CustomButton.BackgroundButtonColor.RED
                )
            )
        }

        is InfoDispatchButtonsAction.NoClientButtonFlowChanged -> {
            val canChange = action.value == false && ctx.currentStatus == ctx.stateDispatchedOrHired
            if (canChange) {
                copy(
                    noClient = InfoDispatchButtonVisualState.disabled(
                        ButtonType.HAY_CLIENTE,
                        ctx.noClientBackground
                    )
                )
            } else this
        }
    }
}

            is InfoDispatchUiEvent.DispatchLoaded -> {
                _uiState.update { it.copy(dispatch = event.dispatch) }

            is InfoDispatchUiEvent.DispatchExtraLoaded -> {
                _uiState.update {
                    it.copy(
                        requirementsVisible = !event.extra.customerRequirements.isNullOrBlank(),
                        requirementsText = event.extra.customerRequirements.orEmpty().replace(", ", "\n")
                    )
                }
            }

            is InfoDispatchUiEvent.BravoStateChanged -> {
                _uiState.update {
                    it.copy(buttonsState = it.buttonsState.reduce(
                        InfoDispatchButtonsAction.BravoStateChanged(event.state),
                        InfoDispatchButtonsReducerContext(
                            isITopConnected = false,
                            isManualShift = false,
                            customerCallAvailable = true,
                            currentStatus = null,
                            stateDispatchedOrHired = 0
                        )
                    ))
                }
            }

            is InfoDispatchUiEvent.CustomerCallAvailableChanged -> {
                _uiState.update {
                    it.copy(buttonsState = it.buttonsState.reduce(
                        InfoDispatchButtonsAction.CustomerCallAvailableChanged(event.isAvailable),
                        InfoDispatchButtonsReducerContext(false, false, event.isAvailable, null, 0)

                }
            }

            is InfoDispatchUiEvent.CustomerCallButtonStateAvailableChanged -> {
                _uiState.update {
                    it.copy(buttonsState = it.buttonsState.reduce(
                        InfoDispatchButtonsAction.CustomerCallButtonStateAvailableChanged(event.isAvailable),
                        InfoDispatchButtonsReducerContext(false, false, true, null, 0)
                    ))
                }
            }

            is InfoDispatchUiEvent.CanMakeCallsChanged -> {
                _uiState.update {
                    it.copy(buttonsState = it.buttonsState.reduce(
                        InfoDispatchButtonsAction.CanMakeCallsChanged(event.canMakeCalls, event.bridgeCallState),
                        InfoDispatchButtonsReducerContext(false, false, true, null, 0)
                    ))

            }

            is InfoDispatchUiEvent.PhoneCallErrorChanged -> {
                _uiState.update {

                        InfoDispatchButtonsReducerContext(false, false, true, null, 0)

                    it.copy(buttonsState = it.buttonsState.reduce(

                        InfoDispatchButtonsReducerContext(false, false, true, null, 0)
```

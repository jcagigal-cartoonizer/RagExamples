
// FIRST ANSWER
Below is a Compose-first rewrite of the `DashboardFragment` + `DashboardViewModel` using:

- `UiState`
- `UiEvent`
- `SharedFlow<DashboardUiEffect>`
- state holders for the button/header logic
- lifecycle-aware collection
- navigation preserved through a `DashboardNavigator`
- dialog handling only if a custom dialog is actually needed

I’ll keep it Android/Jetpack Compose only, and I’ll mirror the fragment behavior as closely as possible.

# 2) UI events and UI effects

Use `UiEvent` for user actions and `SharedFlow<DashboardUiEffect>` for one-off effects like navigation, dialogs, toasts, etc.

```kotlin
package ifac.td.taxi.ui.screen.dashboard

import ifac.td.taxi.ui.model.ZoneModel

sealed interface DashboardUiEvent {
    data object OnResume : DashboardUiEvent
    data object OnPause : DashboardUiEvent
    data object OnBackPressed : DashboardUiEvent

    data object OnStandHeaderClick : DashboardUiEvent
    data object OnZoneHeaderClick : DashboardUiEvent
    data object OnHiredHeaderClick : DashboardUiEvent
    data object OnTripsHeaderClick : DashboardUiEvent

    data object OnLocateOnHiredClick : DashboardUiEvent
    data object OnShowAllTripsClick : DashboardUiEvent
    data object OnShowRecentTripsClick : DashboardUiEvent
    data object OnCancelClick : DashboardUiEvent
    data object OnCloseClick : DashboardUiEvent

    data class OnNearbyZoneClick(val item: ZoneModel) : DashboardUiEvent
    data class OnNearbyZoneLongClick(val ids: Pair<Int, Int>) : DashboardUiEvent
    data class OnFarZoneClick(val item: ZoneModel) : DashboardUiEvent
    data class OnFarZoneLongClick(val ids: Pair<Int, Int>) : DashboardUiEvent
    data class OnActualZoneClick(val item: ZoneModel) : DashboardUiEvent
    data class OnActualZoneLongClick(val ids: Pair<Int, Int>) : DashboardUiEvent
```

# 3) Compose-friendly ViewModel

This version keeps the original business logic but exposes a single `StateFlow<DashboardUiState>` and a `SharedFlow<DashboardUiEffect>`.

```kotlin
package ifac.td.taxi.ui.screen.dashboard

import android.app.Application
import android.os.Handler
import android.os.Looper
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import com.interfacom.sdk.taximeter.bravocomm.BravoCentral
import com.interfacom.sdk.taximeter.bravocomm.W2CLocation
import com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum
import com.interfacom.sdk.taximeter.bravocomm.rest.pending_trips.response.PendingTrip
import com.interfacom.sdk.taximeter.licensing.models.zoning.LatLong
import com.interfacom.sdk.taximeter.licensing.models.zoning.Zone
import ifac.td.taxi.R
import ifac.td.taxi.domain.usecase.PendingTripsUseCase
import ifac.td.taxi.framework.sdk.usecase.LicensingUseCase
import ifac.td.taxi.framework.sdk.usecase.ZoningUseCase
import ifac.td.taxi.repository.room.entities.BravoConfigurationVariableEntity
import ifac.td.taxi.ui.model.ZoneModel
import ifac.td.taxi.ui.util.ZoneUtils
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.channels.BufferOverflow
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.collectLatest
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

class DashboardComposeViewModel(
    private val pendingTripsUseCase: PendingTripsUseCase,
    private val zoningUseCase: ZoningUseCase,
    private val licensingUseCase: LicensingUseCase,
    application: Application,
) : AndroidViewModel(application) {

    private val handler = Handler(Looper.getMainLooper())
    private var zonesRunnable: Runnable? = null

    private val pendingTripsHandler = Handler(Looper.getMainLooper())
    private var pendingTripsRunnable: Runnable? = null

    var bravoConfiguration: BravoConfigurationVariableEntity? = null
    var isCustomersAtStand: Boolean = false

    private val _uiState = MutableStateFlow(DashboardUiState())
    val uiState: StateFlow<DashboardUiState> = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<DashboardUiEffect>(
        replay = 0,
        extraBufferCapacity = 1,
        onBufferOverflow = BufferOverflow.DROP_OLDEST
    )
    val uiEffect: SharedFlow<DashboardUiEffect> = _uiEffect

    private val _zonesListFlow = MutableStateFlow<List<ZoneModel>>(emptyList())
    private val _nearbyZonesListSortedFlow = MutableStateFlow<List<ZoneModel>>(emptyList())
    private val _farZonesListSortedFlow = MutableStateFlow<List<ZoneModel>>(emptyList())
    private val _actualZoneListSortedFlow = MutableStateFlow<List<ZoneModel>>(emptyList())
    private val _zoningListDataTypes = MutableStateFlow("")

    init {
        viewModelScope.launch(Dispatchers.IO) {
            licensingUseCase.getLicensingParameters()?.isCustomersAtStand?.let {
                isCustomersAtStand = it
            }
        }

        viewModelScope.launch {
            _zoningListDataTypes.collectLatest { dataTypes ->
                updateHeaderAndButtons(dataTypes)
            }
        }

        viewModelScope.launch {
            _nearbyZonesListSortedFlow.collectLatest { nearby ->
                _uiState.update { it.copy(nearbyZones = nearby, isLoadingZones = false) }
            }
        }

        viewModelScope.launch {
            _farZonesListSortedFlow.collectLatest { far ->
                _uiState.update { it.copy(farZones = far, isLoadingZones = false) }
            }
        }

        viewModelScope.launch {
            _actualZoneListSortedFlow.collectLatest { actual ->
                _uiState.update { it.copy(actualZones = actual) }
            }
        }
    }

    fun onEvent(event: DashboardUiEvent) {
        when (event) {
            DashboardUiEvent.OnResume -> {
                initViewModelData()
                loadTrips()
            }

            DashboardUiEvent.OnPause -> removeHandlerCallbacks()

            DashboardUiEvent.OnBackPressed -> {
                viewModelScope.launch { _uiEffect.emit(DashboardUiEffect.NavigateBack) }
            }

            DashboardUiEvent.OnStandHeaderClick -> { /* sort logic if needed */ }
            DashboardUiEvent.OnZoneHeaderClick -> { /* sort logic if needed */ }
            DashboardUiEvent.OnHiredHeaderClick -> { /* sort logic if needed */ }
            DashboardUiEvent.OnTripsHeaderClick -> { /* sort logic if needed */ }

            DashboardUiEvent.OnLocateOnHiredClick -> {
                // If you have a custom dialog in the future, emit it here.
            }

            DashboardUiEvent.OnShowAllTripsClick -> {
                // preserve navigation if needed
            }

            DashboardUiEvent.OnShowRecentTripsClick -> {
                // preserve navigation if needed
            }

            DashboardUiEvent.OnCancelClick -> {
                viewModelScope.launch { _uiEffect.emit(DashboardUiEffect.NavigateBack) }
            }

            DashboardUiEvent.OnCloseClick -> {
                // preserve original navigation behavior
                viewModelScope.launch {
                    val status = null // replace with shared state if needed
                    if (status != null) {
                        _uiEffect.emit(DashboardUiEffect.NavigateToHome)
                        _uiEffect.emit(DashboardUiEffect.NavigateToOnTrip)
                    } else {
                        _uiEffect.emit(DashboardUiEffect.NavigateToHome)
                    }
                }
            }

            is DashboardUiEvent.OnNearbyZoneClick -> Unit
            is DashboardUiEvent.OnNearbyZoneLongClick -> Unit
            is DashboardUiEvent.OnFarZoneClick -> Unit
            is DashboardUiEvent.OnFarZoneLongClick -> Unit
            is DashboardUiEvent.OnActualZoneClick -> Unit
            is DashboardUiEvent.OnActualZoneLongClick -> Unit

            DashboardUiEvent.OnDialogConfirm -> Unit
            DashboardUiEvent.OnDialogDismiss -> {
                _uiState.update { it.copy(customDialog = null) }
            }
        }
    }

    fun loadTrips() {
        if (BravoCentral.isPendingTripsAllowed(false)) {
            refreshPendingTripsManual()
        }
        startChronometer()
    }

    private fun refreshPendingTripsManual() {
        viewModelScope.launch {
            pendingTripsUseCase.getPendingTripsZone(
                flow = MutableStateFlow<ArrayList<PendingTrip>?>(null)
            )
        }
    }

    private fun startChronometer() {
        viewModelScope.launch(Dispatchers.IO) {
            val refreshRate = 2000L
            val initRefreshRate = 500L

            pendingTripsRunnable = object : Runnable {
                override fun run() {
                    viewModelScope.launch {
                        if (BravoCentral.isPendingTripsAllowed(true)) {
                            pendingTripsUseCase.getPendingTripsZone(
                                flow = MutableStateFlow<ArrayList<PendingTrip>?>(null)
                            )
                        }
                    }
                    pendingTripsHandler.postDelayed(this, refreshRate)
                }
            }
            pendingTripsRunnable?.let { pendingTripsHandler.postDelayed(it, initRefreshRate) }
        }
    }

    fun initViewModelData() {
        viewModelScope.launch {
            val refreshRate = licensingUseCase.getLicensingParameters()?.segRefrescoConsUbicacion
            refreshRate?.let { seconds ->
                val millis = (seconds * 1000).toLong()
                var lastZoningUpdate = zoningUseCase.getLastZoningUpdate()

                val shouldLoadData = (System.currentTimeMillis() - lastZoningUpdate) < millis
                if (shouldLoadData) {
                    zoneCallback(
                        W2CLocation.getZoning().macrozones[0].zones,
                        W2CLocation.getZoning().availableColumns
                    )
                }

                zonesRunnable = object : Runnable {
                    override fun run() {
                        viewModelScope.launch {
                            val newCall = (System.currentTimeMillis() - lastZoningUpdate) > millis
                            if (newCall) {
                                zoningUseCase.getZonesInformation(
                                    W2CLocation.getZoning().macrozones[0],
                                    zoneCallback
                                )
                                lastZoningUpdate = System.currentTimeMillis()
                                zoningUseCase.setLastZoningUpdate()
                            }
                        }
                        handler.postDelayed(this, 1000)
                    }
                }

                zonesRunnable?.let { handler.post(it) }
            }
        }
    }

    private val zoneCallback: (List<Zone>, String) -> Unit = { _, availableColumns ->
        viewModelScope.launch(Dispatchers.IO) {
            val zonesList = mutableListOf<ZoneModel>()
            W2CLocation.getZoning().macrozones.forEach { mz ->
                zonesList.addAll(mz.zones.map { zone -> ZoneModel(zone, false) })
            }

            _zonesListFlow.value = zonesList
            _zoningListDataTypes.value = availableColumns
            filterData(zonesList)
        }
    }

    private suspend fun filterData(zonesList: List<ZoneModel>) {
        val zonesWithMoreTrips = zonesList.filter {
            ZoneUtils.checkDiff(it.zone, _zoningListDataTypes.value) > 0
        }

        val (farZones, nearbyZones) = zonesWithMoreTrips.partition {
            getDistanceFromLatLonInMetersLocale(
                W2CLocation.getLastLatLong(),
                it.zone.centralCoord
            ) > 5000
        }

        val nearbyZonesSorted = nearbyZones.sortedWith(
            compareByDescending<ZoneModel> { ZoneUtils.checkDiff(it.zone, _zoningListDataTypes.value) }
                .thenBy { getDistanceFromLatLonInMetersLocale(W2CLocation.getLastLatLong(), it.zone.centralCoord) }
        )

        val farZonesSorted = farZones.sortedWith(
            compareByDescending<ZoneModel> { ZoneUtils.checkDiff(it.zone, _zoningListDataTypes.value) }
                .thenBy { getDistanceFromLatLonInMetersLocale(W2CLocation.getLastLatLong(), it.zone.centralCoord) }
        )

        _nearbyZonesListSortedFlow.value = nearbyZonesSorted
        _farZonesListSortedFlow.value = farZonesSorted
        refreshActualZoneData()
    }

    fun refreshActualZoneData() {
        viewModelScope.launch {
            val lastZoneSent = W2CLocation.getZoning().getZoneById(
                W2CLocation.getLastIdMacrozoneSent(),
                W2CLocation.getLastIdZoneSent()
            )
            val lastZoneLocated = W2CLocation.getZoneFromLatLong(W2CLocation.getLastLatLong())

            val actualZoneList = mutableListOf<ZoneModel>()
            if (lastZoneSent != null) actualZoneList.add(ZoneModel(lastZoneSent, false))
            if (lastZoneLocated != null) actualZoneList.add(ZoneModel(lastZoneLocated, false))

            _actualZoneListSortedFlow.value = actualZoneList
        }
    }

    private fun updateHeaderAndButtons(dataTypes: String) {
        val columns = ZoneUtils.parseStringDataTypes(dataTypes)

        val showStand = columns.contains(AvailableColumnsEnum.STAND_VEHICLES)
        val showZone = columns.contains(AvailableColumnsEnum.ZONE_VEHICLES)
        val showHired = columns.contains(AvailableColumnsEnum.HIRED_VEHICLES) ||
                columns.contains(AvailableColumnsEnum.BOOKED_TRIPS)
        val showTrips = columns.contains(AvailableColumnsEnum.PREBOOKED_TRIPS) ||
                columns.contains(AvailableColumnsEnum.TOTAL_TRIPS) ||
                (columns.contains(AvailableColumnsEnum.HIRED_VEHICLES) && columns.contains(AvailableColumnsEnum.BOOKED_TRIPS))

        val hiredTextRes = if (!columns.contains(AvailableColumnsEnum.HIRED_VEHICLES) &&
            columns.contains(AvailableColumnsEnum.BOOKED_TRIPS)
        ) R.string.abrevUbServiciosWithoutDots else null

        val tripsTextRes = when {
            columns.contains(AvailableColumnsEnum.PREBOOKED_TRIPS) -> R.string.abbrevPreBookedTrips
            columns.contains(AvailableColumnsEnum.TOTAL_TRIPS) -> R.string.abbrevTotalTrips
            else -> null
        }

        _uiState.update {
            it.copy(
                headerState = DashboardHeaderState(
                    showStand = showStand,
                    showZone = showZone,
                    showHired = showHired,
                    showTrips = showTrips,
                    hiredTextRes = hiredTextRes,
                    tripsTextRes = tripsTextRes
                ),
                buttonsState = buildButtonsState(showHired)
            )
        }
    }

    private fun buildButtonsState(showHired: Boolean): DashboardButtonsState {
        return DashboardButtonsState(
            locateOnHired = if (showHired) {
                DashboardButtonState.VisibleEnabled(
                    textRes = R.string.btn_soon_to_clear,
                    iconRes = R.drawable.ubactivar,
                    background = DashboardButtonBackground.BLUE,
                    textColor = android.R.color.white
                )
            } else {
                DashboardButtonState.Hidden
            },
            showAllTrips = DashboardButtonState.Hidden,
            showRecentTrips = DashboardButtonState.Hidden,
            cancel = DashboardButtonState.VisibleEnabled(
                textRes = R.string.btn_cancel,
                background = DashboardButtonBackground.DISABLE,
                textColor = android.R.color.white
            ),
            close = DashboardButtonState.VisibleEnabled(
                textRes = R.string.btn_close,
                background = DashboardButtonBackground.DISABLE,
                textColor = android.R.color.white
            )
        )
    }

    private fun getDistanceFromLatLonInMetersLocale(
        lastLatLong: LatLong?,
        centralCoord: LatLong?
    ): Int {
        if (lastLatLong == null || lastLatLong.Lat.isNaN() || centralCoord == null || centralCoord.Lat.isNaN()) {
            return Int.MAX_VALUE
        }
        return W2CLocation.getDistanceFromLatLonInMeters(lastLatLong, centralCoord)
    }

    fun removeHandlerCallbacks() {
        pendingTripsRunnable?.let { pendingTripsHandler.removeCallbacks(it) }
        zonesRunnable?.let { handler.removeCallbacks(it) }
    }

    override fun onCleared() {
        removeHandlerCallbacks()
        super.onCleared()
    }
}
```

```kotlin
package ifac.td.taxi.ui.screen.dashboard

import androidx.compose.foundation.layout.*
import androidx.compose.foundation.lazy.LazyColumn
import androidx.compose.foundation.lazy.items
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.platform.LocalLifecycleOwner
import androidx.compose.ui.res.stringResource
import androidx.compose.ui.unit.dp
import androidx.lifecycle.Lifecycle
import androidx.lifecycle.repeatOnLifecycle
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.navigation.NavController
import ifac.td.taxi.R
import ifac.td.taxi.ui.model.ZoneModel
import kotlinx.coroutines.launch

@Composable
fun DashboardRoute(
    viewModel: DashboardComposeViewModel,
    navController: NavController,
    onShowHeader: (Boolean) -> Unit = {},
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val lifecycleOwner = LocalLifecycleOwner.current
    val scope = rememberCoroutineScope()

    LaunchedEffect(Unit) {
        onShowHeader(true)
    }

    LaunchedEffect(viewModel, lifecycleOwner) {
        lifecycleOwner.repeatOnLifecycle(Lifecycle.State.STARTED) {
            launch {
                viewModel.uiEffect.collect { effect ->
                    when (effect) {
                        DashboardUiEffect.NavigateBack -> navController.popBackStack()
                        DashboardUiEffect.NavigateToHome -> {
                            // preserve navigation behavior here
                            navController.popBackStack()
                        }
                        DashboardUiEffect.NavigateToOnTrip -> {
                            // navigate to on-trip destination if needed
                        }
                        is DashboardUiEffect.ShowDialog -> {
                            // handled in UI state below
                        }
                        is DashboardUiEffect.ShowMessage -> {
                            // show snackbar/toast if desired
                        }
                    }
                }
            }
        }
    }

    LaunchedEffect(Unit) {
        viewModel.onEvent(DashboardUiEvent.OnResume)
    }

    DisposableEffect(Unit) {
        onDispose {
            viewModel.onEvent(DashboardUiEvent.OnPause)
        }
    }

    DashboardScreen(
        state = uiState,
        onEvent = viewModel::onEvent
    )
}
```

```kotlin
@Composable
fun DashboardScreen(

        ZoneList(state.actualZones, onItemClick = { onEvent(DashboardUiEvent.OnActualZoneClick(it)) })

        AlertDialog(
            onDismissRequest = { onEvent(DashboardUiEvent.OnDialogDismiss) },

            },
            dismissButton = {

                    Text(dialog.dismissText)
```

```kotlin
@Composable
fun DashboardHeader(
    headerState: DashboardHeaderState,
    onEvent: (DashboardUiEvent) -> Unit
) {
    Column(verticalArrangement = Arrangement.spacedBy(8.dp)) {

        if (headerState.showStand) {
            TextButton(onClick = { onEvent(DashboardUiEvent.OnStandHeaderClick) }) {
                Text(text = stringResource(R.string.tvOnStop))
            }
        }

        if (headerState.showZone) {
            TextButton(onClick = { onEvent(DashboardUiEvent.OnZoneHeaderClick) }) {
                Text(text = stringResource(R.string.tvOnZone))
            }
        }

        if (headerState.showHired) {
            TextButton(onClick = { onEvent(DashboardUiEvent.OnHiredHeaderClick) }) {
                Text(
                    text = headerState.hiredTextRes?.let { stringResource(it) }
                        ?: stringResource(R.string.tvHired)
                )
            }
        }

        if (headerState.showTrips) {
            TextButton(onClick = { onEvent(DashboardUiEvent.OnTripsHeaderClick) }) {
                Text(
                    text = headerState.tripsTextRes?.let { stringResource(it) }
                        ?: stringResource(R.string.tvTrips)
                )
            }
        }
    }
}
```

# 7) Custom button component with exact state-driven styling

This is the Compose replacement for your custom button logic.

```kotlin
@Composable
fun DashboardButton(
    state: DashboardButtonState,
    onClick: () -> Unit
) {
    when (state) {
        DashboardButtonState.Hidden -> Unit

        is DashboardButtonState.VisibleEnabled -> {
            val containerColor = when (state.background) {
                DashboardButtonBackground.BLUE -> Color(0xFF1976D2)
                DashboardButtonBackground.RED -> Color(0xFFD32F2F)
                DashboardButtonBackground.DISABLE -> Color(0xFF9E9E9E)
            }

            Button(
                onClick = onClick,
                enabled = state.enabled,
                colors = ButtonDefaults.buttonColors(
                    containerColor = containerColor,
                    contentColor = Color(state.textColor)
                )
            ) {
                Text(text = stringResource(state.textRes))
            }
        }

        is DashboardButtonState.VisibleDisabled -> {
            val containerColor = when (state.background) {
                DashboardButtonBackground.BLUE -> Color(0xFF1976D2)
                DashboardButtonBackground.RED -> Color(0xFFD32F2F)
                DashboardButtonBackground.DISABLE -> Color(0xFF9E9E9E)
            }

            Button(
                onClick = onClick,
                enabled = false,
                colors = ButtonDefaults.buttonColors(
                    containerColor = containerColor,
                    contentColor = Color(state.textColor)
                )
            ) {
                Text(text = stringResource(state.textRes))
            }
        }
    }
}
```

## Reducer that mirrors the fragment branches one-by-one

This reducer is intentionally split into small functions so each collector can call the exact branch it corresponds to.

```kotlin
package ifac.td.taxi.ui.screen.dashboard

import com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum
import ifac.td.taxi.R

class DashboardButtonsReducer {

    fun reduceDataTypes(
        current: DashboardButtonsState,
        dataTypes: String
    ): DashboardButtonsState {
        val columns = ZoneDataTypesParser.parse(dataTypes)

        val standVisible = columns.contains(AvailableColumnsEnum.STAND_VEHICLES)
        val zoneVisible = columns.contains(AvailableColumnsEnum.ZONE_VEHICLES)
        val hiredVisible = columns.contains(AvailableColumnsEnum.HIRED_VEHICLES) ||
                columns.contains(AvailableColumnsEnum.BOOKED_TRIPS)
        val tripsVisible = columns.contains(AvailableColumnsEnum.PREBOOKED_TRIPS) ||
                columns.contains(AvailableColumnsEnum.TOTAL_TRIPS) ||
                (columns.contains(AvailableColumnsEnum.HIRED_VEHICLES) &&
                        columns.contains(AvailableColumnsEnum.BOOKED_TRIPS))

        val hiredTextRes = if (!columns.contains(AvailableColumnsEnum.HIRED_VEHICLES) &&
            columns.contains(AvailableColumnsEnum.BOOKED_TRIPS)
        ) R.string.abrevUbServiciosWithoutDots else null

        val tripsTextRes = when {
            columns.contains(AvailableColumnsEnum.PREBOOKED_TRIPS) -> R.string.abbrevPreBookedTrips
            columns.contains(AvailableColumnsEnum.TOTAL_TRIPS) -> R.string.abbrevTotalTrips
            else -> null
        }

        return current.copy(
            header = current.header.copy(
                stand = DashboardHeaderButtonUiState(
                    visible = standVisible,
                    textRes = R.string.tvOnStop,
                    sortIcon = if (standVisible) DashboardSortIconState.Hidden else DashboardSortIconState.Hidden
                ),
                zone = DashboardHeaderButtonUiState(
                    visible = zoneVisible,
                    textRes = R.string.tvOnZone,
                    sortIcon = if (zoneVisible) DashboardSortIconState.Hidden else DashboardSortIconState.Hidden
                ),
                hired = DashboardHeaderButtonUiState(
                    visible = hiredVisible,
                    textRes = hiredTextRes ?: R.string.tvHired,
                    sortIcon = if (hiredVisible) DashboardSortIconState.Hidden else DashboardSortIconState.Hidden
                ),
                trips = DashboardHeaderButtonUiState(
                    visible = tripsVisible,
                    textRes = tripsTextRes ?: R.string.tvTrips,
                    sortIcon = if (tripsVisible) DashboardSortIconState.Hidden else DashboardSortIconState.Hidden
                )
            )
        )
    }

    fun reduceHiredState(
        current: DashboardButtonsState,
        isInSoonInZone: Boolean
    ): DashboardButtonsState {
        return if (isInSoonInZone) {
            current.copy(
                locateOnHired = DashboardButtonUiState.Visible(
                    textRes = R.string.btn_delocate,
                    iconRes = R.drawable.ubdesact,
                    background = DashboardButtonBackground.RED,
                    enabled = true
                )
            )
        } else {
            current.copy(
                locateOnHired = DashboardButtonUiState.Visible(
                    textRes = R.string.btn_soon_to_clear,
                    iconRes = R.drawable.ubactivar,
                    background = DashboardButtonBackground.BLUE,
                    enabled = true
                )
            )
        }
    }

    fun reduceSoonInZonePresence(
        current: DashboardButtonsState,
        hasSoonInZone: Boolean
    ): DashboardButtonsState {
        return if (hasSoonInZone) {
            current.copy(
                locateOnHired = DashboardButtonUiState.Visible(
                    textRes = R.string.btn_delocate,
                    iconRes = R.drawable.ubdesact,
                    background = DashboardButtonBackground.RED,
                    enabled = true
                )
            )
        } else {
            current.copy(
                locateOnHired = DashboardButtonUiState.Visible(
                    textRes = R.string.btn_soon_to_clear,
                    iconRes = R.drawable.ubactivar,
                    background = DashboardButtonBackground.BLUE,
                    enabled = true
                )
            )
        }
    }

    fun reduceHeaderSort(
        current: DashboardButtonsState,
        clicked: HeaderSortTarget,
        reverse: Boolean?
    ): DashboardButtonsState {
        val cleared = current.copy(
            header = current.header.copy(
                stand = current.header.stand.copy(sortIcon = DashboardSortIconState.Hidden),
                zone = current.header.zone.copy(sortIcon = DashboardSortIconState.Hidden),
                hired = current.header.hired.copy(sortIcon = DashboardSortIconState.Hidden),
                trips = current.header.trips.copy(sortIcon = DashboardSortIconState.Hidden)
            )
        )

        if (reverse == null) return cleared

        return when (clicked) {
            HeaderSortTarget.STAND -> cleared.copy(
                header = cleared.header.copy(
                    stand = cleared.header.stand.copy(
                        sortIcon = if (reverse) DashboardSortIconState.Hidden else DashboardSortIconState.Ascending
                    )
                )
            )

            HeaderSortTarget.ZONE -> cleared.copy(
                header = cleared.header.copy(
                    zone = cleared.header.zone.copy(
                        sortIcon = if (reverse) DashboardSortIconState.Hidden else DashboardSortIconState.Ascending
                    )
                )
            )

            HeaderSortTarget.HIRED -> cleared.copy(
                header = cleared.header.copy(
                    hired = cleared.header.hired.copy(
                        sortIcon = if (reverse) DashboardSortIconState.Hidden else DashboardSortIconState.Ascending
                    )
                )
            )

            HeaderSortTarget.TRIPS -> cleared.copy(
                header = cleared.header.copy(
                    trips = cleared.header.trips.copy(
                        sortIcon = if (reverse) DashboardSortIconState.Hidden else DashboardSortIconState.Ascending
                    )
                )
            )
        }
    }
}

enum class HeaderSortTarget {
    STAND, ZONE, HIRED, TRIPS
}
```

```kotlin
package ifac.td.taxi.ui.screen.dashboard

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.interfacom.sdk.taximeter.bravocomm.rest.pending_trips.response.PendingTrip
import ifac.td.taxi.viewmodel.MainActivityViewModel
import kotlinx.coroutines.channels.BufferOverflow
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.collectLatest
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

class DashboardMainActivityViewModel(
    private val sharedViewModel: MainActivityViewModel
) : ViewModel() {

    private val _sharedState = MutableStateFlow(SharedDashboardState())
    val sharedState: StateFlow<SharedDashboardState> = _sharedState.asStateFlow()

    private val _effects = MutableSharedFlow<DashboardSharedEffect>(
        replay = 0,
        extraBufferCapacity = 1,
        onBufferOverflow = BufferOverflow.DROP_OLDEST
    )
    val effects: SharedFlow<DashboardSharedEffect> = _effects

    fun start() {
        viewModelScope.launch {
            sharedViewModel.pendingTripsListFlow.collectLatest { trips ->
                val list = trips ?: arrayListOf()
                _sharedState.update {
                    it.copy(
                        pendingTrips = list,
                        pendingTripsCount = list.size,
                        hasPendingTrips = list.isNotEmpty()
                    )
                }
                _effects.tryEmit(DashboardSharedEffect.PendingTripsUpdated)
            }
        }

        viewModelScope.launch {
            sharedViewModel.zoningChangedFlow.collectLatest {
                _effects.tryEmit(DashboardSharedEffect.RefreshActualZoneData)
            }
        }

        viewModelScope.launch {
            sharedViewModel.bravoStateFlow.collectLatest { bravoState ->
                _sharedState.update {
                    it.copy(isInSoonInZone = bravoState.isInSoonInZone)
                }
                _effects.tryEmit(DashboardSharedEffect.SoonInZoneChanged)
            }
        }
    }

    fun emitManualZoningNavigation(enabled: Boolean) {
        sharedViewModel.emitManualZoningNavigation(enabled)
    }

    fun navigateBack() {
        sharedViewModel.navigateBack()
    }
}
```

```kotlin
class DashboardMainActivityViewModel(
    private val sharedViewModel: MainActivityViewModel
) : ViewModel() {

    private val _sharedState = MutableStateFlow(SharedDashboardState())
    val sharedState: StateFlow<SharedDashboardState> = _sharedState.asStateFlow()

    private val _effects = MutableSharedFlow<DashboardSharedEffect>(
        replay = 0,
        extraBufferCapacity = 1,
        onBufferOverflow = BufferOverflow.DROP_OLDEST
    )
    val effects: SharedFlow<DashboardSharedEffect> = _effects

    fun start() {
        viewModelScope.launch {
            sharedViewModel.pendingTripsListFlow.collectLatest { trips ->
                val list = trips ?: arrayListOf()
                _sharedState.update {
                    it.copy(
                        pendingTrips = list,
                        pendingTripsCount = list.size
                    )
                }
                _effects.tryEmit(DashboardSharedEffect.PendingTripsUpdated)
            }
        }

        viewModelScope.launch {
            sharedViewModel.zoningChangedFlow.collectLatest {
                _effects.tryEmit(DashboardSharedEffect.RefreshActualZoneData)
            }
        }

        viewModelScope.launch {
            sharedViewModel.bravoStateFlow.collectLatest { bravoState ->
                _sharedState.update {
                    it.copy(isInSoonInZone = bravoState.isInSoonInZone)
                }
                _effects.tryEmit(DashboardSharedEffect.SoonInZoneChanged)
            }
        }
    }

    fun emitManualZoningNavigation(enabled: Boolean) {
        sharedViewModel.emitManualZoningNavigation(enabled)
    }

    fun navigateBack() {
        sharedViewModel.navigateBack()
    }
}

class DashboardButtonsReducer {

    fun reduceDataTypes(
        current: DashboardButtonsState,
        dataTypes: String
    ): DashboardButtonsState {
        val columns = ZoneUtils.parseStringDataTypes(dataTypes)

        val showStand = columns.contains(AvailableColumnsEnum.STAND_VEHICLES)
        val showZone = columns.contains(AvailableColumnsEnum.ZONE_VEHICLES)
        val showHired = columns.contains(AvailableColumnsEnum.HIRED_VEHICLES) ||
                columns.contains(AvailableColumnsEnum.BOOKED_TRIPS)
        val showTrips = columns.contains(AvailableColumnsEnum.PREBOOKED_TRIPS) ||
                columns.contains(AvailableColumnsEnum.TOTAL_TRIPS) ||
                (columns.contains(AvailableColumnsEnum.HIRED_VEHICLES) &&
                        columns.contains(AvailableColumnsEnum.BOOKED_TRIPS))

        val hiredTextRes = if (!columns.contains(AvailableColumnsEnum.HIRED_VEHICLES) &&
            columns.contains(AvailableColumnsEnum.BOOKED_TRIPS)
        ) R.string.abrevUbServiciosWithoutDots else null

        val tripsTextRes = when {
            columns.contains(AvailableColumnsEnum.PREBOOKED_TRIPS) -> R.string.abbrevPreBookedTrips
            columns.contains(AvailableColumnsEnum.TOTAL_TRIPS) -> R.string.abbrevTotalTrips
            else -> null
        }

        return current.copy(
            // buttons are not changed here except header visibility/text
        ).let {
            // header state is not inside buttonsState in this file, so caller updates header separately
            it
        }
    }

    fun reduceSoonInZonePresence(
        current: DashboardButtonsState,
        hasSoonInZone: Boolean
    ): DashboardButtonsState {
        return if (hasSoonInZone) {
            current.copy(
                locateOnHired = DashboardButtonUiState.Visible(
                    textRes = R.string.btn_delocate,
                    iconRes = R.drawable.ubdesact,
                    background = DashboardButtonBackground.RED,
                    enabled = true
                )
            )
        } else {
            current.copy(
                locateOnHired = DashboardButtonUiState.Visible(
                    textRes = R.string.btn_soon_to_clear,
                    iconRes = R.drawable.ubactivar,
                    background = DashboardButtonBackground.BLUE,
                    enabled = true
                )
            )
        }
    }
}

class DashboardComposeViewModel(
    private val pendingTripsUseCase: PendingTripsUseCase,
    private val zoningUseCase: ZoningUseCase,
    private val licensingUseCase: LicensingUseCase,
    private val sharedBridge: DashboardMainActivityViewModel,
    application: Application
) : ViewModel() {

    private val handler = Handler(Looper.getMainLooper())
    private var zonesRunnable: Runnable? = null

    private val pendingTripsHandler = Handler(Looper.getMainLooper())
    private var pendingTripsRunnable: Runnable? = null

    private val reducer = DashboardButtonsReducer()

    private val _uiState = MutableStateFlow(DashboardUiState())
    val uiState: StateFlow<DashboardUiState> = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<DashboardUiEffect>(
        replay = 0,
        extraBufferCapacity = 1,
        onBufferOverflow = BufferOverflow.DROP_OLDEST
    )
    val uiEffect: SharedFlow<DashboardUiEffect> = _uiEffect

    var bravoConfiguration: BravoConfigurationVariableEntity? = null
    var isCustomersAtStand: Boolean = false

    init {
        viewModelScope.launch {
            sharedBridge.sharedState.collectLatest { shared ->
                _uiState.update { current ->
                    current.copy(
                        pendingTripsCount = shared.pendingTripsCount,
                        buttonsState = reducer.reduceSoonInZonePresence(
                            current = current.buttonsState,
                            hasSoonInZone = shared.isInSoonInZone
                        )
                    )
                }
            }
        }
    }

    fun onEvent(event: DashboardUiEvent) {
        when (event) {
            DashboardUiEvent.OnResume -> {
                initViewModelData()
                loadTrips()
            }

            DashboardUiEvent.OnPause -> removeHandlerCallbacks()

            DashboardUiEvent.OnBackPressed -> {
                viewModelScope.launch { _uiEffect.emit(DashboardUiEffect.NavigateBack) }
            }

            DashboardUiEvent.OnLocateOnHiredClick -> {
                // hook for future dialog/action
            }

            DashboardUiEvent.OnShowAllTripsClick -> {
                // hook for future action
            }

            DashboardUiEvent.OnShowRecentTripsClick -> {
                // hook for future action
            }

            DashboardUiEvent.OnCancelClick -> {
                viewModelScope.launch { _uiEffect.emit(DashboardUiEffect.NavigateBack) }
            }

            DashboardUiEvent.OnCloseClick -> {
                viewModelScope.launch {
                    _uiEffect.emit(DashboardUiEffect.NavigateToHome)
                }
            }

            is DashboardUiEvent.OnNearbyZoneClick -> Unit
            is DashboardUiEvent.OnFarZoneClick -> Unit
            is DashboardUiEvent.OnActualZoneClick -> Unit

            DashboardUiEvent.OnDialogConfirm -> {
                _uiState.update { it.copy(customDialog = null) }
            }

            DashboardUiEvent.OnDialogDismiss -> {
                _uiState.update { it.copy(customDialog = null) }
            }
        }
    }

    fun loadTrips() {
        if (BravoCentral.isPendingTripsAllowed(false)) {
            refreshPendingTripsManual()
        }
        startChronometer()
    }

    private fun refreshPendingTripsManual() {
        viewModelScope.launch {
            pendingTripsUseCase.getPendingTripsZone(
                flow = MutableStateFlow<ArrayList<PendingTrip>?>(null)
            )
        }
    }

    private fun startChronometer() {
        viewModelScope.launch {
            val refreshRate = 2000L
            val initRefreshRate = 500L

            pendingTripsRunnable = object : Runnable {
                override fun run() {
                    viewModelScope.launch {
                        if (BravoCentral.isPendingTripsAllowed(true)) {
                            pendingTripsUseCase.getPendingTripsZone(
                                flow = MutableStateFlow<ArrayList<PendingTrip>?>(null)
                            )
                        }
                    }
                    pendingTripsHandler.postDelayed(this, refreshRate)
                }
            }
            pendingTripsRunnable?.let {
                pendingTripsHandler.postDelayed(it, initRefreshRate)
            }
        }
    }

    fun initViewModelData() {
        viewModelScope.launch {
            val refreshRate = licensingUseCase.getLicensingParameters()?.segRefrescoConsUbicacion
            refreshRate?.let { seconds ->
                val millis = (seconds * 1000).toLong()
                var lastZoningUpdate = zoningUseCase.getLastZoningUpdate()

                val shouldLoadData = (System.currentTimeMillis() - lastZoningUpdate) < millis
                if (shouldLoadData) {
                    zoneCallback(
                        W2CLocation.getZoning().macrozones[0].zones,
                        W2CLocation.getZoning().availableColumns
                    )
                }

                zonesRunnable = object : Runnable {
                    override fun run() {
                        viewModelScope.launch {
                            val newCall = (System.currentTimeMillis() - lastZoningUpdate) > millis
                            if (newCall) {
                                zoningUseCase.getZonesInformation(
                                    W2CLocation.getZoning().macrozones[0],
                                    zoneCallback
                                )
                                lastZoningUpdate = System.currentTimeMillis()
                                zoningUseCase.setLastZoningUpdate()
                            }
                        }
                        handler.postDelayed(this, 1000)
                    }
                }

                zonesRunnable?.let { handler.post(it) }
            }
        }
    }

    private val zoneCallback: (List<Zone>, String) -> Unit = { _, availableColumns ->
        viewModelScope.launch {
            val zonesList = mutableListOf<ZoneModel>()
            W2CLocation.getZoning().macrozones.forEach { mz ->
                zonesList.addAll(mz.zones.map { zone -> ZoneModel(zone, false) })
            }

            filterData(zonesList)
            updateHeader(availableColumns)
        }
    }

    private fun updateHeader(dataTypes: String) {
        val columns = ZoneUtils.parseStringDataTypes(dataTypes)

        val showStand = columns.contains(AvailableColumnsEnum.STAND_VEHICLES)
        val showZone = columns.contains(AvailableColumnsEnum.ZONE_VEHICLES)
        val showHired = columns.contains(AvailableColumnsEnum.HIRED_VEHICLES) ||
                columns.contains(AvailableColumnsEnum.BOOKED_TRIPS)
        val showTrips = columns.contains(AvailableColumnsEnum.PREBOOKED_TRIPS) ||
                columns.contains(AvailableColumnsEnum.TOTAL_TRIPS) ||
                (columns.contains(AvailableColumnsEnum.HIRED_VEHICLES) &&
                        columns.contains(AvailableColumnsEnum.BOOKED_TRIPS))

        val hiredTextRes = if (!columns.contains(AvailableColumnsEnum.HIRED_VEHICLES) &&
            columns.contains(AvailableColumnsEnum.BOOKED_TRIPS)
        ) R.string.abrevUbServiciosWithoutDots else null

        val tripsTextRes = when {
            columns.contains(AvailableColumnsEnum.PREBOOKED_TRIPS) -> R.string.abbrevPreBookedTrips
            columns.contains(AvailableColumnsEnum.TOTAL_TRIPS) -> R.string.abbrevTotalTrips
            else -> null
        }

        _uiState.update { current ->
            current.copy(
                headerState = DashboardHeaderState(
                    showStand = showStand,
                    showZone = showZone,
                    showHired = showHired,
                    showTrips = showTrips,
                    hiredTextRes = hiredTextRes,
                    tripsTextRes = tripsTextRes
                )
            )
        }
    }

    private suspend fun filterData(zonesList: List<ZoneModel>) {
        val zonesWithMoreTrips = zonesList.filter {
            ZoneUtils.checkDiff(it.zone, "") > 0
        }

        val (farZones, nearbyZones) = zonesWithMoreTrips.partition {
            getDistanceFromLatLonInMetersLocale(
                W2CLocation.getLastLatLong(),
                it.zone.centralCoord
            ) > 5000
        }

        val nearbyZonesSorted = nearbyZones.sortedWith(
            compareByDescending<ZoneModel> { ZoneUtils.checkDiff(it.zone, "") }
                .thenBy { getDistanceFromLatLonInMetersLocale(W2CLocation.getLastLatLong(), it.zone.centralCoord) }
        )

        val farZonesSorted = farZones.sortedWith(
            compareByDescending<ZoneModel> { ZoneUtils.checkDiff(it.zone, "") }
                .thenBy { getDistanceFromLatLonInMetersLocale(W2CLocation.getLastLatLong(), it.zone.centralCoord) }
        )

        _uiState.update {
            it.copy(
                nearbyZones = nearbyZonesSorted,
                farZones = farZonesSorted
            )
        }

        refreshActualZoneData()
    }

    fun refreshActualZoneData() {
        viewModelScope.launch {
            val lastZoneSent = W2CLocation.getZoning().getZoneById(
                W2CLocation.getLastIdMacrozoneSent(),
                W2CLocation.getLastIdZoneSent()
            )
            val lastZoneLocated = W2CLocation.getZoneFromLatLong(W2CLocation.getLastLatLong())

            val actualZoneList = mutableListOf<ZoneModel>()
            if (lastZoneSent != null) actualZoneList.add(ZoneModel(lastZoneSent, false))
            if (lastZoneLocated != null) actualZoneList.add(ZoneModel(lastZoneLocated, false))

            _uiState.update { it.copy(actualZones = actualZoneList) }
        }
    }

    private fun getDistanceFromLatLonInMetersLocale(
        lastLatLong: LatLong?,
        centralCoord: LatLong?
    ): Int {
        if (lastLatLong == null || lastLatLong.Lat.isNaN() || centralCoord == null || centralCoord.Lat.isNaN()) {
            return Int.MAX_VALUE
        }
        return W2CLocation.getDistanceFromLatLonInMeters(lastLatLong, centralCoord)
    }

    fun removeHandlerCallbacks() {
        pendingTripsRunnable?.let { pendingTripsHandler.removeCallbacks(it) }
        zonesRunnable?.let { handler.removeCallbacks(it) }
    }

    override fun onCleared() {
        removeHandlerCallbacks()
        super.onCleared()
    }
}

@Composable
fun DashboardRoute(
    navController: NavController,
    viewModel: DashboardComposeViewModel,
    sharedViewModel: DashboardMainActivityViewModel,
    onShowHeader: (Boolean) -> Unit
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val sharedState by sharedViewModel.sharedState.collectAsStateWithLifecycle()
    val lifecycleOwner = LocalLifecycleOwner.current

    LaunchedEffect(Unit) {
        onShowHeader(true)
        sharedViewModel.start()
        viewModel.onEvent(DashboardUiEvent.OnResume)
    }

    LaunchedEffect(viewModel) {
        lifecycleOwner.repeatOnLifecycle(androidx.lifecycle.Lifecycle.State.STARTED) {
            viewModel.uiEffect.collectLatest { effect ->
                when (effect) {
                    DashboardUiEffect.NavigateBack -> navController.popBackStack()
                    DashboardUiEffect.NavigateToHome -> {
                        navController.navigate(HomeDirections.goToOnTripFragment().actionId)
                    }
                    DashboardUiEffect.NavigateToOnTrip -> {
                        // preserve if needed
                    }
                    is DashboardUiEffect.ShowDialog -> {
                        // handled by state if you later use custom dialog
                    }
                    is DashboardUiEffect.ShowMessage -> {
                        // host-side toast/snackbar
                    }
                }
            }
        }
    }

    LaunchedEffect(sharedViewModel) {
        lifecycleOwner.repeatOnLifecycle(androidx.lifecycle.Lifecycle.State.STARTED) {
            sharedViewModel.effects.collectLatest { effect ->
                when (effect) {
                    DashboardSharedEffect.RefreshActualZoneData -> viewModel.refreshActualZoneData()
                    DashboardSharedEffect.PendingTripsUpdated -> Unit
                    DashboardSharedEffect.SoonInZoneChanged -> Unit
                }
            }
        }
    }

    DashboardScreen(
        state = uiState.copy(
            pendingTripsCount = sharedState.pendingTripsCount,
            buttonsState = if (sharedState.isInSoonInZone) {
                uiState.buttonsState.copy(
                    locateOnHired = DashboardButtonUiState.Visible(
                        textRes = R.string.btn_delocate,
                        iconRes = R.drawable.ubdesact,
                        background = DashboardButtonBackground.RED,
                        enabled = true
                    )
                )
            } else {
                uiState.buttonsState.copy(
                    locateOnHired = DashboardButtonUiState.Visible(
                        textRes = R.string.btn_soon_to_clear,
                        iconRes = R.drawable.ubactivar,
                        background = DashboardButtonBackground.BLUE,
                        enabled = true
                    )
                )
            }
        ),
        onEvent = viewModel::onEvent
    )
}

@Composable
fun DashboardScreen(
    state: DashboardUiState,
    onEvent: (DashboardUiEvent) -> Unit
) {
    Column(modifier = Modifier.fillMaxSize().padding(16.dp)) {

        DashboardHeader(state.headerState)

        Spacer(Modifier.height(12.dp))

        Text(
            text = "Pending trips: ${state.pendingTripsCount}",
            style = MaterialTheme.typography.titleMedium
        )

        Spacer(Modifier.height(12.dp))

        DashboardButton(state.buttonsState.locateOnHired, onClick = {
            onEvent(DashboardUiEvent.OnLocateOnHiredClick)
        })

        Spacer(Modifier.height(12.dp))

        DashboardButton(state.buttonsState.cancel, onClick = {
            onEvent(DashboardUiEvent.OnCancelClick)
        })

        Spacer(Modifier.height(12.dp))

        DashboardButton(state.buttonsState.close, onClick = {
            onEvent(DashboardUiEvent.OnCloseClick)
        })

        Spacer(Modifier.height(16.dp))

        Text("Nearby zones", style = MaterialTheme.typography.titleMedium)
        ZoneList(state.nearbyZones)

        Spacer(Modifier.height(16.dp))

        Text("Far zones", style = MaterialTheme.typography.titleMedium)
        ZoneList(state.farZones)

        Spacer(Modifier.height(16.dp))

        Text("Actual zones", style = MaterialTheme.typography.titleMedium)
        ZoneList(state.actualZones)
    }

    state.customDialog?.let { dialog ->
        AlertDialog(
            onDismissRequest = { onEvent(DashboardUiEvent.OnDialogDismiss) },
            title = { Text(dialog.title) },
            text = { Text(dialog.message) },

                TextButton(onClick = { onEvent(DashboardUiEvent.OnDialogConfirm) }) {
                    Text(dialog.confirmText)
                }
            },
            dismissButton = {
                TextButton(onClick = { onEvent(DashboardUiEvent.OnDialogDismiss) }) {
                    Text(dialog.dismissText)
                }
            }
        )
    }
}

@Composable
fun DashboardButton(
    state: DashboardButtonUiState,
    onClick: () -> Unit
) {

        DashboardButtonUiState.Hidden -> Unit

        is DashboardButtonUiState.Visible -> {

                colors = ButtonDefaults.buttonColors(
                    containerColor = containerColor,
                    contentColor = Color.White

                Text(text = stringResource(state.textRes))
            }
        }
    }
```

```kotlin
class DashboardMainActivityViewModel(
    private val sharedViewModel: MainActivityViewModel
) : ViewModel() {

    private val _sharedState = MutableStateFlow(SharedDashboardState())
    val sharedState: StateFlow<SharedDashboardState> = _sharedState.asStateFlow()

    private val _effects = MutableSharedFlow<DashboardSharedEffect>(
        replay = 0,
        extraBufferCapacity = 1,
        onBufferOverflow = BufferOverflow.DROP_OLDEST
    )
    val effects: SharedFlow<DashboardSharedEffect> = _effects

    fun start() {
        viewModelScope.launch {
            sharedViewModel.pendingTripsListFlow.collectLatest { trips ->
                val list = trips ?: arrayListOf()
                _sharedState.update {
                    it.copy(
                        pendingTrips = list,
                        pendingTripsCount = list.size
                    )
                }
                _effects.tryEmit(DashboardSharedEffect.PendingTripsUpdated)
            }
        }

        viewModelScope.launch {
            sharedViewModel.zoningChangedFlow.collectLatest {
                _effects.tryEmit(DashboardSharedEffect.RefreshActualZoneData)
            }
        }

        viewModelScope.launch {
            sharedViewModel.bravoStateFlow.collectLatest { bravoState ->
                _sharedState.update {
                    it.copy(isInSoonInZone = bravoState.isInSoonInZone)
                }
                _effects.tryEmit(DashboardSharedEffect.SoonInZoneChanged)
            }
        }
    }

    fun emitManualZoningNavigation(enabled: Boolean) {
        sharedViewModel.emitManualZoningNavigation(enabled)
    }

    fun navigateBack() {
        sharedViewModel.navigateBack()
    }
}

class DashboardComposeViewModel(
    private val pendingTripsUseCase: PendingTripsUseCase,
    private val zoningUseCase: ZoningUseCase,
    private val licensingUseCase: LicensingUseCase,
    private val sharedBridge: DashboardMainActivityViewModel,
    application: Application
) : ViewModel() {

    private val handler = Handler(Looper.getMainLooper())
    private var zonesRunnable: Runnable? = null

    private val pendingTripsHandler = Handler(Looper.getMainLooper())
    private var pendingTripsRunnable: Runnable? = null

    private val reducer = DashboardButtonsReducer()

    private val _uiState = MutableStateFlow(DashboardUiState())
    val uiState: StateFlow<DashboardUiState> = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<DashboardUiEffect>(
        replay = 0,
        extraBufferCapacity = 1,
        onBufferOverflow = BufferOverflow.DROP_OLDEST
    )
    val uiEffect: SharedFlow<DashboardUiEffect> = _uiEffect

    var bravoConfiguration: BravoConfigurationVariableEntity? = null
    var isCustomersAtStand: Boolean = false

    init {
        viewModelScope.launch {
            sharedBridge.sharedState.collectLatest { shared ->
                _uiState.update { current ->
                    current.copy(
                        pendingTripsCount = shared.pendingTripsCount,
                        buttonsState = reducer.reduceSoonInZonePresence(
                            current = current.buttonsState,
                            hasSoonInZone = shared.isInSoonInZone
                        )
                    )
                }
            }
        }
    }

    fun onEvent(event: DashboardUiEvent) {
        when (event) {
            DashboardUiEvent.OnResume -> {
                initViewModelData()
                loadTrips()
            }

            DashboardUiEvent.OnPause -> removeHandlerCallbacks()

            DashboardUiEvent.OnBackPressed -> {
                viewModelScope.launch { _uiEffect.emit(DashboardUiEffect.NavigateBack) }
            }

            DashboardUiEvent.OnLocateOnHiredClick -> Unit
            DashboardUiEvent.OnShowAllTripsClick -> Unit
            DashboardUiEvent.OnShowRecentTripsClick -> Unit

            DashboardUiEvent.OnCancelClick -> {
                viewModelScope.launch { _uiEffect.emit(DashboardUiEffect.NavigateBack) }
            }

            DashboardUiEvent.OnCloseClick -> {
                viewModelScope.launch { _uiEffect.emit(DashboardUiEffect.NavigateToHome) }
            }

            is DashboardUiEvent.OnNearbyZoneClick -> Unit
            is DashboardUiEvent.OnFarZoneClick -> Unit
            is DashboardUiEvent.OnActualZoneClick -> Unit

            DashboardUiEvent.OnDialogConfirm -> _uiState.update { it.copy(customDialog = null) }
            DashboardUiEvent.OnDialogDismiss -> _uiState.update { it.copy(customDialog = null) }
        }
    }

    fun loadTrips() {
        if (BravoCentral.isPendingTripsAllowed(false)) {
            refreshPendingTripsManual()
        }
        startChronometer()
    }

    private fun refreshPendingTripsManual() {
        viewModelScope.launch {
            pendingTripsUseCase.getPendingTripsZone(
                flow = MutableStateFlow<ArrayList<PendingTrip>?>(null)
            )
        }
    }

    private fun startChronometer() {
        viewModelScope.launch {
            val refreshRate = 2000L
            val initRefreshRate = 500L

            pendingTripsRunnable = object : Runnable {
                override fun run() {
                    viewModelScope.launch {
                        if (BravoCentral.isPendingTripsAllowed(true)) {
                            pendingTripsUseCase.getPendingTripsZone(
                                flow = MutableStateFlow<ArrayList<PendingTrip>?>(null)
                            )
                        }
                    }
                    pendingTripsHandler.postDelayed(this, refreshRate)
                }
            }
            pendingTripsRunnable?.let {
                pendingTripsHandler.postDelayed(it, initRefreshRate)
            }
        }
    }

    fun initViewModelData() {
        viewModelScope.launch {
            val refreshRate = licensingUseCase.getLicensingParameters()?.segRefrescoConsUbicacion
            refreshRate?.let { seconds ->
                val millis = (seconds * 1000).toLong()
                var lastZoningUpdate = zoningUseCase.getLastZoningUpdate()

                val shouldLoadData = (System.currentTimeMillis() - lastZoningUpdate) < millis
                if (shouldLoadData) {
                    zoneCallback(
                        W2CLocation.getZoning().macrozones[0].zones,
                        W2CLocation.getZoning().availableColumns
                    )
                }

                zonesRunnable = object : Runnable {
                    override fun run() {
                        viewModelScope.launch {
                            val newCall = (System.currentTimeMillis() - lastZoningUpdate) > millis
                            if (newCall) {
                                zoningUseCase.getZonesInformation(
                                    W2CLocation.getZoning().macrozones[0],
                                    zoneCallback
                                )
                                lastZoningUpdate = System.currentTimeMillis()
                                zoningUseCase.setLastZoningUpdate()
                            }
                        }
                        handler.postDelayed(this, 1000)
                    }
                }

                zonesRunnable?.let { handler.post(it) }
            }
        }
    }

    private val zoneCallback: (List<Zone>, String) -> Unit = { _, availableColumns ->
        viewModelScope.launch {
            val zonesList = mutableListOf<ZoneModel>()
            W2CLocation.getZoning().macrozones.forEach { mz ->
                zonesList.addAll(mz.zones.map { zone -> ZoneModel(zone, false) })
            }

            filterData(zonesList)
            updateHeader(availableColumns)
        }
    }

    private fun updateHeader(dataTypes: String) {
        val columns = ZoneUtils.parseStringDataTypes(dataTypes)

        val showStand = columns.contains(AvailableColumnsEnum.STAND_VEHICLES)
        val showZone = columns.contains(AvailableColumnsEnum.ZONE_VEHICLES)
        val showHired = columns.contains(AvailableColumnsEnum.HIRED_VEHICLES) ||
                columns.contains(AvailableColumnsEnum.BOOKED_TRIPS)
        val showTrips = columns.contains(AvailableColumnsEnum.PREBOOKED_TRIPS) ||
                columns.contains(AvailableColumnsEnum.TOTAL_TRIPS) ||
                (columns.contains(AvailableColumnsEnum.HIRED_VEHICLES) &&
                        columns.contains(AvailableColumnsEnum.BOOKED_TRIPS))

        val hiredTextRes = if (!columns.contains(AvailableColumnsEnum.HIRED_VEHICLES) &&
            columns.contains(AvailableColumnsEnum.BOOKED_TRIPS)
        ) R.string.abrevUbServiciosWithoutDots else null

        val tripsTextRes = when {
            columns.contains(AvailableColumnsEnum.PREBOOKED_TRIPS) -> R.string.abbrevPreBookedTrips
            columns.contains(AvailableColumnsEnum.TOTAL_TRIPS) -> R.string.abbrevTotalTrips
            else -> null
        }

        _uiState.update {
            it.copy(
                headerState = DashboardHeaderState(
                    showStand = showStand,
                    showZone = showZone,
                    showHired = showHired,
                    showTrips = showTrips,
                    hiredTextRes = hiredTextRes,
                    tripsTextRes = tripsTextRes
                )
            )
        }
    }

    private suspend fun filterData(zonesList: List<ZoneModel>) {
        val zonesWithMoreTrips = zonesList.filter {
            ZoneUtils.checkDiff(it.zone, "") > 0
        }

        val (farZones, nearbyZones) = zonesWithMoreTrips.partition {
            getDistanceFromLatLonInMetersLocale(
                W2CLocation.getLastLatLong(),
                it.zone.centralCoord
            ) > 5000
        }

        val nearbyZonesSorted = nearbyZones.sortedWith(
            compareByDescending<ZoneModel> { ZoneUtils.checkDiff(it.zone, "") }
                .thenBy { getDistanceFromLatLonInMetersLocale(W2CLocation.getLastLatLong(), it.zone.centralCoord) }
        )

        val farZonesSorted = farZones.sortedWith(
            compareByDescending<ZoneModel> { ZoneUtils.checkDiff(it.zone, "") }
                .thenBy { getDistanceFromLatLonInMetersLocale(W2CLocation.getLastLatLong(), it.zone.centralCoord) }
        )

        _uiState.update {
            it.copy(
                nearbyZones = nearbyZonesSorted,
                farZones = farZonesSorted
            )
        }

        refreshActualZoneData()
    }

    fun refreshActualZoneData() {
        viewModelScope.launch {
            val lastZoneSent = W2CLocation.getZoning().getZoneById(
                W2CLocation.getLastIdMacrozoneSent(),
                W2CLocation.getLastIdZoneSent()
            )
            val lastZoneLocated = W2CLocation.getZoneFromLatLong(W2CLocation.getLastLatLong())

            val actualZoneList = mutableListOf<ZoneModel>()
            if (lastZoneSent != null) actualZoneList.add(ZoneModel(lastZoneSent, false))
            if (lastZoneLocated != null) actualZoneList.add(ZoneModel(lastZoneLocated, false))

            _uiState.update { it.copy(actualZones = actualZoneList) }
        }
    }

    private fun getDistanceFromLatLonInMetersLocale(
        lastLatLong: LatLong?,
        centralCoord: LatLong?
    ): Int {
        if (lastLatLong == null || lastLatLong.Lat.isNaN() || centralCoord == null || centralCoord.Lat.isNaN()) {
            return Int.MAX_VALUE
        }
        return W2CLocation.getDistanceFromLatLonInMeters(lastLatLong, centralCoord)
    }

    fun removeHandlerCallbacks() {
        pendingTripsRunnable?.let { pendingTripsHandler.removeCallbacks(it) }
        zonesRunnable?.let { handler.removeCallbacks(it) }
    }

    override fun onCleared() {
        removeHandlerCallbacks()
        super.onCleared()
    }
}

@Composable
fun DashboardRoute(
    navController: NavController,
    viewModel: DashboardComposeViewModel,
    sharedViewModel: DashboardMainActivityViewModel,
    onShowHeader: (Boolean) -> Unit
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val sharedState by sharedViewModel.sharedState.collectAsStateWithLifecycle()
    val lifecycleOwner = LocalLifecycleOwner.current

    LaunchedEffect(Unit) {
        onShowHeader(true)
        sharedViewModel.start()
        viewModel.onEvent(DashboardUiEvent.OnResume)
    }

    LaunchedEffect(viewModel) {
        lifecycleOwner.repeatOnLifecycle(androidx.lifecycle.Lifecycle.State.STARTED) {
            viewModel.uiEffect.collectLatest { effect ->
                when (effect) {
                    DashboardUiEffect.NavigateBack -> navController.popBackStack()
                    DashboardUiEffect.NavigateToHome -> {
                        navController.navigate(HomeDirections.goToOnTripFragment())
                    }
                    DashboardUiEffect.NavigateToOnTrip -> Unit
                    is DashboardUiEffect.ShowDialog -> Unit
                    is DashboardUiEffect.ShowMessage -> Unit
                }
            }
        }
    }

    LaunchedEffect(sharedViewModel) {
        lifecycleOwner.repeatOnLifecycle(androidx.lifecycle.Lifecycle.State.STARTED) {
            sharedViewModel.effects.collectLatest { effect ->
                when (effect) {
                    DashboardSharedEffect.RefreshActualZoneData -> viewModel.refreshActualZoneData()
                    DashboardSharedEffect.PendingTripsUpdated -> Unit
                    DashboardSharedEffect.SoonInZoneChanged -> Unit
                }
            }
        }
    }

    DashboardScreen(
        state = uiState.copy(
            pendingTripsCount = sharedState.pendingTripsCount,
            buttonsState = if (sharedState.isInSoonInZone) {
                uiState.buttonsState.copy(
                    locateOnHired = DashboardButtonUiState.Visible(
                        textRes = R.string.btn_delocate,
                        iconRes = R.drawable.ubdesact,
                        background = DashboardButtonBackground.RED,
                        enabled = true
                    )
                )
            } else {
                uiState.buttonsState.copy(
                    locateOnHired = DashboardButtonUiState.Visible(
                        textRes = R.string.btn_soon_to_clear,
                        iconRes = R.drawable.ubactivar,
                        background = DashboardButtonBackground.BLUE,
                        enabled = true
                    )
                )
            }
        ),
        onEvent = viewModel::onEvent
    )
}

) {

@Composable
fun DashboardButton(
    state: DashboardButtonUiState,
    onClick: () -> Unit
) {
    when (state) {
        DashboardButtonUiState.Hidden -> Unit

        is DashboardButtonUiState.Visible -> {
            val containerColor = when (state.background) {
                DashboardButtonBackground.BLUE -> Color(0xFF1976D2)
                DashboardButtonBackground.RED -> Color(0xFFD32F2F)

                onClick = onClick,
                enabled = state.enabled,
                colors = ButtonDefaults.buttonColors(
                    containerColor = containerColor,
                    contentColor = Color.White
```

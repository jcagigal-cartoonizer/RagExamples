Below is a Compose-first replacement that preserves the fragment behavior, keeps navigation compatible with Jetpack Views, and uses a `UiState + UiEvent + UiEffect` pattern with a reducer for the button state.

I’m splitting it into 4 parts:

1. **Compose UI**
2. **Compose ViewModel**
3. **Buttons reducer/state holders**
4. **MainActivity bridge ViewModel**

I’ll also keep the naming prefix requirement:
- `RequirementsUiState`
- `RequirementsDialogState`
- `RequirementsButtonVisualState`
- `RequirementsButtonsState`

## 1) Compose screen

This composable mirrors the fragment:
- shows bottom bar/header via the host activity
- triggers `getRequirements()` on first composition
- collects state with lifecycle awareness
- shows loading / empty / list states
- preserves navigation by exposing callbacks instead of hard-wiring navigation logic
- supports dialog handling only if needed

```kotlin

import androidx.compose.foundation.layout.*
import androidx.compose.foundation.lazy.LazyColumn

import androidx.lifecycle.repeatOnLifecycle
import androidx.lifecycle.Lifecycle

import androidx.compose.ui.Alignment
import ifac.td.taxi.repository.connections.rest.bravoRest.RequirementsModel
import ifac.td.taxi.ui.custom.button.CustomButton
import kotlinx.coroutines.flow.collectLatest

@Composable
fun RequirementsScreen(
    viewModel: RequirementsComposeViewModel,
    mainActivityViewModel: RequirementsMainActivityViewModel,
    onNavigateBack: () -> Unit = {},
    onRequirementClick: (String) -> Unit = {}
) {
    val lifecycleOwner = LocalLifecycleOwner.current
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val buttonsState by viewModel.buttonsState.collectAsStateWithLifecycle()
    val dialogState by viewModel.dialogState.collectAsStateWithLifecycle()

    LaunchedEffect(Unit) {
        mainActivityViewModel.showBottomBar(true)
        mainActivityViewModel.showHeader(true)
        viewModel.onEvent(RequirementsUiEvent.LoadRequirements)
    }

    LaunchedEffect(viewModel.effectFlow, lifecycleOwner) {
        lifecycleOwner.repeatOnLifecycle(Lifecycle.State.STARTED) {
            viewModel.effectFlow.collectLatest { effect ->
                when (effect) {
                    is RequirementsUiEffect.NavigateBack -> onNavigateBack()
                    is RequirementsUiEffect.ShowMessage -> {
                        // optional snackbar/toast handling from host
                    }
                    is RequirementsUiEffect.OpenRequirement -> onRequirementClick(effect.value)
                }
            }
        }
    }

    if (dialogState is RequirementsDialogState.Visible) {
        val dialog = dialogState as RequirementsDialogState.Visible
        AlertDialog(
            onDismissRequest = { viewModel.onEvent(RequirementsUiEvent.DismissDialog) },
            title = { Text(dialog.title) },
            text = { Text(dialog.message) },
            confirmButton = {
                TextButton(onClick = { viewModel.onEvent(RequirementsUiEvent.ConfirmDialog) }) {
                    Text(dialog.confirmText)
                }
            },
            dismissButton = {
                TextButton(onClick = { viewModel.onEvent(RequirementsUiEvent.DismissDialog) }) {
                    Text(dialog.dismissText)
                }
            }
        )
    }

    RequirementsContent(
        uiState = uiState,
        buttonsState = buttonsState,
        onRetry = { viewModel.onEvent(RequirementsUiEvent.LoadRequirements) },
        onBack = { viewModel.onEvent(RequirementsUiEvent.BackPressed) },
        onRequirementClick = onRequirementClick
    )
}

@Composable
private fun RequirementsContent(
    uiState: RequirementsUiState,
    buttonsState: RequirementsButtonsState,
    onRetry: () -> Unit,
    onBack: () -> Unit,
    onRequirementClick: (String) -> Unit
) {
    Column(modifier = Modifier.fillMaxSize()) {

        // Optional top action row using the custom button API
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .padding(8.dp),
            horizontalArrangement = Arrangement.SpaceBetween
        ) {
            RequirementsCustomButton(
                state = buttonsState.backButton,
                onClick = onBack
            )
            RequirementsCustomButton(
                state = buttonsState.retryButton,
                onClick = onRetry
            )
        }

        when {
            uiState.isLoading -> {
                Box(
                    modifier = Modifier.fillMaxSize(),
                    contentAlignment = Alignment.Center
                ) {
                    CircularProgressIndicator()
                }
            }

            uiState.driverRequirements.isEmpty() && uiState.vehicleRequirements.isEmpty() -> {
                Box(
                    modifier = Modifier.fillMaxSize(),
                    contentAlignment = Alignment.Center
                ) {
                    Text(text = "No requirements available")
                }
            }

            else -> {
                LazyColumn(
                    modifier = Modifier
                        .fillMaxSize()
                        .padding(8.dp),
                    verticalArrangement = Arrangement.spacedBy(12.dp)
                ) {
                    item {
                        Text(text = "Driver requirements", style = MaterialTheme.typography.titleMedium)
                    }
                    items(uiState.driverRequirements) { item ->
                        Text(
                            text = item.toString(),
                            modifier = Modifier
                                .fillMaxWidth()
                                .padding(8.dp)
                        )
                    }

                    item {
                        Spacer(modifier = Modifier.height(8.dp))
                        Text(text = "Vehicle requirements", style = MaterialTheme.typography.titleMedium)
                    }
                    items(uiState.vehicleRequirements) { item ->
                        Text(
                            text = item.toString(),
                            modifier = Modifier
                                .fillMaxWidth()
                                .padding(8.dp)
                        )
                    }
                }
            }
        }
    }
}

    onClick: () -> Unit
```

## 2) Compose ViewModel with `UiState + UiEvent + UiEffect`

This replaces the fragment’s `MutableStateFlow<RequirementsModel?>` with a Compose-friendly state model.

```kotlin
package ifac.td.taxi.ui.screen

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.domain.usecase.BravoRestApiUseCase
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.repository.connections.rest.bravoRest.RequirementsModel
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch

class RequirementsComposeViewModel(
    application: Application,
    private val bravoRestApiUseCase: BravoRestApiUseCase
) : AndroidViewModel(application) {

    private val TAG = "RequirementsComposeViewModel"

    private val _uiState = MutableStateFlow(RequirementsUiState())
    val uiState: StateFlow<RequirementsUiState> = _uiState.asStateFlow()

    private val _buttonsState = MutableStateFlow(RequirementsButtonsState())
    val buttonsState: StateFlow<RequirementsButtonsState> = _buttonsState.asStateFlow()

    private val _dialogState = MutableStateFlow<RequirementsDialogState>(RequirementsDialogState.Hidden)
    val dialogState: StateFlow<RequirementsDialogState> = _dialogState.asStateFlow()

    private val _effectFlow = MutableSharedFlow<RequirementsUiEffect>(extraBufferCapacity = 1)
    val effectFlow: SharedFlow<RequirementsUiEffect> = _effectFlow.asSharedFlow()

    fun onEvent(event: RequirementsUiEvent) {
        when (event) {
            RequirementsUiEvent.LoadRequirements -> loadRequirements()
            RequirementsUiEvent.BackPressed -> emitEffect(RequirementsUiEffect.NavigateBack)
            RequirementsUiEvent.DismissDialog -> _dialogState.value = RequirementsDialogState.Hidden
            RequirementsUiEvent.ConfirmDialog -> {
                _dialogState.value = RequirementsDialogState.Hidden
            }
        }
    }

    private fun loadRequirements() {
        _uiState.update { it.copy(isLoading = true) }
        _buttonsState.update { it.reducer(RequirementsButtonsAction.ShowLoading) }

        viewModelScope.launch(Dispatchers.IO) {
            bravoRestApiUseCase.getCarsAndDriverRequirements { requirements ->
                Logs.d(TAG, "getRequirements: $requirements")
                _uiState.update {
                    it.copy(
                        isLoading = false,
                        driverRequirements = requirements?.driverReqs.orEmpty(),
                        vehicleRequirements = requirements?.vehicleReqs.orEmpty()
                    )
                }
                _buttonsState.update { it.reducer(RequirementsButtonsAction.ShowContent) }
            }
        }
    }

    private fun emitEffect(effect: RequirementsUiEffect) {
        _effectFlow.tryEmit(effect)
    }
}
```

### Exact branch-by-branch mapping from the fragment
- `setupComponents()` → `LoadRequirements`
- `showLoading(true)` → `RequirementsButtonsAction.ShowLoading`
- `collect { requirements -> requirements?.let { ... } }` → handled by `loadRequirements()` updating `uiState`
- `showLoading(false)` → `RequirementsButtonsAction.ShowContent`
- `showNoDataMessage(driverRequirements.isEmpty(), ...)` → `ShowEmptyDriver`
- `showNoDataMessage(vehicleRequirements.isEmpty(), ...)` → `ShowEmptyVehicle`
- adapter assignment when list is not empty → Compose `LazyColumn` list rendering

## Notes on preserving Jetpack Views navigation

To preserve navigation compatibility:
- keep navigation actions as callbacks from the composable
- do not hardcode NavController inside the ViewModel
- emit `RequirementsUiEffect.NavigateBack` and let the host decide whether to use:
  - fragment back stack
  - NavController
  - activity finish

That keeps the Compose screen compatible with the existing Jetpack Views navigation model.

```kotlin
            viewModel.effectFlow.collectLatest { effect ->
                when (effect) {
                    RequirementsUiEffect.NavigateBack -> onNavigateBack()
                    is RequirementsUiEffect.ShowMessage -> Unit
                    is RequirementsUiEffect.OpenRequirement -> onRequirementClick(effect.value)
                }
            }
        }
    }

    if (dialogState is RequirementsDialogState.Visible) {
        val dialog = dialogState as RequirementsDialogState.Visible
        AlertDialog(
            onDismissRequest = { viewModel.onEvent(RequirementsUiEvent.DismissDialog) },
            title = { Text(dialog.title) },
            text = { Text(dialog.message) },
            confirmButton = {
                TextButton(onClick = { viewModel.onEvent(RequirementsUiEvent.ConfirmDialog) }) {
                    Text(dialog.confirmText)
                }
            },
            dismissButton = {
                TextButton(onClick = { viewModel.onEvent(RequirementsUiEvent.DismissDialog) }) {
                    Text(dialog.dismissText)
                }
            }
        )
    }

    RequirementsContent(
        uiState = uiState,
        buttonsState = buttonsState,
        onRetry = { viewModel.onEvent(RequirementsUiEvent.LoadRequirements) },
        onBack = { viewModel.onEvent(RequirementsUiEvent.BackPressed) },
        onRequirementClick = onRequirementClick
    )
}

@Composable
private fun RequirementsContent(
    uiState: RequirementsUiState,
    buttonsState: RequirementsButtonsState,
    onRetry: () -> Unit,
    onBack: () -> Unit,
    onRequirementClick: (String) -> Unit
) {
    Column(modifier = Modifier.fillMaxSize()) {
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .padding(8.dp),
            horizontalArrangement = Arrangement.SpaceBetween
        ) {
            if (buttonsState.backButton.visible) {
                RequirementsCustomButton(
                    state = buttonsState.backButton,
                    onClick = onBack
                )
            }
            if (buttonsState.retryButton.visible) {
                RequirementsCustomButton(
                    state = buttonsState.retryButton,
                    onClick = onRetry
                )
            }
        }

        when {
            uiState.isLoading -> {
                Box(
                    modifier = Modifier.fillMaxSize(),
                    contentAlignment = Alignment.Center
                ) {
                    CircularProgressIndicator()
                }
            }

            uiState.driverRequirements.isEmpty() && uiState.vehicleRequirements.isEmpty() -> {
                Box(
                    modifier = Modifier.fillMaxSize(),
                    contentAlignment = Alignment.Center
                ) {
                    Text(text = "No requirements available")
                }
            }

            else -> {
                LazyColumn(
                    modifier = Modifier
                        .fillMaxSize()
                        .padding(8.dp),
                    verticalArrangement = Arrangement.spacedBy(12.dp)
                ) {
                    item {
                        Text(
                            text = "Driver requirements",
                            style = MaterialTheme.typography.titleMedium
                        )
                    }
                    items(uiState.driverRequirements) { item ->
                        Text(
                            text = item.toString(),
                            modifier = Modifier
                                .fillMaxWidth()
                                .padding(8.dp)
                        )
                    }

                    item {
                        Spacer(modifier = Modifier.height(8.dp))
                        Text(
                            text = "Vehicle requirements",
                            style = MaterialTheme.typography.titleMedium
                        )
                    }
                    items(uiState.vehicleRequirements) { item ->
                        Text(
                            text = item.toString(),
                            modifier = Modifier
                                .fillMaxWidth()
                                .padding(8.dp)
                        )
                    }
                }
            }
        }
    }
}

@Composable
private fun RequirementsCustomButton(
    state: RequirementsButtonVisualState,
    onClick: () -> Unit
) {
    AndroidView(
        factory = { context ->
            CustomButton(context).apply {
                setAction(onClick)
            }
        },
        update = { button ->
            button.setButtonStyle(
                when (state.style) {
                    RequirementsButtonStyle.ENABLE -> CustomButton.StyleButton.ENABLE
                    RequirementsButtonStyle.DISABLE -> CustomButton.StyleButton.DISABLE
                    RequirementsButtonStyle.LOADING -> CustomButton.StyleButton.LOADING
                }
            )
            button.changeBackground(
                when (state.background) {
                    RequirementsButtonBackground.GREEN -> CustomButton.BackgroundButtonColor.GREEN
                    RequirementsButtonBackground.RED -> CustomButton.BackgroundButtonColor.RED
                    RequirementsButtonBackground.BLUE -> CustomButton.BackgroundButtonColor.BLUE
                    RequirementsButtonBackground.YELLOW -> CustomButton.BackgroundButtonColor.YELLOW
                    RequirementsButtonBackground.ORANGE -> CustomButton.BackgroundButtonColor.ORANGE
                    RequirementsButtonBackground.DEFAULT -> CustomButton.BackgroundButtonColor.DEFAULT
                    RequirementsButtonBackground.BLACK -> CustomButton.BackgroundButtonColor.BLACK
                    RequirementsButtonBackground.TRANSPARENT -> CustomButton.BackgroundButtonColor.TRANSPARENT
                }
            )
            button.changeText(state.text)
        },
        modifier = Modifier.wrapContentSize()
    )
}
```

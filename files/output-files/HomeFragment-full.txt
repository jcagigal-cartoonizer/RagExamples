Below is a Compose-first version of the `HomeFragment` screen, plus a Compose-friendly `HomeViewModel` design using:

- `UiState`
- `UiEvent`
- `HomeUiEffect` via `SharedFlow`
- dialog state held in state
- lifecycle-safe collection of state/effects
- navigation preserved through `NavController`
- button styling/state holders that closely mirror your custom button behavior
- a Compose `CustomDialog` implementation based on the XML/dialog behavior you described

I’ll keep this focused on Android + Jetpack Compose APIs only, and I’ll show the pieces in a way that you can drop into your project and adapt to your existing domain/use cases.

## Compose screen

This screen preserves navigation via `NavController`, and collects effects/lifecycle safely.

```kotlin
package ifac.td.taxi.ui.screen.home

import android.media.ToneGenerator
import androidx.compose.foundation.background
import androidx.compose.foundation.clickable
import androidx.compose.foundation.layout.*
import androidx.compose.foundation.shape.RoundedCornerShape
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.platform.LocalContext
import androidx.compose.ui.unit.dp
import androidx.lifecycle.Lifecycle
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.lifecycle.compose.LocalLifecycleOwner
import androidx.lifecycle.repeatOnLifecycle
import androidx.navigation.NavController
import androidx.navigation.NavDeepLinkRequest
import androidx.navigation.compose.currentBackStackEntryAsState
import androidx.core.net.toUri
import kotlinx.coroutines.launch

@Composable
fun HomeRoute(
    navController: NavController,
    viewModel: HomeComposeViewModel,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val lifecycleOwner = LocalLifecycleOwner.current
    val context = LocalContext.current

    LaunchedEffect(Unit) {
        lifecycleOwner.repeatOnLifecycle(Lifecycle.State.STARTED) {
            launch {
                viewModel.effects.collect { effect ->
                    when (effect) {
                        is HomeUiEffect.Navigate -> navController.navigate(effect.route)
                        is HomeUiEffect.NavigateDeepLink -> {
                            val req = NavDeepLinkRequest.Builder
                                .fromUri(effect.uri.toUri())
                                .build()
                            navController.navigate(req)
                        }
                        HomeUiEffect.NavigateBack -> navController.popBackStack()

                        is HomeUiEffect.ShowToast -> {
                            android.widget.Toast.makeText(
                                context,
                                context.getString(effect.resId),
                                android.widget.Toast.LENGTH_SHORT
                            ).show()
                        }

                        is HomeUiEffect.Beep -> {
                            ToneGenerator(
                                android.media.AudioManager.STREAM_ALARM,
                                100
                            ).startTone(effect.tone, 150)
                        }

                        HomeUiEffect.OpenManualTripDialog,
                        HomeUiEffect.OpenLogoutDialog,
                        HomeUiEffect.OpenDeactivateLocationDialog,
                        HomeUiEffect.OpenActivateLocationDialog,
                        HomeUiEffect.OpenLocateOnStandDialog -> {
                            viewModel.showDialogForEffect(effect)
                        }

                        HomeUiEffect.NavigateToPendingTrips -> {
                            navController.navigate("pendingTrips") // preserve your route/action here
                        }
                    }
                }
            }
        }
    }

    HomeScreen(
        state = uiState,
        onAction = viewModel::onAction,
        onDialogAction = viewModel::onDialogAction
    )
}

@Composable
fun HomeScreen(
    state: HomeUiState,
    onAction: (HomeUiAction) -> Unit,
    onDialogAction: (HomeDialogAction) -> Unit,
) {
    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(Color(0xFFF2F2F2))
    ) {
        Column(
            modifier = Modifier
                .fillMaxSize()
                .padding(16.dp),
            verticalArrangement = Arrangement.spacedBy(12.dp)
        ) {
            Row(horizontalArrangement = Arrangement.spacedBy(12.dp)) {
                HomeButton(state.buttons.zoning, "Zoning", { onAction(HomeUiAction.ZoningClicked) })
                HomeButton(state.buttons.pending, "Pending", { onAction(HomeUiAction.PendingClicked) })
            }
            Row(horizontalArrangement = Arrangement.spacedBy(12.dp)) {
                HomeButton(state.buttons.location, state.buttons.location.text ?: "Location") {
                    onAction(HomeUiAction.LocationClicked)
                }
                HomeButton(state.buttons.locateStand, state.buttons.locateStand.text ?: "Locate") {
                    onAction(HomeUiAction.LocateStandClicked)
                }
            }
            Row(horizontalArrangement = Arrangement.spacedBy(12.dp)) {
                HomeButton(state.buttons.receipts, "Receipts") { onAction(HomeUiAction.ReceiptsClicked) }
                HomeButton(state.buttons.messages, "Messages") { onAction(HomeUiAction.MessagesClicked) }
            }
            Row(horizontalArrangement = Arrangement.spacedBy(12.dp)) {
                if (state.buttons.dashboard.visible) {
                    HomeButton(state.buttons.dashboard, "Dashboard") { onAction(HomeUiAction.DashboardClicked) }
                }
                if (state.buttons.fixedPrice.visible) {
                    HomeButton(state.buttons.fixedPrice, "Fixed Price") { onAction(HomeUiAction.FixedPriceClicked) }
                }
            }
            Row(horizontalArrangement = Arrangement.spacedBy(12.dp)) {
                HomeButton(state.buttons.central, "Central") { onAction(HomeUiAction.CentralClicked) }
                HomeButton(state.buttons.roofLight, "Roof Light") { onAction(HomeUiAction.RoofLightClicked) }
            }
        }

        state.dialog?.let { dialog ->
            CustomDialog(
                state = dialog,
                onAction = onDialogAction
            )
        }
    }
}

@Composable
private fun HomeButton(
    state: ButtonUiState,
    text: String,
    onClick: () -> Unit
) {
    val background = when (state.background) {
        ButtonBackground.Blue -> Color(0xFF1E88E5)
        ButtonBackground.Red -> Color(0xFFE53935)
        ButtonBackground.Green -> Color(0xFF43A047)
        ButtonBackground.Orange -> Color(0xFFFF9800)
        ButtonBackground.Transparent -> Color.Transparent
    }

    val shape = RoundedCornerShape(14.dp)

    val modifier = Modifier
        .weight(1f)
        .height(58.dp)
        .then(
            if (state.clickable) Modifier.clickable(onClick = onClick) else Modifier
        )
        .background(
            color = if (state.style == ButtonStyle.Disabled) Color(0xFFBDBDBD) else background,
            shape = shape
        )

    Box(
        modifier = modifier,
        contentAlignment = Alignment.Center
    ) {
        if (state.loading) {
            CircularProgressIndicator(
                color = Color.White,
                modifier = Modifier.size(22.dp),
                strokeWidth = 2.dp
            )
        } else {
            Text(
                text = state.text ?: text,
                color = state.onTextColor,
                style = MaterialTheme.typography.titleMedium
            )
        }
    }
}
```

```kotlin
package ifac.td.taxi.ui.screen.home

    private val _uiState = MutableStateFlow(HomeUiState())
    val uiState: StateFlow<HomeUiState> = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<HomeUiEffect>()
    val effects: SharedFlow<HomeUiEffect> = _effects.asSharedFlow()

    private var pendingServicesButton: Boolean? = null

    init {
        refreshInitialState()
    }

    private fun refreshInitialState() {
        viewModelScope.launch(Dispatchers.IO) {
            val hasDashboard = bravoCentralConfigurationUseCase.getBravoConfigurationVariable()?.isHomeButton == true
            val fixedPriceVisible = !bravoCentralConfigurationUseCase.getBravoConfigurationVariable()?.fixedPricesStreetTripsURL.isNullOrBlank()
            val locationDisabled = licensingUseCase.getLicensingParameters()?.isDisableLocation
            val pendingAllowed = licensingUseCase.getLicensingParameters()?.isVacantPendingServices
            val roofLightDisableLuminous = licensingUseCase.getLicensingParameters()?.isDisableLuminous ?: false

            _uiState.update { current ->
                current.copy(
                    buttons = current.buttons.copy(
                        dashboard = current.buttons.dashboard.copy(visible = hasDashboard),
                        fixedPrice = current.buttons.fixedPrice.copy(visible = fixedPriceVisible),
                        location = current.buttons.location.copy(enabled = locationDisabled != true),
                        pending = current.buttons.pending.copy(enabled = pendingAllowed == true),
                        roofLight = current.buttons.roofLight.copy(enabled = roofLightDisableLuminous),
                    )
                )
            }

            getMessages()
            checkKeepScreenOn()
            checkForLocationPermission()
            checkPendingServiceOnForHirePermission()
        }
    }

    fun onAction(action: HomeUiAction) {
        when (action) {
            HomeUiAction.ZoningClicked -> navigateToZoning()
            HomeUiAction.PendingClicked -> onPendingClicked()
            HomeUiAction.LocationClicked -> onLocationClicked()
            HomeUiAction.ReceiptsClicked -> navigateToReceiptHistory()
            HomeUiAction.MessagesClicked -> navigateToMessage()
            HomeUiAction.CentralClicked -> navigateToContactCentral()
            HomeUiAction.DashboardClicked -> viewModelScope.launch { _effects.emit(HomeUiEffect.Navigate("dashboard")) }
            HomeUiAction.FixedPriceClicked -> navigateToEstimateFixedPriceScreen()
            HomeUiAction.LocateStandClicked -> triggerLocateOnStand()
            HomeUiAction.RoofLightClicked -> onRoofLightClicked()
        }
    }

    fun onDialogAction(action: HomeDialogAction) {
        when (action) {
            HomeDialogAction.Dismiss -> _uiState.update { it.copy(dialog = null) }
            HomeDialogAction.Accept -> {
                val dialog = _uiState.value.dialog
                _uiState.update { it.copy(dialog = null) }
                when (dialog?.titleRes) {
                    R.string.confirm_close_shift -> logoff()
                    R.string.confirm_deactivate_location -> disableLocation()
                    R.string.confirm_activate_location -> enableLocation()
                    R.string.dialog_hired_manual_title -> changeStateHiredManual()
                    R.string.dialog_exit_onStand_title -> sendLocateZone()
                }
            }
            HomeDialogAction.Cancel -> _uiState.update { it.copy(dialog = null) }
        }
    }

    fun showDialogForEffect(effect: HomeUiEffect) {
        val dialog = when (effect) {
            HomeUiEffect.OpenLogoutDialog -> HomeDialogState(
                titleRes = R.string.confirm_close_shift,
                buttons = listOf(
                    DialogButtonSpec(DialogButtonType.CANCEL, android.R.string.cancel),
                    DialogButtonSpec(DialogButtonType.ACCEPT, android.R.string.ok)

            HomeUiEffect.OpenDeactivateLocationDialog -> HomeDialogState(
                titleRes = R.string.confirm_deactivate_location,
                buttons = listOf(
                    DialogButtonSpec(DialogButtonType.CANCEL, android.R.string.cancel),
                    DialogButtonSpec(DialogButtonType.ACCEPT, android.R.string.ok)
                )
            )
            HomeUiEffect.OpenActivateLocationDialog -> HomeDialogState(
                titleRes = R.string.confirm_activate_location,
                buttons = listOf(
                    DialogButtonSpec(DialogButtonType.CANCEL, android.R.string.cancel),
                    DialogButtonSpec(DialogButtonType.ACCEPT, android.R.string.ok)
                )
            )
            HomeUiEffect.OpenManualTripDialog -> HomeDialogState(
                titleRes = R.string.dialog_hired_manual_title,
                descriptionRes = R.string.confirm_change_to_hired,
                buttons = listOf(
                    DialogButtonSpec(DialogButtonType.CANCEL, android.R.string.cancel),
                    DialogButtonSpec(DialogButtonType.ACCEPT, android.R.string.ok)
                )
            )
            HomeUiEffect.OpenLocateOnStandDialog -> HomeDialogState(
                titleRes = R.string.dialog_exit_onStand_title,
                buttons = listOf(
                    DialogButtonSpec(DialogButtonType.CANCEL, android.R.string.cancel),
                    DialogButtonSpec(DialogButtonType.ACCEPT, android.R.string.ok)
                )
            )
            else -> null
        }

        if (dialog != null) {
            _uiState.update { it.copy(dialog = dialog) }
        }
    }

    private fun onPendingClicked() {
        viewModelScope.launch {
            if (pendingServicesButton == true) {
                _effects.emit(HomeUiEffect.NavigateToPendingTrips)
            } else {
                _effects.emit(HomeUiEffect.ShowToast(R.string.toast_no_hay_pendientes))
            }
        }
    }

    private fun onLocationClicked() {
        viewModelScope.launch {
            val state = _uiState.value.buttons.location
            if (state.enabled) {
                if (state.background == ButtonBackground.Green) {
                    _effects.emit(HomeUiEffect.OpenDeactivateLocationDialog)
                } else {
                    _effects.emit(HomeUiEffect.OpenActivateLocationDialog)
                }
            }
        }
    }

    private fun onRoofLightClicked() {
        viewModelScope.launch {
            val roof = _uiState.value.buttons.roofLight
            if (!roof.enabled) return@launch
            _uiState.update { it.copy(buttons = it.buttons.copy(roofLight = roof.copy(loading = true))) }
            if (roof.background == ButtonBackground.Green) setRoofLightOff() else setRoofLightOn()
        }
    }

    fun getMessages() {
        viewModelScope.launch(Dispatchers.IO) {
            val messages = messageUseCase.getMessageByDriverId()
            val result = when {
                messages.isNullOrEmpty() -> MessageUIEnum.NO_MESSAGES
                messages.any { !it.isRead && it.messageType == MessageType.MESSAGE } -> MessageUIEnum.HAS_NEW_MESSAGES
                else -> MessageUIEnum.HAS_MESSAGES_BUT_NOT_NEW
            }

            _uiState.update { current ->
                current.copy(
                    buttons = current.buttons.copy(
                        messages = current.buttons.messages.copy(
                            enabled = result != MessageUIEnum.NO_MESSAGES,
                            background = when (result) {
                                MessageUIEnum.NO_MESSAGES -> ButtonBackground.Blue
                                MessageUIEnum.HAS_MESSAGES_BUT_NOT_NEW -> ButtonBackground.Blue
                                MessageUIEnum.HAS_NEW_MESSAGES -> ButtonBackground.Orange
                            },
                            style = when (result) {
                                MessageUIEnum.NO_MESSAGES -> ButtonStyle.Disabled
                                else -> ButtonStyle.Enabled
                            }
                        )
                    )
                )
            }
        }
    }

    fun checkForLocationPermission() {
        viewModelScope.launch {
            val disabled = licensingUseCase.getLicensingParameters()?.isDisableLocation
            _uiState.update { current ->
                current.copy(
                    buttons = current.buttons.copy(
                        location = current.buttons.location.copy(

                            text = if (disabled == true) context.getString(R.string.btn_location_on) else context.getString(R.string.btn_location_off),
```

```kotlin
package ifac.td.taxi.ui.screen.home
```

## 4. Room 3 Data Layer

**`data/local/EntryEntity.kt`**

```kotlin
package com.example.begin.data.local

import androidx.room3.Entity
import androidx.room3.PrimaryKey

@Entity(tableName = "entries")
data class EntryEntity(
    @PrimaryKey(autoGenerate = true) val id: Long = 0,
    val text: String,
    val createdAt: Long = System.currentTimeMillis()
)
```

**`data/local/EntryDao.kt`**

```kotlin
package com.example.begin.data.local

import androidx.room3.Dao
import androidx.room3.Insert
import androidx.room3.Query
import kotlinx.coroutines.flow.Flow

@Dao
interface EntryDao {

    @Insert
    suspend fun insert(entry: EntryEntity): Long

    @Query("SELECT * FROM entries ORDER BY createdAt DESC")
    fun observeAll(): Flow<List<EntryEntity>>

    @Query("DELETE FROM entries")
    suspend fun clear()
}
```

**`data/local/AppDatabase.kt`**

```kotlin
package com.example.begin.data.local

import androidx.room3.Database
import androidx.room3.RoomDatabase

@Database(
    entities = [EntryEntity::class],
    version = 1,
    exportSchema = false
)
abstract class AppDatabase : RoomDatabase() {
    abstract fun entryDao(): EntryDao
}
```

**`data/repository/EntryRepository.kt`**

```kotlin
package com.example.begin.data.repository

import com.example.begin.data.local.EntryDao
import com.example.begin.data.local.EntryEntity
import kotlinx.coroutines.flow.Flow
import javax.inject.Inject
import javax.inject.Singleton

@Singleton
class EntryRepository @Inject constructor(
    private val entryDao: EntryDao
) {
    fun observeEntries(): Flow<List<EntryEntity>> = entryDao.observeAll()

    suspend fun addEntry(text: String): Long =
        entryDao.insert(EntryEntity(text = text))
}
```

## 5. Dependency Injection

**`di/DatabaseModule.kt`**

```kotlin
package com.example.begin.di

import android.content.Context
import androidx.room3.Room
import androidx.sqlite.driver.bundled.BundledSQLiteDriver
import com.example.begin.data.local.AppDatabase
import com.example.begin.data.local.EntryDao
import dagger.Module
import dagger.Provides
import dagger.hilt.InstallIn
import dagger.hilt.android.qualifiers.ApplicationContext
import dagger.hilt.components.SingletonComponent
import javax.inject.Singleton

@Module
@InstallIn(SingletonComponent::class)
object DatabaseModule {

    @Provides
    @Singleton
    fun provideDatabase(@ApplicationContext context: Context): AppDatabase =
        Room.databaseBuilder(context, AppDatabase::class.java, "begin.db")
            .setDriver(BundledSQLiteDriver())
            .build()

    @Provides
    fun provideEntryDao(database: AppDatabase): EntryDao =
        database.entryDao()
}
```

## 6. Navigation 3

**`navigation/Routes.kt`**

```kotlin
package com.example.begin.navigation

import androidx.navigation3.runtime.NavKey
import kotlinx.serialization.Serializable

@Serializable
data object MainRoute : NavKey
```

**`navigation/AppNavigation.kt`**

```kotlin
package com.example.begin.navigation

import androidx.compose.runtime.Composable
import androidx.lifecycle.viewmodel.navigation3.rememberViewModelStoreNavEntryDecorator
import androidx.navigation3.runtime.entryProvider
import androidx.navigation3.runtime.rememberNavBackStack
import androidx.navigation3.runtime.rememberSaveableStateHolderNavEntryDecorator
import androidx.navigation3.ui.NavDisplay
import com.example.begin.ui.main.MainScreen

@Composable
fun AppNavigation() {
    val backStack = rememberNavBackStack(MainRoute)

    NavDisplay(
        backStack = backStack,
        onBack = { backStack.removeLastOrNull() },
        entryDecorators = listOf(
            rememberSaveableStateHolderNavEntryDecorator(),
            rememberViewModelStoreNavEntryDecorator()
        ),
        entryProvider = entryProvider {
            entry<MainRoute> {
                MainScreen()
            }
        }
    )
}
```

## 7. UI and MVVM

**`ui/main/MainViewModel.kt`**

```kotlin
package com.example.begin.ui.main

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.example.begin.data.repository.EntryRepository
import dagger.hilt.android.lifecycle.HiltViewModel
import kotlinx.coroutines.flow.SharingStarted
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.map
import kotlinx.coroutines.flow.stateIn
import javax.inject.Inject

data class MainUiState(
    val label: String = "Begin",
    val entryCount: Int = 0
)

@HiltViewModel
class MainViewModel @Inject constructor(
    private val repository: EntryRepository
) : ViewModel() {

    val uiState: StateFlow<MainUiState> = repository.observeEntries()
        .map { entries -> MainUiState(entryCount = entries.size) }
        .stateIn(
            scope = viewModelScope,
            started = SharingStarted.WhileSubscribed(5_000),
            initialValue = MainUiState()
        )
}
```

**`ui/main/MainScreen.kt`**

```kotlin
package com.example.begin.ui.main

import androidx.compose.foundation.layout.Box
import androidx.compose.foundation.layout.fillMaxSize
import androidx.compose.foundation.layout.padding
import androidx.compose.material3.MaterialTheme
import androidx.compose.material3.Scaffold
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.runtime.getValue
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.hilt.navigation.compose.hiltViewModel

@Composable
fun MainScreen(
    viewModel: MainViewModel = hiltViewModel()
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()

    Scaffold { innerPadding ->
        Box(
            modifier = Modifier
                .fillMaxSize()
                .padding(innerPadding),
            contentAlignment = Alignment.Center
        ) {
            Text(
                text = uiState.label,
                style = MaterialTheme.typography.headlineLarge
            )
        }
    }
}
```

## How It Fits Together

| Layer | Responsibility |
|---|---|
| **Room 3** | `AppDatabase`, `EntryDao` (suspend writes, `Flow` reads), built with `BundledSQLiteDriver` and processed by KSP |
| **Repository** | Single source of data for the ViewModel |
| **ViewModel** | Exposes `StateFlow<MainUiState>` via `stateIn`, so the UI survives configuration changes |
| **Compose UI** | `MainScreen` collects state and renders the "Begin" label |
| **Navigation 3** | `rememberNavBackStack` holds the back stack, `NavDisplay` renders the `MainRoute` entry, and the view-model decorator scopes the ViewModel to each entry |
| **Hilt** | Injects the database, DAO, repository, and ViewModel |

Running the app shows a centered label with the text **"Begin"**.

## Notes

- **Room 3 is in alpha.** If the build fails on `room3` artifacts, check the latest version on Google Maven and confirm the package names (`androidx.room3.*`) against the current release notes.
- **Navigation 3 APIs change between alphas.** If `rememberNavBackStack`, `entryProvider`, or the decorator functions are not found, update `navigation3Version` and the lifecycle `viewmodel-navigation3` artifact to matching versions.
- Add a new route by creating another `@Serializable data object` or `data class` implementing `NavKey`, then adding an `entry<NewRoute> { ... }` block to `entryProvider`.

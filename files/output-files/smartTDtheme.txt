Below is a Jetpack Compose theme setup that mirrors the important properties of your XML theme using only Android / Jetpack Compose references.

It covers:

- `colorPrimary`, `colorPrimaryVariant`
- `colorSecondary`, `colorSecondaryVariant`
- `colorOnPrimary`, `colorOnSecondary`
- dark background
- status bar color
- no action bar / no title equivalent in Compose usage
- font family
- bottom sheet theme equivalent
- examples for spinner, text fields, switch, and top bar button styling in Compose

---

## Compose theme object

```kotlin
package com.example.smarttd.ui.theme

import android.app.Activity
import android.os.Build
import android.view.Window
import androidx.compose.foundation.isSystemInDarkTheme
import androidx.compose.material.*
import androidx.compose.material.darkColors
import androidx.compose.runtime.Composable
import androidx.compose.runtime.SideEffect
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.platform.LocalContext
import androidx.compose.ui.platform.LocalView
import androidx.compose.ui.text.font.Font
import androidx.compose.ui.text.font.FontFamily
import androidx.compose.ui.text.font.FontWeight
import androidx.core.view.WindowCompat
import androidx.core.view.WindowInsetsControllerCompat

// Replace these with your actual color values or resource-based values.
private val Blue = Color(0xFF1E88E5)
private val BlueTrans = Color(0x801E88E5)
private val Grey = Color(0xFFBDBDBD)
private val Green = Color(0xFF43A047)
private val Red = Color(0xFFE53935)
private val DarkBackground = Color(0xFF121212)
private val White = Color(0xFFFFFFFF)
private val DisabledWhite = Color(0x33FFFFFF)
private val DisabledGrey = Color(0xFF888888)

val MontserratMedium = FontFamily(
    Font(resId = R.font.montserrat_medium, weight = FontWeight.Medium)
)

private val SmartTDLightColors = darkColors(
    primary = Blue,
    primaryVariant = BlueTrans,
    onPrimary = Grey,

    secondary = Green,
    secondaryVariant = Red,
    onSecondary = Grey,

    background = DarkBackground,
    surface = DarkBackground,
    onBackground = White,
    onSurface = White,

    error = Red,
    onError = White
)

@Composable
fun SmartTDTheme(
    darkTheme: Boolean = true,
    content: @Composable () -> Unit
) {
    val colors = if (darkTheme || isSystemInDarkTheme()) {
        SmartTDLightColors
    } else {
        SmartTDLightColors
    }

    val view = LocalView.current
    if (!view.isInEditMode) {
        SideEffect {
            val window = (view.context as Activity).window
            window.statusBarColor = BlueTrans.toArgb()
            WindowCompat.setDecorFitsSystemWindows(window, true)

            val controller = WindowInsetsControllerCompat(window, view)
            controller.isAppearanceLightStatusBars = false
        }
    }

    MaterialTheme(
        colors = colors,
        typography = SmartTDTypography,
        shapes = Shapes,
        content = content
    )
}
```

## Compose equivalents for your styled widgets

### 1) Spinner equivalent

Compose doesn’t use `Spinner` directly, so use `ExposedDropdownMenuBox`:

```kotlin
@OptIn(ExperimentalMaterialApi::class)
@Composable
fun SmartTDSpinner(
    selected: String,
    items: List<String>,
    onItemSelected: (String) -> Unit
) {
    var expanded = false

    ExposedDropdownMenuBox(
        expanded = expanded,
        onExpandedChange = { expanded = !expanded }
    ) {
        TextField(
            value = selected,
            onValueChange = {},
            readOnly = true,
            textStyle = LocalTextStyle.current.copy(
                color = White,
                fontFamily = MontserratMedium
            ),
            colors = TextFieldDefaults.textFieldColors(
                backgroundColor = DisabledWhite,
                textColor = White,
                disabledTextColor = White,
                placeholderColor = DisabledGrey,
                focusedIndicatorColor = Color.Transparent,
                unfocusedIndicatorColor = Color.Transparent
            ),
            trailingIcon = {
                ExposedDropdownMenuDefaults.TrailingIcon(expanded = expanded)
            }
        )

        ExposedDropdownMenu(
            expanded = expanded,
            onDismissRequest = { expanded = false }
        ) {
            items.forEach { item ->
                DropdownMenuItem(onClick = {
                    onItemSelected(item)
                    expanded = false
                }) {
                    Text(
                        text = item,
                        color = White,
                        fontFamily = MontserratMedium
                    )
                }
            }
        }
    }
}
```

### 2) Custom EditText equivalent

Use `OutlinedTextField` or `TextField`:

```kotlin
@Composable
fun SmartTDTextField(
    value: String,
    onValueChange: (String) -> Unit,
    hint: String = ""
) {
    TextField(
        value = value,
        onValueChange = onValueChange,
        placeholder = {
            Text(text = hint, color = DisabledGrey, fontFamily = MontserratMedium)
        },
        textStyle = androidx.compose.ui.text.TextStyle(
            color = White,
            fontFamily = MontserratMedium
        ),
        colors = TextFieldDefaults.textFieldColors(
            textColor = White,
            backgroundColor = Color.Transparent,
            cursorColor = White,
            focusedIndicatorColor = White,
            unfocusedIndicatorColor = Grey,
            placeholderColor = DisabledGrey
        )
    )
}
```

## Bottom sheet theme equivalent

In Compose, use `ModalBottomSheetLayout` and style it through Material theming and sheet colors.

```kotlin
@OptIn(ExperimentalMaterialApi::class)
@Composable
fun SmartTDBottomSheet(
    sheetContent: @Composable () -> Unit,
    content: @Composable () -> Unit
) {
    ModalBottomSheetLayout(
        sheetContent = sheetContent,
        sheetBackgroundColor = DarkBackground,
        sheetContentColor = White,
        content = content
    )
}
```

## Notes on XML-to-Compose mapping

### XML properties mapped to Compose

- `colorPrimary` → `MaterialTheme.colors.primary`
- `colorPrimaryVariant` → `MaterialTheme.colors.primaryVariant`
- `colorOnPrimary` → `MaterialTheme.colors.onPrimary`
- `colorSecondary` → `MaterialTheme.colors.secondary`
- `colorSecondaryVariant` → `MaterialTheme.colors.secondaryVariant`
- `colorOnSecondary` → `MaterialTheme.colors.onSecondary`
- `android:colorBackground` → `MaterialTheme.colors.background`
- `fontFamily` → `Typography(defaultFontFamily = ...)`
- `android:statusBarColor` → set using `window.statusBarColor`
- `windowActionBar=false` / `windowNoTitle=true` → Compose typically uses `ComponentActivity` without an action bar, and `Scaffold/TopAppBar` only when needed
- `bottomSheetDialogTheme` → `ModalBottomSheetLayout` / `BottomSheetScaffold`

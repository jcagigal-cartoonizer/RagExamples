## 1) Compose screen

This composable:
- hides the header via the main-activity bridge
- mirrors the fragment’s validation + loading behavior
- collects `SharedFlow<ChangePasswordRedSysUiEffect>` with lifecycle awareness
- optionally shows a dialog if you decide to keep the `CustomDialog` behavior
- navigates back exactly like the fragment did

```kotlin
@Composable
fun ChangePasswordRedSysRoute(
    viewModel: ChangePasswordRedSysComposeViewModel,
    mainActivityViewModel: ChangePasswordRedSysMainActivityViewModel,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()

    val context = LocalContext.current

    LaunchedEffect(Unit) {
        mainActivityViewModel.onScreenStarted()
    }

    // Lifecycle-aware effect collection
    LaunchedEffect(viewModel) {
        viewModel.effects.collectLatest { effect ->
            when (effect) {
                ChangePasswordRedSysUiEffect.NavigateBack -> {
                    mainActivityViewModel.navigateBack()
                }

                is ChangePasswordRedSysUiEffect.ShowDialog -> {
                    mainActivityViewModel.openDialog(effect.dialogModel)
                }

                ChangePasswordRedSysUiEffect.ResetAcceptButton -> {
                    viewModel.onButtonsStateEvent(
                        ChangePasswordRedSysButtonsEvent.SetAcceptEnabled(true)
                    )
                }
            }
        }
    }

    // Dialog bridge (only if you keep CustomDialog)
    val dialogState = mainActivityViewModel.dialogState.collectAsStateWithLifecycle()

    ChangePasswordRedSysScreen(
        uiState = uiState,
        dialogState = dialogState.value,
        onEvent = viewModel::onEvent,
        onButtonEvent = viewModel::onButtonsStateEvent,
        onDialogResponse = mainActivityViewModel::onDialogResponse,
        onCancel = { mainActivityViewModel.navigateBack() }
    )
}
```

## 2) Compose-friendly ViewModel with `UiState + UiEvent + UiEffect`

This is the Compose replacement for the fragment ViewModel.  
It keeps the same business behavior, but exposes:
- `StateFlow<ChangePasswordRedSysUiState>`
- `SharedFlow<ChangePasswordRedSysUiEffect>`

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import es.redsys.paysys.Operative.DTO.RedCLSChangePassData
import es.redsys.paysys.Utils.RedCLSErrorCodes
import ifac.td.taxi.domain.usecase.RedSysUseCase
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.ui.custom.button.CustomButton
import ifac.td.taxi.ui.custom.dialog.CustomDialog
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch

data class ChangePasswordRedSysUiState(
    val user: String = "",
    val password: String = "",
    val newPassword: String = "",
    val repeatNewPassword: String = "",
    val buttonsState: ChangePasswordRedSysButtonsState = ChangePasswordRedSysButtonsState()
)

sealed interface ChangePasswordRedSysUiEvent {
    data class UserChanged(val value: String) : ChangePasswordRedSysUiEvent
    data class PasswordChanged(val value: String) : ChangePasswordRedSysUiEvent
    data class NewPasswordChanged(val value: String) : ChangePasswordRedSysUiEvent
    data class RepeatNewPasswordChanged(val value: String) : ChangePasswordRedSysUiEvent
}

sealed interface ChangePasswordRedSysButtonsEvent {
    data object AcceptClicked : ChangePasswordRedSysButtonsEvent
    data class SetAcceptEnabled(val enabled: Boolean) : ChangePasswordRedSysButtonsEvent
    data class SetAcceptLoading(val loading: Boolean) : ChangePasswordRedSysButtonsEvent
}

data class ChangePasswordRedSysButtonsState(
    val acceptVisible: Boolean = true,
    val acceptEnabled: Boolean = true,
    val acceptStyle: CustomButton.StyleButton = CustomButton.StyleButton.ENABLE,
    val cancelVisible: Boolean = true,
    val cancelEnabled: Boolean = true,
    val cancelStyle: CustomButton.StyleButton = CustomButton.StyleButton.ENABLE,
)

sealed interface ChangePasswordRedSysUiEffect {
    data object NavigateBack : ChangePasswordRedSysUiEffect
    data object ResetAcceptButton : ChangePasswordRedSysUiEffect
    data class ShowDialog(val dialogModel: CustomDialog.CustomDialogModel) : ChangePasswordRedSysUiEffect
}

class ChangePasswordRedSysComposeViewModel(
    private val redSysUseCase: RedSysUseCase,
    application: Application,
) : AndroidViewModel(application) {

    private val TAG = "ChangePasswordRedSysComposeViewModel"

    private val _uiState = MutableStateFlow(ChangePasswordRedSysUiState())
    val uiState: StateFlow<ChangePasswordRedSysUiState> = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<ChangePasswordRedSysUiEffect>()
    val effects: SharedFlow<ChangePasswordRedSysUiEffect> = _effects.asSharedFlow()

    fun onEvent(event: ChangePasswordRedSysUiEvent) {
        when (event) {
            is ChangePasswordRedSysUiEvent.UserChanged ->
                _uiState.update { it.copy(user = event.value) }

            is ChangePasswordRedSysUiEvent.PasswordChanged ->
                _uiState.update { it.copy(password = event.value) }

            is ChangePasswordRedSysUiEvent.NewPasswordChanged ->
                _uiState.update { it.copy(newPassword = event.value) }

            is ChangePasswordRedSysUiEvent.RepeatNewPasswordChanged ->
                _uiState.update { it.copy(repeatNewPassword = event.value) }
        }
    }

    fun onButtonsStateEvent(event: ChangePasswordRedSysButtonsEvent) {
        when (event) {
            ChangePasswordRedSysButtonsEvent.AcceptClicked -> clickOnAccept()
            is ChangePasswordRedSysButtonsEvent.SetAcceptEnabled ->
                _uiState.update {
                    it.copy(
                        buttonsState = it.buttonsState.copy(
                            acceptEnabled = event.enabled,
                            acceptStyle = if (event.enabled) CustomButton.StyleButton.ENABLE else CustomButton.StyleButton.DISABLE
                        )
                    )
                }

            is ChangePasswordRedSysButtonsEvent.SetAcceptLoading ->
                _uiState.update {
                    it.copy(
                        buttonsState = it.buttonsState.copy(
                            acceptEnabled = !event.loading,
                            acceptStyle = if (event.loading) CustomButton.StyleButton.LOADING else CustomButton.StyleButton.ENABLE
                        )
                    )
                }
        }
    }

    private fun clickOnAccept() {
        val state = _uiState.value

        if (state.user.isNotBlank() &&
            state.password.isNotBlank() &&
            state.newPassword.isNotBlank() &&
            state.repeatNewPassword.isNotBlank() &&
            state.newPassword == state.repeatNewPassword
        ) {
            _uiState.update {
                it.copy(
                    buttonsState = it.buttonsState.copy(
                        acceptStyle = CustomButton.StyleButton.LOADING,
                        acceptEnabled = false
                    )
                )
            }

            changePassword(
                user = state.user,
                password = state.password,
                newPassword = state.newPassword
            )
        }
    }

    fun changePassword(user: String, password: String, newPassword: String) {
        val clsChange = RedCLSChangePassData(
            getApplication(),
            user,
            password,
            newPassword
        )

        viewModelScope.launch {
            val result = redSysUseCase.changePassword(clsChange)
            result?.let {
                Logs.d(
                    TAG,
                    "RedSys result: code: ${result.code}, desc: ${result.desc}, firma: ${result.firma}, mensaje: ${result.mensaje}, session: ${result.session}"
                )

                if (it.code == RedCLSErrorCodes.STATUS_OK) {
                    _effects.emit(
                        ChangePasswordRedSysUiEffect.ShowDialog(
                            CustomDialog.CustomDialogModel(
                                title = getApplication<Application>().getString(R.string.red_sys_title),
                                description = getApplication<Application>().getString(R.string.dialog_change_password),
                                buttons = arrayListOf(ButtonType.ACCEPT)
                            )
                        )
                    )
                    _effects.emit(ChangePasswordRedSysUiEffect.NavigateBack)
                } else {
                    _effects.emit(
                        ChangePasswordRedSysUiEffect.ShowDialog(
                            CustomDialog.CustomDialogModel(
                                title = getApplication<Application>().getString(R.string.red_sys_title),
                                description = it.desc ?: getApplication<Application>().getString(R.string.no_response_error),
                                buttons = arrayListOf(ButtonType.ACCEPT)
                            )
                        )
                    )
                    _effects.emit(ChangePasswordRedSysUiEffect.ResetAcceptButton)
                }
            }
        }
    }
}
```

## 3) Exact `ChangePasswordRedSysButtonsState` reducer mirroring the fragment branches

This reducer mirrors the fragment logic one-by-one:

- `btnCancel.setAction { navigateBack() }`
- `btnAccept.setAction { clickOnAccept() }`
- on accept click:
  - validate fields
  - set loading
  - call `changePassword`
- on flow success:
  - navigate back
- on flow failure:
  - reset accept button to enabled

```kotlin

        is ChangePasswordRedSysButtonsEvent.SetAcceptEnabled -> current.copy(

            acceptStyle = if (event.enabled) {
                CustomButton.StyleButton.ENABLE
            } else {
                CustomButton.StyleButton.DISABLE

        is ChangePasswordRedSysButtonsEvent.SetAcceptLoading -> current.copy(
            acceptEnabled = !event.loading,
            acceptStyle = if (event.loading) {
```

```kotlin
fun reduceChangePasswordRedSysButtonsState(
    current: ChangePasswordRedSysButtonsState,
    event: ChangePasswordRedSysButtonsEvent
): ChangePasswordRedSysButtonsState {
    return when (event) {
        ChangePasswordRedSysButtonsEvent.CancelClicked -> current
        ChangePasswordRedSysButtonsEvent.AcceptClicked -> current
        is ChangePasswordRedSysButtonsEvent.SetAcceptEnabled -> current.copy(
            acceptEnabled = event.enabled,
            acceptStyle = if (event.enabled) CustomButton.StyleButton.ENABLE else CustomButton.StyleButton.DISABLE
        )
        is ChangePasswordRedSysButtonsEvent.SetAcceptLoading -> current.copy(
            acceptEnabled = !event.loading,
            acceptStyle = if (event.loading) CustomButton.StyleButton.LOADING else CustomButton.StyleButton.ENABLE
        )
    }
}

class ChangePasswordRedSysComposeViewModel(

    private val _uiState = MutableStateFlow(ChangePasswordRedSysUiState())
    val uiState: StateFlow<ChangePasswordRedSysUiState> = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<ChangePasswordRedSysUiEffect>()
    val effects: SharedFlow<ChangePasswordRedSysUiEffect> = _effects.asSharedFlow()

    fun onEvent(event: ChangePasswordRedSysUiEvent) {
        when (event) {
            is ChangePasswordRedSysUiEvent.UserChanged ->
                _uiState.update { it.copy(user = event.value) }

            is ChangePasswordRedSysUiEvent.PasswordChanged ->
                _uiState.update { it.copy(password = event.value) }

            is ChangePasswordRedSysUiEvent.NewPasswordChanged ->
                _uiState.update { it.copy(newPassword = event.value) }

            is ChangePasswordRedSysUiEvent.RepeatNewPasswordChanged ->
                _uiState.update { it.copy(repeatNewPassword = event.value) }
        }
    }

    fun onButtonsEvent(event: ChangePasswordRedSysButtonsEvent) {
        when (event) {
            ChangePasswordRedSysButtonsEvent.CancelClicked -> {
                viewModelScope.launch { _effects.emit(ChangePasswordRedSysUiEffect.NavigateBack) }
            }

            ChangePasswordRedSysButtonsEvent.AcceptClicked -> clickOnAccept()

            is ChangePasswordRedSysButtonsEvent.SetAcceptEnabled,
            is ChangePasswordRedSysButtonsEvent.SetAcceptLoading -> {
                _uiState.update {
                    it.copy(buttonsState = reduceChangePasswordRedSysButtonsState(it.buttonsState, event))
                }
            }
        }
    }

    private fun clickOnAccept() {
        val state = _uiState.value
        if (state.user.isNotBlank() &&
            state.password.isNotBlank() &&
            state.newPassword.isNotBlank() &&
            state.repeatNewPassword.isNotBlank() &&
            state.newPassword == state.repeatNewPassword
        ) {
            _uiState.update {
                it.copy(
                    buttonsState = it.buttonsState.copy(
                        acceptStyle = CustomButton.StyleButton.LOADING,
                        acceptEnabled = false
                    )
                )
            }
            changePassword(state.user, state.password, state.newPassword)
        }
    }

        val clsChange = RedCLSChangePassData(
            getApplication(),
            user,
            password,
            newPassword

            val result = redSysUseCase.changePassword(clsChange)
            result?.let {

                                buttons = arrayListOf(ButtonType.ACCEPT)

                    _effects.emit(ChangePasswordRedSysUiEffect.ResetAcceptButton)
```

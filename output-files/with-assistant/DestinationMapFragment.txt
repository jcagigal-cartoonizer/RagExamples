                coordinates.isNotEmpty() ->
                    navigatorUseCase.openNavigatorWithCoordinates(coordinates)

                !street.isNullOrEmpty() ->
                    navigatorUseCase.openNavigatorApp(street)

                else ->
                    navigatorUseCase.openNavigatorApp()
            }

            _uiEffect.emit(DestinationMapUiEffect.LaunchIntent(navigatorIntent))
        }
    }

## 4) Full `DestinationMapMainActivityViewModel` Compose bridge

This is the Compose equivalent of the fragment’s `sharedViewModel.multiDispatchFlow.collect { ... }` logic.  
It converts dispatches into `RoutePointModel` exactly like your fragment.

```kotlin
package ifac.td.taxi.ui.screen.destinationmap

import android.app.Application
import androidx.core.content.ContextCompat
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.R
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.ui.model.RoutePointModel
import ifac.td.taxi.viewmodel.model.UtilsModel.Companion.getDispatchNumber
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

class DestinationMapMainActivityViewModel(
    application: Application
) : AndroidViewModel(application) {

    private val TAG = "DestinationMapMainActivityViewModel"

    private val _routePoints = MutableStateFlow<List<RoutePointModel>>(emptyList())
    val routePoints = _routePoints.asStateFlow()

    private val _uiEffect = MutableSharedFlow<DestinationMapUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()

    fun onMultiDispatchChanged(arDispatches: List<Any>?) {
        viewModelScope.launch {
            Logs.d(TAG, "dispatchFlow: Starting to collect from multiDispatchFlow.")
            val dispatchColorMap = mutableMapOf<String, Int>()
            var colorIndex = 0

            Logs.d(TAG, "dispatchFlow: Collected new dispatch list with size: ${arDispatches?.size ?: 0}")

            val routePoints = mutableListOf<RoutePointModel>()

            val arrColors = intArrayOf(
                ContextCompat.getColor(getApplication(), R.color.routes_one),
                ContextCompat.getColor(getApplication(), R.color.grey_divider),
                ContextCompat.getColor(getApplication(), R.color.switch_thumb_on),
                ContextCompat.getColor(getApplication(), R.color.routes_four)
            )

            arDispatches?.forEachIndexed { _, infoDispatchModel ->
                val dispatchNumber = (infoDispatchModel as? HasDispatchFields)?.getDispatchNumber() ?: "Default"
                val color = dispatchColorMap.getOrPut(dispatchNumber) {
                    val assignedColor = arrColors[colorIndex % arrColors.size]
                    Logs.d(TAG, "dispatchFlow: Assigned color $assignedColor to dispatchNumber: $dispatchNumber")
                    colorIndex++
                    assignedColor
                }

                val pickUpAddress = infoDispatchModel.pickUpAdress ?: ""
                val pickUpCoordinates = infoDispatchModel.pickUpCoordinates ?: ""

                if (pickUpAddress.isNotEmpty()) {
                    routePoints.add(
                        RoutePointModel(
                            type = RoutePointModel.RoutePointType.PICK_UP,
                            address = pickUpAddress,
                            coordinatesTag = pickUpCoordinates,
                            order = infoDispatchModel.orderPickUp,
                            dispatchNumber = infoDispatchModel.getDispatchNumber(),
                            titleColor = color
                        )
                    )
                }

                val destinyAddressList = infoDispatchModel.destinyAdress ?: emptyList()
                val destinyCoordinatesList = infoDispatchModel.dropOffCoordinates ?: emptyList()

                if (destinyAddressList.size == destinyCoordinatesList.size) {
                    destinyAddressList.forEachIndexed { addrIndex, address ->
                        val coordinates = destinyCoordinatesList[addrIndex]
                        routePoints.add(
                            RoutePointModel(
                                type = RoutePointModel.RoutePointType.DROP_OFF,
                                address = address,
                                coordinatesTag = coordinates,
                                order = infoDispatchModel.orderDropOff,
                                dispatchNumber = infoDispatchModel.getDispatchNumber(),
                                titleColor = color
                            )
                        )
                    }
                } else {
                    Logs.e(TAG, "dispatchFlow: Mismatch between addressList and coordinatesList sizes for dispatchNumber: $dispatchNumber")
                }
            }

            _routePoints.update { routePoints }
        }
    }

    interface HasDispatchFields {
        val pickUpAdress: String?
        val pickUpCoordinates: String?
        val destinyAdress: List<String>?
        val dropOffCoordinates: List<String>?
        val orderPickUp: Int?
        val orderDropOff: Int?
        fun getDispatchNumber(): String?
    }
}
```

> If your dispatch model already has a concrete type, replace `HasDispatchFields` with that exact type. I used an interface only because the original model type wasn’t included in your prompt.

package ifac.td.taxi.ui.screen.destinationmap

import androidx.compose.foundation.clickable
import androidx.compose.foundation.layout.Column

import androidx.compose.ui.Modifier
import androidx.compose.ui.unit.dp
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import ifac.td.taxi.ui.model.RoutePointModel

@Composable
fun DestinationMapRouteScreen(
    mainActivityViewModel: DestinationMapMainActivityViewModel,
    viewModel: DestinationMapViewModel,
    onLaunchIntent: (android.content.Intent) -> Unit,
    onHeaderVisible: (Boolean) -> Unit
) {
    val routePoints by mainActivityViewModel.routePoints.collectAsStateWithLifecycle()
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()

    LaunchedEffect(Unit) {
        onHeaderVisible(true)
    }

    LaunchedEffect(routePoints) {
        viewModel.onRoutePointsChanged(routePoints)
    }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collect { effect ->
            when (effect) {
                is DestinationMapUiEffect.LaunchIntent -> onLaunchIntent(effect.intent)
            }
        }
    }

    DestinationMapRouteContent(
        state = uiState,
        onRoutePointClick = viewModel::onRoutePointClicked
    )
}

@Composable
private fun DestinationMapRouteContent(
    state: DestinationMapUiState,
    onRoutePointClick: (RoutePointModel) -> Unit
) {
    Column(modifier = Modifier.fillMaxSize()) {
        Text(
            text = "PICKUP NAVIGATION TITLE",
            modifier = Modifier
                .fillMaxWidth()
                .padding(10.dp)
        )

        Text(
            text = state.routePoints.firstOrNull { it.type == RoutePointModel.RoutePointType.PICK_UP }?.address
                ?: "Pickup address example",
            modifier = Modifier
                .fillMaxWidth()
                .padding(horizontal = 16.dp, vertical = 16.dp)
        )

        Text(
            text = "DROP OFF NAVIGATION TITLE",
            modifier = Modifier
                .fillMaxWidth()
                .padding(10.dp)
        )

        LazyColumn(modifier = Modifier.fillMaxSize()) {
            items(state.routePoints.filter { it.type == RoutePointModel.RoutePointType.DROP_OFF }) { routePoint ->
                Text(
                    text = routePoint.address,
                    modifier = Modifier
                        .fillMaxWidth()
                        .clickable { onRoutePointClick(routePoint) }
                        .padding(16.dp)
                )
            }
        }
    }
}

                !coordinates.isNullOrEmpty() && !street.isNullOrEmpty() -> {

                    navigatorUseCase.openNavigatorWithCoordinatesAndStreetName(coordinates, street)

                }
                !street.isNullOrEmpty() -> {
                    Logs.d(TAG, "openNavigatorApp: Opening navigator with street: $street")
                    navigatorUseCase.openNavigatorApp(street)
                }
                else -> {
                    Logs.d(TAG, "openNavigatorApp: Opening navigator without specific parameters")
                    navigatorUseCase.openNavigatorApp()
                }
            }

            _navigatorFlow.emit(navigatorIntent)
            if (navigatorIntent != null) _uiEffect.emit(DestinationMapUiEffect.LaunchIntent(navigatorIntent))
        }
    }

@Composable
fun DestinationMapScreen(
    sharedViewModel: ComposeSharedViewModel,
    viewModel: DestinationMapViewModel,
    onLaunchIntent: (Intent) -> Unit,
    onShowHeader: (Boolean) -> Unit
) {
    val dispatch by sharedViewModel.dispatchFlow.collectAsStateWithLifecycle()
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()

    LaunchedEffect(Unit) {
        onShowHeader(true)
    }

    LaunchedEffect(dispatch) {
        val routePoints = buildRoutePointsFromDispatch(dispatch)
        viewModel.setRoutePoints(routePoints)
    }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collect { effect ->
            when (effect) {
                is DestinationMapUiEffect.LaunchIntent -> onLaunchIntent(effect.intent)
            }
        }
    }

    DestinationMapContent(
        state = uiState,
        onRoutePointClick = viewModel::openNavigatorApp
    )
}

private fun buildRoutePointsFromDispatch(dispatch: InfoDispatchModel?): List<RoutePointModel> {
    if (dispatch == null) return emptyList()

    val TAG = "DestinationMapScreen"
    Logs.d(TAG, "dispatchFlow: Starting to collect from multiDispatchFlow.")
    val dispatchColorMap = mutableMapOf<String, Int>()
    var colorIndex = 0

    val routePoints = mutableListOf<RoutePointModel>()

    val arrColors = intArrayOf(
        Color(0xFF2E7D32).toArgb(),
        Color(0xFFBDBDBD).toArgb(),
        Color(0xFF4CAF50).toArgb(),
        Color(0xFF1565C0).toArgb()
    )

    val dispatchNumber = dispatch.getDispatchNumber() ?: "Default"
    val color = dispatchColorMap.getOrPut(dispatchNumber) {
        val assignedColor = arrColors[colorIndex % arrColors.size]
        colorIndex++
        assignedColor
    }

    val pickUpAddress = dispatch.pickUpAdress ?: ""
    val pickUpCoordinates = dispatch.pickUpCoordinates ?: ""

    if (pickUpAddress.isNotEmpty()) {
        routePoints.add(
            RoutePointModel(
                type = RoutePointModel.RoutePointType.PICK_UP,
                address = pickUpAddress,
                coordinatesTag = pickUpCoordinates,
                order = dispatch.orderPickUp,
                dispatchNumber = dispatch.getDispatchNumber(),
                titleColor = color
            )
        )
    }

    val destinyAddressList = dispatch.destinyAdress ?: emptyList()
    val destinyCoordinatesList = dispatch.dropOffCoordinates ?: emptyList()

    if (destinyAddressList.size == destinyCoordinatesList.size) {
        destinyAddressList.forEachIndexed { addrIndex, address ->
            val coordinates = destinyCoordinatesList[addrIndex]
            routePoints.add(
                RoutePointModel(
                    type = RoutePointModel.RoutePointType.DROP_OFF,
                    address = address,
                    coordinatesTag = coordinates,
                    order = dispatch.orderDropOff,
                    dispatchNumber = dispatch.getDispatchNumber(),
                    titleColor = color
                )
            )
        }
    } else {
        Logs.e(TAG, "dispatchFlow: Mismatch between addressList and coordinatesList sizes for dispatchNumber: $dispatchNumber")
    }

    return routePoints
}

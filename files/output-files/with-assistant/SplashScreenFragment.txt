```kotlin
import android.net.Uri
import android.os.Build

import android.widget.VideoView

import androidx.compose.runtime.Composable

import androidx.compose.runtime.LaunchedEffect
import androidx.compose.runtime.getValue
import androidx.compose.runtime.remember
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color as ComposeColor
import androidx.compose.ui.platform.LocalContext
import androidx.compose.ui.viewinterop.AndroidView
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import ifac.td.taxi.R
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.viewmodel.SplashScreenMainActivityViewModel
import ifac.td.taxi.viewmodel.SplashScreenUiEffect
import ifac.td.taxi.viewmodel.SplashScreenUiEvent
import kotlinx.coroutines.flow.collectLatest

@Composable
fun SplashScreenRoute(
    viewModel: SplashScreenMainActivityViewModel,
    onNavigateToWelcome: () -> Unit,
    onNavigateToLegalText: () -> Unit,
    onShowHeader: (Boolean) -> Unit = {},
    onShowBottomBar: (Boolean) -> Unit = {},
    onBackPressedDisabled: () -> Unit = {},
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    LaunchedEffect(Unit) {
        onShowHeader(false)
        onShowBottomBar(false)
        onBackPressedDisabled()
        viewModel.onEvent(
            SplashScreenUiEvent.CheckReconnecting(
                reconnectionFlowValue = uiState.reconnectionFlowValue
            )
        )
    }

    LaunchedEffect(Unit) {
        viewModel.effects.collectLatest { effect ->
            when (effect) {
                SplashScreenUiEffect.NavigateToWelcome -> onNavigateToWelcome()
                SplashScreenUiEffect.NavigateToLegalText -> onNavigateToLegalText()
                SplashScreenUiEffect.EndSplashWithoutCheck -> {
                    // handled by the screen itself if needed; kept as effect for parity
                }
            }
        }
    }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(ComposeColor(android.graphics.Color.parseColor("#000000")))
    ) {
        SplashVideoPlayer(
            rawResId = R.raw.new_splash,
            splashTimeout = 4800L,
            onVideoEnded = {
                viewModel.onEvent(SplashScreenUiEvent.VideoCompleted)
            },
            onVideoError = { what, extra ->
                Logs.e("SplashScreen", "Error during video playback: what=$what, extra=$extra")
                viewModel.onEvent(SplashScreenUiEvent.VideoError(what, extra))
            },
            onPrepared = {
                Logs.d("SplashScreen", "setUpVideo -> Video prepared, starting")
            }
        )
    }
}

@Composable
private fun SplashVideoPlayer(
    @RawRes rawResId: Int,
    splashTimeout: Long,
    onPrepared: () -> Unit,
    onVideoEnded: () -> Unit,
    onVideoError: (what: Int, extra: Int) -> Unit,
) {
    val context = LocalContext.current
    val handler = remember { Handler(Looper.getMainLooper()) }

    DisposableEffect(Unit) {
        onDispose {
            handler.removeCallbacksAndMessages(null)
        }
    }

    AndroidView(
        factory = {
            VideoView(it).apply {
                setZOrderOnTop(true)
                setBackgroundColor(android.graphics.Color.TRANSPARENT)

                setOnPreparedListener { mediaPlayer ->
                    setBackgroundColor(Color.TRANSPARENT)
                    mediaPlayer.isLooping = false
                    mediaPlayer.start()
                    onPrepared()
                }

                setOnCompletionListener {
                    onVideoEnded()
                }

                setOnErrorListener { _, what, extra ->
                    onVideoError(what, extra)
                    true
                }

                val videoUri = Uri.parse("android.resource://${context.packageName}/$rawResId")
                try {
                    setVideoURI(videoUri)
                    handler.postDelayed({
                        onVideoEnded()
                    }, splashTimeout)
                } catch (_: Exception) {
                    onVideoEnded()
                }
            }
        },
        update = { videoView ->
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.LOLLIPOP) {
                videoView.setBackgroundColor(android.graphics.Color.TRANSPARENT)
            }
        },
        modifier = Modifier.fillMaxSize()
    )
}
```

```kotlin
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch

data class SplashScreenUiState(
    val isLoading: Boolean = false,
    val reconnectionFlowValue: Boolean = false,
    val shouldSkipSplash: Boolean = false,
    val isSkipped: Boolean = false,
    val showVideo: Boolean = true,
    val dialogState: SplashScreenDialogState = SplashScreenDialogState.Hidden,
    val buttonsState: SplashScreenButtonsState = SplashScreenButtonsState.initial(),
)

sealed interface SplashScreenUiEvent {
    data class CheckReconnecting(val reconnectionFlowValue: Boolean) : SplashScreenUiEvent
    data object VideoCompleted : SplashScreenUiEvent
    data class VideoError(val what: Int, val extra: Int) : SplashScreenUiEvent
    data object TimeoutReached : SplashScreenUiEvent
    data object EndSplash : SplashScreenUiEvent
    data object EndSplashWithoutCheck : SplashScreenUiEvent
    data class UpdateReconnectionValue(val value: Boolean) : SplashScreenUiEvent
}

sealed interface SplashScreenUiEffect {
    data object NavigateToWelcome : SplashScreenUiEffect
    data object NavigateToLegalText : SplashScreenUiEffect
    data object EndSplashWithoutCheck : SplashScreenUiEffect
}

sealed interface SplashScreenDialogState {
    data object Hidden : SplashScreenDialogState
}

data class SplashScreenButtonVisualState(
    val text: String = "",
    val iconRes: Int? = null,
    val background: BackgroundButtonColor = BackgroundButtonColor.BLUE,
    val style: StyleButton = StyleButton.ENABLE,
    val textVisible: Boolean = true,
    val iconVisible: Boolean = true,
    val onlyTextVisible: Boolean = false,
    val onlyTextValue: String = "",
    val textColorRes: Int = android.R.color.white,
    val iconTintRes: Int = android.R.color.white,
)

data class SplashScreenButtonsState(
    val accept: SplashScreenButtonVisualState = SplashScreenButtonVisualState(),
    val cancel: SplashScreenButtonVisualState = SplashScreenButtonVisualState(),
    val hidden: SplashScreenButtonVisualState = SplashScreenButtonVisualState(
        text = "",
        iconVisible = false,
        textVisible = false,
        background = BackgroundButtonColor.TRANSPARENT
    ),
)

class SplashScreenMainActivityViewModel(
    context: Application,
    private val sessionUseCase: SessionUseCase,
) : BaseViewModel(context) {

    private val TAG = "SplashScreenMainActivityViewModel"

    private val _uiState = MutableStateFlow(SplashScreenUiState())
    val uiState: StateFlow<SplashScreenUiState> = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<SplashScreenUiEffect>()
    val effects: SharedFlow<SplashScreenUiEffect> = _effects.asSharedFlow()

    fun onEvent(event: SplashScreenUiEvent) {
        when (event) {
            is SplashScreenUiEvent.CheckReconnecting -> checkIsReconnecting(event.reconnectionFlowValue)
            SplashScreenUiEvent.VideoCompleted -> endSplash()
            is SplashScreenUiEvent.VideoError -> endSplash()
            SplashScreenUiEvent.TimeoutReached -> endSplash()
            SplashScreenUiEvent.EndSplash -> endSplash()
            SplashScreenUiEvent.EndSplashWithoutCheck -> endSplashWithoutCheck()
            is SplashScreenUiEvent.UpdateReconnectionValue -> {
                _uiState.value = _uiState.value.copy(reconnectionFlowValue = event.value)
            }
        }
    }

    private fun checkIsReconnecting(reconnectionFlowValue: Boolean) {
        viewModelScope.launch(Dispatchers.IO) {
            val serviceWasActive = sessionUseCase.getServiceWasActive()
            Logs.d(
                TAG,
                "checkIsReconnecting serviceWasActive: $serviceWasActive, reconnectionFlowValue: $reconnectionFlowValue"
            )
            delay(50)
            _uiState.value = _uiState.value.copy(shouldSkipSplash = serviceWasActive)
            if (serviceWasActive) {
                _effects.emit(SplashScreenUiEffect.EndSplashWithoutCheck)
            }
        }
    }

    private fun endSplash() {
        _uiState.value = _uiState.value.copy(showVideo = false)
        // fragment behavior: sharedViewModel.showLegalTextWindow(false)
        // in Compose bridge this is handled by the host/shared VM
    }

    private fun endSplashWithoutCheck() {
        _uiState.value = _uiState.value.copy(
            showVideo = false,
            isSkipped = true
        )
        viewModelScope.launch {
            _effects.emit(SplashScreenUiEffect.NavigateToWelcome)
        }
    }
}
```

## 4) Compose bridge for the shared `MainActivityViewModel`

This is the Compose equivalent of the fragment’s `sharedViewModel.showLegalTextFlow.collect { ... }` and `reconnectionFlow.value` usage.

```kotlin
package ifac.td.taxi.viewmodel

import androidx.lifecycle.ViewModel
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow

class SplashScreenMainActivityBridgeViewModel : ViewModel() {

    private val _reconnectionFlow = MutableStateFlow(false)
    val reconnectionFlow: StateFlow<Boolean> = _reconnectionFlow.asStateFlow()

    private val _showLegalTextFlow = MutableStateFlow<Boolean?>(null)
    val showLegalTextFlow: StateFlow<Boolean?> = _showLegalTextFlow.asStateFlow()

    fun setReconnectionValue(value: Boolean) {
        _reconnectionFlow.value = value
    }

    fun showLegalTextWindow(showLegal: Boolean) {
        _showLegalTextFlow.value = showLegal
    }

    fun consumeShowLegalText() {
        _showLegalTextFlow.value = null
    }
}
```

## 5) How to preserve navigation exactly like the fragment

In your host Compose screen or activity, collect `SplashScreenUiEffect` and route them to the same destinations:

```kotlin
LaunchedEffect(Unit) {
    viewModel.effects.collectLatest { effect ->
        when (effect) {
            SplashScreenUiEffect.NavigateToWelcome -> {
                navController.navigate(R.id.action_splashScreenFragment_to_welcomeFragment)
            }
            SplashScreenUiEffect.NavigateToLegalText -> {
                navController.navigate(R.id.action_splashScreenFragment_to_legalTextFragment)
            }
            SplashScreenUiEffect.EndSplashWithoutCheck -> {
                navController.navigate(R.id.action_splashScreenFragment_to_welcomeFragment)
            }
        }
    }
}
```

And if you want the exact fragment behavior:
- when splash ends normally: call `showLegalTextWindow(false)`
- when splash is skipped: call `showLegalTextWindow(true)` then navigate to welcome
- when `showLegalTextFlow` emits `true`, navigate to legal text
- when `showLegalTextFlow` emits `false`, navigate to welcome

```kotlin
class SplashScreenViewModel(
    context: Application,
    private val sessionUseCase: SessionUseCase
) : BaseViewModel(context) {

    private val TAG = "SplashScreenViewModel"

    private val _serviceActiveFlow = MutableSharedFlow<Boolean>()
    val serviceActiveFlow: SharedFlow<Boolean> = _serviceActiveFlow.asSharedFlow()

    fun checkIsReconnecting(reconnectionFlowValue: Boolean) {
        viewModelScope.launch(Dispatchers.IO) {
            val serviceWasActive = sessionUseCase.getServiceWasActive()
            Logs.d(
                TAG,
                "checkIsReconnecting serviceWasActive: $serviceWasActive, reconnectionFlowValue: $reconnectionFlowValue"
            )
            delay(50)
            _serviceActiveFlow.emit(serviceWasActive)
        }
    }
}

class SplashScreenMainActivityViewModel : ViewModel() {
    private val _reconnectionFlow = MutableStateFlow(false)
    val reconnectionFlow: StateFlow<Boolean> = _reconnectionFlow.asStateFlow()

    private val _showLegalTextFlow = MutableStateFlow<Boolean?>(null)
    val showLegalTextFlow: StateFlow<Boolean?> = _showLegalTextFlow.asStateFlow()

    fun setReconnectionValue(value: Boolean) {
        _reconnectionFlow.value = value
    }

    fun showLegalTextWindow(showLegal: Boolean) {
        _showLegalTextFlow.value = showLegal
    }

    fun consumeShowLegalText() {
        _showLegalTextFlow.value = null
    }
}

@Composable
fun SplashScreenRoute(
    viewModel: SplashScreenViewModel,
    sharedViewModel: SplashScreenMainActivityViewModel,
    onNavigateToWelcome: () -> Unit,
    onNavigateToLegalText: () -> Unit,
    onShowHeader: (Boolean) -> Unit = {},
    onShowBottomBar: (Boolean) -> Unit = {},
    onDisableBack: () -> Unit = {},
) {
    val uiState by remember { viewModel.uiStateHolder() }.collectAsStateWithLifecycle()

    LaunchedEffect(Unit) {
        onShowHeader(false)
        onShowBottomBar(false)
        onDisableBack()
        viewModel.checkIsReconnecting(sharedViewModel.reconnectionFlow.value)
    }

    LaunchedEffect(Unit) {
        viewModel.serviceActiveFlow.collect { shouldSkipSplash ->
            Logs.d("SplashScreenFragment", "checkIsReconnecting shouldSkipSplash: $shouldSkipSplash")
            if (shouldSkipSplash) {
                endSplashWithoutCheck(
                    sharedViewModel = sharedViewModel,
                    onNavigateToWelcome = onNavigateToWelcome
                )
            }
        }
    }

    LaunchedEffect(Unit) {
        sharedViewModel.showLegalTextFlow.collect { showLegal ->
            showLegal?.let {
                if (it) onNavigateToLegalText() else onNavigateToWelcome()
                sharedViewModel.consumeShowLegalText()
            }
        }
    }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(ComposeColor(android.graphics.Color.parseColor("#000000")))
    ) {
        SplashVideoPlayer(
            rawResId = R.raw.new_splash,
            splashTimeout = 4800L,
            onPrepared = {
                Logs.d("SplashScreenFragment", "setUpVideo -> Video prepared, starting")
            },
            onVideoEnded = {
                Logs.d("SplashScreenFragment", "setUpVideo -> Video ended")
                endSplash(sharedViewModel)
            },
            onVideoError = { what, extra ->
                Logs.e("SplashScreenFragment", "Error during video playback: what=$what, extra=$extra")
                endSplash(sharedViewModel)
            }
        )
    }
}

@Composable
private fun SplashVideoPlayer(
    @RawRes rawResId: Int,
    splashTimeout: Long,
    onPrepared: () -> Unit,
    onVideoEnded: () -> Unit,
    onVideoError: (what: Int, extra: Int) -> Unit,
) {
    val context = LocalContext.current
    val handler = remember { Handler(Looper.getMainLooper()) }
    var ended by remember { mutableStateOf(false) }

    DisposableEffect(Unit) {
        onDispose {
            handler.removeCallbacksAndMessages(null)
        }
    }

    AndroidView(
        factory = {
            VideoView(it).apply {
                setZOrderOnTop(true)
                setBackgroundColor(it.resources.getColor(R.color.splash_background, null))

                setOnPreparedListener { mediaPlayer ->
                    setBackgroundColor(Color.TRANSPARENT)
                    mediaPlayer.isLooping = false
                    mediaPlayer.start()
                    onPrepared()
                }

                setOnCompletionListener {
                    if (!ended) {
                        ended = true
                        onVideoEnded()
                    }
                }

                setOnErrorListener { _, what, extra ->
                    if (!ended) {
                        ended = true
                        onVideoError(what, extra)
                    }
                    true
                }

                try {
                    val videoUri = Uri.parse("android.resource://${context.packageName}/$rawResId")
                    setVideoURI(videoUri)
                    handler.postDelayed({
                        if (!ended) {
                            ended = true
                            onVideoEnded()
                        }
                    }, splashTimeout)
                } catch (e: Exception) {
                    if (!ended) {
                        ended = true
                        onVideoEnded()
                    }
                }
            }
        },
        modifier = Modifier.fillMaxSize()
    )
}

private fun endSplash(sharedViewModel: SplashScreenMainActivityViewModel) {
    sharedViewModel.showLegalTextWindow(false)

private fun endSplashWithoutCheck(
    sharedViewModel: SplashScreenMainActivityViewModel,
    onNavigateToWelcome: () -> Unit

    sharedViewModel.showLegalTextWindow(true)
    onNavigateToWelcome()
}
```

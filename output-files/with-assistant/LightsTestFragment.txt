```kotlin
@Composable
fun LightsTestRoute(
    viewModel: LightsTestComposeViewModel,
    sharedViewModel: LightsTestMainActivityViewModel,
    onBack: () -> Unit,
    showHeader: (Boolean) -> Unit,
) {
    val lifecycleOwner = LocalLifecycleOwner.current

    LaunchedEffect(Unit) {
        showHeader(true)
        viewModel.getBeeperVolume()
    }

    BackHandler(onBack = onBack)

    // lifecycle collection for effects
    LaunchedEffect(viewModel, sharedViewModel, lifecycleOwner) {
        lifecycleOwner.repeatOnLifecycle(Lifecycle.State.STARTED) {
            launch {
                viewModel.effects.collectLatest { effect ->
                    when (effect) {
                        LightsTestUiEffect.NavigateBack -> onBack()
                    }
                }
            }
        }
    }

    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val buttonsState by viewModel.buttonsState.collectAsStateWithLifecycle()
    val courtesyState by sharedViewModel.courtesyLightState.collectAsStateWithLifecycle()

    // Mirror the fragment's collect branches one-by-one
    LaunchedEffect(courtesyState) {
        viewModel.reduceCourtesyLight(courtesyState.isOn)
    }

    LaunchedEffect(uiState.beeperVolume) {
        // no-op, state already drives UI
    }

    LightsTestScreen(
        uiState = uiState,
        buttonsState = buttonsState,
        onUvClick = { viewModel.onEvent(LightsTestUiEvent.UvLightClicked) },
        onCourtesyClick = { viewModel.onEvent(LightsTestUiEvent.CourtesyLightClicked) },
        onBeeperChanged = { viewModel.onEvent(LightsTestUiEvent.BeeperVolumeChanged(it)) },
    )
}
```

## 2) Compose-friendly `LightsTestViewModel` with `UiState + UiEvent + UiEffect`

This mirrors the original logic, but exposes state in a Compose-friendly way.

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.domain.model.UserPreferences
import ifac.td.taxi.domain.usecase.LightSkyGlassUseCase
import ifac.td.taxi.domain.usecase.UserPreferencesUseCase
import ifac.td.taxi.ui.custom.button.CustomButton
import kotlinx.coroutines.delay
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch

data class LightsTestUiState(
    val beeperVolume: Int? = null,
    val uvLightActive: Boolean = false,
    val courtesyLightOn: Boolean = false,
    val buttonsState: LightsTestButtonsState = LightsTestButtonsState.initial()
)

sealed interface LightsTestUiEvent {
    data object UvLightClicked : LightsTestUiEvent
    data object CourtesyLightClicked : LightsTestUiEvent
    data class BeeperVolumeChanged(val value: Int) : LightsTestUiEvent
    data object ScreenStarted : LightsTestUiEvent
}

sealed interface LightsTestUiEffect {
    data object NavigateBack : LightsTestUiEffect
}

class LightsTestComposeViewModel(
    private val lightSkyGlassUseCase: LightSkyGlassUseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    application: Application,
) : AndroidViewModel(application) {

    private val _uiState = MutableStateFlow(LightsTestUiState())
    val uiState: StateFlow<LightsTestUiState> = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<LightsTestUiEffect>()
    val effects: SharedFlow<LightsTestUiEffect> = _effects.asSharedFlow()

    private val _buttonsState = MutableStateFlow(LightsTestButtonsState.initial())
    val buttonsState: StateFlow<LightsTestButtonsState> = _buttonsState.asStateFlow()

    fun onEvent(event: LightsTestUiEvent) {
        when (event) {
            LightsTestUiEvent.ScreenStarted -> getBeeperVolume()

            LightsTestUiEvent.UvLightClicked -> turnUVLight()

            LightsTestUiEvent.CourtesyLightClicked -> turnCourtesyLight()

            is LightsTestUiEvent.BeeperVolumeChanged -> setBeeperVolume(event.value)
        }
    }

    fun turnUVLight() {
        viewModelScope.launch {
            lightSkyGlassUseCase.turnUVLight()
            _buttonsState.value = _buttonsState.value.copy(
                uv = _buttonsState.value.uv.copy(
                    iconRes = if (_buttonsState.value.uv.iconRes == R.drawable.luz_ultravioleta_blue)
                        R.drawable.luz_ultravioleta
                    else
                        R.drawable.luz_ultravioleta_blue,
                    enabled = false
                ),
                courtesy = _buttonsState.value.courtesy.copy(enabled = false)
            )
            delay(8000L)
            reduceUvLight(false)
        }
    }

    fun turnCourtesyLight() {
        viewModelScope.launch {
            lightSkyGlassUseCase.turnCourtesyLight()
        }
    }

    fun setBeeperVolume(volume: Int?) {
        viewModelScope.launch {
            lightSkyGlassUseCase.setBeeperVolume(volume ?: 50)
            volume?.let { userPreferencesUseCase.insertBeeperVolumeValue(it) }
            _uiState.update { it.copy(beeperVolume = volume ?: 50) }
        }
    }

    fun getBeeperVolume() {
        viewModelScope.launch {
            val preferences = userPreferencesUseCase.getUserPreferences()
            if (preferences != null) {
                _uiState.update { it.copy(beeperVolume = preferences.beeperVolume) }
            } else {
                userPreferencesUseCase.insertUserPreferences(UserPreferences())
                _uiState.update { it.copy(beeperVolume = 50) }
            }
        }
    }

    fun reduceUvLight(isOn: Boolean) {
        _uiState.update { it.copy(uvLightActive = isOn) }
        _buttonsState.update { current ->
            LightsTestButtonsState.reduce(
                current = current,
                effect = LightsTestUiEffectInternal.UvLightCollected(isOn)
            )
        }
    }

    fun reduceCourtesyLight(isOn: Boolean?) {
        _uiState.update { it.copy(courtesyLightOn = isOn == true) }
        _buttonsState.update { current ->
            LightsTestButtonsState.reduce(
                current = current,
                effect = LightsTestUiEffectInternal.CourtesyLightCollected(isOn)
            )
        }
    }

    fun reduceBeeperVolume(volume: Int?) {
        _uiState.update { it.copy(beeperVolume = volume ?: 50) }
        _buttonsState.update { current ->
            LightsTestButtonsState.reduce(
                current = current,
                effect = LightsTestUiEffectInternal.BeeperVolumeCollected(volume)
            )
        }
    }

    private sealed interface LightsTestUiEffectInternal {
        data class UvLightCollected(val lights: Boolean) : LightsTestUiEffectInternal
        data class CourtesyLightCollected(val isOn: Boolean?) : LightsTestUiEffectInternal
        data class BeeperVolumeCollected(val value: Int?) : LightsTestUiEffectInternal
    }
}
```

## 3) Exact `LightsTestButtonsState` reducer

This is the reducer that mirrors the fragment’s `collect {}` branches one-by-one.

```kotlin
package ifac.td.taxi.viewmodel

import ifac.td.taxi.R
import ifac.td.taxi.ui.custom.button.CustomButton

data class LightsTestButtonState(
    val iconRes: Int,
    val enabled: Boolean,
    val background: CustomButton.BackgroundButtonColor,
)

data class LightsTestButtonsState(
    val uv: LightsTestButtonState,
    val courtesy: LightsTestButtonState,
) {
    companion object {
        fun initial() = LightsTestButtonsState(
            uv = LightsTestButtonState(
                iconRes = R.drawable.luz_ultravioleta,
                enabled = true,
                background = CustomButton.BackgroundButtonColor.TRANSPARENT
            ),
            courtesy = LightsTestButtonState(
                iconRes = R.drawable.lumact,
                enabled = true,
                background = CustomButton.BackgroundButtonColor.TRANSPARENT
            )
        )

        /**
         * Exact reducer mirroring the fragment collect branches one-by-one.
         */
        fun reduce(
            current: LightsTestButtonsState,
            effect: Any
        ): LightsTestButtonsState {
            return when (effect) {
                is LightsTestComposeViewModel.LightsTestUiEffectInternal.UvLightCollected -> {
                    if (!effect.lights) {
                        current.copy(
                            uv = current.uv.copy(
                                iconRes = R.drawable.luz_ultravioleta,
                                enabled = true
                            ),
                            courtesy = current.courtesy.copy(
                                enabled = true
                            )
                        )
                    } else current
                }

                is LightsTestComposeViewModel.LightsTestUiEffectInternal.CourtesyLightCollected -> {
                    if (effect.isOn == true) {
                        current.copy(
                            courtesy = current.courtesy.copy(
                                iconRes = R.drawable.lumact_yellow,
                                enabled = current.courtesy.enabled,
                            ),
                            uv = current.uv.copy(enabled = false)
                        )
                    } else {
                        current.copy(
                            courtesy = current.courtesy.copy(
                                iconRes = R.drawable.lumact,
                                enabled = current.courtesy.enabled,
                            ),
                            uv = current.uv.copy(enabled = true)
                        )
                    }
                }

                is LightsTestComposeViewModel.LightsTestUiEffectInternal.BeeperVolumeCollected -> {
                    current
                }

                else -> current
            }
        }
    }
}
```

### Notes on exact branch matching
This reducer mirrors the fragment logic exactly:

- `uvLightFlow.collect { lights -> if (!it) { ... } }`
  - when `false`, UV icon returns to `luz_ultravioleta`
  - UV enabled = `true`
  - Courtesy enabled = `true`

- `courtesyLightFlow.collect { isOn -> if (isOn == true) { ... } else { ... } }`
  - when `true`, courtesy icon = `lumact_yellow`
  - UV enabled = `false`
  - when `false`, courtesy icon = `lumact`
  - UV enabled = `true`

- `beeperLightFlow.collect { beeperValue -> sbBIPVolume.progress = beeperValue ?: 50 }`
  - handled by `uiState.beeperVolume`

```kotlin
                        it.copy(
                            uvLightEnabled = true,
                            courtesyLightEnabled = true,
                            uvIconRes = R.drawable.luz_ultravioleta
                        )
                    }
                }
            }

            is LightsTestButtonsEffect.CourtesyLightCollected -> {
                if (effect.isOn == true) {
                    _uiState.update {
                        it.copy(
                            courtesyLightEnabled = true,
                            uvLightEnabled = false,
                            courtesyIconRes = R.drawable.lumact_yellow
                        )
                    }
                } else {
                    _uiState.update {
                        it.copy(
                            courtesyLightEnabled = true,
                            uvLightEnabled = true,
                            courtesyIconRes = R.drawable.lumact
                        )
                    }
                }
            }

            is LightsTestButtonsEffect.BeeperVolumeCollected -> {
                _uiState.update { it.copy(beeperVolume = effect.value ?: 50) }
            }
        }
    }
}

@Composable
fun LightsTestRoute(
    viewModel: LightsTestComposeViewModel,
    sharedViewModel: LightsTestMainActivityViewModel,
    onBack: () -> Unit,
    showHeader: (Boolean) -> Unit,
) {
    val lifecycleOwner = LocalLifecycleOwner.current
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val buttonsState by viewModel.buttonsState.collectAsStateWithLifecycle()
    val courtesyLightOn by sharedViewModel.courtesyLightFlow.collectAsStateWithLifecycle()

    LaunchedEffect(Unit) {
        showHeader(true)
        viewModel.onEvent(LightsTestUiEvent.ScreenStarted)
    }

    BackHandler(onBack = onBack)

    // lifecycle collection of effects
    LaunchedEffect(viewModel, lifecycleOwner) {
        lifecycleOwner.repeatOnLifecycle(Lifecycle.State.STARTED) {
            launch {
                viewModel.effects.collectLatest { effect ->
                    when (effect) {
                        LightsTestUiEffect.NavigateBack -> onBack()
                    }
                }
            }
        }
    }

    // Mirror fragment's sharedViewModel.courtesyLightFlow.collect { ... }
    LaunchedEffect(courtesyLightOn) {
        Logs.d("LightsTestRoute", "Courtesy light state changed: $courtesyLightOn")
        viewModel.reduce(LightsTestButtonsEffect.CourtesyLightCollected(courtesyLightOn))
    }

    LightsTestScreen(
        uiState = uiState,
        buttonsState = buttonsState,
        onUvClick = { viewModel.onEvent(LightsTestUiEvent.UvLightClicked) },
        onCourtesyClick = { viewModel.onEvent(LightsTestUiEvent.CourtesyLightClicked) },
        onBeeperChanged = { viewModel.onEvent(LightsTestUiEvent.BeeperVolumeChanged(it)) }
    )
}

@Composable
private fun LightsTestScreen(
    uiState: LightsTestUiState,
    buttonsState: LightsTestButtonsState,
    onUvClick: () -> Unit,
    onCourtesyClick: () -> Unit,
    onBeeperChanged: (Int) -> Unit,
) {
    Column(
        modifier = Modifier.fillMaxSize(),
        horizontalAlignment = Alignment.CenterHorizontally,
        verticalArrangement = Arrangement.Center
    ) {
        Row(
            horizontalArrangement = Arrangement.Center,
            verticalAlignment = Alignment.CenterVertically
        ) {
            Column(horizontalAlignment = Alignment.CenterHorizontally) {
                Text(text = "UV LIGHT")
                LightsTestCustomButton(
                    iconRes = buttonsState.uv.iconRes,
                    enabled = buttonsState.uv.enabled,
                    background = buttonsState.uv.background,
                    onClick = onUvClick
                )
            }

            Spacer(modifier = Modifier.width(24.dp))

            Column(horizontalAlignment = Alignment.CenterHorizontally) {
                Text(text = "COURTESY LIGHT")
                LightsTestCustomButton(
                    iconRes = buttonsState.courtesy.iconRes,
                    enabled = buttonsState.courtesy.enabled,
                    background = buttonsState.courtesy.background,
                    onClick = onCourtesyClick
                )
            }
        }

        Spacer(modifier = Modifier.height(24.dp))

        Text(text = "BIP VOLUME")

        Spacer(modifier = Modifier.height(8.dp))

            Image(
                painter = painterResource(id = R.drawable.baseline_volume_up_24),
                contentDescription = null,
                modifier = Modifier.size(24.dp)
            )

            Spacer(modifier = Modifier.width(8.dp))

            AndroidView(
                factory = { context ->
                    SeekBar(context).apply {
                        max = 100
                        progress = uiState.beeperVolume
                        setOnSeekBarChangeListener(object : SeekBar.OnSeekBarChangeListener {
                            override fun onProgressChanged(
                                seekBar: SeekBar,
                                progress: Int,
                                fromUser: Boolean
                            ) {
                                if (fromUser) onBeeperChanged(progress)
                            }

                            override fun onStartTrackingTouch(seekBar: SeekBar) {}
                            override fun onStopTrackingTouch(seekBar: SeekBar) {}
                        })
                    }
                },
                update = { seekBar ->
                    seekBar.max = 100
                    if (seekBar.progress != uiState.beeperVolume) {
                        seekBar.progress = uiState.beeperVolume
                    }
                }
            )
        }
    }
}

@Composable
private fun LightsTestCustomButton(
    iconRes: Int,
    enabled: Boolean,
    background: BackgroundButtonColor,
    onClick: () -> Unit,
) {
    AndroidView(
        factory = { context ->

                setButtonStyle(if (enabled) StyleButton.ENABLE else StyleButton.DISABLE)
                changeBackground(background)
                changeButtonIcon(iconRes)
                setAction(onClick)
            }
        },
        update = { button ->
            button.setButtonStyle(if (enabled) StyleButton.ENABLE else StyleButton.DISABLE)
            button.changeBackground(background)
            button.changeButtonIcon(iconRes)
            button.setAction(onClick)
        }
    )
}
```

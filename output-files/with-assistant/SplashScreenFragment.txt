## 3) Compose-friendly ViewModel

This version exposes:

- `StateFlow<SplashScreenUiState>`
- `SharedFlow<SplashScreenUiEffect>`

It preserves the original behavior:
- checks reconnection state
- emits a skip decision
- lets the UI decide navigation based on shared state from `MainActivityViewModel`

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.domain.usecase.SessionUseCase
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.ui.screen.SplashScreenUiEffect
import ifac.td.taxi.ui.screen.SplashScreenUiEvent
import ifac.td.taxi.ui.screen.SplashScreenUiState
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.delay
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch

class SplashScreenComposeViewModel(
    application: Application,
    private val sessionUseCase: SessionUseCase
) : AndroidViewModel(application) {

    private val TAG = "SplashScreenComposeVM"

    private val _uiState = MutableStateFlow(SplashScreenUiState())
    val uiState: StateFlow<SplashScreenUiState> = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<SplashScreenUiEffect>()
    val uiEffect: SharedFlow<SplashScreenUiEffect> = _uiEffect.asSharedFlow()

    fun onEvent(event: SplashScreenUiEvent) {
        when (event) {
            SplashScreenUiEvent.OnScreenStarted -> checkIsReconnecting()
            SplashScreenUiEvent.OnVideoCompleted -> handleVideoFinished()
            SplashScreenUiEvent.OnVideoError -> handleVideoFinished()
            SplashScreenUiEvent.OnTimeoutReached -> handleVideoFinished()
            SplashScreenUiEvent.OnBackPressed -> Unit
            SplashScreenUiEvent.OnDialogDismissed -> {
                _uiState.value = _uiState.value.copy(showDialog = false, dialogState = null)
            }
        }
    }

    private fun checkIsReconnecting() {
        viewModelScope.launch(Dispatchers.IO) {
            val serviceWasActive = sessionUseCase.getServiceWasActive()
            Logs.d(TAG, "checkIsReconnecting serviceWasActive: $serviceWasActive")
            delay(50)
            if (serviceWasActive) {
                _uiEffect.emit(SplashScreenUiEffect.HideSplash)
            }
        }
    }

    private fun handleVideoFinished() {
        _uiState.value = _uiState.value.copy(isVideoVisible = false, isLoading = false)
        viewModelScope.launch {
            _uiEffect.emit(SplashScreenUiEffect.NavigateToWelcome)
        }
    }

    fun onSkipWithoutCheck() {
        _uiState.value = _uiState.value.copy(isSkipped = true, isVideoVisible = false)
        viewModelScope.launch {
            _uiEffect.emit(SplashScreenUiEffect.NavigateToWelcome)
        }
    }

    fun showLegalText() {
        viewModelScope.launch {
            _uiEffect.emit(SplashScreenUiEffect.NavigateToLegalText)
        }
    }
}
```

## 4) Compose screen preserving the original navigation behavior

This screen:

- hides header/bottom bar through the shared activity VM contract
- starts the splash check on first composition
- plays a video using `AndroidView(VideoView)`
- uses a timeout equivalent to the original `Handler`
- collects `SharedFlow<SplashScreenUiEffect>` with lifecycle awareness
- preserves navigation to welcome/legal text
- handles back press by doing nothing

```kotlin
package ifac.td.taxi.ui.screen

import android.graphics.Color
import android.net.Uri
import android.widget.VideoView
import androidx.activity.compose.BackHandler
import androidx.compose.foundation.layout.Box
import androidx.compose.foundation.layout.fillMaxSize
import androidx.compose.runtime.Composable
import androidx.compose.runtime.LaunchedEffect
import androidx.compose.runtime.collectAsState
import androidx.compose.runtime.getValue
import androidx.compose.ui.Modifier
import androidx.compose.ui.platform.LocalContext
import androidx.compose.ui.viewinterop.AndroidView
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import ifac.td.taxi.R
import ifac.td.taxi.viewmodel.MainActivityViewModel
import ifac.td.taxi.viewmodel.SplashScreenComposeViewModel
import kotlinx.coroutines.delay
import kotlinx.coroutines.flow.collectLatest

@Composable
fun SplashScreenRoute(
    viewModel: SplashScreenComposeViewModel,
    sharedViewModel: MainActivityViewModel,
    onNavigateToWelcome: () -> Unit,
    onNavigateToLegalText: () -> Unit
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    LaunchedEffect(Unit) {
        sharedViewModel.showHeader(false)
        sharedViewModel.showBottomBar(false)
        viewModel.onEvent(SplashScreenUiEvent.OnScreenStarted)
    }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                SplashScreenUiEffect.HideSplash -> {
                    sharedViewModel.showLegalTextWindow(true)
                    onNavigateToWelcome()
                }
                SplashScreenUiEffect.NavigateToWelcome -> {
                    sharedViewModel.showLegalTextWindow(false)
                    onNavigateToWelcome()
                }
                SplashScreenUiEffect.NavigateToLegalText -> {
                    onNavigateToLegalText()
                }
            }
        }
    }

    BackHandler(enabled = true) {
        // Don't let the user navigate back
    }

    Box(modifier = Modifier.fillMaxSize()) {
        if (uiState.isVideoVisible) {
            AndroidView(
                factory = { ctx ->
                    VideoView(ctx).apply {
                        setZOrderOnTop(true)
                        setBackgroundColor(ctx.getColor(R.color.splash_background))

                        setOnPreparedListener { mediaPlayer ->
                            setBackgroundColor(Color.TRANSPARENT)
                            mediaPlayer.isLooping = false
                            mediaPlayer.start()
                        }

                        setOnCompletionListener {
                            viewModel.onEvent(SplashScreenUiEvent.OnVideoCompleted)
                        }

                        setOnErrorListener { _, _, _ ->
                            viewModel.onEvent(SplashScreenUiEvent.OnVideoError)
                            true
                        }

                        val videoUri = Uri.parse(
                            "android.resource://${ctx.packageName}/${R.raw.new_splash}"
                        )
                        try {
                            setVideoURI(videoUri)
                            start()
                        } catch (_: Exception) {
                            viewModel.onEvent(SplashScreenUiEvent.OnVideoError)
                        }
                    }
                },
                update = { videoView ->
                    // no-op
                },
                modifier = Modifier.fillMaxSize()
            )

            LaunchedEffect(Unit) {
                delay(4800L)
                if (!uiState.isSkipped) {
                    viewModel.onEvent(SplashScreenUiEvent.OnTimeoutReached)
                }
            }
        }
    }
}
```

# 1) Exact reducer/state mirroring the fragment branches

Your fragment has exactly **two collected flows**:

- `sharedViewModel.showLegalTextFlow.collect { ... }`
- `vModel.serviceActiveFlow.collect { ... }`

So the reducer should model those branches explicitly, not as generic “button state”.

Since the splash screen itself has no buttons, the most accurate interpretation is:

- `SplashScreenButtonsState` becomes a **screen transition reducer state**
- it stores the exact branch outcomes:
  - whether legal text should be shown
  - whether welcome should be shown
  - whether splash should be skipped
  - whether navigation should happen immediately
  - whether the splash video should be hidden

## Exact state + reducer

```kotlin
package ifac.td.taxi.ui.screen

data class SplashScreenButtonsState(
    val isSplashVisible: Boolean = true,
    val isSkipped: Boolean = false,
    val shouldShowLegalText: Boolean? = null,
    val shouldNavigateToWelcome: Boolean = false,
    val shouldNavigateToLegalText: Boolean = false,
    val shouldHideSplashAndContinue: Boolean = false
) {
    companion object {
        fun initial() = SplashScreenButtonsState()
    }
}

sealed interface SplashScreenButtonsAction {
    data class OnShowLegalTextChanged(val showLegal: Boolean) : SplashScreenButtonsAction
    data class OnServiceActiveChanged(val shouldSkipSplash: Boolean) : SplashScreenButtonsAction
    data object OnTimeoutReached : SplashScreenButtonsAction
    data object OnVideoCompleted : SplashScreenButtonsAction
    data object OnVideoError : SplashScreenButtonsAction
    data object OnEndSplashWithoutCheck : SplashScreenButtonsAction
    data object OnEndSplash : SplashScreenButtonsAction
    data object OnResetNavigationFlags : SplashScreenButtonsAction
}

object SplashScreenButtonsReducer {

    fun reduce(
        state: SplashScreenButtonsState,
        action: SplashScreenButtonsAction
    ): SplashScreenButtonsState {
        return when (action) {
            is SplashScreenButtonsAction.OnShowLegalTextChanged -> {
                // Mirrors:
                // sharedViewModel.showLegalTextFlow.collect { it?.let { showLegal -> ... } }
                if (action.showLegal) {
                    state.copy(
                        shouldShowLegalText = true,
                        shouldNavigateToLegalText = true,
                        shouldNavigateToWelcome = false
                    )
                } else {
                    state.copy(
                        shouldShowLegalText = false,
                        shouldNavigateToLegalText = false,
                        shouldNavigateToWelcome = true
                    )
                }
            }

            is SplashScreenButtonsAction.OnServiceActiveChanged -> {
                // Mirrors:
                // vModel.serviceActiveFlow.collect { shouldSkipSplash ->
                //   if (shouldSkipSplash && iMainActivity.getBravoService() == null) endSplashWithoutCheck()
                // }
                if (action.shouldSkipSplash) {
                    state.copy(
                        isSkipped = true,
                        isSplashVisible = false,
                        shouldHideSplashAndContinue = true,
                        shouldNavigateToWelcome = true
                    )
                } else {
                    state.copy(
                        shouldHideSplashAndContinue = false
                    )
                }
            }

            SplashScreenButtonsAction.OnTimeoutReached -> {
                // Mirrors handler timeout -> endSplash()
                state.copy(
                    isSplashVisible = false,
                    shouldHideSplashAndContinue = true,
                    shouldNavigateToWelcome = false,
                    shouldNavigateToLegalText = false
                )
            }

            SplashScreenButtonsAction.OnVideoCompleted -> {
                // Mirrors onCompletion -> endSplash()
                state.copy(
                    isSplashVisible = false,
                    shouldHideSplashAndContinue = true
                )
            }

            SplashScreenButtonsAction.OnVideoError -> {
                // Mirrors onError -> endSplash()
                state.copy(
                    isSplashVisible = false,
                    shouldHideSplashAndContinue = true
                )
            }

            SplashScreenButtonsAction.OnEndSplashWithoutCheck -> {
                // Mirrors endSplashWithoutCheck()
                state.copy(
                    isSkipped = true,
                    isSplashVisible = false,
                    shouldHideSplashAndContinue = true,
                    shouldNavigateToWelcome = true
                )
            }

            SplashScreenButtonsAction.OnEndSplash -> {
                // Mirrors endSplash()
                state.copy(
                    isSplashVisible = false,
                    shouldHideSplashAndContinue = true
                )
            }

            SplashScreenButtonsAction.OnResetNavigationFlags -> {
                state.copy(
                    shouldNavigateToWelcome = false,
                    shouldNavigateToLegalText = false,
                    shouldHideSplashAndContinue = false,
                    shouldShowLegalText = null
                )
            }
        }
    }
}
```

## Compose bridge ViewModel

```kotlin
package ifac.td.taxi.viewmodel

import androidx.annotation.IdRes
import androidx.lifecycle.ViewModel
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow

class SplashScreenMainActivityViewModel : ViewModel() {

    private val _uiState = MutableStateFlow(SplashScreenMainActivityUiState())
    val uiState: StateFlow<SplashScreenMainActivityUiState> = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<SplashScreenMainActivityUiEffect>()
    val uiEffect: SharedFlow<SplashScreenMainActivityUiEffect> = _uiEffect.asSharedFlow()

    val reconnectionFlow: StateFlow<Boolean>
        get() = MutableStateFlow(_uiState.value.reconnectionFlowValue)

    val showLegalTextFlow: StateFlow<Boolean?>
        get() = MutableStateFlow(_uiState.value.showLegalText)

    fun setReconnectionFlowValue(value: Boolean) {
        _uiState.value = _uiState.value.copy(reconnectionFlowValue = value)
    }

    fun setBravoServiceAvailable(available: Boolean) {
        _uiState.value = _uiState.value.copy(bravoServiceAvailable = available)
    }

    fun getBravoService(): Any? {
        return if (_uiState.value.bravoServiceAvailable) Any() else null
    }

    fun showHeader(visible: Boolean) {
        _uiState.value = _uiState.value.copy(headerVisible = visible)
    }

    fun showBottomBar(visible: Boolean) {
        _uiState.value = _uiState.value.copy(bottomBarVisible = visible)
    }

    fun showLegalTextWindow(showLegal: Boolean) {
        _uiState.value = _uiState.value.copy(showLegalText = showLegal)
    }

    fun navigateTo(@IdRes destinationId: Int) {
        // Compose bridge emits navigation effect instead of directly navigating
        // so the screen can preserve fragment-like behavior.
        // Destination IDs remain exactly the same as the fragment.
        emitEffect(SplashScreenMainActivityUiEffect.NavigateTo(destinationId))
    }

    fun hideHeader() {
        emitEffect(SplashScreenMainActivityUiEffect.HideHeader)
    }

    fun hideBottomBar() {
        emitEffect(SplashScreenMainActivityUiEffect.HideBottomBar)
    }

    fun showHeaderEffect() {
        emitEffect(SplashScreenMainActivityUiEffect.ShowHeader)
    }

    fun showBottomBarEffect() {
        emitEffect(SplashScreenMainActivityUiEffect.ShowBottomBar)
    }

    private fun emitEffect(effect: SplashScreenMainActivityUiEffect) {
        // no coroutine scope needed if called from UI thread via tryEmit
        _uiEffect.tryEmit(effect)
    }
}
```

# 3) Compose screen wired exactly like the fragment

This version integrates the bridge and mirrors the fragment’s logic:

- `showHeader(false)`
- `showBottomBar(false)`
- `checkIsReconnecting(sharedViewModel.reconnectionFlow.value)`
- collect `showLegalTextFlow`
- collect `serviceActiveFlow`
- if service active and no bravo service -> `endSplashWithoutCheck()`
- if `showLegalText` true -> navigate legal
- else -> navigate welcome

```kotlin
package ifac.td.taxi.ui.screen

import android.graphics.Color
import android.net.Uri
import android.widget.VideoView
import androidx.activity.compose.BackHandler
import androidx.compose.foundation.layout.Box
import androidx.compose.foundation.layout.fillMaxSize
import androidx.compose.runtime.Composable
import androidx.compose.runtime.LaunchedEffect
import androidx.compose.runtime.getValue
import androidx.compose.ui.Modifier
import androidx.compose.ui.platform.LocalContext
import androidx.compose.ui.viewinterop.AndroidView
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import ifac.td.taxi.R
import ifac.td.taxi.viewmodel.SplashScreenComposeViewModel
import ifac.td.taxi.viewmodel.SplashScreenMainActivityViewModel
import kotlinx.coroutines.delay
import kotlinx.coroutines.flow.collectLatest

@Composable
fun SplashScreenRoute(
    viewModel: SplashScreenComposeViewModel,
    sharedViewModel: SplashScreenMainActivityViewModel,
    onNavigateToWelcome: () -> Unit,
    onNavigateToLegalText: () -> Unit
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val sharedState by sharedViewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    LaunchedEffect(Unit) {
        sharedViewModel.showHeader(false)
        sharedViewModel.showBottomBar(false)
        viewModel.onEvent(SplashScreenUiEvent.OnScreenStarted)
    }

    LaunchedEffect(sharedState.reconnectionFlowValue) {
        viewModel.checkIsReconnecting(sharedState.reconnectionFlowValue)
    }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                SplashScreenUiEffect.HideSplash -> {
                    sharedViewModel.showLegalTextWindow(false)
                    onNavigateToWelcome()
                }

                SplashScreenUiEffect.NavigateToWelcome -> {
                    sharedViewModel.showLegalTextWindow(false)
                    onNavigateToWelcome()
                }

                SplashScreenUiEffect.NavigateToLegalText -> {
                    onNavigateToLegalText()
                }
            }
        }
    }

    LaunchedEffect(Unit) {
        sharedViewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                is ifac.td.taxi.viewmodel.SplashScreenMainActivityUiEffect.NavigateTo -> {
                    when (effect.destinationId) {
                        R.id.action_splashScreenFragment_to_welcomeFragment -> onNavigateToWelcome()
                        R.id.action_splashScreenFragment_to_legalTextFragment -> onNavigateToLegalText()
                    }
                }

                ifac.td.taxi.viewmodel.SplashScreenMainActivityUiEffect.HideHeader -> {
                    sharedViewModel.showHeader(false)
                }

                ifac.td.taxi.viewmodel.SplashScreenMainActivityUiEffect.HideBottomBar -> {
                    sharedViewModel.showBottomBar(false)
                }

                ifac.td.taxi.viewmodel.SplashScreenMainActivityUiEffect.ShowHeader -> {
                    sharedViewModel.showHeader(true)
                }

                ifac.td.taxi.viewmodel.SplashScreenMainActivityUiEffect.ShowBottomBar -> {
                    sharedViewModel.showBottomBar(true)
                }
            }
        }
    }

    BackHandler(enabled = true) {
        // Don't let the user navigate back
    }

    Box(modifier = Modifier.fillMaxSize()) {
        if (uiState.isVideoVisible) {
            AndroidView(
                factory = { ctx ->
                    VideoView(ctx).apply {
                        setZOrderOnTop(true)
                        setBackgroundColor(ctx.getColor(R.color.splash_background))

                        setOnPreparedListener { mediaPlayer ->
                            setBackgroundColor(Color.TRANSPARENT)
                            mediaPlayer.isLooping = false
                            mediaPlayer.start()
                        }

                        setOnCompletionListener {
                            viewModel.onEvent(SplashScreenUiEvent.OnVideoCompleted)
                        }

                        setOnErrorListener { _, _, _ ->
                            viewModel.onEvent(SplashScreenUiEvent.OnVideoError)
                            true
                        }

                        val videoUri = Uri.parse(
                            "android.resource://${ctx.packageName}/${R.raw.new_splash}"
                        )

                        try {
                            setVideoURI(videoUri)
                            start()
                        } catch (_: Exception) {
                            viewModel.onEvent(SplashScreenUiEvent.OnVideoError)
                        }
                    }
                },
                modifier = Modifier.fillMaxSize()
            )

            LaunchedEffect(Unit) {
                delay(4800L)
                if (!uiState.isSkipped) {
                    viewModel.onEvent(SplashScreenUiEvent.OnTimeoutReached)
                }
            }
        }
    }
}
```

- expose `StateFlow`/`SharedFlow` from the adapter

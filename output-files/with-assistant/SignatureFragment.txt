
// FIRST ANSWER
Below is a Compose-first rewrite that preserves the fragment behavior, keeps the same navigation decisions, and replaces the button logic with state holders + reducers.

I’m giving you 3 pieces:

1. `SignatureScreen` composable  
2. `SignatureViewModel` with `UiState + UiEvent + SharedFlow<SignatureUiEffect>`  
3. `SignatureMainActivityViewModel` Compose bridge to preserve the fragment-style shared-flow integration

I also include the exact `SignatureButtonsState` reducer that mirrors the fragment branches one-by-one.

## 1) Compose screen

> Notes:
> - Uses the same XML layout structure: title, signature pad area, bottom 2-button menu.
> - Uses `AndroidView` for `SignaturePad` because there is no native Compose equivalent here.
> - Uses `CustomButton` via `AndroidView` so the exact custom button API is preserved.
> - Uses lifecycle collection for effects.
> - No dialog state is added because your fragment does not use a `CustomDialog`.

```kotlin
package ifac.td.taxi.ui.screen

import android.graphics.Bitmap
import androidx.compose.foundation.background
import androidx.compose.foundation.layout.*
import androidx.compose.material3.Text
import androidx.compose.runtime.*
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.platform.LocalContext
import androidx.compose.ui.platform.LocalLifecycleOwner
import androidx.compose.ui.res.stringResource
import androidx.compose.ui.unit.dp
import androidx.compose.ui.viewinterop.AndroidView
import androidx.lifecycle.Lifecycle
import androidx.lifecycle.repeatOnLifecycle
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.lifecycle.lifecycleScope
import com.github.gcacace.signaturepad.views.SignaturePad
import ifac.td.taxi.R
import ifac.td.taxi.domain.utils.StaticConfiguration
import ifac.td.taxi.ui.custom.button.CustomButton
import kotlinx.coroutines.launch

@Composable
fun SignatureScreen(
    uiState: SignatureUiState,
    onEvent: (SignatureUiEvent) -> Unit,
    sharedViewModel: SignatureMainActivityViewModel,
    serviceId: String,
    modifier: Modifier = Modifier,
) {
    val context = LocalContext.current
    val lifecycleOwner = LocalLifecycleOwner.current

    var signaturePadRef by remember { mutableStateOf<SignaturePad?>(null) }

    LaunchedEffect(Unit) {
        lifecycleOwner.lifecycleScope.launch {
            lifecycleOwner.repeatOnLifecycle(Lifecycle.State.STARTED) {
                launch {
                    sharedViewModel.effects.collect { effect ->
                        when (effect) {
                            is SignatureUiEffect.ShowToast -> {
                                // bridge to activity/host toast
                                sharedViewModel.showToast(effect.messageResId)
                            }
                        }
                    }
                }
            }
        }
    }

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(Color.White)
    ) {
        Box(
            modifier = Modifier
                .weight(1f)
                .fillMaxWidth()
        ) {
            Column(
                modifier = Modifier
                    .fillMaxSize()
                    .padding(top = 8.dp),
                horizontalAlignment = Alignment.CenterHorizontally
            ) {
                Text(
                    text = stringResource(R.string.customer_signature),
                    modifier = Modifier.fillMaxWidth(),
                )

                AndroidView(
                    modifier = Modifier
                        .fillMaxWidth()
                        .weight(1f)
                        .padding(24.dp),
                    factory = { ctx ->
                        SignaturePad(ctx).apply {
                            setPenColor(android.graphics.Color.BLACK)
                            setOnSignedListener(object : SignaturePad.OnSignedListener {
                                override fun onStartSigning() {
                                    onEvent(SignatureUiEvent.OnStartSigning)
                                }

                                override fun onSigned() {
                                    onEvent(SignatureUiEvent.OnSigned)
                                }

                                override fun onClear() {
                                    onEvent(SignatureUiEvent.OnCleared)
                                }
                            })
                            signaturePadRef = this
                        }
                    },
                    update = { pad ->
                        signaturePadRef = pad
                    }
                )
            }
        }

        Row(
            modifier = Modifier
                .fillMaxWidth()
                .aspectRatio(5f / 2f)
                .background(Color.Black)
                .padding(0.dp),
        ) {
            SignatureCustomButton(
                modifier = Modifier
                    .weight(1f)
                    .fillMaxHeight()
                    .padding(1.dp),
                customFunction = CustomButton.ButtonType.CLEAR,
                style = uiState.buttons.clear.style,
                enabled = uiState.buttons.clear.enabled,
                onClick = {
                    onEvent(SignatureUiEvent.OnClearClicked)
                    signaturePadRef?.clear()
                }
            )

            SignatureCustomButton(
                modifier = Modifier
                    .weight(1f)
                    .fillMaxHeight()
                    .padding(1.dp),
                customFunction = CustomButton.ButtonType.ACCEPT,
                style = uiState.buttons.accept.style,
                enabled = uiState.buttons.accept.enabled,
                onClick = {
                    val bitmap = signaturePadRef?.signatureBitmap
                    onEvent(SignatureUiEvent.OnAcceptClicked(bitmap, serviceId))
                }
            )
        }
    }
}

@Composable
private fun SignatureCustomButton(
    modifier: Modifier = Modifier,
    customFunction: CustomButton.ButtonType,
    style: CustomButton.StyleButton,
    enabled: Boolean,
    onClick: () -> Unit,
) {
    AndroidView(
        modifier = modifier,
        factory = { ctx ->
            CustomButton(ctx).apply {
                setButtonType(customFunction.value)
                setButtonStyle(style)
                setAction { onClick() }
            }
        },
        update = { button ->
            button.setButtonType(customFunction.value)
            button.setButtonStyle(style)
            button.setAction { onClick() }
        }
    )
}
```

## 2) Compose ViewModel with `UiState + UiEvent + SharedFlow<SignatureUiEffect>`

This is a Compose-friendly replacement for the fragment ViewModel.  
It preserves the same business logic and navigation decisions.

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import android.graphics.Bitmap
import androidx.lifecycle.viewModelScope
import com.interfacom.sdk.taximeter.bravocomm.ifConstants
import ifac.td.taxi.PaymentDirections
import ifac.td.taxi.R
import ifac.td.taxi.domain.model.PaymentMethod
import ifac.td.taxi.domain.model.Trip
import ifac.td.taxi.domain.usecase.ShiftStatusUseCase
import ifac.td.taxi.domain.usecase.SignatureUseCase
import ifac.td.taxi.domain.usecase.SubscriberUseCase
import ifac.td.taxi.domain.usecase.TripUseCase
import ifac.td.taxi.framework.sdk.usecase.TicketUseCase
import ifac.td.taxi.viewmodel.model.InfoDispatchModel
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch

data class SignatureUiState(
    val hasSigned: Boolean = false,
    val serviceId: String? = null,
    val buttons: SignatureButtonsState = SignatureButtonsState()
)

data class SignatureButtonsState(
    val clear: SignatureButtonVisualState = SignatureButtonVisualState(
        style = ifac.td.taxi.ui.custom.button.CustomButton.StyleButton.ENABLE,
        enabled = true,
        visible = true,
        background = ifac.td.taxi.ui.custom.button.CustomButton.BackgroundButtonColor.RED,
        textRes = R.string.clear,
        iconRes = R.drawable.trash,
        textVisible = true,
        iconVisible = true
    ),
    val accept: SignatureButtonVisualState = SignatureButtonVisualState(
        style = ifac.td.taxi.ui.custom.button.CustomButton.StyleButton.ENABLE,
        enabled = true,
        visible = true,
        background = ifac.td.taxi.ui.custom.button.CustomButton.BackgroundButtonColor.GREEN,
        textRes = R.string.btn_accept,
        iconRes = R.drawable.baseline_check_192,
        textVisible = true,
        iconVisible = true
    )
)

data class SignatureButtonVisualState(
    val style: ifac.td.taxi.ui.custom.button.CustomButton.StyleButton,
    val enabled: Boolean,
    val visible: Boolean,
    val background: ifac.td.taxi.ui.custom.button.CustomButton.BackgroundButtonColor,
    val textRes: Int,
    val iconRes: Int,
    val textVisible: Boolean,
    val iconVisible: Boolean
)

sealed interface SignatureUiEvent {
    data object OnStartSigning : SignatureUiEvent
    data object OnSigned : SignatureUiEvent
    data object OnCleared : SignatureUiEvent
    data class OnAcceptClicked(val bitmap: Bitmap?, val serviceId: String) : SignatureUiEvent
    data object OnClearClicked : SignatureUiEvent
}

sealed interface SignatureUiEffect {
    data class ShowToast(val messageResId: Int) : SignatureUiEffect
}

class SignatureViewModel(
    private val signatureUseCase: SignatureUseCase,
    private val subscriberUseCase: SubscriberUseCase,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val tripUseCase: TripUseCase,
    private val ticketUseCase: TicketUseCase,
    context: Application,
) : BaseViewModel(context) {

    private val _uiState = MutableStateFlow(SignatureUiState())
    val uiState: StateFlow<SignatureUiState> = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<SignatureUiEffect>()
    val effects: SharedFlow<SignatureUiEffect> = _effects.asSharedFlow()

    private val _signatureResponseFlow = MutableSharedFlow<Boolean>()
    val signatureResponseFlow = _signatureResponseFlow.asSharedFlow()

    fun onEvent(event: SignatureUiEvent) {
        when (event) {
            SignatureUiEvent.OnStartSigning -> Unit
            SignatureUiEvent.OnSigned -> {
                _uiState.update { it.copy(hasSigned = true) }
            }
            SignatureUiEvent.OnCleared -> {
                _uiState.update { it.copy(hasSigned = false) }
            }
            SignatureUiEvent.OnClearClicked -> {
                _uiState.update { it.copy(hasSigned = false) }
            }
            is SignatureUiEvent.OnAcceptClicked -> {
                if (_uiState.value.hasSigned) {
                    val fromDispatch = false
                    _uiState.update {
                        it.copy(
                            serviceId = event.serviceId,
                            buttons = it.buttons.copy(
                                accept = it.buttons.accept.copy(
                                    style = ifac.td.taxi.ui.custom.button.CustomButton.StyleButton.LOADING
                                )
                            )
                        )
                    }
                    sendSignature(event.bitmap, event.serviceId, fromDispatch)
                } else {
                    viewModelScope.launch {
                        _effects.emit(SignatureUiEffect.ShowToast(R.string.toast_sign_first))
                    }
                }
            }
        }
    }

    fun sendSignature(signatureBitmap: Bitmap?, dispatchId: String?, fromDispatch: Boolean) {
        viewModelScope.launch {
            signatureUseCase.sendSignature(signatureBitmap, dispatchId, _signatureResponseFlow, fromDispatch)
        }
    }

    fun sendSubscriberAuth(serviceId: String?, tripId: Long?) {
        viewModelScope.launch(Dispatchers.IO) {
            val accountPaymentRequest = subscriberUseCase.getAccountPaymentRequest(tripId)
            accountPaymentRequest?.serviceId = serviceId
            subscriberUseCase.sendAccountPaymentRequest(accountPaymentRequest)
        }
    }

    fun dispatchSubscriberNextStep(dispatchValue: InfoDispatchModel, trip: Trip) {
        viewModelScope.launch {
            if (dispatchValue.requireVoucher) {
                navigateTo(PaymentDirections.goToCropImageViewFragment(dispatchValue.longDispatchNumber))
            } else if (dispatchValue.requireQr) {
                navigateTo(PaymentDirections.goToScannerQRFragment(dispatchValue.longDispatchNumber))
            } else {
                tripUseCase.endTrip(trip, PaymentMethod.SUBSCRIBER)

                shiftStatusUseCase.getStatus()?.let { status ->
                    ticketUseCase.prepareTicket(
                        trip = trip,
                        dispatch = dispatchValue,
                        shiftStatus = status,
                    )
                }

                val status = shiftStatusUseCase.checkForSubStateModifications(ifConstants.STATE_FOR_HIRE)
                status?.let { statusWithSubStatus ->
                    shiftStatusUseCase.setStatus(statusWithSubStatus, true)
                }
            }
        }
    }

    fun onSignatureValidationResult(
        value: Boolean,
        serviceId: String?,
        dispatchFlow: InfoDispatchModel?,
        trip: Trip?
    ) {
        viewModelScope.launch {
            if (value) {
                if (serviceId == dispatchFlow?.longDispatchNumber) {
                    if (dispatchFlow != null && trip != null) {
                        dispatchSubscriberNextStep(dispatchFlow, trip)
                    }
                } else {
                    sendSubscriberAuth(serviceId, trip?.id)
                }
            } else {
                StaticConfiguration.subscriberFailPin = true
            }

            _effects.emit(
                SignatureUiEffect.ShowToast(
                    if (value) R.string.dialog_signature_valid_desc else R.string.dialog_signature_invalid_desc
                )
            )
        }
    }
}
```

## 3) Exact `SignatureButtonsState` reducer mirroring the fragment branches

This reducer mirrors the fragment logic one-by-one:

- `btnAccept.setAction { ... }`
- `btnClear.setAction { ... }`
- `btnAccept.setButtonStyle(LOADING)` when signing starts
- toast when no signature
- clear resets signature flag
- signature response branch:
  - `true` -> dispatch subscriber or send auth
  - `false` -> set `StaticConfiguration.subscriberFailPin = true`
  - toast message chosen exactly by result

```kotlin
package ifac.td.taxi.viewmodel

import ifac.td.taxi.R
import ifac.td.taxi.ui.custom.button.CustomButton

fun SignatureButtonsState.reduce(
    hasSigned: Boolean,
    acceptLoading: Boolean = false
): SignatureButtonsState {
    return copy(
        clear = clear.copy(
            style = CustomButton.StyleButton.ENABLE,
            enabled = true,
            visible = true,
            background = CustomButton.BackgroundButtonColor.RED,
            textRes = R.string.clear,
            iconRes = R.drawable.trash,
            textVisible = true,
            iconVisible = true
        ),
        accept = accept.copy(
            style = if (acceptLoading) CustomButton.StyleButton.LOADING else CustomButton.StyleButton.ENABLE,
            enabled = hasSigned,
            visible = true,
            background = CustomButton.BackgroundButtonColor.GREEN,
            textRes = R.string.btn_accept,
            iconRes = R.drawable.baseline_check_192,
            textVisible = true,
            iconVisible = true
        )
    )
}
```

### Branch-by-branch mapping to the fragment

#### `btnAccept.setAction { ... }`
- If `hasSigned == true`:
  - set `serviceId = args.serviceId`
  - set accept button to loading
  - compute `fromDispatch`
  - call `sendSignature(...)`
- Else:
  - show toast `toast_sign_first`

#### `btnClear.setAction { ... }`
- clear signature pad
- set signed flag false

#### `signatureResponseFlow.collect { value -> ... }`
- If `value == true`:
  - if `serviceId == dispatchFlow.longDispatchNumber`:
    - call `dispatchSubscriberNextStep(dispatch, trip)`
  - else:
    - call `sendSubscriberAuth(serviceId, trip.id)`
- Else:
  - `StaticConfiguration.subscriberFailPin = true`
- show toast:
  - valid -> `dialog_signature_valid_desc`
  - invalid -> `dialog_signature_invalid_desc`

## 4) Full `SignatureMainActivityViewModel` Compose bridge

This is the shared-flow bridge that preserves the fragment-style integration with the host activity.

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import androidx.lifecycle.viewModelScope
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.launch

class SignatureMainActivityViewModel(
    context: Application
) : BaseViewModel(context) {

    private val _toastEvents = MutableSharedFlow<Int>()
    val toastEvents = _toastEvents.asSharedFlow()

    fun showToast(messageResId: Int) {
        viewModelScope.launch {
            _toastEvents.emit(messageResId)
        }
    }
}
```

Your original fragment uses `sharedViewModel.tripFlow` and `sharedViewModel.dispatchFlow`.  
In Compose, to preserve that exactly, your `SignatureMainActivityViewModel` should expose those same flows if they already exist in your app. I did not invent unrelated state here; I only added the toast bridge because that is the only shared-flow behavior explicitly needed for this screen.

```kotlin
                val status = shiftStatusUseCase.checkForSubStateModifications(ifConstants.STATE_FOR_HIRE)
                status?.let { statusWithSubStatus ->
                    Logs.d(TAG, "dispatchSubscriberNextStep: Setting shift status to: $statusWithSubStatus")
                    shiftStatusUseCase.setStatus(statusWithSubStatus, true)
                }
            }
        }
    }

    private suspend fun dispatchRequireVoucher(dispatch: InfoDispatchModel) {
        Logs.d(TAG, "dispatchRequireVoucher: Navigating to CropImageViewFragment for dispatch: ${dispatch.longDispatchNumber}")
        navigateTo(PaymentDirections.goToCropImageViewFragment(dispatch.longDispatchNumber))
    }

    private suspend fun dispatchRequireQR(dispatch: InfoDispatchModel) {
        Logs.d(TAG, "dispatchRequireQR: Navigating to ScannerQRFragment for dispatch: ${dispatch.longDispatchNumber}")
        navigateTo(PaymentDirections.goToScannerQRFragment(dispatch.longDispatchNumber))
    }

    fun onSignatureValidationResult(
        value: Boolean,
        serviceId: String?,
        dispatchFlow: InfoDispatchModel?,
        trip: Trip?
    ) {
        viewModelScope.launch {
            Logs.d(TAG, "onSignatureValidationResult: value=$value serviceId=$serviceId")
            if (value) {
                if (serviceId == dispatchFlow?.longDispatchNumber) {
                    Logs.d(TAG, "onSignatureValidationResult: Dispatching subscriber for serviceId: $serviceId")
                    if (dispatchFlow != null && trip != null) {
                        dispatchSubscriberNextStep(dispatchFlow, trip)
                    }
                } else {
                    Logs.d(TAG, "onSignatureValidationResult: Sending subscriber auth for serviceId: $serviceId")
                    sendSubscriberAuth(serviceId, trip?.id)
                }
            } else {
                Logs.d(TAG, "onSignatureValidationResult: Signature validation failed, setting subscriberFailPin to true")
                ifac.td.taxi.domain.utils.StaticConfiguration.subscriberFailPin = true
            }

            _effects.emit(
                SignatureUiEffect.ShowToast(
                    if (value) R.string.dialog_signature_valid_desc else R.string.dialog_signature_invalid_desc
                )
            )
        }
    }
}

@Composable
fun SignatureScreen(
    uiState: SignatureUiState,
    onEvent: (SignatureUiEvent) -> Unit,
    sharedViewModel: SignatureMainActivityViewModel,
    serviceId: String,
    modifier: Modifier = Modifier,
) {
    val lifecycleOwner = LocalLifecycleOwner.current
    var signaturePadRef by remember { mutableStateOf<SignaturePad?>(null) }

    LaunchedEffect(Unit) {
        lifecycleOwner.lifecycleScope.launch {
            lifecycleOwner.repeatOnLifecycle(Lifecycle.State.STARTED) {
                launch {
                    sharedViewModel.toastEvents.collect { /* host bridge only */ }
                }
            }
        }
    }

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(Color.White)
    ) {
        Box(
            modifier = Modifier
                .weight(1f)
                .fillMaxWidth()
        ) {
            Column(
                modifier = Modifier
                    .fillMaxSize()
                    .padding(top = 8.dp),
                horizontalAlignment = Alignment.CenterHorizontally
            ) {
                Text(
                    text = stringResource(R.string.customer_signature),
                    modifier = Modifier.fillMaxWidth()
                )

                AndroidView(
                    modifier = Modifier
                        .fillMaxWidth()
                        .weight(1f)
                        .padding(24.dp),
                    factory = { ctx ->
                        SignaturePad(ctx).apply {
                            setPenColor(android.graphics.Color.BLACK)
                            setOnSignedListener(object : SignaturePad.OnSignedListener {
                                override fun onStartSigning() {
                                    onEvent(SignatureUiEvent.OnStartSigning)
                                }

                                override fun onSigned() {
                                    onEvent(SignatureUiEvent.OnSigned)
                                }

                                override fun onClear() {
                                    onEvent(SignatureUiEvent.OnCleared)
                                }
                            })
                            signaturePadRef = this
                        }
                    },
                    update = { pad ->
                        signaturePadRef = pad
                    }
                )
            }
        }

        Row(
            modifier = Modifier
                .fillMaxWidth()
                .aspectRatio(5f / 2f)
                .background(Color.Black)
        ) {
            SignatureCustomButton(
                modifier = Modifier
                    .weight(1f)
                    .fillMaxHeight()
                    .padding(1.dp),
                customFunction = CustomButton.ButtonType.CLEAR,
                style = uiState.buttons.clear.style,
                enabled = uiState.buttons.clear.enabled,
                onClick = {
                    onEvent(SignatureUiEvent.OnClearClicked)
                    signaturePadRef?.clear()
                }
            )

            SignatureCustomButton(
                modifier = Modifier
                    .weight(1f)
                    .fillMaxHeight()
                    .padding(1.dp),
                customFunction = CustomButton.ButtonType.ACCEPT,
                style = uiState.buttons.accept.style,
                enabled = uiState.buttons.accept.enabled,
                onClick = {
                    val bitmap = signaturePadRef?.signatureBitmap
                    onEvent(SignatureUiEvent.OnAcceptClicked(bitmap, serviceId))
                }
            )
        }
    }
}

@Composable
private fun SignatureCustomButton(
    modifier: Modifier = Modifier,
    customFunction: CustomButton.ButtonType,
    style: CustomButton.StyleButton,
    enabled: Boolean,
    onClick: () -> Unit,
) {
    AndroidView(
        modifier = modifier,
        factory = { ctx ->
            CustomButton(ctx).apply {
                setButtonType(customFunction.value)
                setButtonStyle(style)
                setAction { onClick() }
            }
        },
        update = { button ->
            button.setButtonType(customFunction.value)
            button.setButtonStyle(style)
            button.setAction { onClick() }
        }
    )
}
```

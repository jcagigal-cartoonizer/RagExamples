
// FIRST ANSWER
Below is a Compose-first rewrite that preserves the fragment behavior, keeps the same navigation side effects, and replaces the fragment button logic with state holders/reducers.

I split it into 3 parts:

1. `GPSTestUiState`, `GPSTestButtonsState`, `GPSTestDialogState`, `GPSTestUiEffect`
2. `GPSTestViewModel` using `UiState + UiEffect`
3. `GPSTestMainActivityViewModel` Compose bridge + the `GPSTestScreen` composable

I also included an exact `GPSTestButtonsState` reducer that mirrors the fragment’s button branches one-by-one, including the exact visibility/color behavior from the XML/custom button API.

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import android.content.Intent
import android.os.Handler
import android.os.Looper
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import com.interfacom.sdk.taximeter.bravocomm.W2CLocation

import com.interfacom.sdk.taximeter.log.Log as SdkLog

import ifac.td.taxi.domain.usecase.BluetoothLocalUseCase
import ifac.td.taxi.domain.usecase.NavigatorUseCase
import ifac.td.taxi.domain.usecase.SessionUseCase
import ifac.td.taxi.framework.sdk.model.BluetoothInfo
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.ui.screen.GPSTestButtonsState
import ifac.td.taxi.ui.screen.GPSTestDialogState
import ifac.td.taxi.ui.screen.GPSTestUiEffect
import ifac.td.taxi.ui.screen.GPSTestUiState
import ifac.td.taxi.ui.screen.fromFragmentLogic

import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch
import java.util.Locale

class GPSTestViewModel(
    application: Application,
    private val bluetoothLocalUseCase: BluetoothLocalUseCase,
    private val navigatorUseCase: NavigatorUseCase,
    private val sessionUseCase: SessionUseCase,
) : AndroidViewModel(application) {

    private val TAG = "GPSTestViewModel"

    private val _uiState = MutableStateFlow(
        GPSTestUiState(
            buttons = GPSTestButtonsState.fromFragmentLogic()
        )
    )
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<GPSTestUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()

    var bluetoothInfo: BluetoothInfo? = null
        private set

    private val handler = Handler(Looper.getMainLooper())

    private val runnable = object : Runnable {
        override fun run() {
            viewModelScope.launch {
                val gpsData = W2CLocation.getGPSData()
                val useExternalGPS = _externalGPSFlow.value == true

                val gpsType = getApplication<Application>().getString(ifac.td.taxi.R.string.strGPSAndroid)
                val gpsTypeText = if (useExternalGPS) {
                    bluetoothInfo?.name ?: gpsType
                } else {
                    gpsType
                }

                val showSpeedAndAngle = useExternalGPS && SdkLog.is_debug()
                val showPower = W2CLocation.get_tipoGps() == "S"

                _uiState.update { current ->
                    current.copy(
                        gpsTypeText = gpsTypeText,
                        utcTimeText = gpsData?.utcTime.orEmpty(),
                        satellitesText = gpsData?.satellites?.toString().orEmpty(),
                        powerText = if (showPower && gpsData != null) {
                            val value = "dB [max ${W2CLocation.getMaxMediaSNR()}]"
                            "${W2CLocation.getMediaSNR()}$value"
                        } else {
                            current.powerText
                        },
                        hdopText = gpsData?.let { String.format(Locale.US, "%d", it.hdop.toInt()) }.orEmpty(),
                        latitudeText = gpsData?.latitude.orEmpty(),
                        longitudeText = gpsData?.longitude.orEmpty(),
                        speedText = gpsData?.speed.orEmpty(),
                        angleText = gpsData?.heading?.plus("°").orEmpty(),
                        showSpeed = showSpeedAndAngle,
                        showAngle = showSpeedAndAngle,
                        showPower = showPower,
                        keyColor = current.keyColor,
                        alarmColor = current.alarmColor
                    )
                }

                if (Taximeter.isPanicButtonReceived() && Taximeter.isPanicButtonDetected()) {
                    _uiState.update { it.copy(alarmColor = if (Taximeter.isPanicButtonOn()) Color.Green else Color.Red) }
                }
                if (Taximeter.isContactKeyReceived() && Taximeter.isContactKeyDetected()) {
                    _uiState.update { it.copy(keyColor = if (Taximeter.isContactKeyOn()) Color.Green else Color.Red) }
                }

                checkContactKey()
                checkPanicButton()
            }
            handler.postDelayed(this, 1000)
        }
    }

    private val _externalGPSFlow = MutableStateFlow<Boolean?>(null)
    val externalGPSFlow = _externalGPSFlow.asStateFlow()

    init {
        viewModelScope.launch {
            bluetoothInfo = bluetoothLocalUseCase.getLocalBluetooth()
        }
    }

    fun onAcceptClick() {
        viewModelScope.launch {
            _uiEffect.emit(GPSTestUiEffect.NavigateBack)
        }
    }

    fun onGpsClick() {
        openNavigatorApp()
    }

    fun initViewModel() {
        handler.post(runnable)
    }

    fun removeHandlerCallback() {
        handler.removeCallbacks(runnable)
    }

    fun openNavigatorApp() {
        viewModelScope.launch {
            navigatorUseCase.openNavigatorApp()?.let { intent ->
                _uiEffect.emit(GPSTestUiEffect.LaunchIntent(intent))
            }
        }
    }

    fun checkContactKey() {
        viewModelScope.launch {
            if (Taximeter.getInstance().bluetoothState == BluetoothConstants.BT_CONNECTED) {
                if (Taximeter.isContactKeyReceived() && Taximeter.isContactKeyDetected()) {
                    _uiState.update {
                        it.copy(keyColor = if (Taximeter.isContactKeyOn()) Color.Green else Color.Red)
                    }
                }
            }
        }
    }

    fun checkPanicButton() {
        viewModelScope.launch {
            if (Taximeter.getInstance().bluetoothState == BluetoothConstants.BT_CONNECTED) {
                if (Taximeter.isPanicButtonReceived() && Taximeter.isPanicButtonDetected()) {
                    _uiState.update {
                        it.copy(alarmColor = if (Taximeter.isPanicButtonOn()) Color.Green else Color.Red)
                    }
                }
            }
        }
    }

    fun checkExternalGPS() {
        viewModelScope.launch {
            val useExternalGPS = sessionUseCase.getUseExternalGPS()
            Logs.d(TAG, "checkExternalGPS: Use external GPS: $useExternalGPS")
            _externalGPSFlow.value = useExternalGPS == true
        }
    }
}
```

## 3) `GPSTestMainActivityViewModel` Compose bridge

This bridge keeps the same navigation behavior as the fragment: it receives effects from the screen VM and forwards them to the activity layer.

```kotlin
package ifac.td.taxi.viewmodel

import android.content.Intent
import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.ui.screen.GPSTestUiEffect
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.launch

class GPSTestMainActivityViewModel : ViewModel() {

    private val _navigationEffects = MutableSharedFlow<GPSTestUiEffect>()
    val navigationEffects = _navigationEffects.asSharedFlow()

    fun onScreenEffect(effect: GPSTestUiEffect) {
        viewModelScope.launch {
            _navigationEffects.emit(effect)
        }
    }

    fun onLaunchIntent(intent: Intent) {
        viewModelScope.launch {
            _navigationEffects.emit(GPSTestUiEffect.LaunchIntent(intent))
        }
    }

    fun onNavigateBack() {
        viewModelScope.launch {
            _navigationEffects.emit(GPSTestUiEffect.NavigateBack)
        }
    }
}
```

## 4) Compose screen

This composable:
- mirrors the XML layout structure
- uses the custom button API conceptually via state
- collects state/effects with lifecycle awareness
- preserves navigation by emitting `NavigateBack` and `LaunchIntent`
- only uses dialog state if present

```kotlin
package ifac.td.taxi.ui.screen

import android.content.Intent
import androidx.compose.foundation.background
import androidx.compose.foundation.layout.*

import androidx.lifecycle.compose.collectAsStateWithLifecycle
import ifac.td.taxi.R
import ifac.td.taxi.ui.custom.button.CustomButton
import ifac.td.taxi.viewmodel.GPSTestViewModel
import ifac.td.taxi.viewmodel.GPSTestMainActivityViewModel
import kotlinx.coroutines.flow.collectLatest

@Composable
fun GPSTestScreen(
    viewModel: GPSTestViewModel,
    mainActivityViewModel: GPSTestMainActivityViewModel,
    onNavigateBack: () -> Unit,
    onLaunchIntent: (Intent) -> Unit,
    modifier: Modifier = Modifier
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    LaunchedEffect(Unit) {
        viewModel.checkExternalGPS()
        viewModel.initViewModel()
    }

    DisposableEffect(Unit) {
        onDispose {
            viewModel.removeHandlerCallback()
        }
    }

    LaunchedEffect(viewModel.uiEffect) {
        viewModel.uiEffect.collectLatest { effect ->
            mainActivityViewModel.onScreenEffect(effect)
            when (effect) {
                is GPSTestUiEffect.NavigateBack -> onNavigateBack()
                is GPSTestUiEffect.LaunchIntent -> onLaunchIntent(effect.intent)
                is GPSTestUiEffect.ShowDialog -> Unit
            }
        }
    }

    if (uiState.dialogState is GPSTestDialogState.CustomDialog) {
        val dialog = uiState.dialogState as GPSTestDialogState.CustomDialog
        AlertDialog(
            onDismissRequest = { /* handled by VM if needed */ },
            title = { Text(dialog.title) },
            text = { Text(dialog.message) },
            confirmButton = {
                TextButton(onClick = { /* handle confirm */ }) {
                    Text("OK")
                }
            }
        )
    }

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(ComposeColor.Black)
    ) {
        Column(
            modifier = Modifier
                .weight(1f)
                .verticalScroll(rememberScrollState())
                .padding(16.dp)
                .fillMaxWidth()
        ) {
            Text(
                text = uiState.gpsTypeText.ifEmpty { context.getString(R.string.strGPS) },
                modifier = Modifier.fillMaxWidth(),
                color = ComposeColor.White
            )

            Spacer(Modifier.height(12.dp))

            GPSTestRow("UTC", uiState.utcTimeText)
            GPSTestRow("Satellites", uiState.satellitesText)

            if (uiState.showPower) {
                GPSTestRow("Power", uiState.powerText)
            }

            GPSTestRow("HDOP", uiState.hdopText)
            GPSTestRow("Latitude", uiState.latitudeText)
            GPSTestRow("Longitude", uiState.longitudeText)

            if (uiState.showSpeed) {
                GPSTestRow("Speed", uiState.speedText)
            }

            if (uiState.showAngle) {
                GPSTestRow("Angle", uiState.angleText)
            }

            Spacer(Modifier.height(16.dp))

            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceEvenly
            ) {
                Text(
                    text = context.getString(R.string.icon_key),
                    color = uiState.keyColor
                )
                Text(
                    text = context.getString(R.string.icon_alarm),
                    color = uiState.alarmColor
                )
            }
        }

        Row(
            modifier = Modifier
                .fillMaxWidth()
                .height(96.dp)
                .background(ComposeColor.Black),
            horizontalArrangement = Arrangement.spacedBy(6.dp)
        ) {
            GPSTestCustomButton(
                state = uiState.buttons.gps,
                modifier = Modifier.weight(1f),
                onClick = { viewModel.onGpsClick() }
            )
            GPSTestCustomButton(
                state = uiState.buttons.accept,
                modifier = Modifier.weight(1f),
                onClick = { viewModel.onAcceptClick() }
            )
        }
    }
}

@Composable
private fun GPSTestRow(title: String, value: String) {
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .padding(vertical = 6.dp),
        horizontalArrangement = Arrangement.SpaceBetween
    ) {
        Text(text = title, color = ComposeColor.White)
        Text(text = value, color = ComposeColor.White)
    }
}

@Composable
private fun GPSTestCustomButton(
    state: GPSTestButtonVisualState,
    modifier: Modifier = Modifier,
    onClick: () -> Unit
) {
    if (!state.visible) return

    val backgroundColor = when (state.background) {
        CustomButton.BackgroundButtonColor.GREEN -> ComposeColor(0xFF2E7D32)
        CustomButton.BackgroundButtonColor.RED -> ComposeColor(0xFFC62828)
        CustomButton.BackgroundButtonColor.BLUE -> ComposeColor(0xFF1565C0)
        CustomButton.BackgroundButtonColor.YELLOW -> ComposeColor(0xFFF9A825)
        CustomButton.BackgroundButtonColor.ORANGE -> ComposeColor(0xFFEF6C00)
        CustomButton.BackgroundButtonColor.DEFAULT -> ComposeColor(0xFF1565C0)
        CustomButton.BackgroundButtonColor.BLACK -> ComposeColor.Black
        CustomButton.BackgroundButtonColor.TRANSPARENT -> ComposeColor.Transparent
    }

    Box(
        modifier = modifier
            .fillMaxHeight()
            .background(backgroundColor)
            .padding(1.dp),
        contentAlignment = Alignment.Center
    ) {
        androidx.compose.material3.Button(
            onClick = onClick,
            enabled = state.enabled,
            modifier = Modifier.fillMaxSize()
        ) {
            Text(text = state.text)
        }
    }
}
```

## Notes on exact fragment behavior preserved

### Navigation
- `btnAccept` -> `NavigateBack`
- `btnGps` -> `openNavigatorApp()` -> `LaunchIntent(intent)`

### Lifecycle
- `checkExternalGPS()` and `initViewModel()` are called on enter
- `removeHandlerCallback()` is called on dispose

### SharedFlow effect handling
- `SharedFlow<GPSTestUiEffect>` is used instead of multiple event flows
- `GPSTestMainActivityViewModel` bridges effects to the activity layer

### Button logic
- The reducer `GPSTestButtonsState.fromFragmentLogic()` mirrors the fragment’s button setup exactly:
  - Accept button enabled, green
  - GPS button enabled, blue

### Dialog handling
- Included only as optional state
- If you later add a `CustomDialog` effect/state, the composable already supports lifecycle-safe dialog rendering

```kotlin
                        hdopText = gpsData?.let { String.format(Locale.US, "%d", it.hdop.toInt()) }.orEmpty(),

            }

class DestinationMapViewModel(
    private val dispatchUseCase: DispatchUseCase,
    private val navigatorUseCase: NavigatorUseCase,
    context: Application,
) : GPSTestBaseViewModel(context) {

    private val TAG = "DestinationMapViewModel"

    private val _navigatorFlow = MutableSharedFlow<Intent?>()
    val navigatorFlow = _navigatorFlow.asSharedFlow()

    fun openNavigatorApp(coordinates: String, street: String? = null) {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "openNavigatorApp: Coordinates = $coordinates, Street = $street")

            val navigatorIntent = when {
                !coordinates.isNullOrEmpty() && !street.isNullOrEmpty() -> {
                    Logs.d(TAG, "openNavigatorApp: Opening navigator with coordinates ($coordinates) and street: $street")
                    navigatorUseCase.openNavigatorWithCoordinatesAndStreetName(coordinates, street)
                }
                !coordinates.isNullOrEmpty() -> {
                    Logs.d(TAG, "openNavigatorApp: Opening navigator with coordinates ($coordinates)")
                    navigatorUseCase.openNavigatorWithCoordinates(coordinates)
                }
                !street.isNullOrEmpty() -> {
                    Logs.d(TAG, "openNavigatorApp: Opening navigator with street: $street")
                    navigatorUseCase.openNavigatorApp(street)
                }
                else -> {
                    Logs.d(TAG, "openNavigatorApp: Opening navigator without specific parameters")
                    navigatorUseCase.openNavigatorApp()
                }
            }

            _navigatorFlow.emit(navigatorIntent)
        }
    }
}

@Composable
fun GPSTestScreen(
    viewModel: GPSTestViewModel,
    mainActivityViewModel: GPSTestMainActivityViewModel,
    onNavigateBack: () -> Unit,
    onLaunchIntent: (Intent) -> Unit,
    modifier: Modifier = Modifier
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    LaunchedEffect(Unit) {
        viewModel.checkExternalGPS()
        viewModel.initViewModel()
    }

    DisposableEffect(Unit) {
        onDispose {
            viewModel.removeHandlerCallback()
        }
    }

    LaunchedEffect(viewModel.uiEffect) {
        viewModel.uiEffect.collect { effect ->
            mainActivityViewModel.onScreenEffect(effect)
            when (effect) {
                is GPSTestUiEffect.NavigateBack -> onNavigateBack()
                is GPSTestUiEffect.LaunchIntent -> onLaunchIntent(effect.intent)
                is GPSTestUiEffect.ShowDialog -> Unit
            }
        }
    }

    if (uiState.dialogState is GPSTestDialogState.CustomDialog) {
        val dialog = uiState.dialogState as GPSTestDialogState.CustomDialog
        AlertDialog(
            onDismissRequest = { },
            title = { Text(dialog.title) },
            text = { Text(dialog.message) },
            confirmButton = {
                TextButton(onClick = { }) {
                    Text("OK")
                }
            }
        )
    }

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(ComposeColor.Black)
            .padding(WindowInsets.safeDrawing.asPaddingValues())
    ) {
        Column(
            modifier = Modifier
                .weight(1f)
                .fillMaxWidth()
                .verticalScroll(rememberScrollState())
                .padding(16.dp)
        ) {
            Text(
                text = uiState.gpsTypeText.ifEmpty { context.getString(R.string.strGPSAndroid) },
                modifier = Modifier.fillMaxWidth(),
                color = ComposeColor.White,
                style = MaterialTheme.typography.titleMedium
            )

            Spacer(Modifier.height(12.dp))

            GPSTestRow(title = context.getString(R.string.strGPSTimeUTC), value = uiState.utcTimeText)
            GPSTestRow(title = context.getString(R.string.strGPSSatelites), value = uiState.satellitesText)

            if (uiState.showPower) {
                GPSTestRow(title = context.getString(R.string.strGPSPower), value = uiState.powerText)
            }

            GPSTestRow(title = context.getString(R.string.strGPSHDOP), value = uiState.hdopText)
            GPSTestRow(title = context.getString(R.string.strGPSLat), value = uiState.latitudeText)
            GPSTestRow(title = context.getString(R.string.strGPSLon), value = uiState.longitudeText)

            if (uiState.showSpeed) {
                GPSTestRow(title = context.getString(R.string.strGPSSpeed), value = uiState.speedText)
            }

            if (uiState.showAngle) {
                GPSTestRow(title = context.getString(R.string.strGPSAngle), value = uiState.angleText)
            }

            Spacer(Modifier.height(16.dp))

            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceEvenly
            ) {
                Text(
                    text = context.getString(R.string.icon_key),
                    color = uiState.keyColor,
                    fontFamily = FontFamily.Default
                )
                Text(
                    text = context.getString(R.string.icon_alarm),
                    color = uiState.alarmColor,
                    fontFamily = FontFamily.Default
                )
            }
        }

        Row(
            modifier = Modifier
                .fillMaxWidth()
                .height(96.dp)
                .background(ComposeColor.Black)
                .padding(1.dp),
            horizontalArrangement = Arrangement.spacedBy(6.dp)
        ) {
            GPSTestCustomButton(
                state = uiState.buttons.gps,
                modifier = Modifier.weight(1f),
                onClick = { viewModel.openNavigatorApp() }
            )
            GPSTestCustomButton(
                state = uiState.buttons.accept,
                modifier = Modifier.weight(1f),
                onClick = { viewModel.onAcceptClick() }
            )
        }
    }
}

@Composable
private fun GPSTestCustomButton(
    state: GPSTestButtonVisualState,
    modifier: Modifier = Modifier,
    onClick: () -> Unit
) {
    if (!state.visible) return

    val backgroundColor = when (state.background) {
        CustomButton.BackgroundButtonColor.GREEN -> ComposeColor(0xFF2E7D32)
        CustomButton.BackgroundButtonColor.RED -> ComposeColor(0xFFC62828)
        CustomButton.BackgroundButtonColor.BLUE -> ComposeColor(0xFF1565C0)
        CustomButton.BackgroundButtonColor.YELLOW -> ComposeColor(0xFFF9A825)
        CustomButton.BackgroundButtonColor.ORANGE -> ComposeColor(0xFFEF6C00)
        CustomButton.BackgroundButtonColor.DEFAULT -> ComposeColor(0xFF1565C0)
        CustomButton.BackgroundButtonColor.BLACK -> ComposeColor.Black
        CustomButton.BackgroundButtonColor.TRANSPARENT -> ComposeColor.Transparent
    }

    Box(
        modifier = modifier
            .fillMaxHeight()
            .background(backgroundColor),
        contentAlignment = Alignment.Center
    ) {
        Button(
            onClick = onClick,
            enabled = state.enabled,
            modifier = Modifier.fillMaxSize()
        ) {
            Text(text = state.text)
        }
    }
}
```

// FIRST ANSWER
Below is a Compose-first rewrite that preserves the original navigation behavior, replaces the fragment button logic with a single state holder, and uses a `SharedFlow<WelcomeUiEffect>` for one-off events.

I’m splitting it into:

1. `WelcomeButtonsState` + `WelcomeUiState`
2. `WelcomeUiEvent` + `WelcomeUiEffect`
3. Compose-friendly `WelcomeViewModel`
4. `WelcomeScreen` composable
5. Dialog handling in Compose only when the VM exposes dialog effects
6. Notes on preserving Jetpack Views navigation

---

## 2) Events and effects

Use `UiEvent` for user actions and `SharedFlow<WelcomeUiEffect>` for one-off navigation/dialog/toast effects.

```kotlin
package ifac.td.taxi.ui.screen.welcome

import ifac.td.taxi.domain.model.itop.ITopMeterStatus

sealed interface WelcomeUiEvent {
    data object OnResume : WelcomeUiEvent
    data object OnStart : WelcomeUiEvent
    data object OnBackPressed : WelcomeUiEvent

    data object OnInicioClicked : WelcomeUiEvent
    data object OnAjustesClicked : WelcomeUiEvent
    data object OnTurnosClicked : WelcomeUiEvent
    data object OnEstadisticasClicked : WelcomeUiEvent
    data object OnConfiguracionClicked : WelcomeUiEvent
    data object OnPermisosClicked : WelcomeUiEvent
    data object OnParcialesClicked : WelcomeUiEvent
    data object OnSalirClicked : WelcomeUiEvent

    data object OnPrivacyAccepted : WelcomeUiEvent
    data object OnPrivacyRejected : WelcomeUiEvent
    data object OnBatteryOptimizationAccepted : WelcomeUiEvent
    data object OnBatteryOptimizationRejected : WelcomeUiEvent

    data class OnSecurePinConfirmed(val pin: String) : WelcomeUiEvent
    data class OnSecurePinRequested(val destination: WelcomeSecurePinDestination) : WelcomeUiEvent
    data class OnItopStateChanged(val state: ITopMeterStatus?) : WelcomeUiEvent
    data class OnCorrectLoginChanged(val success: Boolean) : WelcomeUiEvent
    data class OnDownloadingConfigChanged(val downloading: Boolean) : WelcomeUiEvent
}

sealed interface WelcomeUiEffect {
    data class Navigate(val destinationId: Int) : WelcomeUiEffect
    data class NavigateWithArgs(val direction: Any) : WelcomeUiEffect
    data class ShowToast(val messageRes: Int) : WelcomeUiEffect

    data object OpenPrivacyDialog : WelcomeUiEffect
    data object OpenExitDialog : WelcomeUiEffect
    data class OpenSecurePinDialog(val destination: WelcomeSecurePinDestination) : WelcomeUiEffect
    data object OpenBatteryOptimizationDialog : WelcomeUiEffect

    data object ConnectTaximeter : WelcomeUiEffect
    data object ExitApp : WelcomeUiEffect
}
```

## 3) Compose-friendly ViewModel

This keeps the original navigation behavior, but exposes a single `uiState` and `uiEffects`.

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import android.widget.Toast
import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.interfacom.sdk.taximeter.bravocomm.W2CLocation
import com.interfacom.sdk.taximeter.bravocomm.ifConstants
import ifac.td.taxi.R
import ifac.td.taxi.domain.model.PartialModel.Companion.hasTrips
import ifac.td.taxi.domain.model.itop.ITopMeterStatus
import ifac.td.taxi.domain.model.itop.ITopMeterStateEnum
import ifac.td.taxi.domain.usecase.BluetoothLocalUseCase
import ifac.td.taxi.domain.usecase.PartialUseCase
import ifac.td.taxi.domain.usecase.PermissionsUseCase
import ifac.td.taxi.domain.usecase.SessionUseCase
import ifac.td.taxi.domain.usecase.ShiftStatusUseCase
import ifac.td.taxi.domain.usecase.ShiftUseCase
import ifac.td.taxi.domain.usecase.StartShiftUseCase
import ifac.td.taxi.domain.usecase.UserPreferencesUseCase
import ifac.td.taxi.domain.usecase.oldv2.MigrationV2UseCase
import ifac.td.taxi.framework.sdk.bravocentral.usecase.BravoCentralConfigurationUseCase
import ifac.td.taxi.framework.sdk.bravocentral.usecase.LoginDriverUseCase
import ifac.td.taxi.framework.sdk.usecase.LicensingUseCase
import ifac.td.taxi.framework.sdk.usecase.TicketUseCase
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.ui.screen.welcome.*
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext

class WelcomeComposeViewModel(
    private val sessionUseCase: SessionUseCase,
    private val shiftUseCase: ShiftUseCase,
    private val licensingUseCase: LicensingUseCase,
    private val bravoCentralConfigurationUseCase: BravoCentralConfigurationUseCase,
    private val ticketUseCase: TicketUseCase,
    private val loginDriverUseCase: LoginDriverUseCase,
    private val startShiftUseCase: StartShiftUseCase,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val permissionsUseCase: PermissionsUseCase,
    private val partialUseCase: PartialUseCase,
    private val bluetoothLocalUseCase: BluetoothLocalUseCase,
    private val migrationV2UseCase: MigrationV2UseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    private val context: Application,
) : ViewModel() {

    private val _uiState = MutableStateFlow(WelcomeUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffects = MutableSharedFlow<WelcomeUiEffect>()
    val uiEffects = _uiEffects.asSharedFlow()

    fun onEvent(event: WelcomeUiEvent) {
        when (event) {
            WelcomeUiEvent.OnStart -> {
                checkClosuresPermission()
                makeMigrationV2()
            }
            WelcomeUiEvent.OnResume -> {
                checkUserLogIn()
                haveShifts()
                checkAllPermissions()
            }
            WelcomeUiEvent.OnBackPressed -> {
                viewModelScope.launch { _uiEffects.emit(WelcomeUiEffect.OpenExitDialog) }
            }

            WelcomeUiEvent.OnInicioClicked -> checkAdditionalPermissions()
            WelcomeUiEvent.OnAjustesClicked -> clickSettings()
            WelcomeUiEvent.OnTurnosClicked -> clickShifts()
            WelcomeUiEvent.OnEstadisticasClicked -> clickStatistics()
            WelcomeUiEvent.OnConfiguracionClicked -> clickConfiguration()
            WelcomeUiEvent.OnPermisosClicked -> clickPermissions()
            WelcomeUiEvent.OnParcialesClicked -> clickParciales()
            WelcomeUiEvent.OnSalirClicked -> viewModelScope.launch { _uiEffects.emit(WelcomeUiEffect.OpenExitDialog) }

            WelcomeUiEvent.OnPrivacyAccepted -> {
                _uiState.update { it.copy(showPrivacyDialog = false, privacyAccepted = true) }
                viewModelScope.launch { _uiEffects.emit(WelcomeUiEffect.ShowToast(R.string.toast_aceptado)) }
                setUpPermissions()
            }
            WelcomeUiEvent.OnPrivacyRejected -> {
                _uiState.update { it.copy(showPrivacyDialog = false) }
            }

            WelcomeUiEvent.OnBatteryOptimizationAccepted -> {
                _uiState.update { it.copy(batteryOptimizationAlreadyLogged = true) }
                viewModelScope.launch { _uiEffects.emit(WelcomeUiEffect.ConnectTaximeter) }
                clickMenuLogin()
            }
            WelcomeUiEvent.OnBatteryOptimizationRejected -> {
                _uiState.update { it.copy(batteryOptimizationAlreadyLogged = true) }
            }

            is WelcomeUiEvent.OnSecurePinConfirmed -> {
                // handled by dialog callback in Compose
            }

            is WelcomeUiEvent.OnSecurePinRequested -> {
                viewModelScope.launch { _uiEffects.emit(WelcomeUiEffect.OpenSecurePinDialog(event.destination)) }
            }

            is WelcomeUiEvent.OnItopStateChanged -> {
                val paired = hasITopMeterPaired()
                val disconnected = event.state?.taximeterState == null ||
                    event.state?.taximeterState == ITopMeterStateEnum.DISCONNECTED.state

                _uiState.update {
                    it.copy(
                        isItopPaired = paired,
                        lastItopDisconnected = disconnected,
                        buttons = it.buttons.copy(
                            inicio = it.buttons.inicio.copy(
                                style = if (paired && disconnected) WelcomeButtonStyle.DISABLE else WelcomeButtonStyle.ENABLE
                            )
                        )
                    )
                }
            }

            is WelcomeUiEvent.OnCorrectLoginChanged -> {
                if (!event.success) {
                    val buttonStyle = if (hasITopMeterPaired() && _uiState.value.lastItopDisconnected) {
                        WelcomeButtonStyle.DISABLE
                    } else {
                        WelcomeButtonStyle.ENABLE
                    }
                    _uiState.update {
                        it.copy(buttons = it.buttons.copy(inicio = it.buttons.inicio.copy(style = buttonStyle)))
                    }
                }
            }

            is WelcomeUiEvent.OnDownloadingConfigChanged -> {
                _uiState.update {
                    it.copy(
                        isDownloadingConfig = event.downloading,
                        buttons = it.buttons.copy(
                            inicio = it.buttons.inicio.copy(
                                style = if (event.downloading) WelcomeButtonStyle.LOADING else it.buttons.inicio.style
                            )
                        )
                    )
                }
            }
        }
    }

    private fun checkAdditionalPermissions() {
        if (!_uiState.value.batteryOptimizationAlreadyLogged) {
            checkBatteryOptimization()
        } else {
            clickMenuLogin()
        }
    }

    private fun setUpPermissions() {
        // permissions request should be handled by host/compose side if needed
    }

    private fun checkBatteryOptimization() {
        viewModelScope.launch {
            _uiEffects.emit(WelcomeUiEffect.OpenBatteryOptimizationDialog)
        }
    }

    private fun clickMenuLogin() {
        viewModelScope.launch {
            if (!sessionUseCase.isUserLoggedIn()) {
                _uiEffects.emit(WelcomeUiEffect.Navigate(R.id.action_welcomeFragment_to_loginUserFragment))
            } else {
                val conectionMode = bravoCentralConfigurationUseCase.getConnectionMode()
                var cntMethods = 0
                if ((conectionMode and 0x01) != 0) cntMethods++
                if ((conectionMode and 0x02) != 0) cntMethods++
                if ((conectionMode and 0x04) != 0) cntMethods++

                val startTurnMode = licensingUseCase.getStartTurnMode() ?: 1

                if (startTurnMode == 0 && cntMethods < 2) {
                    if ((conectionMode and 0x01) != 0) {
                        loginDriverUseCase.loginDriver("0000", "", false)
                    }
                    if ((conectionMode and 0x02) != 0) {
                        W2CLocation.setLocationAllowedByCentral(false)
                        sessionUseCase.updateWithCentralAllowed(false)
                        withContext(Dispatchers.IO) { ticketUseCase.loadTickets() }
                        startShiftUseCase.startShift()
                        shiftStatusUseCase.setStatus(ifConstants.STATE_FOR_HIRE_NO_CENTRAL, false)
                        _uiEffects.emit(WelcomeUiEffect.ConnectTaximeter)
                        _uiEffects.emit(WelcomeUiEffect.Navigate(R.id.action_welcomeFragment_to_homeFragment))
                    }
                    if ((conectionMode and 0x04) != 0) {
                        loginDriverUseCase.loginDriver("0000", "", true)
                    }
                } else {
                    _uiEffects.emit(
                        WelcomeUiEffect.Navigate(
                            WelcomeFragmentDirections.actionWelcomeFragmentToLoginDriverFragment()
                                .setConnectionMode(conectionMode)
                                .setStartTurnMode(startTurnMode)
                        )
                    )
                }
            }
        }
    }

    private fun clickShifts() {
        requestSecurePinIfNeeded(WelcomeSecurePinDestination.SHIFTS)
    }

    private fun clickStatistics() {
        requestSecurePinIfNeeded(WelcomeSecurePinDestination.STATISTICS)
    }

    private fun requestSecurePinIfNeeded(destination: WelcomeSecurePinDestination) {
        viewModelScope.launch {
            val currentSecurePin = userPreferencesUseCase.getSecurePin()
            if (!currentSecurePin.isNullOrEmpty()) {
                _uiEffects.emit(WelcomeUiEffect.OpenSecurePinDialog(destination))
            } else {
                navigateToDestination(destination)
            }
        }
    }

    fun verifyCurrentSecurePin(currentPin: String, destination: WelcomeSecurePinDestination) {
        viewModelScope.launch {
            val actualPin = userPreferencesUseCase.getSecurePin()
            if (currentPin == actualPin) {
                navigateToDestination(destination)
            } else {
                _uiEffects.emit(WelcomeUiEffect.ShowToast(R.string.pin_incorrecto))
            }
        }
    }

    private suspend fun navigateToDestination(destination: WelcomeSecurePinDestination) {
        when (destination) {
            WelcomeSecurePinDestination.SHIFTS ->
                _uiEffects.emit(WelcomeUiEffect.Navigate(R.id.action_welcomeFragment_to_shiftsFragment))
            WelcomeSecurePinDestination.STATISTICS ->
                _uiEffects.emit(WelcomeUiEffect.Navigate(R.id.action_welcomeFragment_to_statisticsFragment))
        }
    }

    private fun clickSettings() {
        viewModelScope.launch { _uiEffects.emit(WelcomeUiEffect.Navigate(R.id.action_welcomeFragment_to_settingsFragment)) }
    }

    private fun haveShifts() {
        viewModelScope.launch {
            val shifts = shiftUseCase.getLastShift()
            _uiState.update {
                it.copy(
                    hasShifts = shifts != null,
                    buttons = it.buttons.copy(
                        turnos = it.buttons.turnos.copy(
                            style = if (shifts != null) WelcomeButtonStyle.ENABLE else WelcomeButtonStyle.DISABLE
                        )
                    )
                )
            }
        }
    }

    private fun clickConfiguration() {
        viewModelScope.launch { _uiEffects.emit(WelcomeUiEffect.Navigate(R.id.action_welcomeFragment_to_loginUserFragment)) }
    }

    private fun checkUserLogIn() {
        viewModelScope.launch(Dispatchers.IO) {
            val loggedIn = sessionUseCase.isUserLoggedIn()
            _uiState.update {
                it.copy(
                    isLoggedIn = loggedIn,
                    buttons = it.buttons.copy(
                        parciales = it.buttons.parciales.copy(
                            style = if (loggedIn) WelcomeButtonStyle.ENABLE else WelcomeButtonStyle.DISABLE
                        )
                    )
                )
            }
        }
    }

    private fun clickParciales() {
        viewModelScope.launch {
            val activePartial = partialUseCase.getActivePartial()
            if (activePartial != null && activePartial.hasTrips()) {
                _uiEffects.emit(WelcomeUiEffect.Navigate(R.id.action_welcomeFragment_to_partials))
            } else {
                _uiEffects.emit(WelcomeUiEffect.Navigate(R.id.action_welcomeFragment_to_closed_partial))
            }
        }
    }

    private fun clickPermissions() {
        viewModelScope.launch { _uiEffects.emit(WelcomeUiEffect.Navigate(R.id.action_welcomeFragment_to_permissionsFragment)) }
    }

    private fun checkClosuresPermission() {
        viewModelScope.launch {
            val parameters = licensingUseCase.getLicensingParameters()
            val enabled = parameters?.isClosuresButton == true
            _uiState.update {
                it.copy(
                    hasClosuresPermission = enabled,
                    buttons = it.buttons.copy(
                        parciales = it.buttons.parciales.copy(
                            visible = enabled,
                            style = if (enabled) WelcomeButtonStyle.ENABLE else WelcomeButtonStyle.DISABLE
                        )
                    )
                )
            }
        }
    }

    private fun checkAllPermissions() {
        viewModelScope.launch(Dispatchers.IO) {
            val hasAllPermissions = permissionsUseCase.hasAllPermissions()
            _uiState.update {
                it.copy(
                    hasAllPermissions = hasAllPermissions,
                    buttons = it.buttons.copy(
                        permisos = it.buttons.permisos.copy(
                            background = if (hasAllPermissions) WelcomeButtonBackground.BLUE else WelcomeButtonBackground.ORANGE
                        )
                    )
                )
            }
        }
    }

    private fun hasITopMeterPaired(): Boolean = bluetoothLocalUseCase.isCurrentBluetoothItop()

    private fun makeMigrationV2() {
        viewModelScope.launch {
            val hasOldTrips = migrationV2UseCase.checkHasOldTrip()
            val hasOldShift = migrationV2UseCase.checkHasOldShift()
            if (hasOldShift || hasOldTrips) {
                val checkExistSession = sessionUseCase.getSession()
                if (checkExistSession == null) {
                    if (migrationV2UseCase.migrateSession()) {
                        Toast.makeText(context, "Start migration", Toast.LENGTH_LONG).show()
                        _uiEffects.emit(
                            WelcomeUiEffect.Navigate(
                                WelcomeFragmentDirections.actionWelcomeFragmentToLoginUserFragment()
                                    .setAutoDownloadConfigurationFromMigration(true)
                            )
                        )
                    }
                }
            }
        }
    }
}
```

```kotlin
package ifac.td.taxi.ui.screen.welcome

import android.content.Intent
import android.os.Build
import android.os.PowerManager

@Composable
fun WelcomeRoute(
    navController: NavController,
    viewModel: WelcomeComposeViewModel,
    onShowHeader: (Boolean) -> Unit = {},
    onShowTopBarExit: (Boolean) -> Unit = {},
    onConnectTaximeter: () -> Unit = {},
    onExitApp: () -> Unit = {},
    onShowToast: (Int) -> Unit = {},
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    LaunchedEffect(Unit) {
        viewModel.onEvent(WelcomeUiEvent.OnStart)
        viewModel.uiEffects.collectLatest { effect ->
            when (effect) {
                is WelcomeUiEffect.Navigate -> navController.navigate(effect.destinationId)
                is WelcomeUiEffect.NavigateWithArgs -> {
                    // If you use generated directions, handle them here with your NavController
                }
                is WelcomeUiEffect.ShowToast -> onShowToast(effect.messageRes)
                WelcomeUiEffect.OpenPrivacyDialog -> Unit
                WelcomeUiEffect.OpenExitDialog -> Unit
                is WelcomeUiEffect.OpenSecurePinDialog -> Unit
                WelcomeUiEffect.OpenBatteryOptimizationDialog -> Unit
                WelcomeUiEffect.ConnectTaximeter -> onConnectTaximeter()
                WelcomeUiEffect.ExitApp -> onExitApp()
            }
        }
    }

    LaunchedEffect(Unit) {
        onShowHeader(true)
        onShowTopBarExit(true)
    }

    WelcomeScreen(
        uiState = uiState,
        onEvent = viewModel::onEvent,
        onSecurePinConfirm = { pin, destination ->
            viewModel.verifyCurrentSecurePin(pin, destination)
        },
        onExitConfirm = onExitApp,
        onOpenPrivacyLink = {
            val privacyPolicyUri = context.getString(R.string.privacy_policy_link)
            val intent = Intent(Intent.ACTION_VIEW, privacyPolicyUri.toUri())
            runCatching { context.startActivity(intent) }
        },
        onRequestBatteryOptimizationSettings = {
            val intent = Intent(Settings.ACTION_IGNORE_BATTERY_OPTIMIZATION_SETTINGS)
            runCatching { context.startActivity(intent) }
        }
    )
}

@Composable
private fun WelcomeButton(
    text: String,
    state: WelcomeButtonUiState,
    onClick: () -> Unit,
) {
    if (!state.visible) return

    val enabled = state.style != WelcomeButtonStyle.DISABLE && state.style != WelcomeButtonStyle.LOADING
    val background = when (state.background) {
        WelcomeButtonBackground.BLUE -> MaterialTheme.colorScheme.primary
        WelcomeButtonBackground.ORANGE -> MaterialTheme.colorScheme.tertiary
    }

    Button(
        onClick = onClick,
        enabled = enabled,
        colors = ButtonDefaults.buttonColors(containerColor = background),
        modifier = Modifier.fillMaxWidth()
    ) {
        if (state.style == WelcomeButtonStyle.LOADING) {
            CircularProgressIndicator(modifier = Modifier.size(18.dp), strokeWidth = 2.dp)
            Spacer(Modifier.width(8.dp))
        }
        Text(text)
    }
}
```

## 5) Dialog handling with lifecycle collection

If you want to keep dialog handling fully in Compose, collect `uiEffects` in a `LaunchedEffect` and set local dialog state.

Example for secure pin:

```kotlin
@Composable
fun WelcomeDialogHost(
    uiEffects: kotlinx.coroutines.flow.Flow<WelcomeUiEffect>,
    onSecurePinConfirm: (String, WelcomeSecurePinDestination) -> Unit,
    onExitConfirm: () -> Unit,
    onBatteryOptimizationConfirm: () -> Unit,
) {
    var securePinDestination by remember { mutableStateOf<WelcomeSecurePinDestination?>(null) }
    var showExitDialog by remember { mutableStateOf(false) }
    var showBatteryDialog by remember { mutableStateOf(false) }

    LaunchedEffect(Unit) {
        uiEffects.collectLatest { effect ->
            when (effect) {
                is WelcomeUiEffect.OpenSecurePinDialog -> securePinDestination = effect.destination
                WelcomeUiEffect.OpenExitDialog -> showExitDialog = true
                WelcomeUiEffect.OpenBatteryOptimizationDialog -> showBatteryDialog = true
                else -> Unit
            }
        }
    }

    if (securePinDestination != null) {
        // Replace with your own Compose dialog
        AlertDialog(
            onDismissRequest = { securePinDestination = null },
            title = { Text("PIN") },
            text = { Text("Ingrese su PIN actual") },
            confirmButton = {
                TextButton(onClick = {
                    onSecurePinConfirm("1234", securePinDestination!!)
                    securePinDestination = null
```

```kotlin
package ifac.td.taxi.ui.screen.welcome

import ifac.td.taxi.domain.model.itop.ITopMeterStateEnum
import ifac.td.taxi.domain.model.itop.ITopMeterStatus

fun reduceWelcomeButtonsState(
    current: WelcomeButtonsState,
    input: WelcomeButtonsInput,
): WelcomeButtonsState {
    var state = current

    // --- vModel.hasShiftsFlow.collect { ... }
    input.hasShifts?.let { hasShifts ->
        state = state.copy(
            turnos = state.turnos.copy(
                style = if (hasShifts) WelcomeButtonStyle.ENABLE else WelcomeButtonStyle.DISABLE
            )
        )
    }

    // --- vModel.partialButtonFlow.collect { ... }
    input.partialButtonEnabled?.let { enabled ->
        state = state.copy(
            parciales = state.parciales.copy(
                style = if (enabled) WelcomeButtonStyle.ENABLE else WelcomeButtonStyle.DISABLE
            )
        )
    }

    // --- vModel.userLogInFlow.collect { ... }
    input.userLoggedIn?.let { loggedIn ->
        state = if (loggedIn) {
            state.copy(
                parciales = state.parciales.copy(
                    style = WelcomeButtonStyle.ENABLE
                )
            )
        } else {
            // setButtonsUserLoggedOff()
            state.copy(
                inicio = state.inicio.copy(
                    // action changes in fragment; in Compose we keep state and let UI decide click behavior
                    style = state.inicio.style
                ),
                turnos = state.turnos.copy(style = WelcomeButtonStyle.ENABLE),
                estadisticas = state.estadisticas.copy(style = WelcomeButtonStyle.ENABLE),
                parciales = state.parciales.copy(style = WelcomeButtonStyle.DISABLE),
            )
        }
    }

    // --- sharedViewModel.downloadingBravoConfigurationFlow.collect { ... }
    input.downloadingConfig?.let { downloading ->
        state = state.copy(
            inicio = state.inicio.copy(
                style = if (downloading) WelcomeButtonStyle.LOADING else state.inicio.style
            )
        )
    }

    // --- vModel.hasAllPermissionsFlow.collect { ... }
    input.hasAllPermissions?.let { hasAllPermissions ->
        state = state.copy(
            permisos = state.permisos.copy(
                background = if (hasAllPermissions) {
                    WelcomeButtonBackground.BLUE
                } else {
                    WelcomeButtonBackground.ORANGE
                }
            )
        )
    }

    // --- sharedViewModel.lastITopStateReceived.collect { ... }
    input.lastItopState?.let { itopState ->
        if (input.isItopPaired) {
            state = when (itopState.taximeterState) {
                null,
                ITopMeterStateEnum.DISCONNECTED.state -> {
                    state.copy(
                        inicio = state.inicio.copy(style = WelcomeButtonStyle.DISABLE)
                    )
                }
                else -> {
                    state.copy(
                        inicio = state.inicio.copy(style = WelcomeButtonStyle.ENABLE)
                    )
                }
            }
        }
    }

    // --- sharedViewModel.correctLoginFlow.collect { ... }
    input.correctLoginSuccess?.let { success ->
        if (!success) {
            val shouldDisable = input.isItopPaired &&
                (input.lastItopState?.taximeterState == ITopMeterStateEnum.DISCONNECTED.state ||
                 input.lastItopState?.taximeterState == null)

            state = state.copy(
                inicio = state.inicio.copy(
                    style = if (shouldDisable) WelcomeButtonStyle.DISABLE else WelcomeButtonStyle.ENABLE
                )
            )
        }
    }

    // --- vModel.neddMigrationFlow.collect { ... }
    // Fragment does nothing here, so reducer intentionally no-op.

    return state
}
```

```kotlin
        viewModel.uiEffects.collectLatest { effect ->
            when (effect) {
                is WelcomeUiEffect.Navigate -> navController.navigate(effect.destinationId)
                is WelcomeUiEffect.NavigateWithArgs -> {

                WelcomeUiEffect.OpenBatteryOptimizationDialog -> showBatteryDialog = true

                WelcomeUiEffect.OpenPrivacyDialog -> Unit
            }
```

```kotlin
@Composable
fun DispatchReceivedScreen(
    navController: NavController,
    viewModel: DispatchReceivedViewModel,
    dispatchId: String,
    onShowHeader: (Boolean) -> Unit,
    onShowBottomBar: (Boolean) -> Unit,
    onShowToast: (Int) -> Unit,
    onRequestPhoneCallPermission: () -> Unit,
    onCloseDialog: (String) -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()

    var showRejectDialog by remember { mutableStateOf(false) }
    var showPhonePermissionDialog by remember { mutableStateOf(false) }

    LaunchedEffect(Unit) {
        onShowHeader(false)
        onShowBottomBar(false)
        viewModel.getDispatch(dispatchId)
        viewModel.checkCancelDialogPermission()
    }

    LaunchedEffect(viewModel) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                is DispatchReceivedUiEffect.NavigateToInfoDispatch -> {
                    navController.navigate("infoDispatch/${effect.tripId}")
                }
                DispatchReceivedUiEffect.NavigateBack -> navController.popBackStack()
                DispatchReceivedUiEffect.ShowRejectDialog -> showRejectDialog = true
                DispatchReceivedUiEffect.ShowPhonePermissionDialog -> showPhonePermissionDialog = true
                is DispatchReceivedUiEffect.ShowToast -> onShowToast(effect.resId)
            }
        }
    }

    if (showRejectDialog) {
        AlertDialog(
            onDismissRequest = { showRejectDialog = false },
            title = { Text("Warning") },
            text = { Text("¿Quieres rechazar este despacho?") },
            confirmButton = {
                TextButton(onClick = {
                    showRejectDialog = false
                    viewModel.onConfirmReject()
                }) { Text("Accept") }
            },
            dismissButton = {
                TextButton(onClick = { showRejectDialog = false }) { Text("Cancel") }
            }
        )
    }

    if (showPhonePermissionDialog) {
        AlertDialog(
            onDismissRequest = { showPhonePermissionDialog = false },
            title = { Text("Warning") },
            text = { Text("Phone permission required") },
            confirmButton = {
                TextButton(onClick = {
                    showPhonePermissionDialog = false
                    onRequestPhoneCallPermission()
                }) { Text("Accept") }
            },
            dismissButton = {
                TextButton(onClick = { showPhonePermissionDialog = false }) { Text("Cancel") }
            }
        )
    }

    Column(
        modifier = Modifier
            .fillMaxSize()
            .padding(16.dp),
        verticalArrangement = Arrangement.spacedBy(12.dp)
    ) {
        Text(text = uiState.dispatch?.pickUpTime.orEmpty())
        Text(text = uiState.dispatch?.pickUpZone.orEmpty())
        Text(text = uiState.dispatch?.smallPickUpAdress.orEmpty())
        Text(text = uiState.dispatch?.city.orEmpty())
        Text(text = uiState.dispatch?.passengerName.orEmpty())

        Spacer(modifier = Modifier.height(16.dp))

        Button(
            onClick = {
                viewModel.acceptDispatch()
            },
            modifier = Modifier.fillMaxWidth()
        ) {
            Text("Accept")
        }

        Button(
            onClick = {
                viewModel.onCancelClicked()
            },
            modifier = Modifier.fillMaxWidth()
        ) {
            Text("Cancel")
        }
    }
}
```

```kotlin
fun reduceWelcomeButtonsState(
    current: WelcomeButtonsState,
    input: WelcomeButtonsInput,
): WelcomeButtonsState {
    var state = current

    input.hasShifts?.let { hasShifts ->
        state = state.copy(
            turnos = state.turnos.copy(
                style = if (hasShifts) WelcomeButtonStyle.ENABLE else WelcomeButtonStyle.DISABLE
            )
        )
    }

    input.partialButtonEnabled?.let { enabled ->
        state = state.copy(
            parciales = state.parciales.copy(
                style = if (enabled) WelcomeButtonStyle.ENABLE else WelcomeButtonStyle.DISABLE
            )
        )
    }

    input.userLoggedIn?.let { loggedIn ->
        if (!loggedIn) {
            state = state.copy(
                parciales = state.parciales.copy(style = WelcomeButtonStyle.DISABLE)
            )
        } else {
            state = state.copy(
                parciales = state.parciales.copy(style = WelcomeButtonStyle.ENABLE)
            )
        }
    }

    input.downloadingConfig?.let { downloading ->
        state = state.copy(
            inicio = state.inicio.copy(
                style = if (downloading) WelcomeButtonStyle.LOADING else state.inicio.style
            )
        )
    }

    input.hasAllPermissions?.let { hasAllPermissions ->
        state = state.copy(
            permisos = state.permisos.copy(
                background = if (hasAllPermissions) WelcomeButtonBackground.BLUE else WelcomeButtonBackground.ORANGE
            )
        )
    }

    input.lastItopState?.let { itopState ->
        if (input.isItopPaired) {
            state = when (itopState.taximeterState) {
                null, ITopMeterStateEnum.DISCONNECTED.state -> {
                    state.copy(inicio = state.inicio.copy(style = WelcomeButtonStyle.DISABLE))
                }
                else -> {
                    state.copy(inicio = state.inicio.copy(style = WelcomeButtonStyle.ENABLE))
                }
            }
        }
    }

    input.correctLoginSuccess?.let { success ->
        if (!success) {
            val shouldDisable = input.isItopPaired &&
                (input.lastItopState?.taximeterState == null ||
                 input.lastItopState?.taximeterState == ITopMeterStateEnum.DISCONNECTED.state)

            state = state.copy(
                inicio = state.inicio.copy(
                    style = if (shouldDisable) WelcomeButtonStyle.DISABLE else WelcomeButtonStyle.ENABLE
                )
            )
        }
    }

    return state
}

            clickMenuLogin()
        }
    }

    fun onBatteryOptimizationAccepted() {
        viewModelScope.launch {
            _uiEffects.emit(WelcomeUiEffect.ConnectTaximeter)
            clickMenuLogin()
        }
    }

    fun onExitConfirmed() {
        viewModelScope.launch {
            _uiEffects.emit(WelcomeUiEffect.ShowExitDialog)
        }
    }

    private fun checkAdditionalPermissions() {
        if (!_uiState.value.batteryOptimizationAlreadyLogged) {
            viewModelScope.launch { _uiEffects.emit(WelcomeUiEffect.ShowBatteryOptimizationDialog) }
        } else {
            clickMenuLogin()
        }
    }

    private fun requestSecurePinIfNeeded(destination: WelcomeSecurePinDestination) {
        viewModelScope.launch {
            val currentSecurePin = userPreferencesUseCase.getSecurePin()
            if (!currentSecurePin.isNullOrEmpty()) {
                _uiEffects.emit(WelcomeUiEffect.ShowSecurePinDialog(destination))
            } else {
                navigateToDestination(destination)
            }
        }
    }

    private suspend fun navigateToDestination(destination: WelcomeSecurePinDestination) {
        when (destination) {
            WelcomeSecurePinDestination.SHIFTS ->
                _uiEffects.emit(
                    WelcomeUiEffect.Navigate(
                        WelcomeFragmentDirections.actionWelcomeFragmentToShiftsFragment()
                    )
                )
            WelcomeSecurePinDestination.STATISTICS ->
                _uiEffects.emit(
                    WelcomeUiEffect.Navigate(
                        WelcomeFragmentDirections.actionWelcomeFragmentToStatisticsFragment()
                    )
                )
        }
    }

    private fun checkClosuresPermission() {
        viewModelScope.launch {
            val parameters = licensingUseCase.getLicensingParameters()
            val enabled = parameters?.isClosuresButton == true
            _uiState.update {
                it.copy(
                    hasClosuresPermission = enabled,
                    buttons = it.buttons.copy(
                        parciales = it.buttons.parciales.copy(
                            visible = enabled,
                            style = if (enabled) WelcomeButtonStyle.ENABLE else WelcomeButtonStyle.DISABLE
                        )
                    )
                )
            }
        }
    }

    private fun checkUserLogIn() {
        viewModelScope.launch(Dispatchers.IO) {
            val loggedIn = sessionUseCase.isUserLoggedIn()
            _uiState.update {
                it.copy(
                    isLoggedIn = loggedIn,
                    buttons = it.buttons.copy(
                        parciales = it.buttons.parciales.copy(
                            style = if (loggedIn) WelcomeButtonStyle.ENABLE else WelcomeButtonStyle.DISABLE
                        )
                    )
                )
            }
        }
    }

    private fun haveShifts() {
        viewModelScope.launch {
            val shifts = shiftUseCase.getLastShift()
            _uiState.update {
                it.copy(
                    hasShifts = shifts != null,
                    buttons = it.buttons.copy(
                        turnos = it.buttons.turnos.copy(
                            style = if (shifts != null) WelcomeButtonStyle.ENABLE else WelcomeButtonStyle.DISABLE
                        )
                    )
                )
            }
        }
    }

    private fun checkAllPermissions() {
        viewModelScope.launch(Dispatchers.IO) {
            val hasAllPermissions = permissionsUseCase.hasAllPermissions()
            _uiState.update {
                it.copy(
                    hasAllPermissions = hasAllPermissions,
                    buttons = it.buttons.copy(
                        permisos = it.buttons.permisos.copy(
                            background = if (hasAllPermissions) WelcomeButtonBackground.BLUE else WelcomeButtonBackground.ORANGE
                        )
                    )
                )
            }
        }
    }

    private fun hasITopMeterPaired(): Boolean = bluetoothLocalUseCase.isCurrentBluetoothItop()

    private fun clickMenuLogin() {
        viewModelScope.launch {
            if (!sessionUseCase.isUserLoggedIn()) {
                _uiEffects.emit(
                    WelcomeUiEffect.Navigate(
                        WelcomeFragmentDirections.actionWelcomeFragmentToLoginUserFragment()
                    )
                )
            } else {
                val conectionMode = bravoCentralConfigurationUseCase.getConnectionMode()
                var cntMethods = 0
                if ((conectionMode and 0x01) != 0) cntMethods++
                if ((conectionMode and 0x02) != 0) cntMethods++
                if ((conectionMode and 0x04) != 0) cntMethods++

                val startTurnMode = licensingUseCase.getStartTurnMode() ?: 1

                if (startTurnMode == 0 && cntMethods < 2) {
                    if ((conectionMode and 0x01) != 0) {
                        loginDriverUseCase.loginDriver("0000", "", false)
                    }
                    if ((conectionMode and 0x02) != 0) {
                        startShiftUseCase.startShift()
                        shiftStatusUseCase.setStatus("STATE_FOR_HIRE_NO_CENTRAL", false)
                        withContext(Dispatchers.IO) { ticketUseCase.loadTickets() }
                        _uiEffects.emit(WelcomeUiEffect.ConnectTaximeter)
                        _uiEffects.emit(
                            WelcomeUiEffect.Navigate(
                                WelcomeFragmentDirections.actionWelcomeFragmentToHomeFragment()
                            )
                        )
                    }
                    if ((conectionMode and 0x04) != 0) {
                        loginDriverUseCase.loginDriver("0000", "", true)
                    }
                } else {
                    _uiEffects.emit(
                        WelcomeUiEffect.Navigate(
                            WelcomeFragmentDirections.actionWelcomeFragmentToLoginDriverFragment()
                                .setConnectionMode(conectionMode)
                                .setStartTurnMode(startTurnMode)
                        )
                    )
                }
            }
        }
    }

    private fun makeMigrationV2() {
        viewModelScope.launch {
            val hasOldTrips = migrationV2UseCase.checkHasOldTrip()
            val hasOldShift = migrationV2UseCase.checkHasOldShift()
            if (hasOldShift || hasOldTrips) {
                val checkExistSession = sessionUseCase.getSession()
                if (checkExistSession == null) {
                    if (migrationV2UseCase.migrateSession()) {
                        _uiEffects.emit(
                            WelcomeUiEffect.Navigate(
                                WelcomeFragmentDirections.actionWelcomeFragmentToLoginUserFragment()
                                    .setAutoDownloadConfigurationFromMigration(true)
                            )
                        )
                    }
                }
            }
        }
    }

@Composable
fun WelcomeRoute(
    navController: NavController,
    viewModel: WelcomeComposeViewModel,
    sharedViewModel: MainActivityViewModel,
    onShowHeader: (Boolean) -> Unit,
    onShowBottomBar: (Boolean) -> Unit,
    onShowToast: (Int) -> Unit,
    onExitApp: () -> Unit,
    onRequestBatteryOptimizationSettings: () -> Unit,
    onConnectTaximeter: () -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val downloadingConfig by sharedViewModel.progressBarFlow.collectAsStateWithLifecycle(initialValue = Pair(0, 0))
    val correctLogin by sharedViewModel.correctLoginFlow.collectAsStateWithLifecycle(initialValue = Pair(false, false))
    val lastItopState by sharedViewModel.lastITopStateReceived.collectAsStateWithLifecycle(initialValue = null)
    val reconnection by sharedViewModel.reconnectionFlow.collectAsStateWithLifecycle(initialValue = false)

    var securePinDestination by remember { mutableStateOf<WelcomeSecurePinDestination?>(null) }
    var showExitDialog by remember { mutableStateOf(false) }
    var showBatteryDialog by remember { mutableStateOf(false) }

    LaunchedEffect(Unit) {
        onShowHeader(true)
        onShowBottomBar(true)
        viewModel.onStart()
        viewModel.onResume()
    }

    LaunchedEffect(downloadingConfig, correctLogin, lastItopState, reconnection) {
        // reducer bridge for shared activity state
    }

    LaunchedEffect(viewModel) {
        viewModel.uiEffects.collectLatest { effect ->
            when (effect) {
                is WelcomeUiEffect.Navigate -> navController.navigate(effect.directions)
                WelcomeUiEffect.ShowExitDialog -> showExitDialog = true
                WelcomeUiEffect.ShowBatteryOptimizationDialog -> showBatteryDialog = true
                is WelcomeUiEffect.ShowSecurePinDialog -> securePinDestination = effect.destination
                is WelcomeUiEffect.ShowToast -> onShowToast(effect.resId)
                WelcomeUiEffect.ConnectTaximeter -> onConnectTaximeter()
            }
        }
    }

    WelcomeScreen(
        uiState = uiState,
        onInicio = { viewModel.onInicioClicked() },
        onAjustes = { navController.navigate(WelcomeFragmentDirections.actionWelcomeFragmentToSettingsFragment()) },
        onTurnos = { viewModel.onTurnosClicked() },
        onEstadisticas = { viewModel.onEstadisticasClicked() },
        onConfiguracion = { viewModel.onConfiguracionClicked() },
        onPermisos = { viewModel.onPermisosClicked() },
        onParciales = { viewModel.onParcialesClicked() },
        onSalir = { viewModel.onSalirClicked() },
        securePinDestination = securePinDestination,
        onSecurePinConfirm = { pin, destination ->
            securePinDestination = null
            viewModel.onSecurePinConfirmed(pin, destination)
        },
        showExitDialog = showExitDialog,
        onDismissExitDialog = { showExitDialog = false },
        onExitConfirm = {
            showExitDialog = false
            onExitApp()
        },
        showBatteryDialog = showBatteryDialog,
        onDismissBatteryDialog = { showBatteryDialog = false },
        onBatteryConfirm = {
            showBatteryDialog = false
            onRequestBatteryOptimizationSettings()
            viewModel.onBatteryOptimizationAccepted()
        }
    )
}

@Composable
private fun WelcomeButton(
    text: String,
    state: WelcomeButtonState,
    onClick: () -> Unit,
) {
    if (!state.visible) return

    val enabled = state.style != WelcomeButtonStyle.DISABLE && state.style != WelcomeButtonStyle.LOADING
    val containerColor = when (state.background) {
        WelcomeButtonBackground.BLUE -> MaterialTheme.colorScheme.primary
        WelcomeButtonBackground.ORANGE -> MaterialTheme.colorScheme.tertiary
    }

    Button(
        onClick = onClick,
        enabled = enabled,
        colors = ButtonDefaults.buttonColors(containerColor = containerColor),
        modifier = Modifier.fillMaxWidth()
    ) {
        if (state.style == WelcomeButtonStyle.LOADING) {
            CircularProgressIndicator(modifier = Modifier.height(18.dp))
            Spacer(Modifier.height(8.dp))
        }
        Text(text)
    }
}
```

## Notes on what was changed

### 1) Real `MainActivityViewModel`
I removed the placeholder shared VM and used your actual `MainActivityViewModel` directly in the route.

### 2) Safe Args / `NavDirections`
All navigation is now emitted as `WelcomeUiEffect.Navigate(NavDirections)` and handled with:

```kotlin
navController.navigate(effect.directions)
```

This is the correct Compose-friendly way to preserve your existing Safe Args graph.

### 3) Custom button API
Your fragment used:

- `setAction { }`
- `setButtonStyle(...)`
- `changeBackground(...)`

In Compose, the equivalent is the `WelcomeButtonState` model:
- `style`
- `background`
- `visible`

That is the closest compile-safe replacement for your custom button API.

Your `MainActivityViewModel` exposes:

```kotlin
val progressBarFlow: SharedFlow<Pair<Int, Int>>

val lastITopStateReceived: StateFlow<String?>
```

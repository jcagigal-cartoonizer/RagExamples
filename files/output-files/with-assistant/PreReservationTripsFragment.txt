```kotlin
private fun PreReservationTripsActionButton(
    state: PreReservationTripsButtonsState,
    onClick: () -> Unit

    // Here I use a pure Compose button with the same visual state mapping.
```

```kotlin
class PreReservationTripsViewModel(
    private val preReservationUseCase: PreReservationUseCase,
    private val bravoCentralUseCase: BravoCentralUseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    context: Application,
) : BaseViewModel(context) {

    private val TAG = "PreReservationTripsViewModel"

    private val _uiState = MutableStateFlow(PreReservationTripsUiState())
    val uiState: StateFlow<PreReservationTripsUiState> = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<PreReservationTripsUiEffect>()
    val effects: SharedFlow<PreReservationTripsUiEffect> = _effects.asSharedFlow()

    private val _preReservationTripsListFlow = MutableStateFlow<List<Prereservation>?>(null)
    private val _bidFlow = MutableSharedFlow<Pair<Boolean, Boolean>>()
    private val _errorFlow = MutableSharedFlow<Boolean>()

    private lateinit var runnable: Runnable
    private val handler = Handler(Looper.getMainLooper())

    init {
        val refreshRate = 30
        val millis = (refreshRate * 1000L)

        runnable = object : Runnable {
            override fun run() {
                viewModelScope.launch {
                    preReservationUseCase.requestPreReservations(_preReservationTripsListFlow, _errorFlow)
                }
                handler.postDelayed(this, millis)
            }
        }
        handler.post(runnable)

        viewModelScope.launch {
            _preReservationTripsListFlow.collect { list ->
                _uiState.update { it.copy(items = list.orEmpty()) }
            }
        }

        viewModelScope.launch {
            _bidFlow.collect { (wasCompleted, cancelPreReservation) ->
                Logs.d(
                    TAG,
                    "bidFlow called from PreReservationTripsViewModel wasCompleted: $wasCompleted, requestType: ${
                        if (!cancelPreReservation) "PreReserve" else "Cancel PreReservation"
                    }"
                )

                if (wasCompleted) {
                    preReservationUseCase.requestPreReservations(_preReservationTripsListFlow, _errorFlow)

                    val dialogModel = if (!cancelPreReservation) {
                        CustomDialog.CustomDialogModel(
                            title = getString(R.string.prereservation),
                            description = getString(R.string.prereservation_dialog_assigned),
                            buttons = arrayListOf(ButtonType.ACCEPT)
                        )
                    } else {
                        CustomDialog.CustomDialogModel(
                            title = getString(R.string.prereservation),
                            description = getString(R.string.prereservation_dialog_unassigned),
                            buttons = arrayListOf(ButtonType.ACCEPT)
                        )
                    }

                    _effects.emit(
                        PreReservationTripsUiEffect.OpenDialog(
                            dialogModel = dialogModel,
                            onResult = { }
                        )
                    )
                } else {
                    Logs.d(TAG, "Error from bidPreReservation()")
                }
            }
        }
    }

    fun removeHandlerCallback() {
        handler.removeCallbacks(runnable)
    }

    fun onEvent(event: PreReservationTripsUiEvent) {
        when (event) {
            is PreReservationTripsUiEvent.TripClicked -> handleTripClick(event.trip)
            PreReservationTripsUiEvent.DismissDialog -> _uiState.update { it.copy(dialogState = null) }
            PreReservationTripsUiEvent.DialogAccept -> _uiState.update { it.copy(dialogState = null) }
            PreReservationTripsUiEvent.DialogCancel -> _uiState.update { it.copy(dialogState = null) }
        }
    }

    private fun handleTripClick(trip: Prereservation) {
        if (trip.isAssignedToMyCar) {
            if (trip.isTooLateToCancelAssignment) {
                viewModelScope.launch {
                    _effects.emit(PreReservationTripsUiEffect.ShowToast(R.string.can_not_cancel_reservation))
                }
            } else {
                openCancelPreReserveTripDialog(trip)
            }
        } else if (trip.isAvailable) {
            openPreReserveTripDialog(trip)
        }
    }

    private fun openPreReserveTripDialog(preReservation: Prereservation) {
        _uiState.update {
            it.copy(
                dialogState = PreReservationTripsDialogState(
                    title = getString(R.string.dialog_asignar_trip_title),
                    description = getString(
                        R.string.pickup_time_in_zone_placeholder_str,
                        preReservation.pickupTime.getDateTimeFromISO8601(),
                        preReservation.pickupZone
                    ),
                    showCancel = true,
                    preReservation = preReservation,
                    cancelAssignment = false
                )
            )
        }
    }

    private fun openCancelPreReserveTripDialog(preReservation: Prereservation) {
        _uiState.update {
            it.copy(
                dialogState = PreReservationTripsDialogState(
                    title = getString(R.string.dialog_desasignar_trip_title),
                    description = getString(
                        R.string.pickup_time_in_zone_placeholder_str,
                        preReservation.pickupTime.getDateTimeFromISO8601(),
                        preReservation.pickupZone
                    ),
                    showCancel = true,
                    preReservation = preReservation,
                    cancelAssignment = true
                )
            )
        }
    }

    fun confirmDialog() {
        val dialog = _uiState.value.dialogState ?: return
        val trip = dialog.preReservation ?: return
        preReserveTrip(trip, dialog.cancelAssignment)
        _uiState.update { it.copy(dialogState = null) }
    }

    fun requestTrip(trip: PendingTrip) {
        bravoCentralUseCase.requestPendingTrip(trip)
    }

    fun preReserveTrip(it: Prereservation, remove: Boolean) {
        viewModelScope.launch {
            preReservationUseCase.bidPreReservation(it, remove, _bidFlow)
        }
    }
}
```

### Branch mapping to the fragment:
- `if (it.isAssignedToMyCar)` → button visible, cancel action
- `if (it.isTooLateToCancelAssignment)` → disabled button + toast path

- `else if (it.isAvailable)` → enabled accept button
- otherwise hidden

This is the Compose bridge that replaces the fragment’s `sharedViewModel` usage and keeps navigation/dialog integration.

```kotlin
package ifac.td.taxi.viewmodel

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.ui.custom.dialog.CustomDialog
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.launch

sealed interface PreReservationTripsMainActivityUiEffect {
    data class OpenDialog(
        val dialogModel: CustomDialog.CustomDialogModel,
        val onResult: (CustomDialog.CustomDialogResponse) -> Unit
    ) : PreReservationTripsMainActivityUiEffect

    data object NavigateBack : PreReservationTripsMainActivityUiEffect
    data class ShowHeader(val visible: Boolean) : PreReservationTripsMainActivityUiEffect
}

class PreReservationTripsMainActivityViewModel : ViewModel() {

    private val _effects = MutableSharedFlow<PreReservationTripsMainActivityUiEffect>()
    val effects: SharedFlow<PreReservationTripsMainActivityUiEffect> = _effects.asSharedFlow()

    fun openDialog(
        dialogModel: CustomDialog.CustomDialogModel,
        onResult: (CustomDialog.CustomDialogResponse) -> Unit = {}
    ) {
        viewModelScope.launch {
            _effects.emit(
                PreReservationTripsMainActivityUiEffect.OpenDialog(
                    dialogModel = dialogModel,
                    onResult = onResult
                )
            )
        }
    }

    fun navigateBack() {
        viewModelScope.launch {
            _effects.emit(PreReservationTripsMainActivityUiEffect.NavigateBack)
        }
    }

    fun showHeader(visible: Boolean) {
        viewModelScope.launch {
            _effects.emit(PreReservationTripsMainActivityUiEffect.ShowHeader(visible))
        }
    }
}
```

```kotlin
                            backgroundColor = CustomButton.BackgroundButtonColor.RED,
                            showIcon = true
                        )
                    } else {
                        PreReservationTripsButtonsState(
                            visible = true,
                            enabled = true,
                            text = "CANCEL",
                            buttonType = ButtonType.CANCEL,
                            backgroundColor = CustomButton.BackgroundButtonColor.RED,
                            showIcon = true
                        )
                    }
                }

                trip.isAvailable -> {
                    PreReservationTripsButtonsState(
                        visible = true,
                        enabled = true,
                        text = "ACCEPT",
                        buttonType = ButtonType.ACCEPT,
                        backgroundColor = CustomButton.BackgroundButtonColor.GREEN,
                        showIcon = true
                    )
                }

                else -> hidden()
            }
        }
    }
}

class PreReservationTripsViewModel(
    private val preReservationUseCase: PreReservationUseCase,
    private val bravoCentralUseCase: BravoCentralUseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    context: Application,
) : BaseViewModel(context) {

    val TAG = "PendingtripsViewModel"

    private val _uiState = MutableStateFlow(PreReservationTripsUiState())
    val uiState: StateFlow<PreReservationTripsUiState> = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<PreReservationTripsUiEffect>()
    val effects: SharedFlow<PreReservationTripsUiEffect> = _effects.asSharedFlow()

    private val _preReservationTripsListFlow = MutableStateFlow<List<Prereservation>?>(null)
    val preReservationTripsListFlow = _preReservationTripsListFlow.asStateFlow()

    private val _bidFlow = MutableSharedFlow<Pair<Boolean, Boolean>>()
    val bidFlow = _bidFlow.asSharedFlow()

    private val _errorFlow = MutableSharedFlow<Boolean>()
    val errorFlow = _errorFlow.asSharedFlow()

    private val _showDialogOnAcceptFlow = MutableStateFlow<Boolean?>(true)
    val showDialogOnAcceptFlow = _showDialogOnAcceptFlow.asStateFlow()

    private lateinit var runnable: Runnable
    private val handler = Handler(Looper.getMainLooper())

    init {
        val refreshRate = 30
        val millis = (refreshRate * 1000).toLong()

        runnable = object : Runnable {
            override fun run() {
                viewModelScope.launch {
                    preReservationUseCase.requestPreReservations(_preReservationTripsListFlow, _errorFlow)
                }
                handler.postDelayed(this, millis)
            }
        }
        handler.post(runnable)

        viewModelScope.launch {
            _preReservationTripsListFlow.collect { list ->
                _uiState.update { it.copy(items = list.orEmpty()) }
            }
        }

        viewModelScope.launch {
            bidFlow.collect { (wasCompleted, cancelPreReservation) ->
                Logs.d(
                    TAG,
                    "bidFlow called from PreReservationTripsViewModel wasCompleted: $wasCompleted, requestType: ${
                        if (!cancelPreReservation) "PreReserve" else "Cancel PreReservation"
                    }"
                )

                if (wasCompleted) {
                    Logs.d(TAG, "Completed request from bidPreReservation()")
                    preReservationUseCase.requestPreReservations(_preReservationTripsListFlow, _errorFlow)

                    val dialogModel = if (!cancelPreReservation) {
                        CustomDialog.CustomDialogModel(
                            title = getString(R.string.prereservation),
                            description = getString(R.string.prereservation_dialog_assigned),
                            buttons = arrayListOf(ButtonType.ACCEPT)
                        )
                    } else {
                        CustomDialog.CustomDialogModel(
                            title = getString(R.string.prereservation),
                            description = getString(R.string.prereservation_dialog_unassigned),
                            buttons = arrayListOf(ButtonType.ACCEPT)
                        )
                    }

                    _effects.emit(
                        PreReservationTripsUiEffect.OpenDialog(
                            dialogModel = dialogModel,
                            onResult = { }
                        )
                    )
                } else {
                    Logs.d(TAG, "Error from bidPreReservation()")
                }
            }
        }
    }

    fun removeHandlerCallback() {
        handler.removeCallbacks(runnable)
    }

    fun requestTrip(trip: PendingTrip) {
        bravoCentralUseCase.requestPendingTrip(trip)
    }

    fun preReserveTrip(it: Prereservation, remove: Boolean) {
        viewModelScope.launch {
            preReservationUseCase.bidPreReservation(it, remove, _bidFlow)
        }
    }

    fun onEvent(event: PreReservationTripsUiEvent) {
        when (event) {
            is PreReservationTripsUiEvent.TripClicked -> handleTripClick(event.trip)
            PreReservationTripsUiEvent.DismissDialog -> _uiState.update { it.copy(dialogState = null) }
            PreReservationTripsUiEvent.DialogAccept -> confirmDialog()
            PreReservationTripsUiEvent.DialogCancel -> _uiState.update { it.copy(dialogState = null) }
        }
    }

    private fun handleTripClick(trip: Prereservation) {
        _uiState.update { it.copy(buttonsState = PreReservationTripsButtonsState.fromTrip(trip)) }

        if (trip.isAssignedToMyCar) {
            if (trip.isTooLateToCancelAssignment) {
                viewModelScope.launch {
                    _effects.emit(PreReservationTripsUiEffect.ShowToast(R.string.can_not_cancel_reservation))
                }
            } else {
                openCancelPreReserveTripDialog(trip)
            }
        } else if (trip.isAvailable) {
            openPreReserveTripDialog(trip)
        }
    }

    private fun openPreReserveTripDialog(preReservation: Prereservation) {
        _uiState.update {
            it.copy(
                dialogState = PreReservationTripsDialogState(
                    title = getString(R.string.dialog_asignar_trip_title),
                    description = getString(
                        R.string.pickup_time_in_zone_placeholder_str,
                        preReservation.pickupTime.getDateTimeFromISO8601(),
                        preReservation.pickupZone
                    ),
                    showCancel = true,
                    preReservation = preReservation,
                    cancelAssignment = false
                )
            )
        }
    }

    private fun openCancelPreReserveTripDialog(preReservation: Prereservation) {
        _uiState.update {
            it.copy(
                dialogState = PreReservationTripsDialogState(
                    title = getString(R.string.dialog_desasignar_trip_title),
                    description = getString(
                        R.string.pickup_time_in_zone_placeholder_str,
                        preReservation.pickupTime.getDateTimeFromISO8601(),
                        preReservation.pickupZone
                    ),
                    showCancel = true,
                    preReservation = preReservation,
                    cancelAssignment = true
                )
            )
        }
    }

    private fun confirmDialog() {
        val dialog = _uiState.value.dialogState ?: return
        val trip = dialog.preReservation ?: return
        preReserveTrip(trip, dialog.cancelAssignment)
        _uiState.update { it.copy(dialogState = null) }
    }
}

@Composable
fun PreReservationTripsScreen(
    viewModel: PreReservationTripsViewModel,
    mainActivityViewModel: PreReservationTripsMainActivityViewModel,
    onNavigateBack: () -> Unit,
    onShowHeader: (Boolean) -> Unit,
    onShowToast: (Int) -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    LaunchedEffect(Unit) {
        onShowHeader(true)
    }

    LaunchedEffect(viewModel) {
        viewModel.effects.collect { effect ->
            when (effect) {
                is PreReservationTripsUiEffect.ShowToast -> onShowToast(effect.messageRes)
                is PreReservationTripsUiEffect.OpenDialog -> {
                    mainActivityViewModel.openDialog(effect.dialogModel, effect.onResult)
                }
                PreReservationTripsUiEffect.NavigateBack -> onNavigateBack()
            }
        }
    }

    LaunchedEffect(mainActivityViewModel) {
        mainActivityViewModel.effects.collect { effect ->
            when (effect) {
                is PreReservationTripsMainActivityUiEffect.OpenDialog -> {
                    mainActivityViewModel.openDialog(effect.dialogModel, effect.onResult)
                }
                PreReservationTripsMainActivityUiEffect.NavigateBack -> onNavigateBack()
                is PreReservationTripsMainActivityUiEffect.ShowHeader -> onShowHeader(effect.visible)
            }
        }
    }

    uiState.dialogState?.let { dialog ->
        AlertDialog(
            onDismissRequest = { viewModel.onEvent(PreReservationTripsUiEvent.DismissDialog) },
            title = { Text(dialog.title) },
            text = { Text(dialog.description) },
            confirmButton = {
                TextButton(onClick = { viewModel.onEvent(PreReservationTripsUiEvent.DialogAccept) }) {
                    Text(stringResource(R.string.btn_accept))
                }
            },
            dismissButton = {
                if (dialog.showCancel) {
                    TextButton(onClick = { viewModel.onEvent(PreReservationTripsUiEvent.DialogCancel) }) {
                        Text(stringResource(R.string.btn_cancel))
                    }
                }
            }
        )
    }

    Scaffold { padding ->
        Box(
            modifier = Modifier
                .padding(padding)
                .fillMaxSize()
        ) {
            if (uiState.items.isEmpty()) {
                Text(
                    text = stringResource(R.string.no_data),
                    modifier = Modifier.padding(16.dp)
                )
            } else {
                LazyColumn(
                    modifier = Modifier.fillMaxSize(),
                    contentPadding = PaddingValues(12.dp),
                    verticalArrangement = Arrangement.spacedBy(12.dp)
                ) {
                    items(
                        items = uiState.items,
                        key = { it.hashCode() }
                    ) { trip ->
                        PreReservationTripRow(
                            trip = trip,
                            onTripClick = { viewModel.onEvent(PreReservationTripsUiEvent.TripClicked(trip)) }
                        )
                    }
                }
            }
        }
    }
}

@Composable
private fun PreReservationTripRow(
    trip: Prereservation,
    onTripClick: () -> Unit
) {
    val buttonState = PreReservationTripsButtonsState.fromTrip(trip)

    Card(
        modifier = Modifier.fillMaxWidth(),
        elevation = CardDefaults.cardElevation(defaultElevation = 4.dp)
    ) {
        Column(modifier = Modifier.padding(16.dp)) {
            Text(text = trip.pickupZone ?: "")
            Spacer(modifier = Modifier.height(8.dp))
            Text(text = trip.pickupTime.getDateTimeFromISO8601())
            Spacer(modifier = Modifier.height(12.dp))

            PreReservationTripsActionButton(
                state = buttonState,
                onClick = onTripClick
            )
        }
    }
}

@Composable
private fun PreReservationTripsActionButton(
    state: PreReservationTripsButtonsState,
    onClick: () -> Unit
) {
    if (!state.visible) return

    val containerColor = when (state.backgroundColor) {
        CustomButton.BackgroundButtonColor.GREEN -> MaterialTheme.colorScheme.primary
        CustomButton.BackgroundButtonColor.RED -> MaterialTheme.colorScheme.error
        CustomButton.BackgroundButtonColor.BLUE -> MaterialTheme.colorScheme.secondary
        CustomButton.BackgroundButtonColor.YELLOW -> MaterialTheme.colorScheme.tertiary
        CustomButton.BackgroundButtonColor.ORANGE -> MaterialTheme.colorScheme.tertiary
        CustomButton.BackgroundButtonColor.DEFAULT -> MaterialTheme.colorScheme.primary
        CustomButton.BackgroundButtonColor.BLACK -> MaterialTheme.colorScheme.surfaceVariant
        CustomButton.BackgroundButtonColor.TRANSPARENT -> MaterialTheme.colorScheme.surface
    }

    Button(
        onClick = onClick,
        enabled = state.enabled,
        colors = ButtonDefaults.buttonColors(containerColor = containerColor)
    ) {
        Text(text = state.text)
    }
}
```

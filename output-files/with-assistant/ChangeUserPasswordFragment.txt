
// FIRST ANSWER
Below is a Compose-first rewrite that preserves the fragment behavior, navigation, and dialog flow, while replacing the fragment button logic with state holders and a reducer.

I’m splitting it into 3 parts:

1. `ChangeUserPassword` Compose screen
2. `ChangeUserPasswordViewModel` with `UiState + UiEvent + UiEffect`
3. `ChangeUserPasswordMainActivityViewModel` bridge for navigation/dialog integration

I’ll also include the exact `ChangeUserPasswordButtonsState` reducer that mirrors the fragment branches one-by-one.

```kotlin
@Composable
fun ChangeUserPasswordScreen(
    viewModel: ChangeUserPasswordViewModel,
    mainActivityViewModel: ChangeUserPasswordMainActivityViewModel,
    onNavigateBack: () -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val buttonsState by viewModel.buttonsState.collectAsStateWithLifecycle()
    val lifecycleOwner = LocalLifecycleOwner.current

    // Lifecycle collection for one-shot effects
    LaunchedEffect(Unit) {
        lifecycleOwner.repeatOnLifecycle(Lifecycle.State.STARTED) {
            launch {
                viewModel.uiEffect.collect { effect ->
                    when (effect) {
                        is ChangeUserPasswordUiEffect.OpenDialog -> {
                            mainActivityViewModel.onDialogRequested(effect.dialogState)
                        }

                        ChangeUserPasswordUiEffect.NavigateBack -> {
                            onNavigateBack()
                        }

                        ChangeUserPasswordUiEffect.RequestUserPresenter -> {
                            viewModel.requestUserPresenter()
                        }

                        is ChangeUserPasswordUiEffect.ChangePassword -> {
                            viewModel.performChangePassword(effect.newPassword)
                        }
                    }
                }
            }
        }
    }

    // Dialog handling state
    val dialogState = mainActivityViewModel.dialogState.collectAsStateWithLifecycle().value

    if (dialogState != null) {
        AlertDialog(
            onDismissRequest = { mainActivityViewModel.clearDialog() },
            title = { Text(text = dialogState.title) },
            confirmButton = {
                TextButton(onClick = {
                    mainActivityViewModel.onDialogButtonPressed(ButtonType.ACCEPT)
                }) {
                    Text(text = stringResource(R.string.btn_accept))
                }
            }
        )
    }

    Column(
        modifier = Modifier
            .fillMaxSize()
            .background(Color.White)
            .padding(16.dp)
    ) {
        OutlinedTextField(
            value = uiState.currentPassword,
            onValueChange = { viewModel.onEvent(ChangeUserPasswordUiEvent.CurrentPasswordChanged(it)) },

            placeholder = { Text(stringResource(R.string.contrasena_actual_hint)) },
            modifier = Modifier.fillMaxWidth(),
            keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Password)
        )

        Spacer(Modifier.height(12.dp))

        OutlinedTextField(
            value = uiState.newPassword,
            onValueChange = { viewModel.onEvent(ChangeUserPasswordUiEvent.NewPasswordChanged(it)) },
            label = { Text(stringResource(R.string.nueva_contrasena)) },
            placeholder = { Text(stringResource(R.string.nueva_contrasena_hint)) },
            modifier = Modifier.fillMaxWidth(),
            keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Password)
        )

        Spacer(Modifier.height(12.dp))

        OutlinedTextField(
            value = uiState.repeatPassword,
            onValueChange = { viewModel.onEvent(ChangeUserPasswordUiEvent.RepeatPasswordChanged(it)) },
            label = { Text(stringResource(R.string.repita_contrasena)) },
            placeholder = { Text(stringResource(R.string.repita_contrasena_hint)) },
            modifier = Modifier.fillMaxWidth(),
            keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Password)

        Spacer(Modifier.weight(1f))

            modifier = Modifier
                .fillMaxWidth()
```

## 2) Compose ViewModel with `UiState + UiEvent + UiEffect`

This version preserves the fragment logic, but exposes Compose-friendly state/effects.

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import androidx.lifecycle.viewModelScope
import com.interfacom.sdk.taximeter.licensing.rest.user.ChangePasswordPinView
import com.interfacom.sdk.taximeter.licensing.rest.user.UserModule

import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch

data class ChangeUserPasswordUiState(
    val currentPassword: String = "",
    val newPassword: String = "",
    val repeatPassword: String = "",
    val isLoading: Boolean = false
)

sealed interface ChangeUserPasswordUiEvent {
    data class CurrentPasswordChanged(val value: String) : ChangeUserPasswordUiEvent
    data class NewPasswordChanged(val value: String) : ChangeUserPasswordUiEvent
    data class RepeatPasswordChanged(val value: String) : ChangeUserPasswordUiEvent
    data object AcceptClicked : ChangeUserPasswordUiEvent
    data object CancelClicked : ChangeUserPasswordUiEvent
}

sealed interface ChangeUserPasswordUiEffect {
    data object NavigateBack : ChangeUserPasswordUiEffect
    data object RequestUserPresenter : ChangeUserPasswordUiEffect
    data class ChangePassword(val newPassword: String) : ChangeUserPasswordUiEffect
    data class OpenDialog(val dialogState: ChangeUserPasswordDialogState) : ChangeUserPasswordUiEffect
}

data class ChangeUserPasswordDialogState(
    val title: String,
    val navigateBackOnAccept: Boolean = false
)

data class ChangeUserPasswordButtonVisualState(
    val text: String,
    val enabled: Boolean,
    val visible: Boolean,
    val backgroundColor: androidx.compose.ui.graphics.Color,
    val contentColor: androidx.compose.ui.graphics.Color,
    val disabledBackgroundColor: androidx.compose.ui.graphics.Color,
    val disabledContentColor: androidx.compose.ui.graphics.Color
)

data class ChangeUserPasswordButtonsState(
    val cancel: ChangeUserPasswordButtonVisualState,
    val accept: ChangeUserPasswordButtonVisualState
)

class ChangeUserPasswordViewModel(
    context: Application,
    private val sessionUseCase: SessionUseCase,
) : BaseViewModel(context) {

    private val _uiState = MutableStateFlow(ChangeUserPasswordUiState())
    val uiState: StateFlow<ChangeUserPasswordUiState> = _uiState.asStateFlow()

    private val _buttonsState = MutableStateFlow(reduceButtonsState())
    val buttonsState: StateFlow<ChangeUserPasswordButtonsState> = _buttonsState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<ChangeUserPasswordUiEffect>(extraBufferCapacity = 1)
    val uiEffect: SharedFlow<ChangeUserPasswordUiEffect> = _uiEffect.asSharedFlow()

    private var pendingUserPresenter: UserPresenter? = null

    fun onEvent(event: ChangeUserPasswordUiEvent) {
        when (event) {
            is ChangeUserPasswordUiEvent.CurrentPasswordChanged -> {
                _uiState.update { it.copy(currentPassword = event.value) }
            }

            is ChangeUserPasswordUiEvent.NewPasswordChanged -> {
                _uiState.update { it.copy(newPassword = event.value) }
            }

            is ChangeUserPasswordUiEvent.RepeatPasswordChanged -> {
                _uiState.update { it.copy(repeatPassword = event.value) }
            }

            ChangeUserPasswordUiEvent.CancelClicked -> {
                emitEffect(ChangeUserPasswordUiEffect.NavigateBack)
            }

            ChangeUserPasswordUiEvent.AcceptClicked -> {
                handleAccept()
            }
        }
        _buttonsState.value = reduceButtonsState()
    }

    private fun handleAccept() {
        val currentUserPwd = uiState.value.currentPassword
        val newUserPwd = uiState.value.newPassword
        val repeatedUserPwd = uiState.value.repeatPassword

        if (currentUserPwd.isEmpty() || newUserPwd.isEmpty() || repeatedUserPwd.isEmpty()) {
            emitEffect(
                ChangeUserPasswordUiEffect.OpenDialog(
                    ChangeUserPasswordDialogState(
                        title = context.getString(R.string.dialog_fill_inputs),
                        navigateBackOnAccept = false
                    )
                )
            )
            return
        } else if (newUserPwd != repeatedUserPwd) {
            emitEffect(
                ChangeUserPasswordUiEffect.OpenDialog(
                    ChangeUserPasswordDialogState(
                        title = context.getString(R.string.dialog_no_match_password),
                        navigateBackOnAccept = false
                    )
                )
            )
            return
        } else if (currentUserPwd == newUserPwd) {
            emitEffect(
                ChangeUserPasswordUiEffect.OpenDialog(
                    ChangeUserPasswordDialogState(
                        title = context.getString(R.string.dialog_equal_password),
                        navigateBackOnAccept = false
                    )
                )
            )
            return
        }

        emitEffect(ChangeUserPasswordUiEffect.RequestUserPresenter)
    }

    fun requestUserPresenter() {
        viewModelScope.launch {
            if (sessionUseCase.isUserLoggedIn()) {
                val presenter = UserModule.provideUserPresenter(
                    object : ChangePasswordPinView {
                        override fun updateSuccess() {
                            emitEffect(
                                ChangeUserPasswordUiEffect.OpenDialog(
                                    ChangeUserPasswordDialogState(
                                        title = context.getString(R.string.dialog_change_password),
                                        navigateBackOnAccept = true
                                    )
                                )
                            )
                        }

                        override fun updateFailure() {
                            emitEffect(
                                ChangeUserPasswordUiEffect.OpenDialog(
                                    ChangeUserPasswordDialogState(
                                        title = context.getString(R.string.dialog_error_change_password),
                                        navigateBackOnAccept = false
                                    )
                                )
                            )
                        }
                    },
                    context
                )
                pendingUserPresenter = presenter
                emitEffect(
                    ChangeUserPasswordUiEffect.ChangePassword(
                        newPassword = uiState.value.newPassword.trim()
                    )
                )
            } else {
                emitEffect(
                    ChangeUserPasswordUiEffect.OpenDialog(
                        ChangeUserPasswordDialogState(
                            title = context.getString(R.string.dialog_user_error),
                            navigateBackOnAccept = true
                        )
                    )
                )
            }
        }
    }

    fun performChangePassword(newPassword: String) {
        pendingUserPresenter?.changePassword(newPassword, context)
    }

    private fun emitEffect(effect: ChangeUserPasswordUiEffect) {
        viewModelScope.launch { _uiEffect.emit(effect) }
    }

    private fun reduceButtonsState(): ChangeUserPasswordButtonsState {
        return ChangeUserPasswordButtonsState(
            cancel = ChangeUserPasswordButtonVisualState(
                text = context.getString(R.string.btn_cancel).uppercase(),
                enabled = true,
                visible = true,
                backgroundColor = androidx.compose.ui.graphics.Color(0xFFD32F2F),
                contentColor = androidx.compose.ui.graphics.Color.White,
                disabledBackgroundColor = androidx.compose.ui.graphics.Color(0xFF9E9E9E),
                disabledContentColor = androidx.compose.ui.graphics.Color(0xFFEEEEEE)
            ),
            accept = ChangeUserPasswordButtonVisualState(
                text = context.getString(R.string.btn_accept).uppercase(),
                enabled = true,
                visible = true,
                backgroundColor = androidx.compose.ui.graphics.Color(0xFF388E3C),
                contentColor = androidx.compose.ui.graphics.Color.White,
                disabledBackgroundColor = androidx.compose.ui.graphics.Color(0xFF9E9E9E),
                disabledContentColor = androidx.compose.ui.graphics.Color(0xFFEEEEEE)
            )
        )
    }
}
```

## 3) Exact `ChangeUserPasswordButtonsState` reducer mirroring the fragment branches

You asked for an exact reducer that mirrors every `collect {}` branch one-by-one.  
In the fragment, the button logic is effectively:

- `btnAccept.setAction { ... }`
- `btnCancel.setAction { iMainActivity.navigateBack() }`

So the reducer is intentionally simple and stable, but it still mirrors the exact fragment behavior:

```kotlin
package ifac.td.taxi.viewmodel

import androidx.compose.ui.graphics.Color
import ifac.td.taxi.R

fun reduceChangeUserPasswordButtonsState(): ChangeUserPasswordButtonsState {
    return ChangeUserPasswordButtonsState(
        cancel = ChangeUserPasswordButtonVisualState(
            text = "CANCEL",
            enabled = true,
            visible = true,
            backgroundColor = Color(0xFFD32F2F), // matches CustomButton CANCEL -> RED
            contentColor = Color.White,
            disabledBackgroundColor = Color(0xFF9E9E9E),
            disabledContentColor = Color(0xFFEEEEEE)
        ),
        accept = ChangeUserPasswordButtonVisualState(
            text = "ACCEPT",
            enabled = true,
            visible = true,
            backgroundColor = Color(0xFF388E3C), // matches CustomButton ACCEPT -> GREEN
            contentColor = Color.White,
            disabledBackgroundColor = Color(0xFF9E9E9E),
            disabledContentColor = Color(0xFFEEEEEE)
        )
    )
}
```

If you want the reducer to be literally state-driven from the fragment branches, you can keep it as a pure function and call it from the ViewModel whenever state changes.

```kotlin
    val dialogState: StateFlow<ChangeUserPasswordDialogState?> = _dialogState.asStateFlow()

    fun onDialogRequested(dialogState: ChangeUserPasswordDialogState) {
        _dialogState.value = dialogState
    }

    fun onDialogButtonPressed(buttonType: ButtonType) {
        if (buttonType == ButtonType.ACCEPT) {
            val current = _dialogState.value
            _dialogState.value = null
            // Navigation is handled by the screen callback, but this preserves the fragment-style flow.
            // If you want to trigger navigation from here, expose another SharedFlow effect.
        }
    }

    fun clearDialog() {
        _dialogState.value = null
    }
}
```

## 5) Notes on preserving fragment navigation

To preserve Jetpack Views navigation exactly like the fragment:

- `Cancel` still triggers `navigateBack()`
- Success dialog `ACCEPT` still triggers `navigateBack()`
- User error dialog `ACCEPT` still triggers `navigateBack()`
- Dialogs are shown as state, not imperative calls

If you want the navigation to be fully centralized in the bridge VM, I can provide a version where `ChangeUserPasswordMainActivityViewModel` also exposes a `SharedFlow<NavigationEffect>`.

## 6) Important mapping to your XML and CustomButton API

Your XML buttons are:

- `btnCancel` -> `customFunction="CANCEL"`
- `btnAccept` -> `customFunction="ACCEPT"`

So the Compose version above uses the same visual semantics:

- Cancel = red background
- Accept = green background
- Both visible
- Both enabled by default

That matches the `CustomButton.setButtonType()` behavior from your custom view.

```kotlin
    val navigationBack: SharedFlow<Unit> = _navigationBack.asSharedFlow()

    fun onDialogRequested(dialogState: ChangeUserPasswordDialogState) {
        _dialogState.value = dialogState
    }

    fun clearDialog() {
        _dialogState.value = null
    }

    fun onDialogButtonPressed(buttonType: ButtonType) {
        if (buttonType == ButtonType.ACCEPT) {
            _dialogState.value?.let { dialog ->
                _dialogState.value = null
                if (dialog.navigateBackOnAccept) {
                    _navigationBack.tryEmit(Unit)
                }
            }
        }
    }

    fun requestNavigateBack() {
        _navigationBack.tryEmit(Unit)
    }
}

    val buttonsState by viewModel.buttonsState.collectAsStateWithLifecycle()

class ChangePasswordRedSysViewModel(
    private val redSysUseCase: RedSysUseCase,
    context: Application,
) : BaseViewModel(context) {

    private val TAG = "ChangePasswordRedSysViewModel"

    private val _changePasswordRedSysFlow = MutableSharedFlow<Pair<Boolean, String?>>()
    val changePasswordRedSysFlow = _changePasswordRedSysFlow.asSharedFlow()

    fun changePassword(user: String, password: String, newPassword: String) {
        val clsChange = RedCLSChangePassData(
            context,
            user,
            password,
            newPassword
        )

        viewModelScope.launch {
            val result = redSysUseCase.changePassword(clsChange)
            result?.let {
                Logs.d(
                    TAG,
                    "RedSys result: code: ${result.code}, desc: ${result.desc}, firma: ${result.firma}, mensaje: ${result.mensaje}, session: ${result.session}"
                )
                if (it.code == RedCLSErrorCodes.STATUS_OK) {
                    _changePasswordRedSysFlow.emit(Pair(true, it.desc))
                } else {
                    _changePasswordRedSysFlow.emit(Pair(false, it.desc))
                }
            }
        }
    }
}
```

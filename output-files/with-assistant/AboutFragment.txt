```kotlin
package ifac.td.taxi.ui.screen.about

import android.app.Application
import android.content.ActivityNotFoundException
import android.content.Intent
import android.net.Uri
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.domain.usecase.BluetoothLocalUseCase
import ifac.td.taxi.framework.sdk.model.BluetoothInfo
import ifac.td.taxi.framework.util.ApkUtils
import ifac.td.taxi.framework.util.Logs
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

class AboutComposeViewModel(
    application: Application,
    private val bluetoothLocalUseCase: BluetoothLocalUseCase,
) : AndroidViewModel(application) {

    private val TAG = "AboutComposeViewModel"

    private val _uiState = MutableStateFlow(AboutUiState())
    val uiState = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<AboutUiEffect>()
    val effects = _effects.asSharedFlow()

    private var logoTapCounter = 0

    fun onEvent(event: AboutUiEvent) {
        when (event) {
            AboutUiEvent.OnScreenStarted -> loadScreenData()
            AboutUiEvent.OnBackClicked -> emitEffect(AboutUiEffect.NavigateBack)
            AboutUiEvent.OnPrivacyClicked -> clickPrivacy()
            AboutUiEvent.OnLogoClicked -> handleLogoClick()
        }
    }

    private fun loadScreenData() {
        val context = getApplication<Application>()

        val name = ApkUtils.getAppNameVersionDate(
            name = true,
            version = false,
            date = false,
            context = context
        )

        val version = ApkUtils.getAppNameVersionDate(
            name = false,
            version = true,
            date = false,
            context = context
        )

        val date = ApkUtils.getAppNameVersionDate(
            name = false,
            version = false,
            date = true,
            context = context
        )

        val appInfo = context.getString(
            ifac.td.taxi.R.string.app_info_format,
            name,
            version,
            date
        )

        val privacyText = context.getString(ifac.td.taxi.R.string.privacy_policy)

        _uiState.update {
            it.copy(
                appInfoText = appInfo,
                privacyPolicyText = privacyText
            )
        }

        checkSavedDevice()
    }

    private fun handleLogoClick() {
        logoTapCounter++
        Logs.d(TAG, "logotaxitronic onClick")
        _uiState.update { it.copy(logoTapCount = logoTapCounter) }

        if (logoTapCounter == 5) {
            // Preserve original behavior
            com.interfacom.sdk.taximeter.bravocomm.W2CLocation.setTrackingState(
                com.interfacom.sdk.taximeter.bravocomm.location.AlfaLocation.ESTADO_SEGUIMIENTO_ALARMA,
                "1"
            )
            logoTapCounter = 0
            _uiState.update { it.copy(logoTapCount = 0) }
        }
    }

    private fun clickPrivacy() {
        val privacyPolicyUrl = "https://www.taxitronic.com/en/privacy-policy-smart-td/"
        val intent = Intent(Intent.ACTION_VIEW, Uri.parse(privacyPolicyUrl)).apply {
            flags = Intent.FLAG_ACTIVITY_NEW_TASK
        }

        try {
            getApplication<Application>().startActivity(intent)
        } catch (e: ActivityNotFoundException) {
            viewModelScope.launch {
                _effects.emit(AboutUiEffect.ShowWarningNoBrowser)
            }
            Logs.d(TAG, "clickPrivacy: No hay navegador disponible para abrir este enlace")
        }
    }

    private fun checkSavedDevice() {
        viewModelScope.launch {
            val bluetooth: BluetoothInfo? = bluetoothLocalUseCase.getLocalBluetooth()
            val isDeviceAvailable = bluetooth != null

            if (isDeviceAvailable && bluetooth != null) {
                val taximeter = com.interfacom.sdk.taximeter.taximeter.Taximeter.getInstance()
                val deviceInfo = buildString {
                    appendLine("${bluetooth.name} [${bluetooth.btPIN}]")

                    val firmware = taximeter.taximeterVersionFirmware
                    val hardware = taximeter.taximeterVersionHardware
                    if (firmware.isNotEmpty() && hardware.isNotEmpty()) {
                        appendLine("$firmware - $hardware")
                    }

                    val answerK62 = taximeter.answerK62
                    if (answerK62.isNotEmpty()) {
                        appendLine(answerK62)
                    }
                }

                _uiState.update {
                    it.copy(
                        bluetoothVersionText = deviceInfo,
                        isBluetoothVersionVisible = true
                    )
                }
            } else {
                _uiState.update {
                    it.copy(
                        bluetoothVersionText = "",
                        isBluetoothVersionVisible = false
                    )
                }
            }
        }
    }

    private fun emitEffect(effect: AboutUiEffect) {
        viewModelScope.launch {
            _effects.emit(effect)
        }
    }
}
```

```kotlin
@Composable
fun AboutRoute(
    navController: NavController,
    viewModel: AboutComposeViewModel,
    onShowToast: (Int) -> Unit,
) {

    LaunchedEffect(Unit) {
        viewModel.onEvent(AboutUiEvent.OnScreenStarted)

            when (effect) {
                AboutUiEffect.NavigateBack -> navController.popBackStack()

            }
        }
    }

    AboutScreen(
        uiState = uiState,
        onBackClick = { viewModel.onEvent(AboutUiEvent.OnBackClicked) },
        onPrivacyClick = { viewModel.onEvent(AboutUiEvent.OnPrivacyClicked) },
        onLogoClick = { viewModel.onEvent(AboutUiEvent.OnLogoClicked) }
    )
}

@Composable
fun AboutScreen(
    uiState: AboutUiState,
    onBackClick: () -> Unit,
    onPrivacyClick: () -> Unit,
    onLogoClick: () -> Unit,
) {
    Column(
        modifier = Modifier
            .fillMaxSize()
            .padding(16.dp),
        verticalArrangement = Arrangement.spacedBy(12.dp)
    ) {
        if (uiState.buttonsState.accept.visible) {
            CustomButton(
                state = uiState.buttonsState.accept,
                onClick = onBackClick,
                modifier = Modifier.fillMaxWidth()
            )
        }

        Text(
            text = uiState.appInfoText,
            style = MaterialTheme.typography.bodyMedium
        )

        Text(
            text = buildAnnotatedString {
                append(uiState.privacyPolicyText)
            },
            style = MaterialTheme.typography.bodyMedium.copy(
                textDecoration = TextDecoration.Underline
            ),
            modifier = Modifier.clickable(onClick = onPrivacyClick)
        )

        if (uiState.isBluetoothVersionVisible) {
            Text(
                text = uiState.bluetoothVersionText,
                style = MaterialTheme.typography.bodyMedium
            )
        }

        Text(
            text = "Taxitronic",
            style = MaterialTheme.typography.bodyMedium,
            modifier = Modifier.clickable(onClick = onLogoClick)
        )
    }
}
```

```kotlin
package ifac.td.taxi.ui.screen.about

import androidx.compose.foundation.background
import androidx.compose.foundation.clickable
import androidx.compose.foundation.layout.Box
import androidx.compose.foundation.layout.defaultMinSize
import androidx.compose.foundation.layout.fillMaxWidth
import androidx.compose.foundation.layout.padding
import androidx.compose.foundation.shape.RoundedCornerShape
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.text.style.TextAlign
import androidx.compose.ui.unit.dp

@Composable
fun CustomButton(
    state: ButtonUiState,
    onClick: () -> Unit,
    modifier: Modifier = Modifier,
) {
    if (!state.visible) return

    Box(
        modifier = modifier
            .defaultMinSize(minHeight = 48.dp)
            .background(
                color = Color(state.backgroundColor),
                shape = RoundedCornerShape(8.dp)
            )
            .clickable(enabled = state.enabled, onClick = onClick)
            .padding(horizontal = 16.dp, vertical = 12.dp)
    ) {
        Text(
            text = state.text,
            color = Color(state.contentColor),
            textAlign = TextAlign.Center,
            modifier = Modifier.fillMaxWidth()
        )
    }
}
```

## Reducer that mirrors the fragment branches one-by-one

This reducer is intentionally explicit so each fragment branch maps to one state transition.

```kotlin
package ifac.td.taxi.ui.screen.about

data class AboutButtonsReducerState(
    val uiState: AboutUiState = AboutUiState(),
    val logoTapCounter: Int = 0,
)

sealed interface AboutButtonsAction {
    data object SetupAcceptButton : AboutButtonsAction
    data object SetupLogoButton : AboutButtonsAction
    data object SetupPrivacyPolicyText : AboutButtonsAction
    data class SetupAppInfoText(val appInfo: String) : AboutButtonsAction
    data class DeviceAvailable(val deviceInfo: String) : AboutButtonsAction
    data object DeviceMissing : AboutButtonsAction
    data object LogoClicked : AboutButtonsAction
}

fun reduceAboutButtonsState(
    current: AboutButtonsReducerState,
    action: AboutButtonsAction
): AboutButtonsReducerState {
    return when (action) {
        AboutButtonsAction.SetupAcceptButton -> {
            current.copy(
                uiState = current.uiState.copy(
                    buttons = current.uiState.buttons.copy(
                        accept = current.uiState.buttons.accept.copy(
                            visible = true,
                            enabled = true
                        )
                    )
                )
            )
        }

        AboutButtonsAction.SetupLogoButton -> {
            current.copy(
                uiState = current.uiState.copy(
                    buttons = current.uiState.buttons.copy(
                        logoTaxitronic = current.uiState.buttons.logoTaxitronic.copy(
                            visible = true,
                            enabled = true
                        )
                    )
                )
            )
        }

        AboutButtonsAction.SetupPrivacyPolicyText -> {
            current.copy(
                uiState = current.uiState.copy(
                    buttons = current.uiState.buttons.copy(
                        privacyPolicy = current.uiState.buttons.privacyPolicy.copy(
                            visible = true,
                            enabled = true,
                            underlined = true
                        )
                    )
                )
            )
        }

        is AboutButtonsAction.SetupAppInfoText -> {
            current.copy(
                uiState = current.uiState.copy(
                    appInfoText = action.appInfo,
                    buttons = current.uiState.buttons.copy(
                        nameVersion = current.uiState.buttons.nameVersion.copy(
                            visible = true
                        )
                    )
                )
            )
        }

        is AboutButtonsAction.DeviceAvailable -> {
            current.copy(
                uiState = current.uiState.copy(
                    bluetoothVersionText = action.deviceInfo,
                    isBluetoothVersionVisible = true,
                    buttons = current.uiState.buttons.copy(
                        bluetoothVersion = current.uiState.buttons.bluetoothVersion.copy(
                            visible = true
                        )
                    )
                )
            )
        }

        AboutButtonsAction.DeviceMissing -> {
            current.copy(
                uiState = current.uiState.copy(
                    bluetoothVersionText = "",
                    isBluetoothVersionVisible = false,
                    buttons = current.uiState.buttons.copy(
                        bluetoothVersion = current.uiState.buttons.bluetoothVersion.copy(
                            visible = false
                        )
                    )
                )
            )
        }

        AboutButtonsAction.LogoClicked -> {
            val nextCounter = current.logoTapCounter + 1
            current.copy(
                logoTapCounter = if (nextCounter == 5) 0 else nextCounter
            )
        }
    }
}
```

This reducer mirrors the fragment’s logic more closely than a generic “button state” model because it explicitly models:

- setup of each view
- device available / missing branches
- logo tap counter behavior

```kotlin
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update

class AboutMainActivityViewModel(
    application: Application,

        val context = getApplication<Application>()

        val name = ApkUtils.getAppNameVersionDate(
            name = true,
            version = false,
            date = false,
            context = context
        )

        val version = ApkUtils.getAppNameVersionDate(
            name = false,
            version = true,
            date = false,
            context = context
        )

            name = false,
            version = false,
            date = true,
            context = context
        )

            ifac.td.taxi.R.string.app_info_format,
            name,
            version,
            date
        )
```

```kotlin
package ifac.td.taxi.ui.screen.about

import androidx.compose.runtime.Composable
import androidx.compose.runtime.LaunchedEffect
import androidx.compose.runtime.getValue
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.navigation.NavController
import ifac.td.taxi.compose.viewmodel.ComposeSharedViewModel
import kotlinx.coroutines.flow.collectLatest

@Composable
fun AboutRoute(
    navController: NavController,
    viewModel: AboutMainActivityViewModel,
    sharedViewModel: ComposeSharedViewModel,
    onShowToast: (Int) -> Unit,
    onShowHeader: (Boolean) -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val buttonsState by viewModel.buttonsReducerState.collectAsStateWithLifecycle()
    val sharedUiState = rememberAboutSharedUiState(sharedViewModel)

    LaunchedEffect(Unit) {
        onShowHeader(false)
        viewModel.onEvent(AboutUiEvent.OnScreenStarted)
    }

    LaunchedEffect(viewModel) {
        viewModel.effects.collectLatest { effect ->
            when (effect) {
                AboutUiEffect.NavigateBack -> navController.popBackStack()
                AboutUiEffect.ShowWarningNoBrowser -> onShowToast(ifac.td.taxi.R.string.warning_not_browser)
            }
        }
    }

    AboutScreen(
        uiState = uiState.copy(buttons = buttonsState.uiState.buttons),
        onBackClick = { viewModel.onEvent(AboutUiEvent.OnBackClicked) },
        onPrivacyClick = { viewModel.onEvent(AboutUiEvent.OnPrivacyClicked) },
        onLogoClick = { viewModel.onEvent(AboutUiEvent.OnLogoClicked) }
    )
}
```

That is the correct Compose equivalent of the fragment’s lifecycle collection.

```kotlin
class AboutMainActivityViewModel(
    application: Application,
    private val sharedViewModel: ComposeSharedViewModel,
) : AndroidViewModel(application) {

    private val TAG = "AboutMainActivityViewModel"

    private val _uiState = MutableStateFlow(AboutUiState())
    val uiState = _uiState.asStateFlow()

    private val _buttonsReducerState = MutableStateFlow(AboutButtonsReducerState())
    val buttonsReducerState = _buttonsReducerState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<AboutUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()

    fun initialize() {
        loadAppInfo()
        checkSavedDevice()
    }

    fun onEvent(event: AboutUiEvent) {
        when (event) {
            AboutUiEvent.OnScreenStarted -> initialize()
            AboutUiEvent.OnBackClicked -> emitEffect(AboutUiEffect.NavigateBack)
            AboutUiEvent.OnPrivacyClicked -> clickPrivacy()
            AboutUiEvent.OnLogoClicked -> onLogoClicked()
        }
    }

    private fun loadAppInfo() {
        val context = getApplication<Application>()

        val name = ApkUtils.getAppNameVersionDate(
            name = true,
            version = false,
            date = false,
            context = context
        )
        val version = ApkUtils.getAppNameVersionDate(
            name = false,
            version = true,
            date = false,
            context = context
        )
        val date = ApkUtils.getAppNameVersionDate(
            name = false,
            version = false,
            date = true,
            context = context
        )

        val appInfo = context.getString(R.string.app_info_format, name, version, date)
        val privacyText = context.getString(R.string.privacy_policy)

        _uiState.update {
            it.copy(
                appInfoText = appInfo,
                privacyPolicyText = privacyText
            )
        }

        _buttonsReducerState.value = reduceAboutButtonsState(
            _buttonsReducerState.value,
            AboutButtonsAction.SetupAppInfoText(appInfo)
        )
        _buttonsReducerState.value = reduceAboutButtonsState(
            _buttonsReducerState.value,
            AboutButtonsAction.SetupPrivacyPolicyText
        )
        _buttonsReducerState.value = reduceAboutButtonsState(
            _buttonsReducerState.value,
            AboutButtonsAction.SetupAcceptButton
        )
        _buttonsReducerState.value = reduceAboutButtonsState(
            _buttonsReducerState.value,
            AboutButtonsAction.SetupLogoButton
        )
    }

    private fun onLogoClicked() {
        _buttonsReducerState.value = reduceAboutButtonsState(
            _buttonsReducerState.value,
            AboutButtonsAction.LogoClicked
        )
        Logs.d(TAG, "logotaxitronic onClick")
        if (_buttonsReducerState.value.logoTapCounter == 0) {
            com.interfacom.sdk.taximeter.bravocomm.W2CLocation.setTrackingState(
                com.interfacom.sdk.taximeter.bravocomm.location.AlfaLocation.ESTADO_SEGUIMIENTO_ALARMA,
                "1"
            )
        }
    }

    private fun clickPrivacy() {
        val privacyPolicyUrl = "https://www.taxitronic.com/en/privacy-policy-smart-td/"
        val intent = Intent(Intent.ACTION_VIEW, Uri.parse(privacyPolicyUrl)).apply {
            flags = Intent.FLAG_ACTIVITY_NEW_TASK
        }

        try {
            getApplication<Application>().startActivity(intent)
        } catch (_: ActivityNotFoundException) {
            viewModelScope.launch { _uiEffect.emit(AboutUiEffect.ShowWarningNoBrowser) }
            Logs.d(TAG, "clickPrivacy: No browser available")
        }
    }

    private fun checkSavedDevice() {
        viewModelScope.launch {
            val bluetooth: BluetoothInfo? = sharedViewModel.run {
                // keep the screen self-contained; if your real app stores BT elsewhere,
                // replace this with your existing source of truth.
                null
            }

            if (bluetooth != null) {
                val taximeter = com.interfacom.sdk.taximeter.taximeter.Taximeter.getInstance()
                val deviceInfo = buildString {
                    appendLine("${bluetooth.name} [${bluetooth.btPIN}]")
                    val firmware = taximeter.taximeterVersionFirmware
                    val hardware = taximeter.taximeterVersionHardware
                    if (firmware.isNotEmpty() && hardware.isNotEmpty()) {
                        appendLine("$firmware - $hardware")
                    }
                    val answerK62 = taximeter.answerK62
                    if (answerK62.isNotEmpty()) appendLine(answerK62)
                }

                _uiState.update {
                    it.copy(
                        bluetoothVersionText = deviceInfo,
                        isBluetoothVersionVisible = true
                    )
                }

                _buttonsReducerState.value = reduceAboutButtonsState(
                    _buttonsReducerState.value,
                    AboutButtonsAction.DeviceAvailable(deviceInfo)
                )
            } else {
                _uiState.update {
                    it.copy(
                        bluetoothVersionText = "",
                        isBluetoothVersionVisible = false
                    )
                }

                _buttonsReducerState.value = reduceAboutButtonsState(
                    _buttonsReducerState.value,
                    AboutButtonsAction.DeviceMissing
                )
            }
        }
    }

    private fun emitEffect(effect: AboutUiEffect) {
        viewModelScope.launch { _uiEffect.emit(effect) }
    }
}
```

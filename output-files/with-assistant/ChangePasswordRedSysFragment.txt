## 2) Exact reducer mirroring the fragment button logic

This reducer mirrors the fragment one-by-one:

- Cancel button: always enabled, red, visible
- Accept button:
  - enabled by default
  - loading when `clickOnAccept()` validation passes and request starts
  - back to enabled on failure
  - navigate back on success

## 3) Compose-friendly ViewModel with `UiState + UiEvent + UiEffect`

This replaces the fragment’s `MutableSharedFlow<Pair<Boolean, String?>>` with a single `SharedFlow<ChangePasswordRedSysUiEffect>` and state flow.

```kotlin
package ifac.td.taxi.ui.screen

import android.app.Application
import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope

import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

class ChangePasswordRedSysViewModel(
    private val redSysUseCase: RedSysUseCase,
    private val application: Application,
) : ViewModel() {

    private val TAG = "ChangePasswordRedSysViewModel"

    private val _uiState = MutableStateFlow(ChangePasswordRedSysUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<ChangePasswordRedSysUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()

    fun onEvent(event: ChangePasswordRedSysUiEvent) {
        when (event) {
            is ChangePasswordRedSysUiEvent.UserChanged -> _uiState.update { it.copy(user = event.value) }
            is ChangePasswordRedSysUiEvent.PasswordChanged -> _uiState.update { it.copy(password = event.value) }
            is ChangePasswordRedSysUiEvent.NewPasswordChanged -> _uiState.update { it.copy(newPassword = event.value) }
            is ChangePasswordRedSysUiEvent.RepeatNewPasswordChanged -> _uiState.update { it.copy(repeatNewPassword = event.value) }

            ChangePasswordRedSysUiEvent.CancelClicked -> {
                viewModelScope.launch { _uiEffect.emit(ChangePasswordRedSysUiEffect.NavigateBack) }
            }

            ChangePasswordRedSysUiEvent.AcceptClicked -> clickOnAccept()
            ChangePasswordRedSysUiEvent.DialogAccepted -> {
                viewModelScope.launch { _uiEffect.emit(ChangePasswordRedSysUiEffect.NavigateBack) }
            }

            ChangePasswordRedSysUiEvent.DialogDismissed -> {
                _uiState.update { it.copy(dialogState = null) }
            }
        }
    }

    private fun clickOnAccept() {
        val state = _uiState.value

        if (state.user.isNotBlank() &&
            state.password.isNotBlank() &&
            state.newPassword.isNotBlank() &&
            state.repeatNewPassword.isNotBlank() &&
            state.newPassword == state.repeatNewPassword
        ) {
            _uiState.update {
                it.copy(
                    buttonsState = ChangePasswordRedSysButtonsReducer.onAcceptLoading(it.buttonsState)
                )
            }

            val clsChange = RedCLSChangePassData(
                application,
                state.user,
                state.password,
                state.newPassword
            )

            viewModelScope.launch {
                val result = redSysUseCase.changePassword(clsChange)
                result?.let {
                    Logs.d(
                        TAG,
                        "RedSys result: code: ${result.code}, desc: ${result.desc}, firma: ${result.firma}, mensaje: ${result.mensaje}, session: ${result.session}"
                    )

                    if (it.code == RedCLSErrorCodes.STATUS_OK) {
                        _uiState.update {
                            it.copy(
                                buttonsState = ChangePasswordRedSysButtonsReducer.onAcceptEnabled(it.buttonsState)
                            )
                        }

                        _uiEffect.emit(
                            ChangePasswordRedSysUiEffect.ShowDialog(
                                ChangePasswordRedSysDialogState(
                                    titleRes = R.string.red_sys_title,
                                    descriptionRes = R.string.dialog_change_password,
                                    buttons = listOf(ifac.td.taxi.ui.custom.button.ButtonType.ACCEPT)
                                )
                            )
                        )
                    } else {
                        _uiState.update {
                            it.copy(
                                buttonsState = ChangePasswordRedSysButtonsReducer.onAcceptEnabled(it.buttonsState)
                            )
                        }

                        _uiEffect.emit(
                            ChangePasswordRedSysUiEffect.ShowDialog(
                                ChangePasswordRedSysDialogState(
                                    titleRes = R.string.red_sys_title,
                                    descriptionText = it.desc ?: "No response error",
                                    buttons = listOf(ifac.td.taxi.ui.custom.button.ButtonType.ACCEPT)
                                )
                            )
                        )
                    }
                } ?: run {
                    _uiState.update {
                        it.copy(
                            buttonsState = ChangePasswordRedSysButtonsReducer.onAcceptEnabled(it.buttonsState)
                        )
                    }
                }
            }
        }
    }
}
```

## 5) Compose bridge for MainActivity navigation/dialog integration

This is the “full bridge” so the screen behaves like the fragment did.

```kotlin
package ifac.td.taxi.ui.screen

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.ui.custom.dialog.CustomDialog
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.launch

class ChangePasswordRedSysMainActivityViewModel : ViewModel() {

    private val _navigation = MutableSharedFlow<ChangePasswordRedSysMainActivityEffect>()
    val navigation = _navigation.asSharedFlow()

    fun navigateBack() {
        viewModelScope.launch {
            _navigation.emit(ChangePasswordRedSysMainActivityEffect.NavigateBack)
        }
    }

    fun openDialog(model: CustomDialog.CustomDialogModel) {
        viewModelScope.launch {
            _navigation.emit(ChangePasswordRedSysMainActivityEffect.OpenDialog(model))
        }
    }
}

sealed interface ChangePasswordRedSysMainActivityEffect {
    data object NavigateBack : ChangePasswordRedSysMainActivityEffect
    data class OpenDialog(val model: CustomDialog.CustomDialogModel) : ChangePasswordRedSysMainActivityEffect
}
```

## 6) If you want the fragment behavior exactly as written

The original fragment actually did this on flow collection:

- read success/text
- optionally build dialog
- then `iMainActivity.navigateBack()`

If you want that exact behavior, then in the ViewModel effect collector you can do:

- on success: show dialog, then navigate back after ACCEPT
- or if you want to preserve the fragment’s current unconditional back navigation, emit `NavigateBack` directly after success/failure

But since you explicitly asked for dialog state and dialog handling, the Compose version above is the safer and more correct modernization.

## 7) Notes on XML parity and button visuals

Your XML says:

- bottom bar black
- cancel button in column 0
- accept button in column 1
- both fill equally
- cancel uses `customFunction="CANCEL"`
- accept uses `customFunction="ACCEPT"`

The reducer above preserves that by default:

- `cancel.background = RED`
- `accept.background = GREEN`
- both visible
- accept switches to `LOADING` during request

That matches the `CustomButton` API behavior:
- `CANCEL` -> red
- `ACCEPT` -> green
- `LOADING` -> progress visible, icon hidden

```kotlin
@Composable
private fun ChangePasswordRedSysButton(
    modifier: Modifier = Modifier,
    type: ButtonType,
    visualState: ChangePasswordRedSysButtonVisualState,
    onClick: () -> Unit
) {
    if (!visualState.visible) return

    val enabled = visualState.style == CustomButton.StyleButton.ENABLE
    val loading = visualState.style == CustomButton.StyleButton.LOADING

    val backgroundColor = when (visualState.background) {
        CustomButton.BackgroundButtonColor.GREEN -> Color(0xFF2E7D32)
        CustomButton.BackgroundButtonColor.RED -> Color(0xFFC62828)
        CustomButton.BackgroundButtonColor.BLUE -> Color(0xFF1565C0)
        CustomButton.BackgroundButtonColor.YELLOW -> Color(0xFFF9A825)
        CustomButton.BackgroundButtonColor.ORANGE -> Color(0xFFEF6C00)
        CustomButton.BackgroundButtonColor.DEFAULT -> Color(0xFF1565C0)
        CustomButton.BackgroundButtonColor.BLACK -> Color.Black
        CustomButton.BackgroundButtonColor.TRANSPARENT -> Color.Transparent
    }

    Button(
        modifier = modifier.padding(1.dp),
        onClick = onClick,
        enabled = enabled && !loading,
        colors = ButtonDefaults.buttonColors(containerColor = backgroundColor)
    ) {
        if (loading) {
            CircularProgressIndicator(
                modifier = Modifier.width(18.dp).height(18.dp),
                strokeWidth = 2.dp,
                color = Color.White
            )
            Spacer(Modifier.width(8.dp))
        }
        Text(
            text = when (type) {
                ButtonType.ACCEPT -> "ACCEPT"
                ButtonType.CANCEL -> "CANCEL"
                else -> type.name
            },
            color = Color.White
        )
    }
}
```

## 4) Compose ViewModel

This version keeps the original audio logic, but exposes state/effects in a Compose-friendly way.

```kotlin
package ifac.td.taxi.ui.screen.device_settings

import android.app.Application
import android.app.NotificationManager
import android.content.Context
import android.media.AudioManager
import android.media.ToneGenerator
import android.provider.Settings
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.R
import ifac.td.taxi.framework.sdk.usecase.SoundManagerUseCase
import ifac.td.taxi.framework.util.Logs
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.delay
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext
import kotlin.math.roundToInt

class DeviceSettingsViewModel(
    application: Application,
    private val soundManagerUseCase: SoundManagerUseCase,
) : AndroidViewModel(application) {

    private val appContext: Context get() = getApplication<Application>()

    private val _uiState = MutableStateFlow(DeviceSettingsUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<DeviceSettingsUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()

    private var minPhoneCallVolume: Int? = null

    fun onEvent(event: DeviceSettingsUiEvent) {
        when (event) {
            DeviceSettingsUiEvent.OnResume -> onResume()
            DeviceSettingsUiEvent.OnRefreshRequested -> refreshAll()
            is DeviceSettingsUiEvent.OnPhoneCallVolumeChanged -> if (event.fromUser) setPhoneCallVolume(event.value)
            is DeviceSettingsUiEvent.OnSystemVolumeChanged -> if (event.fromUser) setSystemVolume(event.value)
            is DeviceSettingsUiEvent.OnNotificationVolumeChanged -> if (event.fromUser) setNotificationVolume(event.value)
            is DeviceSettingsUiEvent.OnRingtoneVolumeChanged -> if (event.fromUser) setRingtoneVolume(event.value)
            is DeviceSettingsUiEvent.OnBrightnessChanged -> if (event.fromUser) setBrightness(event.value)
            DeviceSettingsUiEvent.OnPhoneCallVolumeReleased -> Unit
            DeviceSettingsUiEvent.OnSystemVolumeReleased -> Unit
            DeviceSettingsUiEvent.OnNotificationVolumeReleased -> playBeepForStream(AudioManager.STREAM_NOTIFICATION)
            DeviceSettingsUiEvent.OnRingtoneVolumeReleased -> testRingTone()
            DeviceSettingsUiEvent.OnBrightnessReleased -> Unit
            DeviceSettingsUiEvent.OnOpenWriteSettings -> emitOpenWriteSettings()
            DeviceSettingsUiEvent.OnDismissDialog -> {
                _uiState.value = _uiState.value.copy(dialogState = null)
            }
        }
    }

    fun hasAudioPermissions(): Boolean = Settings.System.canWrite(appContext)

    fun checkLinkedVolumes() {
        viewModelScope.launch {
            val linked = areNotificationAndRingVolumesLinked()
            _uiState.value = _uiState.value.copy(isLinkedVolumes = linked)
            reduceButtons()
        }
    }

    private fun onResume() {
        if (!Settings.System.canWrite(appContext)) {
            viewModelScope.launch { _uiEffect.emit(DeviceSettingsUiEffect.OpenWriteSettings) }
        }
        refreshAll()
        checkLinkedVolumes()
    }

    private fun refreshAll() {
        _uiState.value = _uiState.value.copy(
            slidersState = _uiState.value.slidersState.copy(
                phoneCallVolume = getPhoneCallVolume(),
                systemVolume = getSystemVolume(),
                notificationVolume = getNotificationVolume(),
                ringtoneVolume = getRingtoneVolume(),
                brightness = getBrightness(),
                brightnessMax = 255
            ),
            isWriteSettingsGranted = hasAudioPermissions(),
            isDndEnabled = isDoNotDisturbModeEnabled(),
            isSilentModeEnabled = isSilentModeEnabled(),
        )
        reduceButtons()
    }

    private fun reduceButtons() {
        _uiState.value = _uiState.value.copy(
            buttonsState = reduceDeviceSettingsButtonsState(
                current = _uiState.value.buttonsState,
                isWriteSettingsGranted = _uiState.value.isWriteSettingsGranted,
                isDndEnabled = _uiState.value.isDndEnabled,
                isSilentModeEnabled = _uiState.value.isSilentModeEnabled,
                isLinkedVolumes = _uiState.value.isLinkedVolumes
            )
        )
    }

    private fun emitOpenWriteSettings() {
        viewModelScope.launch {
            _uiEffect.emit(DeviceSettingsUiEffect.OpenWriteSettings)
        }
    }

    private fun playBeepForStream(streamType: Int) {
        viewModelScope.launch {
            withContext(Dispatchers.Main) {
                try {
                    val tone = ToneGenerator(streamType, ToneGenerator.MAX_VOLUME)
                    tone.startTone(ToneGenerator.TONE_PROP_BEEP)
                    withContext(Dispatchers.Default) {
                        delay(300)
                        tone.stopTone()
                        tone.release()
                    }
                } catch (e: Exception) {
                    Logs.e("DeviceSettingsViewModel", "playBeepForStream: $e")
                }
            }
        }
    }

    fun testRingTone() {
        viewModelScope.launch {
            try {
                soundManagerUseCase.playTestSound()
            } catch (e: Exception) {
                Logs.e("DeviceSettingsViewModel", "Error playing ringtone: ${e.message}")
            }
        }
    }

    private fun areNotificationAndRingVolumesLinked(): Boolean {
        return try {
            val audioManager = appContext.getSystemService(Context.AUDIO_SERVICE) as AudioManager
            val notificationVolume = audioManager.getStreamVolume(AudioManager.STREAM_NOTIFICATION)
            val ringVolume = audioManager.getStreamVolume(AudioManager.STREAM_RING)

            val testVolume = if (notificationVolume > 0) notificationVolume - 1 else notificationVolume + 1
            val maxNotificationVolume = audioManager.getStreamMaxVolume(AudioManager.STREAM_NOTIFICATION)
            val safeTestVolume = testVolume.coerceIn(0, maxNotificationVolume)
            audioManager.setStreamVolume(AudioManager.STREAM_NOTIFICATION, safeTestVolume, 0)

            val newRingVolume = audioManager.getStreamVolume(AudioManager.STREAM_RING)
            val areLinked = newRingVolume != ringVolume

            audioManager.setStreamVolume(AudioManager.STREAM_NOTIFICATION, notificationVolume, 0)
            audioManager.setStreamVolume(AudioManager.STREAM_RING, ringVolume, 0)

            areLinked
        } catch (e: Exception) {
            Logs.e("DeviceSettingsViewModel", "Error checking linked volumes: ${e.message}")
            false
        }
    }

    private fun getMinPhoneCallVolume(): Int {
        minPhoneCallVolume?.let { return it }
        val audioManager = appContext.getSystemService(Context.AUDIO_SERVICE) as? AudioManager
        val maxVolume = audioManager?.getStreamMaxVolume(AudioManager.STREAM_VOICE_CALL) ?: 1
        val originalVolume = audioManager?.getStreamVolume(AudioManager.STREAM_VOICE_CALL) ?: 0
        return try {
            audioManager?.setStreamVolume(AudioManager.STREAM_VOICE_CALL, 1, 0)
            val actualMinimum = audioManager?.getStreamVolume(AudioManager.STREAM_VOICE_CALL) ?: 0
            audioManager?.setStreamVolume(AudioManager.STREAM_VOICE_CALL, originalVolume, 0)
            val normalized = normalizeVolume(actualMinimum, maxVolume)
            minPhoneCallVolume = normalized
            normalized
        } catch (_: Exception) {
            20
        }
    }

    private fun getPhoneCallVolume(): Int {
        val audioManager = appContext.getSystemService(Context.AUDIO_SERVICE) as? AudioManager
        val current = audioManager?.getStreamVolume(AudioManager.STREAM_VOICE_CALL) ?: 0
        val max = audioManager?.getStreamMaxVolume(AudioManager.STREAM_VOICE_CALL) ?: 1
        val normalized = normalizeVolume(current, max)
        val min = getMinPhoneCallVolume()
        return if (normalized in 0 until min) min else normalized
    }

    private fun getNotificationVolume(): Int {
        val audioManager = appContext.getSystemService(Context.AUDIO_SERVICE) as? AudioManager
        val current = audioManager?.getStreamVolume(AudioManager.STREAM_NOTIFICATION) ?: 0
        val max = audioManager?.getStreamMaxVolume(AudioManager.STREAM_NOTIFICATION) ?: 1
        val normalized = normalizeVolume(current, max)
        return if (_uiState.value.isLinkedVolumes == true && normalized in 0 until 20) 20 else normalized
    }

    private fun getSystemVolume(): Int {
        val audioManager = appContext.getSystemService(Context.AUDIO_SERVICE) as? AudioManager
        val current = audioManager?.getStreamVolume(AudioManager.STREAM_MUSIC) ?: 0
        val max = audioManager?.getStreamMaxVolume(AudioManager.STREAM_MUSIC) ?: 1
        return normalizeVolume(current, max)
    }

    private fun getRingtoneVolume(): Int {
        val audioManager = appContext.getSystemService(Context.AUDIO_SERVICE) as? AudioManager
        val current = audioManager?.getStreamVolume(AudioManager.STREAM_RING) ?: 0
        val max = audioManager?.getStreamMaxVolume(AudioManager.STREAM_RING) ?: 1
        val normalized = normalizeVolume(current, max)
        return if (normalized in 0 until 20) 20 else normalized
    }

    private fun setPhoneCallVolume(newVolume: Int) {
        if (isDoNotDisturbModeEnabled() || isSilentModeEnabled()) return
        val minVolume = getMinPhoneCallVolume()
        val adjusted = if (newVolume in 0 until minVolume) minVolume else newVolume
        try {
            val audioManager = appContext.getSystemService(Context.AUDIO_SERVICE) as AudioManager
            val max = audioManager.getStreamMaxVolume(AudioManager.STREAM_VOICE_CALL)
            audioManager.setStreamVolume(
                AudioManager.STREAM_VOICE_CALL,
                denormalizeVolume(adjusted, max),
                AudioManager.FLAG_SHOW_UI
            )
            refreshAll()
        } catch (e: Exception) {
            Logs.e("DeviceSettingsViewModel", "setPhoneCallVolume: ${e.message}")
        }
    }

    private fun setNotificationVolume(newVolume: Int) {
        if (isDoNotDisturbModeEnabled() || isSilentModeEnabled()) return
        try {
            val audioManager = appContext.getSystemService(Context.AUDIO_SERVICE) as AudioManager
            val max = audioManager.getStreamMaxVolume(AudioManager.STREAM_NOTIFICATION)
            val adjusted = if (_uiState.value.isLinkedVolumes == true && newVolume in 0 until 20) 20 else newVolume
            audioManager.setStreamVolume(
                AudioManager.STREAM_NOTIFICATION,
                denormalizeVolume(adjusted, max),
                AudioManager.FLAG_SHOW_UI
            )
            refreshAll()
        } catch (e: Exception) {
            Logs.e("DeviceSettingsViewModel", "setNotificationVolume: ${e.message}")
        }
    }

    private fun setSystemVolume(newVolume: Int) {
        if (isDoNotDisturbModeEnabled() || isSilentModeEnabled()) return
        try {
            val audioManager = appContext.getSystemService(Context.AUDIO_SERVICE) as AudioManager
            val max = audioManager.getStreamMaxVolume(AudioManager.STREAM_MUSIC)
            audioManager.setStreamVolume(
                AudioManager.STREAM_MUSIC,
                denormalizeVolume(newVolume, max),
                AudioManager.FLAG_SHOW_UI
            )
            refreshAll()
        } catch (e: Exception) {
            Logs.e("DeviceSettingsViewModel", "setSystemVolume: ${e.message}")
        }
    }

    private fun setRingtoneVolume(newVolume: Int) {
        if (isDoNotDisturbModeEnabled() || isSilentModeEnabled()) return
        val adjusted = if (newVolume in 0 until 20) 20 else newVolume
        try {
            val audioManager = appContext.getSystemService(Context.AUDIO_SERVICE) as AudioManager
            val max = audioManager.getStreamMaxVolume(AudioManager.STREAM_RING)
            audioManager.setStreamVolume(
                AudioManager.STREAM_RING,
                denormalizeVolume(adjusted, max),
                AudioManager.FLAG_SHOW_UI
            )
            refreshAll()
        } catch (e: Exception) {
            Logs.e("DeviceSettingsViewModel", "setRingtoneVolume: ${e.message}")
        }
    }

    private fun setBrightness(newBrightness: Int) {
        if (isAutoBrightnessEnabled(appContext)) {
            Settings.System.putInt(
                appContext.contentResolver,
                Settings.System.SCREEN_BRIGHTNESS_MODE,
                Settings.System.SCREEN_BRIGHTNESS_MODE_MANUAL
            )
        }
        try {
            if (Settings.System.canWrite(appContext)) {
                Settings.System.putInt(appContext.contentResolver, Settings.System.SCREEN_BRIGHTNESS, newBrightness)
                refreshAll()
            }
        } catch (e: Exception) {
            Logs.e("DeviceSettingsViewModel", "setBrightness: ${e.message}")
        }
    }

    private fun isDoNotDisturbModeEnabled(): Boolean {
        val nm = appContext.getSystemService(Context.NOTIFICATION_SERVICE) as? NotificationManager
        return nm?.currentInterruptionFilter != NotificationManager.INTERRUPTION_FILTER_ALL
    }

    private fun isSilentModeEnabled(): Boolean {
        val am = appContext.getSystemService(Context.AUDIO_SERVICE) as? AudioManager
        return am?.ringerMode == AudioManager.RINGER_MODE_SILENT
    }

    private fun isAutoBrightnessEnabled(context: Context?): Boolean {
        if (context == null) return false
        return Settings.System.getInt(
            context.contentResolver,
            Settings.System.SCREEN_BRIGHTNESS_MODE,
            Settings.System.SCREEN_BRIGHTNESS_MODE_MANUAL
        ) == Settings.System.SCREEN_BRIGHTNESS_MODE_AUTOMATIC
    }

    private fun normalizeVolume(volume: Int, maxVolume: Int): Int {
        if (maxVolume <= 0) return 0
        return ((volume.toFloat() * 100) / maxVolume).roundToInt().coerceIn(0, 100)
    }

    private fun denormalizeVolume(normalizedVolume: Int, maxVolume: Int): Int {
        if (maxVolume <= 0) return 0
        return ((normalizedVolume.toFloat() * maxVolume) / 100).roundToInt().coerceIn(0, maxVolume)
    }

    private fun getBrightness(): Int {
        return try {
            Settings.System.getInt(appContext.contentResolver, Settings.System.SCREEN_BRIGHTNESS, 0)
                .coerceIn(0, 255)
        } catch (_: Exception) {
            0
        }
    }
}
```

```kotlin
import ifac.td.taxi.R

@Composable
fun DeviceSettingsScreen(
    navController: NavController,
    viewModel: DeviceSettingsViewModel,
    mainActivityViewModel: DeviceSettingsMainActivityViewModel,
) {
    val state by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    LaunchedEffect(Unit) {
        viewModel.onEvent(DeviceSettingsUiEvent.OnResume)
    }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                DeviceSettingsUiEffect.OpenWriteSettings -> {
                    mainActivityViewModel.onOpenWriteSettings()
                }
                is DeviceSettingsUiEffect.ShowToast -> {
                    mainActivityViewModel.onShowToast(effect.messageRes)
                }
                is DeviceSettingsUiEffect.OpenDialog -> {
                    mainActivityViewModel.onOpenDialog(effect.dialog)
                }
                DeviceSettingsUiEffect.CloseDialog -> {
                    mainActivityViewModel.onCloseDialog()
                }
            }
        }
    }

    Scaffold(
        topBar = {
            // preserve header behavior from fragment
            TopAppBar(title = { Text(text = "Device Settings") })
        }
    ) { padding ->
        Column(
            modifier = Modifier
                .padding(padding)
                .fillMaxSize()
                .verticalScroll(rememberScrollState())
                .padding(16.dp),
            verticalArrangement = Arrangement.spacedBy(12.dp)

            DeviceSettingsSliderItem(
                title = stringResource(R.string.system_volume_title),
                description = stringResource(R.string.system_volume_description),

                onValueChangeFinished = { viewModel.onEvent(DeviceSettingsUiEvent.OnSystemVolumeReleased) }
            )

            DeviceSettingsSliderItem(
                title = stringResource(R.string.notification_volume_title),

                enabled = state.buttonsState.notificationVolume.enabled,

                onValueChangeFinished = { viewModel.onEvent(DeviceSettingsUiEvent.OnNotificationVolumeReleased) }

            DeviceSettingsSliderItem(
                title = stringResource(R.string.ringtone_volume_title),
                description = stringResource(R.string.ringtone_volume_description),

                max = 100,
                enabled = state.buttonsState.ringtoneVolume.enabled,

                onValueChangeFinished = { viewModel.onEvent(DeviceSettingsUiEvent.OnRingtoneVolumeReleased) }
            )

            DeviceSettingsSliderItem(
                title = stringResource(R.string.phone_call_volume_title),
                description = stringResource(R.string.phone_call_volume_description),
                value = state.slidersState.phoneCallVolume,
                max = 100,
                enabled = state.buttonsState.phoneCallVolume.enabled,
                onValueChange = { viewModel.onEvent(DeviceSettingsUiEvent.OnPhoneCallVolumeChanged(it, true)) },
                onValueChangeFinished = { viewModel.onEvent(DeviceSettingsUiEvent.OnPhoneCallVolumeReleased) }
            )

            DeviceSettingsSliderItem(
                title = stringResource(R.string.brightness_title),
                description = stringResource(R.string.brightness_description),
                value = state.slidersState.brightness,
                max = state.slidersState.brightnessMax,
                enabled = state.buttonsState.brightness.enabled,
                onValueChange = { viewModel.onEvent(DeviceSettingsUiEvent.OnBrightnessChanged(it, true)) },
                onValueChangeFinished = { viewModel.onEvent(DeviceSettingsUiEvent.OnBrightnessReleased) }
            )
        }
    }

    state.dialogState?.let { dialog ->
        AlertDialog(
            onDismissRequest = { viewModel.onEvent(DeviceSettingsUiEvent.OnDismissDialog) },
            title = { Text(dialog.title) },
            text = { Text(dialog.message) },

                }
```

```kotlin
package ifac.td.taxi.ui.screen.device_settings

import androidx.lifecycle.ViewModel
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.asSharedFlow

sealed interface DeviceSettingsMainActivityEffect {
    data object ShowHeader : DeviceSettingsMainActivityEffect
    data class ShowToast(val messageRes: Int) : DeviceSettingsMainActivityEffect
    data class OpenWriteSettings(val intent: android.content.Intent) : DeviceSettingsMainActivityEffect
    data class OpenDialog(val dialog: DeviceSettingsDialogState) : DeviceSettingsMainActivityEffect
    data object CloseDialog : DeviceSettingsMainActivityEffect
}
```

```kotlin
class DeviceSettingsViewModel(
    application: Application,
    private val soundManagerUseCase: SoundManagerUseCase,
) : AndroidViewModel(application) {

    private val appContext: Context get() = getApplication<Application>()

    private val _uiState = MutableStateFlow(DeviceSettingsUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<DeviceSettingsUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()

    private var minPhoneCallVolume: Int? = null

    fun onEvent(event: DeviceSettingsUiEvent) {
        when (event) {
            DeviceSettingsUiEvent.OnResume -> onResume()
            DeviceSettingsUiEvent.OnRefreshRequested -> refreshAll()
            is DeviceSettingsUiEvent.OnPhoneCallVolumeChanged -> if (event.fromUser) setPhoneCallVolume(event.value)
            is DeviceSettingsUiEvent.OnSystemVolumeChanged -> if (event.fromUser) setSystemVolume(event.value)
            is DeviceSettingsUiEvent.OnNotificationVolumeChanged -> if (event.fromUser) setNotificationVolume(event.value)
            is DeviceSettingsUiEvent.OnRingtoneVolumeChanged -> if (event.fromUser) setRingtoneVolume(event.value)
            is DeviceSettingsUiEvent.OnBrightnessChanged -> if (event.fromUser) setBrightness(event.value)
            DeviceSettingsUiEvent.OnPhoneCallVolumeReleased -> Unit
            DeviceSettingsUiEvent.OnSystemVolumeReleased -> Unit
            DeviceSettingsUiEvent.OnNotificationVolumeReleased -> playBeepForStream(AudioManager.STREAM_NOTIFICATION)
            DeviceSettingsUiEvent.OnRingtoneVolumeReleased -> testRingTone()
            DeviceSettingsUiEvent.OnBrightnessReleased -> Unit
            DeviceSettingsUiEvent.OnOpenWriteSettings -> emitOpenWriteSettings()
            DeviceSettingsUiEvent.OnDismissDialog -> _uiState.value = _uiState.value.copy(dialogState = null)
        }
    }

    fun hasAudioPermissions(): Boolean = Settings.System.canWrite(appContext)

    fun checkLinkedVolumes() {
        viewModelScope.launch {
            val linked = areNotificationAndRingVolumesLinked()
            _uiState.value = _uiState.value.copy(isLinkedVolumes = linked)
            reduceButtons()
        }
    }

    private fun onResume() {
        if (!Settings.System.canWrite(appContext)) {
            viewModelScope.launch { _uiEffect.emit(DeviceSettingsUiEffect.OpenWriteSettings) }
        }
        refreshAll()
        checkLinkedVolumes()
    }

    private fun refreshAll() {
        _uiState.value = _uiState.value.copy(
            slidersState = _uiState.value.slidersState.copy(
                phoneCallVolume = getPhoneCallVolume(),
                systemVolume = getSystemVolume(),
                notificationVolume = getNotificationVolume(),
                ringtoneVolume = getRingtoneVolume(),
                brightness = getBrightness(),
                brightnessMax = 255
            ),
            isWriteSettingsGranted = hasAudioPermissions(),
            isDndEnabled = isDoNotDisturbModeEnabled(),
            isSilentModeEnabled = isSilentModeEnabled(),
        )
        reduceButtons()
    }

    private fun reduceButtons() {
        _uiState.value = _uiState.value.copy(
            buttonsState = reduceDeviceSettingsButtonsState(
                current = _uiState.value.buttonsState,
                isWriteSettingsGranted = _uiState.value.isWriteSettingsGranted,
                isDndEnabled = _uiState.value.isDndEnabled,
                isSilentModeEnabled = _uiState.value.isSilentModeEnabled,
                isLinkedVolumes = _uiState.value.isLinkedVolumes
            )
        )
    }

    private fun emitOpenWriteSettings() {
        viewModelScope.launch { _uiEffect.emit(DeviceSettingsUiEffect.OpenWriteSettings) }
    }

    private fun playBeepForStream(streamType: Int) {
        viewModelScope.launch {
            withContext(Dispatchers.Main) {
                try {
                    val tone = ToneGenerator(streamType, ToneGenerator.MAX_VOLUME)
                    tone.startTone(ToneGenerator.TONE_PROP_BEEP)
                    withContext(Dispatchers.Default) {
                        delay(300)
                        tone.stopTone()
                        tone.release()
                    }
                } catch (e: Exception) {
                    Logs.e("DeviceSettingsViewModel", "playBeepForStream: $e")
                }
            }
        }
    }

    fun testRingTone() {
        viewModelScope.launch {
            try {
                soundManagerUseCase.playTestSound()
            } catch (e: Exception) {
                Logs.e("DeviceSettingsViewModel", "Error playing ringtone: ${e.message}")
            }
        }
    }

    private fun areNotificationAndRingVolumesLinked(): Boolean {
        return try {
            val audioManager = appContext.getSystemService(Context.AUDIO_SERVICE) as AudioManager
            val notificationVolume = audioManager.getStreamVolume(AudioManager.STREAM_NOTIFICATION)
            val ringVolume = audioManager.getStreamVolume(AudioManager.STREAM_RING)

            val testVolume = if (notificationVolume > 0) notificationVolume - 1 else notificationVolume + 1
            val maxNotificationVolume = audioManager.getStreamMaxVolume(AudioManager.STREAM_NOTIFICATION)
            val safeTestVolume = testVolume.coerceIn(0, maxNotificationVolume)
            audioManager.setStreamVolume(AudioManager.STREAM_NOTIFICATION, safeTestVolume, 0)

            val newRingVolume = audioManager.getStreamVolume(AudioManager.STREAM_RING)
            val areLinked = newRingVolume != ringVolume

            audioManager.setStreamVolume(AudioManager.STREAM_NOTIFICATION, notificationVolume, 0)
            audioManager.setStreamVolume(AudioManager.STREAM_RING, ringVolume, 0)

            areLinked
        } catch (e: Exception) {
            Logs.e("DeviceSettingsViewModel", "Error checking linked volumes: ${e.message}")
            false
        }
    }

    private fun getMinPhoneCallVolume(): Int {
        minPhoneCallVolume?.let { return it }
        val audioManager = appContext.getSystemService(Context.AUDIO_SERVICE) as? AudioManager
        val maxVolume = audioManager?.getStreamMaxVolume(AudioManager.STREAM_VOICE_CALL) ?: 1
        val originalVolume = audioManager?.getStreamVolume(AudioManager.STREAM_VOICE_CALL) ?: 0
        return try {
            audioManager?.setStreamVolume(AudioManager.STREAM_VOICE_CALL, 1, 0)
            val actualMinimum = audioManager?.getStreamVolume(AudioManager.STREAM_VOICE_CALL) ?: 0
            audioManager?.setStreamVolume(AudioManager.STREAM_VOICE_CALL, originalVolume, 0)
            val normalized = normalizeVolume(actualMinimum, maxVolume)
            minPhoneCallVolume = normalized
            normalized
        } catch (_: Exception) {
            20
        }
    }

    private fun getPhoneCallVolume(): Int {
        val audioManager = appContext.getSystemService(Context.AUDIO_SERVICE) as? AudioManager
        val current = audioManager?.getStreamVolume(AudioManager.STREAM_VOICE_CALL) ?: 0
        val max = audioManager?.getStreamMaxVolume(AudioManager.STREAM_VOICE_CALL) ?: 1
        val normalized = normalizeVolume(current, max)
        val min = getMinPhoneCallVolume()
        return if (normalized in 0 until min) min else normalized
    }

    private fun getNotificationVolume(): Int {
        val audioManager = appContext.getSystemService(Context.AUDIO_SERVICE) as? AudioManager
        val current = audioManager?.getStreamVolume(AudioManager.STREAM_NOTIFICATION) ?: 0
        val max = audioManager?.getStreamMaxVolume(AudioManager.STREAM_NOTIFICATION) ?: 1
        val normalized = normalizeVolume(current, max)
        return if (_uiState.value.isLinkedVolumes == true && normalized in 0 until 20) 20 else normalized
    }

    private fun getSystemVolume(): Int {
        val audioManager = appContext.getSystemService(Context.AUDIO_SERVICE) as? AudioManager
        val current = audioManager?.getStreamVolume(AudioManager.STREAM_MUSIC) ?: 0
        val max = audioManager?.getStreamMaxVolume(AudioManager.STREAM_MUSIC) ?: 1
        return normalizeVolume(current, max)
    }

    private fun getRingtoneVolume(): Int {
        val audioManager = appContext.getSystemService(Context.AUDIO_SERVICE) as? AudioManager
        val current = audioManager?.getStreamVolume(AudioManager.STREAM_RING) ?: 0
        val max = audioManager?.getStreamMaxVolume(AudioManager.STREAM_RING) ?: 1
        val normalized = normalizeVolume(current, max)
        return if (normalized in 0 until 20) 20 else normalized
    }

    private fun setPhoneCallVolume(newVolume: Int) {
        if (isDoNotDisturbModeEnabled() || isSilentModeEnabled()) return
        val minVolume = getMinPhoneCallVolume()
        val adjusted = if (newVolume in 0 until minVolume) minVolume else newVolume
        try {
            val audioManager = appContext.getSystemService(Context.AUDIO_SERVICE) as AudioManager
            val max = audioManager.getStreamMaxVolume(AudioManager.STREAM_VOICE_CALL)
            audioManager.setStreamVolume(
                AudioManager.STREAM_VOICE_CALL,
                denormalizeVolume(adjusted, max),
                AudioManager.FLAG_SHOW_UI
            )
            refreshAll()
        } catch (e: Exception) {
            Logs.e("DeviceSettingsViewModel", "setPhoneCallVolume: ${e.message}")
        }
    }

    private fun setNotificationVolume(newVolume: Int) {
        if (isDoNotDisturbModeEnabled() || isSilentModeEnabled()) return
        try {
            val audioManager = appContext.getSystemService(Context.AUDIO_SERVICE) as AudioManager
            val max = audioManager.getStreamMaxVolume(AudioManager.STREAM_NOTIFICATION)
            val adjusted = if (_uiState.value.isLinkedVolumes == true && newVolume in 0 until 20) 20 else newVolume
            audioManager.setStreamVolume(
                AudioManager.STREAM_NOTIFICATION,
                denormalizeVolume(adjusted, max),
                AudioManager.FLAG_SHOW_UI
            )
            refreshAll()
        } catch (e: Exception) {
            Logs.e("DeviceSettingsViewModel", "setNotificationVolume: ${e.message}")
        }
    }

    private fun setSystemVolume(newVolume: Int) {
        if (isDoNotDisturbModeEnabled() || isSilentModeEnabled()) return
        try {
            val audioManager = appContext.getSystemService(Context.AUDIO_SERVICE) as AudioManager
            val max = audioManager.getStreamMaxVolume(AudioManager.STREAM_MUSIC)
            audioManager.setStreamVolume(
                AudioManager.STREAM_MUSIC,
                denormalizeVolume(newVolume, max),
                AudioManager.FLAG_SHOW_UI
            )
            refreshAll()
        } catch (e: Exception) {
            Logs.e("DeviceSettingsViewModel", "setSystemVolume: ${e.message}")
        }
    }

    private fun setRingtoneVolume(newVolume: Int) {
        if (isDoNotDisturbModeEnabled() || isSilentModeEnabled()) return
        val adjusted = if (newVolume in 0 until 20) 20 else newVolume
        try {
            val audioManager = appContext.getSystemService(Context.AUDIO_SERVICE) as AudioManager
            val max = audioManager.getStreamMaxVolume(AudioManager.STREAM_RING)
            audioManager.setStreamVolume(
                AudioManager.STREAM_RING,
                denormalizeVolume(adjusted, max),
                AudioManager.FLAG_SHOW_UI
            )
            refreshAll()
        } catch (e: Exception) {
            Logs.e("DeviceSettingsViewModel", "setRingtoneVolume: ${e.message}")
        }
    }

    private fun setBrightness(newBrightness: Int) {
        if (isAutoBrightnessEnabled(appContext)) {
            Settings.System.putInt(
                appContext.contentResolver,
                Settings.System.SCREEN_BRIGHTNESS_MODE,
                Settings.System.SCREEN_BRIGHTNESS_MODE_MANUAL
            )
        }
        try {
            if (Settings.System.canWrite(appContext)) {
                Settings.System.putInt(appContext.contentResolver, Settings.System.SCREEN_BRIGHTNESS, newBrightness)
                refreshAll()
            }
        } catch (e: Exception) {
            Logs.e("DeviceSettingsViewModel", "setBrightness: ${e.message}")
        }
    }

    private fun isDoNotDisturbModeEnabled(): Boolean {
        val nm = appContext.getSystemService(Context.NOTIFICATION_SERVICE) as? NotificationManager
        return nm?.currentInterruptionFilter != NotificationManager.INTERRUPTION_FILTER_ALL
    }

    private fun isSilentModeEnabled(): Boolean {
        val am = appContext.getSystemService(Context.AUDIO_SERVICE) as? AudioManager
        return am?.ringerMode == AudioManager.RINGER_MODE_SILENT
    }

    private fun isAutoBrightnessEnabled(context: Context?): Boolean {
        if (context == null) return false
        return Settings.System.getInt(
            context.contentResolver,
            Settings.System.SCREEN_BRIGHTNESS_MODE,
            Settings.System.SCREEN_BRIGHTNESS_MODE_MANUAL
        ) == Settings.System.SCREEN_BRIGHTNESS_MODE_AUTOMATIC
    }

    private fun normalizeVolume(volume: Int, maxVolume: Int): Int {
        if (maxVolume <= 0) return 0
        return ((volume.toFloat() * 100) / maxVolume).roundToInt().coerceIn(0, 100)
    }

    private fun denormalizeVolume(normalizedVolume: Int, maxVolume: Int): Int {
        if (maxVolume <= 0) return 0
        return ((normalizedVolume.toFloat() * maxVolume) / 100).roundToInt().coerceIn(0, maxVolume)
    }
}

                onValueChangeFinished = { viewModel.onEvent(DeviceSettingsUiEvent.OnPhoneCallVolumeReleased) }
            )

            DeviceSettingsSliderItem(
                title = stringResource(R.string.brightness_title),
                description = stringResource(R.string.brightness_description),
                value = state.slidersState.brightness,
                max = state.slidersState.brightnessMax,
                enabled = state.buttonsState.brightness.enabled,
                onValueChange = { viewModel.onEvent(DeviceSettingsUiEvent.OnBrightnessChanged(it, true)) },
                onValueChangeFinished = { viewModel.onEvent(DeviceSettingsUiEvent.OnBrightnessReleased) }
            )
        }
    }

    state.dialogState?.let { dialog ->
        AlertDialog(
            onDismissRequest = { viewModel.onEvent(DeviceSettingsUiEvent.OnDismissDialog) },
            title = { Text(dialog.title) },
            text = { Text(dialog.message) },
            confirmButton = {
                TextButton(onClick = { viewModel.onEvent(DeviceSettingsUiEvent.OnDismissDialog) }) {
                    Text(dialog.confirmText)
                }
            },
            dismissButton = {
                TextButton(onClick = { viewModel.onEvent(DeviceSettingsUiEvent.OnDismissDialog) }) {
                    Text(dialog.dismissText)
                }
            }
        )
    }
}
```

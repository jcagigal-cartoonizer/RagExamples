```kotlin
fun reduceMeetingSignButtonsState(
    current: MeetingSignButtonsState,
    event: MeetingSignButtonsEvent,
): MeetingSignButtonsState {
    return when (event) {
        MeetingSignButtonsEvent.CloseMenu -> current.copy(
            isMenuOpen = false,
            fabOptions = current.fabOptions.copy(rotation = 0f),
            fabDispatch = current.fabDispatch.copy(
                visible = false,
                alpha = 0f,
                translationY = 100f,
            ),
            fabEdit = current.fabEdit.copy(
                visible = false,
                alpha = 0f,
                translationY = 100f,
            ),
        )

        MeetingSignButtonsEvent.OpenMenu -> current.copy(
            isMenuOpen = true,
            fabOptions = current.fabOptions.copy(rotation = 180f),
            fabDispatch = current.fabDispatch.copy(
                visible = true,
                alpha = 1f,
                translationY = 0f,
            ),
            fabEdit = current.fabEdit.copy(
                visible = true,
                alpha = 1f,
                translationY = 0f,
            ),
        )

        MeetingSignButtonsEvent.ToggleMenu ->
            if (current.isMenuOpen) reduceMeetingSignButtonsState(current, MeetingSignButtonsEvent.CloseMenu)
            else reduceMeetingSignButtonsState(current, MeetingSignButtonsEvent.OpenMenu)

        MeetingSignButtonsEvent.SetDispatchMode -> current.copy(
            dispatchAction = MeetingSignButtonsState.DispatchAction.NavigateToInfoDispatch,
            fabDispatch = current.fabDispatch.copy(
                iconRes = R.drawable.baseline_info_outline_192,
                backgroundTint = 0xFFFFFFFF.toInt(),
            )
        )

        MeetingSignButtonsEvent.SetBackMode -> current.copy(
            dispatchAction = MeetingSignButtonsState.DispatchAction.NavigateBack,
            fabDispatch = current.fabDispatch.copy(
                iconRes = R.drawable.back,
                backgroundTint = 0xFFFFFFFF.toInt(),
            )
        )
    }
}
```

## 2) Compose ViewModel + MainActivity bridge

This keeps the original `MeetingSignViewModel` responsibilities, but exposes Compose-friendly state/effects.

```kotlin
package ifac.td.taxi.viewmodel.compose

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.R
import ifac.td.taxi.domain.usecase.MeetingSignUseCase
import ifac.td.taxi.domain.usecase.ShiftStatusUseCase
import ifac.td.taxi.ui.screen.compose.*
import ifac.td.taxi.framework.util.Logs
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch

class MeetingSignViewModel(
    application: Application,
    private val meetingSignUseCase: MeetingSignUseCase,
    private val shiftStatusUseCase: ShiftStatusUseCase,
) : AndroidViewModel(application) {

    private val TAG = "MeetingSignViewModel"

    private val _uiState = MutableStateFlow(MeetingSignUiState())
    val uiState: StateFlow<MeetingSignUiState> = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<MeetingSignUiEffect>(extraBufferCapacity = 1)
    val effects: SharedFlow<MeetingSignUiEffect> = _effects.asSharedFlow()

    init {
        viewModelScope.launch {
            val name = meetingSignUseCase.getMessageSignName()
            _uiState.update { it.copy(meetingSignText = name) }
            if (name.isNullOrBlank()) {
                _effects.tryEmit(MeetingSignUiEffect.OpenEditDialog(isCancellable = false))
            }
        }
    }

    fun onTextChanged(text: String) {
        viewModelScope.launch {
            _uiState.update { it.copy(meetingSignText = text) }
            meetingSignUseCase.setMessageSignText(text)
        }
    }

    fun onFabOptionsClicked() {
        _uiState.update {
            it.copy(buttons = reduceMeetingSignButtonsState(it.buttons, MeetingSignButtonsEvent.ToggleMenu))
        }
    }

    fun onFabEditClicked() {
        _effects.tryEmit(MeetingSignUiEffect.OpenEditDialog(isCancellable = true))
    }

    fun onFabDispatchClicked(fromDispatch: Boolean, isInSettings: Boolean) {
        if (fromDispatch && !isInSettings) {
            _effects.tryEmit(MeetingSignUiEffect.NavigateToInfoDispatch)
        } else {
            _effects.tryEmit(MeetingSignUiEffect.NavigateBack)
        }
    }

    fun setDispatchMode() {
        _uiState.update {
            it.copy(buttons = reduceMeetingSignButtonsState(it.buttons, MeetingSignButtonsEvent.SetDispatchMode))
        }
    }

    fun setBackMode() {
        _uiState.update {
            it.copy(buttons = reduceMeetingSignButtonsState(it.buttons, MeetingSignButtonsEvent.SetBackMode))
        }
    }

    fun checkShiftStatusChangeToHired(lastStatus: Int?, currentStatus: Int?): Boolean {
        lastStatus?.let { lstStatus ->
            currentStatus?.let { crntStatus ->
                Logs.d(TAG, "check change to hired last status $lstStatus , currentStatus $currentStatus")
                return shiftStatusUseCase.isVacant(lstStatus) && shiftStatusUseCase.isHired(crntStatus)
            }
        }
        return false
    }

    fun onShiftStatusChanged(
        lastStatus: Int?,
        currentStatus: Int?,
        fromDispatch: Boolean,
        isInSettings: Boolean,
    ) {
        if (checkShiftStatusChangeToHired(lastStatus, currentStatus)) {
            if (!isInSettings && fromDispatch) {
                _effects.tryEmit(MeetingSignUiEffect.NavigateToInfoDispatch)
            }
        }
    }
}
```

```kotlin
@Composable
fun MeetingSignScreen(
    viewModel: MeetingSignViewModel,
    sharedViewModel: MeetingSignMainActivityViewModelBridge,
    navControllerProvider: () -> NavController?,
    isInSettingsProvider: () -> Boolean,
    onShowHeader: (Boolean) -> Unit,
    onShowBottomBar: (Boolean) -> Unit,
    onRequestedOrientation: (Int) -> Unit,
    onNavigateToInfoDispatch: () -> Unit,
    onNavigateBack: () -> Unit,
    onShowToast: (Int) -> Unit,
    onOpenDialog: (CustomDialog.CustomDialogModel, (CustomDialog.CustomDialogResponse) -> Unit) -> Unit,
    textColorArg: Int = 0,
    backgroundColorArg: Int = 0,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val lifecycleOwner = LocalLifecycleOwner.current

    LaunchedEffect(Unit) {
        onShowHeader(false)
        onShowBottomBar(false)
        onRequestedOrientation(ActivityInfo.SCREEN_ORIENTATION_LANDSCAPE)
    }

    val tripState by sharedViewModel.tripFlow.collectAsStateWithLifecycle(initialValue = null)
    val shiftState by sharedViewModel.shiftStatusFlow.collectAsStateWithLifecycle(initialValue = null)

    LaunchedEffect(tripState?.fromDispatch, shiftState?.currentStatus) {
        viewModel.onShiftStatusChanged(
            lastStatus = null,
            currentStatus = shiftState?.currentStatus,
            fromDispatch = tripState?.fromDispatch == true,
            isInSettings = isInSettingsProvider(),
        )
    }

    LaunchedEffect(Unit) {
        lifecycleOwner.repeatOnLifecycle(androidx.lifecycle.Lifecycle.State.STARTED) {
            viewModel.effects.collect { effect ->
                when (effect) {
                    MeetingSignUiEffect.NavigateBack -> onNavigateBack()
                    MeetingSignUiEffect.NavigateToInfoDispatch -> onNavigateToInfoDispatch()
                    is MeetingSignUiEffect.ShowToast -> onShowToast(effect.messageRes)
                    is MeetingSignUiEffect.OpenEditDialog -> {
                        val currentName = uiState.meetingSignText?.takeIf { it.isNotBlank() }
                        onOpenDialog(
                            CustomDialog.CustomDialogModel(
                                title = "Edit message sign",
                                hint = "Message sign",
                                editText = currentName,
                                buttons = arrayListOf(ButtonType.CANCEL, ButtonType.ACCEPT),
                                isCancellable = effect.isCancellable
                            )
                        ) { response ->
                            if (response.buttonPressed == ButtonType.ACCEPT) {
                                if (!response.editTextString.isNullOrBlank()) {
                                    viewModel.onTextChanged(response.editTextString!!)
                                } else {
                                    onShowToast(R.string.editText_error_no_text)
                                }
                            }
                        }
                    }
                }
            }
        }
    }

    val rootBg = if (backgroundColorArg != 0) Color(backgroundColorArg) else Color.White
    val textColor = if (textColorArg != 0) Color(textColorArg) else Color.Black

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(rootBg)
    ) {
        MeetingSignText(
            text = uiState.meetingSignText ?: "",
            color = textColor,
            modifier = Modifier.fillMaxSize()
        )

        MeetingSignFabCluster(
            buttonsState = uiState.buttons,
            onOptionsClick = { viewModel.onFabOptionsClicked() },
            onEditClick = { viewModel.onFabEditClicked() },
            onDispatchClick = {
                viewModel.onFabDispatchClicked(
                    fromDispatch = tripState?.fromDispatch == true,
                    isInSettings = isInSettingsProvider()
                )
            },
            modifier = Modifier
                .align(Alignment.BottomEnd)
                .padding(end = 32.dp, bottom = 32.dp)
        )
    }
}

@Composable
private fun MeetingSignText(
    text: String,
    color: Color,
    modifier: Modifier = Modifier,
) {
    AndroidView(
        modifier = modifier,
        factory = { context ->
            TextView(context).apply {
                setTextColor(color.value.toInt())
                gravity = android.view.Gravity.CENTER
                setBackgroundColor(android.graphics.Color.TRANSPARENT)
            }
        },
        update = { tv ->
            tv.setTextColor(color.value.toInt())
            tv.post {
                tv.setSignTextCompose(text)
            }
        }
    )
}

private fun TextView.setSignTextCompose(text: String) {
    post {
        val availableWidth = width - paddingLeft - paddingRight
        val availableHeight = height - paddingTop - paddingBottom
        if (availableWidth <= 0 || availableHeight <= 0) {
            setText(text)
            return@post
        }

        val words = text.split(" ")
        val paint = TextPaint(paint)

        var low = 10f
        var high = 1000f
        var bestSize = low

        while (low <= high) {
            val mid = (low + high) / 2
            paint.textSize = mid

            val wordTooWide = words.any { paint.measureText(it) > availableWidth }
            if (wordTooWide) {
                high = mid - 1
                continue
            }

            val layout = StaticLayout.Builder
                .obtain(text, 0, text.length, paint, availableWidth)
                .setAlignment(Layout.Alignment.ALIGN_CENTER)
                .setIncludePad(false)
                .build()

            if (layout.height <= availableHeight) {
                bestSize = mid
                low = mid + 1
            } else {
                high = mid - 1
            }
        }

        setTextSize(TypedValue.COMPLEX_UNIT_PX, bestSize)
        setText(text)
    }
}

@Composable
private fun MeetingSignFabCluster(
    buttonsState: MeetingSignButtonsState,
    onOptionsClick: () -> Unit,
    onEditClick: () -> Unit,
    onDispatchClick: () -> Unit,
    modifier: Modifier = Modifier,
) {
    Column(
        modifier = modifier,
        horizontalAlignment = Alignment.End,
        verticalArrangement = Arrangement.spacedBy(8.dp)
    ) {
        AnimatedFab(
            visible = buttonsState.fabDispatch.visible,
            alpha = buttonsState.fabDispatch.alpha,
            translationY = buttonsState.fabDispatch.translationY,
            rotation = buttonsState.fabDispatch.rotation,
            iconRes = buttonsState.fabDispatch.iconRes,
            size = buttonsState.fabDispatch.size,
            onClick = onDispatchClick
        )

        AnimatedFab(
            visible = buttonsState.fabEdit.visible,
            alpha = buttonsState.fabEdit.alpha,
            translationY = buttonsState.fabEdit.translationY,
            rotation = buttonsState.fabEdit.rotation,
            iconRes = buttonsState.fabEdit.iconRes,
            size = buttonsState.fabEdit.size,
            onClick = onEditClick
        )

        AnimatedFab(
            visible = true,
            alpha = buttonsState.fabOptions.alpha,
            translationY = buttonsState.fabOptions.translationY,
            rotation = buttonsState.fabOptions.rotation,
            iconRes = buttonsState.fabOptions.iconRes,
            size = buttonsState.fabOptions.size,
            onClick = onOptionsClick
        )
    }
}

@Composable
private fun AnimatedFab(
    visible: Boolean,
    alpha: Float,
    translationY: Float,
    rotation: Float,
    @androidx.annotation.DrawableRes iconRes: Int,
    size: FabSize,
    onClick: () -> Unit,
) {
    if (!visible) {
        Spacer(modifier = Modifier.size(0.dp))
        return
    }

    val animatedAlpha by animateFloatAsState(targetValue = alpha, label = "fabAlpha")
    val animatedTranslation by animateFloatAsState(targetValue = translationY, label = "fabTranslation")
    val animatedRotation by animateFloatAsState(targetValue = rotation, label = "fabRotation")

    val fabSize = if (size == FabSize.MINI) 40.dp else 56.dp

    FloatingActionButton(
        onClick = onClick,
        modifier = Modifier
            .size(fabSize)
            .alpha(animatedAlpha)
            .offset(y = animatedTranslation.dp),
        containerColor = Color.White,
        contentColor = Color.Unspecified
    ) {
        Icon(
            painter = androidx.compose.ui.res.painterResource(id = iconRes),
            contentDescription = null,
            modifier = Modifier.rotate(animatedRotation)
        )
    }
}
```

```kotlin
                        alpha = 0f,
                        translationY = 100f
                    ),
                    fabEdit = current.fabEdit.copy(
                        visible = false,
                        alpha = 0f,
                        translationY = 100f
                    )
                )
            } else {
                current.copy(
                    isMenuOpen = true,
                    fabOptions = current.fabOptions.copy(rotation = 180f),
                    fabDispatch = current.fabDispatch.copy(
                        visible = true,
                        alpha = 1f,
                        translationY = 0f
                    ),
                    fabEdit = current.fabEdit.copy(
                        visible = true,
                        alpha = 1f,
                        translationY = 0f
                    )
                )
            }
        }

        MeetingSignUiEvent.MenuClosed -> current.copy(
            isMenuOpen = false,
            fabOptions = current.fabOptions.copy(rotation = 0f),
            fabDispatch = current.fabDispatch.copy(
                visible = false,
                alpha = 0f,
                translationY = 100f
            ),
            fabEdit = current.fabEdit.copy(
                visible = false,
                alpha = 0f,
                translationY = 100f
            )
        )

        MeetingSignUiEvent.MenuOpened -> current.copy(
            isMenuOpen = true,
            fabOptions = current.fabOptions.copy(rotation = 180f),
            fabDispatch = current.fabDispatch.copy(
                visible = true,
                alpha = 1f,
                translationY = 0f
            ),
            fabEdit = current.fabEdit.copy(
                visible = true,
                alpha = 1f,
                translationY = 0f
            )
        )

        else -> current
    }
}

class MeetingSignViewModel(
    context: Application,
    private val meetingSignUseCase: MeetingSignUseCase,
    private val shiftStatusUseCase: ShiftStatusUseCase,
) : BaseViewModel(context) {

    private val TAG = "MeetingSignViewModel"

    private val _uiState = MutableStateFlow(MeetingSignUiState())
    val uiState: StateFlow<MeetingSignUiState> = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<MeetingSignUiEffect>(extraBufferCapacity = 1)
    val effects: SharedFlow<MeetingSignUiEffect> = _effects.asSharedFlow()

    private val _meetingSignNameFlow = MutableStateFlow<String?>(null)
    val meetingSignNameFlow = _meetingSignNameFlow.asStateFlow()

    init {
        viewModelScope.launch {
            val name = meetingSignUseCase.getMessageSignName()
            _meetingSignNameFlow.emit(name)
            _uiState.update { it.copy(meetingSignText = name) }
            if (name.isNullOrBlank()) {
                _effects.tryEmit(MeetingSignUiEffect.OpenEditDialog(isCancellable = false))
            }
        }
    }

    fun changeMessageSignText(text: String) {
        viewModelScope.launch {
            _meetingSignNameFlow.emit(text)
            _uiState.update { it.copy(meetingSignText = text) }
            meetingSignUseCase.setMessageSignText(text)
        }
    }

    fun onEvent(event: MeetingSignUiEvent) {
        when (event) {
            MeetingSignUiEvent.ToggleMenu,
            MeetingSignUiEvent.MenuClosed,
            MeetingSignUiEvent.MenuOpened -> {
                _uiState.update { it.copy(buttonsState = reduceMeetingSignButtonsState(it.buttonsState, event)) }
            }

            MeetingSignUiEvent.EditClicked -> {
                _effects.tryEmit(MeetingSignUiEffect.OpenEditDialog(isCancellable = true))
            }

            is MeetingSignUiEvent.DispatchClicked -> {
                if (event.fromDispatch && !event.isInSettings) {
                    _effects.tryEmit(MeetingSignUiEffect.NavigateToInfoDispatch)
                } else {
                    _effects.tryEmit(MeetingSignUiEffect.NavigateBack)
                }
            }

            is MeetingSignUiEvent.TextChanged -> changeMessageSignText(event.text)

            is MeetingSignUiEvent.ShiftStatusChanged -> {
                if (checkShiftStatusChangeToHired(event.lastStatus, event.currentStatus)) {
                    if (!event.isInSettings && event.fromDispatch) {
                        _effects.tryEmit(MeetingSignUiEffect.NavigateToInfoDispatch)
                    }
                }
            }
        }
    }

    fun checkShiftStatusChangeToHired(lastStatus: Int?, currentStatus: Int?): Boolean {
        lastStatus?.let { lstStatus ->
            currentStatus?.let { crntStatus ->
                Logs.d(TAG, "check change to hired last status $lstStatus , currentStatus $currentStatus")
                return shiftStatusUseCase.isVacant(lstStatus) && shiftStatusUseCase.isHired(crntStatus)
            }
        }
        return false
    }
}

@Composable
fun MeetingSignScreen(
    viewModel: MeetingSignViewModel,
    sharedTripFromDispatch: Boolean,
    sharedShiftCurrentStatus: Int?,
    isInSettings: Boolean,
    textColorArg: Int = 0,
    backgroundColorArg: Int = 0,
    onShowHeader: (Boolean) -> Unit,
    onShowBottomBar: (Boolean) -> Unit,
    onRequestedOrientation: (Int) -> Unit,
    onNavigateToInfoDispatch: () -> Unit,
    onNavigateBack: () -> Unit,
    onShowToast: (Int) -> Unit,
    onOpenDialog: (CustomDialog.CustomDialogModel, (CustomDialog.CustomDialogResponse) -> Unit) -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val lifecycleOwner = LocalLifecycleOwner.current

    LaunchedEffect(Unit) {
        onShowHeader(false)
        onShowBottomBar(false)
        onRequestedOrientation(ActivityInfo.SCREEN_ORIENTATION_LANDSCAPE)
    }

    LaunchedEffect(sharedShiftCurrentStatus, sharedTripFromDispatch, isInSettings) {
        viewModel.onEvent(
            MeetingSignUiEvent.ShiftStatusChanged(
                lastStatus = null,
                currentStatus = sharedShiftCurrentStatus,
                fromDispatch = sharedTripFromDispatch,
                isInSettings = isInSettings
            )
        )
    }

    LaunchedEffect(Unit) {
        lifecycleOwner.repeatOnLifecycle(Lifecycle.State.STARTED) {
            viewModel.effects.collect { effect ->
                when (effect) {
                    MeetingSignUiEffect.NavigateBack -> onNavigateBack()
                    MeetingSignUiEffect.NavigateToInfoDispatch -> onNavigateToInfoDispatch()
                    is MeetingSignUiEffect.ShowToast -> onShowToast(effect.messageRes)
                    is MeetingSignUiEffect.OpenEditDialog -> {
                        val currentName = uiState.meetingSignText?.takeIf { it.isNotBlank() }
                        onOpenDialog(
                            CustomDialog.CustomDialogModel(
                                title = "Edit message sign",
                                hint = "Message sign",
                                editText = currentName,
                                buttons = arrayListOf(ButtonType.CANCEL, ButtonType.ACCEPT),
                                isCancellable = effect.isCancellable
                            )
                        ) { response ->
                            if (response.buttonPressed == ButtonType.ACCEPT) {
                                if (!response.editTextString.isNullOrBlank()) {
                                    viewModel.onEvent(MeetingSignUiEvent.TextChanged(response.editTextString!!))
                                } else {
                                    onShowToast(R.string.editText_error_no_text)
                                }
                            }
                        }
                    }
                }
            }
        }
    }

    val rootBg = if (backgroundColorArg != 0) Color(backgroundColorArg) else Color.White
    val textColor = if (textColorArg != 0) Color(textColorArg) else Color.Black

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(rootBg)
    ) {
        MeetingSignText(
            text = uiState.meetingSignText.orEmpty(),
            color = textColor,
            modifier = Modifier.fillMaxSize()
        )

        MeetingSignFabCluster(
            buttonsState = uiState.buttonsState,
            onOptionsClick = { viewModel.onEvent(MeetingSignUiEvent.ToggleMenu) },
            onEditClick = { viewModel.onEvent(MeetingSignUiEvent.EditClicked) },
            onDispatchClick = {
                viewModel.onEvent(
                    MeetingSignUiEvent.DispatchClicked(
                        fromDispatch = sharedTripFromDispatch,
                        isInSettings = isInSettings
                    )
                )
            },
            modifier = Modifier
                .align(Alignment.BottomEnd)
                .size(0.dp) // container only; actual padding is inside cluster
        )
    }
}

@Composable
private fun MeetingSignText(
    text: String,
    color: Color,
    modifier: Modifier = Modifier,
) {
    AndroidView(
        modifier = modifier,
        factory = { context ->
            TextView(context).apply {
                gravity = android.view.Gravity.CENTER
                setTextColor(color.value.toInt())
            }
        },
        update = { tv ->
            tv.setTextColor(color.value.toInt())
            tv.post { tv.setSignTextCompose(text) }
        }
    )
}

private fun TextView.setSignTextCompose(text: String) {
    post {
        val availableWidth = width - paddingLeft - paddingRight
        val availableHeight = height - paddingTop - paddingBottom

        if (availableWidth <= 0 || availableHeight <= 0) {
            setText(text)
            return@post
        }

        val words = text.split(" ")
        val paint = TextPaint(paint)

        var low = 10f
        var high = 1000f
        var bestSize = low

        while (low <= high) {
            val mid = (low + high) / 2
            paint.textSize = mid

            val wordTooWide = words.any { paint.measureText(it) > availableWidth }
            if (wordTooWide) {
                high = mid - 1
                continue
            }

            val layout = StaticLayout.Builder
                .obtain(text, 0, text.length, paint, availableWidth)
                .setAlignment(Layout.Alignment.ALIGN_CENTER)
                .setIncludePad(false)
                .build()

            if (layout.height <= availableHeight) {
                bestSize = mid
                low = mid + 1
            } else {
                high = mid - 1
            }
        }

        setTextSize(TypedValue.COMPLEX_UNIT_PX, bestSize)
        setText(text)
    }
}

@Composable
private fun MeetingSignFabCluster(

@Composable
private fun AnimatedFab(
    visible: Boolean,
    alpha: Float,
    translationY: Float,
    rotation: Float,
    @DrawableRes iconRes: Int,
    size: FabSize,
    onClick: () -> Unit,
) {
    if (!visible) {
        Spacer(modifier = Modifier.size(0.dp))
        return
    }

    val animatedAlpha by animateFloatAsState(targetValue = alpha, label = "fabAlpha")
    val animatedTranslation by animateFloatAsState(targetValue = translationY, label = "fabTranslation")
    val animatedRotation by animateFloatAsState(targetValue = rotation, label = "fabRotation")

    val fabSize = if (size == FabSize.MINI) 40.dp else 56.dp

    FloatingActionButton(
        onClick = onClick,
        containerColor = Color.White,
        modifier = Modifier
            .size(fabSize)
            .alpha(animatedAlpha)
            .offset(y = animatedTranslation.dp)
    ) {
        Icon(
            painter = painterResource(id = iconRes),
            contentDescription = null,
            modifier = Modifier
                .size(if (size == FabSize.MINI) 20.dp else 24.dp)
        )
    }
}
```

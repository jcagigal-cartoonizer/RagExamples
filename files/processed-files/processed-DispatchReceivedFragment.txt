### `DispatchReceivedComposeViewModel.kt`
```kotlin
package ifac.td.taxi.ui.screen.dispatchreceived
// # Block 96-3: import android.app.Application
import android.app.Application
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.domain.usecase.DispatchUseCase
import ifac.td.taxi.domain.usecase.UserPreferencesUseCase
import ifac.td.taxi.framework.sdk.usecase.LicensingUseCase
import ifac.td.taxi.repository.room.dao.BravoConfigurationVariableDao
import ifac.td.taxi.viewmodel.BaseViewModel
import ifac.td.taxi.viewmodel.model.UtilsModel.Companion.toDispatchReceivedModel
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch
class DispatchReceivedComposeViewModel(
    private val dispatchUseCase: DispatchUseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    private val bravoConfigurationVariableDao: BravoConfigurationVariableDao,
    private val licensingUsecase: LicensingUseCase,
    context: Application,
) : BaseViewModel(context) {
    private val _uiState = MutableStateFlow(DispatchReceivedUiState())
    val uiState = _uiState.asStateFlow()
    private val _uiEffect = MutableSharedFlow<DispatchReceivedUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()
    private var actualDispatch: DispatchReceivedModelUi? = null
    private var idDispatch: Long? = null
    init {
        viewModelScope.launch {
            bravoConfigurationVariableDao.getBravoConfigurationVariable()?.let {
                // you can map to state if needed
            }
        }
    }
    fun onScreenStarted(id: Long) {
        getDispatch(id)
        checkCancelDialogPermission()
        checkIsDirRecogidaAceptaDespacho()
    }
    fun getDispatch(id: Long) {
        viewModelScope.launch {
            idDispatch = id
            val dispatch = dispatchUseCase.getDispatchById(id)
            if (dispatch != null) {
                val mapped = dispatch.toDispatchReceivedModel(id)
                actualDispatch = DispatchReceivedModelUi(
                    idDispatch = id,
                    pickUpTime = mapped.pickUpTime,
                    pickUpZone = mapped.pickUpZone,
                    smallPickUpAdress = mapped.smallPickUpAdress,
                    city = mapped.city,
                    passengerName = mapped.passengerName
                )
                _uiState.update { it.copy(dispatch = actualDispatch) }
            }
        }
    }
    fun checkCancelDialogPermission() {
        viewModelScope.launch {
            val preferences = userPreferencesUseCase.getUserPreferences()
            val showDialog = preferences?.showConfRejectDispatch == true
            _uiState.update { it.copy(showDialog = showDialog) }
        }
    }
    fun checkIsDirRecogidaAceptaDespacho() {
        viewModelScope.launch {
            val preferences = licensingUsecase.getLicensingParameters()
            val canShowPickupAddress = preferences?.isAddressTakenAcceptsDispatch == true
            _uiState.update { it.copy(canShowPickupAddress = canShowPickupAddress) }
        }
    }
    fun onAcceptClick() {
        viewModelScope.launch {
            // preserve navigation/logic from old VM
            acceptDispatch()
            stopCallTimer()
        }
    }
    fun onCancelClick() {
        viewModelScope.launch {
            val showDialog = uiState.value.showDialog
            if (showDialog) {
                _uiEffect.emit(
                    DispatchReceivedUiEffect.ShowDialog(
                        CustomDialogState(
                            title = "Advertencia",
                            description = "¿Quieres rechazar este despacho?",
                            acceptText = "Aceptar",
                            cancelText = "Cancelar"
                        )
                    )
                )
            } else {
                rejectManualDispatch()
                stopCallTimer()
            }
        }
    }
    fun onDialogAcceptedCancelDispatch() {
        viewModelScope.launch {
            rejectManualDispatch()
            stopCallTimer()
            _uiEffect.emit(DispatchReceivedUiEffect.DismissDialog)
        }
    }
    fun onDialogDismissed() {
        viewModelScope.launch {
            _uiEffect.emit(DispatchReceivedUiEffect.DismissDialog)
        }
    }
    fun requestPhoneCallPermissionIfNeeded() {
        viewModelScope.launch {
            _uiEffect.emit(DispatchReceivedUiEffect.RequestPhoneCallPermission)
        }
    }
    fun navigateToInfoDispatch(tripId: Long) {
        viewModelScope.launch {
            _uiEffect.emit(DispatchReceivedUiEffect.NavigateToInfoDispatch(tripId))
        }
    }
    fun acceptDispatch() {
        // call your domain/usecase implementation here
    }
    fun rejectManualDispatch() {
        // call your domain/usecase implementation here
    }
    fun stopCallTimer() {
        // keep existing behavior here
    }
    fun getDispatchProvider(idDispatch: Long, dataToString: List<String>) {
        viewModelScope.launch(Dispatchers.IO) {
            var provider: String? = null
            for (str in dataToString.reversed()) {
                val trimmed = str.trim()
                if (trimmed.startsWith("OPERADOR:")) {
                    provider = trimmed.substringAfter("OPERADOR:").trim()
                    break
                }
            }
            provider?.let {
                dispatchUseCase.updateDispatchProvider(idDispatch = idDispatch, provider = it)
            }
        }
    }
}
```

### `DispatchReceivedScreen.kt`
```kotlin
package ifac.td.taxi.ui.screen.dispatchreceived
// # Block 249-4: import androidx.compose.foundation.layout.*
import androidx.compose.foundation.layout.*
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Modifier
import androidx.compose.ui.unit.dp
import androidx.lifecycle.Lifecycle
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.lifecycle.compose.LifecycleStartEffect
import androidx.navigation.NavController
@Composable
fun DispatchReceivedScreen(
    navController: NavController,
    idDispatch: Long,
    viewModel: DispatchReceivedComposeViewModel,
    onRequestPhonePermission: () -> Unit = {}
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val dialogState = uiState.dialogState
    LaunchedEffect(idDispatch) {
        viewModel.onScreenStarted(idDispatch)
    }
    LaunchedEffect(Unit) {
        viewModel.uiEffect.collect { effect ->
            when (effect) {
                is DispatchReceivedUiEffect.NavigateBack -> navController.navigateUp()
                is DispatchReceivedUiEffect.NavigateToInfoDispatch -> {
                    // preserve Jetpack Navigation behavior
                    // navController.navigate(...)
                }
                is DispatchReceivedUiEffect.RequestPhoneCallPermission -> onRequestPhonePermission()
                is DispatchReceivedUiEffect.ShowDialog -> {
                    // dialog is state-driven; handled below
                }
                is DispatchReceivedUiEffect.DismissDialog -> {
                    // handled by dialog state cleanup if you store it
                }
            }
        }
    }
    Box(modifier = Modifier.fillMaxSize()) {
        DispatchReceivedContent(
            uiState = uiState,
            onAccept = viewModel::onAcceptClick,
            onCancel = viewModel::onCancelClick
        )
        if (dialogState != null) {
            CustomDialog(
                state = dialogState,
                onDismiss = viewModel::onDialogDismissed,
                onAccept = viewModel::onDialogAcceptedCancelDispatch
            )
        }
    }
}
```

```kotlin
        if (state.accept.visible) {
            CustomActionButton(
                modifier = Modifier.weight(1f),
                text = state.accept.text,
                containerColor = state.accept.containerColor,
                contentColor = state.accept.contentColor,
                borderColor = state.accept.borderColor,
                enabled = state.accept.enabled,
                onClick = onAccept
            )
        }
    }
}

    contentColor: androidx.compose.ui.graphics.Color,
    borderColor: androidx.compose.ui.graphics.Color,
    enabled: Boolean,
    onClick: () -> Unit,
    shape: Shape = RoundedCornerShape(12.dp)
) {
    Button(
        modifier = modifier
            .height(52.dp),
        onClick = onClick,
        enabled = enabled,
        shape = shape,
```

```kotlin
                    style = MaterialTheme.typography.bodyMedium,
                    textAlign = TextAlign.Start

                    modifier = Modifier.fillMaxWidth(),
```

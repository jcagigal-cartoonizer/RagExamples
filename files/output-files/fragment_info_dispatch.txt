## 1) Compose Theme object matching `theme.xml`

This is a reusable Compose theme that maps typical app theme properties such as colors, typography, and spacing. It uses your XML resources where possible.

```kotlin
package com.example.ui.theme

import androidx.compose.foundation.isSystemInDarkTheme
import androidx.compose.material3.MaterialTheme
import androidx.compose.material3.darkColorScheme
import androidx.compose.material3.lightColorScheme
import androidx.compose.runtime.Composable
import androidx.compose.ui.graphics.Color

private val LightColorScheme = lightColorScheme(
    primary = Color(0xFF1976D2),
    onPrimary = Color.White,
    secondary = Color(0xFF03A9F4),
    onSecondary = Color.White,
    background = Color.White,
    onBackground = Color(0xFF111111),
    surface = Color.White,
    onSurface = Color(0xFF111111)
)

private val DarkColorScheme = darkColorScheme(
    primary = Color(0xFF90CAF9),
    onPrimary = Color.Black,
    secondary = Color(0xFF4FC3F7),
    onSecondary = Color.Black,
    background = Color(0xFF121212),
    onBackground = Color.White,
    surface = Color(0xFF1E1E1E),
    onSurface = Color.White
)

object SmartTDTheme {
    @Composable
    operator fun invoke(
        darkTheme: Boolean = isSystemInDarkTheme(),
        content: @Composable () -> Unit
    ) {
        val colors = if (darkTheme) DarkColorScheme else LightColorScheme

        MaterialTheme(
            colorScheme = colors,
            typography = Typography,
            content = content
        )
    }
}
```

## 4) Supporting composables

These replace repeated `LinearLayout + ImageView + TextView` patterns from the XML.

```kotlin
@Composable
private fun InfoPill(
    text: String,
    modifier: Modifier = Modifier
) {
    Row(
        modifier = modifier,
        verticalAlignment = Alignment.CenterVertically
    ) {
        Box(
            modifier = Modifier
                .size(30.dp)
                .background(Color.LightGray, RoundedCornerShape(15.dp))
        )
        Spacer(Modifier.width(5.dp))
        Text(text = text, style = MaterialTheme.typography.bodyLarge)
    }
}

@Composable
private fun MenuButtonCell(
    text: String,
    modifier: Modifier = Modifier,
    onClick: () -> Unit = {}
) {
    Box(
        modifier = modifier
            .fillMaxHeight()
            .padding(1.dp)
            .background(Color.DarkGray)
            .clickable { onClick() },
        contentAlignment = Alignment.Center
    ) {
        Text(
            text = text,
            color = Color.White,
            style = MaterialTheme.typography.labelLarge,
            textAlign = TextAlign.Center
        )
    }
}
```

## Notes / mapping from XML to Compose

- `NestedScrollView` → `Column` with `verticalScroll(...)`
- `RecyclerView` → usually `LazyColumn`; I used simple `forEach` placeholders for clarity
- `GridLayout` → `Column` with two `Row`s, each with three weighted cells
- `ConstraintLayout` structures → simplified to `Row`, `Column`, and `Box`
- `visibility="gone"` → conditional `if (...) { ... }`
- `@dimen/paddingHorizontalNestedScroll` → `dimensionResource(R.dimen.paddingHorizontalNestedScroll)`
- `@string/...` → `stringResource(...)`
- `@color/...` → `Color(...)` or `colorResource(...)`

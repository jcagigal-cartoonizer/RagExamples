## 1) High-level idea

Your code editor launches the agent as a subprocess, then talks to it like this:

- editor writes JSON-RPC requests to the agent’s **stdin**
- agent reads them, processes them, and writes JSON-RPC responses to **stdout**
- logs should go to **stderr** so they do not corrupt JSON-RPC messages

A typical flow:

- `initialize`
- `tools/list`
- `chat/completions` or `agent/run` style method
- optional `shutdown`

Because “ACP-compatible” can vary depending on the exact protocol version, the safest approach is to implement a **JSON-RPC 2.0 transport** and expose ACP-like methods.

## 4) Example implementation

### `Main.java`

```java
package example;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ObjectNode;
import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiChatModel;

import java.io.*;
import java.nio.charset.StandardCharsets;
import java.util.*;

public class AcpCodeAgent {

    private static final ObjectMapper MAPPER = new ObjectMapper();

    public static void main(String[] args) throws Exception {
        String apiKey = System.getenv("OPENAI_API_KEY");
        if (apiKey == null || apiKey.isBlank()) {
            System.err.println("OPENAI_API_KEY is not set");
            System.exit(1);
        }

        ChatLanguageModel model = OpenAiChatModel.builder()
                .apiKey(apiKey)
                .modelName("gpt-4o-mini")
                .build();

        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in, StandardCharsets.UTF_8));
        BufferedWriter writer = new BufferedWriter(new OutputStreamWriter(System.out, StandardCharsets.UTF_8));

        String line;
        while ((line = reader.readLine()) != null) {
            try {
                JsonNode request = MAPPER.readTree(line);
                JsonNode response = handleRequest(request, model);
                if (response != null) {
                    writer.write(MAPPER.writeValueAsString(response));
                    writer.write("\n");
                    writer.flush();
                }
            } catch (Exception e) {
                // If parsing fails and we can't recover request id, log to stderr only
                System.err.println("Error handling request: " + e.getMessage());
            }
        }
    }

    private static JsonNode handleRequest(JsonNode request, ChatLanguageModel model) {
        String jsonrpc = request.path("jsonrpc").asText();
        String method = request.path("method").asText(null);
        JsonNode idNode = request.get("id");

        if (!"2.0".equals(jsonrpc) || method == null) {
            return errorResponse(idNode, -32600, "Invalid Request");
        }

        switch (method) {
            case "initialize":
                return successResponse(idNode, initializeResult());

            case "tools/list":
                return successResponse(idNode, toolsListResult());

            case "chat":
                return successResponse(idNode, chatResult(request, model));

            case "shutdown":
                return successResponse(idNode, Map.of("ok", true));

            default:
                return errorResponse(idNode, -32601, "Method not found: " + method);
        }
    }

    private static Map<String, Object> initializeResult() {
        return Map.of(
                "protocolVersion", "0.1",
                "serverInfo", Map.of(
                        "name", "langchain4j-openai-agent",
                        "version", "1.0.0"
                ),
                "capabilities", Map.of(
                        "tools", true,
                        "chat", true
                )
        );
    }

    private static Map<String, Object> toolsListResult() {
        return Map.of(
                "tools", List.of(
                        Map.of(
                                "name", "read_file",
                                "description", "Read a file from the local workspace",
                                "inputSchema", Map.of(
                                        "type", "object",
                                        "properties", Map.of(
                                                "path", Map.of("type", "string")
                                        ),
                                        "required", List.of("path")
                                )
                        ),
                        Map.of(
                                "name", "write_file",
                                "description", "Write content to a file in the local workspace",
                                "inputSchema", Map.of(
                                        "type", "object",
                                        "properties", Map.of(
                                                "path", Map.of("type", "string"),
                                                "content", Map.of("type", "string")
                                        ),
                                        "required", List.of("path", "content")
                                )
                        )
                )
        );
    }

    private static Map<String, Object> chatResult(JsonNode request, ChatLanguageModel model) {
        JsonNode params = request.path("params");
        String prompt = params.path("prompt").asText("");
        if (prompt.isBlank()) {
            return Map.of("error", "Missing prompt");
        }

        // Simple single-turn example
        String answer = model.generate(prompt);

        return Map.of(
                "message", answer
        );
    }

    private static ObjectNode successResponse(JsonNode idNode, Object result) {
        ObjectNode response = MAPPER.createObjectNode();
        response.put("jsonrpc", "2.0");
        response.set("id", idNode == null ? MAPPER.nullNode() : idNode);
        response.set("result", MAPPER.valueToTree(result));
        return response;
    }

    private static ObjectNode errorResponse(JsonNode idNode, int code, String message) {
        ObjectNode response = MAPPER.createObjectNode();
        response.put("jsonrpc", "2.0");
        response.set("id", idNode == null ? MAPPER.nullNode() : idNode);

        ObjectNode error = MAPPER.createObjectNode();
        error.put("code", code);
        error.put("message", message);

        response.set("error", error);
        return response;
    }
}
```

## 8) Important stdio rules

When using stdio for JSON-RPC:

- **stdout**: only JSON-RPC messages
- **stderr**: logs, debug output, stack traces
- use line-delimited JSON unless your editor expects another framing format
- avoid printing anything extra to stdout, even startup banners

If your editor’s ACP implementation expects a different framing than newline-delimited JSON, adapt the transport accordingly.

# 5) Tool implementation

## `WorkspaceTools.java`

These are local tools the model can call.

```java
package example;

import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.*;
import java.util.List;
import java.util.stream.Collectors;
import java.util.stream.Stream;

public class WorkspaceTools {

    private final Path workspaceRoot;

    public WorkspaceTools(Path workspaceRoot) {
        this.workspaceRoot = workspaceRoot.toAbsolutePath().normalize();
    }

    public String readFile(String path) throws IOException {
        Path resolved = resolveSafe(path);
        return Files.readString(resolved, StandardCharsets.UTF_8);
    }

    public void writeFile(String path, String content) throws IOException {
        Path resolved = resolveSafe(path);
        Files.createDirectories(resolved.getParent());
        Files.writeString(resolved, content, StandardCharsets.UTF_8,
                StandardOpenOption.CREATE, StandardOpenOption.TRUNCATE_EXISTING);
    }

    public List<String> listFiles(String path) throws IOException {
        Path resolved = resolveSafe(path);
        try (Stream<Path> stream = Files.list(resolved)) {
            return stream
                    .map(p -> workspaceRoot.relativize(p.toAbsolutePath().normalize()).toString())
                    .collect(Collectors.toList());
        }
    }

    private Path resolveSafe(String path) {
        Path resolved = workspaceRoot.resolve(path).normalize();
        if (!resolved.startsWith(workspaceRoot)) {
            throw new IllegalArgumentException("Path escapes workspace root: " + path);
        }
        return resolved;
    }
}
```

# 7) ACP-style JSON-RPC protocol

This is the protocol layer the editor talks to.

## `AcpProtocol.java`

```java
package example;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ObjectNode;

import java.util.List;
import java.util.Map;

public class AcpProtocol {

    private static final ObjectMapper MAPPER = new ObjectMapper();

    public static JsonNode handle(JsonNode req, ProtocolHandler handler) {
        String jsonrpc = req.path("jsonrpc").asText();
        String method = req.path("method").asText(null);
        JsonNode id = req.get("id");

        if (!"2.0".equals(jsonrpc) || method == null) {
            return error(id, -32600, "Invalid Request");
        }

        try {
            return switch (method) {
                case "initialize" -> ok(id, handler.initialize());
                case "tools/list" -> ok(id, handler.toolsList());
                case "tools/call" -> ok(id, handler.toolsCall(req.path("params")));
                case "chat/stream" -> ok(id, handler.chatStream(req.path("params")));
                case "shutdown" -> ok(id, Map.of("ok", true));
                default -> error(id, -32601, "Method not found: " + method);
            };
        } catch (Exception e) {
            return error(id, -32000, e.getMessage());
        }
    }

    private static ObjectNode ok(JsonNode id, Object result) {
        ObjectNode node = MAPPER.createObjectNode();
        node.put("jsonrpc", "2.0");
        node.set("id", id == null ? MAPPER.nullNode() : id);
        node.set("result", MAPPER.valueToTree(result));
        return node;
    }

    private static ObjectNode error(JsonNode id, int code, String message) {
        ObjectNode node = MAPPER.createObjectNode();
        node.put("jsonrpc", "2.0");
        node.set("id", id == null ? MAPPER.nullNode() : id);

        ObjectNode err = MAPPER.createObjectNode();
        err.put("code", code);
        err.put("message", message);
        node.set("error", err);
        return node;
    }

    public interface ProtocolHandler {
        Map<String, Object> initialize();
        Map<String, Object> toolsList();
        Map<String, Object> toolsCall(JsonNode params);
        Map<String, Object> chatStream(JsonNode params);
    }
}
```

# 8) Streaming transport

The transport writes JSON messages to stdout and flushes immediately.

## `JsonRpcTransport.java`

```java
package example;

import com.fasterxml.jackson.databind.ObjectMapper;

import java.io.BufferedWriter;
import java.io.IOException;
import java.io.OutputStreamWriter;
import java.nio.charset.StandardCharsets;

public class JsonRpcTransport {

    private static final ObjectMapper MAPPER = new ObjectMapper();
    private final BufferedWriter writer;

    public JsonRpcTransport() {
        this.writer = new BufferedWriter(new OutputStreamWriter(System.out, StandardCharsets.UTF_8));
    }

    public synchronized void send(Object message) throws IOException {
        writer.write(MAPPER.writeValueAsString(message));
        writer.write("\n");
        writer.flush();
    }
}
```

# 9) Main application with streaming + tool-calling

This version:
- receives `chat/stream`
- sends partial output events
- uses langchain4j streaming model
- supports tool calls through langchain4j tool execution
- emits JSON-RPC notifications

## `Main.java`

```java
package example;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ObjectNode;
import dev.langchain4j.model.chat.StreamingChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiStreamingChatModel;

import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.nio.charset.StandardCharsets;
import java.nio.file.Path;
import java.util.List;
import java.util.Map;

public class Main {

    private static final ObjectMapper MAPPER = new ObjectMapper();

    public static void main(String[] args) throws Exception {
        String apiKey = System.getenv("OPENAI_API_KEY");
        String workspace = System.getenv().getOrDefault("WORKSPACE_ROOT", ".");
        if (apiKey == null || apiKey.isBlank()) {
            System.err.println("OPENAI_API_KEY is missing");
            System.exit(1);
        }

        Path workspaceRoot = Path.of(workspace);
        WorkspaceTools workspaceTools = new WorkspaceTools(workspaceRoot);

        StreamingChatLanguageModel model = OpenAiStreamingChatModel.builder()
                .apiKey(apiKey)
                .modelName("gpt-4o-mini")
                .build();

        AgentService.CodingAgent agent = AgentService.create(model, workspaceTools);
        JsonRpcTransport transport = new JsonRpcTransport();

        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in, StandardCharsets.UTF_8));
        String line;

        while ((line = reader.readLine()) != null) {
            try {
                JsonNode request = MAPPER.readTree(line);
                JsonNode response = AcpProtocol.handle(request, new AcpProtocol.ProtocolHandler() {
                    @Override
                    public Map<String, Object> initialize() {
                        return Map.of(
                                "protocolVersion", "0.1",
                                "serverInfo", Map.of(
                                        "name", "acp-langchain4j-agent",
                                        "version", "1.0.0"
                                ),
                                "capabilities", Map.of(
                                        "tools", true,
                                        "chatStreaming", true
                                )
                        );
                    }

                    @Override
                    public Map<String, Object> toolsList() {
                        return Map.of(
                                "tools", List.of(
                                        Map.of(
                                                "name", "read_file",
                                                "description", "Read a file from the workspace",
                                                "inputSchema", Map.of(
                                                        "type", "object",
                                                        "properties", Map.of(
                                                                "path", Map.of("type", "string")
                                                        ),
                                                        "required", List.of("path")
                                                )
                                        ),
                                        Map.of(
                                                "name", "write_file",
                                                "description", "Write content to a file in the workspace",
                                                "inputSchema", Map.of(
                                                        "type", "object",
                                                        "properties", Map.of(
                                                                "path", Map.of("type", "string"),
                                                                "content", Map.of("type", "string")
                                                        ),
                                                        "required", List.of("path", "content")
                                                )
                                        ),
                                        Map.of(
                                                "name", "list_files",
                                                "description", "List files in a directory",
                                                "inputSchema", Map.of(
                                                        "type", "object",
                                                        "properties", Map.of(
                                                                "path", Map.of("type", "string")
                                                        ),
                                                        "required", List.of("path")
                                                )
                                        )
                                )
                        );
                    }

                    @Override
                    public Map<String, Object> toolsCall(JsonNode params) {
                        String name = params.path("name").asText();
                        JsonNode arguments = params.path("arguments");

                        try {
                            return switch (name) {
                                case "read_file" -> Map.of("content",
                                        workspaceTools.readFile(arguments.path("path").asText()));
                                case "write_file" -> {
                                    workspaceTools.writeFile(
                                            arguments.path("path").asText(),
                                            arguments.path("content").asText()
                                    );
                                    yield Map.of("status", "ok");
                                }
                                case "list_files" -> Map.of("files",
                                        workspaceTools.listFiles(arguments.path("path").asText()));
                                default -> Map.of("error", "Unknown tool: " + name);
                            };
                        } catch (Exception e) {
                            return Map.of("error", e.getMessage());
                        }
                    }

                    @Override
                    public Map<String, Object> chatStream(JsonNode params) {
                        String requestId = params.path("requestId").asText("");
                        String prompt = params.path("prompt").asText("");

                        try {
                            // Notify editor that generation started
                            transport.send(notification("chat/started", Map.of(
                                    "requestId", requestId
                            )));

                            // For streaming, we let the model produce output incrementally.
                            // langchain4j streaming callbacks are model-specific; here we show the pattern.
                            StringBuilder full = new StringBuilder();

                            // Simple approach: use the agent synchronously for final answer.
                            // In a real implementation, wire the streaming callbacks from the model.
                            String answer = agent.chat(prompt);

                            // Send incremental chunks (simulated here by chunking the final answer)
                            for (String chunk : chunk(answer, 40)) {
                                full.append(chunk);
                                transport.send(notification("chat/partial", Map.of(
                                        "requestId", requestId,
                                        "delta", chunk
                                )));
                            }

                            transport.send(notification("chat/final", Map.of(
                                    "requestId", requestId,
                                    "message", full.toString()
                            )));

                            return Map.of(
                                    "requestId", requestId,
                                    "status", "completed"
                            );
                        } catch (Exception e) {
                            try {
                                transport.send(notification("chat/error", Map.of(
                                        "requestId", requestId,
                                        "message", e.getMessage()
                                )));
                            } catch (Exception ignored) {}
                            return Map.of("error", e.getMessage());
                        }
                    }
                });

                if (response != null) {
                    transport.send(response);
                }
            } catch (Exception e) {
                System.err.println("Failed to process request: " + e.getMessage());
            }
        }
    }

    private static ObjectNode notification(String method, Map<String, Object> params) {
        ObjectNode node = MAPPER.createObjectNode();
        node.put("jsonrpc", "2.0");
        node.put("method", method);
        node.set("params", MAPPER.valueToTree(params));
        return node;
    }

    private static List<String> chunk(String text, int size) {
        java.util.ArrayList<String> chunks = new java.util.ArrayList<>();
        for (int i = 0; i < text.length(); i += size) {
            chunks.add(text.substring(i, Math.min(text.length(), i + size)));
        }
        return chunks;
    }
}
```

# 3) Workspace tools

These are the actual file operations.

## `src/main/java/example/WorkspaceTools.java`

```java
package example;

import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.*;
import java.util.List;
import java.util.stream.Collectors;
import java.util.stream.Stream;

public class WorkspaceTools {

    private final Path root;

    public WorkspaceTools(Path root) {
        this.root = root.toAbsolutePath().normalize();
    }

    public String readFile(String path) {
        try {
            return Files.readString(resolve(path), StandardCharsets.UTF_8);
        } catch (IOException e) {
            throw new RuntimeException(e);
        }
    }

    public String writeFile(String path, String content) {
        try {
            Path p = resolve(path);
            Path parent = p.getParent();
            if (parent != null) {
                Files.createDirectories(parent);
            }
            Files.writeString(p, content, StandardCharsets.UTF_8,
                    StandardOpenOption.CREATE, StandardOpenOption.TRUNCATE_EXISTING);
            return "OK";
        } catch (IOException e) {
            throw new RuntimeException(e);
        }
    }

    public String listFiles(String path) {
        try {
            Path dir = resolve(path);
            try (Stream<Path> stream = Files.list(dir)) {
                List<String> files = stream
                        .map(p -> root.relativize(p.toAbsolutePath().normalize()).toString())
                        .collect(Collectors.toList());
                return String.join("\n", files);
            }
        } catch (IOException e) {
            throw new RuntimeException(e);
        }
    }

    private Path resolve(String path) {
        Path resolved = root.resolve(path).normalize();
        if (!resolved.startsWith(root)) {
            throw new IllegalArgumentException("Path escapes workspace root: " + path);
        }
        return resolved;
    }
}
```

# 6) Protocol handler

## `src/main/java/example/Protocol.java`

```java
package example;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ObjectNode;

import java.util.List;
import java.util.Map;

public class Protocol {

    private static final ObjectMapper MAPPER = new ObjectMapper();

    public interface Handler {
        Map<String, Object> initialize();
        Map<String, Object> toolsList();
        Map<String, Object> chatStream(JsonNode params);
        Map<String, Object> shutdown();
    }

    public static JsonNode handle(JsonNode req, Handler handler) {
        String jsonrpc = req.path("jsonrpc").asText();
        String method = req.path("method").asText(null);
        JsonNode id = req.get("id");

        if (!"2.0".equals(jsonrpc) || method == null) {
            return error(id, -32600, "Invalid Request");
        }

        try {
            return switch (method) {
                case "initialize" -> ok(id, handler.initialize());
                case "tools/list" -> ok(id, handler.toolsList());
                case "chat/stream" -> ok(id, handler.chatStream(req.path("params")));
                case "shutdown" -> ok(id, handler.shutdown());
                default -> error(id, -32601, "Method not found: " + method);
            };
        } catch (Exception e) {
            return error(id, -32000, e.getMessage());
        }
    }

    private static ObjectNode ok(JsonNode id, Object result) {
        ObjectNode node = MAPPER.createObjectNode();
        node.put("jsonrpc", "2.0");
        node.set("id", id == null ? MAPPER.nullNode() : id);
        node.set("result", MAPPER.valueToTree(result));
        return node;
    }

    private static ObjectNode error(JsonNode id, int code, String message) {
        ObjectNode node = MAPPER.createObjectNode();
        node.put("jsonrpc", "2.0");
        node.set("id", id == null ? MAPPER.nullNode() : id);

        ObjectNode err = MAPPER.createObjectNode();
        err.put("code", code);
        err.put("message", message);
        node.set("error", err);
        return node;
    }
}
```

## `src/main/java/example/Main.java`

```java
package example;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ObjectNode;
import dev.langchain4j.data.message.AiMessage;
import dev.langchain4j.data.message.UserMessage;
import dev.langchain4j.model.chat.StreamingChatLanguageModel;
import dev.langchain4j.model.output.Response;
import dev.langchain4j.model.output.TokenStream;
import dev.langchain4j.model.output.TokenStreamHandler;

import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.nio.charset.StandardCharsets;
import java.nio.file.Path;
import java.util.List;
import java.util.Map;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.atomic.AtomicReference;

public class Main {

    private static final ObjectMapper MAPPER = new ObjectMapper();

    public static void main(String[] args) throws Exception {
        String apiKey = System.getenv("OPENAI_API_KEY");
        String workspace = System.getenv().getOrDefault("WORKSPACE_ROOT", ".");
        if (apiKey == null || apiKey.isBlank()) {
            System.err.println("OPENAI_API_KEY is missing");
            System.exit(1);
        }

        WorkspaceTools workspaceTools = new WorkspaceTools(Path.of(workspace));
        StreamingChatLanguageModel streamingModel = AgentFactory.createStreamingModel(apiKey);
        StreamingAssistant assistant = AgentFactory.createAssistant(apiKey, workspaceTools);
        JsonRpcTransport transport = new JsonRpcTransport();

        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in, StandardCharsets.UTF_8));
        String line;

        while ((line = reader.readLine()) != null) {
            JsonNode req;
            try {
                req = MAPPER.readTree(line);
            } catch (Exception e) {
                System.err.println("Invalid JSON: " + e.getMessage());
                continue;
            }

            JsonNode response = Protocol.handle(req, new Protocol.Handler() {
                @Override
                public Map<String, Object> initialize() {
                    return Map.of(
                            "protocolVersion", "0.1",
                            "serverInfo", Map.of(
                                    "name", "acp-langchain4j-agent",
                                    "version", "1.0.0"
                            ),
                            "capabilities", Map.of(
                                    "tools", true,
                                    "chatStreaming", true
                            )
                    );
                }

                @Override
                public Map<String, Object> toolsList() {
                    return Map.of(
                            "tools", List.of(
                                    Map.of(
                                            "name", "read_file",
                                            "description", "Read a file from the workspace",
                                            "inputSchema", Map.of(
                                                    "type", "object",
                                                    "properties", Map.of(
                                                            "path", Map.of("type", "string")
                                                    ),
                                                    "required", List.of("path")
                                            )
                                    ),
                                    Map.of(
                                            "name", "write_file",
                                            "description", "Write content to a file in the workspace",
                                            "inputSchema", Map.of(
                                                    "type", "object",
                                                    "properties", Map.of(
                                                            "path", Map.of("type", "string"),
                                                            "content", Map.of("type", "string")
                                                    ),
                                                    "required", List.of("path", "content")
                                            )
                                    ),
                                    Map.of(
                                            "name", "list_files",
                                            "description", "List files in a directory",
                                            "inputSchema", Map.of(
                                                    "type", "object",
                                                    "properties", Map.of(
                                                            "path", Map.of("type", "string")
                                                    ),
                                                    "required", List.of("path")
                                            )
                                    )
                            )
                    );
                }

                @Override
                public Map<String, Object> chatStream(JsonNode params) {
                    String requestId = params.path("requestId").asText();
                    String prompt = params.path("prompt").asText();

                    try {
                        transport.send(notification("chat/started", Map.of(
                                "requestId", requestId
                        )));

                        // REAL streaming callbacks:
                        // We use the streaming model directly and collect tokens as they arrive.
                        CountDownLatch done = new CountDownLatch(1);
                        StringBuilder full = new StringBuilder();
                        AtomicReference<Throwable> errorRef = new AtomicReference<>();

                        TokenStream tokenStream = streamingModel.generate(
                                UserMessage.from(prompt),
                                new TokenStreamHandler() {
                                    @Override
                                    public void onPartialResponse(String partialResponse) {
                                        full.append(partialResponse);
                                        try {
                                            transport.send(notification("chat/partial", Map.of(
                                                    "requestId", requestId,
                                                    "delta", partialResponse
                                            )));
                                        } catch (Exception e) {
                                            errorRef.set(e);
                                        }
                                    }

                                    @Override
                                    public void onCompleteResponse(Response<AiMessage> response) {
                                        done.countDown();
                                    }

                                    @Override
                                    public void onError(Throwable error) {
                                        errorRef.set(error);
                                        done.countDown();
                                    }
                                }
                        );

                        tokenStream.start();

                        done.await();

                        if (errorRef.get() != null) {
                            transport.send(notification("chat/error", Map.of(
                                    "requestId", requestId,
                                    "message", errorRef.get().getMessage()
                            )));
                            return Map.of("error", errorRef.get().getMessage());
                        }

                        transport.send(notification("chat/final", Map.of(
                                "requestId", requestId,
                                "message", full.toString()
                        )));

                        return Map.of(
                                "requestId", requestId,
                                "status", "completed"
                        );
                    } catch (Exception e) {
                        try {
                            transport.send(notification("chat/error", Map.of(
                                    "requestId", requestId,
                                    "message", e.getMessage()
                            )));
                        } catch (Exception ignored) {}
                        return Map.of("error", e.getMessage());
                    }
                }

                @Override
                public Map<String, Object> shutdown() {
                    return Map.of("ok", true);
                }
            });

            try {
                transport.send(response);
            } catch (Exception e) {
                System.err.println("Failed to send response: " + e.getMessage());
            }
        }
    }

    private static ObjectNode notification(String method, Map<String, Object> params) {
        ObjectNode node = MAPPER.createObjectNode();
        node.put("jsonrpc", "2.0");
        node.put("method", method);
        node.set("params", MAPPER.valueToTree(params));
        return node;
    }
}
```

# 5) Workspace-safe tools

## `WorkspaceTools.java`

```java
package example;

import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.*;
import java.util.List;
import java.util.stream.Collectors;
import java.util.stream.Stream;

public class WorkspaceTools {

    private final Path root;

    public WorkspaceTools(Path root) {
        this.root = root.toAbsolutePath().normalize();
    }

    public String readFile(String path) {
        try {
            return Files.readString(resolve(path), StandardCharsets.UTF_8);
        } catch (IOException e) {
            throw new RuntimeException(e.getMessage(), e);
        }
    }

    public String writeFile(String path, String content) {
        try {
            Path p = resolve(path);
            Path parent = p.getParent();
            if (parent != null) Files.createDirectories(parent);
            Files.writeString(p, content, StandardCharsets.UTF_8,
                    StandardOpenOption.CREATE, StandardOpenOption.TRUNCATE_EXISTING);
            return "OK";
        } catch (IOException e) {
            throw new RuntimeException(e.getMessage(), e);
        }
    }

    public String listFiles(String path) {
        try {
            Path dir = resolve(path);
            try (Stream<Path> s = Files.list(dir)) {
                List<String> files = s
                        .map(p -> root.relativize(p.toAbsolutePath().normalize()).toString())
                        .collect(Collectors.toList());
                return String.join("\n", files);
            }
        } catch (IOException e) {
            throw new RuntimeException(e.getMessage(), e);
        }
    }

    private Path resolve(String path) {
        Path resolved = root.resolve(path).normalize();
        if (!resolved.startsWith(root)) {
            throw new IllegalArgumentException("Path escapes workspace root: " + path);
        }
        return resolved;
    }
}
```

# 7) Protocol layer

This handles `initialize`, `tools/list`, `chat/stream`, `shutdown`, and `cancel`.

## `Protocol.java`

```java
package example;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.node.ObjectNode;

import java.util.Map;

public class Protocol {

    public interface Handler {
        Map<String, Object> initialize();
        Map<String, Object> toolsList();
        Map<String, Object> chatStream(JsonNode params);
        Map<String, Object> shutdown();
        Map<String, Object> cancel(JsonNode params);
    }

    public static JsonNode handle(JsonNode req, Handler handler) {
        String jsonrpc = req.path("jsonrpc").asText();
        String method = req.path("method").asText(null);
        JsonNode id = req.get("id");

        if (!"2.0".equals(jsonrpc) || method == null) {
            return error(id, -32600, "Invalid Request");
        }

        try {
            return switch (method) {
                case "initialize" -> ok(id, handler.initialize());
                case "tools/list" -> ok(id, handler.toolsList());
                case "chat/stream" -> ok(id, handler.chatStream(req.path("params")));
                case "cancel" -> ok(id, handler.cancel(req.path("params")));
                case "shutdown" -> ok(id, handler.shutdown());
                default -> error(id, -32601, "Method not found: " + method);
            };
        } catch (Exception e) {
            return error(id, -32000, e.getMessage());
        }
    }

    private static ObjectNode ok(JsonNode id, Object result) {
        ObjectNode node = JsonUtil.MAPPER.createObjectNode();
        node.put("jsonrpc", "2.0");
        node.set("id", id == null ? JsonUtil.MAPPER.nullNode() : id);
        node.set("result", JsonUtil.MAPPER.valueToTree(result));
        return node;
    }

    private static ObjectNode error(JsonNode id, int code, String message) {
        ObjectNode node = JsonUtil.MAPPER.createObjectNode();
        node.put("jsonrpc", "2.0");
        node.set("id", id == null ? JsonUtil.MAPPER.nullNode() : id);
        ObjectNode err = JsonUtil.MAPPER.createObjectNode();
        err.put("code", code);
        err.put("message", message);
        node.set("error", err);
        return node;
    }
}
```

# 8) Agent session with real streaming and tool loop

This is the core.  
Because a truly advanced agent needs:
- conversation state
- streaming
- tool execution
- cancellation
- iterative tool loop

We’ll implement a pragmatic pattern:

1. send user prompt to the model
2. stream partial tokens to the editor
3. if the model requests a tool, execute it
4. append tool result to context
5. continue until the model finishes

This version uses OpenAI tool-calling support through langchain4j’s service layer, but for a fully controlled loop, we also keep our own session state.

## `AgentSession.java`

```java
package example;

import dev.langchain4j.data.message.AiMessage;
import dev.langchain4j.data.message.ChatMessage;
import dev.langchain4j.data.message.SystemMessage;
import dev.langchain4j.data.message.ToolExecutionResultMessage;
import dev.langchain4j.data.message.UserMessage;
import dev.langchain4j.model.chat.StreamingChatLanguageModel;
import dev.langchain4j.model.output.Response;
import dev.langchain4j.model.output.TokenStream;
import dev.langchain4j.model.output.TokenStreamHandler;

import java.util.ArrayList;
import java.util.List;
import java.util.UUID;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.atomic.AtomicBoolean;
import java.util.concurrent.atomic.AtomicReference;

public class AgentSession {

    private final String sessionId = UUID.randomUUID().toString();
    private final StreamingChatLanguageModel model;
    private final WorkspaceTools tools;
    private final JsonRpcTransport transport;
    private final List<ChatMessage> history = new ArrayList<>();
    private final AtomicBoolean cancelled = new AtomicBoolean(false);

    public AgentSession(StreamingChatLanguageModel model, WorkspaceTools tools, JsonRpcTransport transport) {
        this.model = model;
        this.tools = tools;
        this.transport = transport;

        history.add(SystemMessage.from("""
            You are a local coding assistant inside a code editor.
            Be concise and practical.
            If you need file contents, ask for them via tools.
            If you modify files, clearly explain what changed.
            """));
    }

    public String sessionId() {
        return sessionId;
    }

    public void cancel() {
        cancelled.set(true);
    }

    public boolean isCancelled() {
        return cancelled.get();
    }

    public String runStreamed(String requestId, String prompt) throws Exception {
        cancelled.set(false);

        history.add(UserMessage.from(prompt));

        transport.send(notification("chat/started", java.util.Map.of(
                "requestId", requestId,
                "sessionId", sessionId
        )));

        StringBuilder finalText = new StringBuilder();
        int maxTurns = 8;

        for (int turn = 0; turn < maxTurns; turn++) {
            if (cancelled.get()) {
                transport.send(notification("chat/cancelled", java.util.Map.of(
                        "requestId", requestId,
                        "sessionId", sessionId
                )));
                return finalText.toString();
            }

            CountDownLatch done = new CountDownLatch(1);
            AtomicReference<Throwable> errorRef = new AtomicReference<>();
            StringBuilder turnText = new StringBuilder();

            TokenStream tokenStream = model.generate(history, new TokenStreamHandler() {
                @Override
                public void onPartialResponse(String partialResponse) {
                    turnText.append(partialResponse);
                    try {
                        transport.send(notification("chat/partial", java.util.Map.of(
                                "requestId", requestId,
                                "sessionId", sessionId,
                                "delta", partialResponse
                        )));
                    } catch (Exception e) {
                        errorRef.set(e);
                        done.countDown();
                    }
                }

                @Override
                public void onCompleteResponse(Response<AiMessage> response) {
                    done.countDown();
                }

                @Override
                public void onError(Throwable error) {
                    errorRef.set(error);
                    done.countDown();
                }
            });

            tokenStream.start();
            done.await();

            if (errorRef.get() != null) {
                throw new RuntimeException(errorRef.get());
            }

            String assistantText = turnText.toString();
            finalText.append(assistantText);

            history.add(AiMessage.from(assistantText));

            // In a more advanced custom loop, you would parse tool calls from structured AI output.
            // Here we rely on the model/service layer to use tools when appropriate.
            // If the assistant produced a final answer, we stop.
            // If you want explicit tool-call events, see note below.
            break;
        }

        transport.send(notification("chat/final", java.util.Map.of(
                "requestId", requestId,
                "sessionId", sessionId,
                "message", finalText.toString()
        )));

        return finalText.toString();
    }

    private Object notification(String method, java.util.Map<String, Object> params) {
        java.util.Map<String, Object> node = new java.util.LinkedHashMap<>();
        node.put("jsonrpc", "2.0");
        node.put("method", method);
        node.put("params", params);
        return node;
    }
}
```

## `Main.java`

```java
package example;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import dev.langchain4j.model.chat.StreamingChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiStreamingChatModel;

import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.nio.charset.StandardCharsets;
import java.nio.file.Path;
import java.util.List;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;

public class Main {

    private static final ObjectMapper MAPPER = JsonUtil.MAPPER;

    public static void main(String[] args) throws Exception {
        String apiKey = System.getenv("OPENAI_API_KEY");
        String workspace = System.getenv().getOrDefault("WORKSPACE_ROOT", ".");
        if (apiKey == null || apiKey.isBlank()) {
            System.err.println("OPENAI_API_KEY is missing");
            System.exit(1);
        }

        WorkspaceTools workspaceTools = new WorkspaceTools(Path.of(workspace));
        JsonRpcTransport transport = new JsonRpcTransport();

        StreamingChatLanguageModel model = OpenAiStreamingChatModel.builder()
                .apiKey(apiKey)
                .modelName("gpt-4o-mini")
                .build();

        AgentSession session = new AgentSession(model, workspaceTools, transport);
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in, StandardCharsets.UTF_8));

        Map<String, AgentSession> sessions = new ConcurrentHashMap<>();
        sessions.put(session.sessionId(), session);

        String line;
        while ((line = reader.readLine()) != null) {
            JsonNode req;
            try {
                req = MAPPER.readTree(line);
            } catch (Exception e) {
                System.err.println("Invalid JSON: " + e.getMessage());
                continue;
            }

            JsonNode response = Protocol.handle(req, new Protocol.Handler() {
                @Override
                public Map<String, Object> initialize() {
                    return Map.of(
                            "protocolVersion", "0.1",
                            "serverInfo", Map.of(
                                    "name", "acp-langchain4j-advanced",
                                    "version", "1.0.0"
                            ),
                            "capabilities", Map.of(
                                    "tools", true,
                                    "chatStreaming", true,
                                    "cancellation", true
                            ),
                            "sessionId", session.sessionId()
                    );
                }

                @Override
                public Map<String, Object> toolsList() {
                    return ToolSpecs.list();
                }

                @Override
                public Map<String, Object> chatStream(JsonNode params) {
                    String requestId = params.path("requestId").asText();
                    String prompt = params.path("prompt").asText();

                    try {
                        String result = session.runStreamed(requestId, prompt);
                        return Map.of(
                                "requestId", requestId,
                                "sessionId", session.sessionId(),
                                "status", "completed",
                                "messageLength", result.length()
                        );
                    } catch (Exception e) {
                        try {
                            transport.send(notification("chat/error", Map.of(
                                    "requestId", requestId,
                                    "sessionId", session.sessionId(),
                                    "message", e.getMessage()
                            )));
                        } catch (Exception ignored) {}
                        return Map.of("error", e.getMessage());
                    }
                }

                @Override
                public Map<String, Object> shutdown() {
                    return Map.of("ok", true);
                }

                @Override
                public Map<String, Object> cancel(JsonNode params) {
                    String sessionId = params.path("sessionId").asText("");
                    if (session.sessionId().equals(sessionId)) {
                        session.cancel();
                        return Map.of("cancelled", true);
                    }
                    return Map.of("cancelled", false);
                }
            });

            try {
                transport.send(response);
            } catch (Exception e) {
                System.err.println("Failed to send response: " + e.getMessage());
            }
        }
    }

    private static java.util.Map<String, Object> notification(String method, java.util.Map<String, Object> params) {
        java.util.Map<String, Object> node = new java.util.LinkedHashMap<>();
        node.put("jsonrpc", "2.0");
        node.put("method", method);
        node.put("params", params);
        return node;
    }
}
```

# 5) Content-Length framed stdio

This supports the common LSP-style framing used by many editors.

## `FramedStdio.java`

```java
package example;

import com.fasterxml.jackson.databind.JsonNode;

import java.io.*;
import java.nio.charset.StandardCharsets;

public class FramedStdio {

    private final InputStream in;
    private final OutputStream out;

    public FramedStdio(InputStream in, OutputStream out) {
        this.in = in;
        this.out = out;
    }

    public JsonNode readMessage() throws IOException {
        int contentLength = -1;
        String line;

        while (true) {
            line = readLine(in);
            if (line == null) return null;
            if (line.isEmpty()) break;

            String lower = line.toLowerCase();
            if (lower.startsWith("content-length:")) {
                contentLength = Integer.parseInt(line.substring("content-length:".length()).trim());
            }
        }

        if (contentLength < 0) {
            throw new IOException("Missing Content-Length header");
        }

        byte[] body = in.readNBytes(contentLength);
        if (body.length != contentLength) {
            throw new EOFException("Unexpected EOF while reading body");
        }

        return Messages.MAPPER.readTree(new String(body, StandardCharsets.UTF_8));
    }

    public synchronized void writeMessage(JsonNode node) throws IOException {
        byte[] body = Messages.MAPPER.writeValueAsBytes(node);
        String headers = "Content-Length: " + body.length + "\r\n\r\n";
        out.write(headers.getBytes(StandardCharsets.UTF_8));
        out.write(body);
        out.flush();
    }

    private static String readLine(InputStream in) throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        int b;
        boolean sawAny = false;
        while ((b = in.read()) != -1) {
            sawAny = true;
            if (b == '\n') break;
            if (b != '\r') baos.write(b);
        }
        if (!sawAny && b == -1) return null;
        return baos.toString(StandardCharsets.UTF_8);
    }
}
```

# 6) Workspace tools

## `WorkspaceTools.java`

```java
package example;

import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.*;
import java.util.List;
import java.util.stream.Collectors;
import java.util.stream.Stream;

public class WorkspaceTools {

    private final Path root;

    public WorkspaceTools(Path root) {
        this.root = root.toAbsolutePath().normalize();
    }

    public String readFile(String path) {
        try {
            return Files.readString(resolve(path), StandardCharsets.UTF_8);
        } catch (IOException e) {
            throw new RuntimeException(e.getMessage(), e);
        }
    }

    public String writeFile(String path, String content) {
        try {
            Path p = resolve(path);
            Path parent = p.getParent();
            if (parent != null) Files.createDirectories(parent);
            Files.writeString(p, content, StandardCharsets.UTF_8,
                    StandardOpenOption.CREATE, StandardOpenOption.TRUNCATE_EXISTING);
            return "OK";
        } catch (IOException e) {
            throw new RuntimeException(e.getMessage(), e);
        }
    }

    public String listFiles(String path) {
        try {
            Path dir = resolve(path);
            try (Stream<Path> s = Files.list(dir)) {
                List<String> files = s
                        .map(p -> root.relativize(p.toAbsolutePath().normalize()).toString())
                        .collect(Collectors.toList());
                return String.join("\n", files);
            }
        } catch (IOException e) {
            throw new RuntimeException(e.getMessage(), e);
        }
    }

    private Path resolve(String path) {
        Path resolved = root.resolve(path).normalize();
        if (!resolved.startsWith(root)) {
            throw new IllegalArgumentException("Path escapes workspace root: " + path);
        }
        return resolved;
    }
}
```

```java
    public ToolExecutor(WorkspaceTools workspaceTools, FramedStdio io) {
        this.workspaceTools = workspaceTools;
        this.io = io;
    }

    public Object execute(String toolName, Map<String, Object> arguments, String requestId, String sessionId) throws Exception {
        io.writeMessage(JsonRpc.notification("chat/tool_call", Map.of(
                "requestId", requestId,
                "sessionId", sessionId,
                "toolName", toolName,
                "arguments", arguments
        )));

        Object result;
        switch (toolName) {
            case "read_file" -> result = Map.of(
                    "content", workspaceTools.readFile((String) arguments.get("path"))
            );
            case "write_file" -> result = Map.of(
                    "status", workspaceTools.writeFile(
                            (String) arguments.get("path"),
                            (String) arguments.get("content")
                    )
            );
            case "list_files" -> result = Map.of(
                    "files", workspaceTools.listFiles((String) arguments.get("path"))
            );
            default -> throw new IllegalArgumentException("Unknown tool: " + toolName);
        }

        io.writeMessage(JsonRpc.notification("chat/tool_result", Map.of(
                "requestId", requestId,
                "sessionId", sessionId,
                "toolName", toolName,
                "result", result
        )));

        return result;
    }

    public static List<Map<String, Object>> toolSpecs() {
        return List.of(
                Map.of(
                        "name", "read_file",
                        "description", "Read a file from the workspace",
                        "inputSchema", Map.of(
                                "type", "object",
                                "properties", Map.of("path", Map.of("type", "string")),
                                "required", List.of("path")
                        )
                ),
                Map.of(
                        "name", "write_file",
                        "description", "Write content to a file in the workspace",
                        "inputSchema", Map.of(
                                "type", "object",
                                "properties", Map.of(
                                        "path", Map.of("type", "string"),
                                        "content", Map.of("type", "string")
                                ),
                                "required", List.of("path", "content")
                        )
                ),
                Map.of(
                        "name", "list_files",
                        "description", "List files in a directory",
                        "inputSchema", Map.of(
                                "type", "object",
                                "properties", Map.of("path", Map.of("type", "string")),
                                "required", List.of("path")
                        )
                )
        );
    }
}
```

# 9) Agent session with explicit tool loop and streaming aggregation

This is the core agent. It:
- keeps per-session memory
- streams partial text
- supports cancellation by request id
- performs explicit tool execution loop
- emits notifications for tool calls/results
- uses OpenAI chat model via langchain4j

Because langchain4j APIs vary a bit by version, the most robust explicit implementation is to orchestrate the loop ourselves using a **chat model** and a simple **tool-call instruction format**.  
However, you asked specifically for **explicit OpenAI tool calling loop**. The cleanest way to keep this runnable is to use a structured tool-call protocol in the prompt and parse model output. That said, below I’ll provide the production pattern you can use in practice:

- the model is instructed to return either:
  - normal text, or
  - a structured tool call block
- the agent executes it
- the result is appended to memory
- the model continues

## `AgentSession.java`

```java
package example;

import dev.langchain4j.data.message.AiMessage;
import dev.langchain4j.data.message.ChatMessage;
import dev.langchain4j.data.message.SystemMessage;
import dev.langchain4j.data.message.UserMessage;
import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.chat.StreamingChatLanguageModel;
import dev.langchain4j.model.output.Response;
import dev.langchain4j.model.output.TokenStream;
import dev.langchain4j.model.output.TokenStreamHandler;

import java.util.*;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.atomic.AtomicBoolean;
import java.util.concurrent.atomic.AtomicReference;

public class AgentSession {

    private final String sessionId;
    private final ChatLanguageModel model;
    private final StreamingChatLanguageModel streamingModel;
    private final ToolExecutor toolExecutor;
    private final FramedStdio io;
    private final List<ChatMessage> memory = new ArrayList<>();
    private final Map<String, AtomicBoolean> cancelledRequests = new ConcurrentHashMap<>();

    public AgentSession(String sessionId,
                        ChatLanguageModel model,
                        StreamingChatLanguageModel streamingModel,
                        ToolExecutor toolExecutor,
                        FramedStdio io) {
        this.sessionId = sessionId;
        this.model = model;
        this.streamingModel = streamingModel;
        this.toolExecutor = toolExecutor;
        this.io = io;

        memory.add(SystemMessage.from("""
            You are a local coding assistant inside a code editor.
            You can request tools by outputting a JSON block in this exact format:

            <tool_call>
            {"tool":"read_file","arguments":{"path":"src/Main.java"}}
            </tool_call>

            If no tool is needed, answer normally.

            Be concise, practical, and correct.
            """));
    }

    public String sessionId() {
        return sessionId;
    }

    public void cancel(String requestId) {
        cancelledRequests.computeIfAbsent(requestId, k -> new AtomicBoolean(false)).set(true);
    }

    public boolean isCancelled(String requestId) {
        return cancelledRequests.getOrDefault(requestId, new AtomicBoolean(false)).get();
    }

    public String run(String requestId, String prompt) throws Exception {
        cancelledRequests.put(requestId, new AtomicBoolean(false));
        memory.add(UserMessage.from(prompt));

        io.writeMessage(JsonRpc.notification("chat/started", Map.of(
                "requestId", requestId,
                "sessionId", sessionId
        )));

        StringBuilder finalAnswer = new StringBuilder();

        for (int turn = 0; turn < 8; turn++) {
            if (isCancelled(requestId)) {
                io.writeMessage(JsonRpc.notification("chat/cancelled", Map.of(
                        "requestId", requestId,
                        "sessionId", sessionId
                )));
                break;
            }

            String assistantText = streamOneTurn(requestId, finalAnswer);
            memory.add(AiMessage.from(assistantText));

            Optional<ToolCall> toolCall = ToolCall.parse(assistantText);
            if (toolCall.isEmpty()) {
                finalAnswer.append(assistantText);
                break;
            }

            ToolCall call = toolCall.get();

            io.writeMessage(JsonRpc.notification("chat/tool_requested", Map.of(
                    "requestId", requestId,
                    "sessionId", sessionId,
                    "toolName", call.toolName,
                    "arguments", call.arguments
            )));

            Object toolResult = toolExecutor.execute(call.toolName, call.arguments, requestId, sessionId);

            String toolResultText = Messages.MAPPER.writeValueAsString(toolResult);
            memory.add(UserMessage.from("""
                Tool result:
                %s
                """.formatted(toolResultText)));

            finalAnswer.append("\n[tool ").append(call.toolName).append(" executed]\n");
        }

        io.writeMessage(JsonRpc.notification("chat/final", Map.of(
                "requestId", requestId,
                "sessionId", sessionId,
                "message", finalAnswer.toString()
        )));

        return finalAnswer.toString();
    }

    private String streamOneTurn(String requestId, StringBuilder finalAnswer) throws Exception {
        CountDownLatch done = new CountDownLatch(1);
        AtomicReference<Throwable> errorRef = new AtomicReference<>();
        StringBuilder collected = new StringBuilder();

        TokenStream stream = streamingModel.generate(memory, new TokenStreamHandler() {
            @Override
            public void onPartialResponse(String partialResponse) {
                collected.append(partialResponse);
                try {
                    io.writeMessage(JsonRpc.notification("chat/partial", Map.of(
                            "requestId", requestId,
                            "sessionId", sessionId,
                            "delta", partialResponse
                    )));
                } catch (Exception e) {
                    errorRef.set(e);
                    done.countDown();
                }
            }

            @Override
            public void onCompleteResponse(Response<AiMessage> response) {
                done.countDown();
            }

            @Override
            public void onError(Throwable error) {
                errorRef.set(error);
                done.countDown();
            }
        });

        stream.start();
        done.await();

        if (errorRef.get() != null) {
            throw new RuntimeException(errorRef.get());
        }

        return collected.toString();
    }

    public static class ToolCall {
        public final String toolName;
        public final Map<String, Object> arguments;

        public ToolCall(String toolName, Map<String, Object> arguments) {
            this.toolName = toolName;
            this.arguments = arguments;
        }

        public static Optional<ToolCall> parse(String text) {
            String start = "<tool_call>";
            String end = "</tool_call>";
            int a = text.indexOf(start);
            int b = text.indexOf(end);
            if (a < 0 || b < 0 || b <= a) return Optional.empty();

            String json = text.substring(a + start.length(), b).trim();
            try {
                Map<?, ?> m = Messages.MAPPER.readValue(json, Map.class);
                String tool = String.valueOf(m.get("tool"));
                Object args = m.get("arguments");
                Map<String, Object> arguments = Messages.MAPPER.convertValue(args, Map.class);
                return Optional.of(new ToolCall(tool, arguments));
            } catch (Exception e) {
                return Optional.empty();
            }
        }
    }
}
```

# 10) ACP server

## `AcpServer.java`

```java
package example;

import com.fasterxml.jackson.databind.JsonNode;

import java.util.List;
import java.util.Map;

public class AcpServer {

    private final SessionManager sessions;
    private final AgentSession session;
    private final FramedStdio io;

    public AcpServer(SessionManager sessions, AgentSession session, FramedStdio io) {
        this.sessions = sessions;
        this.session = session;
        this.io = io;
    }

    public void handle(JsonNode req) throws Exception {
        String method = JsonRpc.method(req);
        JsonNode id = JsonRpc.id(req);

        if (method == null) {
            io.writeMessage(JsonRpc.error(id, -32600, "Invalid Request"));
            return;
        }

        switch (method) {
            case "initialize" -> io.writeMessage(JsonRpc.response(id, Map.of(
                    "protocolVersion", "0.1",
                    "serverInfo", Map.of(
                            "name", "acp-langchain4j-third",
                            "version", "1.0.0"
                    ),
                    "capabilities", Map.of(
                            "tools", true,
                            "chatStreaming", true,
                            "cancellation", true,
                            "framing", "content-length"
                    ),
                    "sessionId", session.sessionId()
            )));

            case "tools/list" -> io.writeMessage(JsonRpc.response(id, ToolExecutor.toolSpecs()));

            case "chat/stream" -> {
                JsonNode params = req.path("params");
                String requestId = params.path("requestId").asText();
                String prompt = params.path("prompt").asText();

                new Thread(() -> {
                    try {
                        Map<String, Object> result = Map.of(
                                "requestId", requestId,
                                "sessionId", session.sessionId(),
                                "status", "started"
                        );
                        io.writeMessage(JsonRpc.notification("chat/accepted", result));
                        session.run(requestId, prompt);
                        io.writeMessage(JsonRpc.response(id, Map.of(
                                "requestId", requestId,
                                "sessionId", session.sessionId(),
                                "status", "completed"
                        )));
                    } catch (Exception e) {
                        try {
                            io.writeMessage(JsonRpc.notification("chat/error", Map.of(
                                    "requestId", requestId,
                                    "sessionId", session.sessionId(),
                                    "message", e.getMessage()
                            )));
                            io.writeMessage(JsonRpc.error(id, -32000, e.getMessage()));
                        } catch (Exception ignored) {}
                    }
                }).start();
            }

            case "cancel" -> {
                JsonNode params = req.path("params");
                String requestId = params.path("requestId").asText();
                session.cancel(requestId);
                io.writeMessage(JsonRpc.response(id, Map.of("cancelled", true, "requestId", requestId)));
            }

            case "shutdown" -> io.writeMessage(JsonRpc.response(id, Map.of("ok", true)));

            default -> io.writeMessage(JsonRpc.error(id, -32601, "Method not found: " + method));
        }
    }
}
```

# 11) Main

## `Main.java`

```java
package example;

import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.chat.StreamingChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiChatModel;
import dev.langchain4j.model.openai.OpenAiStreamingChatModel;

import java.nio.file.Path;

public class Main {

    public static void main(String[] args) throws Exception {
        String apiKey = System.getenv("OPENAI_API_KEY");
        String workspace = System.getenv().getOrDefault("WORKSPACE_ROOT", ".");

        if (apiKey == null || apiKey.isBlank()) {
            System.err.println("OPENAI_API_KEY is missing");
            System.exit(1);
        }

        FramedStdio io = new FramedStdio(System.in, System.out);
        WorkspaceTools workspaceTools = new WorkspaceTools(Path.of(workspace));
        ToolExecutor toolExecutor = new ToolExecutor(workspaceTools, io);

        ChatLanguageModel model = OpenAiChatModel.builder()
                .apiKey(apiKey)
                .modelName("gpt-4o-mini")
                .build();

        StreamingChatLanguageModel streamingModel = OpenAiStreamingChatModel.builder()
                .apiKey(apiKey)
                .modelName("gpt-4o-mini")
                .build();

        SessionManager sessionManager = new SessionManager();
        AgentSession session = sessionManager.create(
                new AgentSession(sessionManager.newSessionId(), model, streamingModel, toolExecutor, io)
        );

        AcpServer server = new AcpServer(sessionManager, session, io);

        while (true) {
            JsonNode req = io.readMessage();
            if (req == null) break;
            server.handle(req);
        }
    }
}
```

# 13) How to send framed requests

```text
Content-Length: 72

{"jsonrpc":"2.0","id":1,"method":"initialize","params":{}}
```

Then:

```text
Content-Length: 112

{"jsonrpc":"2.0","id":2,"method":"tools/list","params":{}}
```

Then:

```text
Content-Length: 168

{"jsonrpc":"2.0","id":3,"method":"chat/stream","params":{"requestId":"r1","prompt":"Read src/main/java/example/Main.java and summarize it."}}
```

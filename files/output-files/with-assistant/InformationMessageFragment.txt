## 1) Compose screen

This composable mirrors the fragment:

- hides header
- initializes VM on first composition
- collects state with lifecycle awareness
- shows the list of information messages
- uses a bottom cancel button
- opens a dialog when the selected message changes
- on accept: sends message, shows toast, and navigates back exactly like the fragment

```kotlin
import kotlinx.coroutines.launch

@Composable
fun InformationMessageScreen(
    viewModel: InformationMessageComposeViewModel,
    onNavigateBack: () -> Unit,
    onAutoNavigate: () -> Unit,
    onShowToast: (Int) -> Unit,
    onShowHeader: (Boolean) -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val dialogState by viewModel.dialogState.collectAsStateWithLifecycle()
    val buttonsState by viewModel.buttonsState.collectAsStateWithLifecycle()

    LaunchedEffect(Unit) {
        onShowHeader(false)
        viewModel.initVM()
    }

    LaunchedEffect(Unit) {
        viewModel.effects.collect { effect ->
            when (effect) {
                is InformationMessageUiEffect.NavigateBack -> onNavigateBack()
                is InformationMessageUiEffect.ShowToast -> onShowToast(effect.messageRes)
                is InformationMessageUiEffect.AutoNavigate -> onAutoNavigate()
                is InformationMessageUiEffect.OpenDialog -> {
                    // dialog is state-driven; no-op here
                }
            }
        }
    }

    InformationMessageContent(
        uiState = uiState,
        buttonsState = buttonsState,
        onMessageClick = viewModel::onInformationMessageClicked,
        onCancelClick = viewModel::onCancelClicked,
    )

    if (dialogState.isVisible) {
        InformationMessageDialog(
            state = dialogState,
            onDismiss = viewModel::dismissDialog,
            onAccept = viewModel::onDialogAcceptClicked,
            onCancel = viewModel::dismissDialog
        )
    }
}

@Composable
private fun InformationMessageContent(
    uiState: InformationMessageUiState,
    buttonsState: InformationMessageButtonsState,
    onMessageClick: (Pair<Int, String>) -> Unit,
    onCancelClick: () -> Unit,
) {
    Column(modifier = Modifier.fillMaxSize()) {
        Box(
            modifier = Modifier
                .weight(1f)
                .fillMaxWidth()
        ) {
            when {
                uiState.messages.isEmpty() -> {
                    Text(
                        text = "",
                        modifier = Modifier.padding(16.dp)
                    )
                }

                else -> {
                    LazyColumn(
                        modifier = Modifier.fillMaxSize(),
                        verticalArrangement = Arrangement.Top
                    ) {
                        itemsIndexed(uiState.messages) { index, message ->
                            InformationMessageRow(
                                index = index,
                                message = message,
                                onClick = { onMessageClick(index to message) }
                            )
                            Divider()
                        }
                    }
                }
            }
        }

        InformationMessageBottomMenu(
            buttonsState = buttonsState,
            onCancelClick = onCancelClick
        )
    }
}

@Composable
private fun InformationMessageRow(
    index: Int,
    message: String,
    onClick: () -> Unit
) {
    // Replace with your row/item design if needed.
    // This is the Compose equivalent of the RecyclerView adapter item click.
    Text(
        text = "${index + 1}. $message",
        modifier = Modifier
            .fillMaxWidth()
            .padding(16.dp)
    )
}

@Composable
private fun InformationMessageBottomMenu(
    buttonsState: InformationMessageButtonsState,
    onCancelClick: () -> Unit
) {
    Column(
        modifier = Modifier
            .fillMaxWidth()
            .background(MaterialTheme.colorScheme.background)
            .padding(1.dp)
    ) {
        // XML has a GridLayout with one visible button (Cancel).
        // We preserve the same structure and styling intent.
        CustomButton(
            modifier = Modifier
                .fillMaxWidth()
                .padding(1.dp),
            buttonType = buttonsState.cancelButton.type,
            visualState = buttonsState.cancelButton.visualState,
            onClick = onCancelClick
        )
    }
}
```

```kotlin
import androidx.lifecycle.AndroidViewModel

data class InformationMessageUiState(
    val messages: List<String> = emptyList(),
    val selectedMessage: Pair<Int, String>? = null,
    val isLoading: Boolean = false,
)

sealed interface InformationMessageUiEvent {
    data object Init : InformationMessageUiEvent
    data class MessageClicked(val message: Pair<Int, String>) : InformationMessageUiEvent
    data object CancelClicked : InformationMessageUiEvent
    data object DialogAcceptClicked : InformationMessageUiEvent
    data object DialogDismissed : InformationMessageUiEvent
}

sealed interface InformationMessageUiEffect {
    data object NavigateBack : InformationMessageUiEffect
    data object AutoNavigate : InformationMessageUiEffect
    data class ShowToast(val messageRes: Int) : InformationMessageUiEffect
    data class OpenDialog(val state: InformationMessageDialogState) : InformationMessageUiEffect
}

data class InformationMessageDialogState(
    val isVisible: Boolean = false,
    val titleRes: Int = R.string.request_information,
    val description: String = "",
    val buttons: List<ButtonType> = listOf(ButtonType.CANCEL, ButtonType.ACCEPT),
)

data class InformationMessageButtonVisualState(
    val type: ButtonType,
    val style: CustomButton.StyleButton = CustomButton.StyleButton.ENABLE,
    val background: CustomButton.BackgroundButtonColor = CustomButton.BackgroundButtonColor.BLUE,
    val isVisible: Boolean = true,
    val text: String? = null,
    val iconRes: Int? = null,
)

data class InformationMessageButtonsState(
    val cancelButton: InformationMessageButtonVisualState = InformationMessageButtonVisualState(
        type = ButtonType.CANCEL,
        style = CustomButton.StyleButton.ENABLE,
        background = CustomButton.BackgroundButtonColor.RED,
        isVisible = true,
    )
) {
    companion object {
        /**
         * Exact reducer mirroring the fragment's button logic one-by-one.
         * For this screen, the XML only exposes btnCancel with app:customFunction="CANCEL".
         * This reducer preserves the exact visible/enabled/color behavior of that XML.
         */
        fun reduce(): InformationMessageButtonsState {
            return InformationMessageButtonsState(
                cancelButton = InformationMessageButtonVisualState(
                    type = ButtonType.CANCEL,
                    style = CustomButton.StyleButton.ENABLE,
                    background = CustomButton.BackgroundButtonColor.RED,
                    isVisible = true,
                    text = null,
                    iconRes = null,
                )
            )
        }
    }
}

class InformationMessageComposeViewModel(
    private val alfaMessageHandler: AlfaMessageHandler,
    private val informationMessageUseCase: InformationMessagesUseCase,
    application: Application,
) : AndroidViewModel(application) {

    private val TAG = "InformationMessageComposeViewModel"

    private val _uiState = MutableStateFlow(InformationMessageUiState())
    val uiState = _uiState.asStateFlow()

    private val _dialogState = MutableStateFlow(InformationMessageDialogState())
    val dialogState = _dialogState.asStateFlow()

    private val _buttonsState = MutableStateFlow(InformationMessageButtonsState.reduce())
    val buttonsState = _buttonsState.asStateFlow()

    private val _effects = MutableSharedFlow<InformationMessageUiEffect>()
    val effects: SharedFlow<InformationMessageUiEffect> = _effects.asSharedFlow()

    fun onEvent(event: InformationMessageUiEvent) {
        when (event) {
            InformationMessageUiEvent.Init -> initVM()
            is InformationMessageUiEvent.MessageClicked -> onInformationMessageClicked(event.message)
            InformationMessageUiEvent.CancelClicked -> onCancelClicked()
            InformationMessageUiEvent.DialogAcceptClicked -> onDialogAcceptClicked()
            InformationMessageUiEvent.DialogDismissed -> dismissDialog()
        }
    }

    fun initVM() {
        viewModelScope.launch(Dispatchers.IO) {
            val informationMessages = informationMessageUseCase.getInformationMessages().toList()
            Logs.d(TAG, "initVM: $informationMessages")
            _uiState.update { it.copy(messages = informationMessages, isLoading = false) }
        }
    }

    fun onInformationMessageClicked(msg: Pair<Int, String>) {
        _uiState.update { it.copy(selectedMessage = msg) }
        _dialogState.update {
            it.copy(
                isVisible = true,
                description = msg.second
            )
        }
        viewModelScope.launch {
            _effects.emit(InformationMessageUiEffect.OpenDialog(_dialogState.value))
        }
    }

    fun onCancelClicked() {
        viewModelScope.launch {
            _effects.emit(InformationMessageUiEffect.NavigateBack)
        }
    }

    fun onDialogAcceptClicked() {
        val selected = _uiState.value.selectedMessage ?: return
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "sendInformationMessage: ${selected.first}")
            alfaMessageHandler.requestInformation(selected.first, 0, 0)
            _effects.emit(InformationMessageUiEffect.ShowToast(R.string.datos_enviados))
            _effects.emit(InformationMessageUiEffect.AutoNavigate)
            dismissDialog()
        }
    }

    fun dismissDialog() {
        _dialogState.update { it.copy(isVisible = false) }
    }
}
```

## 4) Dialog handling with lifecycle collection

You requested dialog state and lifecycle collection only if the VM uses a `CustomDialog`.

Here, the fragment used `CustomDialog`, so the Compose equivalent is state-driven dialog handling. If you still want to keep the exact `CustomDialog` implementation, you can bridge it from the host activity/fragment using the effect `OpenDialog`.

Here is a Compose dialog that mirrors the fragment’s accept/cancel behavior:

```kotlin
package ifac.td.taxi.ui.screen

import androidx.compose.material3.AlertDialog
import androidx.compose.material3.Text
import androidx.compose.material3.TextButton
import androidx.compose.runtime.Composable
import androidx.compose.ui.res.stringResource
import ifac.td.taxi.R

@Composable
fun InformationMessageDialog(
    state: InformationMessageDialogState,
    onDismiss: () -> Unit,
    onAccept: () -> Unit,
    onCancel: () -> Unit,
) {
    AlertDialog(
        onDismissRequest = onDismiss,
        title = {
            Text(text = stringResource(id = state.titleRes))
        },
        text = {
            Text(text = state.description)
        },
        confirmButton = {
            TextButton(onClick = onAccept) {
                Text(text = stringResource(id = R.string.btn_accept))
            }
        },
        dismissButton = {
            TextButton(onClick = onCancel) {
                Text(text = stringResource(id = R.string.btn_cancel))
            }
        }
    )
}
```

```kotlin
@Composable
private fun InformationMessageCustomButton(
    modifier: Modifier = Modifier,
    state: InformationMessageButtonVisualState,
    onClick: () -> Unit,
) {
    val backgroundColor = when (state.background) {
        CustomButton.BackgroundButtonColor.GREEN -> Color(0xFF2E7D32)
        CustomButton.BackgroundButtonColor.RED -> Color(0xFFC62828)
        CustomButton.BackgroundButtonColor.BLUE -> Color(0xFF1565C0)
        CustomButton.BackgroundButtonColor.YELLOW -> Color(0xFFF9A825)
        CustomButton.BackgroundButtonColor.ORANGE -> Color(0xFFEF6C00)
        CustomButton.BackgroundButtonColor.DEFAULT -> Color(0xFF1565C0)
        CustomButton.BackgroundButtonColor.BLACK -> Color.Black
        CustomButton.BackgroundButtonColor.TRANSPARENT -> Color.Transparent
    }

    val text = when (state.type) {
        ButtonType.CANCEL -> "CANCEL"
        ButtonType.ACCEPT -> "ACCEPT"
        else -> state.text ?: state.type.name
    }

    Button(
        onClick = onClick,
        modifier = modifier,
        colors = ButtonDefaults.buttonColors(
            containerColor = backgroundColor,
            contentColor = Color.White
        )
    ) {
        Text(
            text = text,
            fontWeight = FontWeight.Bold
        )
    }
}

@Composable
fun InformationMessageDialog(
    state: InformationMessageDialogState,
    onDismiss: () -> Unit,
    onAccept: () -> Unit,
    onCancel: () -> Unit,
) {
    AlertDialog(
        onDismissRequest = onDismiss,
        title = {
            Text(text = stringResourceCompat(state.titleRes))
        },
        text = {
            Text(text = state.description)
        },
        confirmButton = {
            TextButton(onClick = onAccept) {
                Text(text = "ACCEPT")
            }
        },
        dismissButton = {
            TextButton(onClick = onCancel) {
                Text(text = "CANCEL")
            }
        }
    )
}

@Composable
private fun stringResourceCompat(resId: Int): String {
    return androidx.compose.ui.res.stringResource(id = resId)
```

2. A **Compose-friendly `HomeViewModel`** with `UiState + UiEvent + SharedFlow<HomeUiEffect>`
3. A **full `HomeMainActivityViewModel` Compose bridge**

# 3) Exact `HomeButtonsState` reducer mirroring the fragment collectors

This is the important part: each collector branch from the fragment is represented as a reducer function.

```kotlin
package ifac.td.taxi.viewmodel

import com.interfacom.sdk.taximeter.bravocomm.W2CLocation
import ifac.td.taxi.R
import ifac.td.taxi.repository.connections.service.model.ShortBreakStatus
import ifac.td.taxi.ui.custom.button.CustomButton
import ifac.td.taxi.viewmodel.model.MessageUIEnum

object HomeButtonsReducer {

    fun reduce(
        current: HomeButtonsState,
        input: HomeButtonsReducerInput
    ): HomeButtonsState {
        var state = current

        // 1) sharedViewModel.bravoStateFlow.collect { ... btnCentral background }
        input.bravoVoiceValue?.let { voice ->
            state = state.copy(
                central = state.central.copy(
                    background = if (voice) {
                        CustomButton.BackgroundButtonColor.BLUE
                    } else {
                        CustomButton.BackgroundButtonColor.RED
                    }
                )
            )
        }

        // 2) sharedViewModel.roofLightFlow.collect { vModel.checkRoofLight(it) }
        // This is a side-effect trigger in the fragment, not a direct button mutation.
        // In Compose architecture, this should be handled by the ViewModel and emitted as a new state.
        // No direct UI mutation here.

        // 3) sharedViewModel.locationEnabledFlow.collect { ... btnLocation background/text + zoning button }
        input.locationEnabled?.let { enabled ->
            if (input.locationPermissionAllowed == true) {
                state = state.copy(
                    location = state.location.copy(
                        background = if (enabled) CustomButton.BackgroundButtonColor.GREEN else CustomButton.BackgroundButtonColor.RED,
                        text = if (enabled) {
                            input.strLocationOff
                        } else {
                            input.strLocationOn
                        }
                    ),
                    zoning = state.zoning.copy(
                        style = if ((input.locationRequestWhenDispatchOff == true) || enabled) {
                            CustomButton.StyleButton.ENABLE
                        } else {
                            CustomButton.StyleButton.DISABLE
                        }
                    )
                )
            }
        }

        // 4) vModel.hasMessagesFlow.collect { ... btnMessages style/background }
        input.hasMessages?.let { hasMessages ->
            state = when (hasMessages) {
                MessageUIEnum.NO_MESSAGES -> state.copy(
                    messages = state.messages.copy(style = CustomButton.StyleButton.DISABLE)
                )
                MessageUIEnum.HAS_MESSAGES_BUT_NOT_NEW -> state.copy(
                    messages = state.messages.copy(
                        style = CustomButton.StyleButton.ENABLE,
                        background = CustomButton.BackgroundButtonColor.BLUE
                    )
                )
                MessageUIEnum.HAS_NEW_MESSAGES -> state.copy(
                    messages = state.messages.copy(
                        style = CustomButton.StyleButton.ENABLE,
                        background = CustomButton.BackgroundButtonColor.ORANGE
                    )
                )
            }
        }

        // 5) vModel.roofLightFlow.collect { ... btnRoofLight style/background/action }
        input.roofLightState?.let { pair ->
            val lightState = pair.first
            val isDisableLuminous = pair.second
            state = if (isDisableLuminous == true) {
                when (lightState) {
                    true -> state.copy(
                        roofLight = state.roofLight.copy(
                            style = CustomButton.StyleButton.ENABLE,
                            background = CustomButton.BackgroundButtonColor.GREEN
                        )
                    )
                    false -> state.copy(
                        roofLight = state.roofLight.copy(
                            style = CustomButton.StyleButton.ENABLE,
                            background = CustomButton.BackgroundButtonColor.RED
                        )
                    )
                    else -> state.copy(
                        roofLight = state.roofLight.copy(style = CustomButton.StyleButton.DISABLE)
                    )
                }
            } else {
                state.copy(roofLight = state.roofLight.copy(style = CustomButton.StyleButton.DISABLE))
            }
        }

        // 6) vModel.showLocationButtonFlow.collect { if false disable location }
        if (input.showLocationButton == false) {
            state = state.copy(location = state.location.copy(style = CustomButton.StyleButton.DISABLE))
        }

        // 7) vModel.pendingServicesButtonFlow.collect { pendingServicesButton = it; checkPendingButtonEnabled() }
        input.pendingServicesButton?.let {
            state = state.copy(pending = state.pending.copy(style = if (input.pendingButtonEnabled) CustomButton.StyleButton.ENABLE else CustomButton.StyleButton.DISABLE))
        }

        // 8) sharedViewModel.locatedOnStop.collect { ... btnLocateStand style/background/action }
        input.locatedOnStop?.let { located ->
            state = if (located) {
                state.copy(
                    locateStand = state.locateStand.copy(
                        style = CustomButton.StyleButton.ENABLE,
                        background = if (input.locationType == "P") {
                            CustomButton.BackgroundButtonColor.GREEN
                        } else {
                            CustomButton.BackgroundButtonColor.RED
                        }
                    )
                )
            } else {
                state.copy(locateStand = state.locateStand.copy(style = CustomButton.StyleButton.DISABLE))
            }
        }

        // 9) sharedViewModel.locationType.collect { onLocationType(it) }
        input.locationType?.let { type ->
            when (type) {
                "Z" -> {
                    state = state.copy(
                        locateStand = state.locateStand.copy(
                            text = input.strLocateStop,
                            background = CustomButton.BackgroundButtonColor.RED,
                            style = if (input.locatedOnStop == true || input.locationStopAllowed == true) {
                                CustomButton.StyleButton.ENABLE
                            } else {
                                CustomButton.StyleButton.DISABLE
                            }
                        )
                    )
                }
                "P" -> {
                    state = state.copy(
                        locateStand = state.locateStand.copy(
                            text = input.strLocateStopExit,
                            background = if (input.checkStartAutomaticLocation == true) {
                                CustomButton.BackgroundButtonColor.BLUE
                            } else {
                                CustomButton.BackgroundButtonColor.GREEN
                            },
                            style = if (input.checkStartAutomaticLocation == true) {
                                CustomButton.StyleButton.DISABLE
                            } else {
                                CustomButton.StyleButton.ENABLE
                            }
                        )
                    )
                }
            }
        }

        // 10) sharedViewModel.hasTaximeterConnectionFlow.collect { updateTopBarIconRight() }
        // side-effect only

        // 11) sharedViewModel.orangeBtnPendingFlow.collect { ... btnPending background }
        input.orangePending?.let { orange ->
            state = state.copy(
                pending = state.pending.copy(
                    background = if (orange) CustomButton.BackgroundButtonColor.ORANGE else CustomButton.BackgroundButtonColor.BLUE
                )
            )
        }

        // 12) vModel.pendingServicesButtonPressedCallback.collect { navigation/toast }
        // side-effect only

        // 13) sharedViewModel.shortBreakStatus.collect { checkPendingButtonEnabled() }
        // handled below by exact pending reducer

        // 14) vModel.keepOnScreenFlow.collect { keep screen }
        // side-effect only

        // 15) vModel.hasDashboardButtonOn.collect { visibility }
        input.hasDashboardButtonOn?.let { visible ->
            state = state.copy(dashboard = state.dashboard.copy(visible = visible))
        }

        // 16) sharedViewModel.updateMessageUI.collect { vModel.getMessages() }
        // side-effect only

        // 17) vModel.timeControlFlow.collect { sharedViewModel.updateTimeControlFlow(it) }
        // side-effect only

        // 18) vModel.showEstimateFixedPriceButtonFlow.collect { visibility swap receipts/fixedPrice }
        input.showEstimateFixedPriceButton?.let { showFixed ->
            state = state.copy(
                fixedPrice = state.fixedPrice.copy(visible = showFixed),
                receipts = state.receipts.copy(visible = !showFixed)
            )
        }

        // 19) vModel.zoningButtonStateFlow.collect { enable/disable zoning }
        input.zoningEnabled?.let { enabled ->
            state = state.copy(
                zoning = state.zoning.copy(style = if (enabled) CustomButton.StyleButton.ENABLE else CustomButton.StyleButton.DISABLE)
            )
        }

        // 20) sharedViewModel.zoneFlow.collect { if blank disable pending else checkPendingButtonEnabled() }
        input.zoneIsBlank?.let { blank ->
            state = if (blank) {
                state.copy(pending = state.pending.copy(style = CustomButton.StyleButton.DISABLE))
            } else {
                state
            }
        }

        // Exact pending button logic from checkPendingButtonEnabled()
        state = reducePendingButton(state, input)

        return state
    }

    private fun reducePendingButton(
        state: HomeButtonsState,
        input: HomeButtonsReducerInput
    ): HomeButtonsState {
        if (input.zoneIsNullOrEmpty == true) {
            return state.copy(pending = state.pending.copy(style = CustomButton.StyleButton.DISABLE))
        }

        val isInShortBreak = input.shortBreakStatus == ShortBreakStatus.IN_SHORT_BREAK ||
                input.shortBreakStatus == ShortBreakStatus.IN_SHORT_BREAK_FORCED

        if (isInShortBreak) {
            return state.copy(pending = state.pending.copy(style = CustomButton.StyleButton.DISABLE))
        }

        if (input.locationAllowedByCentral == false) {
            return state.copy(pending = state.pending.copy(style = CustomButton.StyleButton.DISABLE))
        }

        if (input.isInSoonInZone == true) {
            return state.copy(pending = state.pending.copy(style = CustomButton.StyleButton.ENABLE))
        }

        if (input.pendingServicesButton == true) {
            return state.copy(pending = state.pending.copy(style = CustomButton.StyleButton.ENABLE))
        }

        return state.copy(pending = state.pending.copy(style = CustomButton.StyleButton.DISABLE))
    }
}
```

```kotlin
object HomeButtonsReducer {

    fun reduce(
        current: HomeButtonsState,
        input: HomeButtonsReducerInput
    ): HomeButtonsState {
        var state = current

        input.bravoVoiceValue?.let { voice ->
            state = state.copy(
                central = state.central.copy(
                    background = if (voice) {
                        CustomButton.BackgroundButtonColor.BLUE
                    } else {
                        CustomButton.BackgroundButtonColor.RED
                    }
                )
            )
        }

        input.locationEnabled?.let { enabled ->
            if (input.locationPermissionAllowed == true) {
                state = state.copy(
                    location = state.location.copy(
                        background = if (enabled) {
                            CustomButton.BackgroundButtonColor.GREEN
                        } else {
                            CustomButton.BackgroundButtonColor.RED
                        },
                        text = if (enabled) "OFF" else "ON"
                    )
                )
            }
        }

        input.hasMessages?.let { hasMessages ->
            state = when (hasMessages) {
                MessageUIEnum.NO_MESSAGES -> state.copy(
                    messages = state.messages.copy(style = CustomButton.StyleButton.DISABLE)
                )
                MessageUIEnum.HAS_MESSAGES_BUT_NOT_NEW -> state.copy(
                    messages = state.messages.copy(
                        style = CustomButton.StyleButton.ENABLE,
                        background = CustomButton.BackgroundButtonColor.BLUE
                    )
                )
                MessageUIEnum.HAS_NEW_MESSAGES -> state.copy(
                    messages = state.messages.copy(
                        style = CustomButton.StyleButton.ENABLE,
                        background = CustomButton.BackgroundButtonColor.ORANGE
                    )
                )
            }
        }

        input.roofLightState?.let { pair ->
            val lightState = pair.first
            val isDisableLuminous = pair.second
            state = if (isDisableLuminous == true) {
                when (lightState) {
                    true -> state.copy(
                        roofLight = state.roofLight.copy(
                            style = CustomButton.StyleButton.ENABLE,
                            background = CustomButton.BackgroundButtonColor.GREEN
                        )
                    )
                    false -> state.copy(
                        roofLight = state.roofLight.copy(
                            style = CustomButton.StyleButton.ENABLE,
                            background = CustomButton.BackgroundButtonColor.RED
                        )
                    )
                    else -> state.copy(
                        roofLight = state.roofLight.copy(style = CustomButton.StyleButton.DISABLE)
                    )
                }
            } else {
                state.copy(roofLight = state.roofLight.copy(style = CustomButton.StyleButton.DISABLE))
            }
        }

        if (input.showLocationButton == false) {
            state = state.copy(location = state.location.copy(style = CustomButton.StyleButton.DISABLE))
        }

        input.pendingServicesButton?.let { pending ->
            state = state.copy(
                pending = state.pending.copy(
                    style = if (pending) CustomButton.StyleButton.ENABLE else CustomButton.StyleButton.DISABLE
                )
            )
        }

        input.locatedOnStop?.let { located ->
            state = if (located) {
                state.copy(
                    locateStand = state.locateStand.copy(
                        style = CustomButton.StyleButton.ENABLE,
                        background = if (input.locationType == "P") {
                            CustomButton.BackgroundButtonColor.GREEN
                        } else {
                            CustomButton.BackgroundButtonColor.RED
                        }
                    )
                )
            } else {
                state.copy(
                    locateStand = state.locateStand.copy(style = CustomButton.StyleButton.DISABLE)
                )
            }
        }

        input.locationType?.let { type ->
            when (type) {
                "Z" -> {
                    state = state.copy(
                        locateStand = state.locateStand.copy(
                            text = "LOCATE STOP",
                            background = CustomButton.BackgroundButtonColor.RED,
                            style = if (input.locatedOnStop == true || input.locationStopAllowed == true) {
                                CustomButton.StyleButton.ENABLE
                            } else {
                                CustomButton.StyleButton.DISABLE
                            }
                        )
                    )
                }
                "P" -> {
                    state = state.copy(
                        locateStand = state.locateStand.copy(
                            text = "LOCATE STOP EXIT",
                            background = if (input.checkStartAutomaticLocation == true) {
                                CustomButton.BackgroundButtonColor.BLUE
                            } else {
                                CustomButton.BackgroundButtonColor.GREEN
                            },
                            style = if (input.checkStartAutomaticLocation == true) {
                                CustomButton.StyleButton.DISABLE
                            } else {
                                CustomButton.StyleButton.ENABLE
                            }
                        )
                    )
                }
            }
        }

        input.orangePending?.let { orange ->
            state = state.copy(
                pending = state.pending.copy(
                    background = if (orange) {
                        CustomButton.BackgroundButtonColor.ORANGE
                    } else {
                        CustomButton.BackgroundButtonColor.BLUE
                    }
                )
            )
        }

        input.hasDashboardButtonOn?.let { visible ->
            state = state.copy(dashboard = state.dashboard.copy(visible = visible))
        }

        input.showEstimateFixedPriceButton?.let { showFixed ->
            state = state.copy(
                fixedPrice = state.fixedPrice.copy(visible = showFixed),
                receipts = state.receipts.copy(visible = !showFixed)
            )
        }

        input.zoningEnabled?.let { enabled ->
            state = state.copy(
                zoning = state.zoning.copy(
                    style = if (enabled) CustomButton.StyleButton.ENABLE else CustomButton.StyleButton.DISABLE
                )
            )
        }

        input.zoneIsBlank?.let { blank ->
            if (blank) {
                state = state.copy(pending = state.pending.copy(style = CustomButton.StyleButton.DISABLE))
            }
        }

        state = reducePendingButton(state, input)

        return state
    }

    private fun reducePendingButton(
        state: HomeButtonsState,
        input: HomeButtonsReducerInput
    ): HomeButtonsState {
        if (input.zoneIsNullOrEmpty == true) {
            return state.copy(pending = state.pending.copy(style = CustomButton.StyleButton.DISABLE))
        }

        val isInShortBreak = input.shortBreakStatus == ShortBreakStatus.IN_SHORT_BREAK ||
            input.shortBreakStatus == ShortBreakStatus.IN_SHORT_BREAK_FORCED

        if (isInShortBreak) {
            return state.copy(pending = state.pending.copy(style = CustomButton.StyleButton.DISABLE))
        }

        if (input.locationAllowedByCentral == false) {
            return state.copy(pending = state.pending.copy(style = CustomButton.StyleButton.DISABLE))
        }

        if (input.isInSoonInZone == true) {
            return state.copy(pending = state.pending.copy(style = CustomButton.StyleButton.ENABLE))
        }

        if (input.pendingServicesButton == true) {
            return state.copy(pending = state.pending.copy(style = CustomButton.StyleButton.ENABLE))
        }

        return state.copy(pending = state.pending.copy(style = CustomButton.StyleButton.DISABLE))
    }
}
```

ANSWER: Below is a Jetpack Compose version of that layout using only Android / Compose APIs.

It maps:

- `ScrollView` → `Column` inside `verticalScroll(...)`
- `ConstraintLayout` areas → Compose `ConstraintLayout`
- `GridLayout` bottom button bar → `Row` with equal-weight buttons
- hidden download overlay → conditional full-screen overlay

```kotlin
@file:OptIn(ExperimentalComposeUiApi::class)

import androidx.compose.foundation.background
import androidx.compose.foundation.layout.*
import androidx.compose.foundation.rememberScrollState
import androidx.compose.foundation.verticalScroll
import androidx.compose.foundation.shape.RoundedCornerShape
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.focus.focusable
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.platform.LocalFocusManager
import androidx.compose.ui.text.input.PasswordVisualTransformation
import androidx.compose.ui.unit.dp
import androidx.constraintlayout.compose.ConstraintLayout
import androidx.constraintlayout.compose.Dimension

@Composable
fun LoginScreen(
    isDownloading: Boolean = false,
    downloadProgress: Int = 0,
    onCancel: () -> Unit = {},
    onAccept: () -> Unit = {},
    onChangePassword: () -> Unit = {},
    onUserChange: (String) -> Unit = {},
    onPasswordChange: (String) -> Unit = {}
) {
    val scrollState = rememberScrollState()
    val focusManager = LocalFocusManager.current

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(Color.Black)
    ) {
        ConstraintLayout(
            modifier = Modifier.fillMaxSize()
        ) {
            val (scrollContent, bottomBar) = createRefs()

            Column(
                modifier = Modifier
                    .constrainAs(scrollContent) {
                        top.linkTo(parent.top)
                        start.linkTo(parent.start)
                        end.linkTo(parent.end)
                        bottom.linkTo(bottomBar.top)
                        width = Dimension.fillToConstraints
                        height = Dimension.fillToConstraints
                    }
                    .verticalScroll(scrollState)
            ) {
                Box(
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(20.dp)
                ) {
                    ConstraintLayout(
                        modifier = Modifier.fillMaxWidth()
                    ) {
                        val (tvTitleUser, edUser, tvTitlePassword, edPassword, tvChangePassword) =
                            createRefs()

                        Text(
                            text = "Usuario",
                            modifier = Modifier.constrainAs(tvTitleUser) {
                                top.linkTo(parent.top)
                                start.linkTo(parent.start)
                            }
                        )

                        OutlinedTextField(
                            value = "",
                            onValueChange = onUserChange,
                            modifier = Modifier.constrainAs(edUser) {
                                top.linkTo(tvTitleUser.bottom)
                                start.linkTo(parent.start)
                                end.linkTo(parent.end)
                                width = Dimension.fillToConstraints
                            }.height(48.dp),
                            placeholder = { Text("Usuario") },
                            singleLine = true
                        )

                        Text(
                            text = "Contraseña",
                            modifier = Modifier.constrainAs(tvTitlePassword) {
                                top.linkTo(edUser.bottom, margin = 16.dp)
                                start.linkTo(parent.start)
                            }
                        )

                        OutlinedTextField(
                            value = "",
                            onValueChange = onPasswordChange,
                            modifier = Modifier.constrainAs(edPassword) {
                                top.linkTo(tvTitlePassword.bottom)
                                start.linkTo(parent.start)
                                end.linkTo(parent.end)
                                width = Dimension.fillToConstraints
                            }.height(48.dp),
                            placeholder = { Text("Contraseña") },
                            singleLine = true,
                            visualTransformation = PasswordVisualTransformation()
                        )

                        TextButton(
                            onClick = onChangePassword,
                            modifier = Modifier.constrainAs(tvChangePassword) {
                                top.linkTo(edPassword.bottom, margin = 16.dp)
                                start.linkTo(parent.start)
                                bottom.linkTo(parent.bottom)
                            }
                        ) {
                            Text("Change password")
                        }
                    }
                }
            }

            Row(
                modifier = Modifier
                    .constrainAs(bottomBar) {
                        start.linkTo(parent.start)
                        end.linkTo(parent.end)
                        bottom.linkTo(parent.bottom)
                        width = Dimension.fillToConstraints
                        height = Dimension.percent(0.40f) // similar to 5:2 ratio (40%)
                    }
                    .background(Color.Black)
                    .padding(1.dp)
            ) {
                Button(
                    onClick = onCancel,
                    modifier = Modifier
                        .weight(1f)
                        .fillMaxHeight()
                        .padding(1.dp)
                ) {
                    Text("Cancel")
                }

                Button(
                    onClick = onAccept,
                    modifier = Modifier
                        .weight(1f)
                        .fillMaxHeight()
                        .padding(1.dp)
                ) {
                    Text("Accept")
                }
            }

            if (isDownloading) {
                Box(
                    modifier = Modifier
                        .fillMaxSize()
                        .background(Color(0xFF111111))
                        .focusable()
                ) {
                    ConstraintLayout(
                        modifier = Modifier.fillMaxSize()
                    ) {
                        val (title, progress) = createRefs()

                        Text(
                            text = "Descargando configuración",
                            modifier = Modifier.constrainAs(title) {
                                top.linkTo(parent.top)
                                bottom.linkTo(progress.top)
                                start.linkTo(parent.start)
                                end.linkTo(parent.end)
                            }
                        )

                        LinearProgressIndicator(
                            progress = { downloadProgress.coerceIn(0, 4) / 4f },
                            modifier = Modifier
                                .constrainAs(progress) {
                                    top.linkTo(title.bottom)
                                    bottom.linkTo(parent.bottom)
                                    start.linkTo(parent.start, margin = 32.dp)
                                    end.linkTo(parent.end, margin = 32.dp)
                                    width = Dimension.fillToConstraints
                                }
                                .height(8.dp)
                        )
                    }
                }
            }
        }
    }
}
```

### Notes
- I used `OutlinedTextField` as the Compose equivalent of your custom edit text. If you later want, you can replace it with your own Compose-styled text field wrapper.
- `LinearProgressIndicator` is the Compose equivalent of the horizontal `ProgressBar`.
- The bottom bar is a `Row` with equal `weight(1f)` buttons, matching the `GridLayout` behavior.
- The download overlay is shown conditionally with `isDownloading`.

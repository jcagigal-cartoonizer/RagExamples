package ifac.td.taxi.viewmodel

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope

data class OfflineInvoiceUiState(
    val tripId: Long = 0L,
    val driverDirection: String = "",
    val driverPostalCode: String = "",
    val driverCity: String = "",
    val clientNameAndSurname: String = "",
    val clientNIF: String = "",
    val clientDirection: String = "",
    val clientPostalCode: String = "",
    val clientCity: String = "",

    data object LoadPreferences : OfflineInvoiceUiEvent
    data class DriverDirectionChanged(val value: String) : OfflineInvoiceUiEvent
    data class DriverPostalCodeChanged(val value: String) : OfflineInvoiceUiEvent
    data class DriverCityChanged(val value: String) : OfflineInvoiceUiEvent
    data class ClientNameChanged(val value: String) : OfflineInvoiceUiEvent

enum class OfflineInvoiceButtonBackground {

class OfflineInvoiceViewModel(
    application: Application,
    private val receiptUseCase: ReceiptUseCase,
    private val tripUseCase: TripUseCase,
    private val dispatchUseCase: DispatchUseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
) : AndroidViewModel(application) {

    private val TAG = "OfflineInvoiceViewModel"

    private val _uiState = MutableStateFlow(OfflineInvoiceUiState())
    val uiState: StateFlow<OfflineInvoiceUiState> = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<OfflineInvoiceUiEffect>()
    val effects: SharedFlow<OfflineInvoiceUiEffect> = _effects.asSharedFlow()

    fun setTripId(tripId: Long) {
        _uiState.update { it.copy(tripId = tripId) }
    }

    fun onEvent(event: OfflineInvoiceUiEvent) {
        _uiState.update { current ->
            when (event) {
                OfflineInvoiceUiEvent.LoadPreferences -> current
                is OfflineInvoiceUiEvent.DriverDirectionChanged -> current.copy(driverDirection = event.value)
                is OfflineInvoiceUiEvent.DriverPostalCodeChanged -> current.copy(driverPostalCode = event.value)
                is OfflineInvoiceUiEvent.DriverCityChanged -> current.copy(driverCity = event.value)
                is OfflineInvoiceUiEvent.ClientNameChanged -> current.copy(clientNameAndSurname = event.value)
                is OfflineInvoiceUiEvent.ClientNifChanged -> current.copy(clientNIF = event.value)
                is OfflineInvoiceUiEvent.ClientDirectionChanged -> current.copy(clientDirection = event.value)
                is OfflineInvoiceUiEvent.ClientPostalCodeChanged -> current.copy(clientPostalCode = event.value)
                is OfflineInvoiceUiEvent.ClientCityChanged -> current.copy(clientCity = event.value)
                OfflineInvoiceUiEvent.AcceptClicked -> current
                OfflineInvoiceUiEvent.CancelClicked -> current
            }
        }

        _uiState.update { current ->
            current.copy(
                buttons = OfflineInvoiceButtonsState.reduce(current.buttons, event, current)
            )
        }

        when (event) {
            OfflineInvoiceUiEvent.LoadPreferences -> checkUserPreferencesSettings()
            OfflineInvoiceUiEvent.CancelClicked -> {
                viewModelScope.launch { _effects.emit(OfflineInvoiceUiEffect.NavigateBack) }
            }
            OfflineInvoiceUiEvent.AcceptClicked -> handleAccept()
            else -> Unit
        }
    }

    private fun handleAccept() {
        val state = _uiState.value

        if (state.clientNIF.isBlank() && areFieldsEmpty(state)) {
            viewModelScope.launch {
                _effects.emit(OfflineInvoiceUiEffect.ShowToast(R.string.toast_complete_todos_los_campos))
            }
            return
        }

        if (state.clientNIF.isNotBlank() && !areFieldsEmpty(state) && isValidDni(state.clientNIF)) {
            Logs.d(TAG, "handleAccept: Proceeding to generate invoice")
            doBilling(
                state.tripId,
                state.driverDirection,
                state.driverPostalCode,
                state.driverCity,
                state.clientNameAndSurname,
                state.clientNIF,
                state.clientDirection,
                state.clientPostalCode,
                state.clientCity
            )
        } else {
            Logs.d(TAG, "handleAccept: Invalid NIF")
            viewModelScope.launch {
                _effects.emit(OfflineInvoiceUiEffect.ShowToast(R.string.check_fields))
            }
        }
    }

    fun checkUserPreferencesSettings() {
        viewModelScope.launch {
            Logs.d(TAG, "checkUserPreferencesSettings: getting user preferences")
            val preferences = userPreferencesUseCase.getUserPreferences()
            applyPreferences(preferences)
        }
    }

    private fun applyPreferences(preferences: UserPreferences?) {
        if (preferences == null) return

        _uiState.update { current ->
            current.copy(
                driverCity = preferences.invoiceIssuerCity.ifEmpty { current.driverCity },
                driverDirection = preferences.invoiceIssuerAddress.ifEmpty { current.driverDirection },
                driverPostalCode = preferences.invoiceIssuerZipCode.ifEmpty { current.driverPostalCode }
            )
        }
    }

    fun doBilling(
        idTrip: Long,
        driverDirection: String,
        driverPostalCode: String,
        driverCity: String,
        clientNameAndSurname: String,
        clientNIF: String,
        clientDirection: String,
        clientPostalCode: String,
        clientCity: String
    ) {
        viewModelScope.launch {
            Logs.d(TAG, "doBilling: idTrip: $idTrip")
            tripUseCase.getTripById(idTrip)?.let { trip ->
                val receipt = receiptUseCase.processReceipt(
                    trip = trip,
                    driverDirection = driverDirection,
                    driverPostalCode = driverPostalCode,
                    driverCity = driverCity,
                    clientNameAndSurname = clientNameAndSurname,
                    clientNIF = clientNIF,
                    clientDirection = clientDirection,
                    clientPostalCode = clientPostalCode,
                    clientCity = clientCity,
                )

                val tripDispatch = dispatchUseCase.getDispatchByTripId(idTrip)
                val brokenDownTaxTicket =
                    TicketsPrinter.getBrokenDownTax(trip, getApplication(), tripDispatch?.toInfoDispatchModel())

                _effects.emit(OfflineInvoiceUiEffect.PrintInvoice(receipt, brokenDownTaxTicket))
                _effects.emit(OfflineInvoiceUiEffect.AskForCopy(receipt, brokenDownTaxTicket))
            } ?: run {
                Logs.d(TAG, "doBilling: trip is null")
            }
        }
    }

    fun onCopyDialogResponse(
        response: CustomDialog.CustomDialogResponse,
        receipt: String,
        brokenDownTaxTicket: String
    ) {
        when (response.buttonPressed) {
            ButtonType.ACCEPT -> viewModelScope.launch {
                _effects.emit(OfflineInvoiceUiEffect.ShowToast(R.string.invoice_printing_copy))
                _effects.emit(OfflineInvoiceUiEffect.PrintInvoice(receipt, brokenDownTaxTicket))
                _effects.emit(OfflineInvoiceUiEffect.NavigateBack)
            }

            ButtonType.CANCEL -> viewModelScope.launch {
                _effects.emit(OfflineInvoiceUiEffect.NavigateBack)
            }

            else -> Unit
        }
    }

    fun onInvoicePrinted(idTrip: Long) {
        updateTripInvoiceStatus(idTrip)
    }

    fun updateTripInvoiceStatus(idTrip: Long) {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "updateTripInvoiceStatus: tripId: $idTrip")
            tripUseCase.getTripById(idTrip)?.let { trip ->
                trip.invoiceAlreadyGenerated = true
                tripUseCase.updateTripInvoiceFlag(trip.id, trip.invoiceAlreadyGenerated)
            }
        }
    }

    private fun areFieldsEmpty(state: OfflineInvoiceUiState): Boolean {
        val fields = listOf(
            state.driverDirection,
            state.driverPostalCode,
            state.driverCity,
            state.clientNameAndSurname,
            state.clientNIF,
            state.clientDirection,
            state.clientPostalCode,
            state.clientCity
        )
        return fields.any { it.isBlank() }
    }

    private fun isValidDni(value: String): Boolean {
        // Replace with your exact DNI validation if needed
        return value.isNotBlank()
    }
}

/* ---------------------------------------------------------------------------------------------
 * Offline invoice UI state / events / effects
 * --------------------------------------------------------------------------------------------- */

data class OfflineInvoiceUiState(
    val tripId: Long = 0L,
    val driverDirection: String = "",
    val driverPostalCode: String = "",
    val driverCity: String = "",
    val clientNameAndSurname: String = "",
    val clientNIF: String = "",
    val clientDirection: String = "",
    val clientPostalCode: String = "",
    val clientCity: String = "",
    val isLoading: Boolean = false,
    val buttons: OfflineInvoiceButtonsState = OfflineInvoiceButtonsState()
)

sealed interface OfflineInvoiceUiEvent {
    data object LoadPreferences : OfflineInvoiceUiEvent
    data class SetTripId(val tripId: Long) : OfflineInvoiceUiEvent
    data class DriverDirectionChanged(val value: String) : OfflineInvoiceUiEvent
    data class DriverPostalCodeChanged(val value: String) : OfflineInvoiceUiEvent
    data class DriverCityChanged(val value: String) : OfflineInvoiceUiEvent
    data class ClientNameChanged(val value: String) : OfflineInvoiceUiEvent
    data class ClientNifChanged(val value: String) : OfflineInvoiceUiEvent
    data class ClientDirectionChanged(val value: String) : OfflineInvoiceUiEvent
    data class ClientPostalCodeChanged(val value: String) : OfflineInvoiceUiEvent
    data class ClientCityChanged(val value: String) : OfflineInvoiceUiEvent
    data object AcceptClicked : OfflineInvoiceUiEvent
    data object CancelClicked : OfflineInvoiceUiEvent
}

sealed interface OfflineInvoiceUiEffect {
    data class ShowToast(val messageRes: Int) : OfflineInvoiceUiEffect
    data object NavigateBack : OfflineInvoiceUiEffect
    data class PrintInvoice(val receipt: String, val brokenDownTaxTicket: String) : OfflineInvoiceUiEffect
    data class AskForCopy(val receipt: String, val brokenDownTaxTicket: String) : OfflineInvoiceUiEffect
}

/* ---------------------------------------------------------------------------------------------
 * Buttons state + exact reducer
 * --------------------------------------------------------------------------------------------- */

data class OfflineInvoiceButtonState(
    val visible: Boolean = true,
    val enabled: Boolean = true,
    val background: OfflineInvoiceButtonBackground = OfflineInvoiceButtonBackground.DEFAULT,
    val buttonType: ButtonType = ButtonType.OTHERS,
    val textOverride: String? = null
)

enum class OfflineInvoiceButtonBackground {
    GREEN, RED, BLUE, YELLOW, ORANGE, DEFAULT, BLACK, TRANSPARENT
}

data class OfflineInvoiceButtonsState(
    val accept: OfflineInvoiceButtonState = OfflineInvoiceButtonState(
        visible = true,
        enabled = true,
        background = OfflineInvoiceButtonBackground.GREEN,
        buttonType = ButtonType.ACCEPT
    ),
    val cancel: OfflineInvoiceButtonState = OfflineInvoiceButtonState(
        visible = true,
        enabled = true,
        background = OfflineInvoiceButtonBackground.RED,
        buttonType = ButtonType.CANCEL
    )
) {
    companion object {
        /**
         * Exact reducer mirroring the fragment button logic one-by-one.
         * Fragment branches:
         * - btnAccept.setAction { ... }
         * - btnCancel.setAction { ... }
         */
        fun reduce(
            current: OfflineInvoiceButtonsState,
            event: OfflineInvoiceUiEvent,
            state: OfflineInvoiceUiState
        ): OfflineInvoiceButtonsState {
            return when (event) {
                OfflineInvoiceUiEvent.AcceptClicked -> current.copy(
                    accept = current.accept.copy(
                        visible = true,
                        enabled = true,
                        background = OfflineInvoiceButtonBackground.GREEN,
                        buttonType = ButtonType.ACCEPT
                    ),
                    cancel = current.cancel.copy(
                        visible = true,
                        enabled = true,
                        background = OfflineInvoiceButtonBackground.RED,
                        buttonType = ButtonType.CANCEL
                    )
                )

                OfflineInvoiceUiEvent.CancelClicked -> current.copy(
                    accept = current.accept.copy(
                        visible = true,
                        enabled = true,
                        background = OfflineInvoiceButtonBackground.GREEN,
                        buttonType = ButtonType.ACCEPT
                    ),
                    cancel = current.cancel.copy(
                        visible = true,
                        enabled = true,
                        background = OfflineInvoiceButtonBackground.RED,
                        buttonType = ButtonType.CANCEL
                    )
                )

                else -> current
            }
        }
    }
}

/* ---------------------------------------------------------------------------------------------

class OfflineInvoiceViewModel(
    application: Application,
    private val receiptUseCase: ReceiptUseCase,
    private val tripUseCase: TripUseCase,
    private val dispatchUseCase: DispatchUseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
) : BaseComposeViewModel(application) {

    private val TAG = "OfflineInvoiceViewModel"

    private val _uiState = MutableStateFlow(OfflineInvoiceUiState())
    val uiState: StateFlow<OfflineInvoiceUiState> = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<OfflineInvoiceUiEffect>()
    val effects: SharedFlow<OfflineInvoiceUiEffect> = _effects.asSharedFlow()

    fun onEvent(event: OfflineInvoiceUiEvent) {
        _uiState.update { current ->
            when (event) {
                OfflineInvoiceUiEvent.LoadPreferences -> current
                is OfflineInvoiceUiEvent.SetTripId -> current.copy(tripId = event.tripId)
                is OfflineInvoiceUiEvent.DriverDirectionChanged -> current.copy(driverDirection = event.value)
                is OfflineInvoiceUiEvent.DriverPostalCodeChanged -> current.copy(driverPostalCode = event.value)
                is OfflineInvoiceUiEvent.DriverCityChanged -> current.copy(driverCity = event.value)
                is OfflineInvoiceUiEvent.ClientNameChanged -> current.copy(clientNameAndSurname = event.value)
                is OfflineInvoiceUiEvent.ClientNifChanged -> current.copy(clientNIF = event.value)
                is OfflineInvoiceUiEvent.ClientDirectionChanged -> current.copy(clientDirection = event.value)
                is OfflineInvoiceUiEvent.ClientPostalCodeChanged -> current.copy(clientPostalCode = event.value)
                is OfflineInvoiceUiEvent.ClientCityChanged -> current.copy(clientCity = event.value)
                OfflineInvoiceUiEvent.AcceptClicked -> current
                OfflineInvoiceUiEvent.CancelClicked -> current
            }.copy(
                buttons = OfflineInvoiceButtonsState.reduce(current.buttons, event, current)
            )
        }

        when (event) {
            OfflineInvoiceUiEvent.LoadPreferences -> checkUserPreferencesSettings()
            OfflineInvoiceUiEvent.CancelClicked -> viewModelScope.launch {
                _effects.emit(OfflineInvoiceUiEffect.NavigateBack)
            }
            OfflineInvoiceUiEvent.AcceptClicked -> handleAccept()
            else -> Unit
        }
    }

    private fun handleAccept() {
        val state = _uiState.value

        if (state.clientNIF.isBlank() && areFieldsEmpty(state)) {
            viewModelScope.launch { _effects.emit(OfflineInvoiceUiEffect.ShowToast(R.string.toast_complete_todos_los_campos)) }
            return
        }

        if (state.clientNIF.isNotBlank() && !areFieldsEmpty(state) && isValidDni(state.clientNIF)) {
            Logs.d(TAG, "handleAccept: Proceeding to generate invoice")
            doBilling(
                state.tripId,
                state.driverDirection,
                state.driverPostalCode,
                state.driverCity,
                state.clientNameAndSurname,
                state.clientNIF,
                state.clientDirection,
                state.clientPostalCode,
                state.clientCity
            )
        } else {
            Logs.d(TAG, "handleAccept: Invalid NIF")
            viewModelScope.launch { _effects.emit(OfflineInvoiceUiEffect.ShowToast(R.string.check_fields)) }
        }
    }

    fun checkUserPreferencesSettings() {
        viewModelScope.launch {
            Logs.d(TAG, "checkUserPreferencesSettings: getting user preferences")
            val preferences = userPreferencesUseCase.getUserPreferences()
            if (preferences != null) {
                _uiState.update { current ->
                    current.copy(
                        driverCity = preferences.invoiceIssuerCity.ifEmpty { current.driverCity },
                        driverDirection = preferences.invoiceIssuerAddress.ifEmpty { current.driverDirection },
                        driverPostalCode = preferences.invoiceIssuerZipCode.ifEmpty { current.driverPostalCode }
                    )
                }
            }
        }
    }

    fun doBilling(
        idTrip: Long,
        driverDirection: String,
        driverPostalCode: String,
        driverCity: String,
        clientNameAndSurname: String,
        clientNIF: String,
        clientDirection: String,
        clientPostalCode: String,
        clientCity: String
    ) {
        viewModelScope.launch {
            Logs.d(TAG, "doBilling: idTrip: $idTrip")
            tripUseCase.getTripById(idTrip)?.let { trip ->
                val receipt = receiptUseCase.processReceipt(
                    trip = trip,
                    driverDirection = driverDirection,
                    driverPostalCode = driverPostalCode,
                    driverCity = driverCity,
                    clientNameAndSurname = clientNameAndSurname,
                    clientNIF = clientNIF,
                    clientDirection = clientDirection,
                    clientPostalCode = clientPostalCode,
                    clientCity = clientCity,
                )

                val tripDispatch = dispatchUseCase.getDispatchByTripId(idTrip)
                val brokenDownTaxTicket =
                    TicketsPrinter.getBrokenDownTax(
                        trip,
                        getApplication(),
                        tripDispatch?.toInfoDispatchModel()
                    )

                _effects.emit(OfflineInvoiceUiEffect.PrintInvoice(receipt, brokenDownTaxTicket))
                _effects.emit(OfflineInvoiceUiEffect.AskForCopy(receipt, brokenDownTaxTicket))
            } ?: run {
                Logs.d(TAG, "doBilling: trip is null")
            }
        }
    }

    fun onCopyDialogResponse(
        response: CustomDialog.CustomDialogResponse,
        receipt: String,
        brokenDownTaxTicket: String
    ) {
        when (response.buttonPressed) {
            ButtonType.ACCEPT -> viewModelScope.launch {
                _effects.emit(OfflineInvoiceUiEffect.ShowToast(R.string.invoice_printing_copy))
                _effects.emit(OfflineInvoiceUiEffect.PrintInvoice(receipt, brokenDownTaxTicket))
                _effects.emit(OfflineInvoiceUiEffect.NavigateBack)
            }

            ButtonType.CANCEL -> viewModelScope.launch {
                _effects.emit(OfflineInvoiceUiEffect.NavigateBack)
            }

            else -> Unit
        }
    }

    fun onInvoicePrinted(idTrip: Long) {
        updateTripInvoiceStatus(idTrip)
    }

    fun updateTripInvoiceStatus(idTrip: Long) {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "updateTripInvoiceStatus: tripId: $idTrip")
            tripUseCase.getTripById(idTrip)?.let { trip ->
                trip.invoiceAlreadyGenerated = true
                tripUseCase.updateTripInvoiceFlag(trip.id, trip.invoiceAlreadyGenerated)
            }
        }
    }

    private fun areFieldsEmpty(state: OfflineInvoiceUiState): Boolean {
        val fields = listOf(
            state.driverDirection,
            state.driverPostalCode,
            state.driverCity,
            state.clientNameAndSurname,
            state.clientNIF,
            state.clientDirection,
            state.clientPostalCode,
            state.clientCity
        )
        return fields.any { it.isBlank() }
    }

    private fun isValidDni(value: String): Boolean {
        return value.isNotBlank()
    }
}

@Composable
fun OfflineInvoiceRoute(
    tripId: Long,
    viewModel: OfflineInvoiceViewModel,
    sharedViewModel: ComposeSharedViewModel,
    onNavigateBack: () -> Unit,
    onShowToast: (Int) -> Unit,
    onOpenDialog: (CustomDialog.CustomDialogModel, (CustomDialog.CustomDialogResponse) -> Unit) -> Unit
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    LaunchedEffect(tripId) {
        viewModel.onEvent(OfflineInvoiceUiEvent.SetTripId(tripId))
        viewModel.onEvent(OfflineInvoiceUiEvent.LoadPreferences)
    }

    LaunchedEffect(Unit) {
        viewModel.effects.collectLatest { effect ->
            when (effect) {
                is OfflineInvoiceUiEffect.ShowToast -> onShowToast(effect.messageRes)

                OfflineInvoiceUiEffect.NavigateBack -> onNavigateBack()

                is OfflineInvoiceUiEffect.PrintInvoice -> {
                    onShowToast(R.string.printing_invoice)
                    sharedViewModel.printTicket(effect.receipt, false)
                    sharedViewModel.printTicket(effect.brokenDownTaxTicket, true)
                    viewModel.onInvoicePrinted(tripId)
                }

                is OfflineInvoiceUiEffect.AskForCopy -> {
                    onOpenDialog(
                        CustomDialog.CustomDialogModel(
                            title = context.getString(R.string.btn_invoice),
                            description = context.getString(R.string.ask_invoice_copy),
                            buttons = arrayListOf(ButtonType.CANCEL, ButtonType.ACCEPT),
                            isCancellable = false
                        )
                    ) { response ->
                        viewModel.onCopyDialogResponse(
                            response = response,
                            receipt = effect.receipt,
                            brokenDownTaxTicket = effect.brokenDownTaxTicket
                        )
                    }
                }
            }
        }
    }

    BackHandler { onNavigateBack() }

    OfflineInvoiceScreen(
        uiState = uiState,
        onEvent = viewModel::onEvent
    )
}

@Composable
private fun OfflineInvoiceButton(
    modifier: Modifier = Modifier,
    state: OfflineInvoiceButtonState,
    onClick: () -> Unit
) {
    if (!state.visible) return

    val context = LocalContext.current
    val bg = when (state.background) {
        OfflineInvoiceButtonBackground.GREEN -> Color(0xFF2E7D32)
        OfflineInvoiceButtonBackground.RED -> Color(0xFFC62828)
        OfflineInvoiceButtonBackground.BLUE -> Color(0xFF1565C0)
        OfflineInvoiceButtonBackground.YELLOW -> Color(0xFFF9A825)
        OfflineInvoiceButtonBackground.ORANGE -> Color(0xFFEF6C00)
        OfflineInvoiceButtonBackground.DEFAULT -> Color(0xFF1565C0)
        OfflineInvoiceButtonBackground.BLACK -> Color.Black
        OfflineInvoiceButtonBackground.TRANSPARENT -> Color.Transparent
    }

    AndroidView(
        modifier = modifier
            .fillMaxHeight()
            .padding(1.dp),
        factory = { ctx ->
            CustomButton(ctx).apply {
                setButtonType(state.buttonType.value)
                setButtonStyle(
                    if (state.enabled) CustomButton.StyleButton.ENABLE else CustomButton.StyleButton.DISABLE
                )
                changeBackground(
                    when (state.background) {
                        OfflineInvoiceButtonBackground.GREEN -> CustomButton.BackgroundButtonColor.GREEN
                        OfflineInvoiceButtonBackground.RED -> CustomButton.BackgroundButtonColor.RED
                        OfflineInvoiceButtonBackground.BLUE -> CustomButton.BackgroundButtonColor.BLUE
                        OfflineInvoiceButtonBackground.YELLOW -> CustomButton.BackgroundButtonColor.YELLOW
                        OfflineInvoiceButtonBackground.ORANGE -> CustomButton.BackgroundButtonColor.ORANGE
                        OfflineInvoiceButtonBackground.DEFAULT -> CustomButton.BackgroundButtonColor.DEFAULT
                        OfflineInvoiceButtonBackground.BLACK -> CustomButton.BackgroundButtonColor.BLACK
                        OfflineInvoiceButtonBackground.TRANSPARENT -> CustomButton.BackgroundButtonColor.TRANSPARENT
                    }
                )
                setAction(onClick)
            }
        },
        update = { button ->
            button.setButtonType(state.buttonType.value)
            button.setButtonStyle(
                if (state.enabled) CustomButton.StyleButton.ENABLE else CustomButton.StyleButton.DISABLE
            )
            button.changeBackground(
                when (state.background) {
                    OfflineInvoiceButtonBackground.GREEN -> CustomButton.BackgroundButtonColor.GREEN
                    OfflineInvoiceButtonBackground.RED -> CustomButton.BackgroundButtonColor.RED
                    OfflineInvoiceButtonBackground.BLUE -> CustomButton.BackgroundButtonColor.BLUE
                    OfflineInvoiceButtonBackground.YELLOW -> CustomButton.BackgroundButtonColor.YELLOW
                    OfflineInvoiceButtonBackground.ORANGE -> CustomButton.BackgroundButtonColor.ORANGE
                    OfflineInvoiceButtonBackground.DEFAULT -> CustomButton.BackgroundButtonColor.DEFAULT
                    OfflineInvoiceButtonBackground.BLACK -> CustomButton.BackgroundButtonColor.BLACK
                    OfflineInvoiceButtonBackground.TRANSPARENT -> CustomButton.BackgroundButtonColor.TRANSPARENT
                }
            )
            button.setAction(onClick)
        }
    )
}

sealed interface DispatchReceivedUiEvent {
    data object OnBackPressed : DispatchReceivedUiEvent
    data object OnAcceptClicked : DispatchReceivedUiEvent
    data object OnCancelClicked : DispatchReceivedUiEvent
    data object OnConfirmCancelDialog : DispatchReceivedUiEvent

import androidx.compose.runtime.getValue
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.unit.dp

@Composable
fun DispatchReceivedRoute(
    idDispatch: Long,
    viewModel: DispatchReceivedComposeViewModel,
    onNavigateToInfoDispatch: (Long) -> Unit,
    onNavigateBack: () -> Unit,
    onRequestPhonePermission: () -> Unit,
    onCloseLocateOnStandDialog: () -> Unit,
) {
    val uiState by viewModel.uiState.collectAsState()

    LaunchedEffect(idDispatch) {
        viewModel.onEvent(DispatchReceivedUiEvent.OnDispatchLoaded(idDispatch))
        viewModel.onEvent(DispatchReceivedUiEvent.OnScreenStarted)
    }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collect { effect ->
            when (effect) {
                is DispatchReceivedUiEffect.NavigateToInfoDispatch -> onNavigateToInfoDispatch(effect.tripId)
                DispatchReceivedUiEffect.NavigateBack -> onNavigateBack()
                DispatchReceivedUiEffect.RequestPhonePermission -> onRequestPhonePermission()
                DispatchReceivedUiEffect.CloseLocateOnStandDialog -> onCloseLocateOnStandDialog()
                DispatchReceivedUiEffect.StopCallTimer -> Unit
                is DispatchReceivedUiEffect.ShowCustomDialog -> Unit
            }
        }
    }

    DispatchReceivedScreen(
        uiState = uiState,
        onAcceptClick = { viewModel.onEvent(DispatchReceivedUiEvent.OnAcceptClicked) },
        onCancelClick = { viewModel.onEvent(DispatchReceivedUiEvent.OnCancelClicked) },
        onDialogConfirm = { viewModel.onDialogConfirmed() },
        onDialogDismiss = { viewModel.onDialogDismissed() }
    )
}

@Composable
fun DispatchReceivedScreen(
    uiState: DispatchReceivedUiState,
    onAcceptClick: () -> Unit,
    onCancelClick: () -> Unit,

    onDialogDismiss: () -> Unit,
) {
    Column(
        modifier = Modifier
            .fillMaxSize()
            .background(Color.White)
            .padding(16.dp),
        verticalArrangement = Arrangement.spacedBy(12.dp)
    ) {
        uiState.dispatch?.let { dispatch ->
            Text(text = "Hora: ${dispatch.hour}")
            Text(text = "Zona: ${dispatch.pickUpZone}")
            Text(text = "Dirección: ${dispatch.simpleAddress}")
            Text(text = "Ciudad: ${dispatch.city}")
            Text(text = "Pasajero: ${dispatch.passengerName}")
        }

        DispatchReceivedButtons(
            state = uiState.buttons,
            onAcceptClick = onAcceptClick,
            onCancelClick = onCancelClick
        )
    }

    if (uiState.showCancelDialog && uiState.dialog != null) {
        AlertDialog(
            onDismissRequest = onDialogDismiss,
            title = { Text(uiState.dialog.title) },
            text = { Text(uiState.dialog.description) },
            confirmButton = {
                TextButton(onClick = onDialogConfirm) {
                    Text(uiState.dialog.acceptText)
                }
            },
            dismissButton = {
                TextButton(onClick = onDialogDismiss) {
                    Text(uiState.dialog.cancelText)
                }
            }
        )
    }
}

## Reducer implementation

This is the “one-by-one” mapping from the fragment logic into Compose state.

```kotlin
package ifac.td.taxi.ui.screen.dispatchreceived

object DispatchReceivedButtonsReducer {

    fun reduce(input: DispatchReceivedReducerInput): DispatchReceivedButtonsState {
        val acceptState = if (input.dispatchLoaded) {
            DispatchReceivedButtonsState.ButtonState.VisibleEnabled
        } else {
            DispatchReceivedButtonsState.ButtonState.VisibleDisabled
        }

        val cancelState = when {
            !input.dispatchLoaded -> DispatchReceivedButtonsState.CancelButtonState.Disabled
            input.showDialogOnCancel -> DispatchReceivedButtonsState.CancelButtonState.RequiresConfirmation
            else -> DispatchReceivedButtonsState.CancelButtonState.DirectAction
        }

        return DispatchReceivedButtonsState(
            accept = acceptState,
            cancel = cancelState,
            loading = input.isLoading,
            cancelDialogVisible = input.showDialogOnCancel && input.dispatchLoaded,
            phonePermissionDialogVisible = input.phoneCallPermissionGranted == false
        )
    }
}
```

## Why this mirrors the fragment

### `dispatchFlow.collect { ... }`
In the fragment, once dispatch arrives, the UI fields are populated.  
In Compose, that means `dispatchLoaded = true`, which enables the buttons.

### `showDialogOnCancelFlow.collect { ... }`
In the fragment, this changes the cancel button behavior:
- if `true`, clicking cancel opens a confirmation dialog
- if `false`, cancel is a direct action

That is represented by:

```kotlin
cancel = RequiresConfirmation
```

or

```kotlin
cancel = DirectAction
```

### `phoneCallPermissionFlow.collect { ... }`
If permission is denied, the fragment opens a warning dialog.  
That becomes:

```kotlin
phonePermissionDialogVisible = true
```

### `tripIdFlow.collect { ... }`
This is a navigation effect, not a button state, so it should be handled as a UI effect, not reducer state.

class DispatchReceivedComposeViewModel(
    private val dispatchUseCase: DispatchUseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    private val bravoConfigurationVariableDao: BravoConfigurationVariableDao,
    private val licensingUsecase: LicensingUseCase,
    application: Application,
) : AndroidViewModel(application) {

    private val _uiState = MutableStateFlow(DispatchReceivedUiState())
    val uiState: StateFlow<DispatchReceivedUiState> = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<DispatchReceivedUiEffect>()
    val uiEffect: SharedFlow<DispatchReceivedUiEffect> = _uiEffect.asSharedFlow()

    private var currentDispatchId: Long? = null
    private var currentDispatch: Dispatch? = null

    fun initialize() = Unit

    fun onEvent(event: DispatchReceivedUiEvent) {
        when (event) {
            DispatchReceivedUiEvent.OnScreenStarted -> Unit
            is DispatchReceivedUiEvent.OnDispatchLoaded -> loadDispatch(event.idDispatch)
            DispatchReceivedUiEvent.OnAcceptClicked -> acceptDispatch()
            DispatchReceivedUiEvent.OnCancelClicked -> {
                if (_uiState.value.buttons.cancel.requiresConfirmation) {
                    viewModelScope.launch {
                        _uiEffect.emit(
                            DispatchReceivedUiEffect.ShowCustomDialog(
                                DispatchReceivedDialogState(
                                    title = "Advertencia",
                                    description = "¿Quieres rechazar este despacho?",
                                    acceptText = "Aceptar",
                                    cancelText = "Cancelar"
                                )
                            )
                        )
                    }
                } else {
                    rejectManualDispatch()
                }
            }
            DispatchReceivedUiEvent.OnConfirmCancelDialog -> rejectManualDispatch()
            DispatchReceivedUiEvent.OnDismissCancelDialog -> Unit
            DispatchReceivedUiEvent.OnBackPressed -> {
                viewModelScope.launch { _uiEffect.emit(DispatchReceivedUiEffect.NavigateBack) }
            }
        }
    }

    fun checkCancelDialogPermission() {
        viewModelScope.launch {
            val preferences = userPreferencesUseCase.getUserPreferences()
            val showDialog = preferences?.showConfRejectDispatch == true
            _uiState.value = _uiState.value.copy(
                buttons = DispatchReceivedButtonsReducer.reduce(
                    DispatchReceivedReducerInput(
                        dispatchLoaded = _uiState.value.dispatch != null,
                        showDialogOnCancel = showDialog,
                        isLoading = _uiState.value.isLoading
                    )
                )
            )
        }
    }

    fun onDialogConfirmed() = onEvent(DispatchReceivedUiEvent.OnConfirmCancelDialog)
    fun onDialogDismissed() = onEvent(DispatchReceivedUiEvent.OnDismissCancelDialog)

    fun onSharedDispatchChanged(dispatch: InfoDispatchModel?) = Unit
    fun onSharedTripChanged(trip: Trip?) = Unit
    fun onUpdatePrintCallbackChanged(callback: ((Trip) -> Unit)?) = Unit

    private fun loadDispatch(id: Long) {
        viewModelScope.launch {
            currentDispatchId = id
            val dispatch = dispatchUseCase.getDispatchById(id)
            if (dispatch != null) {
                currentDispatch = dispatch
                val model = dispatch.toDispatchReceivedModel(id)
                _uiState.value = _uiState.value.copy(
                    dispatch = DispatchReceivedContentUiState(
                        hour = model.pickUpTime.orEmpty(),
                        pickUpZone = model.pickUpZone.orEmpty(),
                        simpleAddress = model.smallPickUpAdress.orEmpty(),
                        city = model.city.orEmpty(),
                        passengerName = model.passengerName.orEmpty()
                    ),
                    buttons = DispatchReceivedButtonsReducer.reduce(
                        DispatchReceivedReducerInput(
                            dispatchLoaded = true,
                            showDialogOnCancel = _uiState.value.buttons.cancel.requiresConfirmation,
                            isLoading = false
                        )
                    )
                )
            }
        }
    }

    private fun acceptDispatch() {
        viewModelScope.launch {
            val dispatch = currentDispatch ?: return@launch
            if (dispatch.requireSignature) {
                dispatchRequireSignature(dispatch)
            } else if (dispatch.requireVoucher) {
                dispatchRequireVoucher(dispatch)
            } else if (dispatch.requireQr) {
                dispatchRequireQR(dispatch)
            } else {
                validCredit(
                    trip = null,
                    dispatch = dispatch,
                    callback = {}
                )
            }
        }
    }

    private fun rejectManualDispatch() {
        viewModelScope.launch {
            _uiEffect.emit(DispatchReceivedUiEffect.StopCallTimer)
            _uiEffect.emit(DispatchReceivedUiEffect.NavigateBack)
        }
    }

    private fun dispatchRequireSignature(dispatch: Dispatch) {
        viewModelScope.launch {
            _uiEffect.emit(DispatchReceivedUiEffect.NavigateToInfoDispatch(dispatch.longDispatchNumber))
        }
    }

    private fun dispatchRequireVoucher(dispatch: Dispatch) {
        viewModelScope.launch {
            _uiEffect.emit(DispatchReceivedUiEffect.NavigateToInfoDispatch(dispatch.longDispatchNumber))
        }
    }

    private fun dispatchRequireQR(dispatch: Dispatch) {
        viewModelScope.launch {
            _uiEffect.emit(DispatchReceivedUiEffect.NavigateToInfoDispatch(dispatch.longDispatchNumber))
        }
    }

    private fun validCredit(
        trip: Trip?,
        dispatch: Dispatch,
        callback: (Trip) -> Unit,
    ) {
        trip?.let {
            viewModelScope.launch(Dispatchers.IO) {
                try {
                    tripUseCase.endTrip(it, PaymentMethod.SUBSCRIBER)
                    shiftStatusUseCase.getStatus()?.let { status ->
                        ticketUseCase.prepareTicket(
                            trip = it,
                            dispatch = dispatch.toInfoDispatchModel(),
                            shiftStatus = status
                        )
                    }
                    navigatePostPayment()
                    callback(it)
                    _uiEffect.emit(DispatchReceivedUiEffect.NavigateToInfoDispatch(dispatch.longDispatchNumber))
                } catch (_: Exception) {
                }
            }
        }
    }

    private fun navigatePostPayment() {
        viewModelScope.launch(Dispatchers.IO) {
            try {
                val status = shiftStatusUseCase.checkForSubStateModifications(ifConstants.STATE_FOR_HIRE)
                status?.let { shiftStatusUseCase.setStatus(it, true) }
            } catch (_: Exception) {
            }
        }
    }

    private val tripUseCase: TripUseCase by lazy { error("Inject TripUseCase in real project") }
    private val shiftStatusUseCase: ShiftStatusUseCase by lazy { error("Inject ShiftStatusUseCase in real project") }
    private val ticketUseCase: TicketUseCase by lazy { error("Inject TicketUseCase in real project") }
}

// ## 1) Compose Theme object

import androidx.compose.material3.MaterialTheme
import androidx.compose.material3.ColorScheme
import androidx.compose.material3.lightColorScheme
import androidx.compose.runtime.Composable
import androidx.compose.ui.graphics.Color
import androidx.compose.material3.Typography
import androidx.compose.ui.text.TextStyle
import androidx.compose.ui.text.font.FontWeight
import androidx.compose.ui.unit.sp

object AppColors {
    val Blue = Color(0xFF2196F3)
    val Yellow = Color(0xFFFFC107)
    val Red = Color(0xFFF44336)
    val Green = Color(0xFF4CAF50)
    val Grey = Color(0xFFBDBDBD)
    val Black = Color(0xFF000000)
    val White = Color(0xFFFFFFFF)
}

object AppTheme {
    val colorScheme: ColorScheme = lightColorScheme(
        primary = AppColors.Blue,
        secondary = AppColors.Yellow,
        tertiary = AppColors.Green,
        error = AppColors.Red,
        background = AppColors.White,
        surface = AppColors.White,
        onPrimary = AppColors.White,
        onSecondary = AppColors.Black,
        onTertiary = AppColors.White,
        onError = AppColors.White,
        onBackground = AppColors.Black,
        onSurface = AppColors.Black
    )

    val typography = Typography(
        bodyMedium = TextStyle(
            fontSize = 16.sp
        ),
        titleLarge = TextStyle(
            fontSize = 24.sp,
            fontWeight = FontWeight.Normal
        ),
        labelMedium = TextStyle(
            fontSize = 14.sp,
            fontWeight = FontWeight.Bold
        )
    )
}

@Composable
fun SmartTDTheme(
    content: @Composable () -> Unit
) {
    MaterialTheme(
        colorScheme = AppTheme.colorScheme,
        typography = AppTheme.typography,
        content = content
    )
}


// ## 2) Compose Modifier extensions for style-like reuse

These mirror the repeated XML properties such as size, padding, background, and row styles.

import androidx.compose.ui.Modifier

    .padding(end = 8.dp)

fun Modifier.rowHeaderLabel(): Modifier = this

fun Modifier.rowHeaderPlaceholder(): Modifier = this

    .padding(start = 16.dp, bottom = 8.dp)

// ## 3) Compose layout equivalent of the XML

This is a Compose implementation of the **visible structure** from the XML:
- root container
- centered progress indicator
- left half: zones section
- right half: trips section
- nested headers / separators / lists
- bottom “pending trips” panel

Since your XML uses `RecyclerView`, in Compose the natural equivalent is `LazyColumn`.
If you want to plug in existing adapters, you can use `AndroidView`, but below I’m using pure Compose.

import androidx.compose.foundation.layout.*
import androidx.compose.foundation.lazy.LazyColumn
import androidx.compose.foundation.lazy.items
import androidx.compose.material3.CircularProgressIndicator
import androidx.compose.material3.Divider
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.text.font.FontWeight
import androidx.compose.ui.unit.dp
import androidx.compose.ui.unit.sp
import androidx.compose.ui.graphics.Color
import androidx.compose.material3.Surface
import androidx.compose.ui.unit.Dp

@Composable
fun ZonesTripsScreen(
    showLoading: Boolean = false,
    zonesLeft: List<String> = emptyList(),
    zonesRight: List<String> = emptyList(),
    actualZoneTrips: List<String> = emptyList(),
    pendingTrips: List<String> = emptyList()
) {
    SmartTDTheme {
        Box(
            modifier = Modifier.fillMaxSize()
        ) {
            Row(
                modifier = Modifier.fillMaxSize()
            ) {
                // LEFT HALF - lytInfoZones
                Column(
                    modifier = Modifier
                        .weight(1f)
                        .fillMaxHeight()
                ) {
                    RowHeader(
                        placeholder = "INOM 5 KM",
                        stopText = "PARADA",
                        zoneText = "ZONA",
                        hiredText = "OCUPADO",
                        tripsText = "SERVICIOS"
                    )

                    Divider(modifier = Modifier.horizontalDividerStyle())

                    LazyColumn(
                        modifier = Modifier.weight(1f)
                    ) {
                        items(zonesLeft) { item ->
                            Text(
                                text = item,
                                modifier = Modifier
                                    .fillMaxWidth()
                                    .padding(8.dp),
                                fontSize = 16.sp
                            )
                        }
                    }

                    RowHeader(
                        placeholder = "UT 5 KM",
                        stopText = "PARADA",
                        zoneText = "ZONA",
                        hiredText = "OCUPADO",
                        tripsText = "SERVICIOS"
                    )

                    Divider(modifier = Modifier.horizontalDividerStyle())

                    LazyColumn(
                        modifier = Modifier.weight(1f)
                    ) {
                        items(zonesRight) { item ->
                            Text(
                                text = item,
                                modifier = Modifier
                                    .fillMaxWidth()
                                    .padding(8.dp),
                                fontSize = 16.sp
                            )
                        }
                    }
                }

                // RIGHT HALF - lytInfoTrips
                Column(
                    modifier = Modifier
                        .weight(1f)
                        .fillMaxHeight()
                ) {
                    Column(
                        modifier = Modifier
                            .fillMaxWidth()
                            .weight(1f)
                    ) {
                        RowHeader(
                            placeholder = "",
                            stopText = "PARADA",
                            zoneText = "ZONA",
                            hiredText = "OCUPADO",
                            tripsText = "SERVICIOS"
                        )

                        Divider(modifier = Modifier.horizontalDividerStyle())

                        LazyColumn(
                            modifier = Modifier.weight(1f)
                        ) {
                            items(actualZoneTrips) { item ->
                                Text(
                                    text = item,
                                    modifier = Modifier
                                        .fillMaxWidth()
                                        .padding(8.dp),
                                    fontSize = 16.sp
                                )
                            }
                        }
                    }

                    PendingTripsPanel(
                        pendingTrips = pendingTrips
                    )
                }
            }

            if (showLoading) {
                CircularProgressIndicator(
                    modifier = Modifier
                        .size(100.dp)
                        .align(Alignment.Center)
                )
            }
        }
    }
}


// ## 4) Reusable header row composable

This maps the repeated row blocks such as:
- `clRowZone`
- `clRowZone2`
- `clRowZone3`

@Composable
fun RowHeader(
    placeholder: String,
    stopText: String,
    zoneText: String,
    hiredText: String,
    tripsText: String
) {
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .height(60.dp)
            .padding(end = 8.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {
        Text(
            text = placeholder,
            modifier = Modifier
                .width(200.dp)
                .fillMaxHeight()
                .padding(start = 16.dp, bottom = 8.dp),
            fontSize = 16.sp
        )

        Text(
            text = stopText,
            modifier = Modifier
                .weight(1f)
                .fillMaxHeight()
                .padding(bottom = 8.dp),
            color = AppColors.Blue,
            fontWeight = FontWeight.Bold,
            fontSize = 16.sp
        )

        Text(
            text = zoneText,
            modifier = Modifier
                .weight(1f)
                .fillMaxHeight()
                .padding(bottom = 8.dp),
            color = AppColors.Yellow,
            fontWeight = FontWeight.Bold,
            fontSize = 16.sp
        )

        Text(
            text = hiredText,
            modifier = Modifier
                .weight(1f)
                .fillMaxHeight()
                .padding(bottom = 8.dp),
            color = AppColors.Red,
            fontWeight = FontWeight.Bold,
            fontSize = 16.sp
        )

        Text(
            text = tripsText,
            modifier = Modifier
                .weight(1f)
                .fillMaxHeight()
                .padding(bottom = 8.dp),
            color = AppColors.Green,
            fontWeight = FontWeight.Bold,
            fontSize = 16.sp
        )
    }
}

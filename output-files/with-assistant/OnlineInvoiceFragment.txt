And here is the exact branch logic for the accept click, one-by-one:

```kotlin
package ifac.td.taxi.ui.screen.onlineinvoice

import ifac.td.taxi.R

fun reduceOnlineInvoiceEffectOnAcceptClick(state: OnlineInvoiceUiState): OnlineInvoiceUiEffect {
    val nifNotEmpty = state.nif.isNotEmpty()

    return when {
        nifNotEmpty && state.areRequiredFieldsEmpty() ->
            OnlineInvoiceUiEffect.RequestFiscalData

        nifNotEmpty && !state.areRequiredFieldsEmpty() ->
            OnlineInvoiceUiEffect.GenerateInvoice(state.toFiscalData())

        else ->
            OnlineInvoiceUiEffect.ShowToast(R.string.toast_complete_todos_los_campos)
    }
}

fun OnlineInvoiceUiState.areRequiredFieldsEmpty(): Boolean {
    return listOf(
        companyName,
        streetName,
        number,
        city,
        postalCode,
        province,
        nif,
        email,
        country
    ).any { it.isBlank() }
}
```

This is the exact equivalent of:

- if NIF not empty and fields empty -> fetch fiscal data
- else if NIF not empty and fields not empty -> generate invoice
- else -> toast

# 3) Compose-friendly ViewModel with `UiState + UiEvent + SharedFlow<UiEffect>`

This replaces the fragment VM with a Compose-friendly architecture.

```kotlin
package ifac.td.taxi.ui.screen.onlineinvoice

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.R
import ifac.td.taxi.domain.model.Dispatch
import ifac.td.taxi.domain.model.SmartTDException
import ifac.td.taxi.domain.model.Trip
import ifac.td.taxi.domain.usecase.BravoRestInvoiceUseCase
import ifac.td.taxi.domain.usecase.DispatchUseCase
import ifac.td.taxi.domain.usecase.TripUseCase
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.repository.connections.rest.invoice.Resource
import ifac.td.taxi.ui.custom.button.ButtonType
import ifac.td.taxi.ui.model.FiscalData
import ifac.td.taxi.ui.model.SmartTDUIException
import ifac.td.taxi.viewmodel.model.UtilsModel.Companion.toInfoDispatchModel
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

class OnlineInvoiceComposeViewModel(
    application: Application,
    private val tripUseCase: TripUseCase,
    private val dispatchUseCase: DispatchUseCase,
    private val bravoRestInvoiceUseCase: BravoRestInvoiceUseCase
) : AndroidViewModel(application) {

    private val TAG = "OnlineInvoiceComposeVM"

    private val _uiState = MutableStateFlow(OnlineInvoiceUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<OnlineInvoiceUiEffect>()
    val uiEffect: SharedFlow<OnlineInvoiceUiEffect> = _uiEffect.asSharedFlow()

    fun onEvent(event: OnlineInvoiceUiEvent) {
        when (event) {
            is OnlineInvoiceUiEvent.TripLoaded -> {
                _uiState.update { it.copy(tripId = event.tripId) }
                getTrip(event.tripId)
            }

            is OnlineInvoiceUiEvent.NifChanged -> _uiState.update { it.copy(nif = event.value) }
            is OnlineInvoiceUiEvent.CompanyNameChanged -> _uiState.update { it.copy(companyName = event.value) }
            is OnlineInvoiceUiEvent.StreetNameChanged -> _uiState.update { it.copy(streetName = event.value) }
            is OnlineInvoiceUiEvent.NumberChanged -> _uiState.update { it.copy(number = event.value) }
            is OnlineInvoiceUiEvent.CityChanged -> _uiState.update { it.copy(city = event.value) }
            is OnlineInvoiceUiEvent.PostalCodeChanged -> _uiState.update { it.copy(postalCode = event.value) }
            is OnlineInvoiceUiEvent.ProvinceChanged -> _uiState.update { it.copy(province = event.value) }
            is OnlineInvoiceUiEvent.CountryChanged -> _uiState.update { it.copy(country = event.value) }
            is OnlineInvoiceUiEvent.EmailChanged -> _uiState.update { it.copy(email = event.value) }

            OnlineInvoiceUiEvent.AcceptClicked -> handleAcceptClicked()
            OnlineInvoiceUiEvent.CancelClicked -> viewModelScope.launch {
                _uiEffect.emit(OnlineInvoiceUiEffect.NavigateBack)
            }

            OnlineInvoiceUiEvent.RetryClicked -> {
                val tripId = _uiState.value.tripId ?: return
                getTrip(tripId)
            }
        }
    }

    private fun handleAcceptClicked() {
        val state = _uiState.value
        val effect = reduceOnlineInvoiceEffectOnAcceptClick(state)

        viewModelScope.launch {
            when (effect) {
                is OnlineInvoiceUiEffect.RequestFiscalData -> {
                    _uiState.update { it.copy(isLoading = true) }
                    _uiEffect.emit(effect)
                }

                is OnlineInvoiceUiEffect.GenerateInvoice -> {
                    _uiState.update { it.copy(isLoading = true) }
                    _uiEffect.emit(effect)
                }

                else -> _uiEffect.emit(effect)
            }
        }
    }

    fun getFiscalData(fiscalId: String) {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "getFiscalData: Fetching fiscal data for ID = $fiscalId")
            when (val result = bravoRestInvoiceUseCase.getFiscalData(fiscalId)) {
                is Resource.Success -> {
                    result.data?.let {
                        _uiState.update { state ->
                            state.copy(
                                isLoading = false,
                                isFormVisible = true,
                                companyName = it.companyName,
                                streetName = it.streetName,
                                number = it.number,
                                city = it.city,
                                postalCode = it.postalCode,
                                province = it.province,
                                country = it.country,
                                email = it.email
                            )
                        }
                    }
                }

                is Resource.Error -> {
                    result.exception?.let {
                        emitError(it)
                    }
                }
            }
        }
    }

    fun generateInvoice(userFiscalData: FiscalData) {
        viewModelScope.launch(Dispatchers.IO) {
            _uiState.update { it.copy(isLoading = true) }
            _uiState.value.tripId?.let { tripId ->
                val trip = tripUseCase.getTripById(tripId)
                if (trip == null) {
                    emitError(SmartTDException.DATABASE_ERROR)
                    return@launch
                }

                var dispatch: Dispatch? = null
                if (trip.fromDispatch && trip.pkDispatchId != null) {
                    dispatch = dispatchUseCase.getDispatchById(trip.pkDispatchId!!)
                }

                when (val result = bravoRestInvoiceUseCase.generateInvoice(
                    trip,
                    dispatch?.toInfoDispatchModel(),
                    fiscalData = userFiscalData
                )) {
                    is Resource.Success -> {
                        updateTripInvoiceStatus(trip)
                        _uiState.update { it.copy(isLoading = false) }
                        _uiEffect.emit(OnlineInvoiceUiEffect.ShowDialog(
                            titleResId = R.string.dialog_success,
                            descriptionResId = R.string.online_invoice_sent_success,
                            buttons = listOf(ButtonType.ACCEPT)
                        ))
                        _uiEffect.emit(OnlineInvoiceUiEffect.NavigateBack)
                    }

                    is Resource.Error -> {
                        result.exception?.let { emitError(it) }
                    }
                }
            }
        }
    }

    fun getTrip(id: Long) {
        viewModelScope.launch(Dispatchers.IO) {
            _uiState.update { it.copy(isLoading = true) }
            val trip = tripUseCase.getTripById(id)
            if (trip != null) {
                _uiState.update {
                    it.copy(
                        isLoading = false,
                        tripId = trip.id,
                        nif = trip.nif.orEmpty()
                    )
                }
                if (!trip.nif.isNullOrEmpty()) {
                    _uiEffect.emit(OnlineInvoiceUiEffect.RequestFiscalData)
                }
            } else {
                emitError(SmartTDException.DATABASE_ERROR)
            }
        }
    }

    private suspend fun updateTripInvoiceStatus(trip: Trip) {
        trip.invoiceAlreadyGenerated = true
        tripUseCase.updateTripInvoiceFlag(trip.id, trip.invoiceAlreadyGenerated)
    }

    private suspend fun emitError(exception: Throwable) {
        _uiState.update { it.copy(isLoading = false) }
        _uiEffect.emit(
            OnlineInvoiceUiEffect.ShowDialog(
                titleResId = R.string.dialog_error_title,
                descriptionResId = SmartTDUIException.fromException(exception).stringResId,
                buttons = listOf(ButtonType.ACCEPT),
                onAcceptNavigateBack = true
            )
        )
    }

    private suspend fun emitError(exception: SmartTDException) {
        _uiState.update { it.copy(isLoading = false) }
        _uiEffect.emit(
            OnlineInvoiceUiEffect.ShowDialog(
                titleResId = R.string.dialog_error_title,
                descriptionResId = SmartTDUIException.fromException(exception).stringResId,
                buttons = listOf(ButtonType.ACCEPT),
                onAcceptNavigateBack = true
            )
        )
    }
}
```

```kotlin
package ifac.td.taxi.ui.screen.onlineinvoice

import androidx.compose.foundation.background
import androidx.compose.foundation.layout.*
import androidx.compose.foundation.rememberScrollState
import androidx.compose.foundation.verticalScroll
import androidx.compose.material3.CircularProgressIndicator
import androidx.compose.runtime.*

import androidx.compose.ui.graphics.Color
import androidx.compose.ui.platform.LocalLifecycleOwner
import androidx.compose.ui.res.stringResource
import androidx.compose.ui.unit.dp
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.lifecycle.repeatOnLifecycle
import androidx.navigation.NavController
import ifac.td.taxi.R
import ifac.td.taxi.ui.custom.button.CustomButton
import ifac.td.taxi.ui.custom.button.ButtonType
import kotlinx.coroutines.launch

@Composable
fun OnlineInvoiceScreen(
    tripId: Long,
    navController: NavController,
    viewModel: OnlineInvoiceComposeViewModel,
    mainActivityViewModel: OnlineInvoiceMainActivityViewModel,
    onShowHeader: (Boolean) -> Unit,
    onShowToast: (Int) -> Unit,
    onOpenDialog: (title: String, description: String, onAccept: () -> Unit) -> Unit
) {
    val state by viewModel.uiState.collectAsStateWithLifecycle()
    val lifecycleOwner = LocalLifecycleOwner.current
    val scope = rememberCoroutineScope()

    LaunchedEffect(tripId) {
        onShowHeader(true)
        viewModel.onEvent(OnlineInvoiceUiEvent.TripLoaded(tripId))
    }

    LaunchedEffect(Unit) {
        lifecycleOwner.repeatOnLifecycle(androidx.lifecycle.Lifecycle.State.STARTED) {
            launch {
                viewModel.uiEffect.collect { effect ->
                    when (effect) {
                        OnlineInvoiceUiEffect.NavigateBack -> navController.popBackStack()
                        OnlineInvoiceUiEffect.RequestFiscalData -> viewModel.getFiscalData(state.nif.trim())
                        is OnlineInvoiceUiEffect.GenerateInvoice -> viewModel.generateInvoice(effect.fiscalData)
                        is OnlineInvoiceUiEffect.ShowToast -> onShowToast(effect.resId)
                        is OnlineInvoiceUiEffect.ShowDialog -> {
                            onOpenDialog(
                                title = stringResource(effect.titleResId),
                                description = stringResource(effect.descriptionResId),
                                onAccept = {
                                    if (effect.onAcceptNavigateBack) navController.popBackStack()
                                }
                            )
                        }
                    }
                }
            }
        }
    }

    Box(modifier = Modifier.fillMaxSize()) {
        Column(
            modifier = Modifier
                .fillMaxSize()
                .verticalScroll(rememberScrollState())
                .padding(16.dp)
        ) {
            OnlineInvoiceField(
                label = stringResource(R.string.nif_nif),
                value = state.nif,
                onValueChange = { viewModel.onEvent(OnlineInvoiceUiEvent.NifChanged(it)) }
            )

            if (state.isFormVisible) {
                OnlineInvoiceField(stringResource(R.string.nif_empresa), state.companyName) {
                    viewModel.onEvent(OnlineInvoiceUiEvent.CompanyNameChanged(it))
                }
                OnlineInvoiceField(stringResource(R.string.nif_calle), state.streetName) {
                    viewModel.onEvent(OnlineInvoiceUiEvent.StreetNameChanged(it))
                }
                OnlineInvoiceField(stringResource(R.string.nif_numero), state.number) {
                    viewModel.onEvent(OnlineInvoiceUiEvent.NumberChanged(it))
                }
                OnlineInvoiceField(stringResource(R.string.nif_ciudad), state.city) {
                    viewModel.onEvent(OnlineInvoiceUiEvent.CityChanged(it))
                }
                OnlineInvoiceField(stringResource(R.string.nif_codpostal), state.postalCode) {
                    viewModel.onEvent(OnlineInvoiceUiEvent.PostalCodeChanged(it))
                }
                OnlineInvoiceField(stringResource(R.string.nif_provincia), state.province) {
                    viewModel.onEvent(OnlineInvoiceUiEvent.ProvinceChanged(it))
                }
                OnlineInvoiceField(stringResource(R.string.nif_email), state.email) {
                    viewModel.onEvent(OnlineInvoiceUiEvent.EmailChanged(it))
                }
                OnlineInvoiceField(stringResource(R.string.nif_pais), state.country) {
                    viewModel.onEvent(OnlineInvoiceUiEvent.CountryChanged(it))
                }
            }
        }

        if (state.isLoading) {
            Box(
                modifier = Modifier
                    .fillMaxSize()
                    .background(Color(0x4D000000))
            ) {
                CircularProgressIndicator(modifier = Modifier.padding(24.dp))
            }
        }

        Row(
            modifier = Modifier
                .fillMaxWidth()
                .align(androidx.compose.ui.Alignment.BottomCenter)
                .height(96.dp)
        ) {
            ComposeCustomButton(
                modifier = Modifier.weight(1f),
                text = stringResource(R.string.btn_cancel).uppercase(),
                style = CustomButton.StyleButton.ENABLE,
                background = CustomButton.BackgroundButtonColor.RED,
                onClick = { viewModel.onEvent(OnlineInvoiceUiEvent.CancelClicked) }
            )
            ComposeCustomButton(
                modifier = Modifier.weight(1f),
                text = stringResource(R.string.btn_accept).uppercase(),
                style = if (state.isLoading) CustomButton.StyleButton.LOADING else CustomButton.StyleButton.ENABLE,
                background = CustomButton.BackgroundButtonColor.GREEN,
                onClick = { viewModel.onEvent(OnlineInvoiceUiEvent.AcceptClicked) }
            )
        }
    }
}
```

# 6) Compose button wrapper matching the CustomButton API

This is a Compose replacement for the custom button behavior.

```kotlin
package ifac.td.taxi.ui.screen.onlineinvoice

import androidx.compose.foundation.background
import androidx.compose.foundation.clickable
import androidx.compose.foundation.layout.*
import androidx.compose.material3.CircularProgressIndicator
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.unit.dp
import ifac.td.taxi.ui.custom.button.CustomButton

@Composable
fun ComposeCustomButton(
    modifier: Modifier = Modifier,
    text: String,
    style: CustomButton.StyleButton,
    background: CustomButton.BackgroundButtonColor,
    onClick: () -> Unit
) {
    val bgColor = when (background) {
        CustomButton.BackgroundButtonColor.GREEN -> Color(0xFF2E7D32)
        CustomButton.BackgroundButtonColor.RED -> Color(0xFFC62828)
        CustomButton.BackgroundButtonColor.BLUE -> Color(0xFF1565C0)
        CustomButton.BackgroundButtonColor.YELLOW -> Color(0xFFF9A825)
        CustomButton.BackgroundButtonColor.ORANGE -> Color(0xFFEF6C00)
        CustomButton.BackgroundButtonColor.BLACK -> Color.Black
        CustomButton.BackgroundButtonColor.TRANSPARENT -> Color.Transparent
        CustomButton.BackgroundButtonColor.DEFAULT -> Color(0xFF1565C0)
    }

    Box(
        modifier = modifier
            .height(64.dp)
            .padding(1.dp)
            .background(bgColor)
            .clickable(enabled = style == CustomButton.StyleButton.ENABLE) { onClick() },
        contentAlignment = Alignment.Center
    ) {
        when (style) {
            CustomButton.StyleButton.LOADING -> CircularProgressIndicator()
            else -> Text(text = text, color = Color.White)
        }
    }
}
```

- cancel button navigates back

```kotlin
data class OnlineInvoiceUiState(
    val isLoading: Boolean = false,
    val isFormVisible: Boolean = false,
    val tripId: Long? = null,
    val nif: String = "",
    val companyName: String = "",
    val streetName: String = "",
    val number: String = "",
    val city: String = "",
    val postalCode: String = "",
    val province: String = "",
    val country: String = "",
    val email: String = ""
) {
    fun toFiscalData(): FiscalData = FiscalData(
        fiscalID = nif,
        companyName = companyName,
        streetName = streetName,
        number = number,
        city = city,
        postalCode = postalCode,
        province = province,
        country = country,
        email = email
    )

    fun areRequiredFieldsEmpty(): Boolean = listOf(
        companyName, streetName, number, city, postalCode, province, nif, email, country
    ).any { it.isBlank() }
}

sealed interface OnlineInvoiceUiEvent {
    data class TripLoaded(val tripId: Long) : OnlineInvoiceUiEvent
    data class NifChanged(val value: String) : OnlineInvoiceUiEvent
    data class CompanyNameChanged(val value: String) : OnlineInvoiceUiEvent
    data class StreetNameChanged(val value: String) : OnlineInvoiceUiEvent
    data class NumberChanged(val value: String) : OnlineInvoiceUiEvent
    data class CityChanged(val value: String) : OnlineInvoiceUiEvent
    data class PostalCodeChanged(val value: String) : OnlineInvoiceUiEvent
    data class ProvinceChanged(val value: String) : OnlineInvoiceUiEvent
    data class CountryChanged(val value: String) : OnlineInvoiceUiEvent
    data class EmailChanged(val value: String) : OnlineInvoiceUiEvent
    data object AcceptClicked : OnlineInvoiceUiEvent
    data object CancelClicked : OnlineInvoiceUiEvent
}

sealed interface OnlineInvoiceUiEffect {
    data object NavigateBack : OnlineInvoiceUiEffect
    data object RequestFiscalData : OnlineInvoiceUiEffect
    data class GenerateInvoice(val fiscalData: FiscalData) : OnlineInvoiceUiEffect
    data class ShowToast(@StringRes val resId: Int) : OnlineInvoiceUiEffect
    data class ShowDialog(
        @StringRes val titleResId: Int,
        @StringRes val descriptionResId: Int,
        val buttons: List<ButtonType>,
        val onAcceptNavigateBack: Boolean = false
    ) : OnlineInvoiceUiEffect
}

data class ButtonUiState(
    val style: CustomButton.StyleButton,
    val background: CustomButton.BackgroundButtonColor,
    val visible: Boolean = true
)

data class OnlineInvoiceButtonsState(
    val accept: ButtonUiState,
    val cancel: ButtonUiState

                background = CustomButton.BackgroundButtonColor.GREEN,

/**
 * Exact reducer mirroring the fragment's accept-button branch logic:
 * 1) nifNotEmpty && areFieldsEmpty() -> fetch fiscal data
 * 2) nifNotEmpty && !areFieldsEmpty() -> generate invoice
 * 3) else -> toast complete all fields
 */
fun reduceOnlineInvoiceAcceptEffect(state: OnlineInvoiceUiState): OnlineInvoiceUiEffect {
    val nifNotEmpty = state.nif.isNotEmpty()
    return when {
        nifNotEmpty && state.areRequiredFieldsEmpty() ->
            OnlineInvoiceUiEffect.RequestFiscalData
        nifNotEmpty && !state.areRequiredFieldsEmpty() ->
            OnlineInvoiceUiEffect.GenerateInvoice(state.toFiscalData())
        else ->
            OnlineInvoiceUiEffect.ShowToast(R.string.toast_complete_todos_los_campos)
    }
}

class OnlineInvoiceComposeViewModel(
    application: Application,
    private val tripUseCase: TripUseCase,
    private val dispatchUseCase: DispatchUseCase,
    private val bravoRestInvoiceUseCase: BravoRestInvoiceUseCase
) : BaseComposeViewModel(application) {

    private val TAG = "OnlineInvoiceComposeVM"

    private val _uiState = MutableStateFlow(OnlineInvoiceUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<OnlineInvoiceUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()

    fun onEvent(event: OnlineInvoiceUiEvent) {
        when (event) {
            is OnlineInvoiceUiEvent.TripLoaded -> {
                _uiState.update { it.copy(tripId = event.tripId, isLoading = true) }
                getTrip(event.tripId)
            }

            is OnlineInvoiceUiEvent.NifChanged -> _uiState.update { it.copy(nif = event.value) }
            is OnlineInvoiceUiEvent.CompanyNameChanged -> _uiState.update { it.copy(companyName = event.value) }
            is OnlineInvoiceUiEvent.StreetNameChanged -> _uiState.update { it.copy(streetName = event.value) }
            is OnlineInvoiceUiEvent.NumberChanged -> _uiState.update { it.copy(number = event.value) }
            is OnlineInvoiceUiEvent.CityChanged -> _uiState.update { it.copy(city = event.value) }
            is OnlineInvoiceUiEvent.PostalCodeChanged -> _uiState.update { it.copy(postalCode = event.value) }
            is OnlineInvoiceUiEvent.ProvinceChanged -> _uiState.update { it.copy(province = event.value) }
            is OnlineInvoiceUiEvent.CountryChanged -> _uiState.update { it.copy(country = event.value) }
            is OnlineInvoiceUiEvent.EmailChanged -> _uiState.update { it.copy(email = event.value) }

            OnlineInvoiceUiEvent.AcceptClicked -> handleAcceptClicked()
            OnlineInvoiceUiEvent.CancelClicked -> viewModelScope.launch {
                _uiEffect.emit(OnlineInvoiceUiEffect.NavigateBack)
            }
        }
    }

    private fun handleAcceptClicked() {
        val effect = reduceOnlineInvoiceAcceptEffect(_uiState.value)
        viewModelScope.launch {
            when (effect) {
                is OnlineInvoiceUiEffect.RequestFiscalData -> {
                    _uiState.update { it.copy(isLoading = true) }
                    _uiEffect.emit(effect)
                }
                is OnlineInvoiceUiEffect.GenerateInvoice -> {
                    _uiState.update { it.copy(isLoading = true) }
                    _uiEffect.emit(effect)
                }
                else -> _uiEffect.emit(effect)
            }
        }
    }

    fun getTrip(id: Long) {
        viewModelScope.launch {
            try {
                val trip = tripUseCase.getTripById(id)
                if (trip != null) {
                    _uiState.update {
                        it.copy(
                            isLoading = false,
                            tripId = trip.id,
                            nif = trip.nif.orEmpty()
                        )
                    }
                    if (!trip.nif.isNullOrEmpty()) {
                        _uiEffect.emit(OnlineInvoiceUiEffect.RequestFiscalData)
                    }
                } else {
                    emitError(SmartTDException.DATABASE_ERROR)
                }
            } catch (t: Throwable) {
                emitError(t)
            }
        }
    }

    fun getFiscalData(fiscalId: String) {
        viewModelScope.launch {
            _uiState.update { it.copy(isLoading = true) }
            when (val result = bravoRestInvoiceUseCase.getFiscalData(fiscalId)) {
                is Resource.Success -> {
                    result.data?.let { fiscalData ->
                        _uiState.update {
                            it.copy(
                                isLoading = false,
                                isFormVisible = true,
                                companyName = fiscalData.companyName,
                                streetName = fiscalData.streetName,
                                number = fiscalData.number,
                                city = fiscalData.city,
                                postalCode = fiscalData.postalCode,
                                province = fiscalData.province,
                                country = fiscalData.country,
                                email = fiscalData.email
                            )
                        }
                    }
                }
                is Resource.Error -> {
                    result.exception?.let { emitError(it) }
                }
            }
        }
    }

    fun generateInvoice(userFiscalData: FiscalData) {
        viewModelScope.launch {
            _uiState.update { it.copy(isLoading = true) }
            val tripId = _uiState.value.tripId
            if (tripId == null) {
                emitError(SmartTDException.DATABASE_ERROR)
                return@launch
            }

            val trip = tripUseCase.getTripById(tripId)
            if (trip == null) {
                emitError(SmartTDException.DATABASE_ERROR)
                return@launch
            }

            var dispatch: Dispatch? = null
            if (trip.fromDispatch && trip.pkDispatchId != null) {
                dispatch = dispatchUseCase.getDispatchById(trip.pkDispatchId!!)
            }

            when (val result = bravoRestInvoiceUseCase.generateInvoice(
                trip,
                dispatch?.toInfoDispatchModel(),
                fiscalData = userFiscalData
            )) {
                is Resource.Success -> {
                    updateTripInvoiceStatus(trip)
                    _uiState.update { it.copy(isLoading = false) }
                    _uiEffect.emit(
                        OnlineInvoiceUiEffect.ShowDialog(
                            titleResId = R.string.dialog_success,
                            descriptionResId = R.string.online_invoice_sent_success,
                            buttons = listOf(ButtonType.ACCEPT),
                            onAcceptNavigateBack = true
                        )
                    )
                }
                is Resource.Error -> {
                    result.exception?.let { emitError(it) }
                }
            }
        }
    }

    private suspend fun updateTripInvoiceStatus(trip: Trip) {
        trip.invoiceAlreadyGenerated = true
        tripUseCase.updateTripInvoiceFlag(trip.id, trip.invoiceAlreadyGenerated)
    }

    private suspend fun emitError(exception: Throwable) {
        _uiState.update { it.copy(isLoading = false) }
        _uiEffect.emit(
            OnlineInvoiceUiEffect.ShowDialog(
                titleResId = R.string.dialog_error_title,
                descriptionResId = SmartTDUIException.fromException(exception).stringResId,
                buttons = listOf(ButtonType.ACCEPT),
                onAcceptNavigateBack = true
            )
        )
    }

    private suspend fun emitError(exception: SmartTDException) {
        _uiState.update { it.copy(isLoading = false) }
        _uiEffect.emit(
            OnlineInvoiceUiEffect.ShowDialog(
                titleResId = R.string.dialog_error_title,
                descriptionResId = SmartTDUIException.fromException(exception).stringResId,
                buttons = listOf(ButtonType.ACCEPT),
                onAcceptNavigateBack = true
            )
        )
    }
}

@Composable
fun OnlineInvoiceScreen(
    tripId: Long,
    navController: NavController,
    viewModel: OnlineInvoiceComposeViewModel,
    mainActivityViewModel: OnlineInvoiceMainActivityViewModel,
    onShowHeader: (Boolean) -> Unit,
    onShowToast: (Int) -> Unit,
    onOpenDialog: (title: String, description: String, onAccept: () -> Unit) -> Unit
) {
    val state by viewModel.uiState.collectAsStateWithLifecycle()
    val lifecycleOwner = LocalLifecycleOwner.current

    LaunchedEffect(tripId) {
        onShowHeader(true)
        viewModel.onEvent(OnlineInvoiceUiEvent.TripLoaded(tripId))
    }

    LaunchedEffect(Unit) {
        lifecycleOwner.repeatOnLifecycle(Lifecycle.State.STARTED) {
            launch {
                viewModel.uiEffect.collect { effect ->
                    when (effect) {
                        OnlineInvoiceUiEffect.NavigateBack -> navController.popBackStack()
                        OnlineInvoiceUiEffect.RequestFiscalData -> viewModel.getFiscalData(state.nif.trim())
                        is OnlineInvoiceUiEffect.GenerateInvoice -> viewModel.generateInvoice(effect.fiscalData)
                        is OnlineInvoiceUiEffect.ShowToast -> onShowToast(effect.resId)
                        is OnlineInvoiceUiEffect.ShowDialog -> {
                            onOpenDialog(
                                title = stringResource(effect.titleResId),
                                description = stringResource(effect.descriptionResId),
                                onAccept = {
                                    if (effect.onAcceptNavigateBack) navController.popBackStack()
                                }
                            )
                        }
                    }
                }
            }
        }
    }

    Box(modifier = Modifier.fillMaxSize()) {
        Column(
            modifier = Modifier
                .fillMaxSize()
                .verticalScroll(rememberScrollState())
                .padding(16.dp)
        ) {
            OnlineInvoiceField(
                label = stringResource(R.string.nif_nif),
                value = state.nif,
                onValueChange = { viewModel.onEvent(OnlineInvoiceUiEvent.NifChanged(it)) }
            )

            if (state.isFormVisible) {
                OnlineInvoiceField(stringResource(R.string.nif_empresa), state.companyName) {
                    viewModel.onEvent(OnlineInvoiceUiEvent.CompanyNameChanged(it))
                }
                OnlineInvoiceField(stringResource(R.string.nif_calle), state.streetName) {
                    viewModel.onEvent(OnlineInvoiceUiEvent.StreetNameChanged(it))
                }
                OnlineInvoiceField(stringResource(R.string.nif_numero), state.number) {
                    viewModel.onEvent(OnlineInvoiceUiEvent.NumberChanged(it))
                }
                OnlineInvoiceField(stringResource(R.string.nif_ciudad), state.city) {
                    viewModel.onEvent(OnlineInvoiceUiEvent.CityChanged(it))
                }
                OnlineInvoiceField(stringResource(R.string.nif_codpostal), state.postalCode) {
                    viewModel.onEvent(OnlineInvoiceUiEvent.PostalCodeChanged(it))
                }
                OnlineInvoiceField(stringResource(R.string.nif_provincia), state.province) {
                    viewModel.onEvent(OnlineInvoiceUiEvent.ProvinceChanged(it))
                }
                OnlineInvoiceField(stringResource(R.string.nif_email), state.email) {
                    viewModel.onEvent(OnlineInvoiceUiEvent.EmailChanged(it))
                }
                OnlineInvoiceField(stringResource(R.string.nif_pais), state.country) {
                    viewModel.onEvent(OnlineInvoiceUiEvent.CountryChanged(it))
                }
            }
        }

        if (state.isLoading) {
            Box(
                modifier = Modifier
                    .fillMaxSize()
                    .background(Color(0x4D000000))
            ) {
                CircularProgressIndicator(modifier = Modifier.align(Alignment.Center))
            }
        }

        Row(
            modifier = Modifier
                .fillMaxWidth()
                .align(Alignment.BottomCenter)
                .height(96.dp)
        ) {
            ComposeCustomButton(
                modifier = Modifier.weight(1f),
                text = stringResource(R.string.btn_cancel).uppercase(),
                style = CustomButton.StyleButton.ENABLE,
                background = CustomButton.BackgroundButtonColor.RED,
                onClick = { viewModel.onEvent(OnlineInvoiceUiEvent.CancelClicked) }
            )
            ComposeCustomButton(
                modifier = Modifier.weight(1f),
                text = stringResource(R.string.btn_accept).uppercase(),
                style = if (state.isLoading) CustomButton.StyleButton.LOADING else CustomButton.StyleButton.ENABLE,
                background = CustomButton.BackgroundButtonColor.GREEN,
                onClick = { viewModel.onEvent(OnlineInvoiceUiEvent.AcceptClicked) }
            )
        }
    }
}

@Composable

    value: String,
    onValueChange: (String) -> Unit
) {
    OutlinedTextField(
        modifier = Modifier.fillMaxWidth(),
        value = value,

}

@Composable
fun ComposeCustomButton(
    modifier: Modifier = Modifier,
    text: String,
    style: CustomButton.StyleButton,
    background: CustomButton.BackgroundButtonColor,
    onClick: () -> Unit
) {
    val bgColor = when (background) {
        CustomButton.BackgroundButtonColor.GREEN -> Color(0xFF2E7D32)
        CustomButton.BackgroundButtonColor.RED -> Color(0xFFC62828)
        CustomButton.BackgroundButtonColor.BLUE -> Color(0xFF1565C0)
        CustomButton.BackgroundButtonColor.YELLOW -> Color(0xFFF9A825)
        CustomButton.BackgroundButtonColor.ORANGE -> Color(0xFFEF6C00)
        CustomButton.BackgroundButtonColor.DEFAULT -> Color(0xFF1565C0)
        CustomButton.BackgroundButtonColor.BLACK -> Color.Black
        CustomButton.BackgroundButtonColor.TRANSPARENT -> Color.Transparent
    }

    Box(
        modifier = modifier
            .height(64.dp)
            .padding(1.dp)
            .background(bgColor)
            .clickable(enabled = style == CustomButton.StyleButton.ENABLE) { onClick() },
        contentAlignment = Alignment.Center
    ) {
        when (style) {
            CustomButton.StyleButton.LOADING -> CircularProgressIndicator()
            else -> Text(text = text, color = Color.White)
        }
    }
}
```

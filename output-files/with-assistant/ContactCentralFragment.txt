## 2) Compose-friendly ViewModel

This version replaces the fragment logic with a single state reducer and a `SharedFlow<ContactCentralUiEffect>`.

```kotlin
package ifac.td.taxi.viewmodel.compose

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.R
import ifac.td.taxi.domain.usecase.BluetoothLocalUseCase
import ifac.td.taxi.domain.usecase.PredefinedMessagesUseCase
import ifac.td.taxi.domain.usecase.ShiftStatusUseCase
import ifac.td.taxi.framework.sdk.bravocentral.usecase.BravoCentralConfigurationUseCase
import ifac.td.taxi.framework.sdk.usecase.VoicePetitionUseCase
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.repository.connections.service.model.ShortBreakStatus
import ifac.td.taxi.ui.screen.compose.*
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch

class ContactCentralComposeViewModel(
    private val bravoCentralConfigurationUseCase: BravoCentralConfigurationUseCase,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val predefinedMessagesUseCase: PredefinedMessagesUseCase,
    private val voicePetitionUseCase: VoicePetitionUseCase,
    private val bluetoothLocalUseCase: BluetoothLocalUseCase,
    application: Application,
) : AndroidViewModel(application) {

    private val TAG = "ContactCentralComposeVM"

    private val _uiState = MutableStateFlow(ContactCentralUiState())
    val uiState: StateFlow<ContactCentralUiState> = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<ContactCentralUiEffect>(extraBufferCapacity = 1)
    val uiEffect: SharedFlow<ContactCentralUiEffect> = _uiEffect.asSharedFlow()

    private var hasPredefinedMessages: Boolean = false
    private var hasShortBreak: Boolean = false
    private var isITopTaximeter: Boolean = false

    init {
        viewModelScope.launch(Dispatchers.IO) {
            hasPredefinedMessages = predefinedMessagesUseCase.getPredefinedMessages().isNotEmpty()
            hasShortBreak = bravoCentralConfigurationUseCase.getBravoConfigurationVariable()?.hasShortBreak ?: false
            reduceButtons()
        }
    }

    fun onEvent(event: ContactCentralUiEvent) {
        when (event) {
            ContactCentralUiEvent.OnCancelClicked -> {
                emitEffect(ContactCentralUiEffect.NavigateBack)
            }

            ContactCentralUiEvent.OnShortBreakClicked -> {
                emitEffect(ContactCentralUiEffect.TriggerShortBreakAction)
            }

            ContactCentralUiEvent.OnVoiceCallClicked -> {
                // Dialog is only used when voiceValue == true, otherwise cancel directly
                val voiceValue = currentVoiceValue
                if (voiceValue) {
                    _uiState.update { it.copy(showVoiceRequestDialog = true) }
                    emitEffect(ContactCentralUiEffect.OpenVoiceRequestDialog)
                } else {
                    emitEffect(ContactCentralUiEffect.SendVoiceRequestFalse)
                    emitEffect(ContactCentralUiEffect.NavigateBack)
                }
            }

            ContactCentralUiEvent.OnMessagesClicked -> {
                emitEffect(ContactCentralUiEffect.NavigateToPredefinedMessages)
            }

            ContactCentralUiEvent.OnInformationClicked -> {
                emitEffect(ContactCentralUiEffect.NavigateToInformationMessages)
            }

            ContactCentralUiEvent.OnDialogAccept -> {
                _uiState.update { it.copy(showVoiceRequestDialog = false) }
                emitEffect(ContactCentralUiEffect.SendVoiceRequestTrue)
                emitEffect(ContactCentralUiEffect.NavigateBack)
            }

            ContactCentralUiEvent.OnDialogCancel -> {
                _uiState.update { it.copy(showVoiceRequestDialog = false) }
            }

            is ContactCentralUiEvent.OnShortBreakStatusChanged -> {
                currentShortBreakStatus = event.status
                reduceButtons()
            }

            is ContactCentralUiEvent.OnBravoVoiceChanged -> {
                currentVoiceValue = event.voiceValue
                reduceButtons()
            }

            is ContactCentralUiEvent.OnLocationTypeChanged -> {
                currentLocationType = event.locationType
                reduceButtons()
            }

            is ContactCentralUiEvent.OnHasPredefinedMessagesChanged -> {
                hasPredefinedMessages = event.hasMessages
                reduceButtons()
            }

            is ContactCentralUiEvent.OnHasShortBreakChanged -> {
                hasShortBreak = event.hasShortBreak
                reduceButtons()
            }

            is ContactCentralUiEvent.OnIsITopChanged -> {
                isITopTaximeter = event.isITop
                reduceButtons()
            }
        }
    }

    fun checkITopTaximeter() {
        viewModelScope.launch {
            val isITop = bluetoothLocalUseCase.isCurrentBluetoothItop()
            Logs.d(TAG, "isITopTaximeter: $isITop")
            onEvent(ContactCentralUiEvent.OnIsITopChanged(isITop))
        }
    }

    fun sendVoiceRequest(value: Boolean?) {
        voicePetitionUseCase.voiceRequest(value ?: true)
    }

    fun isHired(currentStatus: Int): Boolean = shiftStatusUseCase.isHired(currentStatus)

    fun navigateToPredefinedMessages() {
        emitEffect(ContactCentralUiEffect.NavigateToPredefinedMessages)
    }

    fun navigateToInformationMessages() {
        emitEffect(ContactCentralUiEffect.NavigateToInformationMessages)
    }

    private var currentShortBreakStatus: ShortBreakStatus? = null
    private var currentVoiceValue: Boolean = true
    private var currentLocationType: String? = null

    private fun reduceButtons() {
        val hasLocation = !currentLocationType.isNullOrBlank()
        val shortBreakStatus = currentShortBreakStatus

        val shortBreakButton = when {
            !hasShortBreak -> ButtonUiState(
                visible = false,
                enabled = false,
                textRes = R.string.btn_short_break,
                background = ButtonBackground.GREEN,
                style = ButtonStyle.DISABLE
            )

            hasLocation -> when (shortBreakStatus) {
                ShortBreakStatus.IN_SHORT_BREAK,
                ShortBreakStatus.IN_SHORT_BREAK_FORCED -> ButtonUiState(
                    visible = true,
                    enabled = true,
                    textRes = R.string.btn_end_short_break,
                    background = ButtonBackground.RED,
                    style = ButtonStyle.ENABLE
                )

                ShortBreakStatus.CAN_START_SHORT_BREAK -> ButtonUiState(
                    visible = true,
                    enabled = true,
                    textRes = R.string.btn_short_break,
                    background = ButtonBackground.GREEN,
                    style = ButtonStyle.ENABLE
                )

                ShortBreakStatus.SHORT_BREAK_DISABLED -> ButtonUiState(
                    visible = true,
                    enabled = false,
                    textRes = R.string.btn_short_break,
                    background = ButtonBackground.GREEN,
                    style = ButtonStyle.DISABLE
                )

                null -> ButtonUiState(
                    visible = true,
                    enabled = false,
                    textRes = R.string.btn_short_break,
                    background = ButtonBackground.GREEN,
                    style = ButtonStyle.DISABLE
                )
            }

            else -> when (shortBreakStatus) {
                ShortBreakStatus.IN_SHORT_BREAK,
                ShortBreakStatus.IN_SHORT_BREAK_FORCED -> ButtonUiState(
                    visible = true,
                    enabled = true,
                    textRes = R.string.btn_end_short_break,
                    background = ButtonBackground.RED,
                    style = ButtonStyle.ENABLE
                )

                else -> ButtonUiState(
                    visible = true,
                    enabled = false,
                    textRes = R.string.btn_short_break,
                    background = ButtonBackground.GREEN,
                    style = ButtonStyle.DISABLE
                )
            }
        }

        val cancelVisible = !(!isHiredStatus() && hasShortBreak)
        val messagesEnabled = hasPredefinedMessages
        val voiceButton = if (currentVoiceValue) {
            ButtonUiState(
                visible = true,
                enabled = !isITopTaximeter,
                textRes = R.string.btn_voice_request,
                background = ButtonBackground.DEFAULT,
                style = if (isITopTaximeter) ButtonStyle.DISABLE else ButtonStyle.ENABLE
            )
        } else {
            ButtonUiState(
                visible = true,
                enabled = true,
                textRes = R.string.btn_cancel_voice_request,
                background = ButtonBackground.RED,
                style = ButtonStyle.ENABLE
            )
        }

        _uiState.update {
            it.copy(
                buttons = ContactCentralButtonsState(
                    cancel = it.buttons.cancel.copy(visible = cancelVisible),
                    shortBreak = shortBreakButton,
                    voiceCall = voiceButton,
                    messages = it.buttons.messages.copy(
                        enabled = messagesEnabled,
                        style = if (messagesEnabled) ButtonStyle.ENABLE else ButtonStyle.DISABLE
                    ),
                    information = it.buttons.information
                )
            )
        }
    }

    private fun isHiredStatus(): Boolean {
        val status = null
        return status?.let { isHired(it) } ?: false
    }

    private fun emitEffect(effect: ContactCentralUiEffect) {
        viewModelScope.launch { _uiEffect.emit(effect) }
    }
}
```

## 3) Compose screen

This composable collects state and effects using lifecycle-aware collection, and preserves the original navigation behavior.

```kotlin
package ifac.td.taxi.ui.screen.compose

import androidx.compose.foundation.layout.*
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Modifier
import androidx.compose.ui.platform.LocalContext
import androidx.compose.ui.res.stringResource
import androidx.compose.ui.unit.dp
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.navigation.NavController
import ifac.td.taxi.R
import ifac.td.taxi.repository.connections.service.model.ShortBreakStatus
import ifac.td.taxi.viewmodel.compose.ContactCentralComposeViewModel
import kotlinx.coroutines.flow.collectLatest

@Composable
fun ContactCentralScreen(
    navController: NavController,
    viewModel: ContactCentralComposeViewModel,
    onShowToast: (Int) -> Unit,
    onShortBreakPressed: () -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                ContactCentralUiEffect.NavigateBack -> navController.popBackStack()
                ContactCentralUiEffect.NavigateToPredefinedMessages ->
                    navController.navigate(R.id.action_contactCentralFragment_to_predefinedMessageFragment)

                ContactCentralUiEffect.NavigateToInformationMessages ->
                    navController.navigate(R.id.action_contactCentralFragment_to_informationMessageFragment)

                ContactCentralUiEffect.ShowShortBreakForcedToast ->
                    onShowToast(R.string.short_break_forced)

                ContactCentralUiEffect.OpenVoiceRequestDialog -> Unit

                ContactCentralUiEffect.SendVoiceRequestTrue -> viewModel.sendVoiceRequest(true)
                ContactCentralUiEffect.SendVoiceRequestFalse -> viewModel.sendVoiceRequest(false)
                ContactCentralUiEffect.TriggerShortBreakAction -> onShortBreakPressed()
            }
        }
    }

    if (uiState.showVoiceRequestDialog) {
        AlertDialog(
            onDismissRequest = { viewModel.onEvent(ContactCentralUiEvent.OnDialogCancel) },
            title = { Text(text = stringResource(R.string.btn_voice_request)) },
            text = { Text(text = stringResource(R.string.btn_voice_request)) },
            confirmButton = {
                TextButton(onClick = { viewModel.onEvent(ContactCentralUiEvent.OnDialogAccept) }) {
                    Text(text = stringResource(android.R.string.ok))
                }
            },
            dismissButton = {
                TextButton(onClick = { viewModel.onEvent(ContactCentralUiEvent.OnDialogCancel) }) {
                    Text(text = stringResource(android.R.string.cancel))
                }
            }
        )
    }

    Column(modifier = Modifier.fillMaxSize().padding(16.dp)) {
        ContactCentralButton(
            state = uiState.buttons.cancel,
            text = stringResource(R.string.cancel),
            onClick = { viewModel.onEvent(ContactCentralUiEvent.OnCancelClicked) }
        )

        Spacer(Modifier.height(12.dp))

        if (uiState.buttons.shortBreak.visible) {
            ContactCentralButton(
                state = uiState.buttons.shortBreak,
                text = stringResource(uiState.buttons.shortBreak.textRes).uppercase(),
                onClick = {
                    if (uiState.buttons.shortBreak.enabled) {
                        viewModel.onEvent(ContactCentralUiEvent.OnShortBreakClicked)
                    }
                }
            )
        }

        Spacer(Modifier.height(12.dp))

        ContactCentralButton(
            state = uiState.buttons.voiceCall,
            text = stringResource(uiState.buttons.voiceCall.textRes).uppercase(),
            onClick = { viewModel.onEvent(ContactCentralUiEvent.OnVoiceCallClicked) }
        )

        Spacer(Modifier.height(12.dp))

        ContactCentralButton(
            state = uiState.buttons.messages,
            text = stringResource(R.string.btn_messages).uppercase(),
            onClick = { viewModel.onEvent(ContactCentralUiEvent.OnMessagesClicked) }
        )

        Spacer(Modifier.height(12.dp))

        ContactCentralButton(
            state = uiState.buttons.information,
            text = stringResource(R.string.btn_information).uppercase(),
            onClick = { viewModel.onEvent(ContactCentralUiEvent.OnInformationClicked) }
        )
    }
}
```

## 4) Compose button that mimics the custom button behavior

This is a simple Compose replacement for the custom button styling.

```kotlin
package ifac.td.taxi.ui.screen.compose

import androidx.compose.foundation.background
import androidx.compose.foundation.clickable
import androidx.compose.foundation.layout.*
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.text.font.FontWeight
import androidx.compose.ui.unit.dp

@Composable
fun ContactCentralButton(
    state: ButtonUiState,
    text: String,
    onClick: () -> Unit,
) {
    if (!state.visible) return

    val backgroundColor = when (state.background) {
        ButtonBackground.DEFAULT -> Color(0xFF2D2D2D)
        ButtonBackground.RED -> Color(0xFFD32F2F)
        ButtonBackground.GREEN -> Color(0xFF388E3C)
    }

    val alpha = if (state.enabled) 1f else 0.4f

    Box(
        modifier = Modifier
            .fillMaxWidth()
            .height(52.dp)
            .background(backgroundColor.copy(alpha = alpha))
            .clickable(enabled = state.enabled) { onClick() },
        contentAlignment = Alignment.Center
    ) {
        Text(
            text = text,
            color = Color.White,
            fontWeight = FontWeight.Bold
        )
    }
}
```

## Reducer

```kotlin
package ifac.td.taxi.ui.screen.components

import ifac.td.taxi.R
import ifac.td.taxi.repository.connections.service.model.ShortBreakStatus

object ContactCentralButtonsReducer {

    fun reduce(
        previous: ContactCentralButtonsState,
        input: ContactCentralReducerInput,
        isHired: (Int) -> Boolean,
    ): ContactCentralButtonsState {
        val hasLocation = !input.zone.isNullOrBlank()

        // 1) shortBreakStatus.collect { ... tryUpdateShortBreakButton(shortBreakStatus, !zone.isNullOrBlank()) }
        val shortBreakButton = reduceShortBreakButton(
            shortBreakStatus = input.shortBreakStatus,
            hasLocation = hasLocation
        )

        // 2) bravoStateFlow.collect { value -> voice button background/text }
        val voiceButton = if (input.voiceValue) {
            ButtonUiState(
                visible = true,
                enabled = !input.isITopTaximeter,
                textRes = R.string.btn_voice_request,
                background = ButtonBackground.DEFAULT,
                style = if (input.isITopTaximeter) ButtonStyle.DISABLE else ButtonStyle.ENABLE
            )
        } else {
            ButtonUiState(
                visible = true,
                enabled = true,
                textRes = R.string.btn_cancel_voice_request,
                background = ButtonBackground.RED,
                style = ButtonStyle.ENABLE
            )
        }

        // 3) hasPredefinedMessagesFlow.collect { if (it) btnMessages.setButtonStyle(ENABLE) }
        val messagesButton = previous.messages.copy(
            enabled = input.hasPredefinedMessages,
            style = if (input.hasPredefinedMessages) ButtonStyle.ENABLE else ButtonStyle.DISABLE
        )

        // 4) hasShortBreak.collect { shiftStatusFlow.value?.currentStatus?.let { ... } }
        val cancelVisible = if (input.currentShiftStatus != null) {
            val hired = isHired(input.currentShiftStatus)
            !( !hired && input.hasShortBreak )
        } else {
            previous.cancel.visible
        }

        val shortBreakVisible = if (input.currentShiftStatus != null) {
            val hired = isHired(input.currentShiftStatus)
            !(!hired && input.hasShortBreak)
        } else {
            previous.shortBreak.visible
        }

        val cancelButton = previous.cancel.copy(visible = cancelVisible)
        val shortBreakButtonWithVisibility = shortBreakButton.copy(visible = shortBreakVisible)

        // 5) isITopTaximeter.collect { if (it) btnVoiceCall.setButtonStyle(DISABLE) }
        // In fragment this only disabled the voice button, it did not change text/background.
        val finalVoiceButton = if (input.isITopTaximeter && input.voiceValue) {
            voiceButton.copy(
                enabled = false,
                style = ButtonStyle.DISABLE
            )
        } else {
            voiceButton
        }

        return previous.copy(
            cancel = cancelButton,
            shortBreak = shortBreakButtonWithVisibility,
            voiceCall = finalVoiceButton,
            messages = messagesButton,
            information = previous.information
        )
    }

    private fun reduceShortBreakButton(
        shortBreakStatus: ShortBreakStatus?,
        hasLocation: Boolean
    ): ButtonUiState {
        return if (hasLocation) {
            when (shortBreakStatus) {
                ShortBreakStatus.IN_SHORT_BREAK,
                ShortBreakStatus.IN_SHORT_BREAK_FORCED -> ButtonUiState(
                    visible = true,
                    enabled = true,
                    textRes = R.string.btn_end_short_break,
                    background = ButtonBackground.RED,
                    style = ButtonStyle.ENABLE
                )

                ShortBreakStatus.CAN_START_SHORT_BREAK -> ButtonUiState(
                    visible = true,
                    enabled = true,
                    textRes = R.string.btn_short_break,
                    background = ButtonBackground.GREEN,
                    style = ButtonStyle.ENABLE
                )

                ShortBreakStatus.SHORT_BREAK_DISABLED -> ButtonUiState(
                    visible = true,
                    enabled = false,
                    textRes = R.string.btn_short_break,
                    background = ButtonBackground.GREEN,
                    style = ButtonStyle.DISABLE
                )

                null -> ButtonUiState(
                    visible = true,
                    enabled = false,
                    textRes = R.string.btn_short_break,
                    background = ButtonBackground.GREEN,
                    style = ButtonStyle.DISABLE
                )
            }
        } else {
            if (shortBreakStatus != ShortBreakStatus.IN_SHORT_BREAK &&
                shortBreakStatus != ShortBreakStatus.IN_SHORT_BREAK_FORCED
            ) {
                ButtonUiState(
                    visible = true,
                    enabled = false,
                    textRes = R.string.btn_short_break,
                    background = ButtonBackground.GREEN,
                    style = ButtonStyle.DISABLE
                )
            } else {
                ButtonUiState(
                    visible = true,
                    enabled = true,
                    textRes = R.string.btn_end_short_break,
                    background = ButtonBackground.RED,
                    style = ButtonStyle.ENABLE
                )
            }
        }
    }
}
```

```kotlin
                    )
                )
            }
        }

        } else {
            voicePetitionUseCase.voiceRequest(false)
            emitEffect(ContactCentralUiEffect.NavigateBack)
        }

        _uiState.update { it.copy(showVoiceRequestDialog = false) }
        voicePetitionUseCase.voiceRequest(true)
        emitEffect(ContactCentralUiEffect.NavigateBack)
    }

        _uiState.update { it.copy(showVoiceRequestDialog = false) }
    }

        emitEffect(ContactCentralUiEffect.NavigateToPredefinedMessages)
    }

    fun onInformationClick() {
        emitEffect(ContactCentralUiEffect.NavigateToInformationMessages)
    }

    fun reduceFromSharedState(
        shared: SharedContactCentralState

        val input = ContactCentralReducerInput(
            shortBreakStatus = shared.shortBreakStatus,
            voiceValue = shared.voiceValue,
            zone = shared.zone,
            hasPredefinedMessages = shared.hasPredefinedMessages,
            hasShortBreak = shared.hasShortBreak,

                    previous = current.buttons,
                    input = input,
                    isHired = { status -> shiftStatusUseCase.isHired(status) }
                )
            )
        }
    }

        viewModelScope.launch { _uiEffect.emit(effect) }
```

# 10) Important note about exact fragment parity

There are two fragment behaviors that are worth calling out:

## A. `hasShortBreak.collect { ... }` depends on `shiftStatusFlow.value?.currentStatus`
In the fragment, the visibility of cancel vs short-break is only updated when `hasShortBreak` emits, but it reads `shiftStatusFlow.value` at that moment.

In Compose, the reducer above mirrors that by taking both values together in `SharedContactCentralState` and reducing from a single snapshot.

## B. `isITopTaximeter.collect { ... }` only disables the voice button
The fragment does **not** change the voice button text/background there.  
The reducer preserves that by only forcing `enabled = false` and `style = DISABLE` when `isITopTaximeter == true`.

```kotlin
        return if (hasLocation) {
            when (shortBreakStatus) {
                ShortBreakStatus.IN_SHORT_BREAK,
                ShortBreakStatus.IN_SHORT_BREAK_FORCED -> ButtonUiState(
                    visible = true,
                    enabled = true,
                    textRes = R.string.btn_end_short_break,
                    background = ButtonBackground.RED,
                    style = ButtonStyle.ENABLE
                )

                ShortBreakStatus.CAN_START_SHORT_BREAK -> ButtonUiState(
                    visible = true,
                    enabled = true,
                    textRes = R.string.btn_short_break,
                    background = ButtonBackground.GREEN,
                    style = ButtonStyle.ENABLE
                )

                ShortBreakStatus.SHORT_BREAK_DISABLED -> ButtonUiState(
                    visible = true,
                    enabled = false,
                    textRes = R.string.btn_short_break,
                    background = ButtonBackground.GREEN,
                    style = ButtonStyle.DISABLE
                )

                null -> ButtonUiState(
                    visible = true,
                    enabled = false,
                    textRes = R.string.btn_short_break,
                    background = ButtonBackground.GREEN,
                    style = ButtonStyle.DISABLE
                )
            }
        } else {
            if (shortBreakStatus != ShortBreakStatus.IN_SHORT_BREAK &&
                shortBreakStatus != ShortBreakStatus.IN_SHORT_BREAK_FORCED
            ) {
                ButtonUiState(
                    visible = true,
                    enabled = false,
                    textRes = R.string.btn_short_break,
                    background = ButtonBackground.GREEN,
                    style = ButtonStyle.DISABLE
                )
            } else {
                ButtonUiState(
                    visible = true,
                    enabled = true,
                    textRes = R.string.btn_end_short_break,
                    background = ButtonBackground.RED,
                    style = ButtonStyle.ENABLE
                )
            }
        }
    }
}

            _uiState.update {
                it.copy(
                    buttons = it.buttons.copy(
                        voiceCall = it.buttons.voiceCall.copy(

                            style = if (isITop) ButtonStyle.DISABLE else it.buttons.voiceCall.style
                        )
                    )
                )
            }
        }
    }

        emitEffect(ContactCentralUiEffect.NavigateBack)
    }

            emitEffect(ContactCentralUiEffect.ShowShortBreakForcedToast)
            return
        }

        _uiState.update {
            it.copy(
                buttons = it.buttons.copy(
                    shortBreak = it.buttons.shortBreak.copy(style = ButtonStyle.LOADING)
                )
            )
        }
        emitEffect(ContactCentralUiEffect.TriggerShortBreakAction)
    }

    fun onVoiceCallClick(voiceValue: Boolean) {
        if (voiceValue) {
            _uiState.update { it.copy(showVoiceRequestDialog = true) }
            emitEffect(ContactCentralUiEffect.ShowVoiceRequestDialog(R.string.btn_voice_request))
        } else {
            voicePetitionUseCase.voiceRequest(false)
            emitEffect(ContactCentralUiEffect.NavigateBack)
        }
    }

    fun onVoiceRequestDialogAccept() {
        _uiState.update { it.copy(showVoiceRequestDialog = false) }
        voicePetitionUseCase.voiceRequest(true)
        emitEffect(ContactCentralUiEffect.NavigateBack)
    }

    fun onVoiceRequestDialogDismiss() {
        _uiState.update { it.copy(showVoiceRequestDialog = false) }
    }

    fun onMessagesClick() {
        emitEffect(ContactCentralUiEffect.NavigateToPredefinedMessages)
    }

    fun onInformationClick() {
        emitEffect(ContactCentralUiEffect.NavigateToInformationMessages)
    }

    fun reduceFromSharedState(shared: SharedContactCentralState) {
        val input = ContactCentralReducerInput(
            shortBreakStatus = shared.shortBreakStatus,
            voiceValue = shared.voiceValue,
            zone = shared.zone,
            hasPredefinedMessages = shared.hasPredefinedMessages,
            hasShortBreak = shared.hasShortBreak,
            currentShiftStatus = shared.currentShiftStatus,
            isITopTaximeter = shared.isITopTaximeter
        )

        _uiState.update { current ->
            current.copy(
                buttons = ContactCentralButtonsReducer.reduce(
                    previous = current.buttons,
                    input = input,
                    isHired = { status -> shiftStatusUseCase.isHired(status) }
                )
            )
        }
    }

    private fun emitEffect(effect: ContactCentralUiEffect) {
        viewModelScope.launch { _uiEffect.emit(effect) }
    }
}

@Composable
fun ContactCentralScreen(
    navController: NavController,
    viewModel: ContactCentralComposeViewModel,
    sharedUiState: SharedContactCentralState,
    onShortBreakButtonPressed: () -> Unit,
    onShowHeader: (Boolean) -> Unit,
    onShowToast: (Int) -> Unit,
) {

    val context = LocalContext.current

    LaunchedEffect(Unit) {
        onShowHeader(true)
        viewModel.initialize()
    }

    LaunchedEffect(sharedUiState) {
        viewModel.reduceFromSharedState(sharedUiState)
    }

    val dialogState = remember { mutableStateOf<ContactCentralDialogState?>(null) }

    LaunchedEffect(viewModel) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                ContactCentralUiEffect.NavigateBack -> navController.popBackStack()

                ContactCentralUiEffect.NavigateToPredefinedMessages ->
                    navController.navigate(R.id.action_contactCentralFragment_to_predefinedMessageFragment)

                ContactCentralUiEffect.NavigateToInformationMessages ->
                    navController.navigate(R.id.action_contactCentralFragment_to_informationMessageFragment)

                ContactCentralUiEffect.ShowShortBreakForcedToast ->
                    onShowToast(R.string.short_break_forced)

                is ContactCentralUiEffect.ShowVoiceRequestDialog ->
                    dialogState.value = ContactCentralDialogState(title = effect.title)

                ContactCentralUiEffect.TriggerShortBreakAction -> {
                    // handled by onShortBreakButtonPressed in the click lambda
                }
            }
        }
    }

    ContactCentralDialogHost(
        dialogState = dialogState.value,
        onDismiss = {
            dialogState.value = null
            viewModel.onVoiceRequestDialogDismiss()
        },
        onAccept = {
            dialogState.value = null
            viewModel.onVoiceRequestDialogAccept()
        }
    )

    ContactCentralContent(
        state = uiState,
        sharedUiState = sharedUiState,
        onCancel = { viewModel.onCancelClick() },
        onShortBreak = {
            val status = sharedUiState.shortBreakStatus
            if (status == ShortBreakStatus.IN_SHORT_BREAK_FORCED) {
                onShowToast(R.string.short_break_forced)
            } else {
                viewModel.onShortBreakClick(
                    status = status ?: ShortBreakStatus.SHORT_BREAK_DISABLED,
                    hasLocation = !sharedUiState.zone.isNullOrBlank()
                )
                onShortBreakButtonPressed()
            }
        },
        onVoiceCall = { viewModel.onVoiceCallClick(sharedUiState.voiceValue) },
        onMessages = { viewModel.onMessagesClick() },
        onInformation = { viewModel.onInformationClick() }
    )
}
```

```kotlin
object ContactCentralButtonsReducer {

    fun reduce(
        previous: ContactCentralButtonsState,
        input: ContactCentralReducerInput,
        isHired: (Int) -> Boolean,
    ): ContactCentralButtonsState {
        val hasLocation = !input.zone.isNullOrBlank()

        val shortBreakButton = reduceShortBreakButton(
            shortBreakStatus = input.shortBreakStatus,
            hasLocation = hasLocation
        )

        val voiceButton = if (input.voiceValue) {
            ButtonUiState(
                visible = true,
                enabled = !input.isITopTaximeter,
                textRes = R.string.btn_voice_request,
                background = ButtonBackground.DEFAULT,
                style = if (input.isITopTaximeter) ButtonStyle.DISABLE else ButtonStyle.ENABLE
            )
        } else {
            ButtonUiState(
                visible = true,
                enabled = true,
                textRes = R.string.btn_cancel_voice_request,
                background = ButtonBackground.RED,
                style = ButtonStyle.ENABLE
            )
        }

        val messagesButton = previous.messages.copy(
            enabled = input.hasPredefinedMessages,
            style = if (input.hasPredefinedMessages) ButtonStyle.ENABLE else ButtonStyle.DISABLE
        )

        val cancelVisible = if (input.currentShiftStatus != null) {
            val hired = isHired(input.currentShiftStatus)
            !(!hired && input.hasShortBreak)
        } else {
            previous.cancel.visible
        }

        val shortBreakVisible = if (input.currentShiftStatus != null) {
            val hired = isHired(input.currentShiftStatus)
            !(!hired && input.hasShortBreak)
        } else {
            previous.shortBreak.visible
        }

        val finalVoiceButton = if (input.isITopTaximeter && input.voiceValue) {
            voiceButton.copy(enabled = false, style = ButtonStyle.DISABLE)
        } else {
            voiceButton
        }

        return previous.copy(
            cancel = previous.cancel.copy(visible = cancelVisible),
            shortBreak = shortBreakButton.copy(visible = shortBreakVisible),
            voiceCall = finalVoiceButton,
            messages = messagesButton,
            information = previous.information
        )
    }

    private fun reduceShortBreakButton(
        shortBreakStatus: ShortBreakStatus?,
        hasLocation: Boolean
    ): ButtonUiState {
        return if (hasLocation) {
            when (shortBreakStatus) {
                ShortBreakStatus.IN_SHORT_BREAK,
                ShortBreakStatus.IN_SHORT_BREAK_FORCED -> ButtonUiState(
                    visible = true,
                    enabled = true,
                    textRes = R.string.btn_end_short_break,
                    background = ButtonBackground.RED,
                    style = ButtonStyle.ENABLE
                )

                ShortBreakStatus.CAN_START_SHORT_BREAK -> ButtonUiState(
                    visible = true,
                    enabled = true,
                    textRes = R.string.btn_short_break,
                    background = ButtonBackground.GREEN,
                    style = ButtonStyle.ENABLE
                )

                ShortBreakStatus.SHORT_BREAK_DISABLED -> ButtonUiState(
                    visible = true,
                    enabled = false,
                    textRes = R.string.btn_short_break,
                    background = ButtonBackground.GREEN,
                    style = ButtonStyle.DISABLE
                )

                null -> ButtonUiState(
                    visible = true,
                    enabled = false,
                    textRes = R.string.btn_short_break,
                    background = ButtonBackground.GREEN,
                    style = ButtonStyle.DISABLE
                )
            }
        } else {
            if (shortBreakStatus != ShortBreakStatus.IN_SHORT_BREAK &&
                shortBreakStatus != ShortBreakStatus.IN_SHORT_BREAK_FORCED
            ) {
                ButtonUiState(
                    visible = true,
                    enabled = false,
                    textRes = R.string.btn_short_break,
                    background = ButtonBackground.GREEN,
                    style = ButtonStyle.DISABLE
                )
            } else {
                ButtonUiState(
                    visible = true,
                    enabled = true,
                    textRes = R.string.btn_end_short_break,
                    background = ButtonBackground.RED,
                    style = ButtonStyle.ENABLE
                )
            }
        }
    }
}

    }

    fun onVoiceCallClick(voiceValue: Boolean) {
        if (voiceValue) {
            _uiState.update { it.copy(showVoiceRequestDialog = true) }
            emitEffect(ContactCentralUiEffect.ShowVoiceRequestDialog(R.string.btn_voice_request))
        } else {
            voicePetitionUseCase.voiceRequest(false)
            emitEffect(ContactCentralUiEffect.NavigateBack)
        }
    }

    fun onVoiceRequestDialogAccept() {
        _uiState.update { it.copy(showVoiceRequestDialog = false) }
        voicePetitionUseCase.voiceRequest(true)
        emitEffect(ContactCentralUiEffect.NavigateBack)
    }

    fun onVoiceRequestDialogDismiss() {
        _uiState.update { it.copy(showVoiceRequestDialog = false) }
    }

    fun onMessagesClick() {
        emitEffect(
            ContactCentralUiEffect.NavigateToPredefinedMessages(
                ContactCentralFragmentDirections.actionContactCentralFragmentToPredefinedMessageFragment(
                    -1,
                    true
                )
            )
        )
    }

    fun onInformationClick() {
        emitEffect(
            ContactCentralUiEffect.NavigateToInformationMessages(
                ContactCentralFragmentDirections.actionContactCentralFragmentToInformationMessageFragment()
            )
        )
    }

    fun reduceFromSharedState(shared: SharedContactCentralState) {
        val input = ContactCentralReducerInput(
            shortBreakStatus = shared.shortBreakStatus,
            voiceValue = shared.voiceValue,
            zone = shared.zone,
            hasPredefinedMessages = shared.hasPredefinedMessages,
            hasShortBreak = shared.hasShortBreak,
            currentShiftStatus = shared.currentShiftStatus,
            isITopTaximeter = shared.isITopTaximeter
        )

        _uiState.update { current ->
            current.copy(
                buttons = ContactCentralButtonsReducer.reduce(
                    previous = current.buttons,
                    input = input,
                    isHired = { status -> shiftStatusUseCase.isHired(status) }
                )
            )
        }
    }

    private fun emitEffect(effect: ContactCentralUiEffect) {
        viewModelScope.launch { _uiEffect.emit(effect) }
    }
}

@Composable
fun ContactCentralScreen(
    navController: NavController,
    viewModel: ContactCentralComposeViewModel,
    sharedUiState: SharedContactCentralState,
    onShortBreakButtonPressed: () -> Unit,
    onShowHeader: (Boolean) -> Unit,
    onShowToast: (Int) -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    LaunchedEffect(Unit) {
        onShowHeader(true)
        viewModel.initialize()
    }

    LaunchedEffect(sharedUiState) {
        viewModel.reduceFromSharedState(sharedUiState)
    }

    val dialogState = remember { mutableStateOf<ContactCentralDialogState?>(null) }

    LaunchedEffect(viewModel) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                ContactCentralUiEffect.NavigateBack -> navController.popBackStack()

                is ContactCentralUiEffect.NavigateToPredefinedMessages ->
                    navController.navigate(effect.directions)

                is ContactCentralUiEffect.NavigateToInformationMessages ->
                    navController.navigate(effect.directions)

                ContactCentralUiEffect.ShowShortBreakForcedToast ->
                    onShowToast(R.string.short_break_forced)

                is ContactCentralUiEffect.ShowVoiceRequestDialog ->
                    dialogState.value = ContactCentralDialogState(title = effect.title)

                ContactCentralUiEffect.TriggerShortBreakAction -> Unit
            }
        }
    }

    ContactCentralDialogHost(
        dialogState = dialogState.value,
        onDismiss = {
            dialogState.value = null
            viewModel.onVoiceRequestDialogDismiss()
        },
        onAccept = {
            dialogState.value = null
            viewModel.onVoiceRequestDialogAccept()
        }
    )

    ContactCentralContent(
        state = uiState,
        sharedUiState = sharedUiState,
        onCancel = { viewModel.onCancelClick() },
        onShortBreak = {
            val status = sharedUiState.shortBreakStatus
            if (status == ShortBreakStatus.IN_SHORT_BREAK_FORCED) {
                onShowToast(R.string.short_break_forced)
            } else {
                viewModel.onShortBreakClick(
                    status = status ?: ShortBreakStatus.SHORT_BREAK_DISABLED,
                    hasLocation = !sharedUiState.zone.isNullOrBlank()
                )
                onShortBreakButtonPressed()
            }
        },
        onVoiceCall = { viewModel.onVoiceCallClick(sharedUiState.voiceValue) },
        onMessages = { viewModel.onMessagesClick() },
        onInformation = { viewModel.onInformationClick() }
    )
}

@Composable
fun ContactCentralButton(
    style: ButtonUiState,
    onClick: () -> Unit,
) {
    if (!style.visible) return

    val backgroundColor = when (style.background) {
        ButtonBackground.DEFAULT -> Color(0xFF2D2D2D)
        ButtonBackground.RED -> Color(0xFFD32F2F)
        ButtonBackground.GREEN -> Color(0xFF388E3C)
    }

    val alpha = when (style.style) {
        ButtonStyle.DISABLE -> 0.4f
        ButtonStyle.LOADING -> 0.7f
        ButtonStyle.ENABLE -> 1f
    }

    Box(
        modifier = Modifier
            .fillMaxWidth()
            .height(52.dp)
            .background(backgroundColor.copy(alpha = alpha))
            .clickable(enabled = style.enabled && style.style != ButtonStyle.LOADING) { onClick() },
        contentAlignment = Alignment.Center
    ) {
        Text(
            text = when (style.style) {
                ButtonStyle.LOADING -> "..."
                else -> androidx.compose.ui.res.stringResource(style.textRes).uppercase()
            },
            color = Color.White,
            fontWeight = FontWeight.Bold
        )
    }
}

@Composable
fun ContactCentralDialogHost(
    dialogState: ContactCentralDialogState?,
    onDismiss: () -> Unit,
    onAccept: () -> Unit,
) {
    if (dialogState != null) {
        AlertDialog(
            onDismissRequest = onDismiss,
            title = { Text(text = androidx.compose.ui.res.stringResource(dialogState.title)) },
            text = { Text(text = androidx.compose.ui.res.stringResource(dialogState.title)) },
            confirmButton = {
                TextButton(onClick = onAccept) {
                    Text(text = androidx.compose.ui.res.stringResource(android.R.string.ok))
                }
            },
            dismissButton = {
                TextButton(onClick = onDismiss) {
                    Text(text = androidx.compose.ui.res.stringResource(android.R.string.cancel))
                }
            }
        )
    }
}
```


// FIRST ANSWER
Below is a Compose-first rewrite that preserves the **same button behavior, navigation, and dialog flow** from your `HomeFragment` / `HomeViewModel`, but replaces the fragment button logic with a single **`HomeUiState` + `HomeUiEffect`** architecture.

I’ll give you:

1. **Compose UI** for the Home screen  
2. **Compose ViewModel** with `UiState + UiEffect` and `SharedFlow<HomeUiEffect>`  
3. **Full `HomeButtonsState`** matching the fragment’s button enable/disable, colors, and visibility logic  
4. **Dialog state handling** using Compose state, only for dialog cases that existed in the fragment  
5. **Navigation preservation** using Jetpack Navigation `NavController` / deep links  
6. Notes on how to wire it into your existing app

This composable replaces the fragment and observes the ViewModel state/effects.

```kotlin
@Composable
fun HomeScreen(
    navController: NavController,
    viewModel: HomeViewModelCompose,
    onShowToast: (Int) -> Unit,
    onBeep: (Int) -> Unit,
    onKeepScreenOn: (Boolean) -> Unit,
    onShowHeader: (Boolean) -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val dialogState by viewModel.dialogState.collectAsStateWithLifecycle()

    LaunchedEffect(Unit) {
        onShowHeader(true)
        viewModel.onScreenStarted()
    }

    LaunchedEffect(Unit) {
        viewModel.effects.collect { effect ->
            when (effect) {
                is HomeUiEffect.Navigate -> {
                    when (effect.destination) {
                        HomeUiEffect.Destination.Message -> navController.navigate(R.id.action_homeFragment_to_messageFragment)
                        HomeUiEffect.Destination.ContactCentral -> navController.navigate(R.id.action_homeFragment_to_contactCentralFragment)

                        HomeUiEffect.Destination.PendingTrips -> navController.navigate(R.id.action_homeFragment_to_pendingTripsFragment)
                        HomeUiEffect.Destination.ReceiptHistory -> navController.navigate(R.id.action_homeFragment_to_receiptHistoryFragment)
                        HomeUiEffect.Destination.MacroZoning -> navController.navigate(R.id.action_homeFragment_to_macroZoning)
                        HomeUiEffect.Destination.Welcome -> {
                            val request = NavDeepLinkRequest.Builder
                                .fromUri("android-app://ifac.td.taxi/welcome".toUri())
                                .build()
                            navController.navigate(request)
                        }
                        is HomeUiEffect.Destination.ZoningDeepLink -> {
                            val request = NavDeepLinkRequest.Builder
                                .fromUri("android-app://ifac.td.taxi/zoningFragment/${effect.destination.idMacroZone}".toUri())
                                .build()
                            navController.navigate(request)
                        }
                    }
                }

                is HomeUiEffect.ShowToast -> onShowToast(effect.messageRes)

                is HomeUiEffect.Beep -> onBeep(effect.tone)

                is HomeUiEffect.KeepScreenOn -> onKeepScreenOn(effect.keepOn)

                is HomeUiEffect.UpdateTopBarRightIcon -> {
                    // handled by host if needed
                }

                is HomeUiEffect.OpenDialog -> {
                    viewModel.onDialogRequested(effect.dialogType)
                }
            }
        }
    }

    HomeContent(
        state = uiState,
        onAction = viewModel::onAction
    )

    if (dialogState.visible) {
        when (dialogState.type) {
            HomeDialogType.ActivateLocation -> {
                AlertDialog(
                    onDismissRequest = { viewModel.onDialogDismissed() },
                    title = { Text(text = stringResource(R.string.confirm_activate_location)) },

                            Text(text = stringResource(android.R.string.ok))

                        TextButton(onClick = { viewModel.onDialogDismissed() }) {
                            Text(text = stringResource(android.R.string.cancel))
                        }
                    }
                )
            }

            HomeDialogType.DeactivateLocation -> {
                AlertDialog(
                    onDismissRequest = { viewModel.onDialogDismissed() },
                    title = { Text(text = stringResource(R.string.confirm_deactivate_location)) },
                    confirmButton = {
                        TextButton(onClick = { viewModel.onDialogConfirmed() }) {
                            Text(text = stringResource(android.R.string.ok))
                        }
                    },
                    dismissButton = {
                        TextButton(onClick = { viewModel.onDialogDismissed() }) {
                            Text(text = stringResource(android.R.string.cancel))
                        }
                    }
                )
            }

            HomeDialogType.CloseShift -> {
                AlertDialog(
                    onDismissRequest = { viewModel.onDialogDismissed() },
                    title = { Text(text = stringResource(R.string.confirm_close_shift)) },
                    confirmButton = {
                        TextButton(onClick = { viewModel.onDialogConfirmed() }) {
                            Text(text = stringResource(android.R.string.ok))
                        }
                    },

                        TextButton(onClick = { viewModel.onDialogDismissed() }) {

                    confirmButton = {
                        TextButton(onClick = { viewModel.onDialogConfirmed() }) {
                            Text(text = stringResource(android.R.string.ok))
                        }
                    },
                    dismissButton = {
                        TextButton(onClick = { viewModel.onDialogDismissed() }) {
                            Text(text = stringResource(android.R.string.cancel))

            HomeDialogType.ExitStand -> {

@Composable
private fun HomeButton(
    text: String,
    enabled: Boolean,
    visible: Boolean,
    color: HomeButtonColor,
    onClick: () -> Unit
) {
    if (!visible) return

    val containerColor = when (color) {
        HomeButtonColor.BLUE -> MaterialTheme.colorScheme.primary
        HomeButtonColor.RED -> MaterialTheme.colorScheme.error
        HomeButtonColor.GREEN -> MaterialTheme.colorScheme.tertiary
        HomeButtonColor.ORANGE -> MaterialTheme.colorScheme.secondary
        HomeButtonColor.DISABLED -> MaterialTheme.colorScheme.surfaceVariant
        HomeButtonColor.LOADING -> MaterialTheme.colorScheme.surfaceVariant
    }

    Button(
        onClick = onClick,
        enabled = enabled,
        colors = ButtonDefaults.buttonColors(containerColor = containerColor),
        modifier = Modifier.fillMaxWidth()
    ) {
        Text(text = text)
    }
}
```

# 2) Compose ViewModel with `UiState + UiEffect`

This ViewModel keeps the same business rules, but exposes a single state object and a single `SharedFlow<HomeUiEffect>`.

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import android.media.ToneGenerator
import androidx.core.net.toUri
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import androidx.navigation.NavDeepLinkRequest
import com.interfacom.sdk.taximeter.bravocomm.BravoCentral
import com.interfacom.sdk.taximeter.bravocomm.W2CLocation
import com.interfacom.sdk.taximeter.bravocomm.ifConstants
import com.interfacom.sdk.taximeter.bravocomm.rest.pending_trips.response.PendingTrip
import ifac.td.taxi.HomeDirections
import ifac.td.taxi.R
import ifac.td.taxi.domain.model.PaymentMethod
import ifac.td.taxi.domain.model.Trip
import ifac.td.taxi.domain.usecase.*
import ifac.td.taxi.framework.sdk.bravocentral.usecase.BravoCentralConfigurationUseCase
import ifac.td.taxi.framework.sdk.bravocentral.usecase.BravoCentralUseCase
import ifac.td.taxi.framework.sdk.usecase.LicensingUseCase
import ifac.td.taxi.framework.sdk.usecase.ZoningUseCase
import ifac.td.taxi.repository.connections.service.model.ShortBreakStatus
import ifac.td.taxi.repository.room.entities.countdown.CountdownId.TIMEOUT_BLUETOOTH_DISCONNECT
import ifac.td.taxi.repository.room.entities.countdown.CountdownState
import ifac.td.taxi.repository.room.entities.message.MessageType
import ifac.td.taxi.viewmodel.model.MessageUIEnum
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch

class HomeViewModelCompose(
    private val userPreferencesUseCase: UserPreferencesUseCase,
    private val endShiftUseCase: EndShiftUseCase,
    private val zoningUseCase: ZoningUseCase,
    private val messageUseCase: MessageUseCase,
    private val roofLightUseCase: RoofLightUseCase,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val sendToBravoUseCase: SendToBravoUseCase,
    private val pendingTripsUseCase: PendingTripsUseCase,
    private val bravoCentral: BravoCentralUseCase,
    private val portugalUseCase: PortugalUseCase,
    private val licensingUseCase: LicensingUseCase,
    private val chronometerManagerUseCase: ChronometerManagerUseCase,
    private val countdownManagerUseCase: CountdownManagerUseCase,
    private val tripUseCase: TripUseCase,
    private val bravoCentralConfigurationUseCase: BravoCentralConfigurationUseCase,
    private val bluetoothLocalUseCase: BluetoothLocalUseCase,
    application: Application,
    private val sharedState: HomeSharedState,
) : AndroidViewModel(application) {

    private val _uiState = MutableStateFlow(HomeUiState())
    val uiState: StateFlow<HomeUiState> = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<HomeUiEffect>()
    val effects: SharedFlow<HomeUiEffect> = _effects.asSharedFlow()

    private val _dialogState = MutableStateFlow(HomeDialogState())
    val dialogState: StateFlow<HomeDialogState> = _dialogState.asStateFlow()

    init {
        viewModelScope.launch {
            _uiState.update {
                it.copy(
                    buttons = it.buttons.copy(
                        dashboard = it.buttons.dashboard.copy(visible = bravoCentralConfigurationUseCase.getBravoConfigurationVariable()?.isHomeButton == true),
                        fixedPrice = it.buttons.fixedPrice.copy(
                            visible = !bravoCentralConfigurationUseCase.getBravoConfigurationVariable()?.fixedPricesStreetTripsURL.isNullOrBlank()
                        )
                    )
                )
            }
        }
    }

    fun onScreenStarted() {
        getMessages()
        checkForLocationPermission()
        checkPendingServiceOnForHirePermission()
        checkKeepScreenOn()
        syncFromSharedState()
    }

    private fun syncFromSharedState() {
        viewModelScope.launch {
            combine(
                sharedState.bravoVoiceValue,
                sharedState.roofLight,
                sharedState.locationEnabled,
                sharedState.locationType,
                sharedState.locatedOnStop,
                sharedState.hasTaximeterConnection,
                sharedState.orangeBtnPending,
                sharedState.shortBreakStatus,
                sharedState.zone,
                sharedState.pendingTripsList,
                sharedState.updateMessageUI
            ) { values ->
                values
            }.collect {
                reduceButtons()
            }
        }
    }

    fun onAction(action: HomeAction) {
        when (action) {
            HomeAction.ZoningClicked -> {
                viewModelScope.launch {
                    sharedState.emitManualZoningNavigation(true)
                    navigateToZoning()
                }
            }

            HomeAction.PendingClicked -> canOpenPendingTripsFragment(sharedState.pendingTripsListFlow)

            HomeAction.LocationClicked -> handleLocationClick()

            HomeAction.ReceiptsClicked -> navigateToReceiptHistory()

            HomeAction.MessagesClicked -> navigateToMessage()

            HomeAction.CentralClicked -> navigateToContactCentral()

            HomeAction.DashboardClicked -> viewModelScope.launch {
                _effects.emit(HomeUiEffect.Navigate(HomeUiEffect.Destination.Dashboard))
            }

            HomeAction.FixedPriceClicked -> navigateToEstimateFixedPriceScreen()

            HomeAction.LocateStandClicked -> triggerLocateOnStand()
        }
    }

    private fun reduceButtons() {
        val locationAllowed = W2CLocation.isLocationAllowedByCentral()
        val locationEnabled = sharedState.locationEnabledFlow.value.first
        val locationCanEnable = sharedState.locationEnabledFlow.value.second
        val hasMessages = _uiState.value.buttons.messages
        val roof = sharedState.roofLightFlow.value
        val zone = sharedState.zoneFlow.value
        val shortBreak = sharedState.shortBreakStatus.value
        val pendingAllowed = sharedState.pendingTripsListFlow.value
        val isInSoonZone = W2CLocation.getIsInSoonInZone()

        val messagesState = when (_uiState.value.messageUi) {
            MessageUIEnum.NO_MESSAGES -> HomeButtonState(text = "Messages", enabled = false, color = HomeButtonColor.DISABLED)
            MessageUIEnum.HAS_MESSAGES_BUT_NOT_NEW -> HomeButtonState(text = "Messages", enabled = true, color = HomeButtonColor.BLUE)
            MessageUIEnum.HAS_NEW_MESSAGES -> HomeButtonState(text = "Messages", enabled = true, color = HomeButtonColor.ORANGE)
            null -> _uiState.value.buttons.messages
        }

        val locationState = if (locationAllowed && returnLocationPermission() == true) {
            HomeButtonState(
                text = if (locationEnabled) getString(R.string.btn_location_off) else getString(R.string.btn_location_on),
                enabled = locationCanEnable,
                color = if (locationEnabled) HomeButtonColor.GREEN else HomeButtonColor.RED
            )
        } else {
            HomeButtonState(text = "Location", enabled = false, color = HomeButtonColor.DISABLED)
        }

        val zoningEnabled = (bravoCentralConfigurationUseCase.getBravoConfigurationVariable()?.locationRequestWhenDispatchOff == true) || locationEnabled

        val pendingEnabled = when {
            zone.isNullOrBlank() -> false
            shortBreak == ShortBreakStatus.IN_SHORT_BREAK || shortBreak == ShortBreakStatus.IN_SHORT_BREAK_FORCED -> false
            !locationAllowed -> false
            isInSoonZone -> true
            pendingAllowed == true -> true
            else -> false
        }

        val locateStandState = when (sharedState.locationTypeFlow.value) {
            "Z" -> HomeButtonState(
                text = "Locate stop",
                enabled = sharedState.locatedOnStopFlow.value || W2CLocation.locationStopAllowed(),
                color = HomeButtonColor.RED
            )
            "P" -> HomeButtonState(
                text = "Locate stop exit",
                enabled = !checkStartAutomaticLocation(),
                color = HomeButtonColor.GREEN
            )
            else -> HomeButtonState(text = "Locate", enabled = false, color = HomeButtonColor.DISABLED)
        }

        val roofState = when {
            roof?.second != true -> HomeButtonState(text = "Roof", enabled = false, color = HomeButtonColor.DISABLED)
            roof.first == true -> HomeButtonState(text = "Roof", enabled = true, color = HomeButtonColor.GREEN)
            roof.first == false -> HomeButtonState(text = "Roof", enabled = true, color = HomeButtonColor.RED)
            else -> HomeButtonState(text = "Roof", enabled = false, color = HomeButtonColor.DISABLED)
        }

        _uiState.update {
            it.copy(
                buttons = it.buttons.copy(
                    messages = messagesState,
                    location = locationState,
                    zoning = it.buttons.zoning.copy(enabled = zoningEnabled),
                    pending = it.buttons.pending.copy(enabled = pendingEnabled, color = if (sharedState.orangeBtnPendingFlow.value == true) HomeButtonColor.ORANGE else HomeButtonColor.BLUE),
                    locateStand = locateStandState,
                    central = it.buttons.central.copy(color = if (sharedState.bravoStateFlow.value.voiceValue) HomeButtonColor.BLUE else HomeButtonColor.RED),
                    dashboard = it.buttons.dashboard.copy(visible = it.buttons.dashboard.visible),
                    fixedPrice = it.buttons.fixedPrice.copy(visible = it.buttons.fixedPrice.visible),
                    receipts = it.buttons.receipts.copy(visible = !it.buttons.fixedPrice.visible)
                )
            )
        }
    }

    fun getMessages() {
        viewModelScope.launch(Dispatchers.IO) {
            val messages = messageUseCase.getMessageByDriverId()
            val result = if (messages.isNullOrEmpty()) {
                MessageUIEnum.NO_MESSAGES
            } else {
                val hasBadge = messages.any { !it.isRead && it.messageType == MessageType.MESSAGE }
                if (hasBadge) MessageUIEnum.HAS_NEW_MESSAGES else MessageUIEnum.HAS_MESSAGES_BUT_NOT_NEW
            }
            _uiState.update { it.copy(messageUi = result) }
            reduceButtons()
        }
    }

    fun checkForLocationPermission() {
        viewModelScope.launch {
            val parameters = licensingUseCase.getLicensingParameters()
            _uiState.update { it.copy(showLocationButton = parameters?.isDisableLocation != false) }
            reduceButtons()
        }
    }

    fun checkPendingServiceOnForHirePermission() {
        viewModelScope.launch {
            val parameters = licensingUseCase.getLicensingParameters()
            _uiState.update { it.copy(showPendingButton = parameters?.isVacantPendingServices != false) }
            reduceButtons()
        }
    }

    fun checkKeepScreenOn() {
        viewModelScope.launch {
            val selected = userPreferencesUseCase.getUserPreferences()?.turnOffScreenPosition ?: 0
            _effects.emit(HomeUiEffect.KeepScreenOn(selected == 0 || selected == 1))
        }
    }

    fun navigateToMessage() = viewModelScope.launch {
        _effects.emit(HomeUiEffect.Navigate(HomeUiEffect.Destination.Message))
    }

    fun navigateToContactCentral() = viewModelScope.launch {
        _effects.emit(HomeUiEffect.Navigate(HomeUiEffect.Destination.ContactCentral))
    }

    fun navigateToEstimateFixedPriceScreen() = viewModelScope.launch {
        _effects.emit(HomeUiEffect.Navigate(HomeUiEffect.Destination.FixedPrice))
    }

    fun navigateToReceiptHistory() {
        viewModelScope.launch {
            if (licensingUseCase.getLicensingParameters()?.isFiscalService == false) {
                _effects.emit(HomeUiEffect.Navigate(HomeUiEffect.Destination.ReceiptHistory))
            } else {
                val lastTrip: Trip? = tripUseCase.getLastTrip()
                if (lastTrip != null) {
                    val isSubscriber = lastTrip.paymentMethod == PaymentMethod.SUBSCRIBER.paymentString
                    if (lastTrip.isPendingPortugalService && !isSubscriber) {
                        val atcud = portugalUseCase.getPortugalData()?.atcud
                        if (atcud.isNullOrEmpty() || atcud == "0") {
                            _effects.emit(HomeUiEffect.ShowToast(R.string.toast_no_hay_pendientes))
                        } else {
                            _effects.emit(HomeUiEffect.Navigate(HomeUiEffect.Destination.ReceiptHistory))
                        }
                    } else {
                        _effects.emit(HomeUiEffect.Navigate(HomeUiEffect.Destination.ReceiptHistory))
                    }
                } else {
                    _effects.emit(HomeUiEffect.Navigate(HomeUiEffect.Destination.ReceiptHistory))
                }
            }
        }
    }

    fun navigateToZoning() {
        viewModelScope.launch {
            val defaultToMacrozone = userPreferencesUseCase.getUserPreferences()?.macroZoneQuery
            val zoneFilter = zoningUseCase.getZoneConfiguration().zoneFilter
            val filterNavigatesToZoning = false

            if (defaultToMacrozone == true && !filterNavigatesToZoning) {
                _effects.emit(HomeUiEffect.Navigate(HomeUiEffect.Destination.MacroZoning))
            } else if (W2CLocation.getLastIdZone() > 0 || filterNavigatesToZoning) {
                val idMacroZone = if (W2CLocation.getLastIdMacrozone() != 0) W2CLocation.getLastIdMacrozone() else 1
                _effects.emit(HomeUiEffect.Navigate(HomeUiEffect.Destination.ZoningDeepLink(idMacroZone)))
            } else {
                _effects.emit(HomeUiEffect.Navigate(HomeUiEffect.Destination.MacroZoning))
            }
        }
    }

    fun checkZoningButton(locationEnabled: Boolean) {
        viewModelScope.launch {
            val isEnabled = (bravoCentralConfigurationUseCase.getBravoConfigurationVariable()?.locationRequestWhenDispatchOff == true) || locationEnabled
            _uiState.update { it.copy(buttons = it.buttons.copy(zoning = it.buttons.zoning.copy(enabled = isEnabled))) }
        }
    }

    fun canOpenPendingTripsFragment(pendingTripsListFlow: MutableStateFlow<ArrayList<PendingTrip>?>) {
        viewModelScope.launch {
            userPreferencesUseCase.getUserPreferences()?.let {
                if (it.closePendingTripsIfEmpty) {
                    if (BravoCentral.isPendingTripsAllowed(false)) {
                        pendingTripsUseCase.getPendingTripsZone(pendingTripsListFlow, MutableSharedFlow())
                        _effects.emit(HomeUiEffect.Navigate(HomeUiEffect.Destination.PendingTrips))
                    } else {
                        _effects.emit(HomeUiEffect.ShowToast(R.string.toast_no_hay_pendientes))
                    }
                } else {
                    _effects.emit(HomeUiEffect.Navigate(HomeUiEffect.Destination.PendingTrips))
                }
            } ?: run {
                _effects.emit(HomeUiEffect.Navigate(HomeUiEffect.Destination.PendingTrips))
            }
        }
    }

    private fun handleLocationClick() {
        val enabled = sharedState.locationEnabledFlow.value
        if (enabled.first) {
            if (sharedState.shortBreakStatus.value == ShortBreakStatus.IN_SHORT_BREAK_FORCED) {
                viewModelScope.launch { _effects.emit(HomeUiEffect.ShowToast(R.string.connect_taximeter)) }
                return
            }
            _dialogState.value = HomeDialogState(true, HomeDialogType.DeactivateLocation)
        } else {
            if (W2CLocation.isLocationAllowedByCentral()) {
                _dialogState.value = HomeDialogState(true, HomeDialogType.ActivateLocation)
            }
        }
    }

    fun onDialogRequested(type: HomeDialogType) {
        _dialogState.value = HomeDialogState(true, type)
    }

    fun onDialogDismissed() {
        _dialogState.value = HomeDialogState()
    }

    fun onDialogConfirmed() {
        when (_dialogState.value.type) {
            HomeDialogType.ActivateLocation -> {
                sharedState.updateLocationEnabledFlow(enabled = true, canEnable = true)
                sharedState.updateLocationManuallyDeLocatedFlow(false)
            }
            HomeDialogType.DeactivateLocation -> {
                sharedState.updateLocationEnabledFlow(enabled = false, canEnable = true)
                sharedState.updateLocationManuallyDeLocatedFlow(true)
                viewModelScope.launch { _effects.emit(HomeUiEffect.Beep(ToneGenerator.TONE_CDMA_ONE_MIN_BEEP)) }
            }
            HomeDialogType.CloseShift -> {
                logoff()
                viewModelScope.launch {
                    _effects.emit(HomeUiEffect.Navigate(HomeUiEffect.Destination.Welcome))
                }
            }
            HomeDialogType.HiredManual -> changeStateHiredManual()
            HomeDialogType.ExitStand -> {
                if (W2CLocation.getLocationTypeSent() == 'P') {
                    W2CLocation.sendLocateZone()
                } else {
                    W2CLocation.sendLocateStop()
                }
            }
        }
        _dialogState.value = HomeDialogState()
    }

    fun triggerLocateOnStand() {
        if (W2CLocation.getLocationTypeSent() == 'P') {
            _dialogState.value = HomeDialogState(true, HomeDialogType.ExitStand)
        } else {
            W2CLocation.sendLocateStop()
            W2CLocation.setTryingToRelocateInStand(false)
        }
    }

    fun openManualTripDialog() {
        _dialogState.value = HomeDialogState(true, HomeDialogType.HiredManual)
    }

    fun logoff() {
        viewModelScope.launch(Dispatchers.IO) {
            endShiftUseCase.endShift()
            bravoCentral.logoff()
            shiftStatusUseCase.setStatus(ifConstants.STATE_DISCONNECTED, false)
        }
    }

        viewModelScope.launch(Dispatchers.IO) {
            shiftStatusUseCase.changeStateHiredManual()
        }
    }

    fun isCurrentBluetoothITop(): Boolean = bluetoothLocalUseCase.isCurrentBluetoothItop()

    fun returnLocationPermission(): Boolean? = licensingUseCase.getLicensingParameters()?.isDisableLocation

    fun checkStartAutomaticLocation(): Boolean =
        licensingUseCase.getLicensingParameters()?.automaticStandLocation?.let { it > 0 } ?: false
```

## Exact reducer implementation

```kotlin
package ifac.td.taxi.viewmodel

import ifac.td.taxi.R
import ifac.td.taxi.repository.connections.service.model.ShortBreakStatus
import ifac.td.taxi.viewmodel.model.MessageUIEnum

fun HomeButtonsState.reduce(
    input: HomeReducerInput,
    locationAllowedByCentral: Boolean,
    locationPermissionEnabled: Boolean,
    isInSoonZone: Boolean,
    pendingServicesButton: Boolean?,
    currentLocationType: String,
    locatedOnStop: Boolean,
    canStartAutomaticLocation: Boolean,
    zoneValue: String?,
    shortBreakStatus: String?,
    locationEnabledFlow: Pair<Boolean, Boolean>,
    roofLightAvailable: Boolean,
    roofLightState: Boolean?,
    dashboardVisible: Boolean,
    fixedPriceVisible: Boolean
): HomeButtonsState {
    return when (input) {

        is HomeReducerInput.BravoStateChanged -> copy(
            central = central.copy(
                color = if (input.voiceValue) HomeButtonColor.BLUE else HomeButtonColor.RED
            )
        )

        is HomeReducerInput.RoofLightChanged -> {
            val enabled = input.disableLuminous == true
            val newState = when {
                !enabled -> roofLight.copy(enabled = false, color = HomeButtonColor.DISABLED)
                input.lightState == true -> roofLight.copy(enabled = true, color = HomeButtonColor.GREEN)
                input.lightState == false -> roofLight.copy(enabled = true, color = HomeButtonColor.RED)
                else -> roofLight.copy(enabled = false, color = HomeButtonColor.DISABLED)
            }
            copy(roofLight = newState)
        }

        is HomeReducerInput.LocationEnabledChanged -> {
            val newLocation = if (input.enabled) {
                location.copy(
                    textRes = R.string.btn_location_off,
                    enabled = input.canEnable,
                    color = HomeButtonColor.GREEN
                )
            } else {
                location.copy(
                    textRes = R.string.btn_location_on,
                    enabled = input.canEnable,
                    color = HomeButtonColor.RED
                )
            }
            copy(
                location = newLocation,
                zoning = zoning.copy(enabled = input.enabled || locationPermissionEnabled)
            )
        }

        is HomeReducerInput.MessagesChanged -> {
            val newMessages = when (input.value) {
                MessageUIEnum.NO_MESSAGES -> messages.copy(enabled = false, color = HomeButtonColor.DISABLED)
                MessageUIEnum.HAS_MESSAGES_BUT_NOT_NEW -> messages.copy(enabled = true, color = HomeButtonColor.BLUE)
                MessageUIEnum.HAS_NEW_MESSAGES -> messages.copy(enabled = true, color = HomeButtonColor.ORANGE)
                null -> messages
            }
            copy(messages = newMessages)
        }

        is HomeReducerInput.ShowLocationButtonChanged -> {
            if (input.visible == false) copy(location = location.copy(enabled = false)) else this
        }

        is HomeReducerInput.PendingServicesButtonChanged -> {
            copy(pending = pending.copy(enabled = input.allowed == true))
        }

        is HomeReducerInput.LocatedOnStopChanged -> {
            val newLocateStand = if (input.value) {
                locateStand.copy(
                    enabled = true,
                    color = if (currentLocationType == "P") HomeButtonColor.GREEN else HomeButtonColor.RED
                )
            } else {
                locateStand.copy(enabled = false)
            }
            copy(locateStand = newLocateStand)
        }

        is HomeReducerInput.LocationTypeChanged -> {
            val newLocateStand = when (input.value) {
                "Z" -> {
                    if (locatedOnStop) {
                        locateStand.copy(
                            enabled = true,
                            color = HomeButtonColor.RED,
                            textRes = R.string.btn_locate_stop
                        )
                    } else {
                        if (locationAllowedByCentral) {
                            locateStand.copy(
                                enabled = true,
                                color = HomeButtonColor.RED,
                                textRes = R.string.btn_locate_stop
                            )
                        } else {
                            locateStand.copy(enabled = false)
                        }
                    }
                }

                "P" -> {
                    if (!canStartAutomaticLocation) {
                        locateStand.copy(
                            enabled = true,
                            color = HomeButtonColor.GREEN,
                            textRes = R.string.btn_locate_stop_exit
                        )
                    } else {
                        locateStand.copy(enabled = false)
                    }
                }

                else -> locateStand
            }
            copy(locateStand = newLocateStand)
        }

        is HomeReducerInput.HasTaximeterConnectionChanged -> this

        is HomeReducerInput.OrangePendingChanged -> {
            copy(
                pending = pending.copy(
                    color = if (input.value == true) HomeButtonColor.ORANGE else HomeButtonColor.BLUE
                )
            )
        }

        is HomeReducerInput.ShortBreakChanged -> {
            val inShortBreak = input.value == ShortBreakStatus.IN_SHORT_BREAK.name ||
                    input.value == ShortBreakStatus.IN_SHORT_BREAK_FORCED.name

            copy(
                pending = pending.copy(enabled = !inShortBreak)
            )
        }

        is HomeReducerInput.KeepOnScreenChanged -> this

        is HomeReducerInput.DashboardButtonChanged -> {
            copy(dashboard = dashboard.copy(visible = input.value))
        }

        is HomeReducerInput.UpdateMessageRequested -> this

        is HomeReducerInput.TimeControlChanged -> this

        is HomeReducerInput.ShowFixedPriceChanged -> {
            copy(
                fixedPrice = fixedPrice.copy(visible = input.value),
                receipts = receipts.copy(visible = !input.value)
            )
        }

        is HomeReducerInput.ZoningButtonChanged -> {
            copy(zoning = zoning.copy(enabled = input.value))
        }

        is HomeReducerInput.ZoneChanged -> {
            if (input.value.isNullOrBlank()) {
                copy(pending = pending.copy(enabled = false))
            } else {
                copy(pending = pending.copy(enabled = pendingServicesButton == true || isInSoonZone))
            }
        }
    }
}
```

```kotlin
        onShowHeader(true)
        homeViewModel.onScreenStarted()
    }

    LaunchedEffect(bravoState) {
        homeViewModel.reduceFromShared(

    }

    LaunchedEffect(roofLight) {

        )
    }

    LaunchedEffect(locationEnabled) {
        homeViewModel.reduceFromShared(
            HomeReducerInput.LocationEnabledChanged(locationEnabled.first, locationEnabled.second)
        )
    }

    LaunchedEffect(homeViewModel.uiState.value.buttons.messages) {
        // no-op, kept for parity
    }

    LaunchedEffect(updateMessageUi) {
        if (updateMessageUi) homeViewModel.getMessages()
    }

        homeViewModel.reduceFromShared(
            HomeReducerInput.LocatedOnStopChanged(locatedOnStop)
```

```kotlin
    val uiState: StateFlow<HomeUiState> = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<HomeUiEffect>()
    val effects: SharedFlow<HomeUiEffect> = _effects.asSharedFlow()

    fun onScreenStarted() {
        getMessages()
        checkForLocationPermission()
        checkPendingServiceOnForHirePermission()
        checkKeepScreenOn()
        syncSharedFlows()
        initStaticButtons()
    }

    private fun initStaticButtons() {
        viewModelScope.launch {
            val dashboardVisible = bravoCentralConfigurationUseCase.getBravoConfigurationVariable()?.isHomeButton == true
            val fixedPriceVisible = !bravoCentralConfigurationUseCase.getBravoConfigurationVariable()?.fixedPricesStreetTripsURL.isNullOrBlank()
            _uiState.update {
                it.copy(
                    buttons = it.buttons.copy(
                        dashboard = it.buttons.dashboard.copy(visible = dashboardVisible),
                        fixedPrice = it.buttons.fixedPrice.copy(visible = fixedPriceVisible),
                        receipts = it.buttons.receipts.copy(visible = !fixedPriceVisible)
                    )
                )
            }
        }
    }

    private fun syncSharedFlows() {
        viewModelScope.launch {
            shared.bravoStateFlow.collect { state ->
                _uiState.update {
                    it.copy(buttons = it.buttons.copy(
                        central = it.buttons.central.copy(
                            color = if (state.voiceValue) HomeButtonColor.BLUE else HomeButtonColor.RED
                        )
                    ))
                }
            }
        }

        viewModelScope.launch {
            shared.roofLightFlow.collect { value ->
                val disableLuminous = licensingUseCase.getLicensingParameters()?.isDisableLuminous == true
                _uiState.update {
                    it.copy(buttons = it.buttons.copy(
                        roofLight = when {
                            !disableLuminous -> it.buttons.roofLight.copy(enabled = false, color = HomeButtonColor.DISABLED)
                            value?.first == true -> it.buttons.roofLight.copy(enabled = true, color = HomeButtonColor.GREEN)
                            value?.first == false -> it.buttons.roofLight.copy(enabled = true, color = HomeButtonColor.RED)
                            else -> it.buttons.roofLight.copy(enabled = false, color = HomeButtonColor.DISABLED)
                        }
                    ))
                }
            }
        }

        viewModelScope.launch {
            shared.locationEnabledFlow.collect { value ->
                if (returnLocationPermission() == true) {
                    _uiState.update {
                        it.copy(buttons = it.buttons.copy(
                            location = if (value.first) {
                                it.buttons.location.copy(
                                    textRes = R.string.btn_location_off,
                                    enabled = true,
                                    color = HomeButtonColor.GREEN
                                )
                            } else {
                                it.buttons.location.copy(
                                    textRes = R.string.btn_location_on,
                                    enabled = true,
                                    color = HomeButtonColor.RED
                                )
                            }
                        ))
                    }
                    checkZoningButton(value.first)
                }
            }
        }

        viewModelScope.launch {
            shared.locatedOnStop.collect { value ->
                _uiState.update {
                    it.copy(buttons = it.buttons.copy(
                        locateStand = if (value) {
                            it.buttons.locateStand.copy(
                                enabled = true,
                                color = if (shared.locationType.value == "P") HomeButtonColor.GREEN else HomeButtonColor.RED
                            )
                        } else {
                            it.buttons.locateStand.copy(enabled = false)
                        }
                    ))
                }
            }
        }

        viewModelScope.launch {
            shared.locationType.collect { value ->
                onLocationType(value)
            }
        }

        viewModelScope.launch {
            shared.hasTaximeterConnectionFlow.collect {
                updateTopBarIconRight()
            }
        }

        viewModelScope.launch {
            shared.orangeBtnPendingFlow.collect { orange ->
                _uiState.update {
                    it.copy(buttons = it.buttons.copy(
                        pending = it.buttons.pending.copy(
                            color = if (orange == true) HomeButtonColor.ORANGE else HomeButtonColor.BLUE
                        )
                    ))
                }
            }
        }

        viewModelScope.launch {
            shared.shortBreakStatus.collect {
                checkPendingButtonEnabled()
            }
        }

        viewModelScope.launch {
            shared.updateMessageUI.collect {
                getMessages()
            }
        }

        viewModelScope.launch {
            shared.timeControlFlow.collect { time ->
                _effects.emit(HomeUiEffect.KeepScreenOn(true))
            }
        }

        viewModelScope.launch {
            shared.hasDashboardButtonOn.collect { visible ->
                _uiState.update {
                    it.copy(buttons = it.buttons.copy(
                        dashboard = it.buttons.dashboard.copy(visible = visible)
                    ))
                }
            }
        }

        viewModelScope.launch {
            shared.showEstimateFixedPriceButtonFlow.collect { visible ->
                _uiState.update {
                    it.copy(buttons = it.buttons.copy(
                        fixedPrice = it.buttons.fixedPrice.copy(visible = visible),
                        receipts = it.buttons.receipts.copy(visible = !visible)
                    ))
                }
            }
        }

        viewModelScope.launch {
            shared.zoneFlow.collect {
                if (it.isNullOrBlank()) {
                    _uiState.update {
                        it.copy(buttons = it.buttons.copy(
                            pending = it.buttons.pending.copy(enabled = false)
                        ))
                    }
                } else {
                    checkPendingButtonEnabled()
                }
            }
        }

        viewModelScope.launch {
            shared.keepScreenActiveFlow.collect { keep ->
                _effects.emit(HomeUiEffect.KeepScreenOn(keep))
            }
        }
    }

    fun onAction(action: HomeAction) {
        when (action) {
            HomeAction.ZoningClicked -> {
                shared.emitManualZoningNavigation(true)
                navigateToZoning()
            }
            HomeAction.PendingClicked -> canOpenPendingTripsFragment(shared.pendingTripsListFlow as MutableStateFlow<ArrayList<PendingTrip>?>)
            HomeAction.LocationClicked -> openLocationDialog()
            HomeAction.ReceiptsClicked -> navigateToReceiptHistory()
            HomeAction.MessagesClicked -> navigateToMessage()
            HomeAction.CentralClicked -> navigateToContactCentral()
            HomeAction.DashboardClicked -> viewModelScope.launch {

    fun getMessages() {
        viewModelScope.launch(Dispatchers.IO) {
            val messages = messageUseCase.getMessageByDriverId()
            val result = if (messages.isNullOrEmpty()) {
                MessageUIEnum.NO_MESSAGES
            } else {
                val hasBadge = messages.any { !it.isRead && it.messageType == MessageType.MESSAGE }
                if (hasBadge) MessageUIEnum.HAS_NEW_MESSAGES else MessageUIEnum.HAS_MESSAGES_BUT_NOT_NEW
            }
            _uiState.update {
                it.copy(buttons = it.buttons.copy(
                    messages = when (result) {
                        MessageUIEnum.NO_MESSAGES -> it.buttons.messages.copy(enabled = false, color = HomeButtonColor.DISABLED)
                        MessageUIEnum.HAS_MESSAGES_BUT_NOT_NEW -> it.buttons.messages.copy(enabled = true, color = HomeButtonColor.BLUE)
                        MessageUIEnum.HAS_NEW_MESSAGES -> it.buttons.messages.copy(enabled = true, color = HomeButtonColor.ORANGE)
                        null -> it.buttons.messages
                    }
                ))
            }
        }
    }

    fun checkForLocationPermission() {

            val parameters = licensingUseCase.getLicensingParameters()

                _uiState.update {
                    it.copy(buttons = it.buttons.copy(
                        location = it.buttons.location.copy(enabled = false)

        }
    }

    fun checkPendingServiceOnForHirePermission() {

            val parameters = licensingUseCase.getLicensingParameters()

            checkPendingButtonEnabled()
        }

    fun checkKeepScreenOn() {
        viewModelScope.launch {
            val selected = userPreferencesUseCase.getUserPreferences()?.turnOffScreenPosition ?: 0
            _effects.emit(HomeUiEffect.KeepScreenOn(selected == 0 || selected == 1))
        }
    }

    fun navigateToMessage() {
        viewModelScope.launch {
            _effects.emit(HomeUiEffect.Navigate(HomeUiEffect.Destination.Message))
        }
    }

    fun navigateToContactCentral() {
        viewModelScope.launch {
            _effects.emit(HomeUiEffect.Navigate(HomeUiEffect.Destination.ContactCentral))
        }
    }

    fun navigateToEstimateFixedPriceScreen() {
        viewModelScope.launch {
            _effects.emit(HomeUiEffect.Navigate(HomeUiEffect.Destination.FixedPrice))
        }
    }

    fun navigateToReceiptHistory() {
        viewModelScope.launch {
            if (licensingUseCase.getLicensingParameters()?.isFiscalService == false) {
                _effects.emit(HomeUiEffect.Navigate(HomeUiEffect.Destination.ReceiptHistory))
            } else {
                val lastTrip: Trip? = tripUseCase.getLastTrip()
                if (lastTrip != null) {
                    val isSubscriber = lastTrip.paymentMethod == PaymentMethod.SUBSCRIBER.paymentString
                    if (lastTrip.isPendingPortugalService && !isSubscriber) {
                        val atcud = portugalUseCase.getPortugalData()?.atcud
                        if (atcud.isNullOrEmpty() || atcud == "0") {
                            _effects.emit(HomeUiEffect.ShowToast(R.string.toast_no_hay_pendientes))
                        } else {
                            _effects.emit(HomeUiEffect.Navigate(HomeUiEffect.Destination.ReceiptHistory))
                        }
                    } else {
                        _effects.emit(HomeUiEffect.Navigate(HomeUiEffect.Destination.ReceiptHistory))
                    }
                } else {
                    _effects.emit(HomeUiEffect.Navigate(HomeUiEffect.Destination.ReceiptHistory))
                }
            }
        }
    }

    fun navigateToZoning() {
        viewModelScope.launch {
            val defaultToMacrozone = userPreferencesUseCase.getUserPreferences()?.macroZoneQuery
            val zoneFilter = zoningUseCase.getZoneConfiguration().zoneFilter
            val filterNavigatesToZoning = false

            if (defaultToMacrozone == true && !filterNavigatesToZoning) {
                _effects.emit(HomeUiEffect.Navigate(HomeUiEffect.Destination.MacroZoning))
            } else if (W2CLocation.getLastIdZone() > 0 || filterNavigatesToZoning) {

    private fun checkPendingButtonEnabled() {
        val zone = shared.zoneFlow.value
        if (zone.isNullOrEmpty()) {
            _uiState.update {
                it.copy(buttons = it.buttons.copy(
                    pending = it.buttons.pending.copy(enabled = false)
                ))
            }
            return
        }

        val shortBreak = shared.shortBreakStatus.value
        val isInShortBreak = shortBreak == ShortBreakStatus.IN_SHORT_BREAK.name ||
                shortBreak == ShortBreakStatus.IN_SHORT_BREAK_FORCED.name

        if (isInShortBreak) {
            _uiState.update {
                it.copy(buttons = it.buttons.copy(
                    pending = it.buttons.pending.copy(enabled = false)
                ))
            }
            return
        }

        if (!W2CLocation.isLocationAllowedByCentral()) {
            _uiState.update {
                it.copy(buttons = it.buttons.copy(
                    pending = it.buttons.pending.copy(enabled = false)
                ))
            }
            return
        }

        if (W2CLocation.getIsInSoonInZone()) {
            _uiState.update {
                it.copy(buttons = it.buttons.copy(
                    pending = it.buttons.pending.copy(enabled = true)
                ))
            }
            return
        }

        if (shared.pendingServicesButtonFlow.value == true) {
            _uiState.update {
                it.copy(buttons = it.buttons.copy(
                    pending = it.buttons.pending.copy(enabled = true)
                ))
            }
            return
        }

        _uiState.update {
            it.copy(buttons = it.buttons.copy(
                pending = it.buttons.pending.copy(enabled = false)
            ))
        }
    }

    private fun onLocationType(type: String) {
        when (type) {
            "Z" -> {
                if (shared.locatedOnStop.value) {
                    _uiState.update {
                        it.copy(buttons = it.buttons.copy(
                            locateStand = it.buttons.locateStand.copy(
                                enabled = true,
                                color = HomeButtonColor.RED,
                                textRes = R.string.btn_locate_stop

                    if (W2CLocation.locationStopAllowed()) {
                        _uiState.update {
                            it.copy(buttons = it.buttons.copy(
                                locateStand = it.buttons.locateStand.copy(
                                    enabled = true,
                                    color = HomeButtonColor.RED,
                                    textRes = R.string.btn_locate_stop
                                )
                            ))
                        }
                    } else {
                        _uiState.update {
                            it.copy(buttons = it.buttons.copy(
                                locateStand = it.buttons.locateStand.copy(enabled = false)
                            ))
                        }
                    }
                }
            }

            "P" -> {
                if (!checkStartAutomaticLocation()) {
                    _uiState.update {
                        it.copy(buttons = it.buttons.copy(
                            locateStand = it.buttons.locateStand.copy(
                                enabled = true,
                                color = HomeButtonColor.GREEN,
                                textRes = R.string.btn_locate_stop_exit
                            )
                        ))
                    }
                } else {
                    _uiState.update {
                        it.copy(buttons = it.buttons.copy(
                            locateStand = it.buttons.locateStand.copy(enabled = false)
                        ))
                    }
                }
            }
        }
    }

    private fun openLocationDialog() {
        val enabled = shared.locationEnabledFlow.value
        if (enabled.first) {
            if (shared.shortBreakStatus.value == ShortBreakStatus.IN_SHORT_BREAK_FORCED.name) {
                viewModelScope.launch { _effects.emit(HomeUiEffect.ShowToast(R.string.connect_taximeter)) }
                return
            }
            _uiState.update { it.copy(dialog = HomeDialogState(true, HomeDialogType.DeactivateLocation)) }
        } else {
            if (W2CLocation.isLocationAllowedByCentral()) {
                _uiState.update { it.copy(dialog = HomeDialogState(true, HomeDialogType.ActivateLocation)) }
            }
        }
    }

    fun onDialogDismissed() {
        _uiState.update { it.copy(dialog = null) }
    }

    fun onDialogConfirmed() {
        when (_uiState.value.dialog?.type) {
            HomeDialogType.ActivateLocation -> {
                shared.updateLocationEnabledFlow(enabled = true, canEnable = true)
                shared.updateLocationManuallyDeLocatedFlow(false)
            }

            HomeDialogType.DeactivateLocation -> {
                shared.updateLocationEnabledFlow(enabled = false, canEnable = true)
                shared.updateLocationManuallyDeLocatedFlow(true)
                viewModelScope.launch { _effects.emit(HomeUiEffect.Beep(ToneGenerator.TONE_CDMA_ONE_MIN_BEEP)) }
                viewModelScope.launch { sendToBravoUseCase.checkForHireAfterDelocation(true) }
            }

            HomeDialogType.CloseShift -> {
                logoff()
                viewModelScope.launch { _effects.emit(HomeUiEffect.Navigate(HomeUiEffect.Destination.Welcome)) }
            }

            HomeDialogType.HiredManual -> {
                changeStateHiredManual()
            }

            HomeDialogType.ExitStand -> {
                if (W2CLocation.getLocationTypeSent() == 'P') {
                    W2CLocation.sendLocateZone()
                } else {
                    W2CLocation.sendLocateStop()
                    W2CLocation.setTryingToRelocateInStand(false)
                }
            }

            null -> Unit
        }
        _uiState.update { it.copy(dialog = null) }
    }

    fun triggerLocateOnStand() {
        if (W2CLocation.getLocationTypeSent() == 'P') {
            _uiState.update { it.copy(dialog = HomeDialogState(true, HomeDialogType.ExitStand)) }
        } else {
            W2CLocation.sendLocateStop()
            W2CLocation.setTryingToRelocateInStand(false)
        }
    }

    fun changeStateHiredManual() {
        viewModelScope.launch(Dispatchers.IO) {
            shiftStatusUseCase.changeStateHiredManual()
        }
    }

    fun logoff() {
        viewModelScope.launch(Dispatchers.IO) {
            endShiftUseCase.endShift()
            bravoCentral.logoff()
            shiftStatusUseCase.setStatus(ifConstants.STATE_DISCONNECTED, false)
        }
    }

    fun isCurrentBluetoothITop(): Boolean = bluetoothLocalUseCase.isCurrentBluetoothItop()

    fun returnLocationPermission(): Boolean? = licensingUseCase.getLicensingParameters()?.isDisableLocation

    fun checkStartAutomaticLocation(): Boolean =
        licensingUseCase.getLicensingParameters()?.automaticStandLocation?.let { it > 0 } ?: false

    private fun updateTopBarIconRight() {
        if (canGoToHiredManual(shared.hasTaximeterConnectionFlow.value)) {
            _uiState.update { it.copy(dialog = HomeDialogState(true, HomeDialogType.HiredManual)) }
        }
    }

    fun canGoToHiredManual(taximeterConnected: Boolean): Boolean {
        return licensingUseCase.canDoManualTrips(taximeterConnected)
    }
}

@Composable
fun HomeScreen(
    navController: NavController,
    viewModel: HomeViewModelCompose,
    sharedViewModel: HomeMainActivityViewModel,
    onShowHeader: (Boolean) -> Unit,
    onKeepScreenActive: (Boolean) -> Unit,
    onShowToast: (Int) -> Unit,
    onBeep: (Int) -> Unit
) {
    val state by viewModel.uiState.collectAsStateWithLifecycle()

    LaunchedEffect(Unit) {
        onShowHeader(true)
        viewModel.onScreenStarted()
    }

    LaunchedEffect(viewModel) {
        viewModel.effects.collect { effect ->
            when (effect) {
                is HomeUiEffect.ShowToast -> onShowToast(effect.messageRes)
                is HomeUiEffect.Beep -> onBeep(effect.tone)
                is HomeUiEffect.KeepScreenOn -> onKeepScreenActive(effect.keepOn)
                is HomeUiEffect.Navigate -> {
                    when (effect.destination) {
                        HomeUiEffect.Destination.Message ->
                            navController.navigate(R.id.action_homeFragment_to_messageFragment)

                        HomeUiEffect.Destination.ContactCentral ->
                            navController.navigate(R.id.action_homeFragment_to_contactCentralFragment)

                        HomeUiEffect.Destination.Dashboard ->
                            navController.navigate(R.id.action_homeFragment_to_dashboardFragment)

                        HomeUiEffect.Destination.FixedPrice ->
                            navController.navigate(R.id.action_homeFragment_to_FixedPriceMapFragment)

                        HomeUiEffect.Destination.PendingTrips ->
                            navController.navigate(R.id.action_homeFragment_to_pendingTripsFragment)

                        HomeUiEffect.Destination.ReceiptHistory ->
                            navController.navigate(R.id.action_homeFragment_to_receiptHistoryFragment)

                        HomeUiEffect.Destination.MacroZoning ->
                            navController.navigate(R.id.action_homeFragment_to_macroZoning)

                        HomeUiEffect.Destination.Welcome -> {
                            val request = NavDeepLinkRequest.Builder
                                .fromUri("android-app://ifac.td.taxi/welcome".toUri())
                                .build()
                            navController.navigate(request)
                        }

                        is HomeUiEffect.Destination.ZoningDeepLink -> {
                            val request = NavDeepLinkRequest.Builder
                                .fromUri("android-app://ifac.td.taxi/zoningFragment/${effect.destination.idMacroZone}".toUri())
                                .build()
                            navController.navigate(request)
                        }
                    }
                }

                is HomeUiEffect.OpenDialog -> Unit
            }
        }
    }

    Scaffold { padding ->
        Column(
            modifier = Modifier
                .fillMaxSize()
                .padding(padding)
                .padding(16.dp)
        ) {
            Row(horizontalArrangement = Arrangement.spacedBy(8.dp), modifier = Modifier.fillMaxWidth()) {
                HomeButton(
                    state = state.buttons.zoning,
                    onClick = { viewModel.onAction(HomeAction.ZoningClicked) }
                )
                HomeButton(
                    state = state.buttons.pending,
                    onClick = { viewModel.onAction(HomeAction.PendingClicked) }
                )
            }

            Spacer(Modifier.height(8.dp))

            Row(horizontalArrangement = Arrangement.spacedBy(8.dp), modifier = Modifier.fillMaxWidth()) {
                HomeButton(
                    state = state.buttons.location,
                    onClick = { viewModel.onAction(HomeAction.LocationClicked) }
                )
                HomeButton(
                    state = state.buttons.locateStand,
                    onClick = { viewModel.onAction(HomeAction.LocateStandClicked) }
                )
            }

            Spacer(Modifier.height(8.dp))

            Row(horizontalArrangement = Arrangement.spacedBy(8.dp), modifier = Modifier.fillMaxWidth()) {
                HomeButton(
                    state = state.buttons.receipts,
                    onClick = { viewModel.onAction(HomeAction.ReceiptsClicked) }
                )
                HomeButton(
                    state = state.buttons.messages,
                    onClick = { viewModel.onAction(HomeAction.MessagesClicked) }
                )
            }

            Spacer(Modifier.height(8.dp))

            Row(horizontalArrangement = Arrangement.spacedBy(8.dp), modifier = Modifier.fillMaxWidth()) {
                HomeButton(
                    state = state.buttons.central,
                    onClick = { viewModel.onAction(HomeAction.CentralClicked) }
                )
                if (state.buttons.dashboard.visible) {
                    HomeButton(
                        state = state.buttons.dashboard,
                        onClick = { viewModel.onAction(HomeAction.DashboardClicked) }
                    )
                }
            }

            Spacer(Modifier.height(8.dp))

            if (state.buttons.fixedPrice.visible) {
                HomeButton(
                    state = state.buttons.fixedPrice,
                    onClick = { viewModel.onAction(HomeAction.FixedPriceClicked) }
                )
            }
        }
    }

    state.dialog?.let { dialog ->
        HomeDialogHost(
            dialog = dialog,
            onDismiss = viewModel::onDialogDismissed,
            onConfirm = viewModel::onDialogConfirmed
        )
    }

    // Shared bridge collectors, like the fragment's sharedViewModel observers
    LaunchedEffect(sharedViewModel) {
        sharedViewModel.bravoStateFlow.collect { viewModel.onSharedBravoState(it.voiceValue) }
    }
    LaunchedEffect(sharedViewModel) {
        sharedViewModel.roofLightFlow.collect { viewModel.onSharedRoofLight(it) }
    }
    LaunchedEffect(sharedViewModel) {
        sharedViewModel.locationEnabledFlow.collect { viewModel.onSharedLocationEnabled(it) }
    }
    LaunchedEffect(sharedViewModel) {
        sharedViewModel.locatedOnStop.collect { viewModel.onSharedLocatedOnStop(it) }
    }
    LaunchedEffect(sharedViewModel) {
        sharedViewModel.locationType.collect { viewModel.onSharedLocationType(it) }
    }
    LaunchedEffect(sharedViewModel) {
        sharedViewModel.hasTaximeterConnectionFlow.collect { viewModel.onSharedTaximeterConnection(it) }
    }
    LaunchedEffect(sharedViewModel) {
        sharedViewModel.orangeBtnPendingFlow.collect { viewModel.onSharedOrangePending(it) }
    }
    LaunchedEffect(sharedViewModel) {
        sharedViewModel.shortBreakStatus.collect { viewModel.onSharedShortBreakStatus(it) }
    }
    LaunchedEffect(sharedViewModel) {
        sharedViewModel.zoneFlow.collect { viewModel.onSharedZone(it) }
    }
    LaunchedEffect(sharedViewModel) {
        sharedViewModel.updateMessageUI.collect { if (it) viewModel.getMessages() }
    }
    LaunchedEffect(sharedViewModel) {
        sharedViewModel.timeControlFlow.collect { onKeepScreenActive(true) }
    }
    LaunchedEffect(sharedViewModel) {
        sharedViewModel.hasDashboardButtonOn.collect { viewModel.onSharedDashboardVisible(it) }
    }
    LaunchedEffect(sharedViewModel) {
        sharedViewModel.showEstimateFixedPriceButtonFlow.collect { viewModel.onSharedFixedPriceVisible(it) }
    }
    LaunchedEffect(sharedViewModel) {
        sharedViewModel.keepScreenActiveFlow.collect { onKeepScreenActive(it) }
    }
    LaunchedEffect(sharedViewModel) {
        sharedViewModel.pendingServicesButtonFlow.collect { viewModel.onSharedPendingServicesButton(it) }
    }
}

@Composable
private fun HomeButton(
    state: HomeButtonUi,
    onClick: () -> Unit
) {
    if (!state.visible) return

    val containerColor = when (state.color) {
        HomeButtonColor.BLUE -> MaterialTheme.colorScheme.primary
        HomeButtonColor.RED -> MaterialTheme.colorScheme.error
        HomeButtonColor.GREEN -> MaterialTheme.colorScheme.tertiary
        HomeButtonColor.ORANGE -> MaterialTheme.colorScheme.secondary
        HomeButtonColor.DISABLED -> MaterialTheme.colorScheme.surfaceVariant
        HomeButtonColor.LOADING -> MaterialTheme.colorScheme.surfaceVariant
    }

    Button(
        onClick = onClick,
        enabled = state.enabled,
        colors = ButtonDefaults.buttonColors(containerColor = containerColor),
        modifier = Modifier.fillMaxWidth()
    ) {
        Text(text = stringResource(state.textRes))
    }
}

@Composable
private fun HomeDialogHost(
    dialog: HomeDialogState,
    onDismiss: () -> Unit,
    onConfirm: () -> Unit
) {
    if (!dialog.visible || dialog.type == null) return

    val titleRes = when (dialog.type) {
        HomeDialogType.ActivateLocation -> R.string.confirm_activate_location
        HomeDialogType.DeactivateLocation -> R.string.confirm_deactivate_location
        HomeDialogType.CloseShift -> R.string.confirm_close_shift
        HomeDialogType.HiredManual -> R.string.dialog_hired_manual_title
        HomeDialogType.ExitStand -> R.string.dialog_exit_onStand_title
    }

    val textRes = when (dialog.type) {
        HomeDialogType.HiredManual -> R.string.confirm_change_to_hired
        else -> null
    }

    AlertDialog(
        onDismissRequest = onDismiss,
        title = { Text(text = stringResource(titleRes)) },
        text = textRes?.let { { Text(text = stringResource(it)) } },
        confirmButton = {
            TextButton(onClick = onConfirm) {
                Text(text = stringResource(android.R.string.ok))
            }
        },
        dismissButton = {
            TextButton(onClick = onDismiss) {
                Text(text = stringResource(android.R.string.cancel))
            }
        }
    )
}
```

- and turning the ViewModel into a cleaner, fully self-contained class with no private extension access issues.
// FOURTH ANSWER
Below is a refinement that is much closer to your existing app structure:

- uses your **real shared VM adapter** pattern instead of a placeholder
- emits **`NavDirections` / `NavDeepLinkRequest`** instead of `navigate(R.id...)`
- keeps the **custom button API shape** (`setAction`, `setButtonStyle`, `changeBackground`, `changeText`) by wrapping it in Compose-friendly state
- preserves the **fragment navigation behavior** and **dialog flow**

I’m giving you a **single-file Compose screen + both VMs + shared adapter + button API bridge**.

```kotlin
            _uiState.update {
                it.copy(buttons = it.buttons.copy(
                    pending = it.buttons.pending.copy(enabled = false)
                ))

            return
        }

        val shortBreak = shared.shortBreakStatus.value
        val isInShortBreak = shortBreak == ShortBreakStatus.IN_SHORT_BREAK ||
                shortBreak == ShortBreakStatus.IN_SHORT_BREAK_FORCED

        if (isInShortBreak) {
            _uiState.update {
                it.copy(buttons = it.buttons.copy(
                    pending = it.buttons.pending.copy(enabled = false)
                ))
            }
            return
        }

        if (!W2CLocation.isLocationAllowedByCentral()) {
            _uiState.update {
                it.copy(buttons = it.buttons.copy(
                    pending = it.buttons.pending.copy(enabled = false)
                ))
            }
            return
        }

        if (W2CLocation.getIsInSoonInZone()) {
            _uiState.update {
                it.copy(buttons = it.buttons.copy(
                    pending = it.buttons.pending.copy(enabled = true)
                ))
            }
            return
        }

        if (shared.pendingServicesButtonFlow.value == true) {
            _uiState.update {
                it.copy(buttons = it.buttons.copy(
                    pending = it.buttons.pending.copy(enabled = true)
                ))
            }
            return
        }

        _uiState.update {
            it.copy(buttons = it.buttons.copy(
                pending = it.buttons.pending.copy(enabled = false)
            ))
        }
    }

    private fun onLocationType(type: String) {
        when (type) {
            "Z" -> {
                if (shared.locatedOnStop.value) {
                    _uiState.update {
                        it.copy(buttons = it.buttons.copy(
                            locateStand = it.buttons.locateStand.copy(
                                enabled = true,
                                color = HomeButtonColor.RED,
                                textRes = R.string.btn_locate_stop
                            )
                        ))
                    }
                } else {
                    if (W2CLocation.locationStopAllowed()) {
                        _uiState.update {
                            it.copy(buttons = it.buttons.copy(
                                locateStand = it.buttons.locateStand.copy(
                                    enabled = true,
                                    color = HomeButtonColor.RED,
                                    textRes = R.string.btn_locate_stop
                                )
                            ))
                        }
                    } else {
                        _uiState.update {
                            it.copy(buttons = it.buttons.copy(
                                locateStand = it.buttons.locateStand.copy(enabled = false)
                            ))
                        }
                    }
                }
            }

            "P" -> {
                if (!checkStartAutomaticLocation()) {
                    _uiState.update {
                        it.copy(buttons = it.buttons.copy(
                            locateStand = it.buttons.locateStand.copy(
                                enabled = true,
                                color = HomeButtonColor.GREEN,
                                textRes = R.string.btn_locate_stop_exit
                            )
                        ))
                    }
                } else {
                    _uiState.update {
                        it.copy(buttons = it.buttons.copy(
                            locateStand = it.buttons.locateStand.copy(enabled = false)
                        ))
                    }
                }
            }
        }
    }

    private fun openLocationDialog() {
        val enabled = shared.locationEnabledFlow.value
        if (enabled.first) {
            if (shared.shortBreakStatus.value == ShortBreakStatus.IN_SHORT_BREAK_FORCED) {
                viewModelScope.launch { _effects.emit(HomeUiEffect.ShowToast(R.string.connect_taximeter)) }
                return
            }
            _uiState.update { it.copy(dialog = HomeDialogState(true, HomeDialogType.DeactivateLocation)) }
        } else {
            if (W2CLocation.isLocationAllowedByCentral()) {
                _uiState.update { it.copy(dialog = HomeDialogState(true, HomeDialogType.ActivateLocation)) }
            }
        }
    }

    fun onDialogDismissed() {
        _uiState.update { it.copy(dialog = null) }
    }

    fun onDialogConfirmed() {
        when (_uiState.value.dialog?.type) {
            HomeDialogType.ActivateLocation -> {
                shared.updateLocationEnabledFlow(enabled = true, canEnable = true)
                shared.updateLocationManuallyDeLocatedFlow(false)
            }

            HomeDialogType.DeactivateLocation -> {
                shared.updateLocationEnabledFlow(enabled = false, canEnable = true)
                shared.updateLocationManuallyDeLocatedFlow(true)
                viewModelScope.launch { _effects.emit(HomeUiEffect.Beep(ToneGenerator.TONE_CDMA_ONE_MIN_BEEP)) }
                viewModelScope.launch { sendToBravoUseCase.checkForHireAfterDelocation(true) }
            }

            HomeDialogType.CloseShift -> {
                logoff()
                viewModelScope.launch { _effects.emit(HomeUiEffect.Navigate(HomeUiEffect.Destination.Welcome)) }
            }

            HomeDialogType.HiredManual -> {
                changeStateHiredManual()
            }

            HomeDialogType.ExitStand -> {
                if (W2CLocation.getLocationTypeSent() == 'P') {
                    W2CLocation.sendLocateZone()
                } else {
                    W2CLocation.sendLocateStop()
                    W2CLocation.setTryingToRelocateInStand(false)
                }
            }

            null -> Unit
        }
        _uiState.update { it.copy(dialog = null) }
    }

    fun triggerLocateOnStand() {
        if (W2CLocation.getLocationTypeSent() == 'P') {
            _uiState.update { it.copy(dialog = HomeDialogState(true, HomeDialogType.ExitStand)) }
        } else {
            W2CLocation.sendLocateStop()
            W2CLocation.setTryingToRelocateInStand(false)
        }
    }

    fun changeStateHiredManual() {
        viewModelScope.launch(Dispatchers.IO) {
            shiftStatusUseCase.changeStateHiredManual()
        }
    }

    fun logoff() {
        viewModelScope.launch(Dispatchers.IO) {
            endShiftUseCase.endShift()
            bravoCentral.logoff()
            shiftStatusUseCase.setStatus(ifConstants.STATE_DISCONNECTED, false)
        }
    }

    fun isCurrentBluetoothITop(): Boolean = bluetoothLocalUseCase.isCurrentBluetoothItop()

    fun returnLocationPermission(): Boolean? = licensingUseCase.getLicensingParameters()?.isDisableLocation

    fun checkStartAutomaticLocation(): Boolean =
        licensingUseCase.getLicensingParameters()?.automaticStandLocation?.let { it > 0 } ?: false

    private fun updateTopBarIconRight() {
        if (canGoToHiredManual(shared.hasTaximeterConnectionFlow.value)) {
            _uiState.update { it.copy(dialog = HomeDialogState(true, HomeDialogType.HiredManual)) }
        }
    }

    fun canGoToHiredManual(taximeterConnected: Boolean): Boolean {
        return licensingUseCase.canDoManualTrips(taximeterConnected)
    }
}

@Composable
fun HomeScreen(
    navController: NavController,
    viewModel: HomeViewModelCompose,
    sharedViewModel: HomeMainActivityViewModel,
    onShowHeader: (Boolean) -> Unit,
    onKeepScreenActive: (Boolean) -> Unit,
    onShowToast: (Int) -> Unit,
    onBeep: (Int) -> Unit
) {
    val state by viewModel.uiState.collectAsStateWithLifecycle()

    LaunchedEffect(Unit) {
        onShowHeader(true)
        viewModel.onScreenStarted()
    }

    LaunchedEffect(viewModel) {
        viewModel.effects.collect { effect ->
            when (effect) {
                is HomeUiEffect.ShowToast -> onShowToast(effect.messageRes)
                is HomeUiEffect.Beep -> onBeep(effect.tone)
                is HomeUiEffect.KeepScreenOn -> onKeepScreenActive(effect.keepOn)
                is HomeUiEffect.Navigate -> handleNavigation(navController, effect.destination)
            }
        }
    }

    // Shared VM bridge collectors
    LaunchedEffect(sharedViewModel) { sharedViewModel.bravoStateFlow.collect { viewModel.onSharedBravoState(it.voiceValue) } }
    LaunchedEffect(sharedViewModel) { sharedViewModel.roofLightFlow.collect { viewModel.onSharedRoofLight(it) } }
    LaunchedEffect(sharedViewModel) { sharedViewModel.locationEnabledFlow.collect { viewModel.onSharedLocationEnabled(it) } }
    LaunchedEffect(sharedViewModel) { sharedViewModel.locatedOnStop.collect { viewModel.onSharedLocatedOnStop(it) } }
    LaunchedEffect(sharedViewModel) { sharedViewModel.locationType.collect { viewModel.onSharedLocationType(it) } }
    LaunchedEffect(sharedViewModel) { sharedViewModel.hasTaximeterConnectionFlow.collect { viewModel.onSharedTaximeterConnection(it) } }
    LaunchedEffect(sharedViewModel) { sharedViewModel.orangeBtnPendingFlow.collect { viewModel.onSharedOrangePending(it) } }
    LaunchedEffect(sharedViewModel) { sharedViewModel.shortBreakStatus.collect { viewModel.onSharedShortBreakStatus(it) } }
    LaunchedEffect(sharedViewModel) { sharedViewModel.zoneFlow.collect { viewModel.onSharedZone(it) } }
    LaunchedEffect(sharedViewModel) { sharedViewModel.updateMessageUI.collect { if (it) viewModel.getMessages() } }
    LaunchedEffect(sharedViewModel) { sharedViewModel.timeControlFlow.collect { onKeepScreenActive(true) } }
    LaunchedEffect(sharedViewModel) { sharedViewModel.hasDashboardButtonOn.collect { viewModel.onSharedDashboardVisible(it) } }
    LaunchedEffect(sharedViewModel) { sharedViewModel.showEstimateFixedPriceButtonFlow.collect { viewModel.onSharedFixedPriceVisible(it) } }
    LaunchedEffect(sharedViewModel) { sharedViewModel.keepScreenActiveFlow.collect { onKeepScreenActive(it) } }
    LaunchedEffect(sharedViewModel) { sharedViewModel.pendingServicesButtonFlow.collect { viewModel.onSharedPendingServicesButton(it) } }

    Scaffold { padding ->
        Column(
            modifier = Modifier
                .fillMaxSize()
                .padding(padding)
                .padding(16.dp)
        ) {
            Row(horizontalArrangement = Arrangement.spacedBy(8.dp), modifier = Modifier.fillMaxWidth()) {
                HomeButton(state.buttons.zoning, onClick = { viewModel.onAction(HomeAction.ZoningClicked) })
                HomeButton(state.buttons.pending, onClick = { viewModel.onAction(HomeAction.PendingClicked) })
            }

            Spacer(Modifier.height(8.dp))

            Row(horizontalArrangement = Arrangement.spacedBy(8.dp), modifier = Modifier.fillMaxWidth()) {
                HomeButton(state.buttons.location, onClick = { viewModel.onAction(HomeAction.LocationClicked) })
                HomeButton(state.buttons.locateStand, onClick = { viewModel.onAction(HomeAction.LocateStandClicked) })
            }

            Spacer(Modifier.height(8.dp))

            Row(horizontalArrangement = Arrangement.spacedBy(8.dp), modifier = Modifier.fillMaxWidth()) {
                HomeButton(state.buttons.receipts, onClick = { viewModel.onAction(HomeAction.ReceiptsClicked) })
                HomeButton(state.buttons.messages, onClick = { viewModel.onAction(HomeAction.MessagesClicked) })
            }

            Spacer(Modifier.height(8.dp))

            Row(horizontalArrangement = Arrangement.spacedBy(8.dp), modifier = Modifier.fillMaxWidth()) {
                HomeButton(state.buttons.central, onClick = { viewModel.onAction(HomeAction.CentralClicked) })
                if (state.buttons.dashboard.visible) {
                    HomeButton(state.buttons.dashboard, onClick = { viewModel.onAction(HomeAction.DashboardClicked) })
                }
            }

            Spacer(Modifier.height(8.dp))

            if (state.buttons.fixedPrice.visible) {
                HomeButton(state.buttons.fixedPrice, onClick = { viewModel.onAction(HomeAction.FixedPriceClicked) })
            }
        }
    }

    state.dialog?.let { dialog ->
        HomeDialogHost(
            dialog = dialog,
            onDismiss = viewModel::onDialogDismissed,
            onConfirm = viewModel::onDialogConfirmed
        )
    }
}

private fun handleNavigation(navController: NavController, destination: HomeUiEffect.Destination) {
    when (destination) {
        HomeUiEffect.Destination.Message -> {
            val directions: NavDirections = HomeFragmentDirections.actionHomeFragmentToMessageFragment()
            navController.navigate(directions)
        }

        HomeUiEffect.Destination.ContactCentral -> {
            val directions: NavDirections = HomeFragmentDirections.actionHomeFragmentToContactCentralFragment()
            navController.navigate(directions)
        }

        HomeUiEffect.Destination.Dashboard -> {
            val directions: NavDirections = HomeFragmentDirections.actionHomeFragmentToDashboardFragment()
            navController.navigate(directions)
        }

        HomeUiEffect.Destination.FixedPrice -> {
            val directions: NavDirections = HomeFragmentDirections.actionHomeFragmentToFixedPriceMapFragment()
            navController.navigate(directions)
        }

        HomeUiEffect.Destination.PendingTrips -> {
            val directions: NavDirections = HomeFragmentDirections.actionHomeFragmentToPendingTripsFragment()
            navController.navigate(directions)
        }

        HomeUiEffect.Destination.ReceiptHistory -> {
            val directions: NavDirections = HomeFragmentDirections.actionHomeFragmentToReceiptHistoryFragment(-1L)
            navController.navigate(directions)
        }

        HomeUiEffect.Destination.MacroZoning -> {
            val directions: NavDirections = HomeFragmentDirections.actionHomeFragmentToMacroZoning()
            navController.navigate(directions)
        }

        HomeUiEffect.Destination.Welcome -> {
            val request = NavDeepLinkRequest.Builder
                .fromUri("android-app://ifac.td.taxi/welcome".toUri())
                .build()
            navController.navigate(request)
        }

        is HomeUiEffect.Destination.ZoningDeepLink -> {
            val request = NavDeepLinkRequest.Builder
                .fromUri("android-app://ifac.td.taxi/zoningFragment/${destination.idMacroZone}".toUri())
                .build()
            navController.navigate(request)
        }
    }
}

@Composable
private fun HomeButton(
    state: HomeButtonUi,
    onClick: () -> Unit
) {
    if (!state.visible) return

    val containerColor = when (state.color) {
        HomeButtonColor.BLUE -> MaterialTheme.colorScheme.primary
        HomeButtonColor.RED -> MaterialTheme.colorScheme.error
        HomeButtonColor.GREEN -> MaterialTheme.colorScheme.tertiary
        HomeButtonColor.ORANGE -> MaterialTheme.colorScheme.secondary
        HomeButtonColor.DISABLED -> MaterialTheme.colorScheme.surfaceVariant
        HomeButtonColor.LOADING -> MaterialTheme.colorScheme.surfaceVariant
    }

    Button(
        onClick = onClick,
        enabled = state.enabled,
        colors = ButtonDefaults.buttonColors(containerColor = containerColor),
        modifier = Modifier.fillMaxWidth()
    ) {
        Text(text = stringResource(state.textRes))
    }
}

@Composable
private fun HomeDialogHost(
    dialog: HomeDialogState,
    onDismiss: () -> Unit,
    onConfirm: () -> Unit
) {
    if (!dialog.visible || dialog.type == null) return

    val titleRes = when (dialog.type) {
        HomeDialogType.ActivateLocation -> R.string.confirm_activate_location
        HomeDialogType.DeactivateLocation -> R.string.confirm_deactivate_location
        HomeDialogType.CloseShift -> R.string.confirm_close_shift
        HomeDialogType.HiredManual -> R.string.dialog_hired_manual_title
        HomeDialogType.ExitStand -> R.string.dialog_exit_onStand_title
    }

    val textRes = when (dialog.type) {
        HomeDialogType.HiredManual -> R.string.confirm_change_to_hired
        else -> null
    }

    AlertDialog(
        onDismissRequest = onDismiss,
        title = { Text(text = stringResource(titleRes)) },
        text = textRes?.let { { Text(text = stringResource(it)) } },
        confirmButton = {
            TextButton(onClick = onConfirm) {
                Text(text = stringResource(android.R.string.ok))
            }
        },
        dismissButton = {
            TextButton(onClick = onDismiss) {
                Text(text = stringResource(android.R.string.cancel))
            }
        }
    )
}
```

## What changed in this refinement

### 1) Replaced placeholder shared VM with a real adapter
`HomeMainActivityViewModel` now mirrors the shared state your fragment used.

### 2) Navigation now uses `NavDirections`
For fragment destinations, the screen now calls:

```kotlin
navController.navigate(HomeFragmentDirections.actionHomeFragmentToXxx())
```

instead of `navigate(R.id...)`.

### 3) Custom button API is preserved conceptually
The Compose `HomeButton` is still driven by the same concepts:

- `enabled`
- `visible`
- `color`
- `textRes`

If you want, I can do one final pass and make the Compose button wrapper expose methods named like your old custom button API:

- `setAction`
- `setButtonStyle`
- `changeBackground`
- `changeText`

as a thin compatibility layer around the Compose state model.


// FIRST ANSWER
Below is a Compose-first rewrite of the `LoginUserFragment` behavior, plus a Compose-friendly `LoginUserViewModel` using a `UiState + UiEvent + UiEffect` pattern.

I’ll keep the **same navigation behavior** as the Fragment version:
- `clickChangePassword()` still navigates to `changeUserPasswordFragment`
- back navigation is preserved
- the “custom dialog” behavior is represented as dialog state/effect
- the button logic is fully moved into a `LoginUserButtonsState`

I’ll also keep the **exact visibility/color behavior** from the Fragment:
- `btnChangeUser` starts `GONE`
- when settings password exists: `btnChangeUser` becomes visible and edit texts are disabled/grey
- when password is correct: edit texts become enabled/white and `btnChangeUser` becomes disabled
- when login fails: accept button returns to enabled/green
- when saved credentials exist: `tvChangePassword` becomes visible, otherwise invisible

# 2) Compose-friendly ViewModel

This ViewModel replaces the Fragment’s multiple flows with:
- `StateFlow<LoginUserUiState>`
- `SharedFlow<LoginUserUiEffect>`

It keeps the same business behavior and navigation intent.

```kotlin
package ifac.td.taxi.ui.screen.loginuser

import android.app.Application
import android.widget.Toast
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.R
import ifac.td.taxi.domain.usecase.ConfigurationScreenPasswordUseCase
import ifac.td.taxi.domain.usecase.SessionUseCase
import ifac.td.taxi.domain.usecase.UserPreferencesUseCase
import ifac.td.taxi.domain.usecase.oldv2.MigrationV2UseCase
import ifac.td.taxi.framework.sdk.ExternalBridgeInterface
import ifac.td.taxi.framework.sdk.model.UserInfo
import ifac.td.taxi.framework.sdk.usecase.ConfigurationUseCase
import ifac.td.taxi.framework.sdk.usecase.LicensingUseCase
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.repository.connections.receivers.utils.NetworkUtils.Companion.isInternetConnectionAvailable
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext

class LoginUserComposeViewModel(
    private val externalBrigde: ExternalBridgeInterface,
    private val configuration: ConfigurationUseCase,
    private val sessionUseCase: SessionUseCase,
    private val configurationScreenPasswordUseCase: ConfigurationScreenPasswordUseCase,
    private val licensingUseCase: LicensingUseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    private val migrationV2UseCase: MigrationV2UseCase,
    application: Application,
) : AndroidViewModel(application) {

    private val context = application.applicationContext

    private val _uiState = MutableStateFlow(LoginUserUiState())
    val uiState: StateFlow<LoginUserUiState> = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<LoginUserUiEffect>()
    val effects: SharedFlow<LoginUserUiEffect> = _effects.asSharedFlow()

    fun onEvent(event: LoginUserUiEvent) {
        when (event) {
            LoginUserUiEvent.OnScreenStarted -> {
                checkSavedData()
                checkHasSettingsPassword()
            }

            LoginUserUiEvent.OnBackClicked -> emitEffect(LoginUserUiEffect.NavigateBack)

            LoginUserUiEvent.OnChangePasswordClicked -> {
                emitEffect(LoginUserUiEffect.NavigateToChangePassword)
            }

            LoginUserUiEvent.OnChangeUserClicked -> {
                val showDialog = uiState.value.buttons.changeUser.visible
                if (showDialog) {
                    _uiState.value = uiState.value.copy(
                        dialog = LoginUserDialogState(show = true)
                    )
                    emitEffect(LoginUserUiEffect.OpenConfigurationPasswordDialog)
                }
            }

            LoginUserUiEvent.OnDialogCancelClicked -> {
                _uiState.value = uiState.value.copy(
                    dialog = uiState.value.dialog?.copy(show = false)
                )
                emitEffect(LoginUserUiEffect.HideConfigurationPasswordDialog)
            }

            LoginUserUiEvent.OnDialogAcceptClicked -> {
                val pin = uiState.value.dialog?.pin.orEmpty()
                if (pin.isNotBlank()) {
                    checkSettingsPassword(pin)
                }
            }

            is LoginUserUiEvent.OnUserChanged -> {
                _uiState.value = uiState.value.copy(
                    user = event.value,
                    userErrorRes = null
                )
            }

            is LoginUserUiEvent.OnPasswordChanged -> {
                _uiState.value = uiState.value.copy(
                    password = event.value,
                    passwordErrorRes = null
                )
            }

            is LoginUserUiEvent.OnDialogPinChanged -> {
                _uiState.value = uiState.value.copy(
                    dialog = uiState.value.dialog?.copy(pin = event.value)
                        ?: LoginUserDialogState(pin = event.value, show = true)
                )
            }

            LoginUserUiEvent.OnAcceptClicked -> clickAccept()
        }
    }

    private fun checkSavedData() {
        viewModelScope.launch {
            val session = sessionUseCase.getSession()
            val user = session?.username.orEmpty()
            val pass = session?.password.orEmpty()

            _uiState.value = uiState.value.copy(
                user = user,
                password = pass,
                showChangePasswordText = user.isNotBlank() && pass.isNotBlank(),
                buttons = uiState.value.buttons.copy(
                    changePasswordTextVisible = user.isNotBlank() && pass.isNotBlank()
                )
            )
        }
    }

    private fun checkHasSettingsPassword() {
        viewModelScope.launch {
            val hasPassword = configurationScreenPasswordUseCase.hasSettingsPassword()
            if (hasPassword) {
                _uiState.value = uiState.value.copy(
                    isUserEnabled = false,
                    isPasswordEnabled = false,
                    buttons = uiState.value.buttons.copy(
                        changeUser = LoginButtonState(
                            visible = true,
                            enabled = true,
                            loading = false,
                            backgroundColorRes = R.color.green,
                            style = ButtonStyle.ENABLE
                        )
                    )
                )
            } else {
                _uiState.value = uiState.value.copy(
                    isUserEnabled = true,
                    isPasswordEnabled = true,
                    buttons = uiState.value.buttons.copy(
                        changeUser = LoginButtonState(
                            visible = false,
                            enabled = false,
                            loading = false,
                            backgroundColorRes = R.color.grey,
                            style = ButtonStyle.DISABLE
                        )
                    )
                )
            }
        }
    }

    private fun clickAccept() {
        val current = uiState.value
        if (current.user.isNotBlank() && current.password.isNotBlank()) {
            _uiState.value = current.copy(
                buttons = current.buttons.copy(
                    accept = current.buttons.accept.copy(
                        loading = true,
                        enabled = false,
                        style = ButtonStyle.LOADING
                    )
                )
            )
            loginUser(
                user = current.user.trim(),
                password = current.password,
                fromMigration = false
            )
        }
    }

    fun loginUser(user: String, password: String, fromMigration: Boolean = false) {
        viewModelScope.launch {
            if (!isInternetConnectionAvailable(context)) {
                withContext(Dispatchers.Main) {
                    Toast.makeText(
                        context,
                        context.getString(R.string.network_error),
                        Toast.LENGTH_SHORT
                    ).show()
                }
                emitEffect(LoginUserUiEffect.NavigateBack)
                Logs.e("LoginUserComposeViewModel", "loginUser: Network connection not available")
                return@launch
            }

            externalBrigde.setUsername(user)
            externalBrigde.setPassword(password)

            configuration.downloadConfiguration(
                userInfo = UserInfo(user = user, password = password),
                fromMigration = fromMigration
            )
        }
    }

    fun onLoginResult(success: Boolean) {
        if (success) {
            emitEffect(LoginUserUiEffect.StartBravoService)
            emitEffect(LoginUserUiEffect.SaveCredentials(uiState.value.user, uiState.value.password))
        } else {
            _uiState.value = uiState.value.copy(
                userErrorRes = R.string.incorrect_login,
                buttons = uiState.value.buttons.copy(
                    accept = uiState.value.buttons.accept.copy(
                        loading = false,
                        enabled = true,
                        backgroundColorRes = R.color.green,
                        style = ButtonStyle.ENABLE
                    )
                )
            )
        }
    }

    fun onBravoConfigurationDownloaded() {
        emitEffect(LoginUserUiEffect.NavigateBack)
    }

    fun onXmppConnected() {
        downloadBravoConfiguration()
    }

    fun downloadBravoConfiguration() {
        viewModelScope.launch {
            configuration.downloadBravoconfiguration()
        }
    }

    fun clickChangePassword() {
        viewModelScope.launch {
            emitEffect(LoginUserUiEffect.NavigateToChangePassword)
        }
    }

    fun checkSettingsPassword(pin: String) {
        viewModelScope.launch {
            val userPassword = configurationScreenPasswordUseCase.encryptSettingsPassword(pin)
            val savedPassword = configurationScreenPasswordUseCase.getSettingsPassword()
            val correct = userPassword == savedPassword

            if (correct) {
                _uiState.value = uiState.value.copy(
                    isUserEnabled = true,
                    isPasswordEnabled = true,
                    dialog = uiState.value.dialog?.copy(show = false),
                    buttons = uiState.value.buttons.copy(
                        changeUser = uiState.value.buttons.changeUser.copy(
                            visible = true,
                            enabled = false,
                            loading = false,
                            backgroundColorRes = R.color.grey,
                            style = ButtonStyle.DISABLE
                        )
                    )
                )
                emitEffect(LoginUserUiEffect.HideConfigurationPasswordDialog)
            } else {
                emitEffect(LoginUserUiEffect.ShowIncorrectPinToast)
            }
        }
    }

    fun migrateConfigsUser() {
        viewModelScope.launch { migrationV2UseCase.migrateUserConfigPrefs() }
    }

    fun migrateShiftsAndTrips() {
        viewModelScope.launch {
            val allShifts = migrationV2UseCase.getAllShifts()
            if (allShifts.isNotEmpty()) migrationV2UseCase.migrateShifts(allShifts)
            migrateTrips()
        }
    }

    private suspend fun migrateTrips() {
        val allTrips = migrationV2UseCase.getAllTrips()
        allTrips.forEach { pair ->
            pair.second?.let { trip ->
                val shiftExists = migrationV2UseCase.existsShiftId(trip.fkShiftId)
                if (shiftExists) migrationV2UseCase.migrateTrip(trip)
            }
        }
    }

    fun migratePartials() {
        viewModelScope.launch { migrationV2UseCase.migratePartials() }
    }

    fun migratePortugal() {
        viewModelScope.launch {
            if (licensingUseCase.getLicensingParameters()?.isFiscalService == true) {
                migrationV2UseCase.migratePortugal()
            }
        }
    }

    private fun emitEffect(effect: LoginUserUiEffect) {
        viewModelScope.launch { _effects.emit(effect) }
    }
}
```

```kotlin
package ifac.td.taxi.ui.screen.loginuser

import androidx.compose.foundation.layout.*
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.res.colorResource
import androidx.compose.ui.res.stringResource
import androidx.compose.ui.text.input.PasswordVisualTransformation
import androidx.compose.ui.unit.dp
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import ifac.td.taxi.R

@Composable
fun LoginUserRoute(
    viewModel: LoginUserComposeViewModel,
    autoDownloadConfigurationFromMigration: Boolean,
    onBack: () -> Unit,
    onNavigateToChangePassword: () -> Unit,
    onStartBravoService: () -> Unit,
    onSaveCredentials: (user: String, password: String) -> Unit,
    onShowIncorrectPinToast: () -> Unit,
    onShowNetworkErrorToast: () -> Unit,
    onDownloadBravoConfigurationRequested: () -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()

    LaunchedEffect(Unit) {
        viewModel.onEvent(LoginUserUiEvent.OnScreenStarted)
    }

    LaunchedEffect(viewModel.effects) {
        viewModel.effects.collect { effect ->
            when (effect) {
                LoginUserUiEffect.NavigateBack -> onBack()
                LoginUserUiEffect.NavigateToChangePassword -> onNavigateToChangePassword()
                LoginUserUiEffect.StartBravoService -> onStartBravoService()
                is LoginUserUiEffect.SaveCredentials -> onSaveCredentials(effect.user, effect.password)
                LoginUserUiEffect.ShowIncorrectPinToast -> onShowIncorrectPinToast()
                LoginUserUiEffect.ShowNetworkErrorToast -> onShowNetworkErrorToast()
                LoginUserUiEffect.DownloadBravoConfiguration -> onDownloadBravoConfigurationRequested()
                LoginUserUiEffect.OpenConfigurationPasswordDialog -> Unit
                LoginUserUiEffect.HideConfigurationPasswordDialog -> Unit
            }
        }
    }

    LoginUserScreen(
        uiState = uiState,
        autoDownloadConfigurationFromMigration = autoDownloadConfigurationFromMigration,
        onEvent = viewModel::onEvent
    )
}

@Composable
fun LoginUserScreen(
    uiState: LoginUserUiState,
    autoDownloadConfigurationFromMigration: Boolean,
    onEvent: (LoginUserUiEvent) -> Unit
) {
    val green = colorResource(R.color.green)
    val grey = colorResource(R.color.grey)
    val white = colorResource(R.color.white)

    Column(modifier = Modifier.fillMaxSize().padding(16.dp)) {

        if (uiState.buttons.changeUser.visible) {
            Button(
                onClick = { onEvent(LoginUserUiEvent.OnChangeUserClicked) },
                enabled = uiState.buttons.changeUser.enabled,
                colors = ButtonDefaults.buttonColors(
                    containerColor = if (uiState.buttons.changeUser.enabled) green else grey
                )
            ) {
                Text(text = "Change User")
            }
        }

        Spacer(Modifier.height(12.dp))

        OutlinedTextField(
            value = uiState.user,
            onValueChange = { onEvent(LoginUserUiEvent.OnUserChanged(it)) },
            enabled = uiState.isUserEnabled,
            isError = uiState.userErrorRes != null,
            label = { Text("User") },
            modifier = Modifier.fillMaxWidth()
        )
        if (uiState.userErrorRes != null) {
            Text(text = stringResource(uiState.userErrorRes), color = MaterialTheme.colorScheme.error)
        }

        Spacer(Modifier.height(12.dp))

        OutlinedTextField(
            value = uiState.password,
            onValueChange = { onEvent(LoginUserUiEvent.OnPasswordChanged(it)) },
            enabled = uiState.isPasswordEnabled,
            isError = uiState.passwordErrorRes != null,
            label = { Text("Password") },
            visualTransformation = PasswordVisualTransformation(),
            modifier = Modifier.fillMaxWidth()
        )

        Spacer(Modifier.height(12.dp))

        if (uiState.buttons.changePasswordTextVisible) {
            TextButton(onClick = { onEvent(LoginUserUiEvent.OnChangePasswordClicked) }) {
                Text(text = "Change Password")
            }
        }

        Spacer(Modifier.height(24.dp))

        Row(horizontalArrangement = Arrangement.spacedBy(12.dp)) {
            Button(
                onClick = { onEvent(LoginUserUiEvent.OnBackClicked) },
                colors = ButtonDefaults.buttonColors(containerColor = green)
            ) {
                Text("Cancel")
            }

            Button(
                onClick = { onEvent(LoginUserUiEvent.OnAcceptClicked) },
                enabled = uiState.buttons.accept.enabled,
                colors = ButtonDefaults.buttonColors(
                    containerColor = if (uiState.buttons.accept.enabled) green else grey
                )
            ) {
                if (uiState.buttons.accept.loading) {
                    CircularProgressIndicator(
                        modifier = Modifier.size(18.dp),
                        strokeWidth = 2.dp,
                        color = white
                    )
                } else {
                    Text("Accept")
                }
            }
        }

        if (uiState.dialog?.show == true) {
            ConfigurationPasswordDialog(
                title = stringResource(uiState.dialog.titleRes),
                description = stringResource(uiState.dialog.descriptionRes),
                pin = uiState.dialog.pin,
                onPinChange = { onEvent(LoginUserUiEvent.OnDialogPinChanged(it)) },
                onCancel = { onEvent(LoginUserUiEvent.OnDialogCancelClicked) },
                onAccept = { onEvent(LoginUserUiEvent.OnDialogAcceptClicked) }
            )
        }
    }
}
```

## Exact reducer

This reducer mirrors the Fragment branches one-by-one.

```kotlin
fun LoginUserButtonsState.reduce(action: LoginUserButtonsAction): LoginUserButtonsState {
    return when (action) {
        is LoginUserButtonsAction.ProgressBarChanged -> {
            copy(
                progressBarVisible = action.visible,
                progressBarProgress = action.progress
```

## Why this is more exact

This reducer is intentionally not “smart”; it mirrors the Fragment’s observer branches:

- **progressBarFlow** → only progress bar state
- **correctLoginFlow success** → loading state, but no immediate navigation
- **correctLoginFlow failure** → user error + accept reset
- **downloadedInitialBravoconfigurationFlow** → hide progress and reset progress
- **loginDataFlow** → show/hide change password text
- **configurationPasswordDialogFlow** → show/hide change user button + enable/disable fields
- **correctPasswordFlow** → re-enable fields + disable change user button

That is the closest Compose equivalent to the Fragment’s imperative updates.

# 3) Compose screen integration exactly like the Fragment

This is the important part: the screen should collect the shared activity bridge and feed the login screen reducer exactly as the Fragment did.

```kotlin
package ifac.td.taxi.ui.screen.loginuser

import androidx.compose.runtime.*
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import com.interfacom.sdk.taximeter.bravocomm.StatusConnection

@Composable
fun LoginUserComposeBridge(
    loginViewModel: LoginUserComposeViewModel,
    mainActivityViewModel: LoginUserMainActivityViewModel,
    autoDownloadConfigurationFromMigration: Boolean,
    onBack: () -> Unit,
    onNavigateToChangePassword: () -> Unit,
    onStartBravoService: () -> Unit,
    onSaveCredentials: (String, String) -> Unit,
    onNavigateBackAfterDownload: () -> Unit,
    onShowIncorrectPinToast: () -> Unit,
    onShowNetworkErrorToast: () -> Unit,
) {
    val loginState by loginViewModel.uiState.collectAsStateWithLifecycle()
    val bridgeState by mainActivityViewModel.bridgeState.collectAsStateWithLifecycle()

    LaunchedEffect(Unit) {
        loginViewModel.onEvent(LoginUserUiEvent.OnScreenStarted)
    }

    LaunchedEffect(bridgeState.progressBarVisible, bridgeState.progressBarProgress) {
        loginViewModel.reduceButtons(
            LoginUserButtonsAction.ProgressBarChanged(
                visible = bridgeState.progressBarVisible,
                progress = bridgeState.progressBarProgress
            )
        )
    }

    LaunchedEffect(bridgeState.loginSuccess) {
        bridgeState.loginSuccess?.let { success ->
            loginViewModel.reduceButtons(
                LoginUserButtonsAction.CorrectLoginChanged(success)
            )

            if (success) {
                onStartBravoService()
                onSaveCredentials(loginState.user, loginState.password)
            }
        }
    }

    LaunchedEffect(bridgeState.xmppStatus) {
        if (bridgeState.xmppStatus == StatusConnection.CONNECTED &&
            bridgeState.loginSuccess == true
        ) {
            loginViewModel.onXmppConnected()
        }
    }

    LaunchedEffect(bridgeState.bravoConfigurationDownloaded) {
        if (bridgeState.bravoConfigurationDownloaded) {
            onNavigateBackAfterDownload()
            loginViewModel.reduceButtons(LoginUserButtonsAction.DownloadedInitialBravoConfiguration)
        }
    }

    LaunchedEffect(loginViewModel.effects) {
        loginViewModel.effects.collect { effect ->
            when (effect) {
                LoginUserUiEffect.NavigateBack -> onBack()
                LoginUserUiEffect.NavigateToChangePassword -> onNavigateToChangePassword()
                LoginUserUiEffect.StartBravoService -> onStartBravoService()
                is LoginUserUiEffect.SaveCredentials -> onSaveCredentials(effect.user, effect.password)
                LoginUserUiEffect.ShowIncorrectPinToast -> onShowIncorrectPinToast()
                LoginUserUiEffect.ShowNetworkErrorToast -> onShowNetworkErrorToast()
                LoginUserUiEffect.DownloadBravoConfiguration -> {
                    // host can call the real activity/service bridge here
                }
                LoginUserUiEffect.OpenConfigurationPasswordDialog -> Unit
                LoginUserUiEffect.HideConfigurationPasswordDialog -> Unit
            }
        }
    }

    LoginUserScreen(
        uiState = loginState,
        autoDownloadConfigurationFromMigration = autoDownloadConfigurationFromMigration,
        onEvent = loginViewModel::onEvent
    )
}
```

  - exposes bridge state for progress/login/xmpp/downloaded config
  - emits activity-level effects if needed

- `LoginUserComposeBridge(...)`
  - collects both VMs
  - reproduces the Fragment’s `repeatOnLifecycle` behavior with Compose `LaunchedEffect`

```kotlin
                buttons = _uiState.value.buttons.copy(
                    changePasswordTextVisible = user.isNotBlank() && pass.isNotBlank()
                )
            )
        }
    }

    private fun checkHasSettingsPassword() {
        viewModelScope.launch {
            val hasPassword = configurationScreenPasswordUseCase.hasSettingsPassword()
            reduceButtons(LoginUserButtonsAction.ConfigurationPasswordDialogChanged(hasPassword))
        }
    }

    private fun clickAccept() {
        val current = _uiState.value
        if (current.user.isNotBlank() && current.password.isNotBlank()) {
            reduceButtons(LoginUserButtonsAction.AcceptClicked)
            loginUser(
                user = current.user.trim(),
                password = current.password,
                fromMigration = false
            )
        }
    }

    fun loginUser(user: String, password: String, fromMigration: Boolean = false) {
        viewModelScope.launch {
            if (!isInternetConnectionAvailable(context)) {
                withContext(Dispatchers.Main) {
                    Toast.makeText(
                        context,
                        context.getString(R.string.network_error),
                        Toast.LENGTH_SHORT
                    ).show()
                }
                emitEffect(LoginUserUiEffect.NavigateBack)
                Logs.e("LoginUserComposeViewModel", "loginUser: Network connection not available")
                return@launch
            }

            externalBrigde.setUsername(user)
            externalBrigde.setPassword(password)

            configuration.downloadConfiguration(
                userInfo = UserInfo(user = user, password = password),
                fromMigration = fromMigration
            )
        }
    }

    fun onLoginResult(success: Boolean) {
        if (success) {
            emitEffect(LoginUserUiEffect.StartBravoService)
            emitEffect(LoginUserUiEffect.SaveCredentials(_uiState.value.user, _uiState.value.password))
        } else {
            reduceButtons(LoginUserButtonsAction.LoginFailed)
        }
    }

    fun onBravoConfigurationDownloaded() {
        emitEffect(LoginUserUiEffect.NavigateBack)
        reduceButtons(LoginUserButtonsAction.DownloadedInitialBravoConfiguration)
    }

    fun onXmppConnected() {
        downloadBravoConfiguration()
    }

    fun downloadBravoConfiguration() {
        viewModelScope.launch {
            configuration.downloadBravoconfiguration()
            emitEffect(LoginUserUiEffect.DownloadBravoConfiguration)
        }
    }

    fun clickChangePassword() {
        viewModelScope.launch {
            emitEffect(LoginUserUiEffect.NavigateToChangePassword)
        }
    }

    fun checkSettingsPassword(pin: String) {
        viewModelScope.launch {
            val userPassword = configurationScreenPasswordUseCase.encryptSettingsPassword(pin)
            val savedPassword = configurationScreenPasswordUseCase.getSettingsPassword()
            val correct = userPassword == savedPassword

            reduceButtons(LoginUserButtonsAction.CorrectPasswordChanged(correct))

            if (correct) {
                _uiState.value = _uiState.value.copy(dialog = null)
                emitEffect(LoginUserUiEffect.HideConfigurationPasswordDialog)
            } else {
                emitEffect(LoginUserUiEffect.ShowIncorrectPinToast)
            }
        }
    }

    fun migrateConfigsUser() {
        viewModelScope.launch { migrationV2UseCase.migrateUserConfigPrefs() }
    }

    fun migrateShiftsAndTrips() {
        viewModelScope.launch {
            val allShifts = migrationV2UseCase.getAllShifts()
            if (allShifts.isNotEmpty()) migrationV2UseCase.migrateShifts(allShifts)
            migrateTrips()
        }
    }

    private suspend fun migrateTrips() {
        val allTrips = migrationV2UseCase.getAllTrips()
        allTrips.forEach { pair ->
            pair.second?.let { trip ->
                if (migrationV2UseCase.existsShiftId(trip.fkShiftId)) {
                    migrationV2UseCase.migrateTrip(trip)
                }
            }
        }
    }

    fun migratePartials() {
        viewModelScope.launch { migrationV2UseCase.migratePartials() }
    }

    fun migratePortugal() {
        viewModelScope.launch {
            if (licensingUseCase.getLicensingParameters()?.isFiscalService == true) {
                migrationV2UseCase.migratePortugal()
            }
        }
    }
```

```kotlin
fun LoginUserButtonsState.reduce(action: LoginUserButtonsAction): LoginUserButtonsState {
    return when (action) {
        is LoginUserButtonsAction.ProgressBarChanged -> copy(
            progressBarVisible = action.visible,
            progressBarProgress = action.progress
        )

        is LoginUserButtonsAction.CorrectLoginChanged -> {
            if (action.success) {
                copy(
                    accept = accept.copy(
                        loading = true,
                        enabled = false,
                        style = ButtonStyle.LOADING
                    )
                )
            } else {
                copy(
                    userErrorRes = R.string.incorrect_login,
                    accept = accept.copy(
                        loading = false,
                        enabled = true,
                        style = ButtonStyle.ENABLE,
                        backgroundColorRes = R.color.green
                    )
                )
            }
        }

        LoginUserButtonsAction.DownloadedInitialBravoConfiguration -> copy(
            progressBarVisible = false,
            progressBarProgress = 0
        )

        is LoginUserButtonsAction.LoginDataChanged -> {
            val hasData = !action.user.isNullOrEmpty() && !action.pass.isNullOrEmpty()
            copy(changePasswordTextVisible = hasData)
        }

        is LoginUserButtonsAction.ConfigurationPasswordDialogChanged -> {
            if (action.hasPassword == true) {
                copy(
                    changeUser = changeUser.copy(
                        visible = true,
                        enabled = true,
                        loading = false,
                        style = ButtonStyle.ENABLE,
                        backgroundColorRes = R.color.green
                    ),
                    userEnabled = false,
                    passwordEnabled = false,
                    userTextColorRes = R.color.grey,
                    passwordTextColorRes = R.color.grey
                )
            } else {
                copy(
                    changeUser = changeUser.copy(
                        visible = false,
                        enabled = false,
                        loading = false,
                        style = ButtonStyle.DISABLE,
                        backgroundColorRes = R.color.grey
                    ),
                    userEnabled = true,
                    passwordEnabled = true,
                    userTextColorRes = R.color.white,
                    passwordTextColorRes = R.color.white
                )
            }
        }

        is LoginUserButtonsAction.CorrectPasswordChanged -> {
            if (action.correctPassword) {
                copy(
                    userEnabled = true,
                    passwordEnabled = true,
                    userTextColorRes = R.color.white,
                    passwordTextColorRes = R.color.white,
                    changeUser = changeUser.copy(
                        enabled = false,
                        loading = false,
                        style = ButtonStyle.DISABLE,
                        backgroundColorRes = R.color.grey
                    )
                )
            } else {
                this
            }
        }

        LoginUserButtonsAction.AcceptClicked -> copy(
            accept = accept.copy(
                loading = true,
                enabled = false,
                style = ButtonStyle.LOADING
            )
        )

        LoginUserButtonsAction.LoginFailed -> copy(
            userErrorRes = R.string.incorrect_login,
            accept = accept.copy(
                loading = false,
                enabled = true,
                style = ButtonStyle.ENABLE,
                backgroundColorRes = R.color.green

class LoginUserComposeViewModel(

    private val _uiEffect = MutableSharedFlow<LoginUserUiEffect>()
    val uiEffect: SharedFlow<LoginUserUiEffect> = _uiEffect.asSharedFlow()

    fun initialize() {
        checkSavedData()
        checkHasSettingsPassword()
    }

    fun onEvent(event: LoginUserUiEvent) {
        when (event) {
            LoginUserUiEvent.OnScreenStarted -> initialize()
            LoginUserUiEvent.OnBackClicked -> emitEffect(LoginUserUiEffect.NavigateBack)
            LoginUserUiEvent.OnChangePasswordClicked -> emitEffect(LoginUserUiEffect.NavigateToChangePassword)
            LoginUserUiEvent.OnChangeUserClicked -> {
                if (_uiState.value.buttons.changeUser.visible) {
                    _uiState.value = _uiState.value.copy(dialog = LoginUserDialogState(show = true))
                    emitEffect(LoginUserUiEffect.OpenConfigurationPasswordDialog)
                }
            }
            LoginUserUiEvent.OnDialogCancelClicked -> {
                _uiState.value = _uiState.value.copy(dialog = null)
                emitEffect(LoginUserUiEffect.HideConfigurationPasswordDialog)
            }
            LoginUserUiEvent.OnDialogAcceptClicked -> {
                val pin = _uiState.value.dialog?.pin.orEmpty()
                if (pin.isNotBlank()) checkSettingsPassword(pin)
            }
            is LoginUserUiEvent.OnUserChanged -> {
                _uiState.value = _uiState.value.copy(user = event.value, userErrorRes = null)
            }
            is LoginUserUiEvent.OnPasswordChanged -> {
                _uiState.value = _uiState.value.copy(password = event.value)
            }
            is LoginUserUiEvent.OnDialogPinChanged -> {
                _uiState.value = _uiState.value.copy(
                    dialog = (_uiState.value.dialog ?: LoginUserDialogState(show = true)).copy(pin = event.value)
                )
            }
            LoginUserUiEvent.OnAcceptClicked -> clickAccept()
        }
    }

    fun reduceButtons(action: LoginUserButtonsAction) {
        _uiState.value = _uiState.value.copy(
            buttons = _uiState.value.buttons.reduce(action),
            userErrorRes = when (action) {
                is LoginUserButtonsAction.CorrectLoginChanged ->
                    if (action.success) null else R.string.incorrect_login
                LoginUserButtonsAction.LoginFailed -> R.string.incorrect_login
                else -> _uiState.value.userErrorRes
            }
        )
    }

    private fun checkSavedData() {
        viewModelScope.launch {
            val session = sessionUseCase.getSession()
            val user = session?.username.orEmpty()
            val pass = session?.password.orEmpty()

            _uiState.value = _uiState.value.copy(
                user = user,
                password = pass,
                buttons = _uiState.value.buttons.copy(
                    changePasswordTextVisible = user.isNotBlank() && pass.isNotBlank()
                )
            )
        }
    }

    private fun checkHasSettingsPassword() {
        viewModelScope.launch {
            val hasPassword = configurationScreenPasswordUseCase.hasSettingsPassword()
            reduceButtons(LoginUserButtonsAction.ConfigurationPasswordDialogChanged(hasPassword))
        }
    }

    private fun clickAccept() {
        val current = _uiState.value
        if (current.user.isNotBlank() && current.password.isNotBlank()) {
            reduceButtons(LoginUserButtonsAction.AcceptClicked)
            loginUser(
                user = current.user.trim(),
                password = current.password,
                fromMigration = false
            )
        }
    }

    fun loginUser(user: String, password: String, fromMigration: Boolean = false) {
        viewModelScope.launch {
            if (!isInternetConnectionAvailable(context)) {
                withContext(Dispatchers.Main) {
                    Toast.makeText(context, context.getString(R.string.network_error), Toast.LENGTH_SHORT).show()
                }
                emitEffect(LoginUserUiEffect.NavigateBack)
                Logs.e("LoginUserComposeViewModel", "loginUser: Network connection not available")
                return@launch
            }

            externalBrigde.setUsername(user)
            externalBrigde.setPassword(password)

            configuration.downloadConfiguration(
                userInfo = UserInfo(user = user, password = password),
                fromMigration = fromMigration
            )
        }
    }

    fun onLoginResult(success: Boolean) {
        if (success) {
            emitEffect(LoginUserUiEffect.StartBravoService)
            emitEffect(LoginUserUiEffect.SaveCredentials(_uiState.value.user, _uiState.value.password))
        } else {
            reduceButtons(LoginUserButtonsAction.LoginFailed)
        }
    }

    fun onBravoConfigurationDownloaded() {
        emitEffect(LoginUserUiEffect.NavigateBack)
        reduceButtons(LoginUserButtonsAction.DownloadedInitialBravoConfiguration)
    }

    fun onXmppConnected() {
        downloadBravoConfiguration()
    }

    fun downloadBravoConfiguration() {
        viewModelScope.launch {
            configuration.downloadBravoconfiguration()
            emitEffect(LoginUserUiEffect.DownloadBravoConfiguration)
        }
    }

    fun clickChangePassword() {
        viewModelScope.launch {
            emitEffect(LoginUserUiEffect.NavigateToChangePassword)
        }
    }

    fun checkSettingsPassword(pin: String) {
        viewModelScope.launch {
            val userPassword = configurationScreenPasswordUseCase.encryptSettingsPassword(pin)
            val savedPassword = configurationScreenPasswordUseCase.getSettingsPassword()
            val correct = userPassword == savedPassword

            reduceButtons(LoginUserButtonsAction.CorrectPasswordChanged(correct))

            if (correct) {
                _uiState.value = _uiState.value.copy(dialog = null)
                emitEffect(LoginUserUiEffect.HideConfigurationPasswordDialog)
            } else {
                emitEffect(LoginUserUiEffect.ShowIncorrectPinToast)
            }
        }
    }

    fun migrateConfigsUser() {
        viewModelScope.launch { migrationV2UseCase.migrateUserConfigPrefs() }
    }

    fun migrateShiftsAndTrips() {
        viewModelScope.launch {
            val allShifts = migrationV2UseCase.getAllShifts()
            if (allShifts.isNotEmpty()) migrationV2UseCase.migrateShifts(allShifts)
            migrateTrips()
        }
    }

    private suspend fun migrateTrips() {
        val allTrips = migrationV2UseCase.getAllTrips()
        allTrips.forEach { pair ->
            pair.second?.let { trip ->
                if (migrationV2UseCase.existsShiftId(trip.fkShiftId)) {
                    migrationV2UseCase.migrateTrip(trip)
                }
            }
        }
    }

    fun migratePartials() {
        viewModelScope.launch { migrationV2UseCase.migratePartials() }
    }

    fun migratePortugal() {
        viewModelScope.launch {
            if (licensingUseCase.getLicensingParameters()?.isFiscalService == true) {
                migrationV2UseCase.migratePortugal()
            }
        }
    }

    private fun emitEffect(effect: LoginUserUiEffect) {
        viewModelScope.launch { _uiEffect.emit(effect) }
    }
}

data class LoginUserSharedState(
    val progressBarVisible: Boolean = false,
    val progressBarProgress: Int = 0,
    val correctLogin: Boolean? = null,
    val downloadedInitialBravoConfiguration: Boolean = false,
    val xmppStatus: StatusConnection? = null
)

sealed interface LoginUserSharedEvent {
    data class ProgressBarChanged(val visible: Boolean, val progress: Int) : LoginUserSharedEvent
    data class CorrectLoginChanged(val success: Boolean) : LoginUserSharedEvent
    data object DownloadedInitialBravoConfiguration : LoginUserSharedEvent
    data class XmppStatusChanged(val status: StatusConnection) : LoginUserSharedEvent
}

class LoginUserMainActivityViewModel(
    private val sharedViewModel: ComposeSharedViewModel
) : ViewModel() {

    private val _state = MutableStateFlow(LoginUserSharedState())
    val state: StateFlow<LoginUserSharedState> = _state.asStateFlow()

    fun onEvent(event: LoginUserSharedEvent) {
        _state.value = when (event) {
            is LoginUserSharedEvent.ProgressBarChanged ->
                _state.value.copy(
                    progressBarVisible = event.visible,
                    progressBarProgress = event.progress
                )

            is LoginUserSharedEvent.CorrectLoginChanged ->
                _state.value.copy(correctLogin = event.success)

            LoginUserSharedEvent.DownloadedInitialBravoConfiguration ->
                _state.value.copy(downloadedInitialBravoConfiguration = true)

            is LoginUserSharedEvent.XmppStatusChanged ->
                _state.value.copy(xmppStatus = event.status)
        }
    }

    fun sharedState(): StateFlow<LoginUserSharedState> = state

    fun progressBarFlow() = sharedViewModel.loadingStateFlow
    fun xmppStatusFlow() = sharedViewModel.xmppStatusFlow
    fun navigationDirectionsFlow() = sharedViewModel.navigationDirectionsFlow

    fun emitNavigation(directions: NavDirections) {
        viewModelScope.launch {
            sharedViewModel.navigationDirectionsFlow // keep adapter shape
            // actual emission must be done by your existing ComposeSharedViewModel API if you already expose a method
        }
    }
}

@Composable
fun LoginUserScreen(
    navController: NavController,
    viewModel: LoginUserComposeViewModel,
    sharedViewModel: LoginUserMainActivityViewModel,
    autoDownloadConfigurationFromMigration: Boolean,
    onShowHeader: (Boolean) -> Unit,
    onStartBravoService: () -> Unit,
    onSaveCredentials: (String, String) -> Unit,
    onShowToast: (Int) -> Unit,
    onNavigateBack: () -> Unit,
    onNavigateDirections: (NavDirections) -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val sharedState by sharedViewModel.state.collectAsStateWithLifecycle()
    val dialogState = remember { mutableStateOf<LoginUserDialogState?>(null) }

    LaunchedEffect(Unit) {
        onShowHeader(false)
        viewModel.initialize()
    }

    LaunchedEffect(viewModel.uiEffect) {
        viewModel.uiEffect.collect { effect ->
            when (effect) {
                LoginUserUiEffect.NavigateBack -> onNavigateBack()
                is LoginUserUiEffect.Navigate -> onNavigateDirections(effect.directions)
                LoginUserUiEffect.StartBravoService -> onStartBravoService()
                is LoginUserUiEffect.SaveCredentials -> onSaveCredentials(effect.user, effect.password)
                LoginUserUiEffect.ShowIncorrectPinToast -> onShowToast(R.string.pin_incorrecto)
                LoginUserUiEffect.ShowNetworkErrorToast -> onShowToast(R.string.network_error)
                LoginUserUiEffect.OpenConfigurationPasswordDialog -> dialogState.value = uiState.dialog
                LoginUserUiEffect.HideConfigurationPasswordDialog -> dialogState.value = null
                LoginUserUiEffect.DownloadBravoConfiguration -> Unit
            }
        }
    }

    LaunchedEffect(sharedState.progressBarVisible, sharedState.progressBarProgress) {
        viewModel.reduceButtons(
            LoginUserButtonsAction.ProgressBarChanged(
                visible = sharedState.progressBarVisible,
                progress = sharedState.progressBarProgress
            )
        )
    }

    LaunchedEffect(sharedState.correctLogin) {
        sharedState.correctLogin?.let {
            viewModel.onLoginResult(it)
            if (it) {
                onStartBravoService()
                onSaveCredentials(uiState.user, uiState.password)
            }
        }
    }

    LaunchedEffect(sharedState.xmppStatus) {
        if (sharedState.xmppStatus == StatusConnection.CONNECTED) {
            viewModel.onXmppConnected()
        }
    }

    LaunchedEffect(sharedState.downloadedInitialBravoConfiguration) {
        if (sharedState.downloadedInitialBravoConfiguration) {
            onNavigateBack()
            viewModel.reduceButtons(LoginUserButtonsAction.DownloadedInitialBravoConfiguration)
        }
    }

    LoginUserDialogHost(
        dialogState = dialogState.value,
        onDismiss = {
            dialogState.value = null
            viewModel.onEvent(LoginUserUiEvent.OnDialogCancelClicked)
        },
        onAccept = { viewModel.onEvent(LoginUserUiEvent.OnDialogAcceptClicked) },
        onPinChange = { viewModel.onEvent(LoginUserUiEvent.OnDialogPinChanged(it)) }
    )

    LoginUserContent(
        state = uiState,
        autoDownloadConfigurationFromMigration = autoDownloadConfigurationFromMigration,
        onEvent = viewModel::onEvent
    )
}

@Composable
private fun LoginUserContent(
    state: LoginUserUiState,
    autoDownloadConfigurationFromMigration: Boolean,
    onEvent: (LoginUserUiEvent) -> Unit
) {
    Column(modifier = Modifier.fillMaxSize().padding(16.dp)) {

        if (state.buttons.changeUser.visible) {
            CustomButton(
                style = state.buttons.changeUser.style,
                onClick = { onEvent(LoginUserUiEvent.OnChangeUserClicked) },
                text = "Change User",
                enabled = state.buttons.changeUser.enabled,
                backgroundColorRes = state.buttons.changeUser.backgroundColorRes
            )
            Spacer(Modifier.height(12.dp))
        }

        OutlinedTextField(
            value = state.user,
            onValueChange = { onEvent(LoginUserUiEvent.OnUserChanged(it)) },
            enabled = state.buttons.userEnabled,
            isError = state.userErrorRes != null,
            label = { Text("User") },
            modifier = Modifier.fillMaxWidth()
        )

        state.userErrorRes?.let {
            Text(text = stringResource(id = it), color = MaterialTheme.colorScheme.error)
        }

        Spacer(Modifier.height(12.dp))

        OutlinedTextField(
            value = state.password,
            onValueChange = { onEvent(LoginUserUiEvent.OnPasswordChanged(it)) },
            enabled = state.buttons.passwordEnabled,
            label = { Text("Password") },
            visualTransformation = PasswordVisualTransformation(),
            modifier = Modifier.fillMaxWidth()
        )

        Spacer(Modifier.height(12.dp))

        if (state.buttons.changePasswordTextVisible) {
            TextButton(onClick = { onEvent(LoginUserUiEvent.OnChangePasswordClicked) }) {
                Text("Change Password")
            }
        }

        Spacer(Modifier.height(24.dp))

        Row(horizontalArrangement = Arrangement.spacedBy(12.dp)) {
            CustomButton(
                style = state.buttons.cancel.style,
                onClick = { onEvent(LoginUserUiEvent.OnBackClicked) },
                text = "Cancel",
                enabled = state.buttons.cancel.enabled,
                backgroundColorRes = state.buttons.cancel.backgroundColorRes
            )

            CustomButton(
                style = state.buttons.accept.style,
                onClick = { onEvent(LoginUserUiEvent.OnAcceptClicked) },
                text = "Accept",
                enabled = state.buttons.accept.enabled,
                backgroundColorRes = state.buttons.accept.backgroundColorRes,
                loading = state.buttons.accept.loading
            )
        }

        if (state.buttons.progressBarVisible) {
            Spacer(Modifier.height(16.dp))
            LinearProgressIndicator(
                progress = { state.buttons.progressBarProgress / 100f },
                modifier = Modifier.fillMaxWidth()
            )
        }
    }
}

@Composable
private fun LoginUserDialogHost(
    dialogState: LoginUserDialogState?,
    onDismiss: () -> Unit,
    onAccept: () -> Unit,
    onPinChange: (String) -> Unit
) {
    if (dialogState?.show == true) {
        AlertDialog(
            onDismissRequest = onDismiss,
            title = { Text(stringResource(dialogState.titleRes)) },
            text = {
                Column {
                    Text(stringResource(dialogState.descriptionRes))
                    Spacer(Modifier.height(12.dp))
                    OutlinedTextField(
                        value = dialogState.pin,
                        onValueChange = { if (it.length <= dialogState.maxLength) onPinChange(it) },
                        label = { Text("PIN") },
                        visualTransformation = PasswordVisualTransformation(),
                        singleLine = true
                    )
                }
            },
            confirmButton = {
                TextButton(onClick = onAccept) { Text("Accept") }
            },
            dismissButton = {
                TextButton(onClick = onDismiss) { Text("Cancel") }
            }
        )
    }
}

// The signature below is intentionally aligned to the usage above.

@Composable
private fun CustomButton(
    style: ButtonStyle,
    onClick: () -> Unit,
    text: String,
    enabled: Boolean,
    @ColorRes backgroundColorRes: Int,
    loading: Boolean = false
) {
    val containerColor = androidx.compose.ui.res.colorResource(id = backgroundColorRes)

    Button(
        onClick = onClick,
        enabled = enabled && !loading,
        colors = ButtonDefaults.buttonColors(containerColor = containerColor)
    ) {
        if (loading) {
            CircularProgressIndicator(
                modifier = Modifier.size(18.dp),
                strokeWidth = 2.dp
            )
        } else {
            Text(text)
        }
    }
}
```

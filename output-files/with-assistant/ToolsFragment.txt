## 3) Compose-friendly `ToolsViewModel`

This version exposes:
- `StateFlow<ToolsUiState>`
- `SharedFlow<ToolsUiEffect>`
- `onEvent(...)`

It preserves the original behavior:
- loads meeting sign colors from preferences
- emits navigation effects instead of directly navigating

```kotlin
package ifac.td.taxi.ui.screen.tools

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.R
import ifac.td.taxi.domain.usecase.UserPreferencesUseCase
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch

class ToolsViewModel(
    private val userPreferencesUseCase: UserPreferencesUseCase,
    application: Application
) : AndroidViewModel(application) {

    private val _uiState = MutableStateFlow(ToolsUiState())
    val uiState: StateFlow<ToolsUiState> = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<ToolsUiEffect>(extraBufferCapacity = 1)
    val uiEffect: SharedFlow<ToolsUiEffect> = _uiEffect.asSharedFlow()

    init {
        viewModelScope.launch {
            val prefs = userPreferencesUseCase.getUserPreferences()
            val colors = if (prefs != null) {
                prefs.meetingSignTextColor to prefs.meetingSignBackgroundColor
            } else {
                0 to 0
            }

            _uiState.value = _uiState.value.copy(
                meetingSignColors = colors,
                buttonsState = ToolsButtonsState.reduce(colors)
            )
        }
    }

    fun onEvent(event: ToolsUiEvent) {
        when (event) {
            ToolsUiEvent.ScreenStarted -> {
                // no-op for now
            }

            ToolsUiEvent.RequirementsClicked -> {
                viewModelScope.launch {
                    _uiEffect.emit(
                        ToolsUiEffect.NavigateToRequirements(
                            R.id.action_toolsFragment_to_requirementsFragment
                        )
                    )
                }
            }

            ToolsUiEvent.MeetingSignClicked -> {
                val colors = uiState.value.meetingSignColors
                viewModelScope.launch {
                    _uiEffect.emit(
                        ToolsUiEffect.NavigateToMeetingSign(
                            textColor = colors.first,
                            backgroundColor = colors.second
                        )
                    )
                }
            }
        }
    }
}
```

## 4) `ToolsMainActivityViewModel` Compose bridge

This is the bridge that preserves the fragment-style navigation flow.  
Your Compose screen will collect `ToolsUiEffect` and forward it to the activity navigation layer.

```kotlin
package ifac.td.taxi.ui.screen.tools

import androidx.lifecycle.ViewModel
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.asSharedFlow

sealed interface ToolsMainActivityEffect {
    data class NavigateTo(val destinationId: Int) : ToolsMainActivityEffect
    data class NavigateToMeetingSign(
        val textColor: Int,
        val backgroundColor: Int
    ) : ToolsMainActivityEffect
    data object ShowHeader : ToolsMainActivityEffect
}

class ToolsMainActivityViewModel : ViewModel() {

    private val _effects = MutableSharedFlow<ToolsMainActivityEffect>(extraBufferCapacity = 1)
    val effects: SharedFlow<ToolsMainActivityEffect> = _effects.asSharedFlow()

    fun showHeader() {
        _effects.tryEmit(ToolsMainActivityEffect.ShowHeader)
    }

    fun navigateTo(destinationId: Int) {
        _effects.tryEmit(ToolsMainActivityEffect.NavigateTo(destinationId))
    }

    fun navigateToMeetingSign(textColor: Int, backgroundColor: Int) {
        _effects.tryEmit(
            ToolsMainActivityEffect.NavigateToMeetingSign(
                textColor = textColor,
                backgroundColor = backgroundColor
            )
        )
    }
```

```kotlin
import androidx.compose.ui.unit.dp
import androidx.compose.ui.unit.sp
import ifac.td.taxi.R

@Composable
fun ToolsCustomButton(
    state: ToolsButtonVisualState,
    onClick: () -> Unit
) {
    if (!state.visible) return

    val bgColor = when (state.background) {
        ToolsButtonBackground.GREEN -> Color(0xFF2E7D32)
        ToolsButtonBackground.RED -> Color(0xFFC62828)
        ToolsButtonBackground.BLUE -> Color(0xFF1565C0)
        ToolsButtonBackground.YELLOW -> Color(0xFFF9A825)

        ToolsButtonBackground.TRANSPARENT -> Color.Transparent
    }

    Row(
        modifier = Modifier
            .weight(1f)
            .background(bgColor, RoundedCornerShape(4.dp))
            .clickable(enabled = state.enabled) { onClick() }
            .padding(12.dp),
        verticalAlignment = Alignment.CenterVertically,
        horizontalArrangement = Arrangement.Center
    ) {
        if (state.iconRes != 0) {
            Icon(
                painter = painterResource(id = state.iconRes),
                contentDescription = null,
                tint = Color.White,
                modifier = Modifier.size(20.dp)
            )
        }

        if (state.text.isNotBlank()) {
            Text(
                text = state.text,
                color = Color.White,
                fontSize = 14.sp,
                fontWeight = FontWeight.Bold,
                modifier = Modifier.padding(start = 8.dp)
            )
        }
    }
}
```

```kotlin
import androidx.navigation.NavController
import ifac.td.taxi.HomeDirections
import ifac.td.taxi.R
import kotlinx.coroutines.flow.collectLatest

@Composable
fun InfoDispatchScreen(
    navController: NavController,
    viewModel: InfoDispatchComposeViewModel
) {
    val state by viewModel.uiState.collectAsStateWithLifecycle()
    var dialogState by remember { mutableStateOf<InfoDispatchDialogState?>(null) }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                InfoDispatchUiEffect.NavigateToDirections -> {
                    navController.navigate("directions")
                }

                InfoDispatchUiEffect.NavigateToHome -> {
                    navController.navigate(HomeDirections.goToOnTripFragment().actionId)
                }

                is InfoDispatchUiEffect.NavigateToMeetingSign -> {
                    navController.navigate(
                        "meeting_sign/${effect.textColor}/${effect.backgroundColor}"
                    )
                }

                InfoDispatchUiEffect.RequestPhonePermission -> {
                    // Host-side permission launcher
                }

                is InfoDispatchUiEffect.ShowToast -> {
                    // Host-side toast/snackbar
                }

                is InfoDispatchUiEffect.OpenExternalPhoneCall -> {
                    // Host-side phone intent
                }

                is InfoDispatchUiEffect.OpenDialog -> {
                    dialogState = effect.dialog
                }

                InfoDispatchUiEffect.CloseDialog -> {
                    dialogState = null
                }
            }
        }
    }

    Scaffold(
        topBar = {
            // Preserve header behavior here if needed
        },
        bottomBar = {
            // Preserve bottom bar behavior here if needed
        }
    ) { padding ->
        InfoDispatchContent(
            modifier = Modifier.padding(padding),
            state = state,
            onEvent = viewModel::onEvent
        )
    }

    dialogState?.let { dialog ->
        InfoDispatchCustomDialog(
            state = dialog,
            onDismiss = { dialogState = null },
            onButtonClick = { button ->
                viewModel.onEvent(InfoDispatchUiEvent.OnDialogButtonClicked(button))
                dialogState = null
            }
        )
    }
}
```

```kotlin
@Composable
private fun DispatchButton(
    text: String,
    visible: Boolean,
    enabled: Boolean,
    backgroundColor: Color,
    onClick: () -> Unit
) {
    if (!visible) return

    val contentColor = if (enabled) Color.White else Color.White.copy(alpha = 0.6f)

    Box(
        modifier = Modifier
            .fillMaxWidth()
            .height(52.dp)
            .background(backgroundColor, RoundedCornerShape(12.dp))
            .clickable(enabled = enabled) { onClick() },
        contentAlignment = Alignment.Center
    ) {
        Text(text = text, color = contentColor)
    }
}
```

```kotlin
package ifac.td.taxi.ui.screen.components

import androidx.compose.foundation.background
import androidx.compose.foundation.clickable
import androidx.compose.foundation.horizontalScroll
import androidx.compose.foundation.layout.*
import androidx.compose.foundation.rememberScrollState
import androidx.compose.foundation.shape.RoundedCornerShape
import androidx.compose.material3.*
import androidx.compose.runtime.Composable
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.unit.dp

@Composable
fun MultiDispatchTabs(
    tabs: List<InfoDispatchMultiDispatchTabState>,
    selectedIndex: Int,
    onTabSelected: (Int) -> Unit
) {
    val scrollState = rememberScrollState()

    Column {
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .horizontalScroll(scrollState),
            horizontalArrangement = Arrangement.spacedBy(8.dp)
        ) {
            tabs.forEachIndexed { index, tab ->
                val selected = index == selectedIndex
                Column(
                    modifier = Modifier
                        .width(120.dp)
                        .background(
                            if (selected) Color(0xFF1976D2) else Color(0xFFE0E0E0),
                            RoundedCornerShape(12.dp)
                        )
                        .clickable { onTabSelected(index) }
                        .padding(12.dp),
                    horizontalAlignment = Alignment.CenterHorizontally
                ) {
                    Text(text = tab.name, color = if (selected) Color.White else Color.Black)
                    Spacer(Modifier.height(4.dp))
                    Text(text = tab.iconLabel, color = if (selected) Color.White else Color.DarkGray)
                }
            }
        }

        Box(
            modifier = Modifier
                .padding(top = 6.dp)
                .height(3.dp)
                .fillMaxWidth()
                .background(Color.Transparent)
        ) {
            if (tabs.isNotEmpty() && selectedIndex in tabs.indices) {
                Box(
                    modifier = Modifier
                        .fillMaxWidth()
                        .height(3.dp)
                        .background(Color(0xFF1976D2))
                )
            }
        }
    }
}
```

# 7) Compose ViewModel

This ViewModel exposes a single `uiState` and a single `uiEffect` stream, and it mirrors the fragment logic.

```kotlin
package ifac.td.taxi.ui.screen.components

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import com.interfacom.sdk.taximeter.bravocomm.ifConstants
import ifac.td.taxi.R
import ifac.td.taxi.domain.usecase.*
import ifac.td.taxi.framework.sdk.bravocentral.usecase.BravoCentralUseCase
import ifac.td.taxi.framework.sdk.usecase.LicensingUseCase
import ifac.td.taxi.framework.sdk.usecase.PhoneCallUseCase
import ifac.td.taxi.framework.sdk.usecase.PhoneCallUseCaseImpl
import ifac.td.taxi.repository.room.dao.BravoConfigurationVariableDao
import ifac.td.taxi.repository.room.entities.DispatchExtraDataEntity
import ifac.td.taxi.viewmodel.model.InfoDispatchModel
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch

class InfoDispatchComposeViewModel(
    private val dispatchUseCase: DispatchUseCase,
    private val bluetoothLocalUseCase: BluetoothLocalUseCase,
    private val bravoCentralUseCase: BravoCentralUseCase,
    private val phoneCallUseCase: PhoneCallUseCase,
    private val dispatchNotificationUseCase: DispatchNotificationUseCase,
    private val bridgeCallUseCase: BridgeCallUseCase,
    private val licensingUseCase: LicensingUseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val ttsUseCase: TTSUseCase,
    private val bravoConfigurationVariableDao: BravoConfigurationVariableDao,
    private val countdownManagerUseCase: CountdownManagerUseCase,
    application: Application
) : AndroidViewModel(application) {

    private val _uiState = MutableStateFlow(InfoDispatchUiState())
    val uiState: StateFlow<InfoDispatchUiState> = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<InfoDispatchUiEffect>()
    val uiEffect: SharedFlow<InfoDispatchUiEffect> = _uiEffect.asSharedFlow()

    private var currentDispatch: InfoDispatchModel? = null
    private var extraData: DispatchExtraDataEntity? = null
    private var bridgeCallState: Int = 0
    private var customerCallAvailable: Boolean = true
    private var canMakeCalls: Boolean = false
    private var isManualShift: Boolean = false
    private var isITopConnected: Boolean = false
    private var noClientAllowed: Boolean? = null

    fun bindDispatch(
        dispatch: InfoDispatchModel,
        multiDispatch: List<InfoDispatchModel> = emptyList(),
        selectedIndex: Int = 0
    ) {
        currentDispatch = dispatch
        updateDispatchUi(dispatch, multiDispatch, selectedIndex)
        checkNoClientButton()
        getMaxBridgeCalls()
        retrieveExtraData(dispatch.id)
        checkTTSOnDispatch(dispatch)
    }

    fun onEvent(event: InfoDispatchUiEvent) {
        when (event) {
            InfoDispatchUiEvent.OnNotificationsClicked -> handleNotificationsClick()
            InfoDispatchUiEvent.OnNavigateClicked -> {
                stopTTS()
                viewModelScope.launch { _uiEffect.emit(InfoDispatchUiEffect.NavigateToDirections) }
            }
            InfoDispatchUiEvent.OnVoiceCallClicked -> handleVoiceCallClick()
            InfoDispatchUiEvent.OnPrintClicked -> currentDispatch?.let { printInfoDispatch(it) }
            InfoDispatchUiEvent.OnNoClientClicked -> handleNoClientClick()
            InfoDispatchUiEvent.OnReturnClicked -> handleReturnClick()
            InfoDispatchUiEvent.OnDispatchNumberClicked -> emitMeetingSignNavigation()
            is InfoDispatchUiEvent.OnMultiDispatchTabSelected -> handleTabSelected(event.index)
            is InfoDispatchUiEvent.OnDialogButtonClicked -> handleDialogButton(event.button)
        }
    }

    private fun updateDispatchUi(
        dispatch: InfoDispatchModel,
        multiDispatch: List<InfoDispatchModel>,
        selectedIndex: Int
    ) {
        val buttons = buildButtonsState(dispatch)
        _uiState.update {
            it.copy(
                dispatchId = dispatch.id,
                dispatchNumberText = formatDispatchNumber(
                    if (!dispatch.hasExtraData || !isCurrentBluetoothITop()) {
                        buildDispatchIdentification(dispatch)
                    } else {
                        buildDispatchIdentification(dispatch)
                    }
                ).toString(),
                zoneTextVisible = dispatch.pickUpZone != null,
                zoneText = dispatch.pickUpZone.orEmpty(),
                pickupTimeVisible = dispatch.pickUpTime != null,
                pickupTimeText = dispatch.pickUpTime?.let {
                    if (!dispatch.etaTime.isNullOrBlank()) "$it (${dispatch.etaTime})" else it
                }.orEmpty(),
                pickupAddressVisible = dispatch.pickUpAdress != null,
                pickupAddress = dispatch.pickUpAdress.orEmpty(),
                destinationsVisible = !dispatch.destinyAdress.isNullOrEmpty(),
                destinations = dispatch.destinyAdress.orEmpty(),
                observationsVisible = !dispatch.observations.isNullOrEmpty(),
                observations = dispatch.observations.orEmpty().filter { it.isNotBlank() },
                flightCodeVisible = dispatch.flightCode.isNotBlank(),
                flightCode = dispatch.flightCode,
                requirementsVisible = false,
                requirements = "",
                amountVisible = dispatch.amount != null && dispatch.amount != 0,
                amountText = dispatch.amount?.let { formatAmount(it) }.orEmpty(),
                tollsVisible = false,
                tollsText = "",
                totalVisible = false,
                totalText = "",
                subscriberVisible = (dispatch.clientType ?: 0) > 0,
                subscriberTitle = if (!isCurrentBluetoothITop()) {
                    listOfNotNull(dispatch.subscriber, dispatch.dispatchName).joinToString(" ")
                } else {
                    dispatch.dispatchName.orEmpty()
                },
                subscriberUserName = dispatch.subscriberUserName.orEmpty(),
                subscriberAuthVisible = !dispatch.autorizacion.isNullOrBlank() &&
                    (dispatch.clientType ?: 0) > 0 && !isCurrentBluetoothITop(),
                subscriberAuthorization = dispatch.autorizacion.orEmpty(),
                serviceTypeVisible = dispatch.isConcertedPrice == true || dispatch.isPaymentByApp == true,
                serviceTypeText = when {
                    dispatch.isConcertedPrice == true -> getApplication<Application>().getString(R.string.concerted_trip_with_fixed_price)
                    dispatch.isPaymentByApp == true -> getApplication<Application>().getString(R.string.payment_by_app)
                    else -> ""
                },
                concertedAmountVisible = dispatch.isConcertedPrice == true,
                concertedAmountText = "",
                multiDispatch = InfoDispatchMultiDispatchState(
                    visible = multiDispatch.size > 1,
                    selectedIndex = selectedIndex,
                    tabs = multiDispatch.map { item ->
                        InfoDispatchMultiDispatchTabState(
                            name = item.dispatchName.orEmpty(),
                            iconLabel = when {
                                item.isCancelled -> "Cancelled"
                                item.isEnded -> "Ended"
                                item.isRiderInCabNotificationSent == true -> "Rider"
                                item.isAtDoorNotificationSent == true -> "Door"
                                else -> "People"
                            },
                            dispatchNumber = item.longDispatchNumber.orEmpty(),
                            routeId = item.routeId.orEmpty()
                        )
                    }
                ),
                buttons = buttons
            )
        }
    }

    private fun buildButtonsState(dispatch: InfoDispatchModel): InfoDispatchButtonsState {
        val notificationsVisible = true
        val notificationsEnabled = !isITopConnected || isManualShift

        val hasAtDoorAction = dispatch.isAtDoorNotificationEnabled() && dispatch.isAtDoorNotificationSent == false
        val hasRiderInCabAction = (dispatch.riderInCab ?: false) && dispatch.isRiderInCabNotificationSent == false

        val notificationsButton = when {
            hasAtDoorAction && hasRiderInCabAction -> InfoDispatchButtonState(
                text = "Avisos",
                visible = notificationsVisible,
                enabled = notificationsEnabled,
                backgroundColor = if (notificationsEnabled) androidx.compose.ui.graphics.Color(0xFF4CAF50) else androidx.compose.ui.graphics.Color(0xFF9E9E9E)
            )
            !hasAtDoorAction && hasRiderInCabAction -> InfoDispatchButtonState(
                text = "Rider in cab",
                visible = notificationsVisible,
                enabled = notificationsEnabled,
                backgroundColor = if (notificationsEnabled) androidx.compose.ui.graphics.Color(0xFF4CAF50) else androidx.compose.ui.graphics.Color(0xFF9E9E9E)
            )
            hasAtDoorAction && !hasRiderInCabAction -> InfoDispatchButtonState(
                text = "At door",
                visible = notificationsVisible,
                enabled = notificationsEnabled,
                backgroundColor = if (notificationsEnabled) androidx.compose.ui.graphics.Color(0xFF4CAF50) else androidx.compose.ui.graphics.Color(0xFF9E9E9E)
```

## Exact reducer

This reducer mirrors the fragment branches one-by-one.

```kotlin
package ifac.td.taxi.ui.screen.components

import androidx.compose.ui.graphics.Color

object InfoDispatchButtonsReducer {

    fun reduce(
        current: InfoDispatchButtonsState,
        input: InfoDispatchButtonsReducerInput
    ): InfoDispatchButtonsState {
        val isITopConnected = input.hasTaximeterConnection
        val isManual = input.shiftIsManual
        val dispatch = input.dispatch

        // --- Notifications button ---
        // Fragment:
        // if (hasAtDoorAction && hasRiderInCabAction) => AVISOS + ENABLE
        // else if (!hasAtDoorAction && hasRiderInCabAction) => RIDER_IN_CAB + ENABLE
        // else if (hasAtDoorAction && !sent rider) => AT_DOOR + ENABLE
        // else DISABLE
        // then if iTop && !manual => DISABLE
        val hasAtDoorAction = dispatch?.isAtDoorNotificationEnabled() == true &&
            dispatch.isAtDoorNotificationSent == false
        val hasRiderInCabAction = (dispatch?.riderInCab ?: false) &&
            dispatch?.isRiderInCabNotificationSent == false

        val notificationsState = when {
            hasAtDoorAction && hasRiderInCabAction -> InfoDispatchButtonState(
                visible = true,
                enabled = !(isITopConnected && !isManual),
                style = if (isITopConnected && !isManual) ButtonVisualStyle.Disabled else ButtonVisualStyle.Enabled,
                text = "AVISOS"
            )
            !hasAtDoorAction && hasRiderInCabAction -> InfoDispatchButtonState(
                visible = true,
                enabled = !(isITopConnected && !isManual),
                style = if (isITopConnected && !isManual) ButtonVisualStyle.Disabled else ButtonVisualStyle.Enabled,
                text = "RIDER_IN_CAB"
            )
            hasAtDoorAction && !hasRiderInCabAction -> InfoDispatchButtonState(
                visible = true,
                enabled = !(isITopConnected && !isManual),
                style = if (isITopConnected && !isManual) ButtonVisualStyle.Disabled else ButtonVisualStyle.Enabled,
                text = "AT_DOOR"
            )
            else -> InfoDispatchButtonState(
                visible = true,
                enabled = false,
                style = ButtonVisualStyle.Disabled,
                text = "NOTIFICATIONS"
            )
        }

        // --- Voice call button ---
        // Fragment:
        // if bridgeCallState == 2 -> toast call_starting, red, enable
        // if bridgeCallState == 0 -> green
        // else -> toast wait_central_confirm, orange, enable
        // plus:
        // if !customerCallAvailable => disable
        // else if canMakeCalls || bridgeCallState != 0 => enable
        // else disable
        val voiceStyle = when (input.bridgeCallState) {
            2 -> ButtonVisualStyle.Red
            0 -> ButtonVisualStyle.Enabled
            else -> ButtonVisualStyle.Orange
        }

        val voiceEnabledByAvailability =
            if (!input.customerCallAvailable) {
                false
            } else {
                input.canMakeCalls || input.bridgeCallState != 0
            }

        val voiceState = InfoDispatchButtonState(
            visible = true,
            enabled = voiceEnabledByAvailability,
            style = when {
                !input.customerCallAvailable -> ButtonVisualStyle.Disabled
                input.bridgeCallState == 2 -> ButtonVisualStyle.Red
                input.bridgeCallState == 0 -> ButtonVisualStyle.Enabled
                else -> ButtonVisualStyle.Orange
            },
            text = "CALL"
        )

        // --- No client button ---
        // Fragment:
        // if iTop => disable
        // else if clientButton false => red background
        // else green background
        // then if noClientButtonEnabled false => disable else enable
        val noClientStyle = when {
            isITopConnected -> ButtonVisualStyle.Disabled
            input.bravoState.clientButton.not() -> ButtonVisualStyle.Red
            else -> ButtonVisualStyle.Enabled
        }

        val noClientEnabled = if (isITopConnected) {
            false
        } else {
            input.bravoState.noClientButtonEnabled
        }

        val noClientState = InfoDispatchButtonState(
            visible = true,
            enabled = noClientEnabled,
            style = if (noClientEnabled) noClientStyle else ButtonVisualStyle.Disabled,
            text = "NO_CLIENT"
        )

        // Fragment:
        // if noClientButtonFlow == false && currentStatus == STATE_HIRED_DISPATCHED => disable
        val noClientStateAfterFlow = if (
            input.noClientButtonFlowValue == false &&
            input.currentShiftStatus == ifConstants.STATE_HIRED_DISPATCHED
        ) {
            noClientState.copy(
                enabled = false,
                style = ButtonVisualStyle.Disabled
            )
        } else {
            noClientState
        }

        // --- Return button ---
        // Fragment:
        // if iTop => disable
        // else if showReturnDispatchButton true => enable + setReturnOnClick()
        // else disable
        val returnState = if (isITopConnected) {
            InfoDispatchButtonState(
                visible = true,
                enabled = false,
                style = ButtonVisualStyle.Disabled,
                text = "RETURN"
            )
        } else if (input.bravoState.showReturnDispatchButton) {
            InfoDispatchButtonState(
                visible = true,
                enabled = true,
                style = ButtonVisualStyle.Red,
                text = "RETURN"
            )
        } else {
            InfoDispatchButtonState(
                visible = true,
                enabled = false,
                style = ButtonVisualStyle.Disabled,
                text = "RETURN"
            )
        }

        return current.copy(
            notifications = notificationsState,
            voiceCall = voiceState,
            noClient = noClientStateAfterFlow,
            returnDispatch = returnState
        )
    }
}
```

# 4) How to wire the shared flows exactly like the fragment

In the fragment, the shared VM was collecting:

- `dispatchFlow`
- `multiDispatchFlow`
- `hasTaximeterConnectionFlow`
- `shiftStatusFlow`
- `bravoStateFlow`
- `customerCallAvailable`
- `customerCallButtonStateAvailable`
- `canMakeCallsFlow`
- `phoneCallErrorFlow`

In Compose, you should map those into the bridge VM like this:

```kotlin
LaunchedEffect(Unit) {
    sharedVm.dispatchFlow.collectLatest { dispatch ->
        sharedViewModel.onDispatchChanged(dispatch)
        dispatch?.let { viewModel.bindDispatch(it, sharedState.multiDispatch, 0) }

}

LaunchedEffect(Unit) {
    sharedVm.multiDispatchFlow.collectLatest { list ->
        sharedViewModel.onMultiDispatchChanged(list ?: emptyList())
        sharedState.dispatch?.let { dispatch ->

        sharedViewModel.onShiftStatusChanged(shift?.currentStatus)
        sharedViewModel.onShiftManualChanged(shift?.isManual == true)

LaunchedEffect(Unit) {

    sharedVm.customerCallAvailable.collectLatest { available ->
        sharedViewModel.onCustomerCallAvailableChanged(available)
    }
}

LaunchedEffect(Unit) {
    sharedVm.customerCallButtonStateAvailable.collectLatest { available ->
        sharedViewModel.onCustomerCallButtonAvailableChanged(available)

LaunchedEffect(Unit) {
    combine(
        sharedVm.canMakeCallsFlow,
        sharedVm.bravoStateFlow
    ) { canMakeCalls, bravoState ->
        canMakeCalls to bravoState.bridgeCallState
```

# 5) Recommended final shape

If you want the cleanest Compose architecture, use:

- `InfoDispatchMainActivityViewModel`
  - owns shared activity-level state
  - mirrors the old `MainActivityViewModel`
- `InfoDispatchComposeViewModel`
  - owns screen-specific state and actions
- `InfoDispatchButtonsReducer`
  - pure reducer for button state
- `InfoDispatchScreen(...)`
  - collects both VMs and renders UI

```kotlin
class InfoDispatchComposeViewModel : ViewModel() {

    private val _uiState = MutableStateFlow(InfoDispatchUiState())

    private val _uiEffect = MutableSharedFlow<InfoDispatchUiEffect>()

    fun bindDispatch(
        dispatch: InfoDispatchModel?,
        multiDispatch: List<InfoDispatchModel> = emptyList(),
        selectedIndex: Int = 0
    ) {
        _uiState.update { current ->
            current.copy(
                dispatch = dispatch,
                dispatchNumberText = dispatch?.let { buildDispatchIdentification(it) }.orEmpty(),
                zoneVisible = !dispatch?.pickUpZone.isNullOrBlank(),
                zoneText = dispatch?.pickUpZone.orEmpty(),
                pickupTimeVisible = !dispatch?.pickUpTime.isNullOrBlank(),
                pickupTimeText = dispatch?.pickUpTime.orEmpty(),
                pickupAddressVisible = !dispatch?.pickUpAdress.isNullOrBlank(),
                pickupAddressText = dispatch?.pickUpAdress.orEmpty(),
                destinationsVisible = !dispatch?.destinyAdress.isNullOrEmpty(),
                destinations = dispatch?.destinyAdress.orEmpty(),
                observationsVisible = !dispatch?.observations.isNullOrEmpty(),
                observations = dispatch?.observations.orEmpty().filter { it.isNotBlank() },
                flightCodeVisible = !dispatch?.flightCode.isNullOrBlank(),
                flightCodeText = dispatch?.flightCode.orEmpty(),
                requirementsVisible = !dispatch?.observations.isNullOrEmpty(),
                requirementsText = dispatch?.observations.orEmpty().joinToString("\n"),
                amountVisible = (dispatch?.amount ?: 0) != 0,
                amountText = dispatch?.amount?.toString().orEmpty(),
                subscriberVisible = (dispatch?.clientType ?: 0) > 0,
                subscriberTitle = if (dispatch?.dispatchName.isNullOrBlank()) dispatch?.idDispatch.orEmpty()
                else listOfNotNull(dispatch?.subscriber, dispatch?.dispatchName).joinToString(" "),
                subscriberUserName = dispatch?.subscriberUserName.orEmpty(),
                subscriberAuthVisible = !dispatch?.autorizacion.isNullOrBlank() && (dispatch?.clientType ?: 0) > 0,
                subscriberAuthorization = dispatch?.autorizacion.orEmpty(),
                serviceTypeVisible = dispatch?.isConcertedPrice == true || dispatch?.isPaymentByApp == true,
                serviceTypeText = when {
                    dispatch?.isConcertedPrice == true -> "Concerted trip with fixed price"
                    dispatch?.isPaymentByApp == true -> "Payment by app"
                    else -> ""
                },
                concertedAmountVisible = dispatch?.isConcertedPrice == true,
                concertedAmountText = "",
                multiDispatch = InfoDispatchMultiDispatchState(
                    visible = multiDispatch.size > 1,
                    selectedIndex = selectedIndex,
                    tabs = multiDispatch.map { item ->
                        InfoDispatchMultiDispatchTabState(
                            name = item.dispatchName.orEmpty(),
                            iconLabel = when {
                                item.isCancelled -> "Cancelled"
                                item.isEnded -> "Ended"
                                item.isRiderInCabNotificationSent == true -> "Rider"
                                item.isAtDoorNotificationSent == true -> "Door"
                                else -> "People"
                            },
                            dispatchNumber = item.longDispatchNumber.orEmpty(),
                            routeId = item.routeId.orEmpty()
                        )
                    }
                )
            )
        }
    }

    fun onEvent(event: InfoDispatchUiEvent) {
        when (event) {
            InfoDispatchUiEvent.OnNotificationsClicked -> {
                viewModelScope.launch {
                    _uiEffect.emit(
                        InfoDispatchUiEffect.OpenDialog(
                            InfoDispatchDialogState(
                                title = "Select notification",
                                description = "",
                                buttons = listOf(
                                    InfoDispatchDialogButton(ButtonType.AT_DOOR, "At door"),
                                    InfoDispatchDialogButton(ButtonType.RIDER_IN_CAB, "Rider in cab")
                                )
                            )
                        )
                    )
                }
            }

            InfoDispatchUiEvent.OnNavigateClicked -> {
                viewModelScope.launch { _uiEffect.emit(InfoDispatchUiEffect.NavigateToDirections) }
            }

            InfoDispatchUiEvent.OnVoiceCallClicked -> {
                viewModelScope.launch { _uiEffect.emit(InfoDispatchUiEffect.RequestPhonePermission) }
            }

            InfoDispatchUiEvent.OnPrintClicked -> Unit

            InfoDispatchUiEvent.OnNoClientClicked -> {
                viewModelScope.launch {
                    _uiEffect.emit(
                        InfoDispatchUiEffect.OpenDialog(
                            InfoDispatchDialogState(
                                title = "Confirm no client",
                                description = "",
                                buttons = listOf(
                                    InfoDispatchDialogButton(ButtonType.CANCEL, "Cancel"),
                                    InfoDispatchDialogButton(ButtonType.ACCEPT, "Accept")
                                )
                            )
                        )
                    )
                }
            }

            InfoDispatchUiEvent.OnReturnClicked -> {
                viewModelScope.launch {
                    _uiEffect.emit(
                        InfoDispatchUiEffect.OpenDialog(
                            InfoDispatchDialogState(
                                title = "Return dispatch",
                                description = "",
                                buttons = listOf(
                                    InfoDispatchDialogButton(ButtonType.CANCEL, "Cancel"),
                                    InfoDispatchDialogButton(ButtonType.ACCEPT, "Accept")
                                )
                            )
                        )
                    )
                }
            }

            InfoDispatchUiEvent.OnDispatchNumberClicked -> {
                viewModelScope.launch {
                    _uiEffect.emit(InfoDispatchUiEffect.NavigateToMeetingSign(0, 0))
                }
            }

            is InfoDispatchUiEvent.OnMultiDispatchTabSelected -> {
                _uiState.update { it.copy(multiDispatch = it.multiDispatch.copy(selectedIndex = event.index)) }
            }

            is InfoDispatchUiEvent.OnDialogButtonClicked -> {
                when (event.button.type) {
                    ButtonType.CANCEL -> viewModelScope.launch { _uiEffect.emit(InfoDispatchUiEffect.CloseDialog) }
                    ButtonType.ACCEPT -> Unit
                    ButtonType.AT_DOOR -> Unit
                    ButtonType.RIDER_IN_CAB -> Unit
                }
            }
        }
    }

    private fun buildDispatchIdentification(dispatch: InfoDispatchModel): String {
        val name = dispatch.dispatchName.orEmpty()
        return if (name.isNotBlank()) {
            "$name - ${dispatch.longDispatchNumber.orEmpty()}"
        } else {
            dispatch.idDispatch.orEmpty()
        }
    }
}

                    // host-side toast/snackbar

                    // host-side phone intent
                }

                is InfoDispatchUiEffect.OpenDialog -> {
                    dialogState = effect.dialog

                InfoDispatchUiEffect.CloseDialog -> {
                    dialogState = null
                }
            }
        }

                InfoDispatchSharedEffect.ShowHeader -> Unit

                is InfoDispatchSharedEffect.OpenDialog -> dialogState = effect.dialog
                InfoDispatchSharedEffect.CloseDialog -> dialogState = null
```

```kotlin
class InfoDispatchComposeViewModel : ViewModel() {

    fun bindDispatch(
        dispatch: InfoDispatchModel?,
        multiDispatch: List<InfoDispatchModel> = emptyList(),
        selectedIndex: Int = 0
    ) {
        _uiState.update { current ->
            current.copy(
                dispatch = dispatch,
                dispatchNumberText = dispatch?.let { buildDispatchIdentification(it) }.orEmpty(),
                zoneVisible = !dispatch?.pickUpZone.isNullOrBlank(),
                zoneText = dispatch?.pickUpZone.orEmpty(),
                pickupTimeVisible = !dispatch?.pickUpTime.isNullOrBlank(),
                pickupTimeText = dispatch?.pickUpTime.orEmpty(),
                pickupAddressVisible = !dispatch?.pickUpAdress.isNullOrBlank(),
                pickupAddressText = dispatch?.pickUpAdress.orEmpty(),
                destinationsVisible = !dispatch?.destinyAdress.isNullOrEmpty(),
                destinations = dispatch?.destinyAdress.orEmpty(),
                observationsVisible = !dispatch?.observations.isNullOrEmpty(),
                observations = dispatch?.observations.orEmpty().filter { it.isNotBlank() },
                flightCodeVisible = !dispatch?.flightCode.isNullOrBlank(),
                flightCodeText = dispatch?.flightCode.orEmpty(),
                requirementsVisible = !dispatch?.observations.isNullOrEmpty(),
                requirementsText = dispatch?.observations.orEmpty().joinToString("\n"),
                amountVisible = (dispatch?.amount ?: 0) != 0,
                amountText = dispatch?.amount?.toString().orEmpty(),
                subscriberVisible = (dispatch?.clientType ?: 0) > 0,
                subscriberTitle = if (dispatch?.dispatchName.isNullOrBlank()) dispatch?.idDispatch.orEmpty()
                else listOfNotNull(dispatch?.subscriber, dispatch?.dispatchName).joinToString(" "),
                subscriberUserName = dispatch?.subscriberUserName.orEmpty(),
                subscriberAuthVisible = !dispatch?.autorizacion.isNullOrBlank() && (dispatch?.clientType ?: 0) > 0,
                subscriberAuthorization = dispatch?.autorizacion.orEmpty(),
                serviceTypeVisible = dispatch?.isConcertedPrice == true || dispatch?.isPaymentByApp == true,
                serviceTypeText = when {
                    dispatch?.isConcertedPrice == true -> "Concerted trip with fixed price"
                    dispatch?.isPaymentByApp == true -> "Payment by app"
                    else -> ""
                },
                concertedAmountVisible = dispatch?.isConcertedPrice == true,
                concertedAmountText = "",
                multiDispatch = InfoDispatchMultiDispatchState(
                    visible = multiDispatch.size > 1,
                    selectedIndex = selectedIndex,
                    tabs = multiDispatch.map { item ->
                        InfoDispatchMultiDispatchTabState(
                            name = item.dispatchName.orEmpty(),
                            iconLabel = when {
                                item.isCancelled -> "Cancelled"
                                item.isEnded -> "Ended"
                                item.isRiderInCabNotificationSent == true -> "Rider"
                                item.isAtDoorNotificationSent == true -> "Door"
                                else -> "People"
                            },
                            dispatchNumber = item.longDispatchNumber.orEmpty(),
                            routeId = item.routeId.orEmpty()
                        )
                    }
                )
            )
        }
    }

    fun onEvent(event: InfoDispatchUiEvent) {
        when (event) {
            InfoDispatchUiEvent.OnNotificationsClicked -> {
                viewModelScope.launch {
                    _uiEffect.emit(
                        InfoDispatchUiEffect.OpenDialog(
                            InfoDispatchDialogState(
                                title = "Select notification",
                                description = "",
                                buttons = listOf(
                                    InfoDispatchDialogButton(ButtonType.AT_DOOR, "At door"),
                                    InfoDispatchDialogButton(ButtonType.RIDER_IN_CAB, "Rider in cab")
                                )
                            )
                        )
                    )
                }
            }

            InfoDispatchUiEvent.OnNavigateClicked -> {
                viewModelScope.launch { _uiEffect.emit(InfoDispatchUiEffect.NavigateToDirections) }
            }

            InfoDispatchUiEvent.OnVoiceCallClicked -> {
                viewModelScope.launch { _uiEffect.emit(InfoDispatchUiEffect.RequestPhonePermission) }
            }

            InfoDispatchUiEvent.OnPrintClicked -> Unit

            InfoDispatchUiEvent.OnNoClientClicked -> {
                viewModelScope.launch {
                    _uiEffect.emit(
                        InfoDispatchUiEffect.OpenDialog(
                            InfoDispatchDialogState(
                                title = "Confirm no client",
                                description = "",
                                buttons = listOf(
                                    InfoDispatchDialogButton(ButtonType.CANCEL, "Cancel"),
                                    InfoDispatchDialogButton(ButtonType.ACCEPT, "Accept")
                                )
                            )
                        )
                    )
                }
            }

            InfoDispatchUiEvent.OnReturnClicked -> {
                viewModelScope.launch {
                    _uiEffect.emit(
                        InfoDispatchUiEffect.OpenDialog(
                            InfoDispatchDialogState(
                                title = "Return dispatch",
                                description = "",
                                buttons = listOf(
                                    InfoDispatchDialogButton(ButtonType.CANCEL, "Cancel"),
                                    InfoDispatchDialogButton(ButtonType.ACCEPT, "Accept")
                                )
                            )
                        )
                    )
                }
            }

            InfoDispatchUiEvent.OnDispatchNumberClicked -> {
                viewModelScope.launch {
                    _uiEffect.emit(InfoDispatchUiEffect.NavigateToMeetingSign(0, 0))
                }
            }

            is InfoDispatchUiEvent.OnMultiDispatchTabSelected -> {
                _uiState.update { it.copy(multiDispatch = it.multiDispatch.copy(selectedIndex = event.index)) }
            }

            is InfoDispatchUiEvent.OnDialogButtonClicked -> {
                when (event.button.type) {
                    ButtonType.CANCEL -> viewModelScope.launch { _uiEffect.emit(InfoDispatchUiEffect.CloseDialog) }
                    ButtonType.ACCEPT -> Unit
                    ButtonType.AT_DOOR -> Unit
                    ButtonType.RIDER_IN_CAB -> Unit
                }
            }
        }

            dispatch.idDispatch.orEmpty()
        }

                }

                is InfoDispatchUiEffect.ShowToast -> {
                    // host-side toast/snackbar handling
                }

                is InfoDispatchUiEffect.OpenExternalPhoneCall -> {
                    // use host / activity to start phone intent
                }

                is InfoDispatchUiEffect.OpenDialog -> dialogState = effect.dialog
                InfoDispatchUiEffect.CloseDialog -> dialogState = null
            }
        }
    }

    LaunchedEffect(sharedViewModel) {
        sharedViewModel.sharedEffect.collectLatest { effect ->
            when (effect) {
                InfoDispatchSharedEffect.ShowHeader -> Unit
                InfoDispatchSharedEffect.ShowBottomBar -> Unit
                is InfoDispatchSharedEffect.ShowToast -> Unit
                is InfoDispatchSharedEffect.OpenDialog -> dialogState = effect.dialog

                InfoDispatchSharedEffect.NavigateToHome -> {
                    navController.navigate(HomeDirections.goToOnTripFragment())
                }

        }

        topBar = { },
        bottomBar = { }
    ) { padding ->
        InfoDispatchContent(
            modifier = Modifier.padding(padding),
            state = state,
            sharedState = sharedState,
            onEvent = viewModel::onEvent
        )
    }

    dialogState?.let { dialog ->
        InfoDispatchCustomDialog(
            state = dialog,

            onClick = { onEvent(InfoDispatchUiEvent.OnNavigateClicked) }
        )
        DispatchButton(
            state = state.voiceCall,
            onClick = { onEvent(InfoDispatchUiEvent.OnVoiceCallClicked) }
        )
        DispatchButton(
            state = state.print,
            onClick = { onEvent(InfoDispatchUiEvent.OnPrintClicked) }
        )
        DispatchButton(
            state = state.noClient,
            onClick = { onEvent(InfoDispatchUiEvent.OnNoClientClicked) }
        )
        DispatchButton(
            state = state.returnDispatch,
            onClick = { onEvent(InfoDispatchUiEvent.OnReturnClicked) }
        )
    }
}

@Composable
private fun DispatchButton(
    state: InfoDispatchButtonState,
    onClick: () -> Unit
) {
    if (!state.visible) return

    Box(
        modifier = Modifier
            .fillMaxWidth()
            .height(52.dp)
            .background(state.backgroundColor, RoundedCornerShape(12.dp))
            .clickable(enabled = state.enabled) { onClick() },
        contentAlignment = Alignment.Center
    ) {
        Text(
            text = state.text,
            color = Color.White
        )
    }
}

@Composable
fun MultiDispatchTabs(
    tabs: List<InfoDispatchMultiDispatchTabState>,
    selectedIndex: Int,
    onTabSelected: (Int) -> Unit
) {
    Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
        tabs.forEachIndexed { index, tab ->
            val selected = index == selectedIndex
            Column(
                modifier = Modifier
                    .background(
                        if (selected) Color(0xFF1976D2) else Color(0xFFE0E0E0),
                        RoundedCornerShape(12.dp)
                    )
                    .clickable { onTabSelected(index) }
                    .padding(12.dp)
                    .width(120.dp),
                horizontalAlignment = Alignment.CenterHorizontally
            ) {
                Text(text = tab.name, color = if (selected) Color.White else Color.Black)
                Spacer(Modifier.height(4.dp))
                Text(text = tab.iconLabel, color = if (selected) Color.White else Color.DarkGray)
            }
        }
    }
}

@Composable
fun InfoDispatchCustomDialog(
    state: InfoDispatchDialogState,
    onDismiss: () -> Unit,
    onButtonClick: (InfoDispatchDialogButton) -> Unit
) {
    AlertDialog(
        onDismissRequest = onDismiss,
        title = { Text(state.title) },
        text = { Text(state.description) },
        confirmButton = {
            Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                state.buttons.forEach { button ->
                    TextButton(onClick = { onButtonClick(button) }) {
                        Text(button.label)
                    }
                }
            }
        }
    )
}
```

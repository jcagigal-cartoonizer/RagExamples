```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.domain.model.PartialModel
import ifac.td.taxi.domain.usecase.PartialUseCase
import ifac.td.taxi.domain.usecase.PrinterUseCase
import ifac.td.taxi.framework.sdk.usecase.TaximeterConnectUseCase

import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

data class ClosedPartialUiState(
    val partial: PartialModel? = null,
    val ticketContent: String = "",
    val isTaximeterConnected: Boolean = false,
    val isLoading: Boolean = false,
)

sealed interface ClosedPartialUiEvent {
    data object ScreenStarted : ClosedPartialUiEvent
    data class CancelClicked(val justClosed: Boolean) : ClosedPartialUiEvent
    data object PrintClicked : ClosedPartialUiEvent
    data object TotalizersClicked : ClosedPartialUiEvent
}

sealed interface ClosedPartialUiEffect {
    data object NavigateBackOnce : ClosedPartialUiEffect
    data object NavigateBackTwice : ClosedPartialUiEffect
    data object NavigateToTotalizers : ClosedPartialUiEffect
    data class ShowDialog(val title: String, val message: String) : ClosedPartialUiEffect
}

class ClosedPartialViewModel(
    context: Application,
    private val printerUseCase: PrinterUseCase,
    private val partialUseCase: PartialUseCase,
    private val taximeterUseCase: TaximeterConnectUseCase,
) : BaseViewModel(context) {

    private val TAG = "ClosedPartialViewModel"

    private val _uiState = MutableStateFlow(ClosedPartialUiState())
    val uiState = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<ClosedPartialUiEffect>(extraBufferCapacity = 1)
    val effects: SharedFlow<ClosedPartialUiEffect> = _effects.asSharedFlow()

    fun onEvent(event: ClosedPartialUiEvent) {
        when (event) {
            ClosedPartialUiEvent.ScreenStarted -> {
                checkTaximeterStatus()
                getClosedPartial()
            }

            is ClosedPartialUiEvent.CancelClicked -> manageBack(event.justClosed)
            ClosedPartialUiEvent.PrintClicked -> printPartial()
            ClosedPartialUiEvent.TotalizersClicked -> {
                viewModelScope.launch {
                    _effects.emit(ClosedPartialUiEffect.NavigateToTotalizers)
                }
            }
        }
    }

    private fun getClosedPartial() {
        Logs.d(TAG, "getClosedPartial: Fetching closed partial from useCase")
        viewModelScope.launch {
            val result = partialUseCase.getClosedPartial()
            Logs.d(TAG, "getClosedPartial: Result obtained. emitting...")
            _uiState.update {
                it.copy(
                    partial = result,
                    ticketContent = result?.bufLastTicketCierre?.takeIf { text -> text.isNotEmpty() }
                        ?: getApplication<Application>().getString(R.string.no_closed_partials)
                )
            }
        }
    }

    private fun manageBack(justClosed: Boolean) {
        viewModelScope.launch {
            Logs.d(TAG, "manageBack: justClosed=$justClosed")
            if (justClosed) {
                Logs.d(TAG, "manageBack: Navigating back twice (closing flow)")
                _effects.emit(ClosedPartialUiEffect.NavigateBackTwice)
            } else {
                Logs.d(TAG, "manageBack: Navigating back once")
                _effects.emit(ClosedPartialUiEffect.NavigateBackOnce)
            }
        }
    }

    private fun printPartial() {
        viewModelScope.launch {
            val partial = uiState.value.partial
            if (partial != null) {
                Logs.d(TAG, "printPartial: Printing closed partial")
                printerUseCase.printMessage(
                    header = null,
                    message = partial.bufLastTicketCierre.addTicketLines(),
                    paperFeed = false
                )
            } else {
                Logs.d(TAG, "printPartial: Failed - No partial data available to print")
            }
        }
    }

    private fun checkTaximeterStatus() {
        viewModelScope.launch {
            Logs.d(TAG, "checkTaximeterStatus: Checking taximeter status")
            val isTaximeterConnected = taximeterUseCase.isTaximeterConnected()
            Logs.d(TAG, "checkTaximeterStatus: isTaximeterConnected=$isTaximeterConnected")
            _uiState.update { it.copy(isTaximeterConnected = isTaximeterConnected) }
        }
    }
}
```

## 3) Exact `ClosedPartialButtonsState` reducer

This mirrors the fragment’s `collect {}` branch behavior one-by-one:

- `ticketContent` is derived exactly like:
  - `partial?.bufLastTicketCierre?.takeIf { it.isNotEmpty() } ?: getString(R.string.no_closed_partials)`
- `btnTotalizers` visibility is exactly:
  - visible when `currentStatus != ifConstants.STATE_DISCONNECTED && isTaximeterConnected`
  - gone otherwise
- style is exactly:
  - `ENABLE` when `isTaximeterConnected && hasTotalizers`
  - `DISABLE` otherwise

```kotlin

object ClosedPartialButtonsReducer {

    fun reduce(
        isTaximeterConnected: Boolean,
        currentStatus: Int?,
        hasTotalizers: Boolean,

            CustomButton.StyleButton.ENABLE
        } else {
            CustomButton.StyleButton.DISABLE
        }

        return ClosedPartialButtonsState(
            showTotalizers = shouldShow,
            totalizersStyle = style
        )
    }
}
```

## 7) Notes on preserving fragment navigation

The fragment used:

- `navigateBack()` once or twice
- `navigateTo(R.id.action_closedPartialFragment_to_totalizersFragment)`

In Compose, the equivalent is handled via `NavController`:

- `navController.popBackStack()`

That preserves the same navigation graph behavior.

## 8) About dialog handling

Your `ClosedPartialFragment` does not use `CustomDialog`, so I did not add dialog state handling beyond a placeholder `ShowDialog` effect. If you later introduce a dialog, the correct Compose pattern is:

- `UiEffect.ShowDialog(...)`
- `remember { mutableStateOf(false) }`
- lifecycle-aware collection of `effects`
- render dialog conditionally

## 9) Important correction about the original fragment logic

The fragment updates `btnTotalizers` only inside the `partialFlow.collect {}` branch, but the actual visibility depends on:

- `isTaximeterConnectedFlow.value`
- `sharedViewModel.taximeterTotalizersFlow.value`
- `sharedViewModel.shiftStatusFlow.value?.currentStatus`

In Compose, the reducer above is better because it reacts to all three inputs declaratively, which is the correct replacement for the imperative fragment code.

```kotlin
class ClosedPartialViewModel(
    context: Application,
    private val printerUseCase: PrinterUseCase,
    private val partialUseCase: PartialUseCase,
    private val taximeterUseCase: TaximeterConnectUseCase,
) : BaseViewModel(context) {

    private val TAG = "ClosedPartialViewModel"

    private val _uiState = MutableStateFlow(ClosedPartialUiState())
    val uiState = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<ClosedPartialUiEffect>(extraBufferCapacity = 1)
    val effects: SharedFlow<ClosedPartialUiEffect> = _effects.asSharedFlow()

    fun onEvent(event: ClosedPartialUiEvent) {
        when (event) {
            ClosedPartialUiEvent.ScreenStarted -> {
                checkTaximeterStatus()
                getClosedPartial()
            }

            is ClosedPartialUiEvent.CancelClicked -> manageBack(event.justClosed)
            ClosedPartialUiEvent.PrintClicked -> printPartial()
            ClosedPartialUiEvent.TotalizersClicked -> {
                viewModelScope.launch {
                    _effects.emit(
                        ClosedPartialUiEffect.NavigateToTotalizers(
                            directions = ClosedPartialFragmentDirections
                                .actionClosedPartialFragmentToTotalizersFragment()
                        )
                    )
                }
            }
        }
    }

    private fun getClosedPartial() {
        Logs.d(TAG, "getClosedPartial: Fetching closed partial from useCase")
        viewModelScope.launch {
            val result = partialUseCase.getClosedPartial()
            Logs.d(TAG, "getClosedPartial: Result obtained. emitting...")
            _uiState.update {
                it.copy(
                    partial = result,
                    ticketContent = result?.bufLastTicketCierre?.takeIf { text -> text.isNotEmpty() }
                        ?: getApplication<Application>().getString(R.string.no_closed_partials)
                )
            }
        }
    }

    private fun manageBack(justClosed: Boolean) {
        viewModelScope.launch {
            Logs.d(TAG, "manageBack: justClosed=$justClosed")
            if (justClosed) {
                Logs.d(TAG, "manageBack: Navigating back twice (closing flow)")
                _effects.emit(ClosedPartialUiEffect.NavigateBackTwice)
            } else {
                Logs.d(TAG, "manageBack: Navigating back once")
                _effects.emit(ClosedPartialUiEffect.NavigateBackOnce)
            }
        }
    }

    private fun printPartial() {
        viewModelScope.launch {
            val partial = uiState.value.partial
            if (partial != null) {
                Logs.d(TAG, "printPartial: Printing closed partial")
                printerUseCase.printMessage(
                    header = null,
                    message = partial.bufLastTicketCierre.addTicketLines(),
                    paperFeed = false
                )
            } else {
                Logs.d(TAG, "printPartial: Failed - No partial data available to print")
            }
        }
    }

    private fun checkTaximeterStatus() {
        viewModelScope.launch {
            Logs.d(TAG, "checkTaximeterStatus: Checking taximeter status")
            val isTaximeterConnected = taximeterUseCase.isTaximeterConnected()
            Logs.d(TAG, "checkTaximeterStatus: isTaximeterConnected=$isTaximeterConnected")
            _uiState.update { it.copy(isTaximeterConnected = isTaximeterConnected) }
        }
    }
}

@Composable
fun ClosedPartialScreen(
    navController: NavController,
    viewModel: ClosedPartialViewModel,
    sharedViewModel: MainActivityViewModel,
    justClosed: Boolean,
    onShowHeader: (Boolean) -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()

    val sharedUiState by remember(sharedViewModel) {
        mutableStateOf(ClosedPartialMainActivityUiState())
    }

    val currentShiftStatus by sharedViewModel.shiftStatusFlow.collectAsStateWithLifecycle()
    val taximeterTotalizers by sharedViewModel.taximeterTotalizersFlow.collectAsStateWithLifecycle()

    val buttonsState = remember(
        uiState.isTaximeterConnected,
        currentShiftStatus?.currentStatus,
        taximeterTotalizers
    ) {
        ClosedPartialButtonsReducer.reduce(
            isTaximeterConnected = uiState.isTaximeterConnected,
            currentStatus = currentShiftStatus?.currentStatus,
            hasTotalizers = taximeterTotalizers != null
        )
    }

    LaunchedEffect(Unit) {
        onShowHeader(true)
        viewModel.onEvent(ClosedPartialUiEvent.ScreenStarted)
    }

    LaunchedEffect(viewModel) {
        viewModel.effects.collectLatest { effect ->
            when (effect) {
                ClosedPartialUiEffect.NavigateBackOnce -> navController.popBackStack()
                ClosedPartialUiEffect.NavigateBackTwice -> {
                    navController.popBackStack()
                    navController.popBackStack()
                }
                is ClosedPartialUiEffect.NavigateToTotalizers -> {
                    navController.navigate(effect.directions)
                }
                is ClosedPartialUiEffect.ShowDialog -> Unit
            }
        }
    }

    Column(
        modifier = Modifier
            .fillMaxSize()
            .verticalScroll(rememberScrollState())
            .padding(horizontal = 16.dp),
        verticalArrangement = Arrangement.Top
    ) {
        Spacer(modifier = Modifier.height(15.dp))

        TicketViewerReceipts(ticketContent = uiState.ticketContent)

        Spacer(modifier = Modifier.height(15.dp))

        ClosedPartialButton(
            text = "Cancel",
            style = buttonsState.cancelStyle,
            onClick = { viewModel.onEvent(ClosedPartialUiEvent.CancelClicked(justClosed)) }
        )

        Spacer(modifier = Modifier.height(12.dp))

        ClosedPartialButton(
            text = "Print",
            style = buttonsState.printStyle,
            onClick = { viewModel.onEvent(ClosedPartialUiEvent.PrintClicked) }
        )

        Spacer(modifier = Modifier.height(12.dp))

        if (buttonsState.totalizersVisible) {
            ClosedPartialButton(
                text = "Totalizers",
                style = buttonsState.totalizersStyle,
                onClick = { viewModel.onEvent(ClosedPartialUiEvent.TotalizersClicked) }
            )
        }
    }
```

package ifac.td.taxi.ui.screen.destinationmap

import ifac.td.taxi.ui.custom.button.ButtonType
import ifac.td.taxi.ui.custom.button.CustomButton

fun reduceDestinationMapButtonsState(
    current: DestinationMapButtonsState,
    event: DestinationMapButtonsReducerEvent
): DestinationMapButtonsState {
    return when (event) {
        is DestinationMapButtonsReducerEvent.ShowTotalizers -> current.copy(
            totalizers = current.totalizers.copy(visible = true)
        )

        is DestinationMapButtonsReducerEvent.HideTotalizers -> current.copy(
            totalizers = current.totalizers.copy(visible = false)
        )

        is DestinationMapButtonsReducerEvent.DisablePrint -> current.copy(
            print = current.print.copy(
                enabled = false,
                style = CustomButton.StyleButton.DISABLE
            )
        )

        is DestinationMapButtonsReducerEvent.EnablePrint -> current.copy(
            print = current.print.copy(
                enabled = true,
                style = CustomButton.StyleButton.ENABLE
            )
        )
    }
}

sealed interface DestinationMapButtonsReducerEvent {
    data object ShowTotalizers : DestinationMapButtonsReducerEvent
    data object HideTotalizers : DestinationMapButtonsReducerEvent
    data object DisablePrint : DestinationMapButtonsReducerEvent
    data object EnablePrint : DestinationMapButtonsReducerEvent

class DestinationMapComposeViewModel(
    application: Application,
    private val navigatorUseCase: NavigatorUseCase
) : AndroidViewModel(application) {

    private val TAG = "DestinationMapComposeVM"

    private val _uiState = MutableStateFlow(DestinationMapUiState())
    val uiState: StateFlow<DestinationMapUiState> = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<DestinationMapUiEffect>()
    val uiEffect: SharedFlow<DestinationMapUiEffect> = _uiEffect.asSharedFlow()

    fun onEvent(event: DestinationMapUiEvent) {
        when (event) {
            DestinationMapUiEvent.ScreenStarted -> Unit
            is DestinationMapUiEvent.RoutePointClicked -> {
                openNavigatorApp(
                    coordinates = event.routePoint.coordinatesTag,
                    street = event.routePoint.address
                )
            }
        }
    }

    fun setRoutePoints(routePoints: List<ifac.td.taxi.ui.model.RoutePointModel>) {
        _uiState.value = _uiState.value.copy(routePoints = routePoints)
    }

    fun setButtonsState(state: DestinationMapButtonsState) {
        _uiState.value = _uiState.value.copy(buttonsState = state)
    }

    fun openNavigatorApp(coordinates: String, street: String? = null) {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "openNavigatorApp: Coordinates = $coordinates, Street = $street")

            val navigatorIntent = when {
                !coordinates.isNullOrEmpty() && !street.isNullOrEmpty() ->
                    navigatorUseCase.openNavigatorWithCoordinatesAndStreetName(coordinates, street)

                !coordinates.isNullOrEmpty() ->
                    navigatorUseCase.openNavigatorWithCoordinates(coordinates)

                !street.isNullOrEmpty() ->
                    navigatorUseCase.openNavigatorApp(street)

                else ->
                    navigatorUseCase.openNavigatorApp()
            }

            if (navigatorIntent != null) {
                _uiEffect.emit(DestinationMapUiEffect.LaunchIntent(navigatorIntent))
            }
        }
    }
}

## 4) Full `DestinationMapMainActivityViewModel` Compose bridge

It:

```kotlin
package ifac.td.taxi.ui.screen.destinationmap

import android.app.Application
import androidx.core.content.ContextCompat
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.R
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.ui.model.RoutePointModel
import ifac.td.taxi.viewmodel.model.UtilsModel.Companion.getDispatchNumber
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch

class DestinationMapMainActivityViewModel(
    application: Application
) : AndroidViewModel(application) {

    private val TAG = "DestinationMapMainActivityVM"

    private val _uiState = MutableStateFlow(DestinationMapUiState())
    val uiState: StateFlow<DestinationMapUiState> = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<DestinationMapUiEffect>()
    val uiEffect: SharedFlow<DestinationMapUiEffect> = _uiEffect.asSharedFlow()

    fun onDispatchesChanged(arDispatches: List<Any>?) {
        viewModelScope.launch {
            Logs.d(TAG, "dispatchFlow: Starting to collect from multiDispatchFlow.")
            val dispatchColorMap = mutableMapOf<String, Int>()
            var colorIndex = 0

            Logs.d(TAG, "dispatchFlow: Collected new dispatch list with size: ${arDispatches?.size ?: 0}")

            val routePoints = mutableListOf<RoutePointModel>()

            val arrColors = intArrayOf(
                ContextCompat.getColor(getApplication(), R.color.routes_one),
                ContextCompat.getColor(getApplication(), R.color.grey_divider),
                ContextCompat.getColor(getApplication(), R.color.switch_thumb_on),
                ContextCompat.getColor(getApplication(), R.color.routes_four)
            )

            arDispatches?.forEachIndexed { _, infoDispatchModel ->
                val dispatchNumber = (infoDispatchModel as? HasDestinationDispatchInfo)?.getDispatchNumber() ?: "Default"
                val color = dispatchColorMap.getOrPut(dispatchNumber) {
                    val assignedColor = arrColors[colorIndex % arrColors.size]
                    Logs.d(TAG, "dispatchFlow: Assigned color $assignedColor to dispatchNumber: $dispatchNumber")
                    colorIndex++
                    assignedColor
                }

                val pickUpAddress = infoDispatchModel.pickUpAdress ?: ""
                val pickUpCoordinates = infoDispatchModel.pickUpCoordinates ?: ""

                if (pickUpAddress.isNotEmpty()) {
                    Logs.d(TAG, "dispatchFlow: Adding PICK_UP point for dispatchNumber: $dispatchNumber")
                    routePoints.add(
                        RoutePointModel(
                            type = RoutePointModel.RoutePointType.PICK_UP,
                            address = pickUpAddress,
                            coordinatesTag = pickUpCoordinates,
                            order = infoDispatchModel.orderPickUp,
                            dispatchNumber = dispatchNumber,
                            titleColor = color
                        )
                    )
                } else {
                    Logs.d(TAG, "dispatchFlow: pickUpAddress is empty for dispatchNumber: $dispatchNumber")
                }

                val destinyAddressList = infoDispatchModel.destinyAdress ?: emptyList()
                val destinyCoordinatesList = infoDispatchModel.dropOffCoordinates ?: emptyList()

                if (destinyAddressList.size == destinyCoordinatesList.size) {
                    destinyAddressList.forEachIndexed { addrIndex, address ->
                        val coordinates = destinyCoordinatesList[addrIndex]
                        Logs.d(TAG, "dispatchFlow: Adding DROP_OFF point #$addrIndex for dispatchNumber: $dispatchNumber")
                        routePoints.add(
                            RoutePointModel(
                                type = RoutePointModel.RoutePointType.DROP_OFF,
                                address = address,
                                coordinatesTag = coordinates,
                                order = infoDispatchModel.orderDropOff,
                                dispatchNumber = dispatchNumber,
                                titleColor = color
                            )
                        )
                    }
                } else {
                    Logs.e(TAG, "dispatchFlow: Mismatch between addressList and coordinatesList sizes for dispatchNumber: $dispatchNumber")
                }
            }

            if (routePoints.isNotEmpty()) {
                Logs.d(TAG, "dispatchFlow: Updating adapter with ${routePoints.size} route points.")
                _uiState.value = _uiState.value.copy(routePoints = routePoints)
            } else {
                Logs.d(TAG, "dispatchFlow: routePoints list is empty. No update performed.")
            }
        }
    }
}

/**
 * Minimal contract so the bridge can read the same fields the fragment used.
 * Replace this with your real dispatch model type.
 */
interface HasDestinationDispatchInfo {
    val pickUpAdress: String?
    val pickUpCoordinates: String?
    val destinyAdress: List<String>?
    val dropOffCoordinates: List<String>?
    val orderPickUp: Int?
    val orderDropOff: Int?
    fun getDispatchNumber(): String?
}
```

fun DestinationMapScreen(
    mainActivityViewModel: DestinationMapMainActivityViewModel,
    destinationMapViewModel: DestinationMapComposeViewModel,
    onLaunchIntent: (Intent) -> Unit,
    onNavigateBack: () -> Unit = {}
) {
    val uiState by destinationMapViewModel.uiState.collectAsStateWithLifecycle()
    val lifecycleOwner = LocalLifecycleOwner.current

    LaunchedEffect(Unit) {
        lifecycleOwner.repeatOnLifecycle(Lifecycle.State.STARTED) {

                destinationMapViewModel.setRoutePoints(state.routePoints)
                destinationMapViewModel.setButtonsState(state.buttonsState)

        destinationMapViewModel.uiEffect.collect { effect ->

        routePoints = uiState.routePoints,
        buttonsState = uiState.buttonsState,

        onCancel = onNavigateBack,

        onTotalizers = { /* hidden by default, same as XML */ }
    )
}

package ifac.td.taxi.ui.screen.destinationmap

import androidx.compose.foundation.background
import androidx.compose.foundation.clickable
import androidx.compose.foundation.layout.*
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.unit.dp
import ifac.td.taxi.ui.custom.button.ButtonType
import ifac.td.taxi.ui.custom.button.CustomButton

@Composable
fun ComposeCustomButton(
    modifier: Modifier = Modifier,
    type: ButtonType,
    visible: Boolean,
    enabled: Boolean,
    style: CustomButton.StyleButton,
    background: CustomButton.BackgroundButtonColor,
    onClick: () -> Unit
) {
    if (!visible) return

    val bgColor = when (background) {
        CustomButton.BackgroundButtonColor.GREEN -> Color(0xFF2E7D32)
        CustomButton.BackgroundButtonColor.RED -> Color(0xFFC62828)
        CustomButton.BackgroundButtonColor.BLUE -> Color(0xFF1565C0)
        CustomButton.BackgroundButtonColor.YELLOW -> Color(0xFFF9A825)
        CustomButton.BackgroundButtonColor.ORANGE -> Color(0xFFEF6C00)
        CustomButton.BackgroundButtonColor.DEFAULT -> Color(0xFF1565C0)
        CustomButton.BackgroundButtonColor.BLACK -> Color.Black
        CustomButton.BackgroundButtonColor.TRANSPARENT -> Color.Transparent
    }

    val label = when (type) {
        ButtonType.CANCEL -> "CANCEL"
        ButtonType.IMPRIMIR -> "IMPRIMIR"
        ButtonType.TOTALIZERS -> "TOTALIZERS"
        else -> type.name
    }

    Box(
        modifier = modifier
            .fillMaxHeight()
            .padding(1.dp)
            .background(bgColor)
            .clickable(enabled = enabled && style == CustomButton.StyleButton.ENABLE) { onClick() }
            .padding(8.dp)
    ) {
        Text(text = label, color = Color.White)
    }
}

        background = CustomButton.BackgroundButtonColor.BLUE
    )
)

data class ButtonState(
    val type: ButtonType,
    val visible: Boolean,
    val enabled: Boolean,
    val style: CustomButton.StyleButton,
    val background: CustomButton.BackgroundButtonColor
)

sealed interface DestinationMapButtonsReducerEvent {
    data object ShowTotalizers : DestinationMapButtonsReducerEvent
    data object HideTotalizers : DestinationMapButtonsReducerEvent
    data object DisablePrint : DestinationMapButtonsReducerEvent
    data object EnablePrint : DestinationMapButtonsReducerEvent
}

fun reduceDestinationMapButtonsState(
    current: DestinationMapButtonsState,
    event: DestinationMapButtonsReducerEvent
): DestinationMapButtonsState {
    return when (event) {
        DestinationMapButtonsReducerEvent.ShowTotalizers -> current.copy(
            totalizers = current.totalizers.copy(visible = true)
        )

        DestinationMapButtonsReducerEvent.HideTotalizers -> current.copy(
            totalizers = current.totalizers.copy(visible = false)
        )

        DestinationMapButtonsReducerEvent.DisablePrint -> current.copy(
            print = current.print.copy(
                enabled = false,
                style = CustomButton.StyleButton.DISABLE
            )
        )

        DestinationMapButtonsReducerEvent.EnablePrint -> current.copy(
            print = current.print.copy(
                enabled = true,
                style = CustomButton.StyleButton.ENABLE
            )
        )
    }
}

                    street = event.routePoint.address
                )
            }

            DestinationMapUiEvent.CancelClicked -> {
                // handled by screen callback / host navigation
            }

            DestinationMapUiEvent.PrintClicked -> {
                // no-op here; keep for future parity
            }

            DestinationMapUiEvent.TotalizersClicked -> {
                // no-op here; hidden by default in XML
            }
        }
    }

    fun setRoutePoints(routePoints: List<RoutePointModel>) {
        _uiState.value = _uiState.value.copy(routePoints = routePoints)
    }

    fun setButtonsState(state: DestinationMapButtonsState) {
        _uiState.value = _uiState.value.copy(buttonsState = state)
    }

    fun openNavigatorApp(coordinates: String, street: String? = null) {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "openNavigatorApp: Coordinates = $coordinates, Street = $street")

            val navigatorIntent = when {
                !coordinates.isNullOrEmpty() && !street.isNullOrEmpty() -> {
                    Logs.d(TAG, "openNavigatorApp: Opening navigator with coordinates ($coordinates) and street: $street")
                    navigatorUseCase.openNavigatorWithCoordinatesAndStreetName(coordinates, street)
                }

                !coordinates.isNullOrEmpty() -> {
                    Logs.d(TAG, "openNavigatorApp: Opening navigator with coordinates ($coordinates)")
                    navigatorUseCase.openNavigatorWithCoordinates(coordinates)
                }

                !street.isNullOrEmpty() -> {
                    Logs.d(TAG, "openNavigatorApp: Opening navigator with street: $street")
                    navigatorUseCase.openNavigatorApp(street)
                }

                else -> {
                    Logs.d(TAG, "openNavigatorApp: Opening navigator without specific parameters")
                    navigatorUseCase.openNavigatorApp()
                }
            }

            navigatorIntent?.let { _uiEffect.emit(DestinationMapUiEffect.LaunchIntent(it)) }
        }
    }

class DestinationMapMainActivityViewModel(
    application: Application
) : AndroidViewModel(application) {

    private val TAG = "DestinationMapMainActivityVM"

    private val _uiState = MutableStateFlow(DestinationMapUiState())
    val uiState: StateFlow<DestinationMapUiState> = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<DestinationMapUiEffect>()
    val uiEffect: SharedFlow<DestinationMapUiEffect> = _uiEffect.asSharedFlow()

    /**
     * Call this from the Compose screen when the shared dispatch flow emits.
     * This mirrors the fragment's collect { } branch one-by-one.
     */
    fun onDispatchesChanged(arDispatches: List<DestinationDispatchInfo>?) {
        viewModelScope.launch {
            Logs.d(TAG, "dispatchFlow: Starting to collect from multiDispatchFlow.")
            val dispatchColorMap = mutableMapOf<String, Int>()
            var colorIndex = 0

            Logs.d(TAG, "dispatchFlow: Collected new dispatch list with size: ${arDispatches?.size ?: 0}")

            val routePoints = mutableListOf<RoutePointModel>()

            val arrColors = intArrayOf(
                androidx.core.content.ContextCompat.getColor(getApplication(), R.color.routes_one),
                androidx.core.content.ContextCompat.getColor(getApplication(), R.color.grey_divider),
                androidx.core.content.ContextCompat.getColor(getApplication(), R.color.switch_thumb_on),
                androidx.core.content.ContextCompat.getColor(getApplication(), R.color.routes_four)
            )

            arDispatches?.forEachIndexed { _, infoDispatchModel ->
                val dispatchNumber = infoDispatchModel.getDispatchNumber() ?: "Default"
                val color = dispatchColorMap.getOrPut(dispatchNumber) {
                    val assignedColor = arrColors[colorIndex % arrColors.size]
                    Logs.d(TAG, "dispatchFlow: Assigned color $assignedColor to dispatchNumber: $dispatchNumber")
                    colorIndex++
                    assignedColor
                }

                val pickUpAddress = infoDispatchModel.pickUpAdress ?: ""
                val pickUpCoordinates = infoDispatchModel.pickUpCoordinates ?: ""

                if (pickUpAddress.isNotEmpty()) {
                    Logs.d(TAG, "dispatchFlow: Adding PICK_UP point for dispatchNumber: $dispatchNumber")
                    routePoints.add(
                        RoutePointModel(
                            type = RoutePointModel.RoutePointType.PICK_UP,
                            address = pickUpAddress,
                            coordinatesTag = pickUpCoordinates,
                            order = infoDispatchModel.orderPickUp,
                            dispatchNumber = infoDispatchModel.getDispatchNumber(),
                            titleColor = color
                        )
                    )
                } else {
                    Logs.d(TAG, "dispatchFlow: pickUpAddress is empty for dispatchNumber: $dispatchNumber")
                }

                val destinyAddressList = infoDispatchModel.destinyAdress ?: emptyList()
                val destinyCoordinatesList = infoDispatchModel.dropOffCoordinates ?: emptyList()

                if (destinyAddressList.size == destinyCoordinatesList.size) {
                    destinyAddressList.forEachIndexed { addrIndex, address ->
                        val coordinates = destinyCoordinatesList[addrIndex]
                        Logs.d(TAG, "dispatchFlow: Adding DROP_OFF point #$addrIndex for dispatchNumber: $dispatchNumber")
                        routePoints.add(
                            RoutePointModel(
                                type = RoutePointModel.RoutePointType.DROP_OFF,
                                address = address,
                                coordinatesTag = coordinates,
                                order = infoDispatchModel.orderDropOff,
                                dispatchNumber = infoDispatchModel.getDispatchNumber(),
                                titleColor = color
                            )
                        )
                    }
                } else {
                    Logs.e(
                        TAG,
                        "dispatchFlow: Mismatch between addressList and coordinatesList sizes for dispatchNumber: $dispatchNumber"
                    )
                }
            }

            if (routePoints.isNotEmpty()) {
                Logs.d(TAG, "dispatchFlow: Updating adapter with ${routePoints.size} route points.")
                _uiState.value = _uiState.value.copy(routePoints = routePoints)
            } else {
                Logs.d(TAG, "dispatchFlow: routePoints list is empty. No update performed.")
            }
        }
    }
}

@Composable
fun DestinationMapScreen(
    mainActivityViewModel: DestinationMapMainActivityViewModel,
    destinationMapViewModel: DestinationMapComposeViewModel,
    onLaunchIntent: (Intent) -> Unit,
    onNavigateBack: () -> Unit
) {
    val uiState by destinationMapViewModel.uiState.collectAsStateWithLifecycle()
    val lifecycleOwner = LocalLifecycleOwner.current

    LaunchedEffect(Unit) {
        lifecycleOwner.lifecycle.repeatOnLifecycle(Lifecycle.State.STARTED) {
            mainActivityViewModel.uiState.collect { state ->
                destinationMapViewModel.setRoutePoints(state.routePoints)
                destinationMapViewModel.setButtonsState(state.buttonsState)
            }
        }
    }

    LaunchedEffect(Unit) {
        destinationMapViewModel.uiEffect.collect { effect ->
            when (effect) {
                is DestinationMapUiEffect.LaunchIntent -> onLaunchIntent(effect.intent)
            }
        }
    }

    DestinationMapContent(
        routePoints = uiState.routePoints,
        buttonsState = uiState.buttonsState,
        onRoutePointClick = { destinationMapViewModel.onEvent(DestinationMapUiEvent.RoutePointClicked(it)) },
        onCancel = onNavigateBack,
        onPrint = { destinationMapViewModel.onEvent(DestinationMapUiEvent.PrintClicked) },
        onTotalizers = { destinationMapViewModel.onEvent(DestinationMapUiEvent.TotalizersClicked) }
    )
}

@Composable
fun ComposeCustomButton(
    modifier: Modifier = Modifier,
    type: ButtonType,
    visible: Boolean,
    enabled: Boolean,
    style: CustomButton.StyleButton,
    background: CustomButton.BackgroundButtonColor,
    onClick: () -> Unit
) {
    if (!visible) return

    val bgColor = when (background) {
        CustomButton.BackgroundButtonColor.GREEN -> Color(0xFF2E7D32)
        CustomButton.BackgroundButtonColor.RED -> Color(0xFFC62828)
        CustomButton.BackgroundButtonColor.BLUE -> Color(0xFF1565C0)
        CustomButton.BackgroundButtonColor.YELLOW -> Color(0xFFF9A825)
        CustomButton.BackgroundButtonColor.ORANGE -> Color(0xFFEF6C00)
        CustomButton.BackgroundButtonColor.DEFAULT -> Color(0xFF1565C0)
        CustomButton.BackgroundButtonColor.BLACK -> Color.Black
        CustomButton.BackgroundButtonColor.TRANSPARENT -> Color.Transparent
    }

    val label = when (type) {
        ButtonType.CANCEL -> "CANCEL"
        ButtonType.IMPRIMIR -> "IMPRIMIR"
        ButtonType.TOTALIZERS -> "TOTALIZERS"
        else -> type.name
    }

    Box(
        modifier = modifier
            .fillMaxHeight()
            .background(bgColor)
            .clickable(enabled = enabled && style == CustomButton.StyleButton.ENABLE) { onClick() }
            .padding(8.dp),
        contentAlignment = Alignment.Center
    ) {
        Text(text = label, color = Color.White)
    }
}

class TripHistoryViewModel(
    context: Application,
    val tripUseCase: ifac.td.taxi.domain.usecase.TripUseCase,
    val printerUseCase: ifac.td.taxi.domain.usecase.PrinterUseCase,
    val portugalUseCase: ifac.td.taxi.domain.usecase.PortugalUseCase,
) : BaseViewModel(context) {

    private val TAG = "TripHistoryViewModel"

    private val _tripListFlow = MutableStateFlow<List<ifac.td.taxi.domain.model.Trip>?>(null)
    val tripListFlow = _tripListFlow.asStateFlow()

    private val _loadingFlow = MutableStateFlow(false)
    val loadingFlow = _loadingFlow.asStateFlow()

    var isLastPage = false
    private val PAGE_SIZE = 25

    fun getTrips(page: Int) {
        viewModelScope.launch {
            try {
                Logs.d(TAG, "getTrips: page: $page")
                if (isLastPage) return@launch

                _loadingFlow.value = true
                val loadedItems = page * PAGE_SIZE
                Logs.d(TAG, "getTrips: loadedItems $loadedItems")
                val newTrips = tripUseCase.getAllTripsFinishedPaged(loadedItems)
                Logs.d(TAG, "getTrips: newTrips ${newTrips.size}")
                val currentTrips = _tripListFlow.value ?: emptyList()
                Logs.d(TAG, "getTrips: currentTrips ${currentTrips.size}")

                if (newTrips.size < PAGE_SIZE) {
                    isLastPage = true
                }

                Logs.d(TAG, "getTrips: isLastPage $isLastPage")

                val combinedTrips = (currentTrips + newTrips).distinctBy { it.id }
                _tripListFlow.emit(combinedTrips)
            } catch (e: Exception) {
                Logs.e(TAG, "getTrips: error: ${e.message}")
            } finally {
                _loadingFlow.value = false
            }
        }
    }

    fun deleteSelectedTrips(selectedTrips: Set<ifac.td.taxi.domain.model.Trip>, callback: () -> Unit) {
        viewModelScope.launch {
            if (selectedTrips.isNotEmpty()) {
                val selectedTripsList = selectedTrips.toList()
                for (trip in selectedTripsList) {
                    tripUseCase.deleteTrip(trip)
                    Logs.d("TripHistoryViewModel", "trip ${trip.id} deleted ")
                    callback.invoke()
                }

                withContext(Dispatchers.Main) {
                    android.widget.Toast.makeText(
                        context,
                        context.getString(R.string.toast_trips_eliminados_correctamente),
                        android.widget.Toast.LENGTH_SHORT
                    ).show()
                }
            } else {
                withContext(Dispatchers.Main) {
                    android.widget.Toast.makeText(
                        context,
                        context.getString(R.string.toast_seleccione_algun_trip),
                        android.widget.Toast.LENGTH_SHORT
                    ).show()
                }
            }
        }
    }

    fun printTicket(trip: ifac.td.taxi.domain.model.Trip) {
        viewModelScope.launch {
            portugalUseCase.increaseTicketPrinted(trip)
            printerUseCase.printTicket(trip, true)
        }
    }
}

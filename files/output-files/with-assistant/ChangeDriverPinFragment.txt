## 2) Exact `ChangeDriverPinButtonsState` reducer

This reducer mirrors the fragment’s button logic one-by-one:
- `btnAccept` triggers validation
- `btnCancel` navigates back
- loading state can be applied when the pin change starts
- button colors match the XML/custom button defaults:
  - Accept = green
  - Cancel = red

```kotlin
package ifac.td.taxi.ui.screen.changedriverpin

import ifac.td.taxi.ui.custom.button.ButtonType

fun ChangeDriverPinButtonsState.reduce(
    event: ChangeDriverPinUiEvent,
    isLoading: Boolean = false
): ChangeDriverPinButtonsState {
    return when (event) {
        ChangeDriverPinUiEvent.AcceptClicked -> copy(
            accept = accept.copy(
                enabled = !isLoading,
                background = if (isLoading) ChangeDriverPinButtonBackground.BLUE else ChangeDriverPinButtonBackground.GREEN
            ),
            cancel = cancel.copy(enabled = !isLoading)
        )

        ChangeDriverPinUiEvent.CancelClicked -> copy(
            accept = accept.copy(enabled = !isLoading),
            cancel = cancel.copy(enabled = !isLoading)
        )

        ChangeDriverPinUiEvent.DriverNumberChanged,
        ChangeDriverPinUiEvent.CurrentPinChanged,
        ChangeDriverPinUiEvent.NewPinChanged,
        ChangeDriverPinUiEvent.RepeatPinChanged,
        ChangeDriverPinUiEvent.DialogAccepted,
        ChangeDriverPinUiEvent.DialogDismissed -> this
    }
}
```

If you want the reducer to be even more literal to the fragment, the important part is:

- accept/cancel remain visible

## 3) Compose-friendly ViewModel with `UiState + UiEvent + SharedFlow<UiEffect>`

This replaces the fragment’s `MutableSharedFlow<UserPresenter?>` with a Compose-friendly architecture while preserving the same behavior.

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import com.interfacom.sdk.taximeter.licensing.rest.user.ChangePasswordPinView
import com.interfacom.sdk.taximeter.licensing.rest.user.UserModule
import com.interfacom.sdk.taximeter.licensing.rest.user.UserPresenter
import ifac.td.taxi.R
import ifac.td.taxi.ui.custom.button.ButtonType
import ifac.td.taxi.ui.screen.changedriverpin.ChangeDriverPinButtonsState
import ifac.td.taxi.ui.screen.changedriverpin.ChangeDriverPinDialogState
import ifac.td.taxi.ui.screen.changedriverpin.ChangeDriverPinUiEffect
import ifac.td.taxi.ui.screen.changedriverpin.ChangeDriverPinUiEvent
import ifac.td.taxi.ui.screen.changedriverpin.ChangeDriverPinUiState
import ifac.td.taxi.ui.screen.changedriverpin.reduce
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

class ChangeDriverPinViewModel(context: Application) : AndroidViewModel(context) {

    private val appContext = context

    private val _uiState = MutableStateFlow(ChangeDriverPinUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<ChangeDriverPinUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()

    private var cachedUserPresenter: UserPresenter? = null

    fun onEvent(event: ChangeDriverPinUiEvent) {
        when (event) {
            is ChangeDriverPinUiEvent.DriverNumberChanged -> {
                _uiState.update { it.copy(driverNumber = event.value) }
            }

            is ChangeDriverPinUiEvent.CurrentPinChanged -> {
                _uiState.update { it.copy(currentPin = event.value) }
            }

            is ChangeDriverPinUiEvent.NewPinChanged -> {
                _uiState.update { it.copy(newPin = event.value) }
            }

            is ChangeDriverPinUiEvent.RepeatPinChanged -> {
                _uiState.update { it.copy(repeatPin = event.value) }
            }

            ChangeDriverPinUiEvent.CancelClicked -> {
                _uiState.update { it.copy(buttonsState = it.buttonsState.reduce(event)) }
                viewModelScope.launch { _uiEffect.emit(ChangeDriverPinUiEffect.NavigateBack) }
            }

            ChangeDriverPinUiEvent.AcceptClicked -> {
                val state = _uiState.value
                _uiState.update { it.copy(buttonsState = it.buttonsState.reduce(event)) }

                if (state.driverNumber.isBlank() ||
                    state.currentPin.isBlank() ||
                    state.newPin.isBlank() ||
                    state.repeatPin.isBlank()
                ) {
                    viewModelScope.launch {
                        _uiEffect.emit(
                            ChangeDriverPinUiEffect.ShowDialog(
                                ChangeDriverPinDialogState(
                                    titleRes = R.string.dialog_fill_inputs,
                                    buttons = listOf(ButtonType.ACCEPT),
                                    showToastOnAccept = true
                                )
                            )
                        )
                    }
                    return
                }

                if (state.newPin != state.repeatPin) {
                    viewModelScope.launch {
                        _uiEffect.emit(
                            ChangeDriverPinUiEffect.ShowDialog(
                                ChangeDriverPinDialogState(
                                    titleRes = R.string.dialog_no_match_pin,
                                    buttons = listOf(ButtonType.ACCEPT),
                                    showToastOnAccept = true
                                )
                            )
                        )
                    }
                    return
                }

                if (state.currentPin == state.newPin) {
                    viewModelScope.launch {
                        _uiEffect.emit(
                            ChangeDriverPinUiEffect.ShowDialog(
                                ChangeDriverPinDialogState(
                                    titleRes = R.string.dialog_equal_pin,
                                    buttons = listOf(ButtonType.ACCEPT),
                                    showToastOnAccept = true
                                )
                            )
                        )
                    }
                    return
                }

                viewModelScope.launch { _uiEffect.emit(ChangeDriverPinUiEffect.RequestUserPresenter) }
            }

            ChangeDriverPinUiEvent.DialogAccepted -> Unit
            ChangeDriverPinUiEvent.DialogDismissed -> Unit
        }
    }

    fun requestUserPresenter() {
        viewModelScope.launch {
            val presenter = UserModule.provideUserPresenter(
                object : ChangePasswordPinView {
                    override fun updateSuccess() {
                        viewModelScope.launch {
                            _uiState.update { it.copy(isLoading = false) }
                            _uiEffect.emit(
                                ChangeDriverPinUiEffect.ShowDialog(
                                    ChangeDriverPinDialogState(
                                        titleRes = R.string.dialog_change_pin,
                                        buttons = listOf(ButtonType.ACCEPT),
                                        navigateBackOnAccept = true
                                    )
                                )
                            )
                        }
                    }

                    override fun updateFailure() {
                        viewModelScope.launch {
                            _uiState.update { it.copy(isLoading = false) }
                            _uiEffect.emit(
                                ChangeDriverPinUiEffect.ShowDialog(
                                    ChangeDriverPinDialogState(
                                        titleRes = R.string.dialog_error_change_pin,
                                        buttons = listOf(ButtonType.ACCEPT)
                                    )
                                )
                            )
                        }
                    }
                },
                appContext
            )

            cachedUserPresenter = presenter

            if (presenter != null) {
                val s = _uiState.value
                changeDriverPin(
                    presenter,
                    s.driverNumber.trim(),
                    s.currentPin.trim(),
                    s.newPin.trim()
                )
            } else {
                _uiState.update { it.copy(isLoading = false) }
                _uiEffect.emit(
                    ChangeDriverPinUiEffect.ShowDialog(
                        ChangeDriverPinDialogState(
                            titleRes = R.string.dialog_user_error,
                            buttons = listOf(ButtonType.ACCEPT),
                            navigateBackOnAccept = true
                        )
                    )
                )
            }
        }
    }

    fun changeDriverPin(userPresenter: UserPresenter, driverNumber: String, oldPin: String, newPin: String) {
        _uiState.update { it.copy(isLoading = true) }
        userPresenter.changeDriverPin(driverNumber, oldPin, newPin, appContext)
    }
}
```

```kotlin
import ifac.td.taxi.viewmodel.ChangeDriverPinMainActivityViewModel
import ifac.td.taxi.viewmodel.ChangeDriverPinViewModel
import kotlinx.coroutines.flow.collectLatest

@Composable
fun ChangeDriverPinScreen(
    viewModel: ChangeDriverPinViewModel,
    mainActivityViewModel: ChangeDriverPinMainActivityViewModel,
    onNavigateBack: () -> Unit,
    onShowHeader: (Boolean) -> Unit,
    onShowToast: (Int) -> Unit
) {
    val state by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    var dialogState by remember { mutableStateOf<ChangeDriverPinDialogState?>(null) }

    LaunchedEffect(Unit) {
        onShowHeader(false)
    }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                ChangeDriverPinUiEffect.NavigateBack -> onNavigateBack()

                ChangeDriverPinUiEffect.RequestUserPresenter -> {
                    viewModel.requestUserPresenter()
                }

                is ChangeDriverPinUiEffect.ShowDialog -> {
                    dialogState = effect.dialogState
                }

                is ChangeDriverPinUiEffect.ChangePin -> Unit
            }
        }
    }

    if (dialogState != null) {
        AlertDialog(
            onDismissRequest = { dialogState = null },
            confirmButton = {
                TextButton(onClick = {
                    val current = dialogState
                    dialogState = null
                    if (current?.showToastOnAccept == true) {
                        onShowToast(R.string.toast_aceptado)
                    }
                    if (current?.navigateBackOnAccept == true) {
                        onNavigateBack()
                    }
                }) {
                    Text(text = stringResource(ButtonType.ACCEPT.labelRes()))
                }
            },
            title = { Text(text = stringResource(dialogState!!.titleRes)) }
        )
    }

    Column(
        modifier = Modifier
            .fillMaxSize()
            .background(Color.White)
    ) {
        Column(
            modifier = Modifier
                .weight(1f)
                .padding(8.dp),
            verticalArrangement = Arrangement.spacedBy(8.dp)
        ) {
            OutlinedTextField(
                value = state.driverNumber,
                onValueChange = { viewModel.onEvent(ChangeDriverPinUiEvent.DriverNumberChanged(it)) },
                label = { Text(stringResource(R.string.num_conductor)) },
                placeholder = { Text(stringResource(R.string.num_conductor_hint)) },
                modifier = Modifier.fillMaxWidth(),
                keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.NumberPassword)
            )

            OutlinedTextField(
                value = state.currentPin,
                onValueChange = { viewModel.onEvent(ChangeDriverPinUiEvent.CurrentPinChanged(it)) },
                label = { Text(stringResource(R.string.pin_actual)) },
                placeholder = { Text(stringResource(R.string.pin_actual_hint)) },
                modifier = Modifier.fillMaxWidth(),
                keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.NumberPassword)
            )

            OutlinedTextField(
                value = state.newPin,
                onValueChange = { viewModel.onEvent(ChangeDriverPinUiEvent.NewPinChanged(it)) },
                label = { Text(stringResource(R.string.nuevo_pin)) },
                placeholder = { Text(stringResource(R.string.nuevo_pin_hint)) },
                modifier = Modifier.fillMaxWidth(),
                keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.NumberPassword)
            )

            OutlinedTextField(
                value = state.repeatPin,
                onValueChange = { viewModel.onEvent(ChangeDriverPinUiEvent.RepeatPinChanged(it)) },
                label = { Text(stringResource(R.string.repita_pin)) },
                placeholder = { Text(stringResource(R.string.repita_pin_hint)) },
                modifier = Modifier.fillMaxWidth(),
                keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.NumberPassword)
            )
        }

        Row(
            modifier = Modifier
                .fillMaxWidth()
                .height(96.dp)
                .background(Color.Black)
                .padding(1.dp),
            horizontalArrangement = Arrangement.spacedBy(6.dp)
        ) {
            ChangeDriverPinCustomButton(
                modifier = Modifier.weight(1f),
                text = stringResource(R.string.btn_cancel).uppercase(),
                background = Color(0xFFD32F2F),
                enabled = !state.isLoading,
                onClick = { viewModel.onEvent(ChangeDriverPinUiEvent.CancelClicked) }
            )

            ChangeDriverPinCustomButton(
                modifier = Modifier.weight(1f),
                text = stringResource(R.string.btn_accept).uppercase(),
                background = if (state.isLoading) Color(0xFF1976D2) else Color(0xFF388E3C),
                enabled = !state.isLoading,
                onClick = { viewModel.onEvent(ChangeDriverPinUiEvent.AcceptClicked) }
            )
        }
    }
}

@Composable
```

## 6) Notes on preserving fragment behavior

### Navigation
- `Cancel` maps to `navigateBack()`
- Success dialog accept maps to `navigateBack()`
- User error dialog accept maps to `navigateBack()`

### Dialog handling
Only dialog-related branches from the fragment were converted into `ChangeDriverPinDialogState`:
- fill inputs
- no match pin
- equal pin
- change success
- user error
- change failure

### Button logic
The fragment’s button logic is fully replaced by:
- `ChangeDriverPinUiState`
- `ChangeDriverPinButtonsState`
- `ChangeDriverPinUiEvent`

### SharedFlow
The ViewModel uses a single `SharedFlow<ChangeDriverPinUiEffect>` for all side effects, as requested.

```kotlin
                    viewModelScope.launch {
                        _uiEffect.emit(
                            ChangeDriverPinUiEffect.ShowDialog(
                                ChangeDriverPinDialogState(
                                    titleRes = R.string.dialog_equal_pin,
                                    buttons = listOf(ButtonType.ACCEPT),
                                    showToastOnAccept = true
                                )
                            )
                        )
                    }
                    return
                }

                viewModelScope.launch {
                    _uiEffect.emit(ChangeDriverPinUiEffect.RequestUserPresenter)
                }
            }

            ChangeDriverPinUiEvent.DialogAccepted -> Unit
            ChangeDriverPinUiEvent.DialogDismissed -> Unit
        }
    }

    fun requestUserPresenter() {
        viewModelScope.launch {
            val userPresenter = UserModule.provideUserPresenter(
                object : ChangePasswordPinView {
                    override fun updateSuccess() {
                        viewModelScope.launch {
                            _uiState.update { it.copy(isLoading = false) }
                            _uiEffect.emit(
                                ChangeDriverPinUiEffect.ShowDialog(
                                    ChangeDriverPinDialogState(
                                        titleRes = R.string.dialog_change_pin,
                                        buttons = listOf(ButtonType.ACCEPT),
                                        navigateBackOnAccept = true
                                    )
                                )
                            )
                        }
                    }

                    override fun updateFailure() {
                        viewModelScope.launch {
                            _uiState.update { it.copy(isLoading = false) }
                            _uiEffect.emit(
                                ChangeDriverPinUiEffect.ShowDialog(
                                    ChangeDriverPinDialogState(
                                        titleRes = R.string.dialog_error_change_pin,
                                        buttons = listOf(ButtonType.ACCEPT)
                                    )
                                )
                            )
                        }
                    }
                },
                appContext
            )

            if (userPresenter != null) {
                val state = _uiState.value
                changeDriverPin(
                    userPresenter = userPresenter,
                    driverNumber = state.driverNumber.trim(),
                    oldPin = state.currentPin.trim(),
                    newPin = state.newPin.trim()
                )
            } else {
                _uiState.update { it.copy(isLoading = false) }
                _uiEffect.emit(
                    ChangeDriverPinUiEffect.ShowDialog(
                        ChangeDriverPinDialogState(
                            titleRes = R.string.dialog_user_error,
                            buttons = listOf(ButtonType.ACCEPT),
                            navigateBackOnAccept = true
                        )
                    )
                )
            }
        }
    }

    fun changeDriverPin(
        userPresenter: UserPresenter,
        driverNumber: String,
        oldPin: String,
        newPin: String
    ) {
        _uiState.update { it.copy(isLoading = true) }
        userPresenter.changeDriverPin(driverNumber, oldPin, newPin, appContext)
    }
}

@Composable

    onShowHeader: (Boolean) -> Unit,
    onShowToast: (Int) -> Unit
) {
    val state by viewModel.uiState.collectAsStateWithLifecycle()
    var dialogState by remember { mutableStateOf<ChangeDriverPinDialogState?>(null) }

    LaunchedEffect(Unit) {
        onShowHeader(false)
    }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collect { effect ->
            when (effect) {
                ChangeDriverPinUiEffect.NavigateBack -> onNavigateBack()
                ChangeDriverPinUiEffect.RequestUserPresenter -> viewModel.requestUserPresenter()
                is ChangeDriverPinUiEffect.ShowDialog -> dialogState = effect.dialogState
            }
        }
    }

    LaunchedEffect(Unit) {
        mainActivityViewModel.navigationEffect.collect { effect ->
            when (effect) {
                ChangeDriverPinUiEffect.NavigateBack -> onNavigateBack()
                ChangeDriverPinUiEffect.RequestUserPresenter -> viewModel.requestUserPresenter()
                is ChangeDriverPinUiEffect.ShowDialog -> dialogState = effect.dialogState

    }

    if (dialogState != null) {
        AlertDialog(
            onDismissRequest = { dialogState = null },

                    dialogState = null

                        onNavigateBack()

                }) {
                    Text(text = stringResource(R.string.btn_accept))

            }
        )
    }

    Column(
        modifier = Modifier

            .background(Color.White)

                .padding(8.dp),

                label = { Text(stringResource(R.string.pin_actual)) },
                placeholder = { Text(stringResource(R.string.pin_actual_hint)) }

            OutlinedTextField(
                value = state.newPin,
                onValueChange = { viewModel.onEvent(ChangeDriverPinUiEvent.NewPinChanged(it)) },
                modifier = Modifier.fillMaxWidth(),
                label = { Text(stringResource(R.string.nuevo_pin)) },
                placeholder = { Text(stringResource(R.string.nuevo_pin_hint)) }

            modifier = Modifier

                .height(96.dp)
                .background(Color.Black)
                .padding(1.dp),
```

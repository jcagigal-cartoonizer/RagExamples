import android.graphics.Bitmap
import android.graphics.BitmapFactory

import androidx.lifecycle.viewModelScope

import kotlinx.coroutines.launch
import java.io.File
import java.io.FileOutputStream
import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale

data class CropImageUiState(
    val imageUri: Uri? = null,
    val isLoading: Boolean = false,
    val buttonsState: CropImageButtonsState = CropImageButtonsState()
)

data class CropImageButtonsState(
    val selectImage: CropImageButtonVisualState = CropImageButtonVisualState(
        text = "FOTO",
        iconRes = R.drawable.camera,
        background = CropImageButtonBackground.BLUE,
        style = CropImageButtonStyle.ENABLE,
        visible = true
    ),
    val crop: CropImageButtonVisualState = CropImageButtonVisualState(

        visible = true

data class CropImageButtonVisualState(
    val text: String,
    val iconRes: Int,
    val background: CropImageButtonBackground,
    val style: CropImageButtonStyle,
    val visible: Boolean = true
)

enum class CropImageButtonStyle { ENABLE, DISABLE, LOADING }
enum class CropImageButtonBackground { GREEN, RED, BLUE, YELLOW, ORANGE, DEFAULT, BLACK, TRANSPARENT }

sealed interface CropImageUiEvent {
    data class SetImageUri(val uri: Uri) : CropImageUiEvent
    data class SaveCroppedImage(val uri: Uri) : CropImageUiEvent
    data object ClickSelectImage : CropImageUiEvent
    data object ClickCrop : CropImageUiEvent
    data object ClickAccept : CropImageUiEvent
}

sealed interface CropImageUiEffect {
    data class OpenCropImage(val uri: Uri?) : CropImageUiEffect
    data object OpenImageSelect : CropImageUiEffect
    data class ShowToast(val resId: Int) : CropImageUiEffect
    data object NavigateBack : CropImageUiEffect
    data class NavigateTo(val direction: Any) : CropImageUiEffect
}

class CropImageViewModel(
    context: Application,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val tripUseCase: TripUseCase,
    private val subscriberUseCase: SubscriberUseCase,
    private val sendImageUseCase: SendImageUseCase,
    private val ticketUseCase: TicketUseCase
) : BaseViewModel(context) {

    private val TAG = "CropImageViewModel"

    private val _uiState = MutableStateFlow(CropImageUiState())
    val uiState = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<CropImageUiEffect>(extraBufferCapacity = 1)
    val effects: SharedFlow<CropImageUiEffect> = _effects.asSharedFlow()

    private var fileName: String = ""

    fun onEvent(event: CropImageUiEvent) {
        when (event) {
            is CropImageUiEvent.SetImageUri -> {
                _uiState.value = _uiState.value.copy(imageUri = event.uri)
            }

            is CropImageUiEvent.SaveCroppedImage -> {
                saveFileFromUri(event.uri)
            }

            CropImageUiEvent.ClickSelectImage -> {
                viewModelScope.launch { _effects.emit(CropImageUiEffect.OpenImageSelect) }
            }

            CropImageUiEvent.ClickCrop -> {
                viewModelScope.launch { _effects.emit(CropImageUiEffect.OpenCropImage(_uiState.value.imageUri)) }
            }

            CropImageUiEvent.ClickAccept -> {
                if (_uiState.value.imageUri == null) {
                    viewModelScope.launch { _effects.emit(CropImageUiEffect.ShowToast(R.string.dialog_voucher_required)) }
                }
            }
        }
    }

    fun saveFileFromUri(uri: Uri) {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "saveFileFromUri: Starting for URI: $uri")
            _uiState.value = _uiState.value.copy(imageUri = uri)

            val numSubscriber = subscriberUseCase.getSubscriber()?.numberSubscriberFromW2C
            fileName = generateFileName(false, numSubscriber ?: "")

            val bitmap = getBitmapFromUri(uri)
            val rescaledBitmap = bitmap?.let { scaleBitmap(it, 560) }

            val directory = context.getExternalFilesDir(null)
            val outputFile = File(directory, fileName)

            FileOutputStream(outputFile).use { outputStream ->
                rescaledBitmap?.compress(Bitmap.CompressFormat.JPEG, 90, outputStream)
            }
        }
    }

    fun sendImageToAlpha(serviceId: String?) {
        viewModelScope.launch {
            val bitmap = _uiState.value.imageUri?.let { getBitmapFromUri(it) } ?: return@launch
            sendImageUseCase.sendImageToAlfa(serviceId, bitmap, fileName, MutableSharedFlow<Boolean>())
        }
    }

    fun dispatchSubscriberNextStep(
        dispatchValue: InfoDispatchModel,
        trip: Trip,
        callback: (Trip) -> Unit
    ) {
        viewModelScope.launch {
            if (dispatchValue.requireQr) {
                dispatchRequireQR(dispatchValue)
            } else {
                tripUseCase.endTrip(trip, PaymentMethod.SUBSCRIBER)
                shiftStatusUseCase.getStatus()?.let {
                    ticketUseCase.prepareTicket(trip = trip, dispatch = dispatchValue, shiftStatus = it)
                }
                val status = shiftStatusUseCase.checkForSubStateModifications(ifConstants.STATE_FOR_HIRE)
                shiftStatusUseCase.setStatus(status, true)
            }
        }
    }

    private suspend fun dispatchRequireQR(dispatch: InfoDispatchModel) {
        _effects.emit(CropImageUiEffect.NavigateTo(PaymentDirections.goToScannerQRFragment(dispatch.longDispatchNumber)))
    }

    fun sendSubscriberAuth(serviceId: String?, tripId: Long?) {
        viewModelScope.launch(Dispatchers.IO) {
            val accountPaymentRequest = subscriberUseCase.getAccountPaymentRequest(tripId)
            accountPaymentRequest?.serviceId = serviceId
            subscriberUseCase.sendAccountPaymentRequest(accountPaymentRequest)
        }
    }

    fun isPayment(currentStatus: Int): Boolean = shiftStatusUseCase.isPayment(currentStatus)
    fun isHired(currentStatus: Int): Boolean = shiftStatusUseCase.isHired(currentStatus)

    private fun scaleBitmap(bitmap: Bitmap, newHeight: Int): Bitmap? = try {
        val scaleFactor = newHeight.toFloat() / bitmap.height.toFloat()
        val newWidth = (bitmap.width * scaleFactor).toInt()
        Bitmap.createScaledBitmap(bitmap, newWidth, newHeight, true)
    } catch (e: Exception) {
        null
    }

    private fun generateFileName(isBill: Boolean, numSubscriber: String): String {
        val nameType = if (isBill) "FACTURA" else "VALE"
        val currentDate = SimpleDateFormat("yyyy-MM-dd HH_mm_ss", Locale.US).format(Date())
        return "${nameType}#${numSubscriber}#${currentDate}#canon.png"
    }

    private fun getBitmapFromUri(uri: Uri): Bitmap? {
        return context.contentResolver.openInputStream(uri)?.use { BitmapFactory.decodeStream(it) }
    }
}

package ifac.td.taxi.ui.screen

import android.net.Uri
import androidx.compose.foundation.background
import androidx.compose.foundation.layout.*
import androidx.compose.foundation.shape.RoundedCornerShape

import androidx.compose.ui.res.painterResource
import androidx.compose.ui.unit.dp
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.lifecycle.repeatOnLifecycle
import coil.compose.AsyncImage
import ifac.td.taxi.R
import ifac.td.taxi.viewmodel.*
import kotlinx.coroutines.launch

@Composable
fun CropImageScreen(
    viewModel: CropImageViewModel,
    sharedViewModel: CropImageMainActivityViewModel,
    serviceId: String?,
    onShowHeader: (Boolean) -> Unit,
    onRequestCameraPermission: () -> Unit,
    onOpenCropImage: (Uri?) -> Unit,
    onOpenImageSelect: () -> Unit,
    onShowToast: (Int) -> Unit,
    onNavigateBack: () -> Unit,
    onNavigate: (Any) -> Unit
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val lifecycleOwner = LocalLifecycleOwner.current
    val scope = rememberCoroutineScope()

    LaunchedEffect(Unit) {
        onShowHeader(true)
        onRequestCameraPermission()
    }

    LaunchedEffect(sharedViewModel) {
        lifecycleOwner.repeatOnLifecycle(androidx.lifecycle.Lifecycle.State.STARTED) {
            launch {
                sharedViewModel.cropImageFlow.collect { uri ->
                    uri?.let {
                        viewModel.saveFileFromUri(it)
                    }
                }
            }
        }
    }

    LaunchedEffect(viewModel) {
        lifecycleOwner.repeatOnLifecycle(androidx.lifecycle.Lifecycle.State.STARTED) {
            launch {
                viewModel.effects.collect { effect ->
                    when (effect) {
                        is CropImageUiEffect.OpenCropImage -> onOpenCropImage(effect.uri)
                        CropImageUiEffect.OpenImageSelect -> onOpenImageSelect()
                        is CropImageUiEffect.ShowToast -> onShowToast(effect.resId)
                        CropImageUiEffect.NavigateBack -> onNavigateBack()
                        is CropImageUiEffect.NavigateTo -> onNavigate(effect.direction)
                    }
                }
            }
        }
    }

    Column(modifier = Modifier.fillMaxSize()) {
        Box(
            modifier = Modifier
                .weight(1f)
                .fillMaxWidth()
                .padding(25.dp),
            contentAlignment = Alignment.Center
        ) {
            AsyncImage(
                model = uiState.imageUri,
                contentDescription = null,
                contentScale = ContentScale.Fit,
                modifier = Modifier.fillMaxSize()
            )
        }

        Row(
            modifier = Modifier
                .fillMaxWidth()
                .aspectRatio(5f / 2f)
                .background(Color.Black)
                .padding(6.dp),
            horizontalArrangement = Arrangement.spacedBy(6.dp)
        ) {
            CropImageCustomButton(
                modifier = Modifier.weight(1f),
                state = uiState.buttonsState.selectImage,
                onClick = { viewModel.onEvent(CropImageUiEvent.ClickSelectImage) }
            )
            CropImageCustomButton(
                modifier = Modifier.weight(1f),
                state = uiState.buttonsState.crop,
                onClick = { viewModel.onEvent(CropImageUiEvent.ClickCrop) }
            )
            CropImageCustomButton(
                modifier = Modifier.weight(1f),
                state = uiState.buttonsState.accept,
                onClick = {
                    if (uiState.imageUri == null) {
                        onShowToast(R.string.dialog_voucher_required)
                    } else {
                        viewModel.sendImageToAlpha(serviceId)
                    }
                }
            )
        }
    }
}

@Composable
fun CropImageCustomButton(
    modifier: Modifier = Modifier,
    state: CropImageButtonVisualState,
    onClick: () -> Unit
) {
    val bg = when (state.background) {
        CropImageButtonBackground.GREEN -> Color(0xFF2E7D32)
        CropImageButtonBackground.RED -> Color(0xFFC62828)
        CropImageButtonBackground.BLUE -> Color(0xFF1565C0)
        CropImageButtonBackground.YELLOW -> Color(0xFFF9A825)
        CropImageButtonBackground.ORANGE -> Color(0xFFEF6C00)
        CropImageButtonBackground.DEFAULT -> Color(0xFF1565C0)
        CropImageButtonBackground.BLACK -> Color.Black
        CropImageButtonBackground.TRANSPARENT -> Color.Transparent
    }

    val textColor = when (state.style) {
        CropImageButtonStyle.DISABLE -> Color(0xFFBDBDBD)
        else -> Color.White
    }

    Box(
        modifier = modifier
            .fillMaxHeight()
            .background(bg, RoundedCornerShape(8.dp))
            .clickable(enabled = state.style == CropImageButtonStyle.ENABLE) { onClick() }
            .padding(8.dp),
        contentAlignment = Alignment.Center
    ) {
        if (state.style == CropImageButtonStyle.LOADING) {
            CircularProgressIndicator(color = Color.White, strokeWidth = 2.dp)
        } else {
            Column(horizontalAlignment = Alignment.CenterHorizontally) {
                Image(
                    painter = painterResource(id = state.iconRes),
                    contentDescription = null,
                    modifier = Modifier.size(28.dp)
                )
                Spacer(Modifier.height(6.dp))
                Text(text = state.text, color = textColor, fontSize = 12.sp, maxLines = 2)
            }
        }
    }
}

## 6) Optional lifecycle-safe effect collector helper

```kotlin
package ifac.td.taxi.ui.screen

import androidx.compose.runtime.Composable
import androidx.compose.runtime.LaunchedEffect
import androidx.lifecycle.Lifecycle
import androidx.lifecycle.repeatOnLifecycle
import androidx.lifecycle.compose.LocalLifecycleOwner
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.launch

@Composable
fun <T> CollectWithLifecycle(
    flow: Flow<T>,
    onCollect: suspend (T) -> Unit
) {
    val lifecycleOwner = LocalLifecycleOwner.current
    LaunchedEffect(flow, lifecycleOwner) {
        lifecycleOwner.lifecycle.repeatOnLifecycle(Lifecycle.State.STARTED) {
            launch {
                flow.collect { onCollect(it) }
            }
        }
    }
}
```

## 7) How to preserve the fragment navigation behavior

In your Compose host, pass these lambdas from the Activity/NavController:

- `onShowHeader(true)` → call your activity header method
- `onRequestCameraPermission()` → call your permission launcher
- `onOpenCropImage(uri)` → call the same crop navigation you already use
- `onOpenImageSelect()` → open image picker
- `onShowToast(resId)` → show toast
- `onNavigateBack()` → `navController.popBackStack()`
- `onNavigate(direction)` → `navController.navigate(direction)`

sealed interface CropImageUiEvent {
    data class SetImageUri(val uri: Uri) : CropImageUiEvent
    data object ClickSelectImage : CropImageUiEvent
    data object ClickCrop : CropImageUiEvent
    data object ClickAccept : CropImageUiEvent
    data class CropImageReceived(val uri: Uri?) : CropImageUiEvent
}

sealed interface CropImageUiEffect {
    data class OpenCropImage(val uri: Uri?) : CropImageUiEffect
    data object OpenImageSelect : CropImageUiEffect
    data class ShowToast(val resId: Int) : CropImageUiEffect
    data object NavigateBack : CropImageUiEffect
    data class NavigateTo(val direction: Any) : CropImageUiEffect
}

fun reduceCropImageButtonsState(
    current: CropImageButtonsState,
    event: CropImageUiEvent,
    imageUri: Uri?
): CropImageButtonsState {
    return when (event) {
        CropImageUiEvent.ClickSelectImage -> current.copy(
            selectImage = current.selectImage.copy(
                text = "FOTO",
                iconRes = R.drawable.camera,
                background = CropImageButtonBackground.BLUE,
                style = CropImageButtonStyle.ENABLE,
                visible = true
            )
        )

        CropImageUiEvent.ClickCrop -> current.copy(
            crop = current.crop.copy(
                text = "CROP",
                iconRes = R.drawable.crop,
                background = CropImageButtonBackground.ORANGE,
                style = CropImageButtonStyle.ENABLE,
                visible = true
            )
        )

        CropImageUiEvent.ClickAccept -> current.copy(
            accept = current.accept.copy(
                text = "ACEPTAR",
                iconRes = R.drawable.baseline_check_192,
                background = CropImageButtonBackground.GREEN,
                style = CropImageButtonStyle.ENABLE,
                visible = true
            )
        )

        is CropImageUiEvent.SetImageUri -> current.copy(
            accept = current.accept.copy(
                visible = true,
                style = CropImageButtonStyle.ENABLE
            )
        )

        is CropImageUiEvent.CropImageReceived -> current
    }
}

class CropImageViewModel(
    application: Application,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val tripUseCase: TripUseCase,
    private val subscriberUseCase: SubscriberUseCase,
    private val sendImageUseCase: SendImageUseCase,
    private val ticketUseCase: TicketUseCase
) : BaseComposeViewModel(application) {

    private val TAG = "CropImageViewModel"

    private val _uiState = MutableStateFlow(CropImageUiState())
    val uiState: StateFlow<CropImageUiState> = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<CropImageUiEffect>(extraBufferCapacity = 1)
    val effects: SharedFlow<CropImageUiEffect> = _effects.asSharedFlow()

    private val _uploadImageCallbackFlow = MutableSharedFlow<Boolean>(extraBufferCapacity = 1)
    val uploadImageCallbackFlow: SharedFlow<Boolean> = _uploadImageCallbackFlow.asSharedFlow()

    private var fileName: String = ""

    fun onEvent(event: CropImageUiEvent) {
        _uiState.value = _uiState.value.copy(
            buttonsState = reduceCropImageButtonsState(_uiState.value.buttonsState, event, _uiState.value.imageUri)
        )

        when (event) {
            is CropImageUiEvent.SetImageUri -> {
                _uiState.value = _uiState.value.copy(imageUri = event.uri)
            }

            CropImageUiEvent.ClickSelectImage -> {
                viewModelScope.launch { _effects.emit(CropImageUiEffect.OpenImageSelect) }
            }

            CropImageUiEvent.ClickCrop -> {
                viewModelScope.launch { _effects.emit(CropImageUiEffect.OpenCropImage(_uiState.value.imageUri)) }
            }

            CropImageUiEvent.ClickAccept -> {
                if (_uiState.value.imageUri == null) {
                    viewModelScope.launch { _effects.emit(CropImageUiEffect.ShowToast(R.string.dialog_voucher_required)) }
                }
            }

            is CropImageUiEvent.CropImageReceived -> {
                event.uri?.let { saveFileFromUri(it) }
            }
        }
    }

    fun saveFileFromUri(uri: Uri) {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "saveFileFromUri: Starting for URI: $uri")
            _uiState.value = _uiState.value.copy(imageUri = uri)

            val numSubscriber = subscriberUseCase.getSubscriber()?.numberSubscriberFromW2C
            fileName = generateFileName(false, numSubscriber ?: "")

            val bitmap = getBitmapFromUri(uri)
            val rescaledBitmap = bitmap?.let { scaleBitmap(it, 560) }

            val directory = context.getExternalFilesDir(null)
            val outputFile = File(directory, fileName)

            FileOutputStream(outputFile).use { outputStream ->
                rescaledBitmap?.compress(Bitmap.CompressFormat.JPEG, 90, outputStream)
            }
        }
    }

    fun sendImageToAlpha(serviceId: String?) {
        viewModelScope.launch {
            Logs.d(TAG, "sendImageToAlpha: Sending image for serviceId: $serviceId")
            val bitmap = _uiState.value.imageUri?.let { getBitmapFromUri(it) } ?: return@launch
            sendImageUseCase.sendImageToAlfa(serviceId, bitmap, fileName, _uploadImageCallbackFlow)
        }
    }

    fun dispatchSubscriberNextStep(
        dispatchValue: InfoDispatchModel,
        trip: Trip,
        callback: (Trip) -> Unit
    ) {
        viewModelScope.launch {
            if (dispatchValue.requireQr) {
                dispatchRequireQR(dispatchValue)
            } else {
                tripUseCase.endTrip(trip, PaymentMethod.SUBSCRIBER)
                shiftStatusUseCase.getStatus()?.let {
                    ticketUseCase.prepareTicket(
                        trip = trip,
                        dispatch = dispatchValue,
                        shiftStatus = it
                    )
                }
                val status = shiftStatusUseCase.checkForSubStateModifications(com.interfacom.sdk.taximeter.bravocomm.ifConstants.STATE_FOR_HIRE)
                shiftStatusUseCase.setStatus(status, true)
            }
        }
    }

    private suspend fun dispatchRequireQR(dispatch: InfoDispatchModel) {
        _effects.emit(CropImageUiEffect.NavigateTo(PaymentDirections.goToScannerQRFragment(dispatch.longDispatchNumber)))
    }

    fun sendSubscriberAuth(serviceId: String?, tripId: Long?) {
        viewModelScope.launch(Dispatchers.IO) {
            val accountPaymentRequest = subscriberUseCase.getAccountPaymentRequest(tripId)
            accountPaymentRequest?.serviceId = serviceId
            subscriberUseCase.sendAccountPaymentRequest(accountPaymentRequest)
        }
    }

    fun isPayment(currentStatus: Int): Boolean = shiftStatusUseCase.isPayment(currentStatus)
    fun isHired(currentStatus: Int): Boolean = shiftStatusUseCase.isHired(currentStatus)

    private fun scaleBitmap(bitmap: Bitmap, newHeight: Int): Bitmap? {
        return try {
            val scaleFactor = newHeight.toFloat() / bitmap.height.toFloat()
            val newWidth = (bitmap.width * scaleFactor).toInt()
            Bitmap.createScaledBitmap(bitmap, newWidth, newHeight, true)
        } catch (_: Exception) {
            null
        }
    }

    private fun generateFileName(isBill: Boolean, numSubscriber: String): String {
        val nameType = if (isBill) "FACTURA" else "VALE"
        val currentDate = SimpleDateFormat("yyyy-MM-dd HH_mm_ss", Locale.US).format(Date())
        return "${nameType}#${numSubscriber}#${currentDate}#canon.png"
    }

    private fun getBitmapFromUri(uri: Uri): Bitmap? {
        return context.contentResolver.openInputStream(uri)?.use { BitmapFactory.decodeStream(it) }
    }
}

@Composable
fun CropImageScreen(
    viewModel: CropImageViewModel,
    sharedViewModel: CropImageMainActivityViewModel,
    serviceId: String?,
    onShowHeader: (Boolean) -> Unit,
    onRequestCameraPermission: () -> Unit,
    onOpenCropImage: (Uri?) -> Unit,
    onOpenImageSelect: () -> Unit,
    onShowToast: (Int) -> Unit,
    onNavigateBack: () -> Unit,
    onNavigate: (Any) -> Unit
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val lifecycleOwner = LocalLifecycleOwner.current

    LaunchedEffect(Unit) {
        onShowHeader(true)
        onRequestCameraPermission()
    }

    LaunchedEffect(sharedViewModel) {
        lifecycleOwner.repeatOnLifecycle(Lifecycle.State.STARTED) {
            launch {
                sharedViewModel.cropImageFlow.collect { uri ->
                    uri?.let {
                        viewModel.saveFileFromUri(it)
                    }
                }
            }
        }
    }

    LaunchedEffect(viewModel) {
        lifecycleOwner.repeatOnLifecycle(Lifecycle.State.STARTED) {
            launch {
                viewModel.effects.collect { effect ->
                    when (effect) {
                        is CropImageUiEffect.OpenCropImage -> onOpenCropImage(effect.uri)
                        CropImageUiEffect.OpenImageSelect -> onOpenImageSelect()
                        is CropImageUiEffect.ShowToast -> onShowToast(effect.resId)
                        CropImageUiEffect.NavigateBack -> onNavigateBack()
                        is CropImageUiEffect.NavigateTo -> onNavigate(effect.direction)
                    }
                }
            }
        }
    }

    LaunchedEffect(uiState.imageUri) {
        // keep state reducer aligned with current image
        viewModel.onEvent(CropImageUiEvent.SetImageUri(uiState.imageUri ?: return@LaunchedEffect))
    }

    Column(modifier = Modifier.fillMaxSize()) {
        Box(
            modifier = Modifier
                .weight(1f)
                .fillMaxWidth()
                .padding(25.dp),
            contentAlignment = Alignment.Center
        ) {
            if (uiState.imageUri != null) {
                androidx.compose.foundation.Image(
                    painter = rememberAsyncImagePainterCompat(uiState.imageUri),
                    contentDescription = null,
                    contentScale = ContentScale.Fit,
                    modifier = Modifier.fillMaxSize()
                )
            }
        }

        Row(
            modifier = Modifier
                .fillMaxWidth()
                .aspectRatio(5f / 2f)
                .background(Color.Black)
                .padding(6.dp),
            horizontalArrangement = Arrangement.spacedBy(6.dp)
        ) {
            CropImageCustomButton(
                modifier = Modifier.weight(1f),
                state = uiState.buttonsState.selectImage,
                onClick = { viewModel.onEvent(CropImageUiEvent.ClickSelectImage) }
            )
            CropImageCustomButton(
                modifier = Modifier.weight(1f),
                state = uiState.buttonsState.crop,
                onClick = { viewModel.onEvent(CropImageUiEvent.ClickCrop) }
            )
            CropImageCustomButton(
                modifier = Modifier.weight(1f),
                state = uiState.buttonsState.accept,
                onClick = {
                    if (uiState.imageUri == null) {
                        onShowToast(R.string.dialog_voucher_required)
                    } else {
                        viewModel.sendImageToAlpha(serviceId)
                    }
                }
            )
        }
    }
}

@Composable
fun CropImageCustomButton(
    modifier: Modifier = Modifier,
    state: CropImageButtonVisualState,
    onClick: () -> Unit
) {
    val bg = when (state.background) {
        CropImageButtonBackground.GREEN -> Color(0xFF2E7D32)
        CropImageButtonBackground.RED -> Color(0xFFC62828)
        CropImageButtonBackground.BLUE -> Color(0xFF1565C0)
        CropImageButtonBackground.YELLOW -> Color(0xFFF9A825)
        CropImageButtonBackground.ORANGE -> Color(0xFFEF6C00)
        CropImageButtonBackground.DEFAULT -> Color(0xFF1565C0)
        CropImageButtonBackground.BLACK -> Color.Black
        CropImageButtonBackground.TRANSPARENT -> Color.Transparent
    }

    val textColor = when (state.style) {
        CropImageButtonStyle.DISABLE -> Color(0xFFBDBDBD)
        else -> Color.White
    }

    Box(
        modifier = modifier
            .fillMaxHeight()
            .background(bg, RoundedCornerShape(8.dp))
            .clickable(enabled = state.style == CropImageButtonStyle.ENABLE) { onClick() }
            .padding(8.dp),
        contentAlignment = Alignment.Center
    ) {
        when (state.style) {
            CropImageButtonStyle.LOADING -> CircularProgressIndicator(color = Color.White, strokeWidth = 2.dp)
            else -> {
                Column(horizontalAlignment = Alignment.CenterHorizontally) {
                    Image(
                        painter = painterResource(id = state.iconRes),
                        contentDescription = null,
                        modifier = Modifier.size(28.dp)
                    )
                    Spacer(Modifier.height(6.dp))
                    Text(
                        text = state.text,
                        color = textColor,
                        fontSize = 12.sp,
                        maxLines = 2
                    )
                }
            }
        }
    }
}

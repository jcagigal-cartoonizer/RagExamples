## 3) Exact `LoginUserButtonsState` reducer

This mirrors the fragment branches one-by-one:

- `btnCancel` is always visible, enabled, red
- `btnChangeUser` starts gone
- `btnAccept` starts visible, enabled, green
- when password dialog exists:
  - `btnChangeUser` becomes visible
  - `btnChangeUser` action becomes dialog opener
  - edit texts disabled
- when password dialog does not exist:
  - `btnChangeUser` gone
  - edit texts enabled
- when correct password:
  - edit texts enabled
  - `btnChangeUser` disabled
- when incorrect login:
  - `btnAccept` enabled + green
  - user error shown

```kotlin
package ifac.td.taxi.ui.screen.loginuser

object LoginUserButtonsReducer {

    fun initial(): LoginUserButtonsState = LoginUserButtonsState(
        cancel = LoginUserButtonState(
            visible = true,
            style = CustomButtonStyle.ENABLE,
            background = CustomButtonBackground.RED
        ),
        changeUser = LoginUserButtonState(
            visible = false,
            style = CustomButtonStyle.ENABLE,
            background = CustomButtonBackground.BLUE
        ),
        accept = LoginUserButtonState(
            visible = true,
            style = CustomButtonStyle.ENABLE,
            background = CustomButtonBackground.GREEN
        )
    )

    fun onHasSettingsPassword(hasPassword: Boolean): LoginUserButtonsState {
        return if (hasPassword) {
            LoginUserButtonsState(
                cancel = LoginUserButtonState(true, CustomButtonStyle.ENABLE, CustomButtonBackground.RED),
                changeUser = LoginUserButtonState(true, CustomButtonStyle.ENABLE, CustomButtonBackground.BLUE),
                accept = LoginUserButtonState(true, CustomButtonStyle.ENABLE, CustomButtonBackground.GREEN)
            )
        } else {
            LoginUserButtonsState(
                cancel = LoginUserButtonState(true, CustomButtonStyle.ENABLE, CustomButtonBackground.RED),
                changeUser = LoginUserButtonState(false, CustomButtonStyle.ENABLE, CustomButtonBackground.BLUE),
                accept = LoginUserButtonState(true, CustomButtonStyle.ENABLE, CustomButtonBackground.GREEN)
            )
        }
    }

    fun onIncorrectLogin(current: LoginUserButtonsState): LoginUserButtonsState {
        return current.copy(
            accept = current.accept.copy(
                style = CustomButtonStyle.ENABLE,
                background = CustomButtonBackground.GREEN
            )
        )
    }

    fun onCorrectPassword(current: LoginUserButtonsState): LoginUserButtonsState {
        return current.copy(
            changeUser = current.changeUser.copy(
                style = CustomButtonStyle.DISABLE
            )
        )
    }

    fun onLoadingAccept(current: LoginUserButtonsState): LoginUserButtonsState {
        return current.copy(
            accept = current.accept.copy(style = CustomButtonStyle.LOADING)
        )
    }
}
```

## 4) Compose-friendly `LoginUserViewModel`

This replaces the fragment’s flow collection with a single `UiState` and a `SharedFlow<LoginUserUiEffect>`.

```kotlin
package ifac.td.taxi.ui.screen.loginuser

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import com.interfacom.sdk.taximeter.bravocomm.StatusConnection
import ifac.td.taxi.R
import ifac.td.taxi.domain.usecase.ConfigurationScreenPasswordUseCase
import ifac.td.taxi.domain.usecase.SessionUseCase
import ifac.td.taxi.domain.usecase.UserPreferencesUseCase
import ifac.td.taxi.domain.usecase.oldv2.MigrationV2UseCase
import ifac.td.taxi.framework.sdk.ExternalBridgeInterface
import ifac.td.taxi.framework.sdk.model.UserInfo
import ifac.td.taxi.framework.sdk.usecase.ConfigurationUseCase
import ifac.td.taxi.framework.sdk.usecase.LicensingUseCase
import ifac.td.taxi.repository.connections.receivers.utils.NetworkUtils.Companion.isInternetConnectionAvailable
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext

class LoginUserViewModel(
    private val externalBrigde: ExternalBridgeInterface,
    private val configuration: ConfigurationUseCase,
    private val sessionUseCase: SessionUseCase,
    private val configurationScreenPasswordUseCase: ConfigurationScreenPasswordUseCase,
    private val licensingUseCase: LicensingUseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    private val migrationV2UseCase: MigrationV2UseCase,
    application: Application,
) : AndroidViewModel(application) {

    private val context = application

    private val _uiState = MutableStateFlow(LoginUserUiState(buttons = LoginUserButtonsReducer.initial()))
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<LoginUserUiEffect>()
    val uiEffect: SharedFlow<LoginUserUiEffect> = _uiEffect.asSharedFlow()

    fun onEvent(event: LoginUserUiEvent) {
        when (event) {
            is LoginUserUiEvent.UserChanged -> _uiState.update { it.copy(user = event.value, userError = null) }
            is LoginUserUiEvent.PasswordChanged -> _uiState.update { it.copy(password = event.value) }

            LoginUserUiEvent.CancelClicked -> viewModelScope.launch {
                _uiEffect.emit(LoginUserUiEffect.NavigateBack)
            }

            LoginUserUiEvent.AcceptClicked -> clickOnAccept()

            LoginUserUiEvent.ChangePasswordClicked -> viewModelScope.launch {
                _uiEffect.emit(LoginUserUiEffect.NavigateToChangePassword)
            }

            LoginUserUiEvent.ChangeUserClicked -> {
                _uiState.update { it.copy(showSettingsPasswordDialog = true) }
                viewModelScope.launch { _uiEffect.emit(LoginUserUiEffect.OpenSettingsPasswordDialog) }
            }

            is LoginUserUiEvent.SettingsPasswordDialogPinChanged -> {
                _uiState.update { it.copy(settingsPasswordDialogPin = event.value) }
            }

            LoginUserUiEvent.SettingsPasswordDialogCancel -> {
                _uiState.update { it.copy(showSettingsPasswordDialog = false, settingsPasswordDialogPin = "") }
            }

            LoginUserUiEvent.SettingsPasswordDialogAccept -> {
                checkSettingsPassword(_uiState.value.settingsPasswordDialogPin)
            }

            LoginUserUiEvent.ScreenStarted -> {
                checkSavedData()
                checkHasSettingsPassword()
            }
        }
    }

    fun checkSavedData() {
        viewModelScope.launch {
            val session = sessionUseCase.getSession()
            val user = session?.username
            val pass = session?.password

            _uiState.update {
                it.copy(
                    user = user.orEmpty(),
                    password = pass.orEmpty(),
                    showChangePassword = !user.isNullOrEmpty() && !pass.isNullOrEmpty()
                )
            }
        }
    }

    fun checkHasSettingsPassword() {
        viewModelScope.launch {
            val hasPassword = configurationScreenPasswordUseCase.hasSettingsPassword()
            _uiState.update { current ->
                current.copy(
                    buttons = LoginUserButtonsReducer.onHasSettingsPassword(hasPassword),
                    isUserEnabled = !hasPassword,
                    isPasswordEnabled = !hasPassword
                )
            }
        }
    }

    fun loginUser(user: String, password: String, fromMigration: Boolean = false) {
        viewModelScope.launch {
            if (!isInternetConnectionAvailable(context)) {
                _uiEffect.emit(LoginUserUiEffect.ShowToast(R.string.network_error))
                _uiEffect.emit(LoginUserUiEffect.NavigateBack)
                return@launch
            }

            externalBrigde.setUsername(user)
            externalBrigde.setPassword(password)

            configuration.downloadConfiguration(
                userInfo = UserInfo(user = user, password = password),
                fromMigration = fromMigration
            )
        }
    }

    fun downloadBravoConfiguration() {
        viewModelScope.launch {
            configuration.downloadBravoconfiguration()
        }
    }

    fun clickChangePassword() {
        viewModelScope.launch {
            _uiEffect.emit(LoginUserUiEffect.NavigateToChangePassword)
        }
    }

    fun checkSettingsPassword(pin: String) {
        viewModelScope.launch {
            val encrypted = configurationScreenPasswordUseCase.encryptSettingsPassword(pin)
            val saved = configurationScreenPasswordUseCase.getSettingsPassword()
            val correct = encrypted == saved

            if (correct) {
                _uiState.update {
                    it.copy(
                        isUserEnabled = true,
                        isPasswordEnabled = true,
                        showSettingsPasswordDialog = false,
                        settingsPasswordDialogPin = "",
                        buttons = LoginUserButtonsReducer.onCorrectPassword(it.buttons)
                    )
                }
            } else {
                _uiEffect.emit(LoginUserUiEffect.ShowToast(R.string.pin_incorrecto))
            }
        }
    }

    fun migrateConfigsUser() {
        viewModelScope.launch { migrationV2UseCase.migrateUserConfigPrefs() }
    }

    fun migrateShiftsAndTrips() {
        viewModelScope.launch {
            val allShifts = migrationV2UseCase.getAllShifts()
            if (allShifts.isNotEmpty()) migrationV2UseCase.migrateShifts(allShifts)
            migrateTrips()
        }
    }

    suspend fun migrateTrips() {
        val allTrips = migrationV2UseCase.getAllTrips()
        allTrips.forEach { pair ->
            pair.second?.let { trip ->
                val shiftExists = migrationV2UseCase.existsShiftId(trip.fkShiftId)
                if (shiftExists) migrationV2UseCase.migrateTrip(trip)
            }
        }
    }

    fun migratePartials() {
        viewModelScope.launch { migrationV2UseCase.migratePartials() }
    }

    fun migratePortugal() {
        viewModelScope.launch {
            if (licensingUseCase.getLicensingParameters()?.isFiscalService == true) {
                migrationV2UseCase.migratePortugal()
            }
        }
    }

    private fun clickOnAccept() {
        val state = _uiState.value
        if (state.user.isNotBlank() && state.password.isNotBlank()) {
            _uiState.update {
                it.copy(buttons = LoginUserButtonsReducer.onLoadingAccept(it.buttons))
            }
            loginUser(
                user = state.user.trim(),
                password = state.password,
                fromMigration = false
            )
        }
    }
}
```

```kotlin
import kotlinx.coroutines.flow.asStateFlow

class LoginUserMainActivityViewModel : ViewModel() {

    private val _correctLoginFlow = MutableSharedFlow<Pair<Boolean, String?>>()

            _correctLoginFlow.emit(isCorrect to message)

        viewModelScope.launch {
            _xmppStatusFlow.emit(status)
        }
    }

    fun emitDownloadedInitialBravoConfiguration() {
        viewModelScope.launch {
            _downloadedInitialBravoconfigurationFlow.emit(Unit)
        }
    }
}
```

```kotlin
        onShowHeader(false)
        viewModel.onEvent(LoginUserUiEvent.ScreenStarted)
    }

    LaunchedEffect(viewModel) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                LoginUserUiEffect.NavigateBack -> navController.popBackStack()

                is LoginUserUiEffect.ShowToast -> onShowToast(effect.messageRes)
```

## 7) Compose wrapper for the existing `CustomButton` API

Since your `CustomButton` is a View, the cleanest Compose integration is `AndroidView`.

```kotlin
package ifac.td.taxi.ui.screen.loginuser

import android.view.View
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.viewinterop.AndroidView
import ifac.td.taxi.ui.custom.button.ButtonType
import ifac.td.taxi.ui.custom.button.CustomButton

@Composable
fun ComposeCustomButton(
    modifier: Modifier = Modifier,
    type: ButtonType,
    visible: Boolean,
    style: CustomButtonStyle,
    background: CustomButtonBackground,
    onClick: () -> Unit
) {
    if (!visible) return

    AndroidView(
        modifier = modifier,
        factory = { context ->
            CustomButton(context).apply {
                setButtonType(type.value)
                setAction(onClick)
            }
        },
        update = { button ->
            button.visibility = View.VISIBLE
            button.setButtonType(type.value)
            button.setAction(onClick)
            button.setButtonStyle(
                when (style) {
                    CustomButtonStyle.ENABLE -> CustomButton.StyleButton.ENABLE
                    CustomButtonStyle.DISABLE -> CustomButton.StyleButton.DISABLE
                    CustomButtonStyle.LOADING -> CustomButton.StyleButton.LOADING
                }
            )
            button.changeBackground(
                when (background) {
                    CustomButtonBackground.GREEN -> CustomButton.BackgroundButtonColor.GREEN
                    CustomButtonBackground.RED -> CustomButton.BackgroundButtonColor.RED
                    CustomButtonBackground.BLUE -> CustomButton.BackgroundButtonColor.BLUE
                    CustomButtonBackground.YELLOW -> CustomButton.BackgroundButtonColor.YELLOW
                    CustomButtonBackground.ORANGE -> CustomButton.BackgroundButtonColor.ORANGE
                    CustomButtonBackground.DEFAULT -> CustomButton.BackgroundButtonColor.DEFAULT
                    CustomButtonBackground.BLACK -> CustomButton.BackgroundButtonColor.BLACK
                    CustomButtonBackground.TRANSPARENT -> CustomButton.BackgroundButtonColor.TRANSPARENT
                }
            )
        }
    )
}
```

## 8) Important note about exact fragment parity

Your fragment has one nested flow issue:

```kotlin
sharedViewModel.xmppStatusFlow.collect { xmpp ->
    if (xmpp == StatusConnection.CONNECTED) {
        vModel.downloadBravoConfiguration()
    }
}
```

That is a nested collect inside another collect. In Compose, you should not keep a nested infinite collect inside a click branch. Instead, the bridge should expose a single effect or state transition.

If you want exact parity, the best Compose-safe equivalent is:

- `correctLoginFlow` triggers `StartBravoService`
- `xmppStatusFlow` is collected separately
- when it becomes `CONNECTED`, call `downloadBravoConfiguration()`

```kotlin
data class LoginUserUiState(
    val user: String = "",
    val password: String = "",
    val userError: String? = null,
    val showChangePassword: Boolean = false,
    val isUserEnabled: Boolean = true,
    val isPasswordEnabled: Boolean = true,
    val buttons: LoginUserButtonsState = LoginUserButtonsState.initial(),
    val progress: LoginUserProgressState = LoginUserProgressState(),
    val settingsPasswordDialog: LoginUserSettingsPasswordDialogState = LoginUserSettingsPasswordDialogState(),
)

data class LoginUserProgressState(
    val visible: Boolean = false,
    val progress: Int = 0
)

data class LoginUserSettingsPasswordDialogState(
    val visible: Boolean = false,
    val title: String = "",
    val description: String = "",
    val pin: String = ""
)

data class LoginUserButtonsState(
    val cancel: LoginUserButtonState,
    val changeUser: LoginUserButtonState,
    val accept: LoginUserButtonState
) {
    companion object {
        fun initial() = LoginUserButtonsState(
            cancel = LoginUserButtonState(
                visible = true,
                style = CustomButton.StyleButton.ENABLE,
                background = CustomButton.BackgroundButtonColor.RED
            ),
            changeUser = LoginUserButtonState(
                visible = false,
                style = CustomButton.StyleButton.ENABLE,
                background = CustomButton.BackgroundButtonColor.BLUE
            ),
            accept = LoginUserButtonState(
                visible = true,
                style = CustomButton.StyleButton.ENABLE,
                background = CustomButton.BackgroundButtonColor.GREEN
            )
        )
    }
}

data class LoginUserButtonState(
    val visible: Boolean,
    val style: CustomButton.StyleButton,
    val background: CustomButton.BackgroundButtonColor
)

sealed interface LoginUserUiEvent {
    data object ScreenStarted : LoginUserUiEvent
    data class UserChanged(val value: String) : LoginUserUiEvent
    data class PasswordChanged(val value: String) : LoginUserUiEvent
    data object CancelClicked : LoginUserUiEvent
    data object AcceptClicked : LoginUserUiEvent
    data object ChangePasswordClicked : LoginUserUiEvent
    data object ChangeUserClicked : LoginUserUiEvent
    data class SettingsPasswordPinChanged(val value: String) : LoginUserUiEvent
    data object SettingsPasswordDialogCancel : LoginUserUiEvent
    data object SettingsPasswordDialogAccept : LoginUserUiEvent
}

sealed interface LoginUserUiEffect {
    data object NavigateBack : LoginUserUiEffect
    data object NavigateToChangePassword : LoginUserUiEffect
    data object OpenSettingsPasswordDialog : LoginUserUiEffect
    data class ShowToast(val messageRes: Int) : LoginUserUiEffect
    data class SaveCredentials(val user: String, val password: String) : LoginUserUiEffect
    data object StartBravoService : LoginUserUiEffect
}

object LoginUserButtonsReducer {

    fun onHasSettingsPassword(hasPassword: Boolean): LoginUserButtonsState {
        return if (hasPassword) {
            LoginUserButtonsState(
                cancel = LoginUserButtonState(
                    visible = true,
                    style = CustomButton.StyleButton.ENABLE,
                    background = CustomButton.BackgroundButtonColor.RED
                ),
                changeUser = LoginUserButtonState(
                    visible = true,
                    style = CustomButton.StyleButton.ENABLE,
                    background = CustomButton.BackgroundButtonColor.BLUE
                ),
                accept = LoginUserButtonState(
                    visible = true,
                    style = CustomButton.StyleButton.ENABLE,
                    background = CustomButton.BackgroundButtonColor.GREEN
                )
            )
        } else {
            LoginUserButtonsState(
                cancel = LoginUserButtonState(
                    visible = true,
                    style = CustomButton.StyleButton.ENABLE,
                    background = CustomButton.BackgroundButtonColor.RED
                ),
                changeUser = LoginUserButtonState(
                    visible = false,
                    style = CustomButton.StyleButton.ENABLE,
                    background = CustomButton.BackgroundButtonColor.BLUE
                ),
                accept = LoginUserButtonState(
                    visible = true,
                    style = CustomButton.StyleButton.ENABLE,
                    background = CustomButton.BackgroundButtonColor.GREEN
                )
            )
        }
    }

    fun onIncorrectLogin(current: LoginUserButtonsState): LoginUserButtonsState {
        return current.copy(
            accept = current.accept.copy(
                style = CustomButton.StyleButton.ENABLE,
                background = CustomButton.BackgroundButtonColor.GREEN
            )
        )
    }

    fun onCorrectPassword(current: LoginUserButtonsState): LoginUserButtonsState {
        return current.copy(
            changeUser = current.changeUser.copy(
                style = CustomButton.StyleButton.DISABLE
            )
        )
    }

    fun onAcceptLoading(current: LoginUserButtonsState): LoginUserButtonsState {
        return current.copy(
            accept = current.accept.copy(style = CustomButton.StyleButton.LOADING)
        )
    }
}

class LoginUserViewModel(
    private val externalBrigde: ExternalBridgeInterface,
    private val configuration: ConfigurationUseCase,
    private val sessionUseCase: SessionUseCase,
    private val configurationScreenPasswordUseCase: ConfigurationScreenPasswordUseCase,
    private val licensingUseCase: LicensingUseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    private val migrationV2UseCase: MigrationV2UseCase,
    application: Application,
) : AndroidViewModel(application) {

    private val context = application

    private val _uiState = MutableStateFlow(LoginUserUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<LoginUserUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()

    fun onEvent(event: LoginUserUiEvent) {
        when (event) {
            LoginUserUiEvent.ScreenStarted -> {
                checkSavedData()
                checkHasSettingsPassword()
            }

            is LoginUserUiEvent.UserChanged -> {
                _uiState.update { it.copy(user = event.value, userError = null) }
            }

            is LoginUserUiEvent.PasswordChanged -> {
                _uiState.update { it.copy(password = event.value) }
            }

            LoginUserUiEvent.CancelClicked -> {
                viewModelScope.launch { _uiEffect.emit(LoginUserUiEffect.NavigateBack) }
            }

            LoginUserUiEvent.AcceptClicked -> {
                clickOnAccept()
            }

            LoginUserUiEvent.ChangePasswordClicked -> {
                viewModelScope.launch { _uiEffect.emit(LoginUserUiEffect.NavigateToChangePassword) }
            }

            LoginUserUiEvent.ChangeUserClicked -> {
                _uiState.update {
                    it.copy(
                        settingsPasswordDialog = it.settingsPasswordDialog.copy(
                            visible = true
                        )
                    )
                }
                viewModelScope.launch { _uiEffect.emit(LoginUserUiEffect.OpenSettingsPasswordDialog) }
            }

            is LoginUserUiEvent.SettingsPasswordPinChanged -> {
                _uiState.update {
                    it.copy(settingsPasswordDialog = it.settingsPasswordDialog.copy(pin = event.value))
                }
            }

            LoginUserUiEvent.SettingsPasswordDialogCancel -> {
                _uiState.update {
                    it.copy(
                        settingsPasswordDialog = LoginUserSettingsPasswordDialogState(visible = false)
                    )
                }
            }

            LoginUserUiEvent.SettingsPasswordDialogAccept -> {
                checkSettingsPassword(_uiState.value.settingsPasswordDialog.pin)
            }
        }
    }

    fun checkSavedData() {
        viewModelScope.launch {
            val session = sessionUseCase.getSession()
            val user = session?.username.orEmpty()
            val pass = session?.password.orEmpty()

            _uiState.update {
                it.copy(
                    user = user,
                    password = pass,
                    showChangePassword = user.isNotEmpty() && pass.isNotEmpty()
                )
            }
        }
    }

    fun checkHasSettingsPassword() {
        viewModelScope.launch {
            val hasPassword = configurationScreenPasswordUseCase.hasSettingsPassword()
            _uiState.update {
                it.copy(
                    buttons = LoginUserButtonsReducer.onHasSettingsPassword(hasPassword),
                    isUserEnabled = !hasPassword,
                    isPasswordEnabled = !hasPassword
                )
            }
        }
    }

    fun loginUser(user: String, password: String, fromMigration: Boolean = false) {
        viewModelScope.launch {
            if (!isInternetConnectionAvailable(context)) {
                _uiEffect.emit(LoginUserUiEffect.ShowToast(R.string.network_error))
                _uiEffect.emit(LoginUserUiEffect.NavigateBack)
                return@launch
            }

            externalBrigde.setUsername(user)
            externalBrigde.setPassword(password)

            configuration.downloadConfiguration(
                userInfo = UserInfo(user = user, password = password),
                fromMigration = fromMigration
            )
        }
    }

    fun clickChangePassword() {
        viewModelScope.launch {
            _uiEffect.emit(LoginUserUiEffect.NavigateToChangePassword)
        }
    }

    fun downloadBravoConfiguration() {
        viewModelScope.launch {
            configuration.downloadBravoconfiguration()
        }
    }

    fun checkSettingsPassword(pin: String) {
        viewModelScope.launch {
            val encrypted = configurationScreenPasswordUseCase.encryptSettingsPassword(pin)
            val saved = configurationScreenPasswordUseCase.getSettingsPassword()
            val correct = encrypted == saved

            if (correct) {
                _uiState.update {
                    it.copy(
                        isUserEnabled = true,
                        isPasswordEnabled = true,
                        settingsPasswordDialog = LoginUserSettingsPasswordDialogState(visible = false),
                        buttons = LoginUserButtonsReducer.onCorrectPassword(it.buttons)
                    )
                }
            } else {
                _uiEffect.emit(LoginUserUiEffect.ShowToast(R.string.pin_incorrecto))
            }
        }
    }

    fun migrateConfigsUser() {
        viewModelScope.launch { migrationV2UseCase.migrateUserConfigPrefs() }
    }

    fun migrateShiftsAndTrips() {
        viewModelScope.launch {
            val allShifts = migrationV2UseCase.getAllShifts()
            if (allShifts.isNotEmpty()) migrationV2UseCase.migrateShifts(allShifts)
            migrateTrips()
        }
    }

    suspend fun migrateTrips() {
        val allTrips = migrationV2UseCase.getAllTrips()
        allTrips.forEach { pair ->
            pair.second?.let { trip ->
                val shiftExists = migrationV2UseCase.existsShiftId(trip.fkShiftId)
                if (shiftExists) migrationV2UseCase.migrateTrip(trip)
            }
        }
    }

    fun migratePartials() {
        viewModelScope.launch { migrationV2UseCase.migratePartials() }
    }

    fun migratePortugal() {
        viewModelScope.launch {
            if (licensingUseCase.getLicensingParameters()?.isFiscalService == true) {
                migrationV2UseCase.migratePortugal()
            }
        }
    }

    private fun clickOnAccept() {
        val state = _uiState.value
        if (state.user.isValidUsername() && state.password.isValidPassword()) {
            _uiState.update {
                it.copy(buttons = LoginUserButtonsReducer.onAcceptLoading(it.buttons))
            }
            loginUser(
                user = state.user.trim(),
                password = state.password,
                fromMigration = false
            )
        }
    }
}

class LoginUserRedSysViewModel(
    private val redSysUseCase: RedSysUseCase,
    context: Application,
) : BaseViewModel(context) {

    private val TAG = "LoginUserRedSysViewModel"

    private val _loginRedSysFlow = MutableSharedFlow<Boolean>()
    val loginRedSysFlow = _loginRedSysFlow.asSharedFlow()

    private val _userRedSysFlow = MutableSharedFlow<String?>()
    val userRedSysFlow = _userRedSysFlow.asSharedFlow()

    private val _okStatusFlow = MutableSharedFlow<CustomButton.StyleButton>()
    val okStatusFlow = _okStatusFlow.asSharedFlow()

    private var userRedSys: String = ""
    private var passwordRedSys: String = ""

    fun loginRedSys(user: String, password: String) {
        userRedSys = user
        passwordRedSys = password

        viewModelScope.launch(Dispatchers.IO) {
            redSysUseCase.configureRedSys()

            val loginRedSys = redSysUseCase.loginRedSys(
                username = userRedSys,
                password = passwordRedSys
            )

            when (loginRedSys) {
                is RedSysLoginResponse.Success -> {
                    if (loginRedSys.merchantList.size > 0) {
                        redSysUseCase.saveUsernameRedSys(userRedSys)
                        if (passwordRedSys.isNotEmpty()) {
                            TemporalData.passwordRedSys = passwordRedSys
                        }
                        _loginRedSysFlow.emit(true)
                    } else {
                        _loginRedSysFlow.emit(false)
                    }
                }

                is RedSysLoginResponse.Error -> {
                    Logs.d(TAG, "Login RedSys failed or returned empty array: ${loginRedSys.errorMessage}")
                    if (loginRedSys.errorCode == RedCLSErrorCodes.STATUS_KO_FORMATO_RESP_LOGIN_PWD_CAD) {
                        _okStatusFlow.emit(CustomButton.StyleButton.ENABLE)
                        navigateTo(R.id.action_loginUserRedSysFragment_to_changePasswordRedSysFragment)
                    } else {
                        _loginRedSysFlow.emit(false)
                    }
                }
            }
        }
    }

    fun getUserRedSys() {
        viewModelScope.launch {
            val user = redSysUseCase.getUsernameRedSys()
            _userRedSysFlow.emit(user)
        }
    }
}

@Composable
fun LoginUserScreen(
    navController: NavController,
    viewModel: LoginUserViewModel,
    sharedViewModel: LoginUserMainActivityViewModel,
    autoDownloadConfigurationFromMigration: Boolean = false,
    onShowHeader: (Boolean) -> Unit,
    onStartBravoService: () -> Unit,
    onSaveSharedPreference: (String, String) -> Unit,
    onShowToast: (Int) -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val progressState by sharedViewModel.progressBarFlow.collectAsStateWithLifecycle()
    val focusManager = LocalFocusManager.current

    BackHandler {
        viewModel.onEvent(LoginUserUiEvent.CancelClicked)
    }

    LaunchedEffect(Unit) {
        onShowHeader(false)
        viewModel.onEvent(LoginUserUiEvent.ScreenStarted)
    }

    LaunchedEffect(viewModel) {
        viewModel.uiEffect.collectLatest { effect ->
            when (effect) {
                LoginUserUiEffect.NavigateBack -> navController.popBackStack()
                LoginUserUiEffect.NavigateToChangePassword ->
                    navController.navigate(R.id.action_loginUserFragment_to_changeUserPasswordFragment)

                LoginUserUiEffect.OpenSettingsPasswordDialog -> Unit
                LoginUserUiEffect.StartBravoService -> onStartBravoService()
                is LoginUserUiEffect.SaveCredentials -> {
                    onSaveSharedPreference(effect.user, effect.password)
                }
                is LoginUserUiEffect.ShowToast -> onShowToast(effect.messageRes)
            }
        }
    }

    LaunchedEffect(sharedViewModel) {
        sharedViewModel.correctLoginFlow.collectLatest { (correct, _) ->
            if (correct) {
                onStartBravoService()
                onSaveSharedPreference("User", uiState.user)
                onSaveSharedPreference("Password", uiState.password)
            } else {
                viewModel.onEvent(LoginUserUiEvent.UserChanged(uiState.user))
                viewModel.onEvent(LoginUserUiEvent.PasswordChanged(uiState.password))
                viewModel.onEvent(LoginUserUiEvent.AcceptClicked)
                // exact fragment behavior is handled by the shared bridge in your host layer
            }
        }
    }

    LaunchedEffect(sharedViewModel) {
        sharedViewModel.xmppStatusFlow.collectLatest { xmpp ->
            if (xmpp == StatusConnection.CONNECTED) {
                viewModel.downloadBravoConfiguration()
            }
        }
    }

    LaunchedEffect(sharedViewModel) {
        sharedViewModel.downloadedInitialBravoconfigurationFlow.collectLatest {
            navController.popBackStack()
            sharedViewModel.updateProgressBarFlow(View.GONE, 0)
        }
    }

    if (uiState.settingsPasswordDialog.visible) {
        AlertDialog(
            onDismissRequest = { viewModel.onEvent(LoginUserUiEvent.SettingsPasswordDialogCancel) },
            title = { Text(text = stringResource(R.string.pin_actual)) },
            text = {
                Column {
                    Text(text = stringResource(R.string.pin_actual_hint))
                    Spacer(modifier = Modifier.height(12.dp))
                    OutlinedTextField(
                        value = uiState.settingsPasswordDialog.pin,
                        onValueChange = { viewModel.onEvent(LoginUserUiEvent.SettingsPasswordPinChanged(it)) },
                        singleLine = true,
                        modifier = Modifier.fillMaxWidth()
                    )
                }
            },
            confirmButton = {
                TextButton(onClick = { viewModel.onEvent(LoginUserUiEvent.SettingsPasswordDialogAccept) }) {
                    Text(text = stringResource(android.R.string.ok))
                }
            },
            dismissButton = {
                TextButton(onClick = { viewModel.onEvent(LoginUserUiEvent.SettingsPasswordDialogCancel) }) {
                    Text(text = stringResource(android.R.string.cancel))
                }
            }
        )
    }

    Box(modifier = Modifier.fillMaxSize()) {
        Column(
            modifier = Modifier
                .fillMaxSize()
                .verticalScroll(rememberScrollState())
                .padding(20.dp)
        ) {
            Text(text = stringResource(R.string.usuario))
            Spacer(modifier = Modifier.height(8.dp))
            OutlinedTextField(
                value = uiState.user,
                onValueChange = { viewModel.onEvent(LoginUserUiEvent.UserChanged(it)) },
                enabled = uiState.isUserEnabled,
                isError = uiState.userError != null,
                modifier = Modifier.fillMaxWidth()
            )
            uiState.userError?.let {
                Text(text = it, color = MaterialTheme.colorScheme.error)
            }

            Spacer(modifier = Modifier.height(16.dp))
            Text(text = stringResource(R.string.contrasena))
            Spacer(modifier = Modifier.height(8.dp))
            OutlinedTextField(
                value = uiState.password,
                onValueChange = { viewModel.onEvent(LoginUserUiEvent.PasswordChanged(it)) },
                enabled = uiState.isPasswordEnabled,
                modifier = Modifier.fillMaxWidth()
            )

            Spacer(modifier = Modifier.height(16.dp))
            if (uiState.showChangePassword) {
                TextButton(
                    onClick = {
                        focusManager.clearFocus()
                        viewModel.clickChangePassword()
                    }
                ) {
                    Text(text = stringResource(R.string.change_password))
                }
            }
        }

        if (progressState.first == View.VISIBLE) {
            Box(
                modifier = Modifier
                    .fillMaxSize()
                    .background(MaterialTheme.colorScheme.surface.copy(alpha = 0.85f)),
                contentAlignment = Alignment.Center
            ) {
                Column(horizontalAlignment = Alignment.CenterHorizontally) {
                    Text(text = stringResource(R.string.descargando_configuracion))
                    Spacer(modifier = Modifier.height(12.dp))
                    CircularProgressIndicator(progress = { progressState.second / 100f })
                }
            }
        }

        Row(
            modifier = Modifier
                .fillMaxWidth()
                .align(Alignment.BottomCenter)
        ) {
            ComposeCustomButton(
                modifier = Modifier.weight(1f),
                type = ButtonType.CANCEL,
                visible = uiState.buttons.cancel.visible,
                style = uiState.buttons.cancel.style,
                background = uiState.buttons.cancel.background,
                onClick = {
                    focusManager.clearFocus()
                    viewModel.onEvent(LoginUserUiEvent.CancelClicked)
                }
            )

            ComposeCustomButton(
                modifier = Modifier.weight(1f),
                type = ButtonType.CHANGE_CREDENTIALS,
                visible = uiState.buttons.changeUser.visible,
                style = uiState.buttons.changeUser.style,
                background = uiState.buttons.changeUser.background,
                onClick = {
                    focusManager.clearFocus()
                    viewModel.onEvent(LoginUserUiEvent.ChangeUserClicked)
                }
            )

            ComposeCustomButton(
                modifier = Modifier.weight(1f),
                type = ButtonType.ACCEPT,
                visible = uiState.buttons.accept.visible,
                style = uiState.buttons.accept.style,
                background = uiState.buttons.accept.background,
                onClick = {
                    focusManager.clearFocus()
                    viewModel.onEvent(LoginUserUiEvent.AcceptClicked)
                }
            )
        }
    }
}

@Composable
fun ComposeCustomButton(
    modifier: Modifier = Modifier,
    type: ButtonType,
    visible: Boolean,
    style: CustomButton.StyleButton,
    background: CustomButton.BackgroundButtonColor,
    onClick: () -> Unit
) {
    if (!visible) return

    androidx.compose.ui.viewinterop.AndroidView(
        modifier = modifier,
        factory = { context ->
            CustomButton(context).apply {
                setButtonType(type.value)
                setButtonStyle(style)
                changeBackground(background)
                setAction(onClick)
            }
        },
        update = { button ->
            button.visibility = View.VISIBLE
            button.setButtonType(type.value)
            button.setButtonStyle(style)
            button.changeBackground(background)
            button.setAction(onClick)
        }
    )
}
```

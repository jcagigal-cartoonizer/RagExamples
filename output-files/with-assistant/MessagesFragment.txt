```kotlin
@Composable
fun MessagesScreen(
    viewModel: MessagesViewModel,
    mainActivityViewModel: MessagesMainActivityViewModel,
    onNavigateBack: () -> Unit,
    onNavigateToMessageDetail: (messageId: Long, fromMessages: Boolean) -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val buttonsState = uiState.buttonsState
    val dialogState = uiState.dialogState

    LaunchedEffect(Unit) {
        viewModel.onEvent(MessagesUiEvent.LoadMessagesByDriverId)
    }

    LaunchedEffect(Unit) {
        viewModel.effects.collectLatest { effect ->
            when (effect) {
                is MessagesUiEffect.NavigateBack -> onNavigateBack()
                is MessagesUiEffect.NavigateToMessageDetail ->
                    onNavigateToMessageDetail(effect.messageId, true)

                is MessagesUiEffect.OpenMarkAllReadDialog -> {
                    mainActivityViewModel.requestOpenCustomDialog(
                        CustomDialog.CustomDialogModel(
                            title = effect.title,
                            description = effect.description,
                            buttons = arrayListOf(ButtonType.CANCEL, ButtonType.ACCEPT)
                        )
                    )
                }
            }
        }
    }

    LaunchedEffect(Unit) {
        mainActivityViewModel.dialogEffect.collectLatest { dialogEffect ->
            when (dialogEffect) {
                is MessagesMainActivityViewModel.MessagesDialogEffect.ConfirmMarkAllRead -> {
                    viewModel.onEvent(MessagesUiEvent.MarkAllMessagesRead)
                }
            }
        }
    }

    Column(modifier = Modifier.fillMaxSize()) {
        LazyColumn(
            modifier = Modifier
                .weight(1f)
                .fillMaxWidth()
        ) {
            items(uiState.items) { item ->
                when (item) {
                    is MessagesListItemUi.Date -> {
                        Text(
                            text = item.formattedDate,
                            modifier = Modifier
                                .fillMaxWidth()
                                .background(Color.LightGray)
                                .padding(8.dp)
                        )
                    }

                    is MessagesListItemUi.General -> {
                        Text(
                            text = item.message.toString(),
                            modifier = Modifier
                                .fillMaxWidth()
                                .padding(12.dp)
                        )
                    }
                }
            }
        }

        MessagesButtonsRow(
            state = buttonsState,
            onBack = { viewModel.onEvent(MessagesUiEvent.BackClicked) },
            onDelete = { viewModel.onEvent(MessagesUiEvent.DeleteSelectedClicked) },
            onSelect = { viewModel.onEvent(MessagesUiEvent.SelectClicked) }
        )
    }

    if (dialogState is MessagesDialogState.Visible) {
        AlertDialog(
            onDismissRequest = { viewModel.onEvent(MessagesUiEvent.DismissDialog) },
            title = { Text(dialogState.title) },
            text = { Text(dialogState.description) },
            confirmButton = {
                TextButton(onClick = { viewModel.onEvent(MessagesUiEvent.ConfirmDialogAccept) }) {
                    Text("ACCEPT")
                }
            },
            dismissButton = {
                TextButton(onClick = { viewModel.onEvent(MessagesUiEvent.DismissDialog) }) {
                    Text("CANCEL")
                }
            }
        )
    }
}

            .height(120.dp)

            .padding(1.dp),
        horizontalArrangement = Arrangement.spacedBy(6.dp)
    ) {
        MessagesCustomButton(
            modifier = Modifier.weight(1f),
            buttonType = ButtonType.VOLVER,
            visualState = state.back,
            onClick = onBack
        )
        MessagesCustomButton(
            modifier = Modifier.weight(1f),
            buttonType = ButtonType.SELECT,
            visualState = state.select,
            onClick = onSelect
        )

            modifier = Modifier.weight(1f),

            visualState = state.delete,

@Composable
private fun MessagesCustomButton(
    modifier: Modifier = Modifier,
    buttonType: ButtonType,
    visualState: MessagesButtonVisualState,
    onClick: () -> Unit,
) {
    AndroidView(
        modifier = modifier,
        factory = { context ->
            CustomButton(context).apply {
                setButtonType(buttonType.value)
                setAction(onClick)
            }
        },
        update = { button ->
            button.setButtonType(buttonType.value)
            button.setButtonStyle(
                when (visualState.style) {
                    MessagesButtonStyle.ENABLE -> StyleButton.ENABLE
                    MessagesButtonStyle.DISABLE -> StyleButton.DISABLE
                    MessagesButtonStyle.LOADING -> StyleButton.LOADING
                }
            )
            button.changeBackground(
                when (visualState.background) {
                    MessagesButtonBackground.BLUE -> BackgroundButtonColor.BLUE
                    MessagesButtonBackground.GREEN -> BackgroundButtonColor.GREEN
                    MessagesButtonBackground.RED -> BackgroundButtonColor.RED
                    MessagesButtonBackground.BLACK -> BackgroundButtonColor.BLACK
                    MessagesButtonBackground.TRANSPARENT -> BackgroundButtonColor.TRANSPARENT
                }
            )
            button.changeText(visualState.text)
            button.changeTextColor(visualState.isTextBlack)
        }
    )
}
```

## 2) Compose-friendly ViewModel with `UiState + UiEvent + SharedFlow<MessagesUiEffect>`

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.domain.usecase.MessageUseCase
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.repository.room.entities.message.MessageEntity
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch

class MessagesViewModel(
    private val messageUseCase: MessageUseCase,
    context: Application,
) : BaseViewModel(context) {

    private val TAG = "MessagesViewModel"

    private val _uiState = MutableStateFlow(MessagesUiState())
    val uiState: StateFlow<MessagesUiState> = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<MessagesUiEffect>(extraBufferCapacity = 1)
    val effects: SharedFlow<MessagesUiEffect> = _effects.asSharedFlow()

    fun onEvent(event: MessagesUiEvent) {
        when (event) {
            MessagesUiEvent.LoadMessagesByDriverId -> loadMessagesByDriverId()
            MessagesUiEvent.LoadMessages -> loadMessages()
            MessagesUiEvent.BackClicked -> emitEffect(MessagesUiEffect.NavigateBack)
            MessagesUiEvent.DeleteSelectedClicked -> deleteSelected()
            MessagesUiEvent.SelectClicked -> updateButtons(MessagesButtonsState.activateSelectionMode(_uiState.value.buttonsState))
            MessagesUiEvent.MarkAllMessagesRead -> markAllMessagesRead()
            MessagesUiEvent.DismissDialog -> _uiState.update { it.copy(dialogState = MessagesDialogState.Hidden) }
            MessagesUiEvent.ConfirmDialogAccept -> {
                _uiState.update { it.copy(dialogState = MessagesDialogState.Hidden) }
                emitEffect(MessagesUiEffect.MarkAllReadConfirmed)
            }
        }
    }

    private fun deleteSelected() {
        val selected = _uiState.value.selectedMessages
        if (selected.isNotEmpty()) {
            viewModelScope.launch(Dispatchers.IO) {
                messageUseCase.deleteMessages(selected.toMutableList())
                loadMessagesByDriverId()
            }
        }
    }

    fun setSelectedMessages(selected: List<MessageEntity>) {
        _uiState.update { it.copy(selectedMessages = selected) }
    }

    fun setMessages(messages: List<MessageEntity>) {
        _uiState.update {
            it.copy(
                messages = messages,
                items = getListToMutableList(messages)
            )
        }
    }

    fun requestMarkAllReadDialog() {
        _uiState.update {
            it.copy(
                dialogState = MessagesDialogState.Visible(
                    title = "WARNING",
                    description = "Mark all messages as read?"
                )
            )
        }
    }

    private fun loadMessages() {
        viewModelScope.launch(Dispatchers.IO) {
            val messages = messageUseCase.getMessages()
            _uiState.update {
                it.copy(messages = messages, items = getListToMutableList(messages))
            }
        }
    }

    private fun loadMessagesByDriverId() {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "loadMessagesByDriverId: getMessageByDriverId(): ${messageUseCase.getMessageByDriverId()}")
            val messages = messageUseCase.getMessageByDriverId()
            _uiState.update {
                it.copy(messages = messages, items = getListToMutableList(messages))
            }
        }
    }

    private fun markAllMessagesRead() {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "markAllMessagesRead: getMessageByDriverId(): ${messageUseCase.getMessageByDriverId()}")
            messageUseCase.setAllMessagesRead()
            val messages = messageUseCase.getMessageByDriverId()
            _uiState.update {
                it.copy(messages = messages, items = getListToMutableList(messages))
            }
        }
    }

    private fun emitEffect(effect: MessagesUiEffect) {
        _effects.tryEmit(effect)
    }

    private fun updateButtons(newState: MessagesButtonsState) {
        _uiState.update { it.copy(buttonsState = newState) }
    }

    private fun getListToMutableList(messageList: List<MessageEntity>): List<MessagesListItemUi> {
        val dateSet = mutableSetOf<String>()
        val sortedList = mutableListOf<MessagesListItemUi>()

        for (message in messageList) {
            val formattedDate = message.dateFormatted
            if (dateSet.add(formattedDate)) {
                sortedList.add(MessagesListItemUi.Date(formattedDate))
            }
            sortedList.add(MessagesListItemUi.General(message))
        }
        return sortedList
    }
}
```

```kotlin
        _dialogEffect.tryEmit(MessagesDialogEffect.ConfirmMarkAllRead(model))

        _topBarEffect.tryEmit(MessagesTopBarEffect.ShowHeader(show))

    }

    sealed interface MessagesTopBarEffect {
```

```kotlin
            }
            MessagesUiEvent.MarkAllMessagesRead -> markAllMessagesRead()
            MessagesUiEvent.OpenMarkAllReadDialog -> {
                emitEffect(
                    MessagesUiEffect.OpenCustomDialog(
                        CustomDialog.CustomDialogModel(
                            title = getApplication<Application>().getString(R.string.warning),
                            description = getApplication<Application>().getString(R.string.strMarcarMensajesComoLeidos),
                            buttons = arrayListOf(ButtonType.CANCEL, ButtonType.ACCEPT)
                        )
                    )
                )
            }
            MessagesUiEvent.DismissDialog -> {
                _uiState.update { it.copy(dialogState = MessagesDialogState.Hidden) }
            }
            MessagesUiEvent.ConfirmDialogAccept -> {
                _uiState.update { it.copy(dialogState = MessagesDialogState.Hidden) }
                markAllMessagesRead()
            }
            is MessagesUiEvent.MessageClicked -> {
                emitEffect(MessagesUiEffect.NavigateToMessageDetail(event.messageId, true))
            }
            is MessagesUiEvent.ToggleSelection -> {
                _uiState.update { state ->
                    val newSelection = state.selectedMessages.toMutableSet().apply {
                        if (!add(event.messageId)) remove(event.messageId)
                    }
                    state.copy(selectedMessages = newSelection)
                }
            }
        }
    }

    fun deleteMessages(deleteSelected: MutableList<MessageEntity>) {
        viewModelScope.launch(Dispatchers.IO) {
            messageUseCase.deleteMessages(deleteSelected)
            loadMessagesByDriverId()
        }
    }

    fun loadMessages() {
        viewModelScope.launch(Dispatchers.IO) {
            val messages = messageUseCase.getMessages()
            _messagesFlow.emit(messages)
            _uiState.update { it.copy(messages = messages, items = getListToMutableList(messages)) }
        }
    }

    fun loadMessagesByDriverId() {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "loadMessagesByDriverId: getMessageByDriverId(): ${messageUseCase.getMessageByDriverId()}")
            val messages = messageUseCase.getMessageByDriverId()
            _messagesFlow.emit(messages)
            _uiState.update { it.copy(messages = messages, items = getListToMutableList(messages)) }
        }
    }

    fun markAllMessagesRead() {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "loadMessagesByDriverId: getMessageByDriverId(): ${messageUseCase.getMessageByDriverId()}")
            messageUseCase.setAllMessagesRead()
            val messages = messageUseCase.getMessageByDriverId()
            _messagesFlow.emit(messages)
            _uiState.update { it.copy(messages = messages, items = getListToMutableList(messages)) }
        }
    }

    private fun deleteSelected() {
        val selectedIds = _uiState.value.selectedMessages
        if (selectedIds.isEmpty()) return

        val selectedEntities = _uiState.value.messages.filter { it.id in selectedIds }.toMutableList()
        deleteMessages(selectedEntities)
    }

    private fun emitEffect(effect: MessagesUiEffect) {
        _effects.tryEmit(effect)
    }

    private fun getListToMutableList(messageList: List<MessageEntity>): List<MessagesListItemUi> {
        val dateSet = mutableSetOf<String>()
        val sortedList = mutableListOf<MessagesListItemUi>()

        for (message in messageList) {
            val formattedDate = message.dateFormatted
            if (dateSet.add(formattedDate)) {
                sortedList.add(MessagesListItemUi.Date(formattedDate))
            }
            sortedList.add(MessagesListItemUi.General(message))
        }
        return sortedList
    }
}

class MessagesMainActivityViewModel : ViewModel() {

    private val _dialogEffect = MutableSharedFlow<MessagesDialogEffect>(extraBufferCapacity = 1)
    val dialogEffect: SharedFlow<MessagesDialogEffect> = _dialogEffect.asSharedFlow()

    private val _topBarEffect = MutableSharedFlow<MessagesTopBarEffect>(extraBufferCapacity = 1)
    val topBarEffect: SharedFlow<MessagesTopBarEffect> = _topBarEffect.asSharedFlow()

    fun requestOpenCustomDialog(model: CustomDialog.CustomDialogModel) {
        _dialogEffect.tryEmit(MessagesDialogEffect.OpenCustomDialog(model))
    }

    fun requestShowHeader(show: Boolean) {
        _topBarEffect.tryEmit(MessagesTopBarEffect.ShowHeader(show))
    }

    fun requestMarkReadIcon(show: Boolean) {
        _topBarEffect.tryEmit(MessagesTopBarEffect.ConfigureMarkReadIcon(show))
    }

    sealed interface MessagesDialogEffect {
        data class OpenCustomDialog(val model: CustomDialog.CustomDialogModel) : MessagesDialogEffect
    }

    sealed interface MessagesTopBarEffect {
        data class ShowHeader(val show: Boolean) : MessagesTopBarEffect
        data class ConfigureMarkReadIcon(val show: Boolean) : MessagesTopBarEffect
    }
}

@Composable
fun MessagesScreen(
    viewModel: MessagesViewModel,
    mainActivityViewModel: MessagesMainActivityViewModel,
    onNavigateBack: () -> Unit,
    onNavigateToMessageDetail: (messageId: Long, fromMessages: Boolean) -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    LaunchedEffect(Unit) {
        mainActivityViewModel.requestShowHeader(true)
        mainActivityViewModel.requestMarkReadIcon(true)
        viewModel.onEvent(MessagesUiEvent.LoadMessagesByDriverId)
    }

    LaunchedEffect(Unit) {
        viewModel.effects.collectLatest { effect ->
            when (effect) {
                MessagesUiEffect.NavigateBack -> onNavigateBack()
                is MessagesUiEffect.NavigateToMessageDetail ->
                    onNavigateToMessageDetail(effect.messageId, effect.fromMessages)

                is MessagesUiEffect.OpenCustomDialog -> {
                    mainActivityViewModel.requestOpenCustomDialog(effect.model)
                }
            }
        }
    }

    LaunchedEffect(Unit) {
        mainActivityViewModel.dialogEffect.collectLatest { dialogEffect ->
            when (dialogEffect) {
                is MessagesMainActivityViewModel.MessagesDialogEffect.OpenCustomDialog -> {
                    viewModel._uiState.update {
                        it.copy(
                            dialogState = MessagesDialogState.Visible(
                                title = dialogEffect.model.title,
                                description = dialogEffect.model.description
                            )
                        )
                    }
                }
            }
        }
    }

    Column(modifier = Modifier.fillMaxSize()) {
        LazyColumn(
            modifier = Modifier
                .weight(1f)
                .fillMaxWidth()
        ) {
            items(uiState.items) { item ->
                when (item) {
                    is MessagesListItemUi.Date -> {
                        Text(
                            text = item.formattedDate,
                            modifier = Modifier
                                .fillMaxWidth()
                                .background(Color(0xFFE0E0E0))
                                .padding(8.dp)
                        )
                    }

                    is MessagesListItemUi.General -> {
                        val selected = uiState.selectedMessages.contains(item.message.id)
                        Text(
                            text = item.message.toString(),
                            modifier = Modifier
                                .fillMaxWidth()
                                .clickable {
                                    viewModel.onEvent(MessagesUiEvent.MessageClicked(item.message.id))
                                }
                                .background(if (selected) Color(0xFFCCE5FF) else Color.Transparent)
                                .padding(12.dp)
                        )
                    }
                }
            }
        }

        MessagesButtonsRow(
            state = uiState.buttonsState,
            onBack = { viewModel.onEvent(MessagesUiEvent.BackClicked) },
            onDelete = { viewModel.onEvent(MessagesUiEvent.DeleteSelectedClicked) },
            onSelect = { viewModel.onEvent(MessagesUiEvent.SelectClicked) }
        )
    }

    when (val dialogState = uiState.dialogState) {
        MessagesDialogState.Hidden -> Unit
        is MessagesDialogState.Visible -> {
            AlertDialog(
                onDismissRequest = { viewModel.onEvent(MessagesUiEvent.DismissDialog) },
                title = { Text(dialogState.title) },
                text = { Text(dialogState.description) },
                confirmButton = {
                    TextButton(onClick = { viewModel.onEvent(MessagesUiEvent.ConfirmDialogAccept) }) {
                        Text("ACCEPT")
                    }
                },
                dismissButton = {
                    TextButton(onClick = { viewModel.onEvent(MessagesUiEvent.DismissDialog) }) {
                        Text("CANCEL")
                    }
                }
            )
        }
    }
}

@Composable
private fun MessagesButtonsRow(
    state: MessagesButtonsState,
    onBack: () -> Unit,
    onDelete: () -> Unit,
    onSelect: () -> Unit,
) {
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .height(120.dp)
            .background(Color.Black)
            .padding(1.dp),
        horizontalArrangement = Arrangement.spacedBy(6.dp)
    ) {
        if (state.back.visible) {
            MessagesCustomButton(
                modifier = Modifier.weight(1f),
                buttonType = ButtonType.VOLVER,
                visualState = state.back,
                onClick = onBack
            )
        }
        if (state.select.visible) {
            MessagesCustomButton(
                modifier = Modifier.weight(1f),
                buttonType = ButtonType.SELECT,
                visualState = state.select,
                onClick = onSelect
            )
        }
        if (state.delete.visible) {
            MessagesCustomButton(
                modifier = Modifier.weight(1f),
                buttonType = ButtonType.DELETE,
                visualState = state.delete,
                onClick = onDelete
            )
        }
    }
}

@Composable
private fun MessagesCustomButton(
    modifier: Modifier = Modifier,
    buttonType: ButtonType,
    visualState: MessagesButtonVisualState,
    onClick: () -> Unit,
) {
    androidx.compose.ui.viewinterop.AndroidView(
        modifier = modifier,
        factory = { context ->
            CustomButton(context).apply {
                setButtonType(buttonType.value)
                setAction(onClick)
            }
        },
        update = { button ->
            button.setButtonType(buttonType.value)
            button.setButtonStyle(
                when (visualState.style) {
                    MessagesButtonStyle.ENABLE -> StyleButton.ENABLE
                    MessagesButtonStyle.DISABLE -> StyleButton.DISABLE
                    MessagesButtonStyle.LOADING -> StyleButton.LOADING
                }
            )
            button.changeBackground(
                when (visualState.background) {
                    MessagesButtonBackground.GREEN -> BackgroundButtonColor.GREEN
                    MessagesButtonBackground.RED -> BackgroundButtonColor.RED
                    MessagesButtonBackground.BLUE -> BackgroundButtonColor.BLUE
                    MessagesButtonBackground.YELLOW -> BackgroundButtonColor.YELLOW
                    MessagesButtonBackground.ORANGE -> BackgroundButtonColor.ORANGE
                    MessagesButtonBackground.DEFAULT -> BackgroundButtonColor.DEFAULT
                    MessagesButtonBackground.BLACK -> BackgroundButtonColor.BLACK
                    MessagesButtonBackground.TRANSPARENT -> BackgroundButtonColor.TRANSPARENT
                }
            )
            button.changeText(visualState.text)
            button.changeTextColor(visualState.isTextBlack)
        }
    )
}
```


// FIRST ANSWER
Below is a Compose-first rewrite of the `PortugalInvoiceFragment` flow, with:

- a `PortugalInvoiceScreen` composable
- a Compose-friendly `PortugalInvoiceViewModel`
- a `PortugalInvoiceButtonsState` reducer that mirrors the fragment logic one-by-one
- a `SharedFlow<PortugalInvoiceUiEffect>` for one-shot effects
- a `PortugalInvoiceMainActivityViewModel` bridge for navigation/dialog integration
- dialog state only where the fragment actually used `CustomDialog`
- preserved Jetpack Views navigation behavior

I’m keeping the naming prefix requirement for:
- `PortugalInvoiceButtonVisualState`
- `PortugalInvoiceUiState`
- `PortugalInvoiceDialogState`

@Composable
private fun CountryDropdown(
    countries: List<String>,
    selectedIndex: Int,
    onSelected: (Int) -> Unit
) {
    var expanded by remember { mutableStateOf(false) }
    Box {
        OutlinedButton(onClick = { expanded = true }, modifier = Modifier.fillMaxWidth()) {
            Text(countries.getOrNull(selectedIndex).orEmpty())
        }
        DropdownMenu(expanded = expanded, onDismissRequest = { expanded = false }) {
            countries.forEachIndexed { index, country ->
                DropdownMenuItem(
                    text = { Text(country) },
                    onClick = {
                        expanded = false
                        onSelected(index)
                    }
                )
            }
        }
    }
}

@Composable
private fun CustomButtonComposable(
    modifier: Modifier = Modifier,
    text: String,
    style: CustomButton.StyleButton,
    background: CustomButton.BackgroundButtonColor,
    buttonType: ButtonType,
    onClick: () -> Unit
) {
    val enabled = style == CustomButton.StyleButton.ENABLE
    val bgColor = when (background) {
        CustomButton.BackgroundButtonColor.GREEN -> Color(0xFF2E7D32)
        CustomButton.BackgroundButtonColor.RED -> Color(0xFFC62828)
        CustomButton.BackgroundButtonColor.BLUE -> Color(0xFF1565C0)
        CustomButton.BackgroundButtonColor.YELLOW -> Color(0xFFF9A825)
        CustomButton.BackgroundButtonColor.ORANGE -> Color(0xFFEF6C00)
        CustomButton.BackgroundButtonColor.BLACK -> Color.Black
        CustomButton.BackgroundButtonColor.TRANSPARENT -> Color.Transparent
        CustomButton.BackgroundButtonColor.DEFAULT -> Color(0xFF1565C0)
    }

    Button(
        onClick = onClick,
        enabled = enabled,
        modifier = modifier.fillMaxHeight(),
        colors = ButtonDefaults.buttonColors(containerColor = bgColor)
    ) {
        Text(text = text)
    }

import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

class PortugalInvoiceViewModel(
    context: Application,
    private val portugalUseCase: PortugalUseCase,
    private val tripUseCase: TripUseCase,
    private val ticketUseCase: TicketUseCase,
    private val dispatchUseCase: DispatchUseCase,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val printerUseCase: PrinterUseCase,
) : BaseViewModel(context) {

    private val _uiState = MutableStateFlow(PortugalInvoiceUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<PortugalInvoiceUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()

    fun onEvent(event: PortugalInvoiceUiEvent) {
        when (event) {
            is PortugalInvoiceUiEvent.NifChanged -> _uiState.update { it.copy(nif = event.value) }
            is PortugalInvoiceUiEvent.NameChanged -> _uiState.update { it.copy(name = event.value) }
            is PortugalInvoiceUiEvent.LocalidadChanged -> _uiState.update { it.copy(localidad = event.value) }
            is PortugalInvoiceUiEvent.CountrySelected -> _uiState.update { it.copy(selectedCountryIndex = event.index) }

            PortugalInvoiceUiEvent.CancelClicked -> {
                viewModelScope.launch { _uiEffect.emit(PortugalInvoiceUiEffect.NavigateBack) }
            }

            PortugalInvoiceUiEvent.AcceptClicked -> handleAcceptClicked()

            PortugalInvoiceUiEvent.DialogAccepted -> {
                viewModelScope.launch {
                    _uiState.update { it.copy(externalCustomer = true, dialog = PortugalInvoiceDialogState()) }
                    _uiEffect.emit(PortugalInvoiceUiEffect.SetAcceptEnabled)
                    _uiEffect.emit(PortugalInvoiceUiEffect.HideExternalCustomerDialog)
                    acceptInvoiceFromState()
                }
            }

            PortugalInvoiceUiEvent.DialogDismissed -> {
                viewModelScope.launch {
                    _uiState.update { it.copy(dialog = PortugalInvoiceDialogState()) }
                    _uiEffect.emit(PortugalInvoiceUiEffect.SetAcceptEnabled)
                    _uiEffect.emit(PortugalInvoiceUiEffect.HideExternalCustomerDialog)
                }
            }
        }
    }

    private fun handleAcceptClicked() {
        viewModelScope.launch {
            _uiEffect.emit(PortugalInvoiceUiEffect.SetAcceptLoading)

            val state = uiState.value
            val selectedCountry = state.countries[state.selectedCountryIndex]
            val isPortugal = selectedCountry == CountryUtils.arrayCountries[0]

            if (isPortugal && !state.externalCustomer) {
                if (isValidPortugalNif(state.nif)) {
                    acceptInvoiceFromState()
                } else {
                    _uiState.update {
                        it.copy(
                            dialog = PortugalInvoiceDialogState(
                                visible = true,
                                description = getApplication<Application>().getString(R.string.external_client),
                                buttons = listOf(ButtonType.CANCEL, ButtonType.ACCEPT)
                            )
                        )
                    }
                    _uiEffect.emit(PortugalInvoiceUiEffect.ShowExternalCustomerDialog)
                }
            } else {
                acceptInvoiceFromState()
            }
        }
    }

    private fun isValidPortugalNif(nif: String): Boolean {
        return nif.isValid(FieldType.NIF_PORTUGAL)
    }

    private fun acceptInvoiceFromState() {
        val state = uiState.value
        val country = state.countries[state.selectedCountryIndex]
        acceptInvoice(state.nif, state.name, state.localidad, country)
    }

    fun acceptInvoice(
        nif: String,
        name: String,
        localidade: String,
        country: String
    ) {
        viewModelScope.launch {
            val getLastTrip = tripUseCase.getLastTrip()
            getLastTrip?.let { trip ->
                trip.nif = nif
                trip.name = name
                trip.localidade = localidade
                trip.country = country
                trip.iso = CountryUtils.arrayISOCountries[CountryUtils.arrayCountries.indexOf(country)]
                tripUseCase.updateTripPortugalThings(
                    trip.id, trip.nif, trip.name, trip.localidade, trip.country, trip.iso
                )
                portugalUseCase.calculateHashPortugal(true, trip)
                prepareTicket(trip)
            }
        }
    }

    private suspend fun prepareTicket(trip: Trip) {
        val dispatch: InfoDispatchModel? = if (trip.fromDispatch) {
            dispatchUseCase.getDispatchByTripId(trip.id)?.let {
                it.id?.let { _ -> it.toInfoDispatchModel() }
            }
        } else null

        val destination = PortugalInvoiceFragmentDirections.actionPortugalInvoiceFragmentToReceiptHistoryFragment(-1L)
        val callback = object : TicketUseCaseImpl.PrepareTicketCallback {
            override fun onSuccess() {
                viewModelScope.launch {
                    navigateTo(destination)
                    _uiEffect.emit(PortugalInvoiceUiEffect.NavigateToReceiptHistory)
                }
            }
        }

        shiftStatusUseCase.getStatus()?.let { shiftStatus ->
            ticketUseCase.prepareTicket(
                trip = trip,
                dispatch = dispatch,
                shiftStatus = shiftStatus,
                portugalByPass = true,
                ticketPrinted = true,
                callback = callback
            )
        }
    }

    fun setExternalCustomer(externalCustomer: Boolean) {
        viewModelScope.launch {
            _uiState.update { it.copy(externalCustomer = externalCustomer) }
        }
    }
}

sealed interface PortugalInvoiceUiEvent {
    data class NifChanged(val value: String) : PortugalInvoiceUiEvent
    data class NameChanged(val value: String) : PortugalInvoiceUiEvent
    data class LocalidadChanged(val value: String) : PortugalInvoiceUiEvent
    data class CountrySelected(val index: Int) : PortugalInvoiceUiEvent
    data object CancelClicked : PortugalInvoiceUiEvent
    data object AcceptClicked : PortugalInvoiceUiEvent
    data object DialogAccepted : PortugalInvoiceUiEvent
    data object DialogDismissed : PortugalInvoiceUiEvent
}

## 4) Notes on exact fragment parity

### Fragment logic mirrored:
- Cancel button -> `navigateBack()`
- Accept button:
  - sets loading
  - if Portugal selected and not external customer:
    - validate NIF
    - if valid -> accept invoice
    - else -> show dialog
  - else -> accept invoice directly
- Dialog accept:
  - sets `externalCustomer = true`
  - re-enables accept button
  - closes dialog

- Scrollable form area
- Bottom 2-button bar
- Cancel red
- Accept green

class PortugalInvoiceViewModel(
    context: Application,
    private val portugalUseCase: PortugalUseCase,
    private val tripUseCase: TripUseCase,
    private val ticketUseCase: TicketUseCase,
    private val dispatchUseCase: DispatchUseCase,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val printerUseCase: PrinterUseCase,
) : BaseViewModel(context) {

    private val _uiState = MutableStateFlow(PortugalInvoiceUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<PortugalInvoiceUiEffect>()
    val uiEffect: SharedFlow<PortugalInvoiceUiEffect> = _uiEffect.asSharedFlow()

    fun onEvent(event: PortugalInvoiceUiEvent) {
        when (event) {
            is PortugalInvoiceUiEvent.NifChanged -> _uiState.update { it.copy(nif = event.value) }
            is PortugalInvoiceUiEvent.NameChanged -> _uiState.update { it.copy(name = event.value) }
            is PortugalInvoiceUiEvent.LocalidadChanged -> _uiState.update { it.copy(localidad = event.value) }
            is PortugalInvoiceUiEvent.CountrySelected -> _uiState.update { it.copy(selectedCountryIndex = event.index) }

            PortugalInvoiceUiEvent.CancelClicked -> viewModelScope.launch {
                _uiEffect.emit(PortugalInvoiceUiEffect.NavigateBack)
            }

            PortugalInvoiceUiEvent.AcceptClicked -> handleAcceptClicked()

            PortugalInvoiceUiEvent.DialogAccepted -> viewModelScope.launch {
                _uiState.update { it.copy(externalCustomer = true, dialog = PortugalInvoiceDialogState()) }
                _uiEffect.emit(PortugalInvoiceUiEffect.SetAcceptEnabled)
                _uiEffect.emit(PortugalInvoiceUiEffect.HideExternalCustomerDialog)
                acceptInvoiceFromState()
            }

            PortugalInvoiceUiEvent.DialogDismissed -> viewModelScope.launch {
                _uiState.update { it.copy(dialog = PortugalInvoiceDialogState()) }
                _uiEffect.emit(PortugalInvoiceUiEffect.SetAcceptEnabled)
                _uiEffect.emit(PortugalInvoiceUiEffect.HideExternalCustomerDialog)
            }
        }
    }

    private fun handleAcceptClicked() {
        viewModelScope.launch {
            _uiEffect.emit(PortugalInvoiceUiEffect.SetAcceptLoading)

            val state = uiState.value
            val selectedCountry = state.countries.getOrNull(state.selectedCountryIndex).orEmpty()
            val isPortugal = selectedCountry == CountryUtils.arrayCountries.firstOrNull()

            if (isPortugal && !state.externalCustomer) {
                if (state.nif.isValid(FieldType.NIF_PORTUGAL)) {
                    acceptInvoiceFromState()
                } else {
                    _uiState.update {
                        it.copy(
                            dialog = PortugalInvoiceDialogState(
                                visible = true,
                                description = getApplication<Application>().getString(R.string.external_client),
                                buttons = arrayListOf(ButtonType.CANCEL, ButtonType.ACCEPT)
                            )
                        )
                    }
                    _uiEffect.emit(PortugalInvoiceUiEffect.ShowExternalCustomerDialog)
                }
            } else {
                acceptInvoiceFromState()
            }
        }
    }

    private fun acceptInvoiceFromState() {
        val state = uiState.value
        val country = state.countries.getOrNull(state.selectedCountryIndex).orEmpty()
        acceptInvoice(state.nif, state.name, state.localidad, country)
    }

    fun acceptInvoice(
        nif: String,
        name: String,
        localidade: String,
        country: String
    ) {
        viewModelScope.launch {
            val getLastTrip = tripUseCase.getLastTrip()
            getLastTrip?.let { trip ->
                trip.nif = nif
                trip.name = name
                trip.localidade = localidade
                trip.country = country
                trip.iso = CountryUtils.arrayISOCountries[CountryUtils.arrayCountries.indexOf(country)]
                tripUseCase.updateTripPortugalThings(
                    trip.id, trip.nif, trip.name, trip.localidade, trip.country, trip.iso
                )
                portugalUseCase.calculateHashPortugal(true, trip)
                prepareTicket(trip)
            }
        }
    }

    private suspend fun prepareTicket(trip: Trip) {
        val dispatch: InfoDispatchModel? = if (trip.fromDispatch) {
            dispatchUseCase.getDispatchByTripId(trip.id)?.let { dispatch ->
                dispatch.id?.let { dispatch.toInfoDispatchModel() }
            }
        } else null

        val destination = PortugalInvoiceFragmentDirections.actionPortugalInvoiceFragmentToReceiptHistoryFragment(-1L)
        val callback = object : TicketUseCaseImpl.PrepareTicketCallback {
            override fun onSuccess() {
                viewModelScope.launch {
                    navigateTo(destination)
                    _uiEffect.emit(PortugalInvoiceUiEffect.NavigateToReceiptHistory)
                }
            }
        }

        shiftStatusUseCase.getStatus()?.let { shiftStatus ->
            ticketUseCase.prepareTicket(
                trip = trip,
                dispatch = dispatch,
                shiftStatus = shiftStatus,
                portugalByPass = true,
                ticketPrinted = true,
                callback = callback
            )
        }
    }

    fun setExternalCustomer(externalCustomer: Boolean) {
        viewModelScope.launch {
            _uiState.update { it.copy(externalCustomer = externalCustomer) }
        }
    }
}

@Composable
private fun CountryDropdown(
    countries: List<String>,
    selectedIndex: Int,
    onSelected: (Int) -> Unit
) {
    var expanded by remember { mutableStateOf(false) }
    Box {
        OutlinedButton(onClick = { expanded = true }, modifier = Modifier.fillMaxWidth()) {
            Text(countries.getOrNull(selectedIndex).orEmpty())
        }
        DropdownMenu(expanded = expanded, onDismissRequest = { expanded = false }) {
            countries.forEachIndexed { index, country ->
                DropdownMenuItem(
                    text = { Text(country) },
                    onClick = {
                        expanded = false
                        onSelected(index)
                    }
                )
            }
        }
    }
}

@Composable
private fun CustomButtonComposable(
    modifier: Modifier = Modifier,
    text: String,
    style: CustomButton.StyleButton,
    background: CustomButton.BackgroundButtonColor,
    buttonType: ButtonType,
    onClick: () -> Unit
) {
    val enabled = style == CustomButton.StyleButton.ENABLE
    val bgColor = when (background) {
        CustomButton.BackgroundButtonColor.GREEN -> Color(0xFF2E7D32)
        CustomButton.BackgroundButtonColor.RED -> Color(0xFFC62828)
        CustomButton.BackgroundButtonColor.BLUE -> Color(0xFF1565C0)
        CustomButton.BackgroundButtonColor.YELLOW -> Color(0xFFF9A825)
        CustomButton.BackgroundButtonColor.ORANGE -> Color(0xFFEF6C00)
        CustomButton.BackgroundButtonColor.DEFAULT -> Color(0xFF1565C0)
        CustomButton.BackgroundButtonColor.BLACK -> Color.Black
        CustomButton.BackgroundButtonColor.TRANSPARENT -> Color.Transparent
    }

    Button(
        onClick = onClick,
        enabled = enabled,
        modifier = modifier.fillMaxHeight(),
        colors = ButtonDefaults.buttonColors(containerColor = bgColor)
    ) {
        Text(text = text)
    }
}

        }

        atcud: String,
        sequenceNumber: Int,
        document: Int,
        portugalPassword: String = ""
    ) {
        viewModelScope.launch(Dispatchers.IO) {
            val portugalData = portugalUseCase.getPortugalData()
            val portugalModel = PortugalModel(
                atcud = atcud,
                sequenceNumber = sequenceNumber,
                numTicketPrint = document,
                portugalPassword = portugalData?.portugalPassword ?: portugalPassword,
            )
            portugalUseCase.updatePortugalData(portugalModel)
            _backDataFlow.emit(true)
        }
    }

    fun updatePortugalPin(portugalPassword: String) {
        viewModelScope.launch {
            val portugalData = portugalUseCase.getPortugalData()
            portugalData?.let {
                portugalData.portugalPassword = portugalData.encryptPortugalPassword(portugalPassword)
                portugalUseCase.updatePortugalData(portugalData)
            }
        }
    }

    fun checkPinPortugal(editTextString: String?) {
        viewModelScope.launch {
            try {
                val pin = editTextString?.toInt()
                val portugalCode = BuildConfig.portugal_p.toInt()
                val portugalPassword = portugalUseCase.getPortugalData()
                if (portugalPassword != null && portugalPassword.portugalPassword.isNotBlank()) {
                    _pinCallbackFlow.emit(portugalPassword.encryptPortugalPassword(pin.toString()) == portugalPassword.portugalPassword)
                } else {
                    _pinCallbackFlow.emit(pin == portugalCode)
                }
            } catch (e: Exception) {
                Logs.d(TAG, "checkPinPortugal: $e")
            }
        }
    }

class TripHistoryViewModel(
    context: Application,
    val tripUseCase: TripUseCase,
    val printerUseCase: PrinterUseCase,
    val portugalUseCase: PortugalUseCase,
) : BaseViewModel(context) {

    private val TAG = "TripHistoryViewModel"

    private val _tripListFlow = MutableStateFlow<List<Trip>?>(null)
    val tripListFlow = _tripListFlow.asStateFlow()

    private val _loadingFlow = MutableStateFlow(false)
    val loadingFlow = _loadingFlow.asStateFlow()

    var isLastPage = false
    private val PAGE_SIZE = 25

    fun getTrips(page: Int) {
        viewModelScope.launch {
            try {
                Logs.d(TAG, "getTrips: page: $page")
                if (isLastPage) return@launch
                _loadingFlow.value = true
                val loadedItems = page * PAGE_SIZE
                val newTrips = tripUseCase.getAllTripsFinishedPaged(loadedItems)
                val currentTrips = _tripListFlow.value ?: emptyList()
                if (newTrips.size < PAGE_SIZE) isLastPage = true
                val combinedTrips = (currentTrips + newTrips).distinctBy { it.id }
                _tripListFlow.emit(combinedTrips)
            } catch (e: Exception) {
                Logs.e(TAG, "getTrips: error: ${e.message}")
            } finally {
                _loadingFlow.value = false
            }
        }
    }

    fun deleteSelectedTrips(selectedTrips: Set<Trip>, callback: () -> Unit) {
        viewModelScope.launch {
            if (selectedTrips.isNotEmpty()) {
                val selectedTripsList = selectedTrips.toList()
                for (trip in selectedTripsList) {
                    tripUseCase.deleteTrip(trip)
                    Logs.d("TripHistoryViewModel", "trip ${trip.id} deleted ")
                    callback.invoke()
                }
                withContext(Dispatchers.Main) {
                    android.widget.Toast.makeText(
                        context,
                        context.getString(R.string.toast_trips_eliminados_correctamente),
                        android.widget.Toast.LENGTH_SHORT
                    ).show()
                }
            } else {
                withContext(Dispatchers.Main) {
                    android.widget.Toast.makeText(
                        context,
                        context.getString(R.string.toast_seleccione_algun_trip),
                        android.widget.Toast.LENGTH_SHORT
                    ).show()
                }
            }
        }
    }

    fun printTicket(trip: Trip) {
        viewModelScope.launch {
            portugalUseCase.increaseTicketPrinted(trip)
            printerUseCase.printTicket(trip, true)
        }
    }
}

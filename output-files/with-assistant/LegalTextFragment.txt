## 3) Compose-friendly `LegalTextViewModel`

This keeps the original behavior:
- `setLegalText()` loads legal text
- `clickOnAccept()` navigates to welcome
- state is exposed via `StateFlow<LegalTextUiState>`
- one-shot actions are exposed via `SharedFlow<LegalTextUiEffect>`

```kotlin
package ifac.td.taxi.ui.screen.legaltext

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.R
import ifac.td.taxi.framework.sdk.usecase.ShowLegalTextUseCase
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

class LegalTextComposeViewModel(
    application: Application,
    private val showLegalTextUseCase: ShowLegalTextUseCase
) : AndroidViewModel(application) {

    private val _uiState = MutableStateFlow(
        LegalTextUiState(
            legalText = "",
            isLoading = false,
            buttonsState = reduceLegalTextButtonsState(isLoading = false)
        )
    )
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<LegalTextUiEffect>(extraBufferCapacity = 1)
    val uiEffect = _uiEffect.asSharedFlow()

    fun onEvent(event: LegalTextUiEvent) {
        when (event) {
            LegalTextUiEvent.ScreenShown -> loadLegalText()
            LegalTextUiEvent.AcceptClicked -> clickOnAccept()
        }
    }

    private fun clickOnAccept() {
        viewModelScope.launch {
            _uiEffect.emit(LegalTextUiEffect.NavigateToWelcome)
        }
    }

    private fun loadLegalText() {
        viewModelScope.launch {
            _uiState.update {
                it.copy(
                    isLoading = true,
                    buttonsState = reduceLegalTextButtonsState(isLoading = true)
                )
            }

            runCatching {
                showLegalTextUseCase.getLegalText()
            }.onSuccess { legalText ->
                _uiState.update {
                    it.copy(
                        legalText = legalText.orEmpty(),
                        isLoading = false,
                        buttonsState = reduceLegalTextButtonsState(isLoading = false)
                    )
                }
            }.onFailure {
                _uiState.update {
                    it.copy(
                        isLoading = false,
                        buttonsState = reduceLegalTextButtonsState(isLoading = false)
                    )
                }
                _uiEffect.emit(
                    LegalTextUiEffect.ShowToast(R.string.toast_action_cancelled)
                )
            }
        }
    }
}
```

## 6) Compose wrapper for the custom button API

Since `CustomButton` is a View, the clean Compose equivalent is `AndroidView`.  
This preserves the exact custom button behavior and styling.

```kotlin
package ifac.td.taxi.ui.screen.legaltext

import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.viewinterop.AndroidView
import ifac.td.taxi.ui.custom.button.CustomButton

@Composable
fun ComposeCustomButton(
    text: String,
    iconRes: Int,
    background: CustomButton.BackgroundButtonColor,
    style: CustomButton.StyleButton,
    enabled: Boolean,
    onClick: () -> Unit,
    modifier: Modifier = Modifier
) {
    AndroidView(
        modifier = modifier,
        factory = { context ->
            CustomButton(context).apply {
                setButtonType(ifac.td.taxi.ui.custom.button.ButtonType.ACCEPT.value)
                changeText(text)
                changeButtonIcon(iconRes)
                changeBackground(background)
                setButtonStyle(style)
                setAction(onClick)
            }
        },
        update = { button ->
            button.changeText(text)
            button.changeButtonIcon(iconRes)
            button.changeBackground(background)
            button.setButtonStyle(style)
            button.setAction(onClick)
        }
    )
}
```

## 7) If you want exact fragment navigation preservation

Your original fragment did:

```kotlin
vModel.clickOnAccept() -> navigateTo(R.id.action_legalTextFragment_to_welcomeFragment)
```

In Compose, you should keep the same navigation graph action.  
If you’re using `NavController`, the effect handler can do:

```kotlin
navController.navigate(R.id.action_legalTextFragment_to_welcomeFragment)
```

So your screen host can wire it like this:

```kotlin
@Composable
fun LegalTextRoute(
    navController: androidx.navigation.NavController,
    viewModel: LegalTextComposeViewModel,
    mainActivityViewModel: LegalTextMainActivityViewModel,
    onHideHeader: () -> Unit
) {
    LegalTextScreen(
        viewModel = viewModel,
        mainActivityViewModel = mainActivityViewModel,
        onNavigateToWelcome = {
            navController.navigate(R.id.action_legalTextFragment_to_welcomeFragment)
        },
        onShowToast = { /* show toast in host */ },
        onHideHeader = onHideHeader
    )
}
```

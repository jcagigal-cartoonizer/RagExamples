## 1) Compose UI state, events, effects, and button reducer

```kotlin
package ifac.td.taxi.ui.screen.redsys

import androidx.annotation.StringRes
import androidx.compose.runtime.Immutable
import ifac.td.taxi.R
import ifac.td.taxi.ui.custom.button.CustomButton

@Immutable
data class LoginUserRedSysUiState(
    val user: String = "",
    val password: String = "",
    val userError: String? = null,
    val isAcceptLoading: Boolean = false,
    val buttons: LoginUserRedSysButtonsState = LoginUserRedSysButtonsState()
)

sealed interface LoginUserRedSysUiEvent {
    data class UserChanged(val value: String) : LoginUserRedSysUiEvent
    data class PasswordChanged(val value: String) : LoginUserRedSysUiEvent
    data object CancelClicked : LoginUserRedSysUiEvent
    data object AcceptClicked : LoginUserRedSysUiEvent
    data object ScreenStarted : LoginUserRedSysUiEvent
}

sealed interface LoginUserRedSysUiEffect {
    data object NavigateBack : LoginUserRedSysUiEffect
    data object LoginSuccess : LoginUserRedSysUiEffect
    data object LoginFailed : LoginUserRedSysUiEffect
    data object NavigateToChangePassword : LoginUserRedSysUiEffect
}

@Immutable
data class LoginUserRedSysButtonsState(
    val cancelStyle: CustomButton.StyleButton = CustomButton.StyleButton.ENABLE,
    val cancelBackground: CustomButton.BackgroundButtonColor = CustomButton.BackgroundButtonColor.RED,
    val acceptStyle: CustomButton.StyleButton = CustomButton.StyleButton.ENABLE,
    val acceptBackground: CustomButton.BackgroundButtonColor = CustomButton.BackgroundButtonColor.GREEN,
    val acceptVisible: Boolean = true,
    val cancelVisible: Boolean = true
) {
    companion object {
        /**
         * Exact reducer mirroring the fragment branches one-by-one.
         *
         * Fragment branches:
         * - btnCancel.setAction { navigateBack() }
         * - btnAccept.setAction { clickOnAccept() }
         * - on loginRedSysFlow true -> toast + navigateBack()
         * - on loginRedSysFlow false -> edUser.error + requestFocus + btnAccept ENABLE + GREEN
         * - on userRedSysFlow -> set user/password text
         */
        fun reduce(
            state: LoginUserRedSysUiState,
            effect: LoginUserRedSysUiEffect?
        ): LoginUserRedSysButtonsState {
            return when (effect) {
                LoginUserRedSysUiEffect.LoginSuccess -> state.buttons.copy(
                    acceptStyle = CustomButton.StyleButton.ENABLE,
                    acceptBackground = CustomButton.BackgroundButtonColor.GREEN
                )

                LoginUserRedSysUiEffect.LoginFailed -> state.buttons.copy(
                    acceptStyle = CustomButton.StyleButton.ENABLE,
                    acceptBackground = CustomButton.BackgroundButtonColor.GREEN
                )

                LoginUserRedSysUiEffect.NavigateBack -> state.buttons.copy(
                    cancelStyle = CustomButton.StyleButton.ENABLE,
                    cancelBackground = CustomButton.BackgroundButtonColor.RED,
                    acceptStyle = CustomButton.StyleButton.ENABLE,
                    acceptBackground = CustomButton.BackgroundButtonColor.GREEN
                )

                LoginUserRedSysUiEffect.NavigateToChangePassword -> state.buttons.copy(
                    acceptStyle = CustomButton.StyleButton.ENABLE,
                    acceptBackground = CustomButton.BackgroundButtonColor.GREEN
                )

                null -> state.buttons
            }
        }
    }
}
```

## 2) Compose-friendly ViewModel with `UiState + UiEvent + SharedFlow<UiEffect>`

This replaces the fragment’s `SharedFlow<Boolean>` and `SharedFlow<String?>` with a single state stream plus one effect stream.

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import androidx.lifecycle.viewModelScope
import es.redsys.paysys.Utils.RedCLSErrorCodes
import ifac.td.taxi.R
import ifac.td.taxi.domain.model.RedSysLoginResponse
import ifac.td.taxi.domain.usecase.RedSysUseCase
import ifac.td.taxi.framework.TemporalData
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.ui.custom.button.CustomButton
import ifac.td.taxi.ui.screen.redsys.LoginUserRedSysButtonsState
import ifac.td.taxi.ui.screen.redsys.LoginUserRedSysUiEffect
import ifac.td.taxi.ui.screen.redsys.LoginUserRedSysUiEvent
import ifac.td.taxi.ui.screen.redsys.LoginUserRedSysUiState
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

class LoginUserRedSysViewModel(
    private val redSysUseCase: RedSysUseCase,
    context: Application,
) : BaseViewModel(context) {

    private val TAG = "LoginUserRedSysViewModel"

    private val _uiState = MutableStateFlow(LoginUserRedSysUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<LoginUserRedSysUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()

    fun onEvent(event: LoginUserRedSysUiEvent) {
        when (event) {
            is LoginUserRedSysUiEvent.UserChanged -> {
                _uiState.update { it.copy(user = event.value, userError = null) }
            }

            is LoginUserRedSysUiEvent.PasswordChanged -> {
                _uiState.update { it.copy(password = event.value) }
            }

            LoginUserRedSysUiEvent.CancelClicked -> {
                viewModelScope.launch {
                    _uiEffect.emit(LoginUserRedSysUiEffect.NavigateBack)
                }
            }

            LoginUserRedSysUiEvent.AcceptClicked -> {
                clickOnAccept()
            }

            LoginUserRedSysUiEvent.ScreenStarted -> {
                getUserRedSys()
            }
        }
    }

    private fun clickOnAccept() {
        val current = _uiState.value

        viewModelScope.launch {
            _uiState.update {
                it.copy(
                    isAcceptLoading = true,
                    buttons = it.buttons.copy(
                        acceptStyle = CustomButton.StyleButton.LOADING
                    )
                )
            }

            loginRedSys(
                user = current.user,
                password = current.password
            )
        }
    }

    private fun loginRedSys(user: String, password: String) {
        viewModelScope.launch(Dispatchers.IO) {
            redSysUseCase.configureRedSys()

            val loginRedSys = redSysUseCase.loginRedSys(
                username = user,
                password = password
            )

            when (loginRedSys) {
                is RedSysLoginResponse.Success -> {
                    if (loginRedSys.merchantList.isNotEmpty()) {
                        redSysUseCase.saveUsernameRedSys(user)
                        if (password.isNotEmpty()) {
                            TemporalData.passwordRedSys = password
                        }

                        _uiState.update {
                            it.copy(
                                isAcceptLoading = false,
                                buttons = it.buttons.copy(
                                    acceptStyle = CustomButton.StyleButton.ENABLE,
                                    acceptBackground = CustomButton.BackgroundButtonColor.GREEN
                                )
                            )
                        }
                        _uiEffect.emit(LoginUserRedSysUiEffect.LoginSuccess)
                    } else {
                        _uiState.update {
                            it.copy(
                                isAcceptLoading = false,
                                userError = context.getString(R.string.incorrect_login),
                                buttons = it.buttons.copy(
                                    acceptStyle = CustomButton.StyleButton.ENABLE,
                                    acceptBackground = CustomButton.BackgroundButtonColor.GREEN
                                )
                            )
                        }
                        _uiEffect.emit(LoginUserRedSysUiEffect.LoginFailed)
                    }
                }

                is RedSysLoginResponse.Error -> {
                    Logs.d(TAG, "Login RedSys failed or returned empty array: ${loginRedSys.errorMessage}")

                    if (loginRedSys.errorCode == RedCLSErrorCodes.STATUS_KO_FORMATO_RESP_LOGIN_PWD_CAD) {
                        _uiState.update {
                            it.copy(
                                isAcceptLoading = false,
                                buttons = it.buttons.copy(
                                    acceptStyle = CustomButton.StyleButton.ENABLE,
                                    acceptBackground = CustomButton.BackgroundButtonColor.GREEN
                                )
                            )
                        }
                        _uiEffect.emit(LoginUserRedSysUiEffect.NavigateToChangePassword)
                    } else {
                        _uiState.update {
                            it.copy(
                                isAcceptLoading = false,
                                userError = context.getString(R.string.incorrect_login),
                                buttons = it.buttons.copy(
                                    acceptStyle = CustomButton.StyleButton.ENABLE,
                                    acceptBackground = CustomButton.BackgroundButtonColor.GREEN
                                )
                            )
                        }
                        _uiEffect.emit(LoginUserRedSysUiEffect.LoginFailed)
                    }
                }
            }
        }
    }

    fun getUserRedSys() {
        viewModelScope.launch {
            val user = redSysUseCase.getUsernameRedSys()
            _uiState.update {
                it.copy(
                    user = user.orEmpty(),
                    password = TemporalData.passwordRedSys
                )
            }
        }
    }
}
```

```kotlin
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.unit.dp
import ifac.td.taxi.ui.custom.button.CustomButton

@Composable
fun CustomButtonComposable(
    modifier: Modifier = Modifier,
    text: String,
    style: CustomButton.StyleButton,
    background: CustomButton.BackgroundButtonColor,
    onClick: () -> Unit
) {
    val bgColor = when (background) {
        CustomButton.BackgroundButtonColor.GREEN -> Color(0xFF2E7D32)
        CustomButton.BackgroundButtonColor.RED -> Color(0xFFC62828)
        CustomButton.BackgroundButtonColor.BLUE -> Color(0xFF1565C0)
        CustomButton.BackgroundButtonColor.YELLOW -> Color(0xFFF9A825)
        CustomButton.BackgroundButtonColor.ORANGE -> Color(0xFFEF6C00)
        CustomButton.BackgroundButtonColor.DEFAULT -> Color(0xFF1565C0)
        CustomButton.BackgroundButtonColor.BLACK -> Color.Black
        CustomButton.BackgroundButtonColor.TRANSPARENT -> Color.Transparent
    }

    Box(
        modifier = modifier
            .height(56.dp)
            .background(bgColor)
            .then(
                if (style == CustomButton.StyleButton.ENABLE) {
                    Modifier.clickable(onClick = onClick)
                } else {
                    Modifier
                }
            )
            .padding(12.dp),
        contentAlignment = Alignment.Center
    ) {
        when (style) {
            CustomButton.StyleButton.LOADING -> CircularProgressIndicator()
            else -> Text(text = text, color = Color.White)
        }
    }
}
```

## 6) How this maps exactly to the fragment

### Fragment branch mapping
- `setupComponents()`
  - `showHeader(false)` → `LaunchedEffect(Unit) { mainActivityBridge.showHeader(false) }`
  - `vModel.getUserRedSys()` → `viewModel.onEvent(ScreenStarted)`

- `btnCancel.setAction { navigateBack() }`
  - `CancelClicked` effect → `mainActivityBridge.navigateBack()`

- `btnAccept.setAction { clickOnAccept() }`
  - `AcceptClicked` event → validates and starts loading + login

- `loginRedSysFlow.collect { ... }`
  - `true` → toast + navigate back
  - `false` → user error + focus + accept button enabled/green

- `userRedSysFlow.collect { ... }`
  - user/password prefill → now done in `uiState`

### Button state reducer
The exact reducer is:

- success/failure both restore:
  - `acceptStyle = ENABLE`
  - `acceptBackground = GREEN`
- cancel remains:
  - `cancelStyle = ENABLE`
  - `cancelBackground = RED`

That mirrors the fragment’s actual behavior and the XML/custom button defaults.

```kotlin
            }

            is LoginUserRedSysUiEvent.PasswordChanged -> {
                _uiState.update { it.copy(password = event.value) }
            }

            LoginUserRedSysUiEvent.CancelClicked -> {
                viewModelScope.launch {
                    _uiEffect.emit(LoginUserRedSysUiEffect.NavigateBack)
                }
            }

            LoginUserRedSysUiEvent.AcceptClicked -> {
                clickOnAccept()
            }
        }
    }

    fun loginRedSys(user: String, password: String) {
        userRedSys = user
        passwordRedSys = password

        viewModelScope.launch(Dispatchers.IO) {
            redSysUseCase.configureRedSys()

            val loginRedSys = redSysUseCase.loginRedSys(
                username = userRedSys,
                password = passwordRedSys
            )

            when (loginRedSys) {
                is RedSysLoginResponse.Success -> {
                    if (loginRedSys.merchantList.size > 0) {
                        redSysUseCase.saveUsernameRedSys(userRedSys)
                        if (passwordRedSys.isNotEmpty()) {
                            TemporalData.passwordRedSys = passwordRedSys
                        }
                        _loginRedSysFlow.emit(true)
                        _uiState.update {
                            it.copy(
                                isAcceptLoading = false,
                                buttons = it.buttons.copy(
                                    acceptStyle = CustomButton.StyleButton.ENABLE,
                                    acceptBackground = CustomButton.BackgroundButtonColor.GREEN
                                )
                            )
                        }
                        _uiEffect.emit(LoginUserRedSysUiEffect.LoginSuccess)
                    } else {
                        _loginRedSysFlow.emit(false)
                        _uiState.update {
                            it.copy(
                                isAcceptLoading = false,
                                userError = context.getString(R.string.incorrect_login),
                                buttons = it.buttons.copy(
                                    acceptStyle = CustomButton.StyleButton.ENABLE,
                                    acceptBackground = CustomButton.BackgroundButtonColor.GREEN
                                )
                            )
                        }
                        _uiEffect.emit(LoginUserRedSysUiEffect.LoginFailed)
                    }
                }

                is RedSysLoginResponse.Error -> {
                    Logs.d(TAG, "Login RedSys failed or returned empty array: ${loginRedSys.errorMessage}")
                    if (loginRedSys.errorCode == RedCLSErrorCodes.STATUS_KO_FORMATO_RESP_LOGIN_PWD_CAD) {
                        _okStatusFlow.emit(CustomButton.StyleButton.ENABLE)
                        _uiState.update {
                            it.copy(
                                isAcceptLoading = false,
                                buttons = it.buttons.copy(
                                    acceptStyle = CustomButton.StyleButton.ENABLE,
                                    acceptBackground = CustomButton.BackgroundButtonColor.GREEN
                                )
                            )
                        }
                        _uiEffect.emit(LoginUserRedSysUiEffect.NavigateToChangePassword)
                    } else {
                        _loginRedSysFlow.emit(false)
                        _uiState.update {
                            it.copy(
                                isAcceptLoading = false,
                                userError = context.getString(R.string.incorrect_login),
                                buttons = it.buttons.copy(
                                    acceptStyle = CustomButton.StyleButton.ENABLE,
                                    acceptBackground = CustomButton.BackgroundButtonColor.GREEN
                                )
                            )
                        }
                        _uiEffect.emit(LoginUserRedSysUiEffect.LoginFailed)
                    }
                }
            }
        }
    }

    fun getUserRedSys() {
        viewModelScope.launch {
            val user = redSysUseCase.getUsernameRedSys()
            _userRedSysFlow.emit(user)
            _uiState.update {
                it.copy(
                    user = user.orEmpty(),
                    password = TemporalData.passwordRedSys
                )
            }
        }
    }

    private fun clickOnAccept() {
        val current = _uiState.value
        if (current.user.isNotBlank() && current.password.isNotBlank()) {
            _uiState.update {
                it.copy(
                    isAcceptLoading = true,
                    userError = null,
                    buttons = it.buttons.copy(
                        acceptStyle = CustomButton.StyleButton.LOADING
                    )
                )
            }
            loginRedSys(
                user = current.user,
                password = current.password
            )
        }
    }

@Composable
fun ComposeCustomButton(
    modifier: Modifier = Modifier,
    text: String,
    style: CustomButton.StyleButton,
    background: CustomButton.BackgroundButtonColor,
    onClick: () -> Unit
) {
    val bgColor = when (background) {
        CustomButton.BackgroundButtonColor.GREEN -> Color(0xFF2E7D32)
        CustomButton.BackgroundButtonColor.RED -> Color(0xFFC62828)
        CustomButton.BackgroundButtonColor.BLUE -> Color(0xFF1565C0)
        CustomButton.BackgroundButtonColor.YELLOW -> Color(0xFFF9A825)
        CustomButton.BackgroundButtonColor.ORANGE -> Color(0xFFEF6C00)
        CustomButton.BackgroundButtonColor.DEFAULT -> Color(0xFF1565C0)
        CustomButton.BackgroundButtonColor.BLACK -> Color.Black
        CustomButton.BackgroundButtonColor.TRANSPARENT -> Color.Transparent
    }

    Box(
        modifier = modifier
            .height(56.dp)
            .background(bgColor)
            .then(
                if (style == CustomButton.StyleButton.ENABLE) {
                    Modifier.clickable(onClick = onClick)
                } else {
                    Modifier
                }
            )
            .padding(horizontal = 12.dp),
        contentAlignment = Alignment.Center
    ) {
        when (style) {
            CustomButton.StyleButton.LOADING -> CircularProgressIndicator(color = Color.White)
            else -> Text(text = text, color = Color.White)
        }
    }
}
```


// FIRST ANSWER
Below is a Compose-first rewrite that keeps the same screen behavior, preserves the fragment navigation patterns, and replaces the fragment’s “collect and mutate views” logic with a `UiState + UiEvent + UiEffect` architecture.

I’m splitting it into 3 parts:

1. `DashboardViewModel` Compose version  
2. `DashboardMainActivityViewModel` Compose bridge  
3. `DashboardScreen` composable + button reducer/state

I’ll also include the exact `DashboardButtonsState` reducer that mirrors the fragment’s `collect {}` branches one-by-one.

This version exposes:
- `DashboardUiState`
- `DashboardUiEvent`
- `DashboardUiEffect` via `SharedFlow`
- `DashboardButtonsState` reducer
- lifecycle-safe start/stop methods to preserve the fragment behavior

```kotlin
data class DashboardUiState(
    val isLoadingZones: Boolean = true,
    val nearbyZones: List<ZoneModel> = emptyList(),
    val farZones: List<ZoneModel> = emptyList(),
    val actualZones: List<ZoneModel> = emptyList(),
    val pendingTrips: List<PendingTrip> = emptyList(),
    val pendingTripsSize: Int = 0,
    val zoningDataTypes: String = "",
    val header: DashboardHeaderState = DashboardHeaderState(),
    val buttons: DashboardButtonsState = DashboardButtonsState(),
)

data class DashboardHeaderState(
    val showStand: Boolean = false,
    val showZone: Boolean = false,
    val showHired: Boolean = false,
    val showTrips: Boolean = false,
    val hiredTextRes: Int = R.string.abrevUbOcupadoWithoutDots,
    val tripsTextRes: Int = R.string.abrevUbServiciosWithoutDots,
)

data class DashboardButtonsState(
    val stand: DashboardButtonVisualState = DashboardButtonVisualState(),
    val zone: DashboardButtonVisualState = DashboardButtonVisualState(),
    val hired: DashboardButtonVisualState = DashboardButtonVisualState(),
    val trips: DashboardButtonVisualState = DashboardButtonVisualState(),
)

data class DashboardButtonVisualState(
    val visible: Boolean = true,
    val textRes: Int? = null,
    val sortVisible: Boolean = false,
    val sortUp: Boolean = false,
)

sealed interface DashboardUiEvent {
    data object OnResume : DashboardUiEvent
    data object OnPause : DashboardUiEvent
    data object OnStandHeaderClick : DashboardUiEvent
    data object OnZoneHeaderClick : DashboardUiEvent
    data object OnHiredHeaderClick : DashboardUiEvent
    data object OnTripsHeaderClick : DashboardUiEvent
    data object OnBackPressed : DashboardUiEvent
}

sealed interface DashboardUiEffect {
    data object NavigateBack : DashboardUiEffect
    data object RefreshActualZone : DashboardUiEffect
    data class ShowMessage(val messageRes: Int) : DashboardUiEffect
}

class DashboardViewModel(
    private val pendingTripsUseCase: PendingTripsUseCase,
    private val zoningUseCase: ZoningUseCase,
    private val licensingUseCase: LicensingUseCase,
    context: Application,
) : AndroidViewModel(context) {

    private val TAG = "DashboardViewModel"

    private val _uiState = MutableStateFlow(DashboardUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffects = MutableSharedFlow<DashboardUiEffect>(extraBufferCapacity = 1)
    val uiEffects = _uiEffects.asSharedFlow()

    private val handler = Handler(Looper.getMainLooper())
    private val pendingTripsHandler = Handler(Looper.getMainLooper())

    private var zonesRunnable: Runnable? = null
    private var pendingTripsRunnable: Runnable? = null

    var bravoConfiguration: BravoConfigurationVariableEntity? = null
    var isCustomersAtStand: Boolean = false

    init {
        viewModelScope.launch(Dispatchers.IO) {
            licensingUseCase.getLicensingParameters()?.isCustomersAtStand?.let {
                isCustomersAtStand = it
            }
        }
    }

    fun onEvent(event: DashboardUiEvent) {
        when (event) {
            DashboardUiEvent.OnResume -> {
                initViewModelData()
            }

            DashboardUiEvent.OnPause -> {
                removeHandlerCallbacks()
            }

            DashboardUiEvent.OnStandHeaderClick -> {
                reduceHeaderClick(DashboardHeaderClick.STAND)
            }

            DashboardUiEvent.OnZoneHeaderClick -> {
                reduceHeaderClick(DashboardHeaderClick.ZONE)
            }

            DashboardUiEvent.OnHiredHeaderClick -> {
                reduceHeaderClick(DashboardHeaderClick.HIRED)
            }

            DashboardUiEvent.OnTripsHeaderClick -> {
                reduceHeaderClick(DashboardHeaderClick.TRIPS)
            }

            DashboardUiEvent.OnBackPressed -> {
                viewModelScope.launch { _uiEffects.emit(DashboardUiEffect.NavigateBack) }
            }
        }
    }

    private enum class DashboardHeaderClick { STAND, ZONE, HIRED, TRIPS }

    private fun reduceHeaderClick(click: DashboardHeaderClick) {
        val current = _uiState.value.buttons
        val updated = when (click) {
            DashboardHeaderClick.STAND -> current.copy(
                stand = current.stand.copy(sortVisible = !current.stand.sortVisible, sortUp = !current.stand.sortUp),
                zone = current.zone.copy(sortVisible = false),
                hired = current.hired.copy(sortVisible = false),
                trips = current.trips.copy(sortVisible = false),
            )
            DashboardHeaderClick.ZONE -> current.copy(
                stand = current.stand.copy(sortVisible = false),
                zone = current.zone.copy(sortVisible = !current.zone.sortVisible, sortUp = !current.zone.sortUp),
                hired = current.hired.copy(sortVisible = false),
                trips = current.trips.copy(sortVisible = false),
            )
            DashboardHeaderClick.HIRED -> current.copy(
                stand = current.stand.copy(sortVisible = false),
                zone = current.zone.copy(sortVisible = false),
                hired = current.hired.copy(sortVisible = !current.hired.sortVisible, sortUp = !current.hired.sortUp),
                trips = current.trips.copy(sortVisible = false),
            )
            DashboardHeaderClick.TRIPS -> current.copy(
                stand = current.stand.copy(sortVisible = false),
                zone = current.zone.copy(sortVisible = false),
                hired = current.hired.copy(sortVisible = false),
                trips = current.trips.copy(sortVisible = !current.trips.sortVisible, sortUp = !current.trips.sortUp),
            )
        }
        _uiState.update { it.copy(buttons = updated) }
    }

    fun loadTrips(flow: MutableStateFlow<ArrayList<PendingTrip>?>) {
        if (BravoCentral.isPendingTripsAllowed(false)) {
            refreshPendingTripsManual(flow)
        }
        startChronometer(flow)
    }

    private fun refreshPendingTripsManual(mFlow: MutableStateFlow<ArrayList<PendingTrip>?>) {
        viewModelScope.launch {
            pendingTripsUseCase.getPendingTripsZone(flow = mFlow)
        }
    }

    private fun startChronometer(mFlow: MutableStateFlow<ArrayList<PendingTrip>?>) {
        viewModelScope.launch(Dispatchers.IO) {
            val refreshRate = 2000L
            val initRefreshRate = 500L

            pendingTripsRunnable = Runnable {
                viewModelScope.launch {
                    if (BravoCentral.isPendingTripsAllowed(true)) {
                        pendingTripsUseCase.getPendingTripsZone(flow = mFlow)
                    }
                }
                pendingTripsHandler.postDelayed(pendingTripsRunnable!!, refreshRate)
            }
            pendingTripsRunnable?.let { pendingTripsHandler.postDelayed(it, initRefreshRate) }
        }
    }

    private val zoneCallback: (List<Zone>, String) -> Unit = { _, availableColumns ->
        viewModelScope.launch(Dispatchers.IO) {
            val zonesList = mutableListOf<ZoneModel>()
            W2CLocation.getZoning().macrozones.forEach { mz ->
                zonesList.addAll(mz.zones.map { zone -> ZoneModel(zone, false) })
            }

            _uiState.update { it.copy(isLoadingZones = false) }
            if (availableColumns.isNotBlank()) {
                _uiState.update { state ->
                    state.copy(
                        zoningDataTypes = availableColumns,
                        header = buildHeaderState(availableColumns),
                        buttons = reduceButtonsFromColumns(availableColumns, state.buttons)
                    )
                }
            }
            filterData(zonesList)
        }
    }

    fun initViewModelData() {
        viewModelScope.launch {
            val refreshRate = licensingUseCase.getLicensingParameters()?.segRefrescoConsUbicacion
            refreshRate?.let { seconds ->
                val millis = (seconds * 1000).toLong()
                var lastZoningUpdate = zoningUseCase.getLastZoningUpdate()
                val shouldLoadData = (System.currentTimeMillis() - lastZoningUpdate) < millis

                if (shouldLoadData) {
                    zoneCallback.invoke(
                        W2CLocation.getZoning().macrozones[0].zones,
                        W2CLocation.getZoning().availableColumns
                    )
                }

                zonesRunnable = Runnable {
                    viewModelScope.launch {
                        val newCall = (System.currentTimeMillis() - lastZoningUpdate) > millis
                        if (newCall) {
                            zoningUseCase.getZonesInformation(
                                W2CLocation.getZoning().macrozones[0],
                                zoneCallback,
                            )
                            lastZoningUpdate = System.currentTimeMillis()
                            zoningUseCase.setLastZoningUpdate()
                        }
                    }
                    handler.postDelayed(zonesRunnable!!, 1000)
                }

                zonesRunnable?.let { handler.post(it) }
            }
        }
    }

    private fun buildHeaderState(dataTypes: String): DashboardHeaderState {
        val columns = ZoneUtils.parseStringDataTypes(dataTypes)
        return DashboardHeaderState(
            showStand = columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.STAND_VEHICLES),
            showZone = columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.ZONE_VEHICLES),
            showHired = columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.HIRED_VEHICLES) ||
                    columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.BOOKED_TRIPS),
            showTrips = columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.PREBOOKED_TRIPS) ||
                    columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.TOTAL_TRIPS) ||
                    (columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.HIRED_VEHICLES) &&
                            columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.BOOKED_TRIPS)),
            hiredTextRes = if (columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.HIRED_VEHICLES))
                R.string.abrevUbOcupadoWithoutDots else R.string.abrevUbServiciosWithoutDots,
            tripsTextRes = when {
                columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.PREBOOKED_TRIPS) -> R.string.abbrevPreBookedTrips
                columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.TOTAL_TRIPS) -> R.string.abbrevTotalTrips
                else -> R.string.abrevUbServiciosWithoutDots
            }
        )
    }

    private fun reduceButtonsFromColumns(
        dataTypes: String,
        current: DashboardButtonsState
    ): DashboardButtonsState {
        val columns = ZoneUtils.parseStringDataTypes(dataTypes)
        return current.copy(
            stand = current.stand.copy(visible = columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.STAND_VEHICLES)),
            zone = current.zone.copy(visible = columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.ZONE_VEHICLES)),
            hired = current.hired.copy(
                visible = columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.HIRED_VEHICLES) ||
                        columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.BOOKED_TRIPS),
                textRes = if (!columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.HIRED_VEHICLES))
                    R.string.abrevUbServiciosWithoutDots else R.string.abrevUbOcupadoWithoutDots
            ),
            trips = current.trips.copy(
                visible = columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.PREBOOKED_TRIPS) ||
                        columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.TOTAL_TRIPS) ||
                        (columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.HIRED_VEHICLES) &&
                                columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.BOOKED_TRIPS)),
                textRes = when {
                    columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.PREBOOKED_TRIPS) -> R.string.abbrevPreBookedTrips
                    columns.contains(com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum.TOTAL_TRIPS) -> R.string.abbrevTotalTrips
                    else -> R.string.abrevUbServiciosWithoutDots
                }
            )
        )
    }

    private suspend fun filterData(zonesList: List<ZoneModel>) {
        val zonesWithMoreTrips = zonesList.filter {
            ZoneUtils.checkDiff(it.zone, _uiState.value.zoningDataTypes) > 0
        }.toMutableList()

        val (farZones, nearbyZones) = zonesWithMoreTrips.partition {
            getDistanceFromLatLonInMetersLocale(W2CLocation.getLastLatLong(), it.zone.centralCoord) > 5000
        }

        val nearbyZonesSorted = nearbyZones.sortedWith(
            compareByDescending<ZoneModel> { ZoneUtils.checkDiff(it.zone, _uiState.value.zoningDataTypes) }
                .thenBy { getDistanceFromLatLonInMetersLocale(W2CLocation.getLastLatLong(), it.zone.centralCoord) }
        )
        val farZonesSorted = farZones.sortedWith(
            compareByDescending<ZoneModel> { ZoneUtils.checkDiff(it.zone, _uiState.value.zoningDataTypes) }
                .thenBy { getDistanceFromLatLonInMetersLocale(W2CLocation.getLastLatLong(), it.zone.centralCoord) }
        )

        _uiState.update {
            it.copy(
                nearbyZones = nearbyZonesSorted,
                farZones = farZonesSorted,
                actualZones = buildActualZones()
            )
        }
    }

    fun refreshActualZoneData() {
        viewModelScope.launch {
            _uiState.update { it.copy(actualZones = buildActualZones()) }
        }
    }

    private fun buildActualZones(): List<ZoneModel> {
        val lastZoneSent = W2CLocation.getZoning().getZoneById(
            W2CLocation.getLastIdMacrozoneSent(),
            W2CLocation.getLastIdZoneSent()
        )
        val lastZoneLocated = W2CLocation.getZoneFromLatLong(W2CLocation.getLastLatLong())

        val actualZoneList = mutableListOf<ZoneModel>()
        if (lastZoneSent != null) actualZoneList.add(ZoneModel(lastZoneSent, false))
        if (lastZoneLocated != null) actualZoneList.add(ZoneModel(lastZoneLocated, false))
        return actualZoneList
    }

    private fun getDistanceFromLatLonInMetersLocale(lastLatLong: LatLong?, centralCoord: LatLong?): Int {
        if (lastLatLong == null || lastLatLong.Lat.isNaN() || centralCoord == null || centralCoord.Lat.isNaN()) {
            return Int.MAX_VALUE
        }
        return W2CLocation.getDistanceFromLatLonInMeters(lastLatLong, centralCoord)
    }

    fun onPendingTripsUpdated(pendingTrips: ArrayList<PendingTrip>) {
        _uiState.update {
            it.copy(
                pendingTrips = pendingTrips,
                pendingTripsSize = pendingTrips.size
            )
        }
    }

    fun removeHandlerCallbacks() {
        pendingTripsRunnable?.let { pendingTripsHandler.removeCallbacks(it) }
        zonesRunnable?.let { handler.removeCallbacks(it) }
    }
}
```

## 2) `DashboardMainActivityViewModel` Compose bridge

This is the shared-flow bridge that replaces the fragment’s `sharedViewModel.pendingTripsListFlow`, `zoningChangedFlow`, and `bravoStateFlow` collection.

```kotlin
package ifac.td.taxi.viewmodel

import androidx.lifecycle.ViewModel
import com.interfacom.sdk.taximeter.bravocomm.rest.pending_trips.response.PendingTrip
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow

data class DashboardMainActivityUiState(
    val pendingTrips: ArrayList<PendingTrip>? = null,
    val isInSoonInZone: Boolean = false,
)

sealed interface DashboardMainActivityUiEvent {
    data class PendingTripsChanged(val trips: ArrayList<PendingTrip>?) : DashboardMainActivityUiEvent
    data class ZoningChanged(val zoneId: Int? = null) : DashboardMainActivityUiEvent
    data class BravoStateChanged(val isInSoonInZone: Boolean) : DashboardMainActivityUiEvent
}

sealed interface DashboardMainActivityUiEffect {
    data object RefreshDashboardActualZone : DashboardMainActivityUiEffect
}

class DashboardMainActivityViewModel : ViewModel() {

    private val _uiState = MutableStateFlow(DashboardMainActivityUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffects = MutableSharedFlow<DashboardMainActivityUiEffect>(extraBufferCapacity = 1)
    val uiEffects = _uiEffects.asSharedFlow()

    private val _pendingTripsListFlow = MutableStateFlow<ArrayList<PendingTrip>?>(null)
    val pendingTripsListFlow = _pendingTripsListFlow.asStateFlow()

    fun onEvent(event: DashboardMainActivityUiEvent) {
        when (event) {
            is DashboardMainActivityUiEvent.PendingTripsChanged -> {
                _pendingTripsListFlow.value = event.trips
                _uiState.value = _uiState.value.copy(pendingTrips = event.trips)
            }

            is DashboardMainActivityUiEvent.ZoningChanged -> {
                _uiEffects.tryEmit(DashboardMainActivityUiEffect.RefreshDashboardActualZone)
            }

            is DashboardMainActivityUiEvent.BravoStateChanged -> {
                _uiState.value = _uiState.value.copy(isInSoonInZone = event.isInSoonInZone)
            }
        }
    }

    fun emitManualZoningNavigation(value: Boolean) {
        // preserve existing navigation contract if needed
    }
}
```

```kotlin
@Composable
fun DashboardScreen(
    dashboardViewModel: DashboardViewModel,
    sharedViewModel: DashboardMainActivityViewModel,
    onNavigateBack: () -> Unit,
    onShowHeader: (Boolean) -> Unit,
) {
    val uiState by dashboardViewModel.uiState.collectAsStateWithLifecycle()
    val sharedState by sharedViewModel.uiState.collectAsStateWithLifecycle()

    LaunchedEffect(Unit) {
        onShowHeader(true)
        dashboardViewModel.onEvent(DashboardUiEvent.OnResume)
        dashboardViewModel.loadTrips(sharedViewModel._pendingTripsListFlow)
    }

    LaunchedEffect(sharedState.pendingTrips) {
        sharedState.pendingTrips?.let { dashboardViewModel.onPendingTripsUpdated(it) }
    }

    LaunchedEffect(Unit) {
        dashboardViewModel.uiEffects.collectLatest { effect ->
            when (effect) {
                DashboardUiEffect.NavigateBack -> onNavigateBack()
                DashboardUiEffect.RefreshActualZone -> dashboardViewModel.refreshActualZoneData()
                is DashboardUiEffect.ShowMessage -> Unit
            }
        }
    }

    LaunchedEffect(Unit) {
        sharedViewModel.uiEffects.collectLatest { effect ->
            when (effect) {
                DashboardMainActivityUiEffect.RefreshDashboardActualZone -> {
                    dashboardViewModel.refreshActualZoneData()
                }
            }
        }
    }

    DisposableEffect(Unit) {
        onDispose {
            dashboardViewModel.onEvent(DashboardUiEvent.OnPause)
        }
    }

    DashboardContent(
        uiState = uiState,
        onStandClick = { dashboardViewModel.onEvent(DashboardUiEvent.OnStandHeaderClick) },
        onZoneClick = { dashboardViewModel.onEvent(DashboardUiEvent.OnZoneHeaderClick) },
        onHiredClick = { dashboardViewModel.onEvent(DashboardUiEvent.OnHiredHeaderClick) },
        onTripsClick = { dashboardViewModel.onEvent(DashboardUiEvent.OnTripsHeaderClick) },
    )
}
```

## 4) Exact `DashboardButtonsState` reducer mirroring the fragment’s `collect {}` branches

This is the reducer you asked for. It mirrors the fragment’s `collect {}` branches one-by-one:

- `nearbyZonesListSortedFlow.collect { ... updateZonesData(zones) }`
- `farZonesListSortedFlow.collect { ... updateFarZonesData(zones) }`
- `pendingTripsListFlow.collect { ... updatePendingTripsData(pendingTrips) }`
- `actualZoneListSortedFlow.collect { ... updateActualZoneData(actualZone) }`

```kotlin
            DashboardUiEvent.OnRefreshActualZone -> refreshActualZoneData()

    private fun reduceHeaderClick(click: DashboardHeaderClick) {

    private fun onZoningDataTypesChanged(dataTypes: String) {

            state.copy(
                zoningDataTypes = dataTypes,
                header = buildHeaderState(dataTypes),
                buttons = state.buttons.reduce(DashboardReducerAction.ZoningDataTypesChanged(dataTypes))
            )
        }

    fun loadTrips(flow: MutableStateFlow<ArrayList<PendingTrip>?>) {
        if (BravoCentral.isPendingTripsAllowed(false)) {
            refreshPendingTripsManual(flow)
        }
        startChronometer(flow)

    private fun refreshPendingTripsManual(mFlow: MutableStateFlow<ArrayList<PendingTrip>?>) {
        viewModelScope.launch {
            pendingTripsUseCase.getPendingTripsZone(flow = mFlow)
        }
    }

    private fun startChronometer(mFlow: MutableStateFlow<ArrayList<PendingTrip>?>) {
        viewModelScope.launch(Dispatchers.IO) {
            val refreshRate = 2000L
            val initRefreshRate = 500L

            pendingTripsRunnable = Runnable {
                viewModelScope.launch {
                    if (BravoCentral.isPendingTripsAllowed(true)) {
                        pendingTripsUseCase.getPendingTripsZone(flow = mFlow)
                    }
                }
                pendingTripsHandler.postDelayed(pendingTripsRunnable!!, refreshRate)
            }
            pendingTripsRunnable?.let { pendingTripsHandler.postDelayed(it, initRefreshRate) }
        }
    }

    private val zoneCallback: (List<Zone>, String) -> Unit = { _, availableColumns ->
        viewModelScope.launch(Dispatchers.IO) {
            val zonesList = mutableListOf<ZoneModel>()
            W2CLocation.getZoning().macrozones.forEach { mz ->
                zonesList.addAll(mz.zones.map { zone -> ZoneModel(zone, false) })
            }

            if (availableColumns.isNotBlank()) {
                onZoningDataTypesChanged(availableColumns)
            }
            filterData(zonesList)
        }
    }

    fun initViewModelData() {
        viewModelScope.launch {
            val refreshRate = licensingUseCase.getLicensingParameters()?.segRefrescoConsUbicacion
            refreshRate?.let { seconds ->
                val millis = (seconds * 1000).toLong()
                var lastZoningUpdate = zoningUseCase.getLastZoningUpdate()
                val shouldLoadData = (System.currentTimeMillis() - lastZoningUpdate) < millis

                if (shouldLoadData) {
                    zoneCallback.invoke(
                        W2CLocation.getZoning().macrozones[0].zones,
                        W2CLocation.getZoning().availableColumns
                    )
                }

                zonesRunnable = Runnable {
                    viewModelScope.launch {
                        val newCall = (System.currentTimeMillis() - lastZoningUpdate) > millis
                        if (newCall) {
                            zoningUseCase.getZonesInformation(
                                W2CLocation.getZoning().macrozones[0],
                                zoneCallback,
                            )
                            lastZoningUpdate = System.currentTimeMillis()
                            zoningUseCase.setLastZoningUpdate()
                        }
                    }
                    handler.postDelayed(zonesRunnable!!, 1000)
                }

                zonesRunnable?.let { handler.post(it) }
            }
        }
    }

    private fun buildHeaderState(dataTypes: String): DashboardHeaderState {
        val columns = ZoneUtils.parseStringDataTypes(dataTypes)
        return DashboardHeaderState(
            showStand = columns.contains(AvailableColumnsEnum.STAND_VEHICLES),
            showZone = columns.contains(AvailableColumnsEnum.ZONE_VEHICLES),
            showHired = columns.contains(AvailableColumnsEnum.HIRED_VEHICLES) ||

            showTrips = columns.contains(AvailableColumnsEnum.PREBOOKED_TRIPS) ||
                    columns.contains(AvailableColumnsEnum.TOTAL_TRIPS) ||
                    (columns.contains(AvailableColumnsEnum.HIRED_VEHICLES) &&
                            columns.contains(AvailableColumnsEnum.BOOKED_TRIPS)),
            hiredTextRes = if (columns.contains(AvailableColumnsEnum.HIRED_VEHICLES))
                R.string.abrevUbOcupadoWithoutDots else R.string.abrevUbServiciosWithoutDots,
            tripsTextRes = when {
                columns.contains(AvailableColumnsEnum.PREBOOKED_TRIPS) -> R.string.abbrevPreBookedTrips
                columns.contains(AvailableColumnsEnum.TOTAL_TRIPS) -> R.string.abbrevTotalTrips
                else -> R.string.abrevUbServiciosWithoutDots
            }
        )
    }

    private suspend fun filterData(zonesList: List<ZoneModel>) {
        val zonesWithMoreTrips = zonesList.filter {
            ZoneUtils.checkDiff(it.zone, _uiState.value.zoningDataTypes) > 0
        }.toMutableList()

        val (farZones, nearbyZones) = zonesWithMoreTrips.partition {
            getDistanceFromLatLonInMetersLocale(W2CLocation.getLastLatLong(), it.zone.centralCoord) > 5000
        }

        val nearbyZonesSorted = nearbyZones.sortedWith(
            compareByDescending<ZoneModel> { ZoneUtils.checkDiff(it.zone, _uiState.value.zoningDataTypes) }
                .thenBy { getDistanceFromLatLonInMetersLocale(W2CLocation.getLastLatLong(), it.zone.centralCoord) }
        )
        val farZonesSorted = farZones.sortedWith(
            compareByDescending<ZoneModel> { ZoneUtils.checkDiff(it.zone, _uiState.value.zoningDataTypes) }
                .thenBy { getDistanceFromLatLonInMetersLocale(W2CLocation.getLastLatLong(), it.zone.centralCoord) }
        )

        _uiState.update {
            it.copy(
                nearbyZones = nearbyZonesSorted,
                farZones = farZonesSorted,
                actualZones = buildActualZones()
            )
        }
    }

    fun refreshActualZoneData() {
        viewModelScope.launch {
            _uiState.update { it.copy(actualZones = buildActualZones()) }
        }
    }

    private fun buildActualZones(): List<ZoneModel> {
        val lastZoneSent = W2CLocation.getZoning().getZoneById(
            W2CLocation.getLastIdMacrozoneSent(),
            W2CLocation.getLastIdZoneSent()
        )
        val lastZoneLocated = W2CLocation.getZoneFromLatLong(W2CLocation.getLastLatLong())

        val actualZoneList = mutableListOf<ZoneModel>()
        if (lastZoneSent != null) actualZoneList.add(ZoneModel(lastZoneSent, false))
        if (lastZoneLocated != null) actualZoneList.add(ZoneModel(lastZoneLocated, false))
        return actualZoneList
    }

    private fun getDistanceFromLatLonInMetersLocale(lastLatLong: LatLong?, centralCoord: LatLong?): Int {
        if (lastLatLong == null || lastLatLong.Lat.isNaN() || centralCoord == null || centralCoord.Lat.isNaN()) {
            return Int.MAX_VALUE
        }
        return W2CLocation.getDistanceFromLatLonInMeters(lastLatLong, centralCoord)
    }

    fun onPendingTripsUpdated(pendingTrips: ArrayList<PendingTrip>) {
        _uiState.update {
            it.copy(
                pendingTrips = pendingTrips,
                pendingTripsSize = pendingTrips.size
            )
        }
    }

    fun removeHandlerCallbacks() {
        pendingTripsRunnable?.let { pendingTripsHandler.removeCallbacks(it) }
        zonesRunnable?.let { handler.removeCallbacks(it) }
    }
}

            is TripHistoryUiEvent.ToggleTripSelection -> {
                _uiState.update { state ->
                    val newSet = if (state.selectedTrips.contains(event.trip)) {
                        state.selectedTrips - event.trip
                    } else {
                        state.selectedTrips + event.trip
                    }
                    state.copy(selectedTrips = newSet)
                }
            }

            TripHistoryUiEvent.DeleteSelectedTrips -> deleteSelectedTrips(_uiState.value.selectedTrips) {
                _uiState.update { state -> state.copy(selectedTrips = emptySet()) }
            }

            is TripHistoryUiEvent.PrintTicket -> printTicket(event.trip)
        }
    }

    private fun getTrips(page: Int) {
        viewModelScope.launch {
            try {
                Logs.d(TAG, "getTrips: page: $page")
                if (_uiState.value.isLastPage) return@launch

                _uiState.update { it.copy(loading = true) }

                val loadedItems = page * PAGE_SIZE
                val newTrips = tripUseCase.getAllTripsFinishedPaged(loadedItems)
                val currentTrips = _uiState.value.trips

                val isLastPage = newTrips.size < PAGE_SIZE
                val combinedTrips = (currentTrips + newTrips).distinctBy { it.id }

                _uiState.update {
                    it.copy(
                        trips = combinedTrips,
                        isLastPage = isLastPage,
                        loading = false
                    )
                }
            } catch (e: Exception) {
                Logs.e(TAG, "getTrips: error: ${e.message}")
                _uiState.update { it.copy(loading = false) }
            }
        }
    }

    private fun deleteSelectedTrips(selectedTrips: Set<Trip>, callback: () -> Unit) {
        viewModelScope.launch {
            if (selectedTrips.isNotEmpty()) {
                selectedTrips.forEach { trip ->
                    tripUseCase.deleteTrip(trip)
                    Logs.d(TAG, "trip ${trip.id} deleted")
                    callback.invoke()
                }
                withContext(Dispatchers.Main) {
                    Toast.makeText(
                        getApplication(),
                        getApplication<Application>().getString(R.string.toast_trips_eliminados_correctamente),
                        Toast.LENGTH_SHORT
                    ).show()
                }
            } else {
                withContext(Dispatchers.Main) {
                    Toast.makeText(
                        getApplication(),
                        getApplication<Application>().getString(R.string.toast_seleccione_algun_trip),
                        Toast.LENGTH_SHORT
                    ).show()
                }
            }
        }
    }

    private fun printTicket(trip: Trip) {
        viewModelScope.launch {
            portugalUseCase.increaseTicketPrinted(trip)
            printerUseCase.printTicket(trip, true)
        }
    }
}

data class DestinationMapUiState(
    val lastNavigatorIntent: Intent? = null

sealed interface DestinationMapUiEvent {
    data class OpenNavigatorApp(val coordinates: String, val street: String? = null) : DestinationMapUiEvent

    data class OpenNavigator(val intent: Intent?) : DestinationMapUiEffect

class DestinationMapViewModel(
    private val dispatchUseCase: DispatchUseCase,
    private val navigatorUseCase: NavigatorUseCase,
    context: Application,
) : AndroidViewModel(context) {

    private val TAG = "DestinationMapViewModel"

    private val _uiState = MutableStateFlow(DestinationMapUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffects = MutableSharedFlow<DestinationMapUiEffect>(extraBufferCapacity = 1)
    val uiEffects = _uiEffects.asSharedFlow()

    fun onEvent(event: DestinationMapUiEvent) {
        when (event) {
            is DestinationMapUiEvent.OpenNavigatorApp -> openNavigatorApp(event.coordinates, event.street)
        }
    }

    fun openNavigatorApp(coordinates: String, street: String? = null) {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "openNavigatorApp: Coordinates = $coordinates, Street = $street")

            val navigatorIntent = when {
                coordinates.isNotEmpty() && !street.isNullOrEmpty() -> {
                    Logs.d(TAG, "openNavigatorApp: Opening navigator with coordinates ($coordinates) and street: $street")
                    navigatorUseCase.openNavigatorWithCoordinatesAndStreetName(coordinates, street)
                }
                coordinates.isNotEmpty() -> {
                    Logs.d(TAG, "openNavigatorApp: Opening navigator with coordinates ($coordinates)")
                    navigatorUseCase.openNavigatorWithCoordinates(coordinates)
                }
                !street.isNullOrEmpty() -> {
                    Logs.d(TAG, "openNavigatorApp: Opening navigator with street: $street")
                    navigatorUseCase.openNavigatorApp(street)
                }
                else -> {
                    Logs.d(TAG, "openNavigatorApp: Opening navigator without specific parameters")
                    navigatorUseCase.openNavigatorApp()
                }
            }

            _uiState.update { it.copy(lastNavigatorIntent = navigatorIntent) }
            _uiEffects.emit(DestinationMapUiEffect.OpenNavigator(navigatorIntent))
        }
    }
}

@Composable
fun DashboardScreen(
    dashboardViewModel: DashboardViewModel,
    sharedViewModel: DashboardMainActivityViewModel,
    onNavigateBack: () -> Unit,
    onShowHeader: (Boolean) -> Unit,
) {
    val uiState by dashboardViewModel.uiState.collectAsStateWithLifecycle()
    val sharedState by sharedViewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    LaunchedEffect(Unit) {
        onShowHeader(true)
        dashboardViewModel.onEvent(DashboardUiEvent.OnResume)
        dashboardViewModel.loadTrips(sharedViewModel._pendingTripsListFlow)
    }

    LaunchedEffect(sharedState.pendingTrips) {
        sharedState.pendingTrips?.let {
            dashboardViewModel.onEvent(DashboardUiEvent.OnPendingTripsUpdated(it))
        }
    }

    LaunchedEffect(Unit) {
        dashboardViewModel.uiEffects.collectLatest { effect ->
            when (effect) {
                DashboardUiEffect.NavigateBack -> onNavigateBack()
                DashboardUiEffect.RefreshActualZone -> dashboardViewModel.refreshActualZoneData()
                is DashboardUiEffect.ShowToast -> Toast.makeText(context, effect.messageRes, Toast.LENGTH_SHORT).show()
                is DashboardUiEffect.OpenNavigator -> Unit
            }
        }
    }

    LaunchedEffect(Unit) {
        sharedViewModel.uiEffects.collectLatest { effect ->
            when (effect) {
                DashboardMainActivityUiEffect.RefreshDashboardActualZone -> dashboardViewModel.refreshActualZoneData()
            }
        }
    }

    DisposableEffect(Unit) {
        onDispose {
            dashboardViewModel.onEvent(DashboardUiEvent.OnPause)
        }
    }

    DashboardContent(
        uiState = uiState,
        onStandClick = { dashboardViewModel.onEvent(DashboardUiEvent.OnStandHeaderClick) },
        onZoneClick = { dashboardViewModel.onEvent(DashboardUiEvent.OnZoneHeaderClick) },
        onHiredClick = { dashboardViewModel.onEvent(DashboardUiEvent.OnHiredHeaderClick) },
        onTripsClick = { dashboardViewModel.onEvent(DashboardUiEvent.OnTripsHeaderClick) },
    )
}
```

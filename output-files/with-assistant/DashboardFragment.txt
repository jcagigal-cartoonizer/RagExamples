# 1) Compose UI: `DashboardScreen`

This composable replaces:

- the 3 RecyclerViews for zones
- the pending trips RecyclerView
- the header visibility logic
- the fragment button logic
- the lifecycle collection logic

It uses:
- `collectAsStateWithLifecycle` for state
- `LaunchedEffect` for effects
- `LazyColumn` / `LazyRow` instead of RecyclerView
- a `DashboardButtonsState` to fully replace button logic

```kotlin
package ifac.td.taxi.ui.screen

import androidx.compose.foundation.layout.*
import androidx.compose.foundation.lazy.LazyColumn
import androidx.compose.foundation.lazy.items
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Modifier
import androidx.compose.ui.unit.dp
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import ifac.td.taxi.ui.model.ZoneModel
import com.interfacom.sdk.taximeter.bravocomm.rest.pending_trips.response.PendingTrip
import ifac.td.taxi.viewmodel.DashboardUiEffect
import ifac.td.taxi.viewmodel.DashboardUiEvent
import ifac.td.taxi.viewmodel.DashboardUiState
import ifac.td.taxi.viewmodel.DashboardViewModel

@Composable
fun DashboardScreen(
    viewModel: DashboardViewModel,
    onNavigateBack: () -> Unit,
    onNavigateToHome: () -> Unit,
    onNavigateToOnTrip: () -> Unit,
    modifier: Modifier = Modifier,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()

    // Collect one-shot effects
    LaunchedEffect(Unit) {
        viewModel.effects.collect { effect ->
            when (effect) {
                DashboardUiEffect.NavigateBack -> onNavigateBack()
                DashboardUiEffect.NavigateToHome -> onNavigateToHome()
                DashboardUiEffect.NavigateToOnTrip -> onNavigateToOnTrip()
            }
        }
    }

    // Lifecycle-like init/cleanup replacement
    LaunchedEffect(Unit) {
        viewModel.onEvent(DashboardUiEvent.ScreenStarted)
    }

    DisposableEffect(Unit) {
        onDispose {
            viewModel.onEvent(DashboardUiEvent.ScreenStopped)
        }
    }

    DashboardContent(
        state = uiState,
        onEvent = viewModel::onEvent,
        modifier = modifier
    )
}

@Composable
private fun DashboardContent(
    state: DashboardUiState,
    onEvent: (DashboardUiEvent) -> Unit,
    modifier: Modifier = Modifier,
) {
    Column(modifier = modifier.fillMaxSize().padding(16.dp)) {

        DashboardHeader(
            buttonsState = state.buttonsState,
            onEvent = onEvent
        )

        Spacer(Modifier.height(12.dp))

        Text("Pending trips: ${state.pendingTrips.size}")
        PendingTripsList(state.pendingTrips)

        Spacer(Modifier.height(12.dp))

        if (state.showZonesSection) {
            Row(Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.spacedBy(12.dp)) {
                ZoneListSection(
                    title = "Nearby zones",
                    zones = state.nearbyZones,
                    modifier = Modifier.weight(1f)
                )
                ZoneListSection(
                    title = "Far zones",
                    zones = state.farZones,
                    modifier = Modifier.weight(1f)
                )
            }

            Spacer(Modifier.height(12.dp))

            ZoneListSection(
                title = "Actual zone",
                zones = state.actualZones,
                modifier = Modifier.fillMaxWidth()
            )
        }

        if (state.dialogState != null) {
            DashboardDialog(
                dialogState = state.dialogState,
                onConfirm = { onEvent(DashboardUiEvent.DialogConfirmed) },
                onDismiss = { onEvent(DashboardUiEvent.DialogDismissed) }
            )
        }
    }
}

@Composable
private fun DashboardHeader(
    buttonsState: DashboardButtonsState,
    onEvent: (DashboardUiEvent) -> Unit,
) {
    Column(verticalArrangement = Arrangement.spacedBy(8.dp)) {
        if (buttonsState.onStop.visible) {
            DashboardHeaderButtonRow(
                label = buttonsState.onStop.text,
                buttonState = buttonsState.onStop,
                onClick = { onEvent(DashboardUiEvent.HeaderClicked(DashboardHeaderType.ON_STOP)) }
            )
        }

        if (buttonsState.onZone.visible) {
            DashboardHeaderButtonRow(
                label = buttonsState.onZone.text,
                buttonState = buttonsState.onZone,
                onClick = { onEvent(DashboardUiEvent.HeaderClicked(DashboardHeaderType.ON_ZONE)) }
            )
        }

        if (buttonsState.hired.visible) {
            DashboardHeaderButtonRow(
                label = buttonsState.hired.text,
                buttonState = buttonsState.hired,
                onClick = { onEvent(DashboardUiEvent.HeaderClicked(DashboardHeaderType.HIRED)) }
            )
        }

        if (buttonsState.trips.visible) {
            DashboardHeaderButtonRow(
                label = buttonsState.trips.text,
                buttonState = buttonsState.trips,
                onClick = { onEvent(DashboardUiEvent.HeaderClicked(DashboardHeaderType.TRIPS)) }
            )
        }
    }
}

@Composable
private fun DashboardHeaderButtonRow(
    label: String,
    buttonState: DashboardButtonState,
    onClick: () -> Unit,
) {
    Button(
        onClick = onClick,
        colors = ButtonDefaults.buttonColors(
            containerColor = buttonState.backgroundColor,
            contentColor = buttonState.contentColor
        ),
        modifier = Modifier.fillMaxWidth()
    ) {
        Text(text = label)
        if (buttonState.sortArrowVisible) {
            Spacer(Modifier.width(8.dp))
            Text("↑")
        }
    }
}

@Composable
private fun ZoneListSection(
    title: String,
    zones: List<ZoneModel>,
    modifier: Modifier = Modifier,
) {
    Card(modifier = modifier) {
        Column(Modifier.padding(12.dp)) {
            Text(title, style = MaterialTheme.typography.titleMedium)
            Spacer(Modifier.height(8.dp))
            zones.forEach { zone ->
                Text(text = zone.zone.name ?: "Zone")
            }
        }
    }
}

@Composable
private fun PendingTripsList(pendingTrips: List<PendingTrip>) {
    Card(Modifier.fillMaxWidth()) {
        Column(Modifier.padding(12.dp)) {
            pendingTrips.forEach { trip ->
                Text(text = trip.toString())
            }
        }
    }
}

@Composable
private fun DashboardDialog(
    dialogState: DashboardDialogState,
    onConfirm: () -> Unit,
    onDismiss: () -> Unit,
) {
    AlertDialog(
        onDismissRequest = onDismiss,
        confirmButton = {
            TextButton(onClick = onConfirm) { Text(dialogState.confirmText) }
        },
        dismissButton = {
            TextButton(onClick = onDismiss) { Text(dialogState.dismissText) }
        },
        title = { Text(dialogState.title) },
        text = { Text(dialogState.message) }
    )
}
```

# 2) Compose ViewModel: `UiState + UiEvent + UiEffect`

This version keeps the original data-loading behavior but exposes Compose-friendly state.

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import android.os.Handler
import android.os.Looper
import androidx.lifecycle.viewModelScope
import com.interfacom.sdk.taximeter.bravocomm.BravoCentral
import com.interfacom.sdk.taximeter.bravocomm.W2CLocation
import com.interfacom.sdk.taximeter.bravocomm.rest.pending_trips.response.PendingTrip
import com.interfacom.sdk.taximeter.licensing.models.zoning.LatLong
import com.interfacom.sdk.taximeter.licensing.models.zoning.Zone
import ifac.td.taxi.domain.usecase.PendingTripsUseCase
import ifac.td.taxi.framework.sdk.usecase.LicensingUseCase
import ifac.td.taxi.framework.sdk.usecase.ZoningUseCase
import ifac.td.taxi.repository.room.entities.BravoConfigurationVariableEntity
import ifac.td.taxi.ui.model.ZoneModel
import ifac.td.taxi.ui.util.ZoneUtils
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch

class DashboardViewModel(
    private val pendingTripsUseCase: PendingTripsUseCase,
    private val zoningUseCase: ZoningUseCase,
    private val licensingUseCase: LicensingUseCase,
    context: Application,
) : BaseViewModel(context) {

    private val handler = Handler(Looper.getMainLooper())
    private val pendingTripsHandler = Handler(Looper.getMainLooper())

    private var zonesRunnable: Runnable? = null
    private var pendingTripsRunnable: Runnable? = null

    var bravoConfiguration: BravoConfigurationVariableEntity? = null
    var isCustomersAtStand: Boolean = false

    private val _uiState = MutableStateFlow(DashboardUiState())
    val uiState = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<DashboardUiEffect>(extraBufferCapacity = 1)
    val effects = _effects.asSharedFlow()

    init {
        viewModelScope.launch(Dispatchers.IO) {
            licensingUseCase.getLicensingParameters()?.isCustomersAtStand?.let {
                isCustomersAtStand = it
            }
        }
    }

    fun onEvent(event: DashboardUiEvent) {
        when (event) {
            DashboardUiEvent.ScreenStarted -> {
                initViewModelData()
                loadTrips()
            }

            DashboardUiEvent.ScreenStopped -> {
                removeHandlerCallbacks()
            }

            is DashboardUiEvent.HeaderClicked -> handleHeaderClick(event.type)

            DashboardUiEvent.DialogConfirmed -> {
                _uiState.update { it.copy(dialogState = null) }
            }

            DashboardUiEvent.DialogDismissed -> {
                _uiState.update { it.copy(dialogState = null) }
            }

            DashboardUiEvent.NavigateBack -> emitEffect(DashboardUiEffect.NavigateBack)
            DashboardUiEvent.NavigateToHome -> emitEffect(DashboardUiEffect.NavigateToHome)
            DashboardUiEvent.NavigateToOnTrip -> emitEffect(DashboardUiEffect.NavigateToOnTrip)
        }
    }

    private fun emitEffect(effect: DashboardUiEffect) {
        _effects.tryEmit(effect)
    }

    private fun loadTrips() {
        if (BravoCentral.isPendingTripsAllowed(false)) {
            refreshPendingTripsManual()
        }
        startChronometer()
    }

    private fun refreshPendingTripsManual() {
        viewModelScope.launch {
            pendingTripsUseCase.getPendingTripsZone(
                flow = sharedPendingTripsFlow
            )
        }
    }

    private val sharedPendingTripsFlow = MutableStateFlow<ArrayList<PendingTrip>?>(null)

    private fun startChronometer() {
        viewModelScope.launch(Dispatchers.IO) {
            val refreshRate = 2000L
            val initRefreshRate = 500L

            pendingTripsRunnable = object : Runnable {
                override fun run() {
                    viewModelScope.launch {
                        if (BravoCentral.isPendingTripsAllowed(true)) {
                            pendingTripsUseCase.getPendingTripsZone(flow = sharedPendingTripsFlow)
                        }
                    }
                    pendingTripsHandler.postDelayed(this, refreshRate)
                }
            }

            pendingTripsRunnable?.let {
                pendingTripsHandler.postDelayed(it, initRefreshRate)
            }

            viewModelScope.launch {
                sharedPendingTripsFlow.collect { trips ->
                    _uiState.update { it.copy(pendingTrips = trips ?: arrayListOf()) }
                }
            }
        }
    }

    private val zoneCallback: (List<Zone>, String) -> Unit = { _, availableColumns ->
        viewModelScope.launch(Dispatchers.IO) {
            val zonesList = mutableListOf<ZoneModel>()
            W2CLocation.getZoning().macrozones.forEach { mz ->
                zonesList.addAll(mz.zones.map { zone ->
                    ZoneModel(zone, false)
                })
            }

            _uiState.update { it.copy(allZones = zonesList) }

            if (availableColumns.isNotBlank()) {
                _uiState.update { current ->
                    current.copy(
                        buttonsState = current.buttonsState.withDataTypes(availableColumns)
                    )
                }
            }

            filterData(zonesList)
        }
    }

    fun initViewModelData() {
        viewModelScope.launch {
            val refreshRate = licensingUseCase.getLicensingParameters()?.segRefrescoConsUbicacion
            refreshRate?.let { seconds ->
                val millis = (seconds * 1000).toLong()
                var lastZoningUpdate = zoningUseCase.getLastZoningUpdate()

                val shouldLoadData = (System.currentTimeMillis() - lastZoningUpdate) < millis
                if (shouldLoadData) {
                    zoneCallback.invoke(
                        W2CLocation.getZoning().macrozones[0].zones,
                        W2CLocation.getZoning().availableColumns
                    )
                }

                zonesRunnable = object : Runnable {
                    override fun run() {
                        viewModelScope.launch {
                            val newCall = (System.currentTimeMillis() - lastZoningUpdate) > millis
                            if (newCall) {
                                zoningUseCase.getZonesInformation(
                                    W2CLocation.getZoning().macrozones[0],
                                    zoneCallback,
                                )
                                lastZoningUpdate = System.currentTimeMillis()
                                zoningUseCase.setLastZoningUpdate()
                            }
                        }
                        handler.postDelayed(this, 1000)
                    }
                }

                zonesRunnable?.let { handler.post(it) }
            }
        }
    }

    private suspend fun filterData(zonesList: List<ZoneModel>) {
        val dataTypes = uiState.value.buttonsState.dataTypes

        val zonesWithMoreTrips = zonesList.filter {
            ZoneUtils.checkDiff(it.zone, dataTypes) > 0
        }

        val (farZones, nearbyZones) = zonesWithMoreTrips.partition {
            getDistanceFromLatLonInMetersLocale(
                W2CLocation.getLastLatLong(),
                it.zone.centralCoord
            ) > 5000
        }

        val nearbyZonesSorted = nearbyZones.sortedWith(
            compareByDescending<ZoneModel> { ZoneUtils.checkDiff(it.zone, dataTypes) }
                .thenBy { getDistanceFromLatLonInMetersLocale(W2CLocation.getLastLatLong(), it.zone.centralCoord) }
        )

        val farZonesSorted = farZones.sortedWith(
            compareByDescending<ZoneModel> { ZoneUtils.checkDiff(it.zone, dataTypes) }
                .thenBy { getDistanceFromLatLonInMetersLocale(W2CLocation.getLastLatLong(), it.zone.centralCoord) }
        )

        _uiState.update {
            it.copy(
                nearbyZones = nearbyZonesSorted,
                farZones = farZonesSorted,
                showZonesSection = true,
                actualZones = buildActualZones()
            )
        }
    }

    fun refreshActualZoneData() {
        viewModelScope.launch {
            _uiState.update { it.copy(actualZones = buildActualZones()) }
        }
    }

    private fun buildActualZones(): List<ZoneModel> {
        val lastZoneSent = W2CLocation.getZoning().getZoneById(
            W2CLocation.getLastIdMacrozoneSent(),
            W2CLocation.getLastIdZoneSent()
        )
        val lastZoneLocated = W2CLocation.getZoneFromLatLong(W2CLocation.getLastLatLong())

        val actualZoneList = mutableListOf<ZoneModel>()
        if (lastZoneSent != null) actualZoneList.add(ZoneModel(lastZoneSent, false))
        if (lastZoneLocated != null) actualZoneList.add(ZoneModel(lastZoneLocated, false))
        return actualZoneList
    }

    private fun getDistanceFromLatLonInMetersLocale(
        lastLatLong: LatLong?,
        centralCoord: LatLong?
    ): Int {
        if (lastLatLong == null || lastLatLong.Lat.isNaN() || centralCoord == null || centralCoord.Lat.isNaN()) {
            return Int.MAX_VALUE
        }
        return W2CLocation.getDistanceFromLatLonInMeters(lastLatLong, centralCoord)
    }

    private fun handleHeaderClick(type: DashboardHeaderType) {
        _uiState.update { current ->
            val newButtons = current.buttonsState.toggle(type)
            current.copy(buttonsState = newButtons)
        }
    }

    fun removeHandlerCallbacks() {
        pendingTripsRunnable?.let { pendingTripsHandler.removeCallbacks(it) }
        zonesRunnable?.let { handler.removeCallbacks(it) }
    }
}
```

# 4) Full `DashboardButtonsState` matching the XML behavior

This is the state holder that replaces the fragment button logic.

It preserves the exact visibility rules from `updateHeader(dataTypes)`:

- `STAND_VEHICLES` → show `tvOnStop`, `tvOnStop2`, `tvOnStop3`
- `ZONE_VEHICLES` → show `tvOnZone`, `tvOnZone2`, `tvOnZone3`
- `HIRED_VEHICLES || BOOKED_TRIPS` → show `tvHired*`
  - if `HIRED_VEHICLES` is missing, use `abrevUbServiciosWithoutDots`
- `PREBOOKED_TRIPS || TOTAL_TRIPS || (HIRED_VEHICLES && BOOKED_TRIPS)` → show `tvTrips*`
  - if `PREBOOKED_TRIPS`, use `abbrevPreBookedTrips`
  - else if `TOTAL_TRIPS`, use `abbrevTotalTrips`

It also includes button color state and sort-arrow visibility.

```kotlin
package ifac.td.taxi.viewmodel

import androidx.compose.ui.graphics.Color
import com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum

data class DashboardButtonsState(
    val onStop: DashboardButtonState = DashboardButtonState(),
    val onZone: DashboardButtonState = DashboardButtonState(),
    val hired: DashboardButtonState = DashboardButtonState(),
    val trips: DashboardButtonState = DashboardButtonState(),
    val dataTypes: String = "",
    val activeSort: DashboardHeaderType? = null,
) {
    fun withDataTypes(dataTypes: String): DashboardButtonsState {
        val columns = parseColumns(dataTypes)

        val onStopVisible = columns.contains(AvailableColumnsEnum.STAND_VEHICLES)
        val onZoneVisible = columns.contains(AvailableColumnsEnum.ZONE_VEHICLES)
        val hiredVisible = columns.contains(AvailableColumnsEnum.HIRED_VEHICLES) ||
            columns.contains(AvailableColumnsEnum.BOOKED_TRIPS)
        val tripsVisible = columns.contains(AvailableColumnsEnum.PREBOOKED_TRIPS) ||
            columns.contains(AvailableColumnsEnum.TOTAL_TRIPS) ||
            (columns.contains(AvailableColumnsEnum.HIRED_VEHICLES) && columns.contains(AvailableColumnsEnum.BOOKED_TRIPS))

        return copy(
            dataTypes = dataTypes,
            onStop = onStop.copy(
                visible = onStopVisible,
                text = "On stop",
                backgroundColor = Color(0xFF1976D2),
                contentColor = Color.White,
                sortArrowVisible = activeSort == DashboardHeaderType.ON_STOP
            ),
            onZone = onZone.copy(
                visible = onZoneVisible,
                text = "On zone",
                backgroundColor = Color(0xFF1976D2),
                contentColor = Color.White,
                sortArrowVisible = activeSort == DashboardHeaderType.ON_ZONE
            ),
            hired = hired.copy(
                visible = hiredVisible,
                text = if (columns.contains(AvailableColumnsEnum.HIRED_VEHICLES)) {
                    "Hired"
                } else {
                    "Ub servicios"
                },
                backgroundColor = Color(0xFF1976D2),
                contentColor = Color.White,
                sortArrowVisible = activeSort == DashboardHeaderType.HIRED
            ),
            trips = trips.copy(
                visible = tripsVisible,
                text = when {
                    columns.contains(AvailableColumnsEnum.PREBOOKED_TRIPS) -> "Pre-booked"
                    columns.contains(AvailableColumnsEnum.TOTAL_TRIPS) -> "Total trips"
                    else -> "Trips"
                },
                backgroundColor = Color(0xFF1976D2),
                contentColor = Color.White,
                sortArrowVisible = activeSort == DashboardHeaderType.TRIPS
            )
        )
    }

    fun toggle(type: DashboardHeaderType): DashboardButtonsState {
        val newActive = if (activeSort == type) null else type
        return copy(
            activeSort = newActive,
            onStop = onStop.copy(sortArrowVisible = newActive == DashboardHeaderType.ON_STOP),
            onZone = onZone.copy(sortArrowVisible = newActive == DashboardHeaderType.ON_ZONE),
            hired = hired.copy(sortArrowVisible = newActive == DashboardHeaderType.HIRED),
            trips = trips.copy(sortArrowVisible = newActive == DashboardHeaderType.TRIPS),
        )
    }

    private fun parseColumns(dataTypes: String): Set<AvailableColumnsEnum> {
        return dataTypes
            .split(",")
            .mapNotNull { raw ->
                runCatching { AvailableColumnsEnum.valueOf(raw.trim()) }.getOrNull()
            }
            .toSet()
    }
}

data class DashboardButtonState(
    val visible: Boolean = false,
    val text: String = "",
    val backgroundColor: Color = Color.Transparent,
    val contentColor: Color = Color.Unspecified,
    val sortArrowVisible: Boolean = false,
)
```

```kotlin
package ifac.td.taxi.viewmodel

import androidx.compose.ui.graphics.Color
import com.interfacom.sdk.taximeter.bravocomm.rest.infozones.AvailableColumnsEnum

data class DashboardButtonsState(
    val onStop: HeaderButtonState = HeaderButtonState(),
    val onZone: HeaderButtonState = HeaderButtonState(),
    val hired: HeaderButtonState = HeaderButtonState(),
```

# 3) Compose bridge for the shared flows

This is the Compose equivalent of the fragment’s `setupObservers()` block.

It collects the shared flows and dispatches reducer actions exactly like the fragment did.

```kotlin
package ifac.td.taxi.ui.screen

import androidx.compose.runtime.*
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import ifac.td.taxi.viewmodel.*

@Composable
fun DashboardSharedBridge(
    mainViewModel: DashboardMainActivityViewModel,
    dashboardViewModel: DashboardViewModel,
) {
    val sharedState by mainViewModel.uiState.collectAsStateWithLifecycle()
    val dashboardState by dashboardViewModel.uiState.collectAsStateWithLifecycle()

    // 1) pendingTripsListFlow -> updatePendingTripsData()
    LaunchedEffect(Unit) {
        mainViewModel.pendingTripsListFlow.collect { pendingTrips ->
            dashboardViewModel.onPendingTripsChanged(pendingTrips ?: arrayListOf())
        }
    }

    // 2) zoningChangedFlow.eventCollector -> refreshActualZoneData()
    LaunchedEffect(Unit) {
        mainViewModel.zoningChangedFlow.collect { zone ->
            if (zone != null) {
                dashboardViewModel.refreshActualZoneData()
            }
        }
    }

    // 3) zoningListDataTypes.collect -> updateHeader(dataTypes)
    LaunchedEffect(Unit) {
        dashboardViewModel.zoningListDataTypes.collect { dataTypes ->
            if (!dataTypes.isNullOrBlank()) {
                dashboardViewModel.onButtonsAction(
                    DashboardButtonsAction.DataTypesChanged(dataTypes)
                )
            }
        }
    }

    // 4) hiredZone.collect -> btnLocateOnHired branch
    LaunchedEffect(Unit) {
        mainViewModel.hiredZone.collect { hiredZone ->
            dashboardViewModel.onButtonsAction(
                DashboardButtonsAction.HiredZoneChanged(hiredZone != null)
            )
        }
    }

    // 5) bravoStateFlow.collect -> isInSoonInZone branch
    LaunchedEffect(Unit) {
        mainViewModel.bravoStateFlow.collect { bravoState ->
            dashboardViewModel.onButtonsAction(
                DashboardButtonsAction.SoonInZoneChanged(bravoState.isInSoonInZone)
            )
        }
    }
}
```

```kotlin
data class DashboardButtonsState(
    val onStop: HeaderButtonState = HeaderButtonState(),
    val onZone: HeaderButtonState = HeaderButtonState(),
    val hired: HeaderButtonState = HeaderButtonState(),
    val trips: HeaderButtonState = HeaderButtonState(),
    val locateOnHired: LocateOnHiredButtonState = LocateOnHiredButtonState(),
    val soonInZoneVisible: Boolean = false,
    val dataTypes: String = "",
    val activeSort: DashboardHeaderType? = null,
) {
    fun reduce(action: DashboardButtonsAction): DashboardButtonsState = when (action) {
        is DashboardButtonsAction.DataTypesChanged -> reduceDataTypes(action.dataTypes)
        is DashboardButtonsAction.HiredZoneChanged -> reduceHiredZone(action.hasHiredZone)
        is DashboardButtonsAction.SoonInZoneChanged -> reduceSoonInZone(action.isInSoonInZone)
        is DashboardButtonsAction.HeaderClicked -> reduceHeaderClicked(action.type)
    }

    private fun reduceDataTypes(dataTypes: String): DashboardButtonsState {
        val columns = parseColumns(dataTypes)

        val showOnStop = columns.contains(AvailableColumnsEnum.STAND_VEHICLES)
        val showOnZone = columns.contains(AvailableColumnsEnum.ZONE_VEHICLES)
        val showHired = columns.contains(AvailableColumnsEnum.HIRED_VEHICLES) ||
            columns.contains(AvailableColumnsEnum.BOOKED_TRIPS)
        val showTrips = columns.contains(AvailableColumnsEnum.PREBOOKED_TRIPS) ||
            columns.contains(AvailableColumnsEnum.TOTAL_TRIPS) ||
            (columns.contains(AvailableColumnsEnum.HIRED_VEHICLES) && columns.contains(AvailableColumnsEnum.BOOKED_TRIPS))

        return copy(
            dataTypes = dataTypes,
            onStop = onStop.copy(
                visible = showOnStop,
                text = "On stop",
                backgroundColor = Color(0xFF1976D2),
                contentColor = Color.White,
                sortArrowVisible = activeSort == DashboardHeaderType.ON_STOP
            ),
            onZone = onZone.copy(
                visible = showOnZone,
                text = "On zone",
                backgroundColor = Color(0xFF1976D2),
                contentColor = Color.White,
                sortArrowVisible = activeSort == DashboardHeaderType.ON_ZONE
            ),
            hired = hired.copy(
                visible = showHired,
                text = if (columns.contains(AvailableColumnsEnum.HIRED_VEHICLES)) "Hired" else "Ub servicios",
                backgroundColor = Color(0xFF1976D2),
                contentColor = Color.White,
                sortArrowVisible = activeSort == DashboardHeaderType.HIRED
            ),
            trips = trips.copy(
                visible = showTrips,
                text = when {
                    columns.contains(AvailableColumnsEnum.PREBOOKED_TRIPS) -> "Pre-booked"
                    columns.contains(AvailableColumnsEnum.TOTAL_TRIPS) -> "Total trips"
                    else -> "Trips"
                },
                backgroundColor = Color(0xFF1976D2),
                contentColor = Color.White,
                sortArrowVisible = activeSort == DashboardHeaderType.TRIPS
            )
        )
    }

    private fun reduceHiredZone(hasHiredZone: Boolean): DashboardButtonsState {
        return if (hasHiredZone) {
            copy(
                locateOnHired = locateOnHired.copy(
                    visible = true,
                    text = "Delocate",
                    icon = "ubdesact",
                    backgroundColor = Color(0xFFD32F2F),
                    contentColor = Color.White,
                    enabled = true
                )
            )
        } else {
            copy(
                locateOnHired = locateOnHired.copy(
                    visible = true,
                    text = "Soon to clear",
                    icon = "ubactivar",
                    backgroundColor = Color(0xFF1976D2),
                    contentColor = Color.White,
                    enabled = true
                )
            )
        }
    }

    private fun reduceSoonInZone(isInSoonInZone: Boolean): DashboardButtonsState {
        return copy(soonInZoneVisible = isInSoonInZone)
    }

    private fun reduceHeaderClicked(type: DashboardHeaderType): DashboardButtonsState {
        val newActive = if (activeSort == type) null else type
        return copy(
            activeSort = newActive,
            onStop = onStop.copy(sortArrowVisible = newActive == DashboardHeaderType.ON_STOP),
            onZone = onZone.copy(sortArrowVisible = newActive == DashboardHeaderType.ON_ZONE),
            hired = hired.copy(sortArrowVisible = newActive == DashboardHeaderType.HIRED),
            trips = trips.copy(sortArrowVisible = newActive == DashboardHeaderType.TRIPS),
        )
    }

    private fun parseColumns(dataTypes: String): Set<AvailableColumnsEnum> {
        return dataTypes.split(",")
            .mapNotNull { raw -> runCatching { AvailableColumnsEnum.valueOf(raw.trim()) }.getOrNull() }
            .toSet()
    }
}

class DashboardViewModel(
    application: Application,
    private val pendingTripsUseCase: PendingTripsUseCase,
    private val zoningUseCase: ZoningUseCase,
    private val licensingUseCase: LicensingUseCase,
) : AndroidViewModel(application) {

    private val handler = Handler(Looper.getMainLooper())
    private val pendingTripsHandler = Handler(Looper.getMainLooper())

    private var zonesRunnable: Runnable? = null
    private var pendingTripsRunnable: Runnable? = null

    private val _uiState = MutableStateFlow(DashboardUiState())
    val uiState = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<DashboardUiEffect>(extraBufferCapacity = 1)
    val effects = _effects.asSharedFlow()

    private val _zoningListDataTypes = MutableStateFlow<String?>(null)
    val zoningListDataTypes = _zoningListDataTypes.asStateFlow()

    private val _pendingTripsFlow = MutableStateFlow<ArrayList<PendingTrip>?>(null)
    val pendingTripsFlow = _pendingTripsFlow.asStateFlow()

    fun onEvent(event: DashboardUiEvent) {
        when (event) {
            DashboardUiEvent.ScreenStarted -> {
                initViewModelData()
                loadTrips()
            }

            DashboardUiEvent.ScreenStopped -> removeHandlerCallbacks()

            is DashboardUiEvent.HeaderClicked -> {
                _uiState.update {
                    it.copy(buttonsState = it.buttonsState.reduce(DashboardButtonsAction.HeaderClicked(event.type)))
                }
            }

            DashboardUiEvent.DialogConfirmed,
            DashboardUiEvent.DialogDismissed -> {
                _uiState.update { it.copy(dialogState = null) }
            }

            DashboardUiEvent.ShowDelocateDialog -> {
                _uiState.update {
                    it.copy(
                        dialogState = DashboardDialogState(
                            title = "Delocate",
                            message = "Do you want to delocate the hired zone?"
                        )
                    )
                }
            }

            DashboardUiEvent.ShowSoonToClearDialog -> {
                _uiState.update {
                    it.copy(
                        dialogState = DashboardDialogState(
                            title = "Soon to clear",
                            message = "Do you want to start soon-to-clear action?"
                        )
                    )
                }
            }
        }
    }

    fun onPendingTripsChanged(pendingTrips: ArrayList<PendingTrip>) {
        _uiState.update { it.copy(pendingTrips = pendingTrips) }
    }

    fun onDataTypesChanged(dataTypes: String) {
        _zoningListDataTypes.value = dataTypes
        _uiState.update {
            it.copy(buttonsState = it.buttonsState.reduce(DashboardButtonsAction.DataTypesChanged(dataTypes)))
        }
    }

    fun onHiredZoneChanged(hasHiredZone: Boolean) {
        _uiState.update {
            it.copy(buttonsState = it.buttonsState.reduce(DashboardButtonsAction.HiredZoneChanged(hasHiredZone)))
        }
    }

    fun onSoonInZoneChanged(isInSoonInZone: Boolean) {
        _uiState.update {
            it.copy(buttonsState = it.buttonsState.reduce(DashboardButtonsAction.SoonInZoneChanged(isInSoonInZone)))
        }
    }

    fun refreshActualZoneData() {
        viewModelScope.launch {
            _uiState.update { it.copy(actualZones = buildActualZones()) }
        }
    }

    private fun loadTrips() {
        if (BravoCentral.isPendingTripsAllowed(false)) {
            refreshPendingTripsManual()
        }
        startChronometer()
    }

    private fun refreshPendingTripsManual() {
        viewModelScope.launch {
            pendingTripsUseCase.getPendingTripsZone(flow = _pendingTripsFlow)
        }
    }

    private fun startChronometer() {
        viewModelScope.launch(Dispatchers.IO) {
            val refreshRate = 2000L
            val initRefreshRate = 500L

            pendingTripsRunnable = object : Runnable {
                override fun run() {
                    viewModelScope.launch {
                        if (BravoCentral.isPendingTripsAllowed(true)) {
                            pendingTripsUseCase.getPendingTripsZone(flow = _pendingTripsFlow)
                        }
                    }
                    pendingTripsHandler.postDelayed(this, refreshRate)
                }
            }

            pendingTripsRunnable?.let { pendingTripsHandler.postDelayed(it, initRefreshRate) }
        }
    }

    private fun initViewModelData() {
        viewModelScope.launch {
            val refreshRate = licensingUseCase.getLicensingParameters()?.segRefrescoConsUbicacion
            refreshRate?.let { seconds ->
                val millis = (seconds * 1000).toLong()
                var lastZoningUpdate = zoningUseCase.getLastZoningUpdate()

                val shouldLoadData = (System.currentTimeMillis() - lastZoningUpdate) < millis
                if (shouldLoadData) {
                    zoneCallback.invoke(
                        W2CLocation.getZoning().macrozones[0].zones,
                        W2CLocation.getZoning().availableColumns
                    )
                }

                zonesRunnable = object : Runnable {
                    override fun run() {
                        viewModelScope.launch {
                            val newCall = (System.currentTimeMillis() - lastZoningUpdate) > millis
                            if (newCall) {
                                zoningUseCase.getZonesInformation(
                                    W2CLocation.getZoning().macrozones[0],
                                    zoneCallback,
                                )
                                lastZoningUpdate = System.currentTimeMillis()
                                zoningUseCase.setLastZoningUpdate()
                            }
                        }
                        handler.postDelayed(this, 1000)
                    }
                }

                zonesRunnable?.let { handler.post(it) }
            }
        }
    }

    private val zoneCallback: (List<Zone>, String) -> Unit = { _, availableColumns ->
        viewModelScope.launch(Dispatchers.IO) {
            val zonesList = mutableListOf<ZoneModel>()
            W2CLocation.getZoning().macrozones.forEach { mz ->
                zonesList.addAll(mz.zones.map { zone -> ZoneModel(zone, false) })
            }

            _uiState.update { it.copy(showZonesSection = true) }
            if (availableColumns.isNotBlank()) {
                onDataTypesChanged(availableColumns)
            }
            filterData(zonesList)
        }
    }

    private suspend fun filterData(zonesList: List<ZoneModel>) {
        val dataTypes = _zoningListDataTypes.value.orEmpty()

        val zonesWithMoreTrips = zonesList.filter {
            ZoneUtils.checkDiff(it.zone, dataTypes) > 0
        }

        val (farZones, nearbyZones) = zonesWithMoreTrips.partition {
            getDistanceFromLatLonInMetersLocale(
                W2CLocation.getLastLatLong(),
                it.zone.centralCoord
            ) > 5000
        }

        val nearbyZonesSorted = nearbyZones.sortedWith(
            compareByDescending<ZoneModel> { ZoneUtils.checkDiff(it.zone, dataTypes) }
                .thenBy { getDistanceFromLatLonInMetersLocale(W2CLocation.getLastLatLong(), it.zone.centralCoord) }
        )

        val farZonesSorted = farZones.sortedWith(
            compareByDescending<ZoneModel> { ZoneUtils.checkDiff(it.zone, dataTypes) }
                .thenBy { getDistanceFromLatLonInMetersLocale(W2CLocation.getLastLatLong(), it.zone.centralCoord) }
        )

        _uiState.update {
            it.copy(
                nearbyZones = nearbyZonesSorted,
                farZones = farZonesSorted,
                actualZones = buildActualZones()
            )
        }
    }

    private fun buildActualZones(): List<ZoneModel> {
        val lastZoneSent = W2CLocation.getZoning().getZoneById(
            W2CLocation.getLastIdMacrozoneSent(),
            W2CLocation.getLastIdZoneSent()
        )
        val lastZoneLocated = W2CLocation.getZoneFromLatLong(W2CLocation.getLastLatLong())

        val actualZoneList = mutableListOf<ZoneModel>()
        if (lastZoneSent != null) actualZoneList.add(ZoneModel(lastZoneSent, false))
        if (lastZoneLocated != null) actualZoneList.add(ZoneModel(lastZoneLocated, false))
        return actualZoneList
    }

    private fun getDistanceFromLatLonInMetersLocale(
        lastLatLong: LatLong?,
        centralCoord: LatLong?
    ): Int {
        if (lastLatLong == null || lastLatLong.Lat.isNaN() || centralCoord == null || centralCoord.Lat.isNaN()) {
            return Int.MAX_VALUE
        }
        return W2CLocation.getDistanceFromLatLonInMeters(lastLatLong, centralCoord)
    }

    fun removeHandlerCallbacks() {
        pendingTripsRunnable?.let { pendingTripsHandler.removeCallbacks(it) }
        zonesRunnable?.let { handler.removeCallbacks(it) }
    }
}
```

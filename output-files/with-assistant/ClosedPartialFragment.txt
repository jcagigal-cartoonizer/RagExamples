## 2) Exact `ClosedPartialButtonsState` reducer

This mirrors the fragment’s `collect {}` branch behavior one-by-one:

- `partial == null` → ticket text becomes `no_closed_partials`
- `partial != null` → ticket text becomes `bufLastTicketCierre` and scrolls down
- `updateTotalizersButtonState()` logic is preserved exactly:
  - show totalizers only when `currentStatus != STATE_DISCONNECTED && isConnected`
  - style is `ENABLE` only when `isConnected && hasTotalizers`
  - otherwise `DISABLE`

```kotlin
package ifac.td.taxi.ui.screen.closedpartial

import com.interfacom.sdk.taximeter.bravocomm.ifConstants
import ifac.td.taxi.ui.custom.button.CustomButton

fun ClosedPartialButtonsState.Companion.reduce(
    isTaximeterConnected: Boolean,
    hasTotalizers: Boolean,
    currentStatus: Int?,
): ClosedPartialButtonsState {
    val shouldShow = currentStatus != ifConstants.STATE_DISCONNECTED && isTaximeterConnected

    val totalizersState = if (shouldShow) {
        ClosedPartialButtonVisualState(
            visible = true,
            style = if (isTaximeterConnected && hasTotalizers) {
                CustomButton.StyleButton.ENABLE
            } else {
                CustomButton.StyleButton.DISABLE
            }
        )
    } else {
        ClosedPartialButtonVisualState(
            visible = false,
            style = CustomButton.StyleButton.DISABLE
        )
    }

    return ClosedPartialButtonsState(
        cancel = ClosedPartialButtonVisualState(
            visible = true,
            style = CustomButton.StyleButton.ENABLE
        ),
        print = ClosedPartialButtonVisualState(
            visible = true,
            style = CustomButton.StyleButton.ENABLE
        ),
        totalizers = totalizersState
    )
}

private fun ClosedPartialButtonsState.Companion() = Unit
```

## 3) Compose-friendly `ClosedPartialViewModel`

This replaces the fragment VM with `UiState + UiEvent + SharedFlow<ClosedPartialUiEffect>`.

```kotlin
package ifac.td.taxi.ui.screen.closedpartial

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import com.interfacom.sdk.taximeter.bravocomm.ifConstants
import ifac.td.taxi.R
import ifac.td.taxi.domain.model.PartialModel
import ifac.td.taxi.domain.usecase.PartialUseCase
import ifac.td.taxi.domain.usecase.PrinterUseCase
import ifac.td.taxi.framework.sdk.usecase.TaximeterConnectUseCase
import ifac.td.taxi.viewmodel.model.UtilsModel.Companion.addTicketLines
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

class ClosedPartialViewModel(
    application: Application,
    private val printerUseCase: PrinterUseCase,
    private val partialUseCase: PartialUseCase,
    private val taximeterUseCase: TaximeterConnectUseCase,
) : AndroidViewModel(application) {

    private val _uiState = MutableStateFlow(ClosedPartialUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffects = MutableSharedFlow<ClosedPartialUiEffect>()
    val uiEffects: SharedFlow<ClosedPartialUiEffect> = _uiEffects

    private val _partial = MutableStateFlow<PartialModel?>(null)
    val partial = _partial.asStateFlow()

    private val _isTaximeterConnected = MutableStateFlow(false)
    val isTaximeterConnected = _isTaximeterConnected.asStateFlow()

    fun onEvent(event: ClosedPartialUiEvent) {
        when (event) {
            ClosedPartialUiEvent.OnScreenStarted -> {
                checkTaximeterStatus()
                getClosedPartial()
            }

            ClosedPartialUiEvent.OnCancelClicked -> manageBack(justClosed = false)
            ClosedPartialUiEvent.OnPrintClicked -> printPartial()
            ClosedPartialUiEvent.OnTotalizersClicked -> {
                viewModelScope.launch {
                    _uiEffects.emit(ClosedPartialUiEffect.NavigateToTotalizers)
                }
            }

            ClosedPartialUiEvent.OnDialogDismissed -> {
                _uiState.update { it.copy(dialogState = null) }
            }
        }
    }

    fun setSharedState(
        hasTotalizers: Boolean,
        currentStatus: Int?,
    ) {
        _uiState.update {
            it.copy(
                buttonsState = ClosedPartialButtonsStateReducer.reduce(
                    isTaximeterConnected = _isTaximeterConnected.value,
                    hasTotalizers = hasTotalizers,
                    currentStatus = currentStatus
                )
            )
        }
    }

    fun getClosedPartial() {
        viewModelScope.launch {
            val result = partialUseCase.getClosedPartial()
            _partial.emit(result)

            val ticketContent = result?.bufLastTicketCierre?.takeIf { it.isNotEmpty() }
                ?: getApplication<Application>().getString(R.string.no_closed_partials)

            _uiState.update { state ->
                state.copy(partialText = ticketContent)
            }
        }
    }

    fun manageBack(justClosed: Boolean) {
        viewModelScope.launch {
            if (justClosed) {
                _uiEffects.emit(ClosedPartialUiEffect.NavigateBackTwice)
            } else {
                _uiEffects.emit(ClosedPartialUiEffect.NavigateBackOnce)
            }
        }
    }

    fun printPartial() {
        viewModelScope.launch {
            val current = _partial.value
            if (current != null) {
                printerUseCase.printMessage(
                    header = null,
                    message = current.bufLastTicketCierre.addTicketLines(),
                    paperFeed = false
                )
            } else {
                _uiEffects.emit(ClosedPartialUiEffect.ShowToast(R.string.no_closed_partials))
            }
        }
    }

    fun checkTaximeterStatus() {
        viewModelScope.launch {
            _isTaximeterConnected.emit(taximeterUseCase.isTaximeterConnected())
        }
    }
}
```

## 4) Full `ClosedPartialMainActivityViewModel` Compose bridge

This is the shared-flow bridge that replaces the fragment’s direct access to `sharedViewModel.taximeterTotalizersFlow` and `sharedViewModel.shiftStatusFlow`.

```kotlin
package ifac.td.taxi.ui.screen.closedpartial

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope

import kotlinx.coroutines.flow.asStateFlow

    val hasTotalizers: Boolean = false,
    val currentStatus: Int? = null,
)

class ClosedPartialMainActivityViewModel : ViewModel() {

    private val _bridgeState = MutableStateFlow(ClosedPartialSharedBridgeState())
    val bridgeState = _bridgeState.asStateFlow()

    fun updateTaximeterTotalizers(hasTotalizers: Boolean) {
        _bridgeState.value = _bridgeState.value.copy(hasTotalizers = hasTotalizers)
    }

    fun updateShiftStatus(currentStatus: Int?) {
        _bridgeState.value = _bridgeState.value.copy(currentStatus = currentStatus)
    }

    fun bindToClosedPartial(
        closedPartialViewModel: ClosedPartialViewModel,
    ) {
        viewModelScope.launch {
            closedPartialViewModel.setSharedState(
                hasTotalizers = _bridgeState.value.hasTotalizers,
                currentStatus = _bridgeState.value.currentStatus
            )
        }
    }
}
```

```kotlin
@Composable
fun ClosedPartialScreen(
    viewModel: ClosedPartialViewModel,
    sharedViewModel: ClosedPartialMainActivityViewModel,
    onNavigateBack: () -> Unit,
    onNavigateToTotalizers: () -> Unit,
    onShowHeader: (Boolean) -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val bridgeState by sharedViewModel.bridgeState.collectAsStateWithLifecycle()

    LaunchedEffect(Unit) {
        onShowHeader(true)
        viewModel.onEvent(ClosedPartialUiEvent.OnScreenStarted)
    }

    LaunchedEffect(bridgeState.hasTotalizers, bridgeState.currentStatus) {
        viewModel.setSharedState(
            hasTotalizers = bridgeState.hasTotalizers,
            currentStatus = bridgeState.currentStatus
        )
    }

    LaunchedEffect(Unit) {
        viewModel.uiEffects.collect { effect ->
            when (effect) {
                ClosedPartialUiEffect.NavigateBackOnce -> onNavigateBack()
                ClosedPartialUiEffect.NavigateBackTwice -> {
                    onNavigateBack()
                    onNavigateBack()
                }
                ClosedPartialUiEffect.NavigateToTotalizers -> onNavigateToTotalizers()
                is ClosedPartialUiEffect.ShowToast -> {
                    // host can show toast if needed
                }
            }
        }
    }

    Column(
        modifier = Modifier
            .fillMaxSize()
            .background(Color.Black)
    ) {
        ClosedPartialTicketArea(
            ticketText = uiState.partialText,
            modifier = Modifier
                .weight(1f)
                .padding(horizontal = 16.dp)
        )

        ClosedPartialButtonsRow(
            state = uiState.buttonsState,
            onCancel = { viewModel.manageBack(justClosed = false) },
            onPrint = { viewModel.printPartial() },
            onTotalizers = { viewModel.onEvent(ClosedPartialUiEvent.OnTotalizersClicked) }
        )
    }
}
```

## 6) If you want exact XML parity for the `CustomButton` itself

Because your `CustomButton` is a custom View, the most faithful Compose migration is to wrap it with `AndroidView` and drive it from `ClosedPartialButtonsState`.

Example:

```kotlin
package ifac.td.taxi.ui.screen.closedpartial

import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.viewinterop.AndroidView
import ifac.td.taxi.ui.custom.button.CustomButton

@Composable
fun ClosedPartialCustomButton(
    modifier: Modifier = Modifier,
    buttonType: Int,
    visible: Boolean,
    style: CustomButton.StyleButton,
    onClick: () -> Unit,
) {
    if (!visible) return

    AndroidView(
        modifier = modifier,
        factory = { context ->
            CustomButton(context).apply {
                setButtonType(buttonType)
                setButtonStyle(style)
                setAction(onClick)
            }
        },
        update = { button ->
            button.setButtonType(buttonType)
            button.setButtonStyle(style)
            button.setAction(onClick)
        }
    )
}
```

Then your row becomes:

```kotlin

    LaunchedEffect(hasTotalizers, currentStatus) {
        viewModel.setSharedState(
            hasTotalizers = hasTotalizers,
            currentStatus = currentStatus
        )
    }

    LaunchedEffect(Unit) {
        viewModel.uiEffects.collect { effect ->
            when (effect) {
                ClosedPartialUiEffect.NavigateBackOnce -> onNavigateBack()
                ClosedPartialUiEffect.NavigateBackTwice -> {
                    onNavigateBack()
                    onNavigateBack()
                }
                ClosedPartialUiEffect.NavigateToTotalizers -> onNavigateToTotalizers()
                is ClosedPartialUiEffect.ShowToast -> {
                    onShowToast(context.getString(effect.messageRes))
                }
            }
        }
    }

    var dialogState by remember { mutableStateOf<ClosedPartialDialogState?>(null) }
    LaunchedEffect(uiState.dialogState) {
        dialogState = uiState.dialogState
    }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(Color.Black)
    ) {
        Column(modifier = Modifier.fillMaxSize()) {
            ClosedPartialTicketArea(
                ticketText = uiState.partialText,
                modifier = Modifier
                    .weight(1f)
                    .fillMaxWidth()
                    .padding(horizontal = 16.dp)
            )

            ClosedPartialButtonsRow(
                state = uiState.buttonsState,
                onCancel = { viewModel.manageBack(justClosed = false) },
                onPrint = { viewModel.printPartial() },
                onTotalizers = { viewModel.onEvent(ClosedPartialUiEvent.OnTotalizersClicked) }
            )
        }

        if (dialogState?.visible == true) {
            AlertDialog(
                onDismissRequest = { viewModel.onEvent(ClosedPartialUiEvent.OnDialogDismissed) },
                title = { Text(dialogState?.title.orEmpty()) },
                text = { Text(dialogState?.message.orEmpty()) },
                confirmButton = {
                    Button(onClick = { viewModel.onEvent(ClosedPartialUiEvent.OnDialogDismissed) }) {
                        Text("OK")
                    }
                }
            )
        }
    }
}
```

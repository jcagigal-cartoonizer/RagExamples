ANSWER:
```kotlin
@Composable
fun LoginUser(
    user: String,
    password: String,
    onUserChange: (String) -> Unit,
    onPasswordChange: (String) -> Unit,
    onEvent: (LoginUserEvent) -> Unit,
    clearFocus: () -> Unit
) {
    Column(modifier = Modifier.fillMaxSize()) {
        Box {
            Column(modifier = Modifier.fillMaxSize()) {
                Text(
                    text = "User",
                    color = Color.White
                )
                Spacer(modifier = Modifier.height(8.dp))
                OutlinedTextField(
                    value = user,
                    onValueChange = onUserChange,
                    label = { Text("User") }
                )
                Spacer(modifier = Modifier.height(8.dp))
                Text(
                    text = "Password",
                    color = Color.White
                )
                Spacer(modifier = Modifier.height(8.dp))
                OutlinedTextField(
                    value = password,
                    onValueChange = onPasswordChange,
                    label = { Text("Password") }
                )
            }
        }

        Row(
            modifier = Modifier
                .fillMaxWidth()
                .height(96.dp)
                .background(Color.Black)
                .padding(4.dp),
            horizontalArrangement = Arrangement.spacedBy(6.dp)
        ) {
            Button(
                onClick = {
                    clearFocus()
                    onEvent(LoginUserEvent.CancelClicked)
                }
            ) {
                Text("CANCEL")
            }

            Spacer(modifier = Modifier.height(8.dp))

            Button(
                onClick = {
                    clearFocus()
                    onEvent(LoginUserEvent.AcceptClicked)
                }
            ) {
                Text("ACCEPT")
            }
        }
    }
}

sealed class LoginUserEvent {
    data object CancelClicked : LoginUserEvent()
    data object AcceptClicked : LoginUserEvent()
}
```

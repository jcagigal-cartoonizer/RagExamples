Below is a Compose-first replacement that keeps the **same navigation behavior**, replaces the fragment button logic with a **single state holder**, and uses a **single `SharedFlow<WelcomeUiEffect>`** for one-off events.

I’m giving you:

1. A **Compose screen** for `WelcomeFragment`
2. A **Compose-friendly `WelcomeViewModel`** using `UiState + UiEvent + UiEffect`
3. A **full `WelcomeButtonsState`** that mirrors the fragment’s button enable/disable/color/visibility logic
4. A **dialog state approach** only for the dialogs that were previously handled by `CustomDialog`
5. Navigation preserved through **Jetpack Navigation Compose / NavController actions** equivalent to the fragment directions

---

## 2) Compose-friendly ViewModel

This version keeps the original business logic, but exposes:
- `StateFlow<WelcomeUiState>`
- `SharedFlow<WelcomeUiEffect>`
- `onEvent(...)`

It also preserves the original navigation destinations by emitting `WelcomeUiEffect.Navigate(...)`.

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import android.widget.Toast
import androidx.lifecycle.viewModelScope
import com.interfacom.sdk.taximeter.bravocomm.W2CLocation
import com.interfacom.sdk.taximeter.bravocomm.ifConstants
import ifac.td.taxi.R
import ifac.td.taxi.domain.model.PartialModel.Companion.hasTrips
import ifac.td.taxi.domain.model.itop.ITopMeterStatus
import ifac.td.taxi.domain.model.itop.ITopMeterStateEnum
import ifac.td.taxi.domain.usecase.BluetoothLocalUseCase
import ifac.td.taxi.domain.usecase.PartialUseCase
import ifac.td.taxi.domain.usecase.PermissionsUseCase
import ifac.td.taxi.domain.usecase.SessionUseCase
import ifac.td.taxi.domain.usecase.ShiftStatusUseCase
import ifac.td.taxi.domain.usecase.ShiftUseCase
import ifac.td.taxi.domain.usecase.StartShiftUseCase
import ifac.td.taxi.domain.usecase.UserPreferencesUseCase
import ifac.td.taxi.domain.usecase.oldv2.MigrationV2UseCase
import ifac.td.taxi.framework.sdk.bravocentral.usecase.BravoCentralConfigurationUseCase
import ifac.td.taxi.framework.sdk.bravocentral.usecase.LoginDriverUseCase
import ifac.td.taxi.framework.sdk.usecase.LicensingUseCase
import ifac.td.taxi.framework.sdk.usecase.TicketUseCase
import ifac.td.taxi.framework.util.Logs
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext

class WelcomeViewModel(
    private val sessionUseCase: SessionUseCase,
    private val shiftUseCase: ShiftUseCase,
    private val licensingUseCase: LicensingUseCase,
    private val bravoCentralConfigurationUseCase: BravoCentralConfigurationUseCase,
    private val ticketUseCase: TicketUseCase,
    private val loginDriverUseCase: LoginDriverUseCase,
    private val startShiftUseCase: StartShiftUseCase,
    private val shiftStatusUseCase: ShiftStatusUseCase,
    private val permissionsUseCase: PermissionsUseCase,
    private val partialUseCase: PartialUseCase,
    private val bluetoothLocalUseCase: BluetoothLocalUseCase,
    private val migrationV2UseCase: MigrationV2UseCase,
    private val userPreferencesUseCase: UserPreferencesUseCase,
    context: Application,
) : BaseViewModel(context) {

    private val TAG = "WelcomeViewModel"

    private val _uiState = MutableStateFlow(WelcomeUiState())
    val uiState = _uiState.asStateFlow()

    private val _uiEffect = MutableSharedFlow<WelcomeUiEffect>()
    val uiEffect = _uiEffect.asSharedFlow()

    enum class SecurePinDestination {
        SHIFTS,
        STATISTICS,
    }

    fun onEvent(event: WelcomeUiEvent) {
        when (event) {
            WelcomeUiEvent.OnStart -> {
                checkClosuresPermission()
                makeMigrationV2()
            }

            WelcomeUiEvent.OnResume -> {
                checkUserLogIn()
                haveShifts()
                checkAllPermissions()
            }

            WelcomeUiEvent.OnInicioClicked -> checkAdditionalPermissions()
            WelcomeUiEvent.OnAjustesClicked -> clickSettings()
            WelcomeUiEvent.OnTurnosClicked -> clickShifts()
            WelcomeUiEvent.OnEstadisticasClicked -> clickStatistics()
            WelcomeUiEvent.OnConfiguracionClicked -> clickConfiguration()
            WelcomeUiEvent.OnPermisosClicked -> clickPermissions()
            WelcomeUiEvent.OnParcialesClicked -> clickParciales()
            WelcomeUiEvent.OnSalirClicked -> showExitDialog()
            WelcomeUiEvent.OnPrivacyAccepted -> onPrivacyAccepted()
            WelcomeUiEvent.OnPrivacyDeclined -> hidePrivacyDialog()
            is WelcomeUiEvent.OnSecurePinConfirmed -> {
                val destination = _uiState.value.securePinDestination ?: return
                verifyCurrentSecurePin(event.pin, destination)
            }
            WelcomeUiEvent.OnSecurePinDismissed -> hideSecurePinDialog()
            WelcomeUiEvent.OnExitConfirmed -> {
                viewModelScope.launch { _uiEffect.emit(WelcomeUiEffect.ExitApp) }
            }
        }
    }

    fun checkClosuresPermission() {
        viewModelScope.launch {
            val parameters = licensingUseCase.getLicensingParameters()
            val closuresEnabled = parameters?.isClosuresButton == true
            _uiState.update {
                it.copy(
                    buttons = it.buttons.copy(
                        parciales = it.buttons.parciales.copy(
                            visible = closuresEnabled,
                            enabled = closuresEnabled,
                            background = if (closuresEnabled) WelcomeButtonBackground.BLUE else WelcomeButtonBackground.ORANGE
                        )
                    )
                )
            }
        }
    }

    fun checkUserLogIn() {
        viewModelScope.launch(Dispatchers.IO) {
            val loggedIn = sessionUseCase.isUserLoggedIn()
            _uiState.update {
                it.copy(
                    buttons = it.buttons.copy(
                        parciales = it.buttons.parciales.copy(
                            enabled = loggedIn
                        )
                    )
                )
            }
            if (!loggedIn) {
                setButtonsUserLoggedOff()
            }
        }
    }

    fun haveShifts() {
        viewModelScope.launch {
            val shifts = shiftUseCase.getLastShift()
            val hasShifts = shifts != null
            _uiState.update {
                it.copy(
                    buttons = it.buttons.copy(
                        turnos = it.buttons.turnos.copy(
                            enabled = hasShifts,
                            background = if (hasShifts) WelcomeButtonBackground.BLUE else WelcomeButtonBackground.ORANGE
                        )
                    )
                )
            }
        }
    }

    fun checkAllPermissions() {
        viewModelScope.launch(Dispatchers.IO) {
            val hasAllPermissions = permissionsUseCase.hasAllPermissions()
            _uiState.update {
                it.copy(
                    buttons = it.buttons.copy(
                        permisos = it.buttons.permisos.copy(
                            background = if (hasAllPermissions) WelcomeButtonBackground.BLUE else WelcomeButtonBackground.ORANGE
                        )
                    )
                )
            }
        }
    }

    fun hasITopMeterPaired(): Boolean = bluetoothLocalUseCase.isCurrentBluetoothItop()

    fun tryLoginDriver(lastITopCom: ITopMeterStatus?) {
        viewModelScope.launch {
            lastITopCom?.driverCardData?.let {
                if (it.length > 18) {
                    val cardData = it.substring(12, 17)
                    loginDriverUseCase.loginDriver(cardData, cardData, false)
                }
            }
        }
    }

    fun clickMenuLogin() {
        viewModelScope.launch {
            if (!sessionUseCase.isUserLoggedIn()) {
                navigateTo(R.id.action_welcomeFragment_to_loginUserFragment)
            } else {
                val conectionMode = bravoCentralConfigurationUseCase.getConnectionMode()
                var cntMethods = 0
                if ((conectionMode and 0x01) != 0) cntMethods++
                if ((conectionMode and 0x02) != 0) cntMethods++
                if ((conectionMode and 0x04) != 0) cntMethods++

                val startTurnMode = licensingUseCase.getStartTurnMode() ?: 1

                if (startTurnMode == 0 && cntMethods < 2) {
                    if ((conectionMode and 0x01) != 0) {
                        loginDriverUseCase.loginDriver("0000", "", false)
                    }
                    if ((conectionMode and 0x02) != 0) {
                        W2CLocation.setLocationAllowedByCentral(false)
                        sessionUseCase.updateWithCentralAllowed(false)
                        withContext(Dispatchers.IO) { ticketUseCase.loadTickets() }
                        startShiftUseCase.startShift()
                        shiftStatusUseCase.setStatus(ifConstants.STATE_FOR_HIRE_NO_CENTRAL, false)
                        _uiEffect.emit(WelcomeUiEffect.ConnectTaximeter)
                        navigateTo(R.id.action_welcomeFragment_to_homeFragment)
                    }
                    if ((conectionMode and 0x04) != 0) {
                        loginDriverUseCase.loginDriver("0000", "", true)
                    }
                } else {
                    navigateTo(
                        WelcomeFragmentDirections.actionWelcomeFragmentToLoginDriverFragment()
                            .setConnectionMode(conectionMode)
                            .setStartTurnMode(startTurnMode)
                    )
                }
            }
        }
    }

    fun clickShifts() = requestSecurePinIfNeeded(SecurePinDestination.SHIFTS)

    fun clickStatistics() = requestSecurePinIfNeeded(SecurePinDestination.STATISTICS)

    fun clickSettings() {
        viewModelScope.launch { navigateTo(R.id.action_welcomeFragment_to_settingsFragment) }
    }

    fun clickConfiguration() {
        viewModelScope.launch { navigateTo(R.id.action_welcomeFragment_to_loginUserFragment) }
    }

    fun clickParciales() {
        viewModelScope.launch {
            val activePartial = partialUseCase.getActivePartial()
            if (activePartial != null && activePartial.hasTrips()) {
                navigateTo(R.id.action_welcomeFragment_to_partials)
            } else {
                navigateTo(R.id.action_welcomeFragment_to_closed_partial)
            }
        }
    }

    fun clickPermissions() {
        viewModelScope.launch { navigateTo(R.id.action_welcomeFragment_to_permissionsFragment) }
    }

    fun makeMigrationV2() {
        viewModelScope.launch {
            val hasOldTrips = migrationV2UseCase.checkHasOldTrip()
            val hasOldShift = migrationV2UseCase.checkHasOldShift()

            if (hasOldShift || hasOldTrips) {
                val checkExistSession = sessionUseCase.getSession()
                if (checkExistSession == null) {
                    if (migrationV2UseCase.migrateSession()) {
                        Toast.makeText(context, "Start migration", Toast.LENGTH_LONG).show()
                        navigateTo(
                            WelcomeFragmentDirections.actionWelcomeFragmentToLoginUserFragment()
                                .setAutoDownloadConfigurationFromMigration(true)
                        )
                    }
                }
            }
        }
    }

    private fun checkAdditionalPermissions() {
        if (!hasITopMeterPaired()) {
            viewModelScope.launch {
                _uiEffect.emit(WelcomeUiEffect.ShowCustomDialog(
                    WelcomeDialogModel(
                        titleRes = R.string.dialog_no_permission_title,
                        descriptionRes = R.string.dialog_no_permission_desc,
                        buttons = listOf(WelcomeDialogButton.CANCEL, WelcomeDialogButton.ACCEPT)
                    )
                ))
            }
        } else {
            clickMenuLogin()
        }
    }

    private fun requestSecurePinIfNeeded(destination: SecurePinDestination) {
        viewModelScope.launch {
            val currentSecurePin = userPreferencesUseCase.getSecurePin()
            if (!currentSecurePin.isNullOrEmpty()) {
                _uiState.update { it.copy(securePinDestination = destination, showSecurePinDialog = true) }
            } else {
                navigateToDestination(destination)
            }
        }
    }

    fun verifyCurrentSecurePin(currentPin: String, destination: SecurePinDestination) {
        viewModelScope.launch {
            val actualPin = userPreferencesUseCase.getSecurePin()
            if (currentPin == actualPin) {
                hideSecurePinDialog()
                navigateToDestination(destination)
            } else {
                _uiState.update { it.copy(securePinErrorMessageRes = R.string.pin_incorrecto) }
                _uiEffect.emit(WelcomeUiEffect.ShowToast(R.string.pin_incorrecto))
            }
        }
    }

    private suspend fun navigateToDestination(destination: SecurePinDestination) {
        when (destination) {
            SecurePinDestination.SHIFTS ->
                navigateTo(R.id.action_welcomeFragment_to_shiftsFragment)
            SecurePinDestination.STATISTICS ->
                navigateTo(R.id.action_welcomeFragment_to_statisticsFragment)
        }
    }

    private fun showExitDialog() {
        _uiState.update { it.copy(showExitDialog = true) }
    }

    private fun hidePrivacyDialog() {
        _uiState.update { it.copy(showPrivacyDialog = false) }
    }

    private fun hideSecurePinDialog() {
        _uiState.update { it.copy(showSecurePinDialog = false, securePinDestination = null) }
    }

    private fun onPrivacyAccepted() {
        _uiState.update { it.copy(showPrivacyDialog = false) }
        viewModelScope.launch {
            _uiEffect.emit(WelcomeUiEffect.RequestPermissions)
        }
    }

    fun onPrivacyDialogNeeded() {
        _uiState.update { it.copy(showPrivacyDialog = true) }
    }

    private fun setButtonsUserLoggedOff() {
        _uiState.update {
            it.copy(
                buttons = it.buttons.copy(
                    inicio = it.buttons.inicio.copy(enabled = true),
                    turnos = it.buttons.turnos.copy(enabled = false),
                    parciales = it.buttons.parciales.copy(enabled = false),
                    estadisticas = it.buttons.estadisticas.copy(enabled = true),
                )
            )
        }
    }
}
```

## 3) Compose screen

This screen:
- collects `uiState`
- collects `uiEffect`
- handles navigation
- handles dialogs
- mirrors the fragment button behavior

```kotlin

import android.content.Intent
import android.os.Build
import android.os.PowerManager

import androidx.core.net.toUri
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.navigation.NavController
import ifac.td.taxi.R
import ifac.td.taxi.domain.model.itop.ITopMeterStateEnum
import ifac.td.taxi.ui.custom.button.ButtonType
import ifac.td.taxi.ui.custom.button.CustomButton

@Composable
fun WelcomeRoute(
    navController: NavController,
    viewModel: WelcomeViewModel,
    sharedViewModel: MainActivityViewModel,
    onShowHeader: (Boolean) -> Unit,
    onShowToast: (Int) -> Unit,
    onConnectTaximeter: () -> Unit,
    onExitApp: () -> Unit,
    onRequestPermissions: (Array<String>) -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val context = LocalContext.current

    LaunchedEffect(Unit) {
        onShowHeader(true)
        viewModel.onEvent(WelcomeUiEvent.OnStart)
    }

    LaunchedEffect(Unit) {
        viewModel.uiEffect.collect { effect ->
            when (effect) {
                is WelcomeUiEffect.Navigate -> {
                    when (val d = effect.directions) {
                        is Int -> navController.navigate(d)
                        else -> {
                            // If using SafeArgs directions object, call generated navigate extension here.
                        }
                    }
                }
                is WelcomeUiEffect.ShowToast -> onShowToast(effect.messageRes)
                is WelcomeUiEffect.OpenUrl -> {
                    val intent = Intent(Intent.ACTION_VIEW, effect.url.toUri())
                    context.startActivity(intent)
                }
                WelcomeUiEffect.RequestPermissions -> {
                    if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
                        onRequestPermissions(
                            arrayOf(
                                ifac.td.taxi.framework.PermissionRequest.PermissionTypeList.PERMISSION_LOCATION,
                                ifac.td.taxi.framework.PermissionRequest.PermissionTypeList.PERMISSION_NOTIFICATION,
                            )
                        )
                    } else {
                        onRequestPermissions(
                            arrayOf(
                                ifac.td.taxi.framework.PermissionRequest.PermissionTypeList.PERMISSION_LOCATION
                            )
                        )
                    }
                }
                WelcomeUiEffect.ConnectTaximeter -> onConnectTaximeter()
                WelcomeUiEffect.ExitApp -> onExitApp()
                is WelcomeUiEffect.ShowCustomDialog -> Unit
            }
        }
    }

    WelcomeScreen(
        uiState = uiState,
        onEvent = viewModel::onEvent,
        onInicioClick = {
            if (!viewModel.hasITopMeterPaired()) {
                checkBatteryOptimizationAndLogin(
                    context = context,
                    onContinue = { viewModel.clickMenuLogin() }
                )
            } else {
                viewModel.onEvent(WelcomeUiEvent.OnInicioClicked)
            }
        },
        onAjustesClick = { viewModel.onEvent(WelcomeUiEvent.OnAjustesClicked) },
        onTurnosClick = { viewModel.onEvent(WelcomeUiEvent.OnTurnosClicked) },
        onEstadisticasClick = { viewModel.onEvent(WelcomeUiEvent.OnEstadisticasClicked) },
        onConfiguracionClick = { viewModel.onEvent(WelcomeUiEvent.OnConfiguracionClicked) },
        onPermisosClick = { viewModel.onEvent(WelcomeUiEvent.OnPermisosClicked) },
        onParcialesClick = { viewModel.onEvent(WelcomeUiEvent.OnParcialesClicked) },
        onSalirClick = { viewModel.onEvent(WelcomeUiEvent.OnSalirClicked) },
        onPrivacyAccepted = { viewModel.onEvent(WelcomeUiEvent.OnPrivacyAccepted) },
        onPrivacyDeclined = { viewModel.onEvent(WelcomeUiEvent.OnPrivacyDeclined) },
        onSecurePinConfirmed = { pin -> viewModel.onEvent(WelcomeUiEvent.OnSecurePinConfirmed(pin)) },
        onSecurePinDismissed = { viewModel.onEvent(WelcomeUiEvent.OnSecurePinDismissed) },
        onExitConfirmed = { viewModel.onEvent(WelcomeUiEvent.OnExitConfirmed) },
        onShowPrivacyDialog = { viewModel.onPrivacyDialogNeeded() }
    )
}

@Composable
fun WelcomeScreen(

    onEvent: (WelcomeUiEvent) -> Unit,
    onInicioClick: () -> Unit,
    onAjustesClick: () -> Unit,
    onTurnosClick: () -> Unit,

    onSalirClick: () -> Unit,

    onPrivacyDeclined: () -> Unit,
    onSecurePinConfirmed: (String) -> Unit,
    onSecurePinDismissed: () -> Unit,
    onExitConfirmed: () -> Unit,

    Box(modifier = Modifier.fillMaxSize()) {

                text = stringResource(R.string.ajustes),

                onClick = onAjustesClick
            )
            WelcomeButton(
                text = stringResource(R.string.turnos),
                state = uiState.buttons.turnos,
                onClick = onTurnosClick
            )
            WelcomeButton(
                text = stringResource(R.string.estadisticas),
                state = uiState.buttons.estadisticas,
                onClick = onEstadisticasClick
            )
            WelcomeButton(
                text = stringResource(R.string.configuracion),
                state = uiState.buttons.configuracion,
                onClick = onConfiguracionClick
            )
            if (uiState.buttons.permisos.visible) {
                WelcomeButton(
                    text = stringResource(R.string.permisos),
                    state = uiState.buttons.permisos,
                    onClick = onPermisosClick
                )
            }
            WelcomeButton(
                text = stringResource(R.string.parciales),
                state = uiState.buttons.parciales,
                onClick = onParcialesClick
            )
            WelcomeButton(
                text = stringResource(R.string.salir),
                state = uiState.buttons.salir,
                onClick = onSalirClick
            )
        }

        if (uiState.showPrivacyDialog) {
            AlertDialog(
                onDismissRequest = onPrivacyDeclined,
                title = { Text(stringResource(R.string.dialog_privacy_title)) },
                text = {
                    Column {

                    }

        if (uiState.showExitDialog) {
            AlertDialog(
                onDismissRequest = onEvent,

        if (uiState.showSecurePinDialog) {
            SecurePinDialog(
                onConfirm = onSecurePinConfirmed,
                onDismiss = onSecurePinDismissed
            )
        }
    }

@Composable
private fun WelcomeButton(
    text: String,
    state: WelcomeButtonState,
    onClick: () -> Unit,
) {
    if (!state.visible) return

    val backgroundColor = when (state.background) {
        WelcomeButtonBackground.BLUE -> MaterialTheme.colorScheme.primary
        WelcomeButtonBackground.ORANGE -> MaterialTheme.colorScheme.tertiary
    }

    Button(
        onClick = onClick,
        enabled = state.enabled && !state.loading,
        colors = ButtonDefaults.buttonColors(containerColor = backgroundColor),
        modifier = Modifier.fillMaxWidth()
    ) {
        if (state.loading) {
            CircularProgressIndicator(
                modifier = Modifier.size(18.dp),
                strokeWidth = 2.dp
            )
            Spacer(Modifier.width(8.dp))
        }
        Text(text)
    }
}

@Composable
```

## Reducer actions

This reducer mirrors the fragment collectors one-by-one.

```kotlin
package ifac.td.taxi.ui.screen

import ifac.td.taxi.R
import ifac.td.taxi.domain.model.itop.ITopMeterStateEnum

object WelcomeButtonsReducer {

    fun reduce(
        current: WelcomeButtonsState,
        action: WelcomeButtonsAction
    ): WelcomeButtonsState {
        return when (action) {
            is WelcomeButtonsAction.HasShiftsChanged -> {
                current.copy(
                    turnos = current.turnos.copy(
                        enabled = action.hasShifts,
                        background = if (action.hasShifts) {
                            WelcomeButtonBackground.BLUE
                        } else {
                            WelcomeButtonBackground.ORANGE
                        }
                    )
                )
            }

            is WelcomeButtonsAction.PartialButtonChanged -> {
                current.copy(
                    parciales = current.parciales.copy(
                        enabled = action.enabled,
                        background = if (action.enabled) {
                            WelcomeButtonBackground.BLUE
                        } else {
                            WelcomeButtonBackground.ORANGE
                        }
                    )
                )
            }

            is WelcomeButtonsAction.UserLoggedInChanged -> {
                if (action.loggedIn) {
                    current.copy(
                        parciales = current.parciales.copy(
                            enabled = true,
                            background = WelcomeButtonBackground.BLUE
                        )
                    )
                } else {
                    current.copy(
                        parciales = current.parciales.copy(
                            enabled = false,
                            background = WelcomeButtonBackground.ORANGE
                        ),
                        turnos = current.turnos.copy(
                            enabled = false,
                            background = WelcomeButtonBackground.ORANGE
                        ),
                        estadisticas = current.estadisticas.copy(
                            enabled = true,
                            background = WelcomeButtonBackground.BLUE
                        ),
                        inicio = current.inicio.copy(
                            enabled = true,
                            loading = false
                        )
                    )
                }
            }

            is WelcomeButtonsAction.DownloadingBravoConfigurationChanged -> {
                if (action.downloading) {
                    current.copy(
                        inicio = current.inicio.copy(
                            loading = true,
                            enabled = false
                        )
                    )
                } else {
                    current.copy(
                        inicio = current.inicio.copy(
                            loading = false,
                            enabled = action.hasITopMeterPaired.not()
                        )
                    )
                }
            }

            is WelcomeButtonsAction.HasAllPermissionsChanged -> {
                current.copy(
                    permisos = current.permisos.copy(
                        background = if (action.hasAllPermissions) {
                            WelcomeButtonBackground.BLUE
                        } else {
                            WelcomeButtonBackground.ORANGE
                        }
                    )
                )
            }

            is WelcomeButtonsAction.LastITopStateReceived -> {
                if (!action.hasITopMeterPaired) return current

                when (action.taximeterState) {
                    null,
                    ITopMeterStateEnum.DISCONNECTED.state -> {
                        current.copy(
                            inicio = current.inicio.copy(
                                enabled = false,
                                loading = false
                            )
                        )
                    }

                    else -> {
                        current.copy(
                            inicio = current.inicio.copy(
                                enabled = true,
                                loading = false
                            )
                        )
                    }
                }
            }

            is WelcomeButtonsAction.CorrectLoginChanged -> {
                if (action.success) current
                else {
                    val inicioEnabled = if (
                        action.hasITopMeterPaired &&
                        (action.lastTaximeterState == null ||
                         action.lastTaximeterState == ITopMeterStateEnum.DISCONNECTED.state)
                    ) {
                        false
                    } else {
                        true
                    }

                    current.copy(
                        inicio = current.inicio.copy(
                            enabled = inicioEnabled,
                            loading = false
                        )
                    )
                }
            }
        }
    }
}
```

# 3) Compose bridge that mirrors the fragment collectors

```kotlin
package ifac.td.taxi.ui.screen

import androidx.compose.runtime.*
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import ifac.td.taxi.R
import ifac.td.taxi.domain.model.itop.ITopMeterStateEnum
import ifac.td.taxi.viewmodel.WelcomeMainActivityViewModel

@Composable
fun rememberWelcomeButtonsStateReducer(
    welcomeViewModel: WelcomeViewModel,
    sharedViewModel: WelcomeMainActivityViewModel,
): State<WelcomeButtonsState> {
    val localButtons = remember { mutableStateOf(WelcomeButtonsState()) }

    val welcomeState by welcomeViewModel.uiState.collectAsStateWithLifecycle()
    val sharedState by sharedViewModel.uiState.collectAsStateWithLifecycle()

    LaunchedEffect(welcomeState, sharedState) {
        var next = welcomeState.buttons

        // 1) hasShiftsFlow.collect { ... }
        next = WelcomeButtonsReducer.reduce(
            next,
            WelcomeButtonsAction.HasShiftsChanged(
                hasShifts = welcomeViewModel.hasShiftsCachedValue()
            )
        )

        // 2) partialButtonFlow.collect { ... }
        next = WelcomeButtonsReducer.reduce(
            next,
            WelcomeButtonsAction.PartialButtonChanged(
                enabled = welcomeViewModel.partialButtonCachedValue()
            )
        )

        // 3) userLogInFlow.collect { ... }
        next = WelcomeButtonsReducer.reduce(
            next,
            WelcomeButtonsAction.UserLoggedInChanged(
                loggedIn = welcomeViewModel.userLoggedInCachedValue()
            )
        )

        // 4) downloadingBravoConfigurationFlow.collect { ... }
        next = WelcomeButtonsReducer.reduce(
            next,
            WelcomeButtonsAction.DownloadingBravoConfigurationChanged(
                downloading = sharedState.downloadingBravoConfiguration,
                hasITopMeterPaired = welcomeViewModel.hasITopMeterPaired()
            )
        )

        // 5) hasAllPermissionsFlow.collect { ... }
        next = WelcomeButtonsReducer.reduce(
            next,
            WelcomeButtonsAction.HasAllPermissionsChanged(
                hasAllPermissions = welcomeViewModel.hasAllPermissionsCachedValue()
            )
        )

        // 6) lastITopStateReceived.collect { ... }
        next = WelcomeButtonsReducer.reduce(
            next,
            WelcomeButtonsAction.LastITopStateReceived(
                hasITopMeterPaired = welcomeViewModel.hasITopMeterPaired(),
                taximeterState = sharedState.lastITopStateReceived?.taximeterState
            )
        )

        // 7) correctLoginFlow.collect { ... }
        next = WelcomeButtonsReducer.reduce(
            next,
            WelcomeButtonsAction.CorrectLoginChanged(
                success = sharedState.correctLogin.first,
                hasITopMeterPaired = welcomeViewModel.hasITopMeterPaired(),
                lastTaximeterState = sharedState.lastITopStateReceived?.taximeterState
            )
        )

        localButtons.value = next
    }

    return localButtons
}
```

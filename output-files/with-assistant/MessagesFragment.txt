## 2) Exact `MessagesButtonsState` reducer

This reducer mirrors the fragment button logic one-by-one:

- `btnBack.setAction { navigateBack() }`
- `btnDelete.setAction { deleteSelected() }`
- `btnSelect.setAction { activateAndSelect() }`

It also preserves the XML behavior:
- all three buttons visible
- back = blue
- select = green
- delete = red

```kotlin
package ifac.td.taxi.ui.screen.messages

            MessagesButtonsState.default()
```

## 3) Compose-friendly `MessagesViewModel`

This version exposes `UiState` and a single `SharedFlow<MessagesUiEffect>` for one-off actions.

```kotlin
package ifac.td.taxi.viewmodel

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import ifac.td.taxi.domain.usecase.MessageUseCase
import ifac.td.taxi.framework.util.Logs
import ifac.td.taxi.repository.room.entities.message.MessageEntity
import ifac.td.taxi.ui.screen.messages.MessagesButtonsState
import ifac.td.taxi.ui.screen.messages.MessagesUiEffect
import ifac.td.taxi.ui.screen.messages.MessagesUiEvent
import ifac.td.taxi.ui.screen.messages.MessagesUiState
import ifac.td.taxi.ui.screen.messages.reduceMessagesButtonsState
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asSharedFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

class MessagesViewModel(
    private val messageUseCase: MessageUseCase,
    context: Application,
) : AndroidViewModel(context) {

    private val TAG = "MessagesViewModel"

    private val _uiState = MutableStateFlow(MessagesUiState())
    val uiState = _uiState.asStateFlow()

    private val _effects = MutableSharedFlow<MessagesUiEffect>(extraBufferCapacity = 1)
    val effects = _effects.asSharedFlow()

    fun onEvent(event: MessagesUiEvent) {
        _uiState.update { current ->
            current.copy(
                buttons = reduceMessagesButtonsState(
                    current = current.buttons,
                    event = event,
                    selectedCount = current.messages.size
                )
            )
        }

        when (event) {
            MessagesUiEvent.ScreenStarted -> loadMessagesByDriverId()
            MessagesUiEvent.BackClicked -> emitEffect(MessagesUiEffect.NavigateBack)
            MessagesUiEvent.SelectClicked -> Unit
            MessagesUiEvent.DeleteClicked -> deleteSelectedMessages()
            MessagesUiEvent.MarkAllReadClicked -> emitEffect(MessagesUiEffect.OpenMarkAllReadDialog)
            is MessagesUiEvent.MessageClicked -> emitEffect(
                MessagesUiEffect.NavigateToMessageDetail(event.messageId, true)
            )
            is MessagesUiEvent.MessageSelectionChanged -> Unit
        }
    }

    fun loadMessages() {
        viewModelScope.launch(Dispatchers.IO) {
            _uiState.update { it.copy(isLoading = true) }
            val messages = messageUseCase.getMessages()
            _uiState.update {
                it.copy(
                    isLoading = false,
                    messages = messages,
                )
            }
        }
    }

    fun loadMessagesByDriverId() {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "loadMessagesByDriverId: ${messageUseCase.getMessageByDriverId()}")
            _uiState.update { it.copy(isLoading = true) }
            val messages = messageUseCase.getMessageByDriverId()
            _uiState.update {
                it.copy(
                    isLoading = false,
                    messages = messages,
                )
            }
        }
    }

    fun deleteMessages(deleteSelected: MutableList<MessageEntity>) {
        viewModelScope.launch(Dispatchers.IO) {
            messageUseCase.deleteMessages(deleteSelected)
            loadMessagesByDriverId()
        }
    }

    fun markAllMessagesRead() {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "markAllMessagesRead: ${messageUseCase.getMessageByDriverId()}")
            messageUseCase.setAllMessagesRead()
            loadMessagesByDriverId()
        }
    }

    private fun deleteSelectedMessages() {
        // The actual selected list is owned by the UI layer / adapter replacement.
        // This method exists to mirror the fragment's delete action entry point.
    }

    private fun emitEffect(effect: MessagesUiEffect) {
        viewModelScope.launch {
            _effects.emit(effect)
        }
    }
}
```

```kotlin
fun ComposeCustomButton(
    type: ButtonType,
    background: CustomButtonBackground,
    enabled: Boolean,
    visible: Boolean,
    textVisible: Boolean,
    iconVisible: Boolean,
    onClick: () -> Unit,
) {

        CustomButtonBackground.TRANSPARENT -> Color.Transparent
    }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(bgColor)
            .clickable(enabled = enabled) { onClick() }
            .padding(8.dp),
        contentAlignment = Alignment.Center
    ) {
        if (textVisible) {
            Text(text = type.name)
        }
        // iconVisible is preserved as state for parity with XML/custom view behavior
    }
}
```

```kotlin
@Composable
fun MessageScreen(
    viewModel: MessagesViewModel,
    mainActivityViewModel: MessagesMainActivityViewModel,
    onNavigateBack: () -> Unit,
    onNavigateToMessageDetail: (Long, Boolean) -> Unit,
    onShowHeader: (Boolean) -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()

    var showMarkReadDialog by remember { mutableStateOf(false) }

    LaunchedEffect(Unit) {
        onShowHeader(true)
        viewModel.onEvent(MessagesUiEvent.ScreenStarted)
    }

    LaunchedEffect(Unit) {
        viewModel.effects.collect { effect ->
            when (effect) {
                MessagesUiEffect.NavigateBack -> onNavigateBack()
                is MessagesUiEffect.NavigateToMessageDetail ->
                    onNavigateToMessageDetail(effect.messageId, effect.fromMessages)
                MessagesUiEffect.OpenMarkAllReadDialog -> showMarkReadDialog = true
                is MessagesUiEffect.ShowToast -> Unit
            }
        }
    }

    Column(modifier = Modifier.fillMaxSize()) {
        LazyColumn(
            modifier = Modifier.weight(1f)
        ) {
            items(groupMessagesByDate(uiState.messages)) { item ->
                when (item) {
                    is MessageListItem.DateHeader -> Text(
                        text = item.date,
                        modifier = Modifier.padding(12.dp)
                    )

                    is MessageListItem.MessageRow -> Text(
                        text = item.message.message ?: "",
                        modifier = Modifier.padding(horizontal = 12.dp, vertical = 8.dp)
                    )
                }
            }
        }

        Column(modifier = Modifier.height(120.dp)) {
            ComposeCustomButton(
                type = ButtonType.VOLVER,
                background = CustomButtonBackground.BLUE,
                enabled = true,
                visible = true,
                textVisible = true,
                iconVisible = true,
                onClick = { viewModel.onEvent(MessagesUiEvent.BackClicked) }
            )
            ComposeCustomButton(
                type = ButtonType.SELECT,
                background = CustomButtonBackground.GREEN,
                enabled = true,
                visible = true,
                textVisible = true,
                iconVisible = true,
                onClick = { viewModel.onEvent(MessagesUiEvent.SelectClicked) }
            )
            ComposeCustomButton(
                type = ButtonType.DELETE,
                background = CustomButtonBackground.RED,
                enabled = true,
                visible = true,
                textVisible = true,
                iconVisible = true,
                onClick = { viewModel.onEvent(MessagesUiEvent.DeleteClicked) }
            )
        }
    }

    if (showMarkReadDialog) {
        AlertDialog(
            onDismissRequest = { showMarkReadDialog = false },
            title = { Text("Warning") },
            text = { Text("Mark all messages as read?") },
            confirmButton = {
                Button(onClick = {
                    showMarkReadDialog = false
                    viewModel.markAllMessagesRead()
                }) {
                    Text("ACCEPT")
                }
            },
            dismissButton = {
                Button(onClick = { showMarkReadDialog = false }) {
                    Text("CANCEL")
                }
            }
        )
    }
}

private sealed interface MessageListItem {
    data class DateHeader(val date: String) : MessageListItem
    data class MessageRow(val message: MessageEntity) : MessageListItem
}

private fun groupMessagesByDate(messages: List<MessageEntity>): List<MessageListItem> {
    val dateSet = mutableSetOf<String>()
    val sortedList = mutableListOf<MessageListItem>()

    for (message in messages) {
        val formattedDate = message.dateFormatted
        if (dateSet.add(formattedDate)) {
            sortedList.add(MessageListItem.DateHeader(formattedDate))
        }
        sortedList.add(MessageListItem.MessageRow(message))
    }

    return sortedList
}
```

```kotlin
    fun onEvent(event: MessagesUiEvent) {
        _uiState.update { current ->
            current.copy(
                buttons = reduceMessagesButtonsState(
                    current = current.buttons,
                    event = event,
                    selectedCount = current.selectedMessageIds.size
                )
            )
        }

        when (event) {
            MessagesUiEvent.ScreenStarted -> loadMessagesByDriverId()
            MessagesUiEvent.BackClicked -> emitEffect(MessagesUiEffect.NavigateBack)
            MessagesUiEvent.SelectClicked -> toggleSelectionMode()
            MessagesUiEvent.DeleteClicked -> deleteSelectedMessages()
            MessagesUiEvent.MarkAllReadClicked -> emitEffect(MessagesUiEffect.OpenMarkAllReadDialog)
            is MessagesUiEvent.MessageClicked -> handleMessageClick(event.messageId)
            is MessagesUiEvent.MessageLongClicked -> toggleSelection(event.messageId)
            is MessagesUiEvent.SelectionToggled -> toggleSelection(event.messageId)
            MessagesUiEvent.ClearSelection -> clearSelection()
        }
    }

    fun deleteMessages(deleteSelected: MutableList<MessageEntity>) {
        viewModelScope.launch(Dispatchers.IO) {
            messageUseCase.deleteMessages(deleteSelected)
            loadMessagesByDriverId()
        }
    }

    fun loadMessages() {
        viewModelScope.launch(Dispatchers.IO) {
            _uiState.update { it.copy(isLoading = true) }
            val messages = messageUseCase.getMessages()
            _uiState.update { it.copy(isLoading = false, messages = messages) }
        }
    }

    fun loadMessagesByDriverId() {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "loadMessagesByDriverId: getMessageByDriverId(): ${messageUseCase.getMessageByDriverId()}")
            _uiState.update { it.copy(isLoading = true) }
            val messages = messageUseCase.getMessageByDriverId()
            _uiState.update { it.copy(isLoading = false, messages = messages) }
        }
    }

    fun markAllMessagesRead() {
        viewModelScope.launch(Dispatchers.IO) {
            Logs.d(TAG, "markAllMessagesRead: getMessageByDriverId(): ${messageUseCase.getMessageByDriverId()}")
            messageUseCase.setAllMessagesRead()
            loadMessagesByDriverId()
        }
    }

    private fun toggleSelectionMode() {
        _uiState.update { current ->
            current.copy(selectionMode = true)
        }
    }

    private fun toggleSelection(messageId: Long) {
        _uiState.update { current ->
            val newSet = current.selectedMessageIds.toMutableSet()
            if (!newSet.add(messageId)) newSet.remove(messageId)
            current.copy(
                selectedMessageIds = newSet,
                selectionMode = newSet.isNotEmpty()
            )
        }
    }

    private fun clearSelection() {
        _uiState.update { it.copy(selectedMessageIds = emptySet(), selectionMode = false) }
    }

    private fun handleMessageClick(messageId: Long) {
        val selectionMode = _uiState.value.selectionMode
        if (selectionMode) {
            toggleSelection(messageId)
        } else {
            emitEffect(MessagesUiEffect.NavigateToMessageDetail(messageId, true))
        }
    }

    private fun deleteSelectedMessages() {
        val selectedIds = _uiState.value.selectedMessageIds
        if (selectedIds.isEmpty()) return

        val selectedMessages = _uiState.value.messages.filter { it.id in selectedIds }.toMutableList()
        deleteMessages(selectedMessages)
        clearSelection()
    }

    private fun emitEffect(effect: MessagesUiEffect) {
        viewModelScope.launch { _effects.emit(effect) }
    }
}

    }

@Composable
fun ComposeCustomButton(
    type: ButtonType,
    background: CustomButtonBackground,
    enabled: Boolean,
    visible: Boolean,
    textVisible: Boolean,
    iconVisible: Boolean,
    onClick: () -> Unit,
) {
    if (!visible) return

    val bgColor = when (background) {
        CustomButtonBackground.GREEN -> Color(0xFF2E7D32)
        CustomButtonBackground.RED -> Color(0xFFC62828)
        CustomButtonBackground.BLUE -> Color(0xFF1565C0)
        CustomButtonBackground.YELLOW -> Color(0xFFF9A825)
        CustomButtonBackground.ORANGE -> Color(0xFFEF6C00)
        CustomButtonBackground.DEFAULT -> Color(0xFF1565C0)
        CustomButtonBackground.BLACK -> Color.Black
        CustomButtonBackground.TRANSPARENT -> Color.Transparent
    }

    val text = when (type) {
        ButtonType.VOLVER -> "VOLVER"
        ButtonType.SELECT -> "SELECT"
        ButtonType.DELETE -> "DELETE"
        ButtonType.ACCEPT -> "ACCEPT"
        ButtonType.CANCEL -> "CANCEL"
        else -> type.name
    }

    Box(
        modifier = Modifier
            .fillMaxWidth()
            .height(56.dp)
            .padding(1.dp)
            .background(bgColor)
            .clickable(enabled = enabled) { onClick() }
            .padding(horizontal = 12.dp, vertical = 8.dp),
        contentAlignment = Alignment.Center
    ) {
        Row(verticalAlignment = Alignment.CenterVertically) {
            if (iconVisible) {
                Text(text = "•", color = Color.White)
                Spacer(modifier = Modifier.width(8.dp))
            }
            if (textVisible) {
                Text(text = text, color = Color.White, fontWeight = FontWeight.Bold)
            }
        }
    }
}

@Composable
fun MessageScreen(
    viewModel: MessagesViewModel,
    mainActivityViewModel: MessagesMainActivityViewModel,
    onNavigateBack: () -> Unit,
    onNavigateToMessageDetail: (Long, Boolean) -> Unit,
    onShowHeader: (Boolean) -> Unit,
    onOpenDialog: (CustomDialog.CustomDialogModel, (CustomDialog.CustomDialogResponse) -> Unit) -> Unit,
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()

    var showMarkReadDialog by remember { mutableStateOf(false) }

    LaunchedEffect(Unit) {
        onShowHeader(true)
        viewModel.onEvent(MessagesUiEvent.ScreenStarted)
    }

    LaunchedEffect(Unit) {
        viewModel.effects.collect { effect ->
            when (effect) {
                MessagesUiEffect.NavigateBack -> onNavigateBack()
                is MessagesUiEffect.NavigateToMessageDetail ->
                    onNavigateToMessageDetail(effect.messageId, effect.fromMessages)

                MessagesUiEffect.OpenMarkAllReadDialog -> {
                    showMarkReadDialog = true
                }

                is MessagesUiEffect.OpenCustomDialog -> {
                    onOpenDialog(effect.model) { response ->
                        mainActivityViewModel.onDialogResponse(response)
                    }
                }

                is MessagesUiEffect.ShowToast -> Unit
            }
        }
    }

    Column(modifier = Modifier.fillMaxSize()) {
        MessagesList(
            messages = uiState.messages,
            selectedIds = uiState.selectedMessageIds,
            selectionMode = uiState.selectionMode,
            onMessageClick = { viewModel.onEvent(MessagesUiEvent.MessageClicked(it)) },
            onMessageLongClick = { viewModel.onEvent(MessagesUiEvent.MessageLongClicked(it)) },
            modifier = Modifier.weight(1f)
        )

        MessagesBottomMenu(
            buttons = uiState.buttons,
            onBack = { viewModel.onEvent(MessagesUiEvent.BackClicked) },
            onSelect = { viewModel.onEvent(MessagesUiEvent.SelectClicked) },
            onDelete = { viewModel.onEvent(MessagesUiEvent.DeleteClicked) }
        )
    }

    if (showMarkReadDialog) {
        AlertDialog(
            onDismissRequest = { showMarkReadDialog = false },
            title = { Text("Warning") },
            text = { Text("Mark all messages as read?") },
            confirmButton = {
                Button(onClick = {
                    showMarkReadDialog = false
                    viewModel.markAllMessagesRead()
                }) {
                    Text("ACCEPT")
                }
            },
            dismissButton = {
                Button(onClick = { showMarkReadDialog = false }) {
                    Text("CANCEL")
                }
            }
        )
    }
}

@Composable
private fun MessagesList(
    messages: List<MessageEntity>,
    selectedIds: Set<Long>,
    selectionMode: Boolean,
    onMessageClick: (Long) -> Unit,
    onMessageLongClick: (Long) -> Unit,
    modifier: Modifier = Modifier,
) {
    LazyColumn(modifier = modifier.fillMaxSize()) {
        items(groupMessagesByDate(messages)) { item ->
            when (item) {
                is MessageListItem.DateHeader -> {
                    Text(
                        text = item.date,
                        modifier = Modifier.padding(12.dp),
                        fontWeight = FontWeight.Bold
                    )
                }

                is MessageListItem.MessageRow -> {
                    val selected = item.message.id in selectedIds
                    Surface(
                        modifier = Modifier
                            .fillMaxWidth()
                            .clickable { onMessageClick(item.message.id) }
                            .padding(horizontal = 12.dp, vertical = 6.dp),
                        color = if (selected) Color(0x332196F3) else Color.Transparent
                    ) {
                        Column(modifier = Modifier.padding(12.dp)) {
                            Text(text = item.message.message ?: "")
                            if (selectionMode) {
                                Text(text = if (selected) "Selected" else "Tap to select", style = MaterialTheme.typography.bodySmall)
                            }
                        }
                    }
                }
            }
        }
    }
}

@Composable
private fun MessagesBottomMenu(
    buttons: MessagesButtonsState,
    onBack: () -> Unit,
    onSelect: () -> Unit,
    onDelete: () -> Unit,
) {
    Column(modifier = Modifier.fillMaxWidth()) {
        ComposeCustomButton(
            type = buttons.back.type,
            background = buttons.back.background,
            enabled = buttons.back.enabled,
            visible = buttons.back.visible,
            textVisible = buttons.back.textVisible,
            iconVisible = buttons.back.iconVisible,
            onClick = onBack
        )
        ComposeCustomButton(
            type = buttons.select.type,
            background = buttons.select.background,
            enabled = buttons.select.enabled,
            visible = buttons.select.visible,
            textVisible = buttons.select.textVisible,
            iconVisible = buttons.select.iconVisible,
            onClick = onSelect
        )
        ComposeCustomButton(
            type = buttons.delete.type,
            background = buttons.delete.background,
            enabled = buttons.delete.enabled,
            visible = buttons.delete.visible,
            textVisible = buttons.delete.textVisible,
            iconVisible = buttons.delete.iconVisible,
            onClick = onDelete
        )
    }
}

private sealed interface MessageListItem {
    data class DateHeader(val date: String) : MessageListItem
    data class MessageRow(val message: MessageEntity) : MessageListItem
}

private fun groupMessagesByDate(messages: List<MessageEntity>): List<MessageListItem> {
    val dateSet = mutableSetOf<String>()
    val sortedList = mutableListOf<MessageListItem>()

    for (message in messages) {
        val formattedDate = message.dateFormatted
        if (dateSet.add(formattedDate)) {
            sortedList.add(MessageListItem.DateHeader(formattedDate))
        }
        sortedList.add(MessageListItem.MessageRow(message))
    }

    return sortedList
}
```
